122static void poeg_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
123{
124 (void)uc;
125 (void)size;
126 const uint64_t off = addr - (uint64_t)k_poeg_base;
127 const uint64_t group = off / (uint64_t)k_poeg_stride;
128 const uint64_t reg = off % (uint64_t)k_poeg_stride;
129 if (group >= (uint64_t)k_poeg_count) {
130 return;
131 }
132 if (reg != (uint64_t)k_poeg_off_poegg) {
133 return;
134 }
135 /* ST is read-only: strip any incoming ST, then re-derive it from the
136 * request flags so a software-stop assert / clear moves ST as hardware would. */
137 const uint32_t writable = (uint32_t)value & ~(uint32_t)k_poegg_st;
138 const uint32_t prev = s_poeg.poegg[group];
139 const uint32_t next = poeg_with_st(writable);
140 s_poeg.poegg[group] = next;
141 if (((prev & (uint32_t)k_poegg_st) == 0U) && ((next & (uint32_t)k_poegg_st) != 0U)) {
142 s_poeg.asserts++;
143 }
144 if (((prev & (uint32_t)k_poegg_st) != 0U) && ((next & (uint32_t)k_poegg_st) == 0U)) {
145 s_poeg.clears++;
146 }
147}
