216[[nodiscard]] static ra8_err_t dotf_demo_sample(uint8_t* out_ok)
217{
218 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
219
220 uint32_t st0 = 0U;
221 const ra8_err_t st0_err = ra8_dotf_run_self_test((uint8_t)k_dotf_demo_ch0, &st0);
222 g_dotf_st0_snap = st0;
223
224 uint32_t st1 = 0U;
225 const ra8_err_t st1_err = ra8_dotf_run_self_test((uint8_t)k_dotf_demo_ch1, &st1);
226 g_dotf_st1_snap = st1;
227
228 uint32_t reg00 = 0U;
229 const ra8_err_t stat_err = ra8_dotf_get_status((uint8_t)k_dotf_demo_ch0, &reg00);
230 g_dotf_reg00 = reg00;
231
232 *out_ok = dotf_demo_verdict(st0_err, st1_err, stat_err);
233 if (st0_err != k_ra8_ok) {
234 return st0_err;
235 }
236 if (st1_err != k_ra8_ok) {
237 return st1_err;
238 }
239 return stat_err;
240}

244int32_t main(void)
245{
246 dotf_demo_setup_or_halt();
247 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
248 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No maskable NVIC
249 * source is armed; DOTF raises no IRQ of its own. */
250 ra8_isr_globals_enable();
251
252 const ra8_err_t init_err = ra8_dotf_init();
253 g_dotf_init_err = (uint32_t)init_err;
254 if (init_err != k_ra8_ok) {
255 dotf_demo_panic_halt();
256 }
257
258 while (1) {
259 uint8_t ok = 0U;
260 const ra8_err_t err = dotf_demo_sample(&ok);
261 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
262 g_dotf_ok = (uint32_t)good;
263 if (good != 0U) {
264 (void)ra8_board_uart_console_write(k_dotf_demo_ok_msg,
265 (size_t)(sizeof(k_dotf_demo_ok_msg) - 1U));
266 (void)ra8_board_led_toggle(k_ra8_board_led1);
267 } else {
268 (void)ra8_board_uart_console_write(k_dotf_demo_bad_msg,
269 (size_t)(sizeof(k_dotf_demo_bad_msg) - 1U));
270 (void)ra8_board_led_toggle(k_ra8_board_led2);
271 }
272 ++g_dotf_heartbeat;
273 ra8_delay_ms(k_dotf_demo_period_ms);
274 }
275 dotf_demo_panic_halt();
276 return 0;
277}
