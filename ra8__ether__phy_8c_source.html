83static ra8_err_t internal_reset_and_wait(void)
84{
85 ra8_err_t err = s_state.io.write(s_state.io.ctx,
86 s_state.phy_address,
87 k_ra8_ether_phy_reg_control,
88 k_ra8_ether_phy_bmcr_reset);
89 if (err != k_ra8_ok) {
90 return err;
91 }
92 for (uint8_t i = 0U; i < k_ra8_ether_phy_reset_poll_max; ++i) { /* GCOVR_EXCL_BR_LINE */
93 uint16_t reg = 0U;
94 err = s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_ether_phy_reg_control, &reg);
95 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
96 return err;
97 }
98 if ((reg & k_ra8_ether_phy_bmcr_reset) == 0U) {
99 return k_ra8_ok;
100 }
101 }
102 return k_ra8_err_hw_timeout;
103}

105ra8_err_t ra8_ether_phy_open(const ra8_ether_phy_cfg_t* cfg)
106{
107 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
108 RA8_CHECK_NULL_PTR(cfg->io.read, s_tag, "io.read required");
109 RA8_CHECK_NULL_PTR(cfg->io.write, s_tag, "io.write required");
110 if (cfg->phy_address > k_ra8_ether_phy_addr_max) {
111 return k_ra8_err_invalid_arg;
112 }
113 if (s_state.opened) {
114 return k_ra8_err_exists;
115 }
116 s_state.opened = true;
117 s_state.phy_address = cfg->phy_address;
118 s_state.mii_type = cfg->mii_type;
119 s_state.reset_wait_us = cfg->reset_wait_us;
120 s_state.io = cfg->io;
121 s_state.last_bmsr = 0U;
122
123 /* Issue BMCR reset and poll for self-clear. */
124 const ra8_err_t err = internal_reset_and_wait();
125 if (err != k_ra8_ok) {
126 s_state.opened = false;
127 return err;
128 }
129 ra8_log_info_val(s_tag, "phy ready addr", (uint32_t)cfg->phy_address);
130 return k_ra8_ok;
131}

165ra8_err_t ra8_ether_phy_auto_negotiate_start(void)
166{
167 if (!s_state.opened) {
168 return k_ra8_err_not_initialized;
169 }
170 /* Re-arm AN: enable + restart. */
171 const uint16_t bmcr =
172 (uint16_t)(k_ra8_ether_phy_bmcr_an_enable | k_ra8_ether_phy_bmcr_an_restart);
173 return s_state.io.write(s_state.io.ctx, s_state.phy_address, k_ra8_ether_phy_reg_control, bmcr);

176ra8_err_t ra8_ether_phy_link_status_get(ra8_ether_phy_link_t* out)
177{
178 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
179 if (!s_state.opened) {
180 return k_ra8_err_not_initialized;
181 }
182 uint16_t bmsr = 0U;
183 ra8_err_t err =
184 s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_ether_phy_reg_status, &bmsr);
185 if (err != k_ra8_ok) {
186 return err;
187 }
188 s_state.last_bmsr = bmsr;
189 out->bmsr = bmsr;
190 out->link_up = (uint8_t)(((bmsr & k_ra8_ether_phy_bmsr_link_up) != 0U) ? 1U : 0U);
191 out->auto_neg_done = (uint8_t)(((bmsr & k_ra8_ether_phy_bmsr_an_complete) != 0U) ? 1U : 0U);
192 out->speed = k_ra8_ether_phy_speed_no_link;
193 if ((out->link_up != 0U) && (out->auto_neg_done != 0U)) {
194 /* Inspect partner ability for resolved speed/duplex. */
195 uint16_t lpa = 0U;
196 err =
197 s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_ether_phy_reg_an_partner, &lpa);
198 if (err == k_ra8_ok) {
199 if ((lpa & k_ra8_ether_phy_anar_100full) != 0U) {
200 out->speed = k_ra8_ether_phy_speed_100f;
201 } else if ((lpa & k_ra8_ether_phy_anar_100half) != 0U) {
202 out->speed = k_ra8_ether_phy_speed_100h;
203 } else if ((lpa & k_ra8_ether_phy_anar_10full) != 0U) {
204 out->speed = k_ra8_ether_phy_speed_10f;
205 } else if ((lpa & k_ra8_ether_phy_anar_10half) != 0U) {
206 out->speed = k_ra8_ether_phy_speed_10h;
207 }
208 }
209 }
210 return k_ra8_ok;
