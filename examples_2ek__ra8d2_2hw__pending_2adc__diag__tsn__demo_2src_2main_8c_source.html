142static uint32_t adc_diag_u32_to_dec(uint8_t* buf, uint32_t val)
143{
144 if (val == 0U) {
145 buf[0] = (uint8_t)'0';
146 return 1U;
147 }
148 uint8_t tmp[k_adc_diag_dec_max];
149 uint32_t n = 0U;
150 uint32_t v = val;
151 while (v != 0U) {
152 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint32_t)k_adc_diag_radix));
153 v = v / (uint32_t)k_adc_diag_radix;
154 n++;
155 }
156 for (uint32_t i = 0U; i < n; i++) {
157 buf[i] = tmp[n - 1U - i];
158 }
159 return n;
160}

239static bool adc_diag_run_cycle(void)
240{
241 bool healthy = true;
242 for (uint32_t i = 0U; i < (uint32_t)k_adc_diag_num_modes; i++) {
243 uint16_t code = 0U;
244 bool pass = false;
245 const ra8_err_t err = ra8_adc_self_diagnose(k_adc_diag_modes[i], &code, &pass);
246 if (err != k_ra8_ok) {
247 healthy = false;
248 pass = false;
249 }
250 if (!pass) {
251 healthy = false;
252 }
253 adc_diag_emit_mode((uint8_t)(i + 1U), code, pass);
254 }
255
256 int32_t milli = 0;
257 const ra8_err_t temp_err = ra8_tsn_read_die_temp_milli_c(&milli);
258 if (temp_err != k_ra8_ok) {
259 healthy = false;
260 }
261 adc_diag_write(k_adc_diag_tsn_prefix, (uint32_t)(sizeof(k_adc_diag_tsn_prefix) - 1U));
262 adc_diag_write_i32(milli);
263 adc_diag_write(k_adc_diag_crlf, (uint32_t)(sizeof(k_adc_diag_crlf) - 1U));
264
265 uint16_t vref = 0U;
266 const ra8_err_t vref_err = ra8_adc_read_internal_channel(k_ra8_adc_chan_int_ref_volt, &vref);
267 if (vref_err != k_ra8_ok) {
268 healthy = false;
269 }
270 adc_diag_write(k_adc_diag_vref_prefix, (uint32_t)(sizeof(k_adc_diag_vref_prefix) - 1U));
271 adc_diag_write_u32((uint32_t)vref);
272 adc_diag_write(k_adc_diag_crlf, (uint32_t)(sizeof(k_adc_diag_crlf) - 1U));
273
274 return healthy;
275}

330[[nodiscard]] static ra8_err_t adc_diag_arm(void)
331{
332 const ra8_err_t adc_err = ra8_adc_init();

340 };

344void main(void)
345{
346 adc_diag_setup_or_halt();
347 ra8_isr_globals_enable();
348
349 if (adc_diag_arm() != k_ra8_ok) {
350 adc_diag_panic_halt();
351 }
352
353 while (1) {
354 const bool healthy = adc_diag_run_cycle();
355 if (healthy) {
356 adc_diag_write(k_adc_diag_verdict_pass, (uint32_t)(sizeof(k_adc_diag_verdict_pass) - 1U));
357 } else {
358 adc_diag_write(k_adc_diag_verdict_fail, (uint32_t)(sizeof(k_adc_diag_verdict_fail) - 1U));
359 }
360 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
361 break;
362 }
363 ra8_delay_ms((uint32_t)k_adc_diag_period_ms);
364 }
365 adc_diag_panic_halt();
366}
