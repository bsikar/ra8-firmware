227RA8_INTERNAL static bool ends_with_ci(const char* name, const char* suffix)
228{
229 const size_t nl = strlen(name);
230 const size_t sl = strlen(suffix);
231 if (sl > nl) {
232 return false;
233 }
234 const char* tail = name + (nl - sl);
235 for (size_t i = 0U; i < sl; ++i) {
236 if (tolower((unsigned char)tail[i]) != tolower((unsigned char)suffix[i])) {
237 return false;
238 }
239 }
240 return true;

269RA8_INTERNAL static size_t
270list_pages(const char* dir, char names[][k_name_max], size_t cap, bool* out_truncated)
271{
272 *out_truncated = false;
273 DIR* d = opendir(dir);
274 if (d == nullptr) {
275 return 0U;
276 }
277 size_t n = 0U;
278 const struct dirent* e = readdir(d);
279 while (e != nullptr) {
280 /* Skip hidden / AppleDouble, and anything that is not a page image so a
281 * dir already holding this tool's output re-packages cleanly. A name that
282 * would not fit is rejected rather than silently truncated (which could
283 * collide two distinct pages onto one entry). */
284 if ((e->d_name[0] != '.') && is_page_image(e->d_name) &&
285 (strlen(e->d_name) < (size_t)k_name_max)) {
286 if (n >= cap) {
287 *out_truncated = true; /* more pages than the fixed table holds */
288 break;
289 }
290 (void)snprintf(names[n], k_name_max, "%s", e->d_name);
291 ++n;
292 }
293 e = readdir(d);
294 }
295 (void)closedir(d);
296 qsort(names, n, k_name_max, name_cmp);
297 return n;
298}

307RA8_INTERNAL static ra8_err_t tar_header(uint8_t* blk, const char* name, size_t size)
308{
309 if (strlen(name) >= (size_t)k_len_name) {
310 /* The ustar name field is 100 bytes; a longer name would silently truncate
311 * and two distinct pages could collide onto one entry. Refuse instead. */
312 return k_ra8_err_invalid_size;
313 }
314 memset(blk, 0, k_tar_block);
315 (void)snprintf((char*)blk + k_off_name, k_len_name, "%s", name);
316 (void)snprintf((char*)blk + k_off_mode, k_len_id, "%07o", (unsigned)k_file_mode);
317 (void)snprintf((char*)blk + k_off_uid, k_len_id, "%07o", 0U);
318 (void)snprintf((char*)blk + k_off_gid, k_len_id, "%07o", 0U);
319 (void)snprintf((char*)blk + k_off_size, k_len_size, "%011zo", size);
320 (void)snprintf((char*)blk + k_off_mtime, k_len_mtime, "%011o", 0U);
321 blk[k_off_type] = '0';
322 memcpy(blk + k_off_magic, k_ustar_magic, sizeof(k_ustar_magic));
323 memcpy(blk + k_off_version, k_ustar_version, sizeof(k_ustar_version));
324
325 memset(blk + k_off_chksum, ' ', k_len_chksum);
326 unsigned sum = 0U;
327 for (size_t i = 0U; i < (size_t)k_tar_block; ++i) {
328 sum += blk[i];
329 }
330 (void)snprintf((char*)blk + k_off_chksum, k_len_chksum - 1U, "%06o", sum);
331 blk[k_off_chksum + k_len_chksum - 1U] = ' ';
332 return k_ra8_ok;
333}

347RA8_INTERNAL static ra8_err_t tar_copy_file(FILE* in, FILE* out, size_t size)
348{
349 uint8_t chunk[k_stream_chunk];
350 size_t remaining = size;
351 while (remaining > 0U) {
352 const size_t want = (remaining < sizeof(chunk)) ? remaining : sizeof(chunk);
353 const size_t got = fread(chunk, 1U, want, in);
354 if ((got == 0U) || (fwrite(chunk, 1U, got, out) != got)) {
355 return k_ra8_fail;
356 }
357 remaining -= got;
358 }
359 const size_t pad = round_block(size) - size;
360 if (pad > 0U) {
361 const uint8_t zeros[k_tar_block] = {};
362 if (fwrite(zeros, 1U, pad, out) != pad) {
363 return k_ra8_fail;
364 }
365 }
366 return k_ra8_ok;
367}

370RA8_INTERNAL static ra8_err_t
371build_tar_to_file(const char* dir, char names[][k_name_max], size_t count, FILE* out)
372{
373 for (size_t i = 0U; i < count; ++i) {
374 char path[PATH_MAX];
375 (void)snprintf(path, sizeof(path), "%s/%s", dir, names[i]);
376 FILE* f = fopen(path, "rb");
377 if (f == nullptr) {
378 return k_ra8_fail;
379 }
380 struct stat st = {};
381 if (fstat(fileno(f), &st) != 0) {
382 (void)fclose(f);
383 return k_ra8_fail;
384 }
385 /* One fstat on the open descriptor sizes the header and bounds the copy, so
386 * the two never disagree (a file racing our output cannot overflow). */
387 const size_t sz = (size_t)st.st_size;
388 uint8_t hdr[k_tar_block];
389 const ra8_err_t hrc = tar_header(hdr, names[i], sz);
390 if (hrc != k_ra8_ok) {
391 (void)fclose(f);
392 return hrc;
393 }
394 ra8_err_t rc = (fwrite(hdr, 1U, sizeof(hdr), out) == sizeof(hdr)) ? k_ra8_ok : k_ra8_fail;
395 if (rc == k_ra8_ok) {
396 rc = tar_copy_file(f, out, sz);
397 }
398 (void)fclose(f);
399 if (rc != k_ra8_ok) {
400 return rc;
401 }
402 }
403 uint8_t trailer[2U * (size_t)k_tar_block] = {}; /* two trailing zero blocks */
404 return (fwrite(trailer, 1U, sizeof(trailer), out) == sizeof(trailer)) ? k_ra8_ok : k_ra8_fail;
405}

429RA8_INTERNAL static bool put_u32le(FILE* f, uint32_t v)
430{
431 uint8_t b[k_u32_bytes] = {};
432 for (size_t i = 0U; i < (size_t)k_u32_bytes; ++i) {
433 b[i] = (uint8_t)(v & (uint32_t)k_byte_mask);
434 v >>= (uint32_t)k_byte_bits;
435 }
436 return fwrite(b, 1U, sizeof(b), f) == sizeof(b);
437}

453RA8_INTERNAL static ra8_err_t gzip_file(const char* in_path, const char* out_path)
454{
455 FILE* in = fopen(in_path, "rb");
456 if (in == nullptr) {
457 return k_ra8_fail;
458 }
459 FILE* out = fopen(out_path, "wb");
460 if (out == nullptr) {
461 (void)fclose(in);
462 return k_ra8_fail;
463 }
464 tdefl_compressor* d = (tdefl_compressor*)malloc(sizeof(*d));
465 if (d == nullptr) {
466 (void)fclose(in);
467 (void)fclose(out);
468 return k_ra8_err_no_mem;
469 }
470 const uint8_t hdr[k_gzip_hdr_len] =
471 {k_gz_id1, k_gz_id2, k_gz_cm, 0U, 0U, 0U, 0U, 0U, 0U, k_gz_os};
472 bool ok = (fwrite(hdr, 1U, sizeof(hdr), out) == sizeof(hdr)) &&
473 (tdefl_init(d, gz_put, out, TDEFL_DEFAULT_MAX_PROBES) == TDEFL_STATUS_OKAY);
474 uint32_t crc = (uint32_t)MZ_CRC32_INIT;
475 uint32_t isize = 0U; /* ISIZE = total input length mod 2^32 */
476 uint8_t chunk[k_stream_chunk];
477 while (ok) {
478 const size_t got = fread(chunk, 1U, sizeof(chunk), in);
479 if (got == 0U) {
480 break;
481 }
482 crc = (uint32_t)mz_crc32(crc, chunk, got);
483 isize += (uint32_t)got;
484 ok = (tdefl_compress_buffer(d, chunk, got, TDEFL_NO_FLUSH) == TDEFL_STATUS_OKAY);
485 }
486 if (ok && (ferror(in) != 0)) {
487 ok = false;
488 }
489 if (ok && (tdefl_compress_buffer(d, nullptr, 0U, TDEFL_FINISH) != TDEFL_STATUS_DONE)) {
490 ok = false;
491 }
492 free(d);
493 ok = ok && put_u32le(out, crc) && put_u32le(out, isize);
494 ok = (fclose(out) == 0) && ok;
495 (void)fclose(in);
496 return ok ? k_ra8_ok : k_ra8_fail;
497}

734 const char* name,
735 size_t idx,
736 char* mani,
737 char* spine,
738 char* nav,
739 size_t cap)

744 return k_ra8_fail; /* untrusted filename must not break the container XML */

747 const int xn = snprintf(xhtml,
748 sizeof(xhtml),

763 }

771 return k_ra8_fail;
772 }
773 return epub_append_frags(mani, spine, nav, cap, esc, epub_media_type(name), idx, n);
774}

820RA8_INTERNAL static ra8_err_t
821export_epub(const char* dir, char names[][k_name_max], size_t count, const char* out_path)
822{
823 const size_t cap = (size_t)k_epub_base_bytes + (count * (size_t)k_epub_per_page_bytes);
824 char* mani = (char*)calloc(1U, cap);
825 char* spine = (char*)calloc(1U, cap);
826 char* nav = (char*)calloc(1U, cap);
827 if ((mani == nullptr) || (spine == nullptr) || (nav == nullptr)) {
828 free(mani);
829 free(spine);
830 free(nav);
831 return k_ra8_err_no_mem;
832 }
833 mz_zip_archive zip;
834 memset(&zip, 0, sizeof(zip));
835 const bool zip_open = (mz_zip_writer_init_file(&zip, out_path, 0) != MZ_FALSE);
836 ra8_err_t rc = zip_open ? k_ra8_ok : k_ra8_fail;
837 if ((rc == k_ra8_ok) && (!epub_add_str(&zip, "mimetype", "application/epub+zip") ||
838 !epub_add_str(&zip, "META-INF/container.xml", k_epub_container_xml))) {
839 rc = k_ra8_fail;
840 }
841 for (size_t i = 0U; (rc == k_ra8_ok) && (i < count); ++i) {
842 rc = epub_add_page(&zip, dir, names[i], i, mani, spine, nav, cap);
843 }
844 if (rc == k_ra8_ok) {
845 rc = epub_add_meta(&zip, mani, spine, nav);
846 }
847 if (zip_open) {
848 (void)mz_zip_writer_end(&zip);
849 }
850 free(mani);
851 free(spine);
852 free(nav);
853 /* A partial EPUB is ::export_atomic's temp to discard, not ours. */
854 return rc;
855}

878 return k_ra8_fail;
879 }
880 const bool ok = (WIFEXITED(status) != 0) && (WEXITSTATUS(status) == 0);
881 return ok ? k_ra8_ok : k_ra8_fail;

883 (void)cbz;
884 (void)out_path;
885 (void)fprintf(stderr, "media_dl: rabook support not built (MDL_EPUB_COMPILE_DIR unset)\n");

910RA8_INTERNAL static ra8_err_t export_dispatch(mdl_format_t fmt,
911 const char* dir,
912 char names[][k_name_max],

914 const char* out_path)

931 case k_mdl_fmt_jof:
932 case k_mdl_fmt_loose:

936 }

949RA8_INTERNAL static ra8_err_t export_atomic(mdl_format_t fmt,
950 const char* dir,
951 char names[][k_name_max],
952 size_t count,
953 const char* out_path)

967ra8_err_t mdl_export_chapter(mdl_format_t fmt, const char* chapter_dir, const char* out_path)
968{
969 if ((chapter_dir == nullptr) || (out_path == nullptr) || (fmt == k_mdl_fmt_loose) ||
970 (fmt == k_mdl_fmt_invalid)) {
971 return k_ra8_err_invalid_arg;
972 }
973 if (fmt == k_mdl_fmt_cbr) {
974 /* `rar` archives the directory itself, so it needs no page table. */
975 return export_atomic(fmt, chapter_dir, nullptr, 0U, out_path);
976 }
977
978 static char s_names[k_max_pages][k_name_max];
979 bool truncated = false;
980 const size_t count = list_pages(chapter_dir, s_names, (size_t)k_max_pages, &truncated);
981 if (truncated) {
982 /* More page images than the fixed table holds: fail rather than silently
983 * package a short chapter the reader would discover missing pages in. */
984 return k_ra8_err_invalid_size;
985 }
986 if (count == 0U) {
987 return k_ra8_err_empty;
988 }
989 if (fmt == k_mdl_fmt_jof) {
990 /* JOF writes one `.jof` sibling per page into chapter_dir; out_path names
991 * no single container (see mdl_format_is_dir_output), so there is no single
992 * file to rename into place -- mdl_export_jof commits each page itself. */
993 return mdl_export_jof(chapter_dir, s_names, count);
994 }
995 return export_atomic(fmt, chapter_dir, s_names, count, out_path);
996}
