86static void kb_print_uint(uint32_t value)
87{
88 uint8_t buf[k_kb_dec_ten];
89 uint32_t n = 0U;
90 if (value == 0U) {
91 buf[n] = '0';
92 n++;
93 }
94 while ((value > 0U) && (n < (uint32_t)k_kb_dec_ten)) {
95 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_kb_dec_ten));
96 n++;
97 value /= (uint32_t)k_kb_dec_ten;
98 }
99 for (uint32_t i = 0U; i < n; i++) {
100 kb_print(&buf[n - 1U - i], 1U);
101 }
102}

127static void kb_tap(ra8_kbd_text_t* t, uint8_t idx, uint32_t* taps)
128{
129 const ra8_ui_rect_t* r = &s_kb.keys[idx].rect;
130 const int32_t cx = r->x + (r->w / (int32_t)k_kb_half);
131 const int32_t cy = r->y + (r->h / (int32_t)k_kb_half);
132 const uint8_t hit = ra8_kbd_hit(&s_kb, cx, cy);
133 (void)ra8_kbd_apply(t, &s_kb, hit);
134 (*taps)++;
135}

156static bool kb_streq(const char* a, const char* b)
157{
158 for (uint32_t i = 0U; i < (uint32_t)k_ra8_kbd_text_max; i++) {
159 if (a[i] != b[i]) {
160 return false;
161 }
162 if (a[i] == '\0') {
163 return true;
164 }
165 }
166 return false;

181int32_t main(void)
182{
183 kb_setup_or_halt();
184 ra8_isr_globals_enable();
185 kb_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
186
187 const ra8_ui_rect_t frame = {.x = k_kb_frame_x,
188 .y = k_kb_frame_y,
189 .w = k_kb_frame_w,
190 .h = k_kb_frame_h};
191 if (ra8_kbd_layout_init(&s_kb, &frame) != k_ra8_ok) {
192 kb_panic_halt(k_msg_lay, (uint32_t)sizeof(k_msg_lay) - 1U);
193 }
194
195 ra8_kbd_text_t t;
196 if (ra8_kbd_text_init(&t) != k_ra8_ok) {
197 kb_panic_halt(k_msg_lay, (uint32_t)sizeof(k_msg_lay) - 1U);
198 }
199
200 /* Type "Hi 9" -- exercises one-shot SHIFT (caps), SPACE, and the 123 layer
201 * toggle to reach a digit, then commits with RETURN. */
202 uint32_t taps = 0U;
203 kb_tap(&t, kb_key_of_kind(k_ra8_kbd_key_shift), &taps); /* -> capitalise next */
204 for (uint32_t i = 0U; i < (uint32_t)(sizeof(k_kb_word) - 1U); i++) {
205 kb_tap(&t, kb_key_of(k_kb_word[i]), &taps); /* h -> H (one-shot), then i */
206 }
207 kb_tap(&t, kb_key_of_kind(k_ra8_kbd_key_space), &taps);
208 kb_tap(&t, kb_key_of_kind(k_ra8_kbd_key_layer), &taps); /* 123 -> numbers layer */
209 kb_tap(&t, kb_key_of('9'), &taps); /* digit, now reachable */
210 kb_tap(&t, kb_key_of_kind(k_ra8_kbd_key_enter), &taps); /* RETURN commits */
211
212 if (!kb_streq(t.buf, k_kb_expected) || !t.committed) {
213 kb_panic_halt(k_msg_miss, (uint32_t)sizeof(k_msg_miss) - 1U);
214 }
215
216 kb_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
217 kb_print((const uint8_t*)t.buf, (uint32_t)t.len);
218 kb_print(k_msg_mid, (uint32_t)sizeof(k_msg_mid) - 1U);
219 kb_print_uint((uint32_t)(t.committed ? 1U : 0U));
220 kb_print(k_msg_taps, (uint32_t)sizeof(k_msg_taps) - 1U);
221 kb_print_uint(taps);
222 kb_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
223
224 while (1) {
225 __asm__ volatile("wfi");
226 }
227}
