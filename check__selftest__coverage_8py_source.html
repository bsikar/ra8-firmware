1#!/usr/bin/env python3
2# SPDX-License-Identifier: MIT
3# Copyright (c) 2026 Brighton Sikarskie
4"""Enforce the ``--selftest`` requirement that nothing used to enforce.
5
6The repo's stated remedy for its dominant defect class -- a detector that has
7quietly stopped matching -- is a ``--selftest`` asserting BOTH directions.
8``scripts/ci.sh`` says so and ``CLAUDE.md`` says so. Nothing checked it, so a
9new checker with no selftest, or one whose gate never ran it, landed clean
10(#531). Two of the checkers that turned out to have no selftest were the two
11that had silently stopped seeing their subject -- ``check_obsolete_standards``
12scanning 0 files and ``audit_init_order`` reaching 11 of 217 apps. That is not
13a coincidence, and it is why this gate exists.
14
15WHAT THE RULE IS, AND WHY IT IS NARROWED
16----------------------------------------
17
18An unenforceable rule in ``CLAUDE.md`` is itself the defect class, so the rule
19is scoped to where it can be both meaningful and true:
20
21 **Rule A (universal).** ANY first-party script a gate body invokes that
22 *has* a selftest must have it RUN by that gate. A selftest nobody executes
23 is documentation. This has no exceptions and no baseline.
24
25 **Rule B (detectors).** Every script under ``scripts/checks/`` -- plus the
26 ``scripts/ci/check_*.py`` meta-checkers -- that a gate body invokes must
27 ACCEPT a selftest.
28
29Rule B is keyed on the taxonomy ``CLAUDE.md`` already documents, in which
30``scripts/`` is organised by the QUESTION a script answers: ``checks/`` is
31"Is the tree OK?" -- read-only, exits non-zero when it is not. Those are the
32detectors, and a detector is exactly the thing that can stop detecting.
33``builders/`` produce a build output, ``report/`` "never fails on content",
34and ``hil/`` drives the bench; demanding a both-directions selftest of
35``scripts/builders/docs.sh`` would be ceremony, and a gate that demands
36ceremony gets disabled. Scoping by the repo's own stated organisation is a
37principle, not an allowlist.
38
39THE BACKLOG IS RATCHETED, NOT WAIVED
40------------------------------------
41
42Turning Rule B on found a real backlog. Failing on all of it at once would
43turn the tree red and teach everyone to skip the gate, so the pre-existing
44offenders are frozen in ``.github/selftest-baseline.txt`` -- the same shape as
45``tidy_ratchet.py`` and ``misra_ratchet.py``. A NEW gate-wired detector with
46no selftest fails immediately. A baseline row whose checker has since GAINED
47a selftest also fails, so the list can only shrink: the ratchet tightens
48itself rather than waiting to be tidied.
49
50Run::
51
52 check_selftest_coverage.py # the gate
53 check_selftest_coverage.py --list # what is scanned, and its status
54 check_selftest_coverage.py --update # re-seed the baseline (shrink only)
55 check_selftest_coverage.py --selftest # prove both directions
56
57Exit 0 if clean, 1 on a violation, 2 when the scan itself collapsed.
58"""

68REPO_ROOT = Path(__file__).resolve().parents[2]
69GATE_DIR = REPO_ROOT / "scripts" / "ci" / "gates"
70BASELINE_FILE = REPO_ROOT / ".github" / "selftest-baseline.txt"

72# A first-party script path as it appears in a gate body.
73INVOKE_RE = re.compile(r"(?<![\w./-])(scripts/[\w./-]+\.(?:py|sh))")
74
75# A selftest IMPLEMENTATION. Both spellings are live in this tree: the flag
76# form (`--selftest`, argparse or a bash `case` arm) and the subcommand form
77# (`monitor.sh selftest`, dispatched through a `cmd_selftest` function).
78# Matching only the flag form would report monitor.sh as having none, which is
79# false -- and a rule that fires on a compliant file is how a gate gets turned
80# off.
81IMPL_RE = re.compile(
82 r"""["']--selftest["']|--selftest\‍)|\bcmd_selftest\b|^\s*selftest\‍)""",
83 re.MULTILINE,
84)
85
86# A selftest INVOCATION on one line of a gate body, in either spelling.
87RUN_RE = re.compile(r"--selftest\b|\bselftest\b")
88
89# Directories whose scripts are DETECTORS and therefore owe a selftest under
90# Rule B. Derived from the scripts/ taxonomy documented in CLAUDE.md.
91DETECTOR_DIRS = ("scripts/checks/",)
92# The sibling meta-checkers, which live one directory up from checks/.
93DETECTOR_META_RE = re.compile(r"^scripts/ci/check_[\w.]+\.py$") # PATHREF-OK: a regex, not a path
94
95# A tree with no gate-invoked scripts means the parser stopped matching, not
96# that the gates stopped calling checkers. Refuse to report clean against
97# nothing. Measured 94 gate-invoked first-party scripts on 2026-07-28.
98MIN_INVOKED = 40

105def is_detector(rel: str) -> bool:
106 """Report whether `rel` is a detector owing a selftest under Rule B.
107
108 Args:
109 rel: Repo-relative script path.
110
111 Returns:
112 True for ``scripts/checks/*`` and the ``scripts/ci/check_*.py`` meta

118def scan_gate_invocations(text: str) -> dict[str, bool]:
119 """Map every first-party script invoked in `text` to whether a selftest ran.
120
121 Args:
122 text: A gate fragment's source.
123
124 Returns:
125 ``script path -> True`` when at least one invocation of that script in
126 this text carried a selftest.
127 """
128 found: dict[str, bool] = {}
129 for line in text.splitlines():
130 if line.lstrip().startswith("#"):
131 continue
132 ran = bool(RUN_RE.search(line))
133 for match in INVOKE_RE.finditer(line):
134 rel = match.group(1)
135 found[rel] = found.get(rel, False) or ran
136 return found

139def collect() -> dict[str, bool]:
140 """Gather every gate-invoked script and whether any gate runs its selftest.
141
142 Gate bodies delegate. ``gate_format`` invokes ``format_code.sh``, and
143 *that* is what drives ``check_comment_format.py`` -- a detector every bit as
144 gate-wired as one named in the fragment itself. Reading only the fragments
145 made such a script invisible: it was neither credited as invoked nor asked
146 for its selftest, so the checker reported clean over a detector no gate ever
147 proved. The walk therefore follows first-party shell helpers to a fixed
148 point, `seen` guarding against a helper cycle.
149
150 Returns:
151 ``script path -> selftest is invoked somewhere along the gate's reach``.
152 """
153 direct: dict[str, bool] = {}
154 for fragment in sorted(GATE_DIR.glob("*.sh")):
155 for rel, ran in scan_gate_invocations(fragment.read_text(encoding="utf-8")).items():
156 direct[rel] = direct.get(rel, False) or ran
157
158 def read(rel: str) -> str | None:
159 path = REPO_ROOT / rel
160 return path.read_text(encoding="utf-8") if path.is_file() else None
161
162 return expand_helpers(direct, read)

165def expand_helpers(
166 direct: dict[str, bool], read_text: Callable[[str], str | None]
167) -> dict[str, bool]:
168 """Extend `direct` with the scripts its shell helpers invoke, transitively.
169
170 Args:
171 direct: ``script path -> a selftest ran`` as read from the gate bodies.
172 read_text: Returns a script's source, or None when it cannot be read.
173
174 Returns:
175 The same mapping, plus every script reachable through a ``.sh`` helper.
176 `seen` makes a helper cycle terminate rather than spin.
177 """
178 out = dict(direct)
179 queue = list(direct)
180 seen: set[str] = set()
181 while queue:
182 rel = queue.pop()
183 if rel in seen or not rel.endswith(".sh"):
184 continue
185 seen.add(rel)
186 text = read_text(rel)
187 if text is None:
188 continue
189 for sub, ran in scan_gate_invocations(text).items():
190 out[sub] = out.get(sub, False) or ran
191 queue.append(sub)
192 return out

196 """Report whether the script at `rel` implements a selftest.

198 Args:
199 rel: Repo-relative script path.

211def evaluate(invoked: dict[str, bool]) -> tuple[list[str], list[str]]:
212 """Split the gate-invoked scripts into Rule A and Rule B offenders.
213
214 Args:
215 invoked: Output of `collect`.
216
217 Returns:
218 ``(rule_a, rule_b)`` -- scripts with an unrun selftest, and detectors
219 with no selftest at all. Both sorted.
220 """
221 rule_a: list[str] = []
222 rule_b: list[str] = []
223 for rel, ran in sorted(invoked.items()):
224 if not (REPO_ROOT / rel).is_file():
225 continue
226 if has_selftest(rel):
227 if not ran:
228 rule_a.append(rel)
229 elif is_detector(rel):
230 rule_b.append(rel)
231 return rule_a, rule_b

236

280 """Print every violation with the fix spelled out.
281
282 Args:
283 rule_a: Scripts whose selftest no gate runs.
284 new_b: Detectors newly missing a selftest.
285 stale: Baseline rows whose checker gained a selftest.

293 for rel in new_b:

298 " must-fire case and a must-stay-quiet case) and run it in the gate.\n"
299 " Do NOT add a row to .github/selftest-baseline.txt -- that list is\n"
300 " frozen pre-existing debt and may only shrink.\n\n"
301 )
302 for rel in stale:
303 sys.stderr.write(

309def run_check() -> int:
310 """Apply Rule A and Rule B against the frozen baseline.
311
312 Returns:
313 0 clean, 1 on a violation, 2 when the scan collapsed.
314 """
315 invoked = collect()
316 if len(invoked) < MIN_INVOKED:
317 sys.stderr.write(
318 f"check_selftest_coverage.py: FATAL -- only {len(invoked)} gate-invoked "
319 f"script(s) found, floor is {MIN_INVOKED}.\n"
320 " A collapsed scan reports full selftest coverage because it saw nothing.\n"
321 )
322 return EXIT_VACUOUS
323
324 rule_a, rule_b = evaluate(invoked)
325 baseline = load_baseline()
326 new_b = [rel for rel in rule_b if rel not in baseline]
327 stale = sorted(rel for rel in baseline if rel not in rule_b)
328
329 if rule_a or new_b or stale:
330 sys.stderr.write("check_selftest_coverage.py: selftest requirement violated\n\n")
331 _report(rule_a, new_b, stale)
332 total = len(rule_a) + len(new_b) + len(stale)
333 sys.stderr.write(f"{total} violation(s).\n")
334 return EXIT_VIOLATION
335
336 print(
337 f"check_selftest_coverage.py: clean -- {len(invoked)} gate-invoked script(s); "
338 f"every selftest present is run; {len(baseline)} detector(s) still owe one "
339 "(frozen in .github/selftest-baseline.txt)."
340 )
341 return EXIT_OK

344def run_list() -> int:
345 """Print every gate-invoked script with its selftest status.
346
347 Returns:
348 Always 0.
349 """
350 invoked = collect()
351 for rel, ran in sorted(invoked.items()):
352 impl = has_selftest(rel)
353 kind = "detector" if is_detector(rel) else "other "
354 print(f"{kind} impl={'Y' if impl else 'n'} run={'Y' if ran else 'n'} {rel}")
355 print(f"total: {len(invoked)}")
356 return EXIT_OK

359def run_update() -> int:
360 """Re-seed the baseline, refusing to grow it.
361
362 Returns:
363 0 on success, 1 when the update would freeze new debt.
364 """
365 _, rule_b = evaluate(collect())
366 current = set(rule_b)
367 seeding = not BASELINE_FILE.is_file()
368 baseline = load_baseline()
369 grew = sorted(current - baseline)
370 if grew and not seeding:
371 sys.stderr.write(
372 f"refusing to --update: {len(grew)} row(s) would be ADDED. The baseline is "
373 "a burn-down; write the missing selftests instead.\n"
374 )
375 for rel in grew:
376 sys.stderr.write(f" {rel}\n")
377 return EXIT_VIOLATION
378 write_baseline(current)
379 print(f"check_selftest_coverage.py: wrote {len(current)} row(s) to {BASELINE_FILE.name}")
380 return EXIT_OK

417def _rule_a_cases() -> list[tuple[str, bool]]:
418 """Rule A over the fixture gate bodies: it must fire on an unrun selftest.
419
420 Returns:
421 ``(label, held)`` per fixture, the label carrying which direction the
422 fixture asserts so a failure names it.
423 """
424 out: list[tuple[str, bool]] = []
425 for label, text, must_fire in _selftest_cases():
426 invoked = scan_gate_invocations(text)
427 fired = any(
428 has_selftest(rel) and not ran
429 for rel, ran in invoked.items()
430 if (REPO_ROOT / rel).is_file()
431 )
432 expectation = "must fire" if must_fire else "must stay quiet"
433 out.append((f"{label} ({expectation})", fired == must_fire))
434 return out

437def _helper_walk_cases() -> list[tuple[str, bool]]:
438 """The helper walk, driven off a fixture filesystem.
439
440 Without the walk a detector invoked through a gate's shell helper is
441 invisible in both directions: never credited as gate-wired, never asked for
442 its selftest.
443
444 Returns:
445 ``(label, held)`` per case.
446 """
447 helper = "scripts/checks/helper.sh" # PATHREF-OK: selftest fixture, not a real script
448 detector = "scripts/checks/check_thing.py" # PATHREF-OK: selftest fixture, not a real script
449 other = "scripts/checks/other.sh" # PATHREF-OK: selftest fixture, not a real script
450 quiet_helper = {helper: f"python3 {detector}\n"}
451 loud_helper = {helper: f"python3 {detector} --selftest\npython3 {detector}\n"}
452 cyclic = {helper: f"bash {other}\n", other: f"bash {helper}\n"}
453
454 reached_quiet = expand_helpers({helper: False}, quiet_helper.get)
455 reached_loud = expand_helpers({helper: False}, loud_helper.get)
456 reached_cycle = expand_helpers({helper: False}, cyclic.get)
457
458 return [
459 (
460 "a detector reached through a gate helper is seen",
461 detector in reached_quiet,
462 ),
463 (
464 "its selftest going unrun through that helper is reported",
465 reached_quiet.get(detector) is False,
466 ),
467 (
468 "its selftest being run through that helper counts",
469 reached_loud.get(detector) is True,
470 ),
471 ("a helper cycle terminates", other in reached_cycle),
472 ]

532def selftest() -> int:
533 """Prove Rule A fires on an unrun selftest and spares a run one.
534
535 Returns:
536 0 when every case holds, 1 otherwise.
537 """
538 families = (_rule_a_cases, _helper_walk_cases, _taxonomy_cases, _live_scan_cases)
539 failures = sum(_report_cases(family()) for family in families)
540 if failures:
541 sys.stderr.write(f"check_selftest_coverage.py --selftest: {failures} case(s) failed.\n")
542 return EXIT_VIOLATION
543 print("check_selftest_coverage.py --selftest: all cases pass (both directions).")
544 return EXIT_OK

547def main() -> int:
548 """Parse arguments and dispatch.
549
550 Returns:
551 A process exit status.
552 """
553 parser = argparse.ArgumentParser(description="enforce the --selftest requirement")
554 parser.add_argument("--check", action="store_true", help="apply the rules (the gate mode)")
555 parser.add_argument("--list", action="store_true", help="print the scanned set and status")
556 parser.add_argument("--update", action="store_true", help="re-seed the baseline (shrink only)")
557 parser.add_argument("--selftest", action="store_true", help="prove both directions, then exit")
558 args = parser.parse_args()
559
560 if not GATE_DIR.is_dir():
561 sys.stderr.write(
562 f"check_selftest_coverage.py: FATAL -- {GATE_DIR} does not exist; the gate "
563 "bodies moved and this checker is scanning nothing.\n"
564 )
565 return EXIT_VACUOUS
566 if args.selftest:
567 return selftest()
568 if args.list:
569 return run_list()
570 if args.update:
571 return run_update()
572 if not args.check:
573 parser.error("one of --check / --list / --update / --selftest is required")
574 return run_check()
