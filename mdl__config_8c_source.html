173RA8_INTERNAL static bool internal_config_parse_u32(const char* val, uint32_t* out)
174{
175 if ((val[0] == '\0') || (val[0] == '+') || (val[0] == '-')) {
176 return false;
177 }
178 uint32_t parsed = 0U;
179 for (size_t index = 0U; val[index] != '\0'; ++index) {
180 if ((val[index] < '0') || (val[index] > '9')) {
181 return false;
182 }
183 const uint32_t digit = (uint32_t)(val[index] - '0');
184 if (parsed > ((UINT32_MAX - digit) / (uint32_t)k_config_dec_base)) {
185 return false;
186 }
187 parsed = (parsed * (uint32_t)k_config_dec_base) + digit;
188 }
189 *out = parsed;
190 return true;
191}

357RA8_INTERNAL static ra8_err_t
358internal_config_next(internal_config_reader_t* reader, uint8_t* out_byte, bool* out_eof)
359{
360 *out_eof = false;
361 while (reader->position == reader->available) {
362 if (reader->eof) {
363 *out_eof = true;
364 return k_ra8_ok;
365 }
366 if (reader->read_calls >= (uint32_t)k_config_read_calls) {
367 return k_ra8_err_invalid_size;
368 }
369 uint32_t count = 0U;
370 const ra8_err_t err = fw_fs_read(&reader->file,
371 reader->storage->io_buffer,
372 reader->storage->io_buffer_bytes,
373 &count);
374 ++reader->read_calls;
375 if (err != k_ra8_ok) {
376 return err;
377 }
378 reader->position = 0U;
379 reader->available = count;
380 reader->eof = count == 0U;
381 }
382 *out_byte = reader->storage->io_buffer[reader->position];
383 ++reader->position;
384 return k_ra8_ok;
385}

406RA8_INTERNAL static ra8_err_t internal_config_read_line(internal_config_reader_t* reader,
407 char* line,
408 uint32_t capacity,
409 bool* out_present)
410{
411 uint32_t length = 0U;
412 *out_present = false;
413 for (;;) {
414 uint8_t byte = 0U;
415 bool eof = false;
416 ra8_err_t err = internal_config_next(reader, &byte, &eof);
417 if (err != k_ra8_ok) {
418 return err;
419 }
420 if (eof) {
421 line[length] = '\0';
422 *out_present = length > 0U;
423 return k_ra8_ok;
424 }
425 *out_present = true;
426 if (byte == (uint8_t)'\n') {
427 line[length] = '\0';
428 return k_ra8_ok;
429 }
430 if ((length + 1U) >= capacity) {
431 return k_ra8_err_invalid_state;
432 }
433 line[length] = (char)byte;
434 ++length;
435 }

520RA8_INTERNAL static bool internal_config_valid(const mdl_site_t* site)
521{
522 const char* selectors[] = {site->series_title_selector,
523 site->series_summary_selector,
524 site->series_author_selector,
525 site->series_artist_selector,
526 site->series_cover_selector,
527 site->chapter_title_selector,
528 site->chapter_number_selector};
529 for (size_t i = 0U; i < (sizeof(selectors) / sizeof(selectors[0])); ++i) {
530 const char* selector = selectors[i];
531 if (selector[0] == '\0') {
532 continue;
533 }
534 const char* prefixes[] = {"meta:", "class:", "label:", "literal:"};
535 bool valid = false;
536 for (size_t prefix_index = 0U; prefix_index < (sizeof(prefixes) / sizeof(prefixes[0]));
537 ++prefix_index) {
538 const size_t prefix_len = strlen(prefixes[prefix_index]);
539 if ((strncmp(selector, prefixes[prefix_index], prefix_len) == 0) &&
540 (selector[prefix_len] != '\0')) {
541 valid = true;
542 break;
543 }
544 }
545 if (!valid) {
546 return false;
547 }
548 }
549 return (site->host[0] != '\0') && (site->language[0] != '\0') &&
550 ((strcmp(site->reading_direction, "ltr") == 0) ||
551 (strcmp(site->reading_direction, "rtl") == 0)) &&
552 (site->img_delay_min <= site->img_delay_max) &&
553 (site->chapter_delay_min <= site->chapter_delay_max) && (site->burst > 0U) &&
554 (site->max_inflight > 0U) && (site->backoff_base_ms <= site->backoff_max_ms);
555}

557ra8_err_t mdl_config_load(mdl_storage_t* storage, const char* path, mdl_site_t* out)
558{
559 if ((storage == nullptr) || (storage->fs == nullptr) || (storage->file_workspace == nullptr) ||
560 (storage->io_buffer == nullptr) || (storage->io_buffer_bytes == 0U) || (path == nullptr) ||
561 (out == nullptr)) {
562 return k_ra8_err_invalid_arg;
563 }
564 internal_config_set_defaults(out);
565
566 internal_config_reader_t reader = {.storage = storage};
567 ra8_err_t err = fw_fs_open(&storage->fs->streams,
568 path,
569 k_fw_fs_open_read,
570 &reader.file,
571 storage->file_workspace,
572 storage->file_workspace_bytes);
573 if (err != k_ra8_ok) {
574 return err;
575 }
576 uint64_t file_bytes = 0U;
577 err = fw_fs_file_size(&reader.file, &file_bytes);
578 if ((err == k_ra8_ok) && (file_bytes > (uint64_t)k_config_file_max)) {
579 err = k_ra8_err_invalid_size;
580 }
581 if (err == k_ra8_ok) {
582 err = internal_config_parse(&reader, out);
583 }
584 const ra8_err_t close_err = fw_fs_close(&reader.file);
585 if (close_err != k_ra8_ok) {
586 return close_err;
587 }
588 if (err != k_ra8_ok) {
589 return err;
590 }
591 return internal_config_valid(out) ? k_ra8_ok : k_ra8_err_invalid_state;
592}
