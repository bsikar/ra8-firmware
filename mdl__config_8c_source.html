43RA8_INTERNAL static char* trim(char* s)
44{
45 while ((*s == ' ') || (*s == '\t') || (*s == '\r') || (*s == '\n')) {
46 ++s;
47 }
48 size_t n = strlen(s);
49 while ((n > 0U) && ((s[n - 1U] == ' ') || (s[n - 1U] == '\t') || (s[n - 1U] == '\r') ||
50 (s[n - 1U] == '\n'))) {
51 s[n - 1U] = '\0';
52 --n;
53 }
54 return s;
55}

64RA8_INTERNAL static bool apply_kv_str(mdl_site_t* s, const char* key, const char* val)
65{
66 const struct {
67 const char* key;
68 char* dst;
69 size_t cap;
70 } fields[] = {

72 {"host", s->host, sizeof(s->host)},
73 {"kind", s->kind, sizeof(s->kind)},
74 {"contact", s->contact, sizeof(s->contact)},
75 {"chapter_url_contains", s->chapter_url_contains, sizeof(s->chapter_url_contains)},
76 {"page_img_attr", s->page_img_attr, sizeof(s->page_img_attr)},
77 {"page_img_url_contains", s->page_img_url_contains, sizeof(s->page_img_url_contains)},
78 {"search_url", s->search_url, sizeof(s->search_url)},
79 {"search_result_contains", s->search_result_contains, sizeof(s->search_result_contains)},
80 {"browse_url", s->browse_url, sizeof(s->browse_url)},
81 };
82 for (size_t i = 0U; i < (sizeof(fields) / sizeof(fields[0])); ++i) {
83 if (strcmp(key, fields[i].key) == 0) {
84 set_str(fields[i].dst, fields[i].cap, val);
85 return true;
86 }
87 }
88 return false;

166RA8_INTERNAL static void config_parse_stream(FILE* fp, mdl_site_t* out)
167{
168 char line[k_line_max];
169 while (fgets(line, (int)sizeof(line), fp) != nullptr) {
170 char* p = trim(line);
171 if ((p[0] == '\0') || (p[0] == '#') || (p[0] == '[')) {
172 continue; /* blank / comment / section header */
173 }
174 char* eq = strchr(p, '=');
175 if (eq == nullptr) {
176 continue; /* not a key=value line */
177 }
178 *eq = '\0';
179 const char* key = trim(p);
180 const char* val = trim(eq + 1);
181 if (key[0] != '\0') {
182 apply_kv(out, key, val);
183 }
184 }
185}

187ra8_err_t mdl_config_load(const char* path, mdl_site_t* out)
188{
189 if ((path == nullptr) || (out == nullptr)) {
190 return k_ra8_err_invalid_arg;
191 }
192 config_set_defaults(out);
193
194 FILE* fp = fopen(path, "r");
195 if (fp == nullptr) {
196 (void)fprintf(stderr, "media_dl: cannot open config '%s'\n", path);
197 return k_ra8_fail;
198 }
199 config_parse_stream(fp, out);
200 (void)fclose(fp);
201
202 if (out->host[0] == '\0') {
203 (void)fprintf(stderr, "media_dl: config missing required 'host'\n");
204 return k_ra8_err_invalid_state;
205 }
206 return k_ra8_ok;
207}
