187static bool s_vint(const uint8_t* buf, size_t len, size_t* pos, uint64_t* val)
188{
189 uint32_t words[2] = {0U, 0U}; /* [0] = low 32 bits, [1] = high 32 bits */
190 uint32_t shift = 0U;
191 const uint32_t hi_bit = (uint32_t)k_rar_hi_sh; /* 32 */
192 const uint32_t straddle = (uint32_t)k_rar_hi_sh - (uint32_t)k_rar_vint_bits; /* 25 */
193 for (uint16_t i = 0U; i < (uint16_t)k_ra8_rar_vint_max; ++i) { /* <=10 bytes */
194 if (*pos >= len) {
195 return false;
196 }
197 const uint8_t b = buf[*pos];
198 *pos += 1U;
199 const uint32_t d = (uint32_t)(b & (uint8_t)k_rar_vint_data);
200 if (shift < hi_bit) {
201 words[0] |= (uint32_t)(d << shift);
202 if (shift > straddle) { /* this 7-bit group straddles the 32-bit boundary */
203 words[1] |= (uint32_t)(d >> (hi_bit - shift));
204 }
205 } else {
206 words[1] |= (uint32_t)(d << (shift - hi_bit));
207 }
208 if ((b & (uint8_t)k_rar_vint_cont) == 0U) {
209 (void)memcpy(val, words, sizeof(*val)); /* LE host + target: word[0] low */
210 return true;
211 }
212 shift += (uint32_t)k_rar_vint_bits;
213 }
214 return false;
215}

320 out->pack_size = pack;

352s_rar4_block(const ra8_rar_t* rar, uint64_t off, char* nbuf, uint16_t ncap, ra8_rar_entry_t* out)
353{
354 uint8_t scratch[k_ra8_rar_hdr_scratch] = {};
355 const size_t n = rar->read(rar->ctx, off, scratch, sizeof(scratch));
356 if (n < (size_t)k_rar4_base_len) {
357 return k_ra8_err_validation_failed;
358 }
359 const uint8_t type = scratch[k_rar4_off_type];
360 const uint16_t flags = s_le16(&scratch[k_rar4_off_flags]);
361 const uint16_t hsize = s_le16(&scratch[k_rar4_off_size]);
362 if (hsize < (uint16_t)k_rar4_base_len) {
363 return k_ra8_err_validation_failed;
364 }
365 const bool islong = (flags & (uint16_t)k_rar4_flag_long) != 0U;
366 uint64_t add = 0U;
367 if (islong) {
368 if (n >= ((size_t)k_rar4_off_add + (size_t)k_rar_w32)) {
369 add = (uint64_t)s_le32(&scratch[k_rar4_off_add]);
370 }
371 }
372 if (type == (uint8_t)k_rar4_type_file) {
373 if (!s_rar4_file(rar, off, scratch, n, flags, hsize, nbuf, ncap, out)) {
374 return k_ra8_err_validation_failed;
375 }
376 } else {
377 out->next_off = off + (uint64_t)hsize + add;
378 }
379 if (out->next_off <= off) {
380 return k_ra8_err_validation_failed;
381 }
382 return k_ra8_ok;
383}

530static ra8_err_t
531s_rar5_block(const ra8_rar_t* rar, uint64_t off, char* nbuf, uint16_t ncap, ra8_rar_entry_t* out)
532{
533 uint8_t scratch[k_ra8_rar_hdr_scratch] = {};
534 const size_t n = rar->read(rar->ctx, off, scratch, sizeof(scratch));
535 size_t pos = (size_t)k_rar_w32; /* skip the 4-byte header CRC32 */
536 uint64_t hsize = 0U;
537 if (!s_vint(scratch, n, &pos, &hsize)) {
538 return k_ra8_err_validation_failed;
539 }
540 const uint64_t data_off = off + (uint64_t)pos + hsize; /* header data ends here */
541 uint64_t htype = 0U;
542 uint64_t hflags = 0U;
543 if (!s_vint(scratch, n, &pos, &htype)) {
544 return k_ra8_err_validation_failed;
545 }
546 if (!s_vint(scratch, n, &pos, &hflags)) {
547 return k_ra8_err_validation_failed;
548 }
549 uint64_t extra = 0U;
550 if ((hflags & (uint64_t)k_rar5_hflag_extra) != 0U) {
551 if (!s_vint(scratch, n, &pos, &extra)) {
552 return k_ra8_err_validation_failed;
553 }
554 }
555 (void)extra; /* extra-area length is skipped via the header-size vint */
556 uint64_t dsize = 0U;
557 if ((hflags & (uint64_t)k_rar5_hflag_data) != 0U) {
558 if (!s_vint(scratch, n, &pos, &dsize)) {
559 return k_ra8_err_validation_failed;
560 }
561 }
562 out->next_off = data_off + dsize;
563 if (htype == (uint64_t)k_rar5_type_file) {
564 if (!s_rar5_file(rar, off, scratch, n, pos, data_off, dsize, nbuf, ncap, out)) {
565 return k_ra8_err_validation_failed;
566 }
567 }
568 if (out->next_off <= off) {
569 return k_ra8_err_validation_failed;
570 }
571 return k_ra8_ok;
572}

595static void s_rar_match_signature(const uint8_t* hdr, size_t got, ra8_rar_t* rar)
596{
597 static const char k_sig5[] = "Rar!\x1A\x07\x01"; /* + implicit NUL = 8-byte RAR5 sig */
598 static const char k_sig4[] = "Rar!\x1A\x07"; /* + implicit NUL = 7-byte RAR4 marker */
599 static_assert(sizeof(k_sig5) == (size_t)k_ra8_rar_sig5_len, "RAR5 signature length");
600 static_assert(sizeof(k_sig4) == (size_t)k_ra8_rar_sig4_len, "RAR4 marker length");
601 if (got >= (size_t)k_ra8_rar_sig5_len) {
602 if (memcmp(hdr, k_sig5, (size_t)k_ra8_rar_sig5_len) == 0) {
603 rar->version = k_ra8_rar_ver_5;
604 rar->first_off = (uint64_t)k_ra8_rar_sig5_len;
605 return;
606 }
607 }
608 if (memcmp(hdr, k_sig4, (size_t)k_ra8_rar_sig4_len) == 0) {
609 rar->version = k_ra8_rar_ver_4;
610 rar->first_off = (uint64_t)k_ra8_rar_sig4_len;
611 }
612}

733 total += r;
734 }
