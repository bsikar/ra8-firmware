169[[nodiscard]] ra8_err_t ra8_lpm_graphics_power_on(uint32_t timeout_iters)
170{
171 if (timeout_iters == 0U) {
172 ra8_log_error(s_tag, "graphics_power_on: timeout_iters == 0");
173 return k_ra8_err_invalid_arg;
174 }
175
176 /* HUM Ch 11.2.14 "PDCTRGD : Graphics Power Domain Control Register", p 452 */
177 /* Already powered (PDPGSF = 0)? Idempotent no-op for multi-driver init. */
178 if ((*ra8_lpm_sysc_reg8(k_ra8_lpm_pdctrgd_off) & (uint8_t)k_ra8_lpm_pdctr_pdpgsf_mask) == 0U) {
179 return k_ra8_ok;
180 }
181
182 internal_graphics_enable_moco();
183
184 ra8_err_t err = internal_graphics_confirm_ready(timeout_iters);
185 RA8_RETURN_ON_ERROR(err, s_tag, "graphics_power_on: not ready"); /* GCOVR_EXCL_BR_LINE */
186
187 internal_graphics_clear_pdde();
188
189 err = internal_graphics_confirm_powered(timeout_iters);
190 RA8_RETURN_ON_ERROR(err, s_tag, "graphics_power_on: still gated"); /* GCOVR_EXCL_BR_LINE */
191
192 ra8_log_info(s_tag, "graphics power domain on");
193 return k_ra8_ok;
