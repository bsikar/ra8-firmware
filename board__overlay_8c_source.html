212static void
213console_tab_rect(int32_t panel_x, int32_t panel_w, uint32_t idx, uint32_t count, int32_t* r)
214{
215 if (r == nullptr) {
216 return;
217 }
218 const uint32_t per_row = console_tabs_per_row(panel_w, count);
219 const uint32_t row = idx / per_row;
220 const uint32_t col = idx % per_row;
221 const int32_t pitch = panel_w / (int32_t)per_row;
222 r[0] = panel_x + ((int32_t)col * pitch);
223 r[1] = (int32_t)k_con_y + ((int32_t)row * (int32_t)k_con_tab_h);
224 r[2] = pitch - (int32_t)k_con_tab_gap;
225 r[3] = (int32_t)k_con_tab_h - (int32_t)k_con_tab_gap;
226 if (r[2] < 0) {
227 r[2] = 0;
228 }
229 if (r[3] < 0) {
230 r[3] = 0;
231 }

377static void draw_console_body(uint16_t* out,
378 uint16_t w,
379 uint16_t h,
380 int32_t panel_x,
381 int32_t body_y,
382 int32_t body_h,
383 const board_status_t* st)
384{
385 const int32_t max_rows = (body_h - (2 * (int32_t)k_con_pad)) / (int32_t)k_con_line_h;
386 int32_t rows = (int32_t)st->console_count;
387 if (rows > max_rows) {

391 /* Bottom row (i=0 from the bottom) is console[0], the newest line. */
392 const int32_t ly = body_y + body_h - (int32_t)k_con_pad - ((i + 1) * (int32_t)k_con_line_h);
393 const uint16_t col = (i == 0) ? (uint16_t)k_ovl_console_new : (uint16_t)k_ovl_console_txt;

398static void draw_console(uint16_t* out, uint16_t w, uint16_t h, int32_t x, const board_status_t* st)
399{
400 draw_console_heading(out, w, h, x, st);
401 const int32_t panel_x = x;
402 const int32_t panel_w = (int32_t)k_ovl_sidebar_w - (2 * (int32_t)k_pad_x);
403 const int32_t panel_h = (int32_t)h - (int32_t)k_con_y - (int32_t)k_con_bottom;
404 if (panel_h < (int32_t)k_con_line_h) {
405 return;
406 }
407 fill_rect(out, w, h, panel_x, (int32_t)k_con_y, panel_w, panel_h, (uint16_t)k_ovl_console_bg);
408 /* The tab bar wraps into as many rows as the channel count needs; the scrolling
409 * body fills whatever is left below it. */
410 draw_console_tabs(out, w, h, panel_x, panel_w, st);
411 uint32_t tab_count = st->console_ch_count;
412 if (tab_count > (uint32_t)k_overlay_console_tabs_max) {
413 tab_count = (uint32_t)k_overlay_console_tabs_max;
414 }
415 const int32_t tab_rows = (int32_t)console_tab_row_count(panel_w, tab_count);
416 const int32_t tab_band = tab_rows * (int32_t)k_con_tab_h;
417 const int32_t body_y = (int32_t)k_con_y + tab_band;
418 const int32_t body_h = panel_h - tab_band;
419 if (body_h < (int32_t)k_con_line_h) {
420 return;
421 }
422 /* Fit as many lines as the body height allows; show the newest at the bottom
423 * so the console scrolls upward like a terminal. console[0] is the newest. */
424 draw_console_body(out, w, h, panel_x, body_y, body_h, st);
425}

655board_overlay_btn_t board_overlay_hit_button(uint16_t x, uint16_t y, uint16_t panel_w)
656{
657 const int32_t cx = (int32_t)x;
658 const int32_t cy = (int32_t)y;
659 const int32_t side_lo = (int32_t)panel_w;
660 const int32_t side_hi = (int32_t)panel_w + (int32_t)k_ovl_sidebar_w;
661 const bool in_sidex = (cx >= side_lo) && (cx < side_hi);
662 /* SW1 / SW2 push-buttons. */
663 if ((cy >= (int32_t)k_btn_y) && (cy < ((int32_t)k_btn_y + (int32_t)k_btn_h))) {
664 const int32_t bx1 = (int32_t)panel_w + (int32_t)k_btn_x_dx;
665 if ((cx >= bx1) && (cx < (bx1 + (int32_t)k_btn_w))) {
666 return k_board_overlay_btn_sw1;
667 }
668 const int32_t bx2 = bx1 + (int32_t)k_btn_w + (int32_t)k_btn_gap;
669 if ((cx >= bx2) && (cx < (bx2 + (int32_t)k_btn_w))) {
670 return k_board_overlay_btn_sw2;
671 }
672 }
673 /* POWER row: the battery slider track and the CHG toggle share one row. */
674 if ((cy >= (int32_t)k_pwr_y) && (cy < ((int32_t)k_pwr_y + (int32_t)k_pwr_h))) {
675 const int32_t sx = (int32_t)panel_w + (int32_t)k_pwr_x_dx;
676 if ((cx >= sx) && (cx < (sx + (int32_t)k_pwr_track_w))) {
677 return k_board_overlay_btn_battery;
678 }
679 const int32_t cgx = sx + (int32_t)k_pwr_chg_off;

681 return k_board_overlay_btn_batt_chg;
682 }
683 }
684 /* Core / low-power toggle: shares the BUTTONS row, right of SW1/SW2. */
685 if ((cy >= (int32_t)k_btn_y) && (cy < ((int32_t)k_btn_y + (int32_t)k_btn_h))) {

687 if ((cx >= lx) && (cx < (lx + (int32_t)k_core_lp_w))) {
688 return k_board_overlay_btn_lowpower;
689 }
690 }

692 * like clicking the Arduino Serial Monitor's pane. */

694 return k_board_overlay_btn_console;
695 }
696 return k_board_overlay_btn_none;
697}

699bool board_overlay_battery_pct_at(uint16_t x, uint16_t panel_w, uint8_t* out_pct)
700{
701 if (out_pct == nullptr) {
702 return false;
703 }
704 const int32_t cx = (int32_t)x;
705 const int32_t sx = (int32_t)panel_w + (int32_t)k_pwr_x_dx;
706 if (cx <= sx) {
707 *out_pct = 0U;
708 } else if (cx >= (sx + (int32_t)k_pwr_track_w)) {
709 *out_pct = (uint8_t)k_pwr_soc_full;
710 } else {
711 *out_pct = (uint8_t)(((cx - sx) * (int32_t)k_pwr_soc_full) / (int32_t)k_pwr_track_w);
712 }
713 return true;
714}
715
716bool board_overlay_hit_console_tab(uint16_t x,
717 uint16_t y,
718 uint16_t panel_w,
719 uint32_t tab_count,
720 uint32_t* out_idx)
721{
722 if (out_idx == nullptr) {
723 return false;
724 }
725 if (tab_count == 0U) {
726 return false;
727 }
728 uint32_t count = tab_count;
729 if (count > (uint32_t)k_overlay_console_tabs_max) {
730 count = (uint32_t)k_overlay_console_tabs_max;
731 }
732 const int32_t cx = (int32_t)x;
733 const int32_t cy = (int32_t)y;
734 const int32_t origin_x = (int32_t)panel_w + (int32_t)k_pad_x;
735 const int32_t pan_w = (int32_t)k_ovl_sidebar_w - (2 * (int32_t)k_pad_x);
736 /* Point-in-cell test against the wrapping grid: each cell carries its own
737 * row y, so a click resolves to the right tab on any row. The four bounds are
738 * split into separate guards (no compound decision) so each edge is its own
739 * branch. */
740 for (uint32_t i = 0U; i < count; i++) {
741 int32_t rect[4] = {0, 0, 0, 0};
742 console_tab_rect(origin_x, pan_w, i, count, rect);
743 if (cx < rect[0]) {
744 continue;
745 }
746 if (cx >= (rect[0] + rect[2])) {
747 continue;
748 }
749 if (cy < rect[1]) {
750 continue;
751 }
752 if (cy >= (rect[1] + rect[3])) {
753 continue;
754 }
755 *out_idx = i;
756 return true;
757 }
758 return false;
759}
