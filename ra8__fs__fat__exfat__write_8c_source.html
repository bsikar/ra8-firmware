73/* `priv_exfat_find_bitmap()`: see header for the documented contract. */
74ra8_err_t priv_exfat_find_bitmap(const ra8_fs_mount_t* m, uint32_t* out_clus, uint32_t* out_len)
75{
76 exfat_cursor_t cur = {.cluster = m->root_cluster, .entry_in_cluster = 0U, .scanned = 0U};
77 while (cur.scanned < (uint32_t)k_exfat_scan_limit) {
78 uint8_t e[k_exfat_entry_bytes] = {};
79 ra8_err_t r = priv_exfat_next_entry(m, &cur, e);
80 if (r != k_ra8_ok) {
81 return r;
82 }
83 if (e[0] == (uint8_t)k_exfat_entry_eod) {
84 return k_ra8_err_not_found;
85 }
86 if (e[0] == (uint8_t)k_exfat_entry_bitmap) {
87 *out_clus = priv_rd32(&e[k_exfat_strm_off_clus]);
88 *out_len = priv_rd32(&e[k_exfat_strm_off_dlen]);
89 return k_ra8_ok;
90 }
91 }
92 return k_ra8_err_not_found; /* GCOVR_EXCL_LINE -- scan sentinel: reaching here requires k_exfat_scan_limit (65536) consecutive non-EOD, non-bitmap root dir entries; no host-side input can manufacture that many raw entries without a separate cluster-allocation API that does not exist */
93}

115static ra8_err_t
116priv_exfat_bitmap_scan(const ra8_fs_mount_t* m, uint32_t bmp_lba, uint32_t need, uint32_t* out_clus)
117{
118 uint32_t run = 0U;
119 uint32_t start = 0U;
120 uint32_t loaded = UINT32_MAX;
121 uint8_t sec[k_ra8_fs_bytes_per_sector] = {};
122 for (uint32_t idx = 0U; idx < m->count_of_clusters; idx++) {
123 const uint32_t lba = bmp_lba + ((idx >> k_exfat_bit_shift) / k_ra8_fs_bytes_per_sector);
124 if (lba != loaded) {
125 ra8_err_t e = priv_read_sector(m, lba, sec);
126 if (e != k_ra8_ok) {
127 return e;
128 }
129 loaded = lba;
130 }
131 const uint32_t byte = (idx >> k_exfat_bit_shift) % k_ra8_fs_bytes_per_sector;
132 const uint32_t bit = idx & k_exfat_bit_mask;
133 if ((((uint32_t)sec[byte] >> bit) & 1U) != 0U) {
134 run = 0U;
135 continue;
136 }
137 if (run == 0U) {
138 start = idx;
139 }
140 run++;
141 if (run >= need) {
142 *out_clus = start + k_cluster_first_data;
143 return k_ra8_ok;
144 }
145 }
146 return k_ra8_err_no_mem;
147}

190static ra8_err_t
191priv_exfat_bitmap_mark(const ra8_fs_mount_t* m, uint32_t bmp_lba, uint32_t clus, uint32_t count)
192{
193 uint32_t loaded = UINT32_MAX;
194 uint8_t sec[k_ra8_fs_bytes_per_sector] = {};
195 for (uint32_t k = 0U; k < count; k++) {
196 const uint32_t idx = (clus - k_cluster_first_data) + k;
197 const uint32_t lba = bmp_lba + ((idx >> k_exfat_bit_shift) / k_ra8_fs_bytes_per_sector);
198 const uint32_t byte = (idx >> k_exfat_bit_shift) % k_ra8_fs_bytes_per_sector;
199 const uint32_t bit = idx & k_exfat_bit_mask;
200 ra8_err_t e = priv_exfat_bmp_switch(m, lba, &loaded, sec);
201 if (e != k_ra8_ok) {
202 return e;
203 }
204 sec[byte] = (uint8_t)(sec[byte] | (uint8_t)(1U << bit));
205 }
206 return priv_write_sector(m, loaded, sec);
207}

230priv_exfat_write_data(const ra8_fs_mount_t* m, uint32_t clus, const uint8_t* data, uint32_t len)
231{
232 const uint32_t base = priv_cluster_to_lba(m, clus);
233 const uint32_t sectors = (len + k_ra8_fs_bytes_per_sector - 1U) / k_ra8_fs_bytes_per_sector;
234 for (uint32_t s = 0U; s < sectors; s++) {
235 uint8_t sec[k_ra8_fs_bytes_per_sector] = {};
236 const uint32_t off = s * k_ra8_fs_bytes_per_sector;
237 uint32_t n = (off < len) ? (len - off) : 0U;
238 if (n > k_ra8_fs_bytes_per_sector) {
239 n = k_ra8_fs_bytes_per_sector;
240 }
241 if (n > 0U) {
242 priv_byte_copy(sec, &data[off], n);
243 }
244 ra8_err_t e = priv_write_sector(m, base + s, sec);
245 if (e != k_ra8_ok) {
246 return e;
247 }
248 }
249 return k_ra8_ok;

335static ra8_err_t priv_exfat_find_dir_space(const ra8_fs_mount_t* m,
336 uint32_t need,
337 uint32_t* out_clus,
338 uint32_t* out_idx)
339{
340 const uint32_t per_cluster =
341 (m->sectors_per_cluster * k_ra8_fs_bytes_per_sector) / (uint32_t)k_exfat_entry_bytes;
342 uint32_t cluster = m->root_cluster;
343 uint32_t guard = 0U;
344 while (guard < (uint32_t)k_exfat_scan_limit) {
345 uint32_t run = 0U;
346 for (uint32_t i = 0U; i < per_cluster; i++) {
347 uint8_t e[k_exfat_entry_bytes] = {};
348 ra8_err_t r = priv_exfat_read_entry(m, cluster, i, e);
349 if (r != k_ra8_ok) {
350 return r;
351 }
352 if (priv_exfat_slot_free(e[0]) == 0U) {
353 run = 0U;
354 continue;
355 }
356 if (run == 0U) {
357 *out_idx = i;
358 }
359 run++;
360 if (run >= need) {
361 *out_clus = cluster;
362 return k_ra8_ok;
363 }
364 }
365 uint32_t next = 0U;
366 ra8_err_t fe = priv_fat_get(m, cluster, &next);
367 if (fe != k_ra8_ok) {
368 return fe;
369 }
370 if (priv_is_eoc(m, next) != 0U) {
371 return k_ra8_err_no_mem;
372 }
373 cluster = next;
374 guard++;
375 }
376 return k_ra8_err_no_mem;
377}

399static uint32_t priv_exfat_build_set(uint8_t* set,
400 const char* path,
401 uint32_t nlen,
402 uint32_t first_clus,
403 uint32_t len)
404{
405 const uint32_t name_entries =
406 (nlen + (uint32_t)k_exfat_name_per_entry - 1U) / (uint32_t)k_exfat_name_per_entry;
407 const uint32_t sec_count = 1U + name_entries;
408 const uint32_t total = (1U + sec_count) * (uint32_t)k_exfat_entry_bytes;
409 for (uint32_t i = 0U; i < total; i++) {
410 set[i] = 0U;
411 }
412 set[0] = (uint8_t)k_exfat_entry_file;
413 set[k_exfat_off_file_secnt] = (uint8_t)sec_count;
414 priv_wr16(&set[k_exfat_off_file_attr], (uint16_t)k_exfat_attr_archive);
415 uint8_t* strm = &set[k_exfat_entry_bytes];
416 strm[0] = (uint8_t)k_exfat_entry_stream;
417 strm[k_exfat_strm_off_flags] = (uint8_t)k_exfat_secflag_alloc;
418 strm[k_exfat_strm_off_nlen] = (uint8_t)nlen;
419 priv_wr16(&strm[k_exfat_off_strm_hash], priv_exfat_name_hash(path, nlen));
420 priv_wr32(&strm[k_exfat_off_strm_valid], len);
421 priv_wr32(&strm[k_exfat_strm_off_clus], first_clus);
422 priv_wr32(&strm[k_exfat_strm_off_dlen], len);
423 for (uint32_t n = 0U; n < name_entries; n++) {
424 uint8_t* ne = &set[(size_t)(2U + n) * (size_t)k_exfat_entry_bytes];
425 ne[0] = (uint8_t)k_exfat_entry_name;
426 for (uint32_t c = 0U; c < (uint32_t)k_exfat_name_per_entry; c++) {
427 const uint32_t pos = (n * (uint32_t)k_exfat_name_per_entry) + c;
428 if (pos < nlen) {
429 ne[k_exfat_name_off + (c * 2U)] = (uint8_t)path[pos];
430 }
431 }
432 }
433 priv_wr16(&set[k_exfat_off_file_csum], priv_exfat_set_checksum(set, total));
434 return total;
435}

550/* `priv_exfat_create()`: see header for the documented contract. */
551ra8_err_t priv_exfat_create(ra8_fs_mount_t* m, const char* path, const uint8_t* data, uint32_t len)
552{
553 /* Strip leading slashes so the stored name matches what the matchers
554 * search for (#93); otherwise the file is created but cannot be reopened. */
555 while (*path == '/') {
556 path++;
557 }
558 const uint32_t nlen = priv_strlen(path);
559 if (nlen == 0U) {
560 return k_ra8_err_invalid_arg;
561 }
562 if (nlen > (uint32_t)k_exfat_name_cap) {
563 return k_ra8_err_invalid_arg;
564 }
565 if (len == 0U) {
566 return k_ra8_err_invalid_arg;
567 }
568 const uint32_t cbytes = m->sectors_per_cluster * k_ra8_fs_bytes_per_sector;
569 const uint32_t nclus = (len + cbytes - 1U) / cbytes;
570 uint32_t start = 0U;
571 ra8_err_t e = priv_exfat_alloc_write(m, data, len, nclus, &start);
572 if (e != k_ra8_ok) {
573 return e;
574 }
575 return priv_exfat_link(m, path, nlen, start, len);
576}
