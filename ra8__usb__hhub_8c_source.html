439RA8_INTERNAL
440static ra8_err_t internal_step_advance(void)
441{
442 switch (s_state.step) {
443 case k_ra8_hhub_step_idle:
444 return internal_do_idle();
445 case k_ra8_hhub_step_bus_reset:
446 return internal_do_bus_reset();
447 case k_ra8_hhub_step_set_address:
448 return internal_do_set_address();
449 case k_ra8_hhub_step_get_dev_desc:
450 return internal_do_get_dev_desc();
451 case k_ra8_hhub_step_get_cfg_desc:
452 return internal_do_get_cfg_desc();
453 case k_ra8_hhub_step_set_config:
454 return internal_do_set_config();
455 case k_ra8_hhub_step_get_hub_desc:
456 return internal_do_get_hub_desc();
457 default:
458 /* Already done; idempotent. */
459 return k_ra8_ok;
460 }

468ra8_err_t ra8_usb_hhub_init(ra8_usb_speed_t speed)
469{
470 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
471 return k_ra8_err_invalid_arg;
472 }
473 const ra8_err_t usb_err = ra8_usb_host_init(speed);
474 if (usb_err != k_ra8_ok) {
475 ra8_log_error_val(s_tag, "ra8_usb_host_init failed", (uint32_t)usb_err);
476 return k_ra8_err_hw_init_failed;
477 }
478
479 s_state.speed = speed;
480 s_state.step = k_ra8_hhub_step_idle;
481 s_state.attached = false;
482 s_state.attach_cb = nullptr;
483 s_state.attach_ctx = nullptr;
484 s_state.device = (ra8_usb_hhub_device_t){};
485 s_state.initialized = true;
486
487 ra8_log_info_val(s_tag, "host-HUB ready", (uint32_t)speed);
488 return k_ra8_ok;
489}
490
491ra8_err_t ra8_usb_hhub_close(void)
492{
493 if (!s_state.initialized) {
494 return k_ra8_err_invalid_state;
495 }
496 (void)ra8_usb_host_set_uact(s_state.speed, false);
497 const ra8_err_t err = ra8_usb_host_deinit(s_state.speed);
498 s_state.initialized = false;
499 s_state.attached = false;
500 s_state.attach_cb = nullptr;
501 s_state.attach_ctx = nullptr;
502 s_state.step = k_ra8_hhub_step_idle;
503 return err;
504}
