89/* Bring up clocks, MSTP, the delay timer and the console, or halt trying. */
90static void wifi_hal_setup_or_halt(void)
91{
92 if (ra8_cgc_init() != k_ra8_ok) {
93 wifi_hal_panic_halt();
94 }
95 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &s_cpuclk_hz) != k_ra8_ok) {
96 wifi_hal_panic_halt();
97 }
98 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &s_pclka_hz) != k_ra8_ok) {
99 wifi_hal_panic_halt();
100 }
101 if (ra8_mstp_init() != k_ra8_ok) {
102 wifi_hal_panic_halt();
103 }
104 if (ra8_time_init(s_cpuclk_hz) != k_ra8_ok) {
105 wifi_hal_panic_halt();
106 }
107 if (ra8_board_uart_console_init((uint32_t)k_wifi_hal_uart_baud) != k_ra8_ok) {
108 wifi_hal_panic_halt();
109 }
110}

112/* Fill an ra8_wifi configuration: the C6 backend plus the NetX DHCP provider. */
113static ra8_err_t wifi_hal_make_cfg(ra8_wifi_cfg_t* out)
114{
115 ra8_wifi_c6link_cfg_t bcfg = {};
116 bcfg.link = &s_link;
117 bcfg.arena = s_arena;
118 bcfg.arena_bytes = (uint32_t)sizeof(s_arena);
119 bcfg.rx_cb = nx_ether_driver_c6_rx;
120 const ra8_err_t bound = ra8_esp_hosted_c6link_bind(&bcfg.transport);
121 if (bound != k_ra8_ok) {
122 return bound;
123 }
124 const ra8_err_t wired = ra8_wifi_c6link_setup(&s_c6, &bcfg, out);
125 if (wired != k_ra8_ok) {
126 return wired;
127 }
128 out->ip_bind = wifi_hal_ip_bind;
129 out->ip_ctx = &s_link;
130 return k_ra8_ok;
131}

147/* The whole network journey: bind the hardware config, run the shared core, and
148 * narrate what it recorded. The orchestration lives in wifi_hal_join_run
149 * (src/wifi_hal_core.c) so the host test drives the identical logic; this
150 * function only builds the config and prints from the result. */

153static bool wifi_hal_run(void)
154{
155 ra8_wifi_cfg_t cfg = {};
156 const ra8_err_t made = wifi_hal_make_cfg(&cfg);
157 if (made != k_ra8_ok) {
158 wifi_hal_puts("wifi_hal: FAIL transport setup err=");
159 wifi_hal_puts(ra8_err_to_str(made));
160 wifi_hal_puts("\r\n");
161 return false;
162 }
163
164 const wifi_hal_run_cfg_t rcfg = {
165 .wifi_cfg = &cfg,
166 .wifi = &s_wifi,
167 .ssid = s_ssid,
168 .psk = s_psk,
169 .poll_budget = (uint32_t)k_wifi_hal_assoc_polls,
170 };
171 wifi_hal_result_t res = {};
172 const bool joined = wifi_hal_join_run(&rcfg, &res);
173 return wifi_hal_report(&res, joined);
174}

176/* Narrate each stage from the result record and judge the run. `joined` is the
177 aggregate verdict join_run returned; it is carried in rather than re-derived
178 so a disagreement between it and the per-stage flags cannot pass silently. */
179static bool wifi_hal_report(const wifi_hal_result_t* res, bool joined)
180{
181 wifi_hal_puts("wifi_hal: init=");
182 wifi_hal_puts(ra8_err_to_str(res->init_err));
183 wifi_hal_puts("\r\n");
184 if (!res->init_ok) {
185 return false;
186 }
187
188 /* cppcheck-suppress knownConditionTrueFalse ; build-dependent credential. */
189 if (s_ssid[0] == '\0') {
190 wifi_hal_puts("wifi_hal: FAIL no Wi-Fi credentials compiled in! Set RA8_C6_WIFI_SSID / "
191 "RA8_C6_WIFI_PSK in env, or in coprocessor/esp32c6/wifi.env, or fetch via "
192 "OpenBao (python3 scripts/secrets/openbao_client.py)\r\n");
193 return false;
194 }
195 if (!res->associated) {
196 wifi_hal_puts("wifi_hal: FAIL association did not complete connect=");
197 wifi_hal_puts(ra8_err_to_str(res->connect_err));
198 wifi_hal_puts("\r\n");
199 return false;
200 }
201
202 wifi_hal_puts("wifi_hal: associated ssid=");
203 wifi_hal_puts(s_ssid);
204 wifi_hal_puts(" mac=");
205 wifi_hal_put_mac(&res->mac);
206 wifi_hal_puts(" chan=");
207 wifi_hal_put_u32((uint32_t)res->ap.channel);
208 wifi_hal_puts("\r\n");
209 if (!res->ip_bound) {
210 wifi_hal_puts("wifi_hal: FAIL dhcp did not bind err=");
211 wifi_hal_puts(ra8_err_to_str(res->ip_err));
212 wifi_hal_puts("\r\n");
213 return false;
214 }
215 wifi_hal_print_lease(&res->lease);
216 if (!joined) {
217 wifi_hal_puts("wifi_hal: FAIL join_run failed with every stage reported ok\r\n");
218 }
219 return joined;
220}

222/* Print a heartbeat forever so the verdict stays visible on the console. */
223static void wifi_hal_heartbeat(bool passed)
224{
225 while (true) {
226 wifi_hal_puts(passed ? "wifi_hal: alive PASS\r\n" : "wifi_hal: alive FAIL\r\n");
227 tx_thread_sleep((ULONG)k_wifi_hal_heartbeat_ms);
228 }
229}

231/* Worker thread: bring up, join, get an IP, judge, then heartbeat. */
232static void wifi_hal_worker_entry(ULONG thread_input)
233{
234 (void)thread_input;
235
236 wifi_hal_puts("wifi_hal: port_init=");
237 wifi_hal_puts(ra8_err_to_str(s_init_err));
238 wifi_hal_puts("\r\n");
239 if (s_init_err != k_ra8_ok) {
240 wifi_hal_puts("wifi_hal: FAIL port init failed -- no transaction attempted\r\n");
241 wifi_hal_heartbeat(false);
242 return;
243 }
244
245 ra8_delay_ms((uint32_t)k_wifi_hal_boot_wait_ms);
246
247 const bool ok = wifi_hal_run();
248 wifi_hal_puts(ok ? k_wifi_hal_pass_line : k_wifi_hal_fail_line);
249 wifi_hal_heartbeat(ok);
250}
