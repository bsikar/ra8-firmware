4"""Gate: ``infra/fleet.yml`` describes a fleet that could actually be built.
5
6The declaration is the single registry of what machines this project runs on
7and how much of each one CI may use, so an error in it is an error in the
8estate. This checks three things a green Ansible run would not:
9
101. **The declaration is internally sound.** Every rule in
11 :func:`fleet_model.validate` -- classes and plays that exist, an address any
12 machine could reach the host at, capacity that fits the declared budget,
13 per-instance floors, a parseable quiet-hours window, and an instance count
14 that is either the sizing formula's or comes with a written reason. A number
15 nobody can re-derive is folklore, and a host addressed by an ssh alias is
16 reachable only from whichever laptop defines it (#526).
17
182. **Nothing tunes a host twice.** A committed ``host_vars`` file may not
19 re-declare a variable the declaration owns. Extra-vars beat ``host_vars``,
20 so a duplicate would not change behaviour -- it would leave a number in the
21 tree that looks authoritative, that somebody will edit, and that will have
22 no effect.
23
243. **The derived variables land somewhere real.** Every ``fleet_capacity_*``
25 and ``dev_slice_*`` name the mapping emits must exist in that role's
26 defaults. A mapping keyed on a spelling no role reads is the same defect as
27 a checker rule keyed on a string no macro produces: it matches nothing and
28 reports success forever.
29
304. **The names both halves must agree on do agree.** The model predicts the dev
31 slice's unit name (``fleet.py`` passes it to the capacity script) and the
32 role creates it. Two spellings would give the host a quiet-hours window that
33 freezes a slice nothing ever made -- a schedule that stands nothing down.
34
355. **No command the tooling builds needs an ssh alias.** Rule 1 checks the
36 INPUT; this checks the derivation, by walking the real ssh argv and the real
37 inventory line for every host and failing on any destination or ProxyJump
38 hop that is a bare label. A future ``-J <fleet name>`` would pass every
39 input rule and still only work on a machine that happened to define that
40 name -- which is the whole of #526, one layer down.
41
42``--selftest`` runs first in the gate and asserts each rule fires on a
43deliberately broken declaration and stays quiet on a legal one. Without it,
44"0 problems" is indistinguishable from "checked nothing".
45"""

83def _check_derived_vars() -> list[str]:
84 """Every variable the mapping emits exists in the role that consumes it.
85
86 Returns:
87 One message per name the role would never read.
88 """
89 data = fm.load()
90 emitted: set[str] = set()
91 for name, host in data["hosts"].items():
92 emitted |= set(fm.role_vars(name, host))
93 problems = []
94 for prefix, role in DERIVED_ROLES.items():
95 declared = set(_role_defaults(role))
96 problems += [
97 f"fleet.py emits '{key}', which is in no {role} default -- the role "
98 "would never read it, so whatever it configures would silently not happen"
99 for key in sorted({k for k in emitted if k.startswith(prefix)} - declared)
100 ]
101 return problems

107 ``fleet_model`` predicts the dev slice's unit name so ``fleet.py`` can pass

126def _is_literal(destination: str) -> bool:
127 """Whether an ssh destination is an address rather than a config alias.
128
129 Args:
130 destination: ``[user@]address`` as it would appear on an ssh command
131 line.
132
133 Returns:
134 True when it carries a dot or a colon, i.e. an IPv4/IPv6 literal or a
135 qualified name; False for a bare label, which only resolves through
136 somebody's ``~/.ssh/config``.
137 """
138 address = destination.rpartition("@")[2]
139 return "." in address or ":" in address

142def _inventory_destinations(entry: str) -> list[str]:
143 """Every host address one generated inventory line hands to Ansible.
144
145 Args:
146 entry: One line of the generated inventory.
147
148 Returns:
149 The ``ansible_host`` value plus every ``ProxyJump`` hop, empty for a
150 ``connection=local`` host, which Ansible never dials.
151 """
152 out = []
153 for field in ("ansible_host=", "-o ProxyJump="):
154 _, found, tail = entry.partition(field)
155 if not found:
156 continue
157 value = tail.split("'")[0].split()[0]
158 out += value.split(",")
159 return out

162def _check_derived_reach(data: dict[str, Any]) -> list[str]:
163 """No ssh command or inventory line the model builds names an alias.
164
165 Args:
166 data: The parsed declaration. The selftest hands it one whose
167 derivation is broken, so a detector that stopped matching cannot
168 report the real fleet clean forever.
169
170 Returns:
171 One message per derived destination that is a bare label.
172 """
173 problems = []
174 for name in data["hosts"]:
175 argv = fr.ssh_target(data, name)
176 derived = {
177 "the ssh command this tooling builds": [argv[-1], *fr.jump_chain(data, name)],
178 "the generated Ansible inventory": _inventory_destinations(
179 fm.inventory_entry(data, name)
180 ),
181 }
182 problems += [
183 f"{name}: {what} dials '{token}', a bare label rather than an address. It "
184 "would resolve only on a machine whose ~/.ssh/config happened to define it, "
185 "which is exactly the fault #526 removed -- one layer further down."
186 for what, tokens in derived.items()
187 for token in tokens
188 if not _is_literal(token)
189 ]
190 return problems

325def _lend(data: dict[str, Any], **override: int) -> None:
326 """Give the selftest's host a dev slice, with one field made wrong.
327
328 The base block is legal on the fixture host -- 2 runners x 8 GB of a 16 GB
329 budget leaves nothing, so the memory field is what has to give: the fixture
330 lends 0 GB is not legal either, hence the budget bump. Each caller then
331 breaks exactly one field, so a rule that stopped firing is attributable.
332
333 Args:
334 data: The declaration being damaged.
335 override: The one field to set to an illegal value.
336 """
337 data["hosts"]["nas"]["budget"]["memory_gb"] = 20
338 data["hosts"]["nas"]["budget"]["swap_gb"] = 2
339 data["hosts"]["nas"]["sizing_note"] = "fixture: budget raised to leave room to lend"
340 slice_: dict[str, int] = {"cpu_weight": 10, "memory_gb": 4, "swap_gb": 2, "max_jobs": 4}
341 slice_.update(override)
342 data["hosts"]["nas"]["dev_slice"] = slice_

373def _check_jump_resolves(host_vars_dir: Path) -> list[str]:
374 """A declared hop must reach the ssh command line as an ADDRESS.
375
376 The mutation table proves a bad hop is rejected; this proves a good one is
377 honoured, and honoured as a literal. ``-J bench`` would satisfy every input
378 rule and still only work on a machine that defined that alias -- the same
379 defect the addresses themselves had.
380
381 Args:
382 host_vars_dir: Empty fixture directory for the validator.
383
384 Returns:
385 One message per way the hop failed to reach the command line.
386 """
387 data = _jumped_declaration()
388 problems = [f" a legal ProxyJump was rejected: {p}" for p in fm.validate(data, host_vars_dir)]
389 argv = fr.ssh_target(data, "nas")
390 if "-J" not in argv:
391 problems.append(" a declared connect.jump produced no -J on the ssh command line")
392 elif argv[argv.index("-J") + 1] != "pi@10.0.0.9":
393 hop = argv[argv.index("-J") + 1]
394 problems.append(f" the ProxyJump hop is '{hop}', not the hop host's address")
395 if "ProxyJump=pi@10.0.0.9" not in fm.inventory_entry(data, "nas"):
396 problems.append(" the generated inventory does not hand Ansible the ProxyJump hop")
397 if "ProxyJump bench" not in fr.render_ssh_config(data):
398 problems.append(" the generated ssh config does not carry the hop")
399 return problems

402def _selftest() -> int:
403 """Assert every rule fires on a broken fleet and none fires on a legal one.
404
405 Returns:
406 0 when the checker demonstrably still has teeth, 1 otherwise.
407 """
408 failures = []
409 with tempfile.TemporaryDirectory() as tmp:
410 empty = Path(tmp)
411 if fm.validate(_good_declaration(), host_vars_dir=empty):
412 failures.append(" a legal declaration was rejected")
413 # A validator that rejected EVERY dev slice would pass every mutation
414 # below while making the feature unusable, so the legal shape is
415 # asserted too -- the same reason the legal declaration above is.
416 legal_lend = _good_declaration()
417 _lend(legal_lend)
418 if fm.validate(legal_lend, host_vars_dir=empty):
419 failures.append(" a legal dev_slice was rejected")
420 failures += _check_jump_resolves(empty)
421 if _check_derived_reach(_jumped_declaration()):
422 failures.append(" a fleet reachable only by address was reported unreachable")
423 # Both directions for the derivation check itself: an alias in the
424 # declaration must come out the far end as an alias on a command line,
425 # or the check is decoration.
426 aliased = _jumped_declaration()
427 aliased["hosts"]["bench"]["connect"]["address"] = "bench"
428 if not _check_derived_reach(aliased):
429 failures.append(" an ssh alias survived into a derived ssh command unreported")
430 for rule, damage in _mutations().items():
431 broken = deepcopy(_good_declaration())
432 damage(broken)
433 if not fm.validate(broken, host_vars_dir=empty):
434 failures.append(f" rule not enforced: {rule}")
435 good = _good_declaration()
436 (empty / "nas.yml").write_text("ci_runner_docker_cpus: '9'\n", encoding="utf-8")
437 if not fm.validate(good, host_vars_dir=empty):
438 failures.append(" a host_vars file re-declaring a fleet-owned knob was accepted")
439 if failures:
440 print("check_fleet_declaration selftest FAILED:", file=sys.stderr)
441 print("\n".join(failures), file=sys.stderr)
442 return 1
443 print(f"selftest OK: {len(_mutations())} rules fire, a legal declaration passes")
444 return 0

447def main(argv: list[str] | None = None) -> int:
448 """Entry point.
449
450 Args:
451 argv: Command line, defaulting to ``sys.argv[1:]``.
452
453 Returns:
454 0 when the declaration is sound, 1 otherwise.
455 """
456 parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
457 parser.add_argument("--selftest", action="store_true", help="prove the rules still fire")
458 args = parser.parse_args(argv)
459 if args.selftest:
460 return _selftest()
461 try:
462 data = fm.load()
463 except fm.FleetError as exc:
464 print(f"check_fleet_declaration: {exc}", file=sys.stderr)
465 return 1
466 problems = (
467 fm.validate(data)
468 + _check_derived_vars()
469 + _check_shared_constants()
470 + _check_derived_reach(data)
471 )
472 if problems:
473 print(f"infra/fleet.yml: {len(problems)} problem(s):", file=sys.stderr)
474 for problem in problems:
475 print(f" {problem}", file=sys.stderr)
476 return 1
477 runners = sum(int((h.get("runners") or {}).get("instances", 0)) for h in data["hosts"].values())
478 print(f"infra/fleet.yml OK: {len(data['hosts'])} host(s), {runners} runner instance(s)")
479 return 0
