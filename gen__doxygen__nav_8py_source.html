4"""Generate the Doxygen navigation trees, mirroring the on-disk layout.
5
6Doxygen renders markdown files (docs/*.md and every example README.md) as
7"pages", never as entries in the Files/directory tree -- so the sidebar's
8directory view is code-only, and the human-readable docs + per-app READMEs
9can only be reached through the "pages" tab. This script builds that pages
10tab to MIRROR THE ON-DISK DIRECTORY TREE instead of an invented taxonomy, so
11navigation reads the same way everywhere (browse by directory) and can never
12drift from the repo:
13
14 1. "Documentation" -- the docs/ subtree. Every docs/*.md is a leaf and
15 every docs/<subdir> (adr, SOUP, qualification, reference, ...) is a
16 sub-node, recursively, exactly as they sit on disk. No doc-to-section
17 map: a new page or a new subdir just appears; a removed one just leaves.
18
19 2. "Examples" -- the examples/ subtree. Every app appears under its real
20 hardware-validation tier directory (hw_validated/hil, hw_validated/manual,
21 hw_pending, _unsupported, ra8p1_foundation, ...), each carrying the
22 one-line description from its own README. Moving an app between tiers on
23 disk moves it in the nav with no edit here.
24
25The only hand-maintained data left is a set of PRETTY-TITLE maps (tier and
26docs-subdir display names); every one degrades gracefully -- an unmapped
27directory falls back to a prettified name and still renders, never breaks and
28never lists a removed item.
29
30The generated .dox files are written under docs/generated/ (gitignored) and
31consumed by the main Doxyfile. build_docs.sh runs this before doxygen, so the
32navigation can never drift from the tree.
33"""
34

46# --- Doxygen id manglers -----------------------------------------------------
47# These reproduce the pinned Doxygen 1.16.1's deterministic naming so the
48# generated @subpage / link targets resolve without editing any source file. If
49# Doxygen ever changed the scheme, the warnings gate would flag the unresolved
50# references.
51
52
53def doxy_page_id(relpath: str) -> str:
54 """Doxygen page id for a markdown file.
55
56 Matches the Doxyfile's CASE_SENSE_NAMES=YES scheme: the "md_" prefix, the
57 path with '/' -> '_2' and '_' -> '__', and the original case preserved
58 (e.g. docs/ACRONYMS.md -> md_docs_2ACRONYMS). The '.md' extension is
59 dropped. Under CASE_SENSE_NAMES=NO doxygen would instead escape every
60 uppercase letter, so this mangler is coupled to that Doxyfile setting.
61 """
62 stem = str(pathlib.PurePosixPath(relpath).with_suffix(""))
63 out = ["md_"]
64 for ch in stem:
65 if ch == "/":
66 out.append("_2")
67 elif ch == "_":
68 out.append("__")
69 else:
70 out.append(ch) # case preserved
71 return "".join(out)

74def doxy_dir_id(relpath: str) -> str:
75 """Doxygen directory-page id: ``dir_`` + md5 of the directory path.
76
77 The path hashed is the STRIP_FROM_PATH-relative one WITH a trailing slash;
78 both details are load-bearing, since doxygen hashes exactly that string and
79 dropping the slash yields a valid-looking id that resolves to nothing.
80
81 Doxygen renders a directory's README.md as that directory page's detailed
82 description -- so this is where an example app's README actually shows,
83 alongside its source-file list (the empty 'README.md File Reference' page
84 is a dead end). Because the Doxyfile sets STRIP_FROM_PATH=., the path fed
85 to the hash is the plain repo-relative one, so the id is machine-independent
86 and matches the same page the Files directory tree links to.
87 """
88 key = relpath.rstrip("/") + "/"
89 # Not security-sensitive: this reproduces doxygen's own directory-page
90 # file-naming hash so links resolve. usedforsecurity=False documents that
91 # and satisfies the lint.
92 digest = hashlib.md5(key.encode("utf-8"), usedforsecurity=False).hexdigest()
93 return "dir_" + digest

146def prettify(name: str) -> str:
147 """Fallback display title for a directory with no entry in the title maps.
148
149 This is what makes the title maps optional rather than authoritative: a
150 tier or docs subdirectory added on disk renders as Title Case immediately,
151 so forgetting to add a pretty name degrades the label and never drops the
152 node out of the navigation.
153 """
154 return name.replace("_", " ").strip().title()

157def clean_desc(text: str) -> str:
158 """Reduce a README paragraph to one safe, plain-text line."""
159 text = text.replace("`", "").replace("**", "").replace("*", "")
160 # collapse markdown links [txt](url) -> txt
161 out = []
162 i = 0
163 while i < len(text):
164 if text[i] == "[":
165 close = text.find("]", i)
166 paren = text.find("(", close) if close != -1 else -1
167 if close != -1 and paren == close + 1:
168 end = text.find(")", paren)
169 if end != -1:
170 out.append(text[i + 1 : close])
171 i = end + 1
172 continue
173 out.append(text[i])
174 i += 1
175 text = "".join(out)
176 text = " ".join(text.split())
177 limit = 200
178 if len(text) > limit:
179 cut = text[:limit].rsplit(" ", 1)[0]
180 text = cut + " ..."
181 # HTML-escape, then neutralize Doxygen-active characters.
182 text = html.escape(text, quote=False)
183 return text.replace("@", "&#64;").replace("\\", "&#92;")

186def read_app_desc(app_dir: pathlib.Path) -> str:
187 """Pull the one-line blurb for an example app out of its own README.
188
189 Takes the first prose paragraph AFTER the H1, which is where these READMEs
190 put their summary. The scan stops at a heading or a table row so an app
191 whose README opens with a status table straight after the title yields an
192 empty description rather than a row of pipes rendered as prose.
193
194 A missing README returns "" rather than raising: an app without one still
195 belongs in the navigation, just without a blurb.
196 """
197 readme = app_dir / "README.md"
198 if not readme.is_file():
199 return ""
200 lines = readme.read_text(encoding="utf-8", errors="replace").splitlines()
201 # skip to after the first H1
202 idx = 0
203 while idx < len(lines) and not lines[idx].startswith("# "):
204 idx += 1
205 idx += 1
206 # skip blanks
207 while idx < len(lines) and not lines[idx].strip():
208 idx += 1
209 para = []
210 while idx < len(lines) and lines[idx].strip():
211 stripped = lines[idx].strip()
212 if stripped.startswith(("#", "|")):
213 break
214 para.append(stripped)
215 idx += 1
216 return clean_desc(" ".join(para))

219# --- example tree ------------------------------------------------------------
220class TierNode:
221 """One directory level of the examples/ tree, holding its apps and children.
222
223 The tree is built from the app directories found on disk, so a node exists
224 only because some app lives at or below it -- there is no declared tier
225 list to fall out of step with the repository.
226 """
227
228 def __init__(self, path: str) -> None:
229 """Create an empty node for the tier at ``path`` relative to examples/.
230
231 The root is spelled "" rather than "." so that child paths concatenate
232 cleanly and the root's page id can be special-cased.
233 """
234 self.path = path # relative to examples/, "" for root
235 self.apps: list[tuple[str, str, str]] = [] # (name, file_id, desc)
236 self.children: dict[str, TierNode] = {}
237
238 def title(self) -> str:
239 """Display title for this tier, falling back to a prettified leaf name.
240
241 Lookup is on the node's FULL path, so two tiers sharing a leaf name at
242 different depths (``hw_validated/manual`` and ``hw_pending/manual``)
243 can be titled independently.
244 """
245 if self.path in TIER_TITLES:
246 return TIER_TITLES[self.path]
247 return prettify(self.path.split("/")[-1])
248
249 def page_id(self) -> str:
250 """Doxygen ``@page`` id for this tier, unique and identifier-safe.
251
252 Every character that is neither alphanumeric nor an underscore is
253 replaced, so a tier directory containing a dot or a dash cannot emit an
254 id doxygen would silently truncate or reject.
255 """
256 if not self.path:
257 return "ra8_examples"
258 safe = self.path.replace("/", "_")
259 safe = "".join(c if (c.isalnum() or c == "_") else "_" for c in safe)
260 return "ra8_ex_" + safe

263def build_example_tree() -> tuple[TierNode, int]:
264 """Discover every example app and assemble the tier tree it implies.
265
266 An app is defined as "a directory containing main.c", which is what makes
267 the navigation self-maintaining: moving an app between tiers on disk moves
268 it in the sidebar with no edit here, and a deleted app simply stops being
269 found. Intermediate tier nodes are created on demand as each path is
270 walked, so only tiers that actually contain apps appear.
271
272 Returns the root node (whose own ``apps`` list holds any app sitting
273 directly under examples/) and the total app count, which the caller prints
274 in the page brief rather than recomputing.
275 """
276 root = TierNode("")
277 count = 0
278 mains = sorted(EXAMPLES_DIR.rglob("main.c"))
279 for main_c in mains:
280 app_dir = main_c.parent
281 tier_rel = app_dir.parent.relative_to(EXAMPLES_DIR).as_posix()
282 if tier_rel == ".":
283 tier_rel = ""
284 node = root
285 acc = ""
286 for seg in [s for s in tier_rel.split("/") if s]:
287 acc = f"{acc}/{seg}" if acc else seg
288 node = node.children.setdefault(acc, TierNode(acc))
289 app_rel = app_dir.relative_to(ROOT).as_posix()
290 # Link to the app's DIRECTORY page: doxygen renders the app README as
291 # that page's description and lists its source files there -- the same
292 # page the Files tree reaches. (The README file page itself is empty.)
293 node.apps.append((app_dir.name, doxy_dir_id(app_rel), read_app_desc(app_dir)))
294 count += 1
295 return root, count

298def emit_subpage_list(out: list[str], entries: list[tuple[str, str]]) -> None:
299 """Emit @subpage links as an HTML bulleted list (one per line).
300
301 entries is a list of (page_id, suffix); suffix is appended after the link
302 (e.g. " (12 apps)") or "" for none.
303 """
304 if not entries:
305 return
306 out.append(" * <ul>")
307 for pid, suffix in entries:
308 out.append(f" * <li>@subpage {pid}{suffix}</li>")
309 out.append(" * </ul>")

312def emit_tier_page(node: TierNode, out: list[str]) -> None:
313 """Append this tier's ``@page`` block to ``out``, then recurse into children.
314
315 Appends in place rather than returning, so one list accumulates the whole
316 tree in traversal order: a parent's page is emitted before its descendants,
317 which is the order doxygen needs to nest the subpage links correctly.
318
319 Apps are rendered as a table and child tiers as a bulleted subpage list, so
320 a tier holding both reads as sections rather than one mixed list.
321 """
322 brief = TIER_BRIEFS.get(node.path, "")
323 out.append("/**")
324 out.append(f" * @page {node.page_id()} {node.title()}")
325 if brief:
326 out.append(f" * @brief {brief}")
327 out.append(" *")
328 child_nodes = [node.children[k] for k in sorted(node.children)]
329 emit_subpage_list(
330 out,
331 [(c.page_id(), f" ({_count_apps(c)} apps)") for c in child_nodes],
332 )
333 if child_nodes and node.apps:
334 out.append(" *")
335 if node.apps:
336 out.append(f" * @par {len(node.apps)} app(s) in this tier:")
337 out.append(" * <table>")
338 out.append(" * <tr><th>App</th><th>Description</th></tr>")
339 for name, fid, desc in sorted(node.apps):
340 link = f'<a href="{fid}.html"><code>{html.escape(name)}</code></a>'
341 out.append(f" * <tr><td>{link}</td><td>{desc}</td></tr>")
342 out.append(" * </table>")
343 out.append(" */")
344 out.append("")
345 for child in child_nodes:
346 emit_tier_page(child, out)
347
348
349def _count_apps(node: TierNode) -> int:
350 return len(node.apps) + sum(_count_apps(c) for c in node.children.values())

387def _docs_subid(rel_under_docs: str) -> str:
388 """Stable @page id for a docs/ subdirectory, keyed by its full sub-path.
389
390 Keying on the full path (not just the leaf name) means two subdirs that
391 share a leaf name at different depths cannot collide. The distinct
392 "docsub" prefix keeps it clear of the "ra8_docs" root and of any page id.
393 """
394 safe = "".join(c if (c.isalnum() or c == "_") else "_" for c in rel_under_docs)
395 return "ra8_docsub_" + safe

398def _walk_docs(dir_path: pathlib.Path, rel_under_docs: str) -> tuple[list[str], str | None, str]:
399 """Emit the @page blocks for a docs/ subtree, mirroring it on disk.
400
401 Returns (page_blocks, subid, title). subid is None when the subtree holds
402 no narrative markdown at all, so the parent omits it -- a directory of only
403 images / JSON never becomes an empty nav node. Sub-directories are listed
404 before files, matching doxygen's own Files-tree convention.
405 """
406 md_files = sorted(
407 (p for p in dir_path.glob("*.md") if p.name.lower() != "readme.md"),
408 key=lambda p: p.name.lower(),
409 )
410 child_dirs = sorted(
411 (p for p in dir_path.iterdir() if p.is_dir() and p.name not in DOCS_SKIP_DIRS),
412 key=lambda p: p.name.lower(),
413 )
414
415 blocks: list[str] = []
416 child_nodes: list[tuple[str, str]] = []
417 for cd in child_dirs:
418 crel = f"{rel_under_docs}/{cd.name}" if rel_under_docs else cd.name
419 cblocks, csubid, ctitle = _walk_docs(cd, crel)
420 if csubid is not None:
421 child_nodes.append((csubid, ctitle))
422 blocks.extend(cblocks)
423
424 leaf_ids = [doxy_page_id(p.relative_to(ROOT).as_posix()) for p in md_files]
425 if not leaf_ids and not child_nodes:
426 return [], None, ""
427
428 subid = _docs_subid(rel_under_docs)
429 title = SUBDIR_TITLES.get(dir_path.name, prettify(dir_path.name))
430 page = ["/**", f" * @page {subid} {title}", " *"]
431 entries = [(cid, "") for cid, _t in child_nodes] + [(pid, "") for pid in leaf_ids]
432 emit_subpage_list(page, entries)
433 page.append(" */")
434 page.append("")
435 # This directory's page first, then its descendants' pages.
436 return page + blocks, subid, title

439def _handwritten_doc_page_ids() -> list[str]:
440 """Discover hand-written ``@page`` ids in docs/*.dox.
441
442 Without this the hand-written pages would float at the top level of the
443 sidebar rather than nesting under Documentation, since doxygen parents a
444 page only where something ``@subpage``s it.
445
446 Only the docs/ root .dox files are scanned (never docs/generated/, which
447 this script owns). The ids are read straight from the files, so a page that
448 is renamed or deleted is picked up or dropped automatically -- nothing is
449 hardcoded, nothing dangles.
450 """
451 ids: list[str] = []
452 for dox in sorted(DOCS_DIR.glob("*.dox"), key=lambda p: p.name.lower()):
453 for line in dox.read_text(encoding="utf-8", errors="replace").splitlines():
454 stripped = line.lstrip(" *")
455 for tag in ("@page ", "\\page "):
456 if stripped.startswith(tag):
457 pid = stripped[len(tag) :].split()[0]
458 if pid:
459 ids.append(pid)
460 return ids

463def gen_docs() -> str:
464 """Render the complete "Documentation" navigation tree as one .dox file body.
465
466 Both the generated per-file pages and the ids harvested from hand-written
467 .dox files are listed as children here, so the two kinds of page appear in
468 one tree rather than the hand-written ones floating at the top level.
469
470 Returns the file text, newline-terminated.
471 """
472 # Top-level docs/*.md become leaves of the "Documentation" root; each
473 # docs/<subdir> that carries markdown becomes a sub-node, recursively --
474 # the tree is exactly the on-disk docs/ layout, no classification map.
475 top_md = sorted(
476 (p for p in DOCS_DIR.glob("*.md") if p.name.lower() != "readme.md"),
477 key=lambda p: p.name.lower(),
478 )
479 top_leaf_ids = [doxy_page_id(p.relative_to(ROOT).as_posix()) for p in top_md]
480 top_leaf_ids += _handwritten_doc_page_ids()
481
482 subdirs = sorted(
483 (p for p in DOCS_DIR.iterdir() if p.is_dir() and p.name not in DOCS_SKIP_DIRS),
484 key=lambda p: p.name.lower(),
485 )
486 child_nodes: list[tuple[str, str]] = []
487 sub_blocks: list[str] = []
488 for sd in subdirs:
489 blocks, subid, _title = _walk_docs(sd, sd.name)
490 if subid is not None:
491 child_nodes.append((subid, _title))
492 sub_blocks.extend(blocks)
493
494 out: list[str] = [
495 "// GENERATED by scripts/gen/gen_doxygen_nav.py -- do not edit.",
496 "",
497 "/**",
498 " * @page ra8_docs Documentation",
499 " * @brief The hand-written documentation, browsable exactly as it sits",
500 " * under docs/ on disk. (The API reference lives under Topics /",
501 " * Data Structures; the source tree lives under Files.)",
502 " *",
503 ]
504 # Sub-directories first, then the top-level pages -- same order as the
505 # Files directory tree.
506 entries = [(subid, "") for subid, _t in child_nodes]
507 entries += [(pid, "") for pid in top_leaf_ids]
508 emit_subpage_list(out, entries)
509 out.append(" */")
510 out.append("")
511 out.extend(sub_blocks)
512
513 return "\n".join(out) + "\n"

516# --- directory descriptions (Files tab) --------------------------------------
517# A plain "@dir src" in a static .dox is ambiguous: many example apps carry a
518# nested src/ (examples/*/src), and doxygen applies the description to the
519# alphabetically-first match instead of the repo-root src/. The other top-level
520# directories have no same-named nested twin, so their @dir blocks stay in the
521# static docs/doxygen_dirs.dox. Here we emit an ABSOLUTE-path @dir for any such
522# colliding top-level directory, which matches exactly one directory.
523COLLIDING_TOP_DIRS: dict[str, str] = {
524 "src": "Shared application internals used by the drivers -- no boot code, "
525 "no main(). Includes the Ring 5 secure-side substrate (src/secure_app, "
526 "e.g. the key vault and secure-OTA commit).",
527}

530def gen_dirs() -> str:
531 """Emit ``@dir`` blocks for top-level directories with a same-named twin.
532
533 A bare ``@dir src`` is ambiguous because many example apps carry their own
534 nested src/, and doxygen silently binds the description to the
535 alphabetically-first match rather than the repo-root one. Only the
536 colliding names need this treatment; the rest keep their static blocks in
537 docs/doxygen_dirs.dox.
538
539 Returns the file text, newline-terminated.
540 """
541 out = ["// GENERATED by scripts/gen/gen_doxygen_nav.py -- do not edit.", ""]
542 # Disambiguate with the "<repo-dir>/src" suffix rather than an absolute
543 # path: it is unique (no example app lives under <repo-dir>/), and it avoids
544 # a doxygen bug that truncates a \dir path at the first dot-prefixed
545 # component (e.g. a ".../.claude/worktrees/.../src" checkout).
546 for name, brief in COLLIDING_TOP_DIRS.items():
547 out.append("/**")
548 out.append(f" * @dir {ROOT.name}/{name}")
549 out.append(f" * @brief {brief}")
550 out.append(" */")
551 out.append("")
552 return "\n".join(out) + "\n"

564def python_module_symbols() -> list[str]:
565 """Collect the module STEMS of every first-party Python file, deduplicated.
566
567 Doxygen parses a .py file as a namespace named after its stem, and those
568 namespaces' classes then appear in the C firmware's Data Structures list.
569 Feeding these to EXCLUDE_SYMBOLS drops the symbols while leaving the files
570 themselves browsable under Files.
571
572 Stems, not paths: EXCLUDE_SYMBOLS matches symbol names, so two like-named
573 modules in different directories collapse to one entry -- which is correct
574 here, since both would produce the same namespace name.
575 """
576 mods: set[str] = set()
577 for top in PY_TOOL_DIRS:
578 base = ROOT / top
579 if not base.is_dir():
580 continue
581 for py in base.rglob("*.py"):
582 if any(part in CONFIG_SKIP for part in py.parts):
583 continue
584 mods.add(py.stem)
585 return sorted(mods)
