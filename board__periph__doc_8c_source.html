96static void doc_apply(uint32_t dodir)
97{
98 const uint32_t mode = (uint32_t)(s_doc.docr & (uint8_t)k_doc_oms_mask);
99 const uint32_t mask = doc_width_mask();
100 const uint32_t in = dodir & mask;
101 const uint32_t cur = s_doc.dodsr0 & mask;
102 if (mode == (uint32_t)k_doc_mode_add) {
103 const uint32_t sum = cur + in;
104 s_doc.dopcf = ((sum & ~mask) != 0U) ? (uint8_t)k_doc_dopcf : s_doc.dopcf;
105 s_doc.dodsr0 = sum & mask;
106 } else if (mode == (uint32_t)k_doc_mode_subtract) {
107 s_doc.dopcf = (in > cur) ? (uint8_t)k_doc_dopcf : s_doc.dopcf; /* borrow */
108 s_doc.dodsr0 = (cur - in) & mask;
109 } else {
110 /* Compare: DCSEL selects the relation that latches DOPCF. DCSEL=1 matches
111 * on equality; everything else (including 0) latches on mismatch. */
112 const uint32_t dcsel =
113 (uint32_t)((s_doc.docr & (uint8_t)k_doc_dcsel_mask) >> (uint32_t)k_doc_dcsel_shift);
114 const bool equal = (in == cur);
115 const bool hit = (dcsel == 1U) ? equal : !equal;
116 if (hit) {
117 s_doc.dopcf = (uint8_t)k_doc_dopcf;
118 }
119 }
120 s_doc.ops++;
121}

155static void doc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
156{
157 (void)uc;
158 (void)size;
159 const uint64_t off = addr - (uint64_t)k_doc_base;
160 if (off == (uint64_t)k_doc_off_docr) {
161 s_doc.docr = (uint8_t)value;
162 } else if (off == (uint64_t)k_doc_off_doscr) {
163 if (((uint8_t)value & (uint8_t)k_doc_dopcf) != 0U) {
164 s_doc.dopcf = 0U; /* DOPCFCL clears the flag. */
165 }
166 } else if (off == (uint64_t)k_doc_off_dodir) {
167 doc_apply((uint32_t)value);
168 } else if (off == (uint64_t)k_doc_off_dodsr0) {
169 s_doc.dodsr0 = (uint32_t)value; /* Seed operand A. */
170 } else if (off == (uint64_t)k_doc_off_dodsr1) {
171 s_doc.dodsr1 = (uint32_t)value;
172 }
173}
