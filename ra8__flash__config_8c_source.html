170static uint32_t internal_popcount32(uint32_t x)
171{
172 /* Avoid relying on __builtin_popcount so the off-target build
173 * does not pick up a different implementation than the firmware
174 * target. NASA Rule 2 -- bounded loop. */
175 uint32_t count = 0U;
176 for (uint32_t i = 0U; i < 32U; ++i) {
177 if ((x & (uint32_t)(1UL << i)) != 0U) {
178 count++;
179 }
180 }
181 return count;
182}

189ra8_err_t ra8_flash_set_startup_area(ra8_flash_startup_t target, bool temporary)
190{
191 if (target > k_ra8_flash_startup_btflg) {
192 return k_ra8_err_invalid_arg;
193 }
194 ra8_err_t err = ra8_flash_enter_pe_mode();
195 if (err != k_ra8_ok) {
196 return err;
197 }
198
199 if (temporary) {
200 /* HUM Ch 59 "MSUACR : Start-Up Area Control Register" p 3593 */
201 const uint16_t swap_bit = (target == k_ra8_flash_startup_alternate)
202 ? k_ra8_msuacr_swap_alternate
203 : k_ra8_msuacr_swap_default;
204 *ra8_mram_reg16(k_ra8_mram_off_msuacr) = (uint16_t)(k_ra8_msuacr_key | swap_bit);
205 } else {
206 /* Permanent: configuration-set write to BTFLG. */
207 /* HUM Ch 7 "Option-Setting Memory" p 278 */
208 uint16_t cfg_words[k_ra8_mram_config_set_word_count];
209 for (uint32_t i = 0U; i < k_ra8_mram_config_set_word_count; ++i) {
210 cfg_words[i] = k_ra8_flash_cfg_word_all_ones;
211 }
212 /* BTFLG occupies bit 15 of word index 3 (FSP MRAM_PRV_CONFIG_SET_BTFLG_OFFSET).
213 * 0 selects alternate, 1 selects default (HUM Ch 7 p 278). */
214 uint16_t btflg_bit = k_ra8_flash_btflg_alternate;
215 if (target == k_ra8_flash_startup_default) {
216 btflg_bit = k_ra8_flash_btflg_default;
217 }
218 /* HUM Ch 7 "OFS SAS region" p 278 */
219 cfg_words[3] = (uint16_t)(btflg_bit | k_ra8_flash_btflg_word_keep);
220 err = ra8_flash_config_set_write(k_ra8_msaddr_config_set_startup, cfg_words);
221 }
222
223 ra8_err_t exit_err = ra8_flash_exit_pe_mode();
224 if (err != k_ra8_ok) {
225 return err;
226 }
227 return exit_err;
228}

246ra8_err_t ra8_flash_config_set_write(uint32_t target_addr, const uint16_t* words)
247{
248 RA8_CHECK_NULL_PTR(words, s_flash_tag, "words must not be nullptr");
249 /* One MACI byte-stream shape -- ``<opener>, N, 8 halfwords, 0xD0`` -- serves
250 * two target regions, and the opener opcode is chosen per region below:
251 * - OFS configuration area (HUM Ch 7 "Option-Setting Memory" p 278) at
252 * 0x02C9F000: the Configuration Set command (HUM Ch 59.7.4.8 p 3594).
253 * - Extra-MRAM option-setting / OTP area (HUM Ch 59.7.4.5 Table 59.15
254 * p 3592) at 0x02E07600: the Program command (HUM Ch 59.7.4.5 "Program
255 * Command" Fig 59.13 p 3591). Config-Set is NOT valid for the data area
256 * -- it raises
257 * MSTATR.CFGSETERR and leaves the target blank, so a later read of the
258 * un-programmed cells bus-faults on the blank-MRAM ECC error.
259 * Accept both ranges; reject everything else. */
260 const uint32_t ofs_end = (uint32_t)k_ra8_flash_ofs_start + (uint32_t)k_ra8_flash_ofs_size;
261 const uint32_t extra_end = (uint32_t)k_ra8_flash_extra_start + (uint32_t)k_ra8_flash_extra_size;
262 const bool in_ofs =
263 (bool)((target_addr >= (uint32_t)k_ra8_flash_ofs_start) && (target_addr < ofs_end));
264 const bool in_extra =
265 (bool)((target_addr >= (uint32_t)k_ra8_flash_extra_start) && (target_addr < extra_end));
266 if (!in_ofs && !in_extra) {
267 return k_ra8_err_invalid_arg;
268 }
269
270 /* HUM Ch 59.5.19 "MSADDR : MACI Command Start Address Register" p 3573 */
271 *ra8_mram_reg32(k_ra8_mram_off_msaddr) = target_addr;
272 /* Opener: Program (0xE8) for the extra-MRAM data area, else Configuration
273 * Set (0x40) for the OFS config area. HUM Ch 59.7.4.5 Fig 59.13 p 3591 /
274 * HUM Ch 59.7.4.8 p 3594. */
275 const uint8_t opener =
276 in_extra ? (uint8_t)k_ra8_maci_cmd_program : (uint8_t)k_ra8_maci_cmd_config_set;
277 ra8_flash_internal_maci_cmd8(opener);
278 ra8_flash_internal_maci_cmd8(k_ra8_maci_cmd_word_count_n);
279
280 /* N = k_ra8_mram_config_set_word_count halfwords of payload.
281 * HUM Ch 59.7.4.5 Fig 59.13 p 3592 / HUM Ch 59.7.4.8 p 3595. */
282 for (uint32_t i = 0U; i < k_ra8_mram_config_set_word_count; ++i) {
283 ra8_flash_internal_maci_cmd16(words[i]);
284 }
285 /* Trailer 0xD0 starts command processing. HUM Ch 59.7.4.5 Fig 59.13 p 3592. */
286 ra8_flash_internal_maci_cmd8(k_ra8_maci_cmd_final);
287
288 ra8_err_t err = ra8_flash_internal_wait_mrdy(k_ra8_flash_maci_spin_limit);
289 if (err != k_ra8_ok) {
290 return err;
291 }
292
293 /* HUM Ch 59 "MSTATR : Extra MRAM Status Register" p 3578 */
294 const uint32_t s = *ra8_mram_reg32(k_ra8_mram_off_mstatr);
295 if ((s & k_ra8_mstatr_mask_any_err) != 0U) {
296 return k_ra8_err_hw_error;
297 }
298 return k_ra8_ok;
299}

367static uint32_t internal_arc_nsec_count(ra8_flash_arc_id_t id)
368{
369 /* HUM Ch 7.2.21 "ARCCS" p 296 */
370 const uint16_t arccs = *(volatile const uint16_t*)k_ra8_flash_ofs_arccs_addr;
371 const uint8_t arcns = (uint8_t)(arccs & (uint16_t)k_ra8_arc_arccs_mask);
372 /* HUM Ch 7.2.23 "ARC_NSEC" p 297 */
373 const volatile uint32_t* base = (const volatile uint32_t*)k_ra8_flash_ofs_arc_nsec_addr;
374 uint32_t words_per = (arcns == (uint8_t)k_ra8_arc_arcns_single) ? 16U : 2U;
375 if (words_per > k_ra8_mram_arc_max_words) {
376 words_per = k_ra8_mram_arc_max_words;
377 }
378 uint32_t base_idx = words_per * 3U;
379 if (id == k_ra8_flash_arc_nsec_0) {
380 base_idx = 0U;
381 } else if (id == k_ra8_flash_arc_nsec_1) {
382 base_idx = words_per;
383 } else if (id == k_ra8_flash_arc_nsec_2) {
384 base_idx = words_per * 2U;
385 } else {
386 /* falls through to 3*words_per default. */
387 }
388 uint32_t count = 0U;
389 for (uint32_t w = 0U; w < words_per; ++w) {
390 count += internal_popcount32(base[base_idx + w]);
391 }
392 return count;
393}

395/* internal_arc_read_locked -- see header for full description. */
396static ra8_err_t internal_arc_read_locked(ra8_flash_arc_id_t id, uint32_t* out_count)
397{
398 uint8_t mcntselr = internal_arc_to_mcntselr(id);
399 uint32_t count = 0U;
400
401 if (id == k_ra8_flash_arc_oembl) {
402 /* OEMBL needs the MACI read so RSIP-E50D can intercept. */
403 ra8_err_t err = internal_arc_cmd(mcntselr, k_ra8_maci_cmd_read_counter);
404 if (err != k_ra8_ok) {
405 return err;
406 }
407 /* HUM Ch 59 "MCNTDTR0 : MRAM Counter Data 0" p 3576 */
408 const uint32_t lo = *ra8_mram_reg32(k_ra8_mram_off_mcntdtr0);
409 /* HUM Ch 59 "MCNTDTR1 : MRAM Counter Data 1" p 3577 */
410 const uint32_t hi = *ra8_mram_reg32(k_ra8_mram_off_mcntdtr1);
411 count = internal_popcount32(lo) + internal_popcount32(hi);
412 } else if (id == k_ra8_flash_arc_sec) {
413 /* HUM Ch 7.2.22 "ARC_SEC" p 296: ARC_SEC is a 64-bit unary counter, i.e.
414 * 2 x uint32_t words. FSP r_mram.c L1196 derives the same count via
415 * BSP_FEATURE_FLASH_ARC_SEC_MAX_COUNT (64) >> 5. The previous loop walked
416 * 8 words, which over-counted into adjacent OFS state. */
417 const volatile uint32_t* p = (const volatile uint32_t*)k_ra8_flash_ofs_arc_sec_addr;
418 const uint32_t sec_words = k_ra8_arc_sec_max_bits >> k_flash_bits_to_words_shift;
419 for (uint32_t w = 0U; w < sec_words; ++w) {
420 count += internal_popcount32(p[w]);
421 }
422 } else {
423 count = internal_arc_nsec_count(id);
424 }
425
426 *out_count = count;
427 (void)mcntselr; /* used in OEMBL branch only */
428 return k_ra8_ok;
429}

431ra8_err_t ra8_flash_arc_increment(ra8_flash_arc_id_t counter)
432{
433 if (counter >= k_ra8_flash_arc_count) {
434 return k_ra8_err_invalid_arg;
435 }
436 RA8_VALIDATE_INIT(s_flash_rt.initialized, s_flash_tag, "arc_inc before init");
437
438 ra8_err_t err = ra8_flash_enter_pe_mode();
439 if (err != k_ra8_ok) {
440 return err;
441 }
442
443 /* NASA Power-of-10 Rule 1 forbids goto, so the read-bound-check-increment
444 * sequence is expressed as a short-circuit chain: each step runs only if
445 * the previous one returned k_ra8_ok, and the unconditional exit-PE step
446 * replaces the prior goto-out label.
447 * HUM Ch 59.4.4 "MACI Increment Counter" p 3550. */
448 uint32_t cur = 0U;
449 err = internal_arc_read_locked(counter, &cur);
450 if (err == k_ra8_ok) {
451 const uint32_t max = internal_arc_max_count(counter);
452 if (cur + 1U > max) {
453 err = k_ra8_err_out_of_range;
454 } else {
455 err = internal_arc_cmd(internal_arc_to_mcntselr(counter), k_ra8_maci_cmd_increment_counter);
456 }
457 }
458
459 const ra8_err_t exit_err = ra8_flash_exit_pe_mode();
460 if (err == k_ra8_ok) {
461 err = exit_err;
462 }
463 return err;
464}

466ra8_err_t ra8_flash_arc_read(ra8_flash_arc_id_t counter, uint32_t* out_count)
467{
468 RA8_CHECK_NULL_PTR(out_count, s_flash_tag, "out_count must not be nullptr");
469 if (counter >= k_ra8_flash_arc_count) {
470 return k_ra8_err_invalid_arg;
471 }
472 RA8_VALIDATE_INIT(s_flash_rt.initialized, s_flash_tag, "arc_read before init");
473
474 if (counter == k_ra8_flash_arc_oembl) {
475 ra8_err_t err = ra8_flash_enter_pe_mode();
476 if (err != k_ra8_ok) {
477 return err;
478 }
479 ra8_err_t r = internal_arc_read_locked(counter, out_count);
480 ra8_err_t exit_err = ra8_flash_exit_pe_mode();
481 if (r != k_ra8_ok) {
482 return r;
483 }
484 return exit_err;
485 }
486 return internal_arc_read_locked(counter, out_count);
487}

680static void priv_pack_config_words(const uint8_t* src,
681 uint32_t len,
682 uint32_t done,
683 uint16_t words[k_ra8_mram_config_set_word_count])
684{
685 for (uint32_t i = 0U; i < k_ra8_mram_config_set_word_count; ++i) {
686 const uint32_t base = done + (i * 2U);
687 const uint8_t lo = (base < len) ? src[base] : k_flash_blank_byte;
688 const uint8_t hi = (base + 1U < len) ? src[base + 1U] : k_flash_blank_byte;
689 words[i] = (uint16_t)((uint16_t)lo | ((uint16_t)hi << 8U));
690 }
691}
692
693ra8_err_t ra8_flash_extra_mram_write(uint32_t mram_addr, const uint8_t* src, uint32_t len)
694{
695 RA8_CHECK_NULL_PTR(src, s_flash_tag, "src must not be nullptr");
696 if (len == 0U || len > k_ra8_mram_write_size_bytes) {
697 return k_ra8_err_invalid_arg;
698 }
699 if (mram_addr < (uint32_t)k_ra8_flash_extra_start) {
700 return k_ra8_err_invalid_arg;
701 }
702 const uint32_t end_excl = (uint32_t)((uint64_t)mram_addr + (uint64_t)len);
703 /* OTP-misuse guard (#397): cap the general-purpose write path at
704 * k_ra8_flash_extra_locked_start. The permanent / irreversible option-setting
705 * structures (PBPS, POFSPS, REVOKE, Zeroization-HUK enable, anti-rollback)
706 * begin there; programming any of them can brick the part or destroy the
707 * wrapped HUK, so they require the deliberate, separately-named
708 * ra8_flash_config_set_write. This bound is stricter than the full window end
709 * (k_ra8_flash_extra_start + k_ra8_flash_extra_size), so the single check also
710 * rejects an overrun past the window. HUM Ch 59.7.4.5 Table 59.15 p 3592. */
711 if (end_excl > (uint32_t)k_ra8_flash_extra_locked_start) {
712 return k_ra8_err_invalid_arg;
713 }
714 const uint32_t page_mask = k_ra8_mram_write_size_bytes - 1U;
715 if ((mram_addr & ~page_mask) != ((end_excl - 1U) & ~page_mask)) {
716 return k_ra8_err_invalid_arg;
717 }
718
719 ra8_err_t err = ra8_flash_enter_pe_mode();
720 if (err != k_ra8_ok) {
721 return err;
722 }
723
724 /* The extra-MRAM data area is programmed with the MACI Program command (HUM
725 * Ch 59.7.4.5 "Program Command" Fig 59.13 p 3591) -- NOT the Configuration
726 * Set command, which is only valid for the OFS config area. One Program
727 * command carries ``k_ra8_mram_config_set_word_count`` halfwords (16 bytes),
728 * so a write spanning more than that issues back-to-back Program commands,
729 * each retargeting MSADDR ``k_ra8_mram_config_set_bytes`` further on, until the
730 * whole ``len`` is programmed. ``ra8_flash_config_set_write`` picks the Program
731 * opener for this extra-MRAM range and sets MSADDR per call, so no separate
732 * MSADDR write is needed here. Tail bytes pad to 0xFFFF. */
733 for (uint32_t done = 0U; done < len; done += (uint32_t)k_ra8_mram_config_set_bytes) {
734 uint16_t cfg_words[k_ra8_mram_config_set_word_count] = {};
735 priv_pack_config_words(src, len, done, cfg_words);
736 err = ra8_flash_config_set_write(mram_addr + done, cfg_words);
737 if (err != k_ra8_ok) {
738 break;
739 }
740 }
741
742 ra8_err_t exit_err = ra8_flash_exit_pe_mode();
743 if (err == k_ra8_ok) {
744 err = exit_err;
745 }
746 return err;
747}

749ra8_err_t ra8_flash_extra_mram_erase(uint32_t mram_addr)
750{
751 if ((mram_addr & (k_ra8_mram_block_size_bytes - 1U)) != 0U) {
752 return k_ra8_err_invalid_arg;
753 }
754 static const uint8_t s_ones[k_ra8_mram_block_size_bytes] = {
755 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU,
756 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU,
757 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU, 0xFFU,
758 };
