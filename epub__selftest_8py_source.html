27# A fixed-layout / image-only EPUB3 is a CBZ in EPUB clothing: each spine
28# document is one full-page <img> with no flowable text. This selftest proves

49def _fl_str(bb: BlobBuilder, off: int) -> str:
50 """Resolve a string-pool offset in builder `bb` back to text."""
51 raw = bb.sp.buf
52 return bytes(raw[off : raw.index(b"\x00", off)]).decode("utf-8")

55def _fl_find_image(bb: BlobBuilder, root_idx: int) -> tuple[str, dict[str, str]] | None:
56 """DFS a chapter DOM for the first <img>/<image>; return (tag, attrs) or None.
57
58 Handles both the bare <img src> page and the <svg><image xlink:href> wrapper
59 page: the tag name distinguishes them, and either src or xlink:href carries
60 the raster reference.
61 """
62 stack = [root_idx]
63 while stack:
64 idx = stack.pop()
65 if idx == NIL:
66 continue
67 node = bb.nodes[idx]
68 if node["kind"] == NODE_ELEMENT:
69 tag = _fl_str(bb, node["name_off"])
70 if tag in ("img", "image"):
71 attrs = {}
72 for k in range(node["attr_count"]):
73 name_off, val_off = bb.attrs[node["first_attr"] + k]
74 attrs[_fl_str(bb, name_off)] = _fl_str(bb, val_off)
75 return tag, attrs
76 stack.append(node["next_sibling"])
77 stack.append(node["first_child"])
78 return None

81def _fl_zip_dir(directory: Path) -> io.BytesIO:
82 """Zip a directory tree in memory (sorted for determinism); return BytesIO."""
83 buf = io.BytesIO()
84 with ZipFile(buf, "w") as zf:
85 for path in sorted(directory.rglob("*")):
86 if path.is_file():
87 zf.writestr(str(path.relative_to(directory)), path.read_bytes())
88 buf.seek(0)
89 return buf

104 want_pages = ["text/page1.xhtml", "text/page2.xhtml", "text/page3.xhtml"]

121def _check_page_elements(bb: BlobBuilder) -> None:
122 """Each page's image element, and the img-src vs manifest-id gap.
123
124 The gap is asserted in both directions on purpose: the verbatim `src` must
125 NOT be a manifest id (cross-directory), and normalising it against the
126 chapter's own directory must recover the id exactly. That pair is the
127 bridge a book-backed image loader has to implement, so a change that made
128 either half true by accident would be caught here.
129 """
130 image_ids = [_fl_str(bb, img[0]) for img in bb.images]
131 want_tags = ["img", "img", "image"] # p1 bare img, p2 bare img, p3 svg<image>
132 for i, (_title_off, href_off, root_idx) in enumerate(bb.chapters):
133 root_node = bb.nodes[root_idx]
134 _fl_require(_fl_str(bb, root_node["name_off"]) == "body", f"chapter root is body [{i}]")
135 hit = _fl_find_image(bb, root_idx)
136 _fl_require(hit is not None, f"chapter has an image element [{i}]")
137 tag, attrs = hit
138 _fl_require(tag == want_tags[i], f"page {i} image tag is <{want_tags[i]}> (got <{tag}>)")
139 src = attrs.get("src") or attrs.get("xlink:href")
140 _fl_require(src is not None, f"image element carries a src/xlink:href [{i}]")
141 # The gap: verbatim src != manifest image id, but normalising the src
142 # against the chapter's own directory recovers the id exactly.
143 chapter_dir = posixpath.dirname(_fl_str(bb, href_off))
144 normalised = posixpath.normpath(posixpath.join(chapter_dir, src))
145 _fl_require(src not in image_ids, f"verbatim src is NOT a manifest id [{i}] (the gap)")
146 _fl_require(normalised in image_ids, f"normalised src recovers the manifest id [{i}]")

149def selftest() -> int:
150 """Compile the committed fixed-layout fixture and check the #196 contract.
151
152 Zips tests/fixtures/rabook_fixed_layout/ in memory, compiles it, then:
153 * asserts the one-image-per-page shape (one spine chapter per page, each a
154 <body> holding a single <img>/<image>, one manifest raster per page);
155 * asserts the EPUB3 properties="cover-image" cover resolved to page 1;
156 * builds an equivalent CBZ from the SAME page rasters, compiles it with
157 cbz_compile, and asserts the 4bpp image payloads are byte-identical
158 EPUB-vs-CBZ (so an image that the render path can find draws the same
159 pixels either way);
160 * documents the img-src gap: the verbatim <img src> does NOT equal the
161 manifest image id (cross-directory), but normalising it against the
162 chapter path does -- the exact bridge a book-backed image loader needs.
163 Exits non-zero on the first failed check.
164 """
165 root = Path(__file__).resolve().parents[2]
166 fixture = root.joinpath(*FIXED_LAYOUT_FIXTURE)
167 _fl_require(fixture.is_dir(), f"fixture missing at {fixture}")
168
169 blob, meta, bb = compile_epub(_fl_zip_dir(fixture))
170 container = wrap_container(blob)
171
172 # Round-trip the container so the whole on-disk path is exercised, not just
173 # the flat blob (mirrors cbz_compile's selftest).
174 from cbz_compile import compile_cbz, unwrap_container # noqa: PLC0415 # avoid import cycle
175
176 _fl_require(unwrap_container(container) == blob, "RBKC container round-trip")
177
178 _check_page_tables(bb, meta)
179 _check_page_elements(bb)
180
181 # Equivalent CBZ from the same page rasters -> identical 4bpp payloads.
182 page_names = [posixpath.basename(_fl_str(bb, img[0])) for img in bb.images]
183 raw_pngs = {name: (fixture / "OEBPS" / "images" / name).read_bytes() for name in page_names}
184 _blob_cbz, _meta_cbz, bb_cbz = compile_cbz(
185 _fl_build_equivalent_cbz(raw_pngs), title="Fixed Layout Spike (#196)"
186 )
187 cbz_payload = {posixpath.basename(_fl_str(bb_cbz, img[0])): img for img in bb_cbz.images}
188 for img in bb.images:
189 name = posixpath.basename(_fl_str(bb, img[0]))
190 _fl_require(name in cbz_payload, f"CBZ has a matching page for {name}")
191 c_img = cbz_payload[name]
192 _fl_require((img[1], img[2]) == (c_img[1], c_img[2]), f"same dims EPUB/CBZ [{name}]")
193 _fl_require(img[3] == c_img[3] == IMG_GRAY4, f"both transcode to 4bpp [{name}]")
194 _fl_require(img[4] == c_img[4], f"4bpp payload byte-identical EPUB/CBZ [{name}]")
195
196 sys.stdout.write(
197 "epub_compile.py selftest: PASS -- fixed-layout shape, EPUB3 cover, "
198 "CBZ payload parity, and the img-src normalisation gap all verified.\n"
199 )
200 return 0
