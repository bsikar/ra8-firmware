120static bool internal_rv_turn(ra8_widget_reflow_view_t* v, int32_t px, int32_t mid)
121{
122 if (px < mid) {
123 if (v->page == (uint16_t)k_ra8_widget_rv_first) {
124 return false;
125 }
126 v->page = (uint16_t)(v->page - (uint16_t)k_ra8_widget_rv_one);
127 return true;
128 }
129 if ((uint16_t)(v->page + (uint16_t)k_ra8_widget_rv_one) >= v->page_count) {
130 return false;
131 }
132 v->page = (uint16_t)(v->page + (uint16_t)k_ra8_widget_rv_one);
133 return true;
134}

187RA8_INTERNAL
188static bool
189internal_rv_link(ra8_widget_t* w, ra8_widget_reflow_view_t* v, const ra8_widget_event_t* ev)
190{
191 if ((v->ops == nullptr) || (v->ops->follow_link == nullptr)) {
192 return false;
193 }
194 uint16_t dest = v->page;
195 if (!v->ops->follow_link(v->ops->user, v->page, ev->x, ev->y, &dest)) {
196 return false;
197 }
198 v->page = internal_rv_clamp(dest, v->page_count);
199 (void)ra8_widget_invalidate(w, k_ra8_widget_refresh_quality);
200 return true;
201}

221RA8_INTERNAL
222static bool internal_rv_on_input(ra8_widget_t* w, const ra8_widget_event_t* ev)
223{
224 ra8_widget_reflow_view_t* v = (ra8_widget_reflow_view_t*)w->ctx;
225 if (v == nullptr) {
226 return false;
227 }
228 if (ev->kind != k_ra8_widget_ev_touch) {
229 return false;
230 }
231 if (internal_rv_link(w, v, ev)) {
232 return true;
233 }
234 const int32_t mid = w->rect.x + (w->rect.w / 2);
235 if (internal_rv_turn(v, ev->x, mid)) {
236 (void)ra8_widget_invalidate(w, k_ra8_widget_refresh_quality);
237 }
238 return true;
239}
