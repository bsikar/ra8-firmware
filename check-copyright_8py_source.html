28def needs_header(path: pathlib.Path) -> bool:
29 """Whether this path is required to carry a licence header.
30
31 Keyed on extension plus the exact basename ``CMakeLists.txt``, which has
32 no distinguishing suffix and would otherwise escape the rule.
33 """
34 if path.suffix.lower() not in EXTENSIONS and path.name != "CMakeLists.txt":
35 return False
36 if path.is_dir():
37 return False
38 if not path.exists():
39 return False
40 return not any(part in EXCLUDED_PARTS for part in path.parts)

43def check(path: pathlib.Path) -> bool:
44 """Whether the file carries both the SPDX identifier and the copyright line.
45
46 Reads the WHOLE file rather than a fixed-size head: some file-level
47 Doxygen blocks in this tree are long enough that the copyright line sits
48 past any prefix worth guessing, and a head-only read reported those as
49 missing.
50
51 Returns True when both are present.
52 """
53 try:
54 # Read the whole file: some file-level Doxygen blocks are long
55 # enough that the copyright line sits past any fixed prefix.
56 head = path.read_text(encoding="utf-8", errors="replace")
57 except OSError:
58 return False
59
60 for needle in COPYRIGHT_SIGNATURES:
61 if needle not in head:
62 print(f"[MISSING] {path}: header does not contain '{needle}'", file=sys.stderr)
63 return False
64 return True

70def main() -> int:
71 """Check the files named on argv for their licence headers.
72
73 Takes its targets from the caller (the pre-commit hook passes the staged

75 2 rather than a clean tree.
76
77 Returns 0 when every named file is compliant, 1 on a missing header, 2 on
78 the usage error.
79 """
80 if len(sys.argv) < MIN_ARGS:
81 print("usage: check-copyright.py FILE [FILE ...]", file=sys.stderr)
82 return 2

84 failures = 0
