1#!/usr/bin/env python3
2# SPDX-License-Identifier: MIT
3# Copyright (c) 2026 Brighton Sikarskie
4"""Gate: no first-party file may introduce a permanent anti-recovery brick.
5
6Owner policy (2026-07-23): this is a personal, open-source project. It must
7NEVER contain code that permanently disables device recovery on the RA8D2.
8Nothing sets any of this today -- the point of this gate is to keep it that
9way forever, by failing CI the moment a change would.
10
11What is FORBIDDEN (a *violation* -- the gate fails)
12---------------------------------------------------
13Only *actions* that move an RA8D2 into a state its own recovery scripts cannot
14undo. Three shapes, all as executed commands / programmatic writes, never as
15prose:
16
17 * Setting the ``ce`` Security Flag ("Disable Initialize Command") -- via
18 ``rfp-cli`` (``-set-security-flag ce``), an option-byte / OSIS / security
19 register write, or any ``set ce`` / ``ce = true`` / ``--disable-initialize``
20 programmatic path. That flag permanently disables the boot-firmware
21 Initialize (erase-chip) recovery that ``scripts/hil/dlm_reset.sh`` relies on.
22 * Transitioning the DLM to a terminal/unrecoverable lock -- ``LCK_BOOT`` (or
23 any lock that renders the SWD-DP permanently unresponsive), e.g.
24 ``rfp-cli -dlm LCK_BOOT`` or a programmatic ``dlm_program(k_dlm_lck_boot)``.
25 * Any ``rfp-cli -dlm <state>`` / security-flag write that moves the device to
26 a state the recovery scripts (``scripts/hil/dlm_reset.sh`` /
27 ``recover.sh``) cannot undo.
28
29What is ALLOWED (must NOT be flagged -- these are pro-recovery / defensive)
30--------------------------------------------------------------------------
31 * The recovery scripts themselves (``scripts/hil/dlm_reset.sh``,
32 ``dlm_reset_local.sh``, ``recover.sh``, ``reflash.sh``). They READ / CHECK
33 that ``ce`` is unset and WARN if recovery failed -- the exact opposite of
34 the forbidden action. They are excluded by path AND the patterns below do
35 not fire on their content (the ``--selftest`` proves both).
36 * Reading / checking DLM state (``rfp-cli -rfo``), and *recoverable* DLM
37 transitions (``OEM_PL0`` / ``OEM_PL1`` debug-lock, which the Initialize
38 command or an authenticated regression can undo).
39 * Any comment, warning string, or negated phrasing that merely *discusses*
40 the danger: "verify ``ce`` is unset", "``ce`` may be set", "must NOT have
41 been set", "do NOT run ``rfp-cli -dlm LCK_BOOT``".
42
43How the ACTION / not-an-action distinction is drawn
44---------------------------------------------------
45Comments are blanked before matching, so a comment that shows the forbidden
46command as a warning cannot trip the gate. The patterns split into two tiers:
47
48 * HARD -- unambiguous executed commands / writes (``rfp-cli -dlm LCK_BOOT``,
49 ``set ce``, ``ce = true``, ``-set-security-flag ce``, ``--disable-initialize``).
50 These fire on any non-comment occurrence; a comment nearby does not excuse
51 an action.
52 * SOFT -- occurrences of a terminal-state NAME near a set-verb (spaced
53 "disable initialize", "... regress ... LCK_BOOT"). A NAME is a violation
54 only when nothing on the same code line marks it as a check / negation
55 (``DEFENSIVE_RE``). The defensive cues are word-boundaried, so a snake_case
56 or camelCase identifier that merely contains "never" / "verify" does NOT
57 earn the exemption -- a function that really does the transition is still a
58 violation regardless of what it is named.
59
60Documentation files (``.md`` / ``.txt``) are prose, not executed, so every
61match in them is treated as SOFT: a doc that WARNS against the brick stays
62clean, while a doc that advocates running it does not.
63
64Run::
65
66 check_no_antirecovery.py # scan the whole tracked tree
67 check_no_antirecovery.py FILE ... # scan listed files
68 check_no_antirecovery.py --selftest # prove the detector fires on brick
69 # actions and stays silent on the real
70 # recovery scripts + defensive checks
71
72Exit 0 when no anti-recovery action is found, 1 otherwise (or on a failing
73selftest, or on an empty/unreadable scan set).
74"""

140# --- Forbidden ACTION patterns -----------------------------------------------
141#
142# Each entry: (rule, compiled regex, hard). `hard` patterns fire on any
143# non-comment occurrence. Non-`hard` (SOFT) patterns are exempted when the same
144# code line carries a DEFENSIVE_RE cue.
145
146# A terminal DLM lock. Only LCK_BOOT is known to render the SWD-DP permanently
147# unresponsive on the RA8D2; the recoverable OEM_PL0/PL1 debug-locks are
148# deliberately NOT here. Keep this a named token so a future terminal state is a
149# one-line addition.
150_TERMINAL = r"lck[_-]?boot"
151
152# A single `=` assignment operator, excluding the comparison operators (==, !=,
153# <=, >=). Used so `x = LCK_BOOT` (an action) fires while `x == LCK_BOOT` (a
154# check) does not.
155_ASSIGN = r"(?<![=!<>])=(?!=)"
156
157# The truthy right-hand sides that set a flag.
158_TRUE = r"(?:1\b|true\b|on\b|yes\b|set\b|enabled?\b)"
159
160# A set/transition verb. The lookarounds treat `_`, whitespace and punctuation
161# as boundaries but NOT letters, so the verb matches as an underscore-delimited
162# identifier component too -- `dlm_program(...)` and `never_regress_to_...` both
163# expose their verb -- while `reset`, `offset`, `clock`, `remove` do not.
164_SET_VERB = r"set|enable|program|transition|move|write|lock|enter|switch|regress|activate|go\s*to"
165_VERB = rf"(?<![A-Za-z])(?:{_SET_VERB})(?![A-Za-z])"
166
167RULES: list[tuple[str, re.Pattern[str], bool]] = [
168 # -- HARD: executed / programmatic ce ("Disable Initialize") set ----------
169 ("ce-set", re.compile(r"\bset[_ -]ce\b", re.IGNORECASE), True),
170 ("ce-set", re.compile(rf"\bce\b\s*(?:{_ASSIGN}|:)\s*{_TRUE}", re.IGNORECASE), True),
171 (
172 "ce-set",
173 re.compile(
174 r"(?:set[_ -]security[_ -]flags?|--?security[_ -]flags?[= ])[^\n]{0,24}\bce\b",
175 re.IGNORECASE,
176 ),
177 True,
178 ),
179 ("ce-set", re.compile(r"--disable[_-]initialize\b", re.IGNORECASE), True),
180 (
181 "ce-set",
182 re.compile(rf"\bdisable[_-]initialize\b\s*(?:{_ASSIGN}|:)\s*{_TRUE}", re.IGNORECASE),
183 True,
184 ),
185 # -- HARD: executed terminal DLM transition (rfp-cli -dlm LCK_BOOT) --------
186 ("dlm-terminal", re.compile(rf"--?dlm[= ]+['\"]?\s*{_TERMINAL}\b", re.IGNORECASE), True),
187 # -- SOFT: terminal-state / flag NAME governed by a set-verb (guarded) -----
188 (
189 "ce-set",
190 re.compile(rf"{_VERB}[^\n]{{0,24}}\bdisable[ _-]?initialize\b", re.IGNORECASE),
191 False,
192 ),
193 (
194 "dlm-terminal",
195 re.compile(rf"{_VERB}[^\n]{{0,24}}{_TERMINAL}\b", re.IGNORECASE),
196 False,
197 ),
198 # -- SOFT: assignment target is the terminal state (`dlm = LCK_BOOT`) ------
199 (
200 "dlm-terminal",
201 re.compile(rf"{_ASSIGN}\s*[^\n;{{}}]{{0,24}}?{_TERMINAL}\b", re.IGNORECASE),
202 False,
203 ),
204]
205
206# A code line that CHECKS / NEGATES / WARNS / COMPARES rather than acts. The
207# word cues are word-boundaried so that an identifier merely containing one of
208# them (e.g. `never_regress_to_lck_boot`) does NOT earn the exemption -- a
209# function that does the transition is a violation whatever it is named. The
210# comparison operators keep a state check (`if (dlm == LCK_BOOT)`) from reading
211# as an action. Guards SOFT rules only (and, for prose files, HARD rules too).
212DEFENSIVE_RE = re.compile(
213 r"==|!=|"
214 r"\b(?:unset|must\s+not|must\s+never|may\s+be\s+set|may\s+have\s+been\s+set|"
215 r"not\s+have\s+been\s+set|not\s+be\s+set|do\s+not|don't|never|should\s+not|"
216 r"shall\s+not|verify|verifies|ensure|is\s+set|was\s+set|been\s+set)\b",
217 re.IGNORECASE,
218)

224def lang_of(rel: str) -> str:
225 """Classify a tracked path's comment style / prose-ness for scanning.
226
227 Returns one of "hash", "c", "xml", "prose", "none", or "skip" (binary /
228 out-of-scope). Extension wins; a few known extensionless basenames (the git
229 hooks) map to "hash".
230 """
231 p = Path(rel)
232 lang = _EXT_LANG.get(p.suffix.lower())
233 if lang is not None:
234 return lang
235 if p.name in HASH_BASENAMES:
236 return "hash"
237 if p.suffix == "" and p.name in HASH_STEM_HINTS:
238 return "hash"
239 return "skip"

242def _blank_hash_line(line: str) -> str:
243 """Blank a shell/python/yaml ``#`` comment, respecting quotes.
244
245 A ``#`` starts a comment only outside quotes and at a word boundary (line
246 start after whitespace, or preceded by whitespace) -- so ``${x#y}`` and
247 ``http://a#b`` are left intact while ``foo # note`` is trimmed.
248 """
249 in_s = in_d = False
250 i, n = 0, len(line)
251 while i < n:
252 c = line[i]
253 if in_s:
254 if c == "'":
255 in_s = False
256 elif in_d:
257 if c == "\\":
258 i += 2
259 continue
260 if c == '"':
261 in_d = False
262 elif c == "'":
263 in_s = True
264 elif c == '"':
265 in_d = True
266 elif c == "#" and (i == 0 or line[i - 1].isspace()):
267 return line[:i] + " " * (n - i)
268 i += 1
269 return line

272def _blank_c_line(line: str) -> str:
273 """Blank a C ``//`` line comment, respecting quotes."""
274 in_s = in_d = False
275 i, n = 0, len(line)
276 while i < n:
277 c = line[i]
278 if in_s:
279 if c == "\\":
280 i += 2
281 continue
282 if c == "'":
283 in_s = False
284 elif in_d:
285 if c == "\\":
286 i += 2
287 continue
288 if c == '"':
289 in_d = False
290 elif c == "'":
291 in_s = True
292 elif c == '"':
293 in_d = True
294 elif c == "/" and i + 1 < n and line[i + 1] == "/":
295 return line[:i] + " " * (n - i)
296 i += 1
297 return line

300def _blank_blocks(text: str, open_tok: str, close_tok: str) -> str:
301 """Blank ``open_tok ... close_tok`` spans, preserving newlines and offsets."""
302 out: list[str] = []
303 i, n = 0, len(text)
304 while i < n:
305 start = text.find(open_tok, i)
306 if start < 0:
307 out.append(text[i:])
308 break
309 out.append(text[i:start])
310 end = text.find(close_tok, start + len(open_tok))
311 end = n if end < 0 else end + len(close_tok)
312 out.append(re.sub(r"[^\n]", " ", text[start:end]))
313 i = end
314 return "".join(out)

317def blank_comments(text: str, lang: str) -> str:
318 """Return `text` with comments blanked (strings preserved), for `lang`.
319
320 Strings are deliberately kept so a quoted CLI argument like
321 ``-dlm "LCK_BOOT"`` is still seen. Prose files are returned unchanged --
322 every match in them is treated as SOFT by the caller.
323 """
324 if lang == "c":
325 text = _blank_blocks(text, "/*", "*/")
326 return "\n".join(_blank_c_line(ln) for ln in text.split("\n"))
327 if lang == "hash":
328 return "\n".join(_blank_hash_line(ln) for ln in text.split("\n"))
329 if lang == "xml":
330 return _blank_blocks(text, "<!--", "-->")
331 return text

334def scan_text(raw: str, rel: str, lang: str) -> list[dict]:
335 """Find forbidden anti-recovery ACTIONS in one file's text.
336
337 `lang == "prose"` forces every rule to be treated as SOFT (documentation is
338 described, not executed), so a doc that warns against the brick stays clean.
339 """
340 code = blank_comments(raw, lang)
341 prose = lang == "prose"
342 found: list[dict] = []
343 for lineno, line in enumerate(code.split("\n"), start=1):
344 defended = DEFENSIVE_RE.search(line) is not None
345 for rule, pattern, hard in RULES:
346 m = pattern.search(line)
347 if not m:
348 continue
349 if (not hard or prose) and defended:
350 continue
351 found.append(
352 {
353 "path": rel,
354 "line": lineno,
355 "rule": rule,
356 "match": m.group(0).strip(),
357 "text": line.strip(),
358 }
359 )
360 break
361 return found

364def tracked_files(explicit: list[str]) -> list[Path]:
365 """Enumerate the tracked first-party files in scope.
366
367 Tracked via ``git ls-files``, not globbed: a glob also sweeps build output
368 and changes the verdict depending on whether the caller has built. Excluded
369 prefixes, the recovery scripts, and this checker are dropped here.
370 """
371 if explicit:
372 return [Path(p) for p in explicit]
373 try:
374 listed = subprocess.run(
375 ["git", "ls-files", "-z"], # noqa: S607 # trusted: fixed git argv
376 capture_output=True,
377 text=True,
378 check=True,
379 cwd=REPO_ROOT,
380 ).stdout
381 except (OSError, subprocess.CalledProcessError) as exc:
382 sys.exit(
383 f"check_no_antirecovery.py: FATAL -- cannot list tracked files: {exc}\n"
384 " This gate enumerates via git and must not fall back to a glob."
385 )
386 out: list[Path] = []
387 for name in listed.split("\0"):
388 if not name:
389 continue
390 if name in EXCLUDED_FILES:
391 continue
392 if any(name.startswith(pfx) for pfx in EXCLUDED_PREFIXES):
393 continue
394 out.append(REPO_ROOT / name)
395 return out

398def analyse(files: list[Path]) -> list[dict]:
399 """Return every forbidden-action finding across the given files."""
400 findings: list[dict] = []
401 for path in files:
402 rel = str(path.relative_to(REPO_ROOT)) if path.is_absolute() else str(path)
403 lang = lang_of(rel)
404 if lang == "skip":
405 continue
406 try:
407 raw = path.read_text(encoding="utf-8")
408 except (OSError, UnicodeDecodeError):
409 continue
410 findings.extend(scan_text(raw, rel, lang))
411 return findings

494def selftest(tmp: Path) -> int:
495 """Assert the detector fires on brick actions and stays silent otherwise.
496
497 Runs BOTH directions -- synthetic brick files must be detected, and the
498 REAL recovery scripts plus a defensive-check fixture must produce nothing.
499 The recovery scripts are scanned by CONTENT here (bypassing the path
500 exclusion) so the assertion proves the PATTERNS keep them silent, not
501 merely that they are excluded.
502 """
503 failures: list[str] = []
504
505 for idx, (body, fname, should_fire) in enumerate(SELFTEST_CASES):
506 p = tmp / f"{idx:02d}_{fname}"
507 p.write_text(body, encoding="utf-8")
508 fired = bool(scan_text(body, fname, lang_of(fname)))
509 if fired != should_fire:
510 verb = "did not fire" if should_fire else "fired"
511 failures.append(f" synthetic [{fname}]: gate {verb} (unexpected)")
512
513 # The real recovery scripts, by content, must be silent on the PATTERNS.
514 for rel in sorted(RECOVERY_SCRIPTS):
515 f = REPO_ROOT / rel
516 if not f.is_file():
517 failures.append(f" recovery script missing: {rel}")
518 continue
519 hits = scan_text(f.read_text(encoding="utf-8"), rel, lang_of(rel))
520 failures.extend(f" recovery script {rel}:{h['line']} matched {h['match']!r}" for h in hits)
521
522 if failures:
523 print("check_no_antirecovery.py --selftest: FAILED\n", file=sys.stderr)
524 print("\n".join(failures), file=sys.stderr)
525 return 1
526
527 fires = sum(1 for c in SELFTEST_CASES if c[2])
528 total = len(SELFTEST_CASES)
529 print(
530 f"check_no_antirecovery.py --selftest: PASS "
531 f"({total} synthetic cases: {fires} must fire, {total - fires} must stay silent; "
532 f"{len(RECOVERY_SCRIPTS)} real recovery scripts silent by pattern)"
533 )
534 return 0

541 file=sys.stderr,

549 print(f" ... {len(findings) - MAX_FINDINGS_SHOWN} more (truncated)", file=sys.stderr)
550 print(
551 "\nOwner policy (2026-07-23): this project must NEVER permanently disable\n"
552 "device recovery. Forbidden actions:\n"
553 " [ce-set] setting the ce Security Flag / Disable Initialize Command\n"

567def main(argv: list[str]) -> int:
568 """Scan first-party files for anti-recovery brick actions, or run the selftest.
569
570 CI runs ``--selftest`` before the scan: a detector whose patterns stopped
571 matching would report a clean tree, and only an assertion in both directions
572 tells that apart from a genuinely clean tree.
573
574 Returns 0 when no forbidden action is found, 1 on a finding, on an empty
575 scan set, or on a failing selftest.
576 """
577 ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
578 ap.add_argument("files", nargs="*", help="specific files to scan")
579 ap.add_argument(
580 "--selftest",
581 action="store_true",
582 help="prove the detector fires on brick actions and not on recovery/defensive code",
583 )
584 args = ap.parse_args(argv[1:])
