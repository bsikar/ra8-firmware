369RA8_INTERNAL
370static const mstp_family_t* mstp_family_for_addr(uint64_t addr, uint32_t* out_idx)
371{
372 for (uint32_t f = 0U; f < (sizeof(k_families) / sizeof(k_families[0])); f++) {
373 const mstp_family_t* fam = &k_families[f];
374 const uint64_t span = fam->stride * (uint64_t)fam->count;
375 if ((addr >= fam->base) && (addr < (fam->base + span))) {
376 *out_idx = (uint32_t)((addr - fam->base) / fam->stride);
377 return fam;
378 }
379 }
380 return nullptr;
381}

384RA8_INTERNAL
385static bool mstp_bit_set(uint16_t id)
386{
387 const uint32_t reg = ((uint32_t)id >> (uint32_t)k_mstp_id_reg_shift) & (uint32_t)k_mstp_byte_mask;
388 const uint32_t bit = (uint32_t)id & (uint32_t)k_mstp_id_bit_mask;
389 if (reg >= (uint32_t)k_mstp_reg_count) {
390 return false; /* defensive: a malformed id gates nothing. */
391 }
392 return (s_mstpcr[reg] & ((uint32_t)1U << bit)) != 0U;
393}

400RA8_PRIV void board_mstp_reset(void)
401{
402 s_mstpcr[0] = (uint32_t)k_mstp_reset_a;
403 for (uint32_t r = 1U; r < (uint32_t)k_mstp_reg_count; r++) {
404 s_mstpcr[r] = (uint32_t)k_mstp_reset_rest;
405 }
406 s_gated_reads = 0U;
407 s_gated_writes = 0U;
408 s_last_gated = "-";
409}

411RA8_PRIV void board_mstp_apply_write(uint64_t off, unsigned size, uint32_t value)
412{
413 for (unsigned i = 0U; i < size; i++) {
414 const uint64_t b = off + (uint64_t)i;
415 if (b >= (uint64_t)k_board_mstp_win_span) {
416 break;
417 }
418 const uint32_t reg = (uint32_t)(b / (uint64_t)k_mstp_bytes_per_reg);
419 const uint32_t sh = (uint32_t)(b % (uint64_t)k_mstp_bytes_per_reg) * (uint32_t)k_mstp_byte_bits;
420 const uint32_t vb =
421 (value >> ((uint32_t)i * (uint32_t)k_mstp_byte_bits)) & (uint32_t)k_mstp_byte_mask;
422 s_mstpcr[reg] = (s_mstpcr[reg] & ~((uint32_t)k_mstp_byte_mask << sh)) | (vb << sh);
423 }
424}
425
426RA8_PRIV uint32_t board_mstp_read_reg(uint64_t off, unsigned size)
427{
428 uint32_t v = 0U;
429 for (unsigned i = 0U; i < size; i++) {
430 const uint64_t b = off + (uint64_t)i;
431 if (b >= (uint64_t)k_board_mstp_win_span) {
432 continue;
433 }
434 const uint32_t reg = (uint32_t)(b / (uint64_t)k_mstp_bytes_per_reg);
435 const uint32_t sh = (uint32_t)(b % (uint64_t)k_mstp_bytes_per_reg) * (uint32_t)k_mstp_byte_bits;
436 const uint32_t vb = (s_mstpcr[reg] >> sh) & (uint32_t)k_mstp_byte_mask;
437 v |= vb << ((uint32_t)i * (uint32_t)k_mstp_byte_bits);
438 }
439 return v;
440}

442RA8_PRIV bool board_mstp_addr_stopped(uint64_t addr)
443{
444 uint32_t idx = 0U;
445 const mstp_family_t* fam = mstp_family_for_addr(addr, &idx);
446 if (fam == nullptr) {
447 return false; /* not gated: unmodelled or a peripheral with no module-stop. */
448 }
449 return mstp_bit_set(fam->ids[idx]);
450}

452RA8_PRIV void board_mstp_note_gated_access(uint64_t addr, bool is_write)
453{
454 uint32_t idx = 0U;
