116static inline void internal_actcsr_write(uint16_t value)
117{
118 const uint16_t key =
119 (uint16_t)((value >> (uint16_t)k_ra8_dual_core_actcsr_key_shift) & k_dc_byte_mask);
120 if (key != (uint16_t)k_ra8_dual_core_actcsr_key_value) {
121 /* Only caller ra8_cpu1_release always presents KEY=0xA5; wrong-key drop is host-unreachable. */
122 return; /* Silent drop -- matches HUM Ch 2.9.1.9 KEY gate. */ /* GCOVR_EXCL_LINE */
123 }
124 uint16_t actcsr = s_fake.actcsr;
125 if ((value & (uint16_t)k_ra8_dual_core_actcsr_actreq_mask) != 0U) {
126 actcsr |= (uint16_t)k_ra8_dual_core_actcsr_act_mask;
127 }
128 s_fake.actcsr = actcsr;
129}

492ra8_err_t ra8_cpu1_release(void* entry, void* sp)
493{
494 if (entry == nullptr || sp == nullptr) {
495 ra8_log_error(s_tag, "release: NULL entry or sp");
496 return k_ra8_err_null_ptr;
497 }
498 if (((uintptr_t)entry & ((uintptr_t)k_ra8_dual_core_align_vtor - 1U)) != 0U) {
499 ra8_log_error(s_tag, "release: entry not 128-byte aligned");
500 return k_ra8_err_invalid_arg;
501 }
502 if (((uintptr_t)sp & ((uintptr_t)k_ra8_dual_core_align_sp - 1U)) != 0U) {
503 ra8_log_error(s_tag, "release: sp not 8-byte aligned");
504 return k_ra8_err_invalid_arg;
505 }
506 if (!internal_is_cpu0()) {
507 /* RA8_BUILD_FOR_CPU1 is never defined in host builds; this branch is compile-time dead. */
508 ra8_log_error(s_tag, "release: caller is not CPU0"); /* GCOVR_EXCL_LINE */
509 return k_ra8_err_not_supported; /* GCOVR_EXCL_LINE */
510 }
511
512 /* HUM Ch 2.9.1.7 "CPU1INITVTOR" p 128-129: program CPU1's initial
513 * vector-table base. CPU1 will latch this into VTOR on the next
514 * reset-release edge driven by ACTCSR.ACTREQ below. The M33 also
515 * fetches its initial MSP from the first word of this table. */
516 internal_initvtor_write((uint32_t)(uintptr_t)entry);
517
518 /* HUM Ch 2.9.1.8 "CPU1WAITCR" p 129: make sure CPUWAIT is cleared so
519 * CPU1 begins executing immediately on activation. Reset value is
520 * already 0 but writing it defensively keeps repeated halt/release
521 * cycles deterministic. */
522 internal_waitcr_write(0U);
523
524 /* HUM Ch 2.9.1.9 "CPU1ACTCSR" p 129-130: writes require KEY=0xA5 in
525 * bits 15:8 alongside the ACTREQ bit. Anything else is silently
526 * dropped by the hardware. */
527 const uint16_t actreq = (uint16_t)(((uint16_t)k_ra8_dual_core_actcsr_key_value
528 << (uint16_t)k_ra8_dual_core_actcsr_key_shift) |
529 (uint16_t)k_ra8_dual_core_actcsr_actreq_mask);
530 internal_actcsr_write(actreq);
531
532 /* Bounded poll for ACT to assert (HUM Ch 2.9.1.9 ACT bit p 130). */
533 return internal_wait_act_set();
534}

559ra8_err_t ra8_cpu1_halt(void)
560{
561 if (!internal_is_cpu0()) {

563 ra8_log_error(s_tag, "halt: caller is not CPU0"); /* GCOVR_EXCL_LINE */
564 return k_ra8_err_not_supported; /* GCOVR_EXCL_LINE */
565 }
566
567 internal_waitcr_write((uint8_t)k_ra8_dual_core_waitcr_cpuwait_mask);
568 return k_ra8_ok;

594bool ra8_cpu1_is_running(void)
595{
596 /* Sequential ifs so the function has zero compound boolean
597 * decisions -- keeps MC/DC test obligations linear per condition
598 * rather than N+1 combinatorial, and matches the safety-checklist
599 * preference for split conditions in libs/ra8_hal/. */
600 const bool active = (internal_actcsr_read() & (uint16_t)k_ra8_dual_core_actcsr_act_mask) != 0U;
601 if (!active) {
602 return false;
603 }
604 const bool stalled =
605 (internal_waitcr_read() & (uint8_t)k_ra8_dual_core_waitcr_cpuwait_mask) != 0U;
606 return !stalled;
607}
