4"""misra_ratchet.py -- MISRA-C 2012 ratchet gate (compare vs committed baseline).
5
6`scripts/checks/misra_check_inner.sh` (make misra) runs cppcheck + the bundled
7misra.py addon and writes one finding per line to `build/misra/results.txt`.
8Until now that audit was advisory-only: no CI job consumed it, so the
9finding count could only grow (issue #240). This script turns the audit
10into a one-way ratchet against a committed baseline:
11
12* NEW findings (any per-file-per-rule count above the baseline, or any
13 file/rule pair absent from the baseline) FAIL the gate.
14* Shrinkage (findings burned down) PASSES with a notice to re-baseline via
15 `make misra-baseline`, which locks the progress in so the debt can never
16 quietly grow back.
17
18Baseline normalization -- per-file-per-rule COUNTS, not raw finding lines:
19raw misra.py findings carry line numbers, which churn on every unrelated
20edit above them, and message text, which embeds identifiers. Fingerprinting
21the violating source line churns whenever that line is touched for any
22reason. A `(file, rule) -> count` map is invariant under both kinds of
23drift, still trips on any new violation of a rule in a file (the bucket
24count rises), keeps the committed baseline small and diffable, and is fully
25deterministic. The accepted coarseness: within one bucket, adding one
26violation while simultaneously fixing another nets zero and passes -- fine
27for a burn-down ratchet whose end state is an empty baseline.
28
29The baseline records the cppcheck version that generated it. cppcheck
30majors disagree on rule coverage (2.13 has no C23 parse; newer versions
31change false-positive sets), so the gate WARNS loudly on a version mismatch
32-- regenerate the baseline on the CI-pinned toolchain (the self-hosted
33runner / dev box cppcheck), never on a drifted local install.
34
35Usage:
36 python3 scripts/checks/misra_ratchet.py --check # gate (default)
37 python3 scripts/checks/misra_ratchet.py --update # rewrite the baseline
38
39Both modes read `build/misra/results.txt`; run the audit first:
40 bash scripts/checks/misra_check_inner.sh
41
42Copyright (c) 2026 Brighton Sikarskie
43SPDX-License-Identifier: MIT

83def load_results(path: Path) -> tuple[Counter[tuple[str, str]], dict[tuple[str, str], list[str]]]:
84 r"""Parse the misra_check.sh TSV into per-(file, rule) counts + raw lines.
85
86 Each row is `rule \\t severity \\t file \\t line \\t message`. Rows that
87 do not split into five columns are ignored (defensive: the awk parser in
88 misra_check.sh already guarantees the shape).
89 """
90 counts: Counter[tuple[str, str]] = Counter()
91 details: dict[tuple[str, str], list[str]] = {}
92 for raw in path.read_text(encoding="utf-8").splitlines():
93 cols = raw.split("\t")
94 if len(cols) != RESULTS_COLUMNS:
95 continue
96 rule, _severity, fname, line, message = cols
97 key = (fname, rule)
98 counts[key] += 1
99 details.setdefault(key, []).append(f"{fname}:{line}: {message} [{rule}]")
100 return counts, details

103def load_baseline(path: Path) -> tuple[Counter[tuple[str, str]], str]:
104 """Parse the committed baseline into counts + the recorded cppcheck version."""
105 counts: Counter[tuple[str, str]] = Counter()
106 version = "unknown"
107 for raw in path.read_text(encoding="utf-8").splitlines():
108 if raw.startswith("# cppcheck:"):
109 version = raw.split(":", 1)[1].strip()
110 continue
111 if not raw or raw.startswith("#"):
112 continue
113 cols = raw.split("\t")
114 if len(cols) != BASELINE_COLUMNS:
115 print(f"misra_ratchet.py: ERROR -- malformed baseline row: {raw!r}")
116 sys.exit(1)
117 fname, rule, count = cols
118 counts[(fname, rule)] = int(count)
119 return counts, version

122def write_baseline(path: Path, counts: Counter[tuple[str, str]]) -> None:
123 """Serialize `counts` (sorted by file, then rule) with a provenance header."""
124 total = sum(counts.values())
125 lines = [
126 "# MISRA-C 2012 ratchet baseline -- per-file-per-rule finding counts.",
127 "# Consumed by scripts/checks/misra_ratchet.py --check (CI job: misra).",
128 "# Regenerate ON THE CI-PINNED CPPCHECK (self-hosted runner / dev box):",
129 "# make misra-baseline",
130 "# and commit the result. Never regenerate to absorb NEW findings --",
131 "# fix those at the root or record a deviation in",
132 "# docs/qualification/MISRA_DEVIATIONS.md first.",
133 f"# cppcheck: {cppcheck_version()}",
134 f"# total findings: {total}",
135 "# columns: file<TAB>rule<TAB>count",
136 ]
137 lines += [f"{fname}\t{rule}\t{count}" for (fname, rule), count in sorted(counts.items())]
138 path.write_text("\n".join(lines) + "\n", encoding="utf-8")

145 """Print each offending (file, rule) bucket with its raw finding lines."""

161def check(counts: Counter[tuple[str, str]], details: dict[tuple[str, str], list[str]]) -> int:
162 """Ratchet `counts` against the committed baseline; return the exit code."""
163 if not BASELINE_FILE.is_file():
164 print(
165 f"misra_ratchet.py: ERROR -- baseline {BASELINE_FILE} missing; "
166 f"generate it with `make misra-baseline` and commit it."
167 )
168 return 1
169 baseline, base_version = load_baseline(BASELINE_FILE)
170
171 current_version = cppcheck_version()
172 if current_version != base_version:
173 print(
174 f"misra_ratchet.py: WARNING -- cppcheck version skew: baseline was "
175 f"generated by {base_version!r} but this run used {current_version!r}. "
176 f"Finding sets are not comparable across cppcheck versions; run the "
177 f"gate on the CI-pinned toolchain before trusting a red or a green."
178 )
179
180 regressions = [
181 (fname, rule, baseline.get((fname, rule), 0), count)
182 for (fname, rule), count in sorted(counts.items())
183 if count > baseline.get((fname, rule), 0)
184 ]
185 if regressions:
186 report_regressions(regressions, details)
187 return 1
188
189 # No regressions past this point, so every current bucket is <= its
190 # baseline bucket and the totals difference is exactly the burn-down.
191 cur_total = sum(counts.values())
192 base_total = sum(baseline.values())
193 shrunk = base_total - cur_total
194 if shrunk > 0:
195 print(
196 f"misra_ratchet.py: PASS -- no new findings, and {shrunk} baseline "
197 f"finding(s) were burned down ({base_total} -> {cur_total}). Lock the "
198 f"progress in: run `make misra-baseline` on the CI-pinned cppcheck "
199 f"and commit the shrunken .github/misra-baseline.txt."
200 )
201 else:
202 print(f"misra_ratchet.py: PASS -- {cur_total} finding(s), all within the baseline.")
203 return 0

206def main() -> int:

221
222 if not RESULTS_TSV.is_file():
223 print(

229 if not counts:

233 "parse in --check mode almost always means the audit itself broke "
234 "(cppcheck missing, addon missing, or dump generation failed)."
235 )
236 return 1
237
238 if args.update:
239 write_baseline(BASELINE_FILE, counts)
240 print(
241 f"misra_ratchet.py: baseline updated -- {sum(counts.values())} finding(s) "
242 f"across {len(counts)} (file, rule) bucket(s) -> {BASELINE_FILE}"
243 )
244 return 0
245 return check(counts, details)
