24def compile_epub(
25 path: str | Path,
26 max_image_edge: int = MAX_IMAGE_EDGE,
27 skip_images: bool = SKIP_IMAGES,
28 pixel_format: int = PIXFMT_GRAY4,

30 """Compile one EPUB into an inflated .rabook blob.
31
32 Assembles in dependency order -- stylesheets, then images (so the cover
33 index is known), then spine chapters -- and takes each chapter's `<body>`
34 subtree, falling back to the whole document when there is none.
35
36 The pass is deliberately lenient, because a book that fails to open is worse
37 than a book missing one asset. A manifest entry absent from the zip is
38 skipped; an image Pillow cannot decode is skipped; a spine idref with no
39 manifest entry is skipped. All four are silent. The consequence worth
40 knowing at 2am: a corrupt EPUB compiles successfully into a blob with
41 missing content rather than reporting an error.
42
43 Recursion limit: `add_element` recurses per level of markup nesting, so the
44 limit is raised to 100000 here. This mutates interpreter global state and is
45 not restored.
46
47 Args:
48 path: Path to the source .epub.
49 max_image_edge: Long-edge downscale cap in pixels; 0 preserves source
50 resolution. See `BlobBuilder.add_raster_image` for why the two
51 paths are not byte-equivalent.
52 skip_images: Drop every image, producing a text-only blob small enough
53 to bake into MRAM as a fixture. The cover is dropped too.
54 pixel_format: Device-profile raster depth (issue #343): PIXFMT_GRAY4
55 (the default 4bpp packing) or PIXFMT_GRAY8 (lossless 8bpp). Only the
56 raster image arm reads it; SVG is unaffected.
57
58 Returns:
59 Tuple of (blob, meta, bb): the serialized inflated blob, the metadata
60 dict, and the BlobBuilder itself so callers can report table sizes.
61
62 Raises:
63 KeyError: Malformed EPUB with no container.xml.
64 xml.etree.ElementTree.ParseError: container.xml or the OPF is not
65 well-formed. A broken TOC is tolerated; a broken OPF is not.
66 zipfile.BadZipFile: `path` is not a zip archive.
67 """
68 sys.setrecursionlimit(100000)
69 bb = BlobBuilder()
70 with ZipFile(path) as zf:
71 opf_dir, meta, manifest, spine, cover_id = parse_opf(zf)
72 labels = parse_toc(zf, opf_dir, manifest)
73 names = set(zf.namelist())
74
75 def resolve(href: str) -> str:
76 return posixpath.normpath(posixpath.join(opf_dir, href))
77
78 # stylesheets (verbatim)
79 for href, media, _props in manifest.values():
80 if media == "text/css":
81 full = resolve(href)
82 if full in names:
83 bb.add_stylesheet(zf.read(full).decode("utf-8", "replace"))
84
85 # images (raster -> 4bpp, svg -> vector); remember manifest-id -> image index
86 id_to_image = {}
87 for mid, (href, media, _props) in () if skip_images else manifest.items():
88 full = resolve(href)
89 if full not in names:
90 continue
91 try:
92 if media == "image/svg+xml":
93 id_to_image[mid] = bb.add_svg_image(href, zf.read(full))
94 elif media.startswith("image/"):
95 id_to_image[mid] = bb.add_raster_image(
96 href, zf.read(full), max_image_edge, pixel_format
97 )
98 except (OSError, ValueError):
99 pass
100 if cover_id and cover_id in id_to_image:
101 bb.cover_index = id_to_image[cover_id]
102
103 # spine chapters -> faithful DOM (body subtree)
104 href_by_id = {mid: h for mid, (h, _m, _p) in manifest.items()}
105 for idref in spine:
106 href = href_by_id.get(idref)
107 if not href:
108 continue
109 full = resolve(href)
110 if full not in names:
111 continue
112 dom = DomBuilder()
113 dom.feed(zf.read(full).decode("utf-8", "replace"))
114 body = find_first(dom.root, "body") or dom.root
115 title = labels.get(full, "")
116 bb.add_chapter(body, title, href)
117
118 return bb.serialize(meta), meta, bb
