55uint32_t ra8_cs_crc32(const uint8_t* data, uint32_t len)
56{
57 if (data == nullptr) {
58 return 0U;
59 }
60 if (len == 0U) {
61 return 0U;
62 }
63 uint32_t crc = (uint32_t)k_ra8_cs_crc32_seed;
64 for (uint32_t i = 0U; i < len; i++) {
65 crc ^= data[i];
66 for (uint8_t b = 0U; b < (uint8_t)k_ra8_cs_crc_bits; b++) {
67 bool lsb = (crc & 1U) != 0U;
68 crc >>= 1U;
69 if (lsb) {
70 crc ^= (uint32_t)k_ra8_cs_crc32_poly;
71 }
72 }
73 }
74 return crc ^ (uint32_t)k_ra8_cs_crc32_seed;
75}

115{

171ra8_err_t ra8_cs_super_write(ra8_cache_store_t* store, uint32_t clean)
172{
173 RA8_CHECK_NULL_PTR(store, s_tag, "store");
174 RA8_CHECK_NULL_PTR(store->staging, s_tag, "staging");
175 ra8_cs_super_t sb = {.magic = (uint32_t)k_ra8_cs_super_magic,
176 .version = (uint32_t)k_ra8_cs_format_version,
177 .seq = store->next_seq,
178 .clean = clean,
179 .entry_count = 0U,
180 .live_sectors = store->live_sectors,
181 .next_seq = store->next_seq,
182 .log_start = store->log_start,
183 .data_capacity = store->data_capacity,
184 .logical_sectors = store->logical_sectors,
185 .crc = 0U};
186 for (uint16_t i = 0U; i < store->index_cap; i++) {
187 if ((store->index[i].flags & (uint8_t)k_ra8_cache_store_flag_in_use) != 0U) {
188 sb.entry_count++;
189 }
190 }
191 sb.crc = ra8_cs_crc32((const uint8_t*)&sb, (uint32_t)(sizeof(sb) - sizeof(sb.crc)));
192 (void)memset(store->staging, 0, (size_t)k_ra8_cache_store_sector_bytes);
193 (void)memcpy(store->staging, &sb, sizeof(sb));
194 return ra8_cs_sector_write(store, 0U, store->staging);
195}

281static uint32_t cs_dir_pack_sector(ra8_cache_store_t* store, uint16_t* slot, uint8_t* out512)
282{
283 if (store == nullptr) {
284 return 0U;
285 }
286 if (out512 == nullptr) {
287 return 0U;
288 }
289 (void)memset(out512, 0, (size_t)k_ra8_cache_store_sector_bytes);
290 uint32_t packed = 0U;
291 for (uint32_t e = 0U; e < (uint32_t)k_ra8_cs_dir_per_sector; e++) {
292 for (; *slot < store->index_cap; (*slot)++) {
293 if ((store->index[*slot].flags & (uint8_t)k_ra8_cache_store_flag_in_use) != 0U) {
294 break;
295 }
296 }
297 if (*slot >= store->index_cap) {
298 break;
299 }
300 const ra8_cache_store_entry_t* src = &store->index[*slot];
301 ra8_cs_dir_ent_t ent = {.key = src->key,
302 .start_sector = src->start_sector,
303 .byte_len = src->byte_len,
304 .sector_count = src->sector_count,
305 .flags =
306 (uint16_t)(src->flags & (uint8_t)k_ra8_cache_store_flag_pinned)};
307 (void)memcpy(&out512[(size_t)e * (uint32_t)k_ra8_cs_dir_ent_bytes], &ent, sizeof(ent));
308 (*slot)++;
309 packed++;
310 }
311 return packed;
312}

428static bool cs_hdr_read(ra8_cache_store_t* store, uint32_t s, ra8_cs_entry_hdr_t* out_hdr)
429{
430 if (ra8_cs_sector_read(store, s, store->staging) != k_ra8_ok) {
431 return false;
432 }
433 ra8_cs_entry_hdr_t h = {};
434 (void)memcpy(&h, store->staging, sizeof(h));
435 if (h.magic != (uint32_t)k_ra8_cs_entry_magic) {
436 return false;
437 }
438 if (h.start_sector != s) {
439 return false;
440 }
441 if (h.sector_count == 0U) {
442 return false;
443 }
444 if (((uint64_t)s + (uint64_t)h.sector_count) > (uint64_t)store->logical_sectors) {
445 return false;
446 }
447 uint32_t want = ra8_cs_crc32((const uint8_t*)&h, (uint32_t)(sizeof(h) - sizeof(h.hdr_crc)));
448 if (h.hdr_crc != want) {
449 return false;
450 }
451 *out_hdr = h;
452 return true;

469static void cs_scan_accept(ra8_cache_store_t* store, const ra8_cs_entry_hdr_t* h, uint32_t* max_seq)
470{
471 bool pinned = (h->flags & (uint16_t)k_ra8_cache_store_flag_pinned) != 0U;
472 if (ra8_cs_index_find(store, h->key) >= 0) {
473 return; /* write-once: a duplicate key on flash is ignored (keep the first). */
474 }
475 if (ra8_cs_index_add(store, h->key, h->start_sector, h->sector_count, h->byte_len, pinned) < 0) {
476 return; /* index full: stop indexing further entries. */
477 }
478 store->live_sectors += h->sector_count;
479 if (h->seq > *max_seq) {
480 *max_seq = h->seq;
481 }
482}

500static ra8_err_t cs_scan_log(ra8_cache_store_t* store)
501{
502 RA8_CHECK_NULL_PTR(store, s_tag, "store");
503 RA8_CHECK_NULL_PTR(store->staging, s_tag, "staging");
504 store->live_sectors = 0U;
505 uint32_t max_seq = 0U;
506 uint32_t s = store->log_start;
507 for (uint32_t guard = store->log_start; guard < store->logical_sectors; guard++) {
508 if (s >= store->logical_sectors) {
509 break;
510 }
511 ra8_cs_entry_hdr_t h = {};
512 if (!cs_hdr_read(store, s, &h)) {
513 s += 1U;
514 continue;
515 }
516 cs_scan_accept(store, &h, &max_seq);
517 s += h.sector_count;
518 }
519 store->next_seq = max_seq + 1U;
520 return k_ra8_ok;
521}

580static ra8_err_t cs_geometry(ra8_cache_store_t* store, const ra8_cache_store_cfg_t* cfg)
581{
582 RA8_CHECK_NULL_PTR(store, s_tag, "store");
583 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg");
584 uint32_t dirs = ((uint32_t)cfg->index_cap + (uint32_t)k_ra8_cs_dir_per_sector - 1U) /
585 (uint32_t)k_ra8_cs_dir_per_sector;
586 store->checkpoint_dirs = (uint16_t)dirs;
587 store->log_start = 1U + dirs;
588 store->logical_sectors = cfg->logical_sectors;
589 if (cfg->logical_sectors < (store->log_start + (uint32_t)k_ra8_cache_store_min_sectors)) {
590 return k_ra8_err_invalid_size;
591 }
592 uint32_t op = (cfg->overprovision_pct != 0U) ? (uint32_t)cfg->overprovision_pct
593 : (uint32_t)k_ra8_cache_store_overprov_pct;
594 uint32_t usable = cfg->logical_sectors - store->log_start;
595 store->data_capacity =
596 (usable * ((uint32_t)k_ra8_cs_pct_full - op)) / (uint32_t)k_ra8_cs_pct_full;
597 if (store->data_capacity == 0U) {
598 store->data_capacity = 1U;
599 }
600 return k_ra8_ok;
601}

618static ra8_err_t cs_open_levelx(ra8_cache_store_t* store, const ra8_cache_store_cfg_t* cfg)
619{
620 RA8_CHECK_NULL_PTR(store, s_tag, "store");
621 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg");
622 if (!s_cs_lx_system_inited) {
623 (void)lx_nor_flash_initialize();
624 s_cs_lx_system_inited = true;
625 }
626 /* LevelX takes a mutable CHAR* name but never writes it. */
627 CHAR* nm = (cfg->name != nullptr) ? (CHAR*)(uintptr_t)cfg->name : (CHAR*)(uintptr_t)"ra8_cache";
628 if (cfg->format) {
629 if (lx_nor_flash_format(store->flash, nm, cfg->nor_driver_init, LX_NULL) != (UINT)LX_SUCCESS) {
630 return k_ra8_err_hw_init_failed;
631 }
632 }
633 if (lx_nor_flash_open(store->flash, nm, cfg->nor_driver_init) != (UINT)LX_SUCCESS) {
634 return k_ra8_err_hw_init_failed;
635 }
636 /* Stamp an empty clean superblock at format time so a fresh mount loads the
637 * (empty) checkpoint instead of scanning the whole -- as-yet unmapped -- log
638 * region, which on LevelX would allocate a physical sector per read. */
639 if (cfg->format) {
640 RA8_RETURN_ON_ERROR(ra8_cs_super_write(store, (uint32_t)k_ra8_cs_clean), s_tag, "format super");
641 }
642 return k_ra8_ok;
643}
