72RA8_INTERNAL
73static mz_zip_archive* s_zip(ra8_comic_t* c)
74{
75 /* Reinterpret the aligned byte arena as the archive via a `void*` seam (as
76 * `ra8_epub` does): a direct `uint8_t* -> mz_zip_archive*` cast would trip
77 * -Wcast-align, and a one-expression `(T*)(void*)p` cast trips
78 * bugprone-casting-through-void, so the `void*` is a named local. */
79 void* const storage = &c->zip_storage[0];
80 return (mz_zip_archive*)storage;
81}

102static size_t s_stream_read(void* opaque, mz_uint64 file_ofs, void* buf, size_t n)
103{
104 const ra8_comic_stream_t* sm = (const ra8_comic_stream_t*)opaque;
105 if (sm == nullptr) {
106 return 0U; /* GCOVR_EXCL_LINE -- open binds a live stream; miniz never passes NULL */
107 }
108 if (sm->read == nullptr) {
109 return 0U; /* GCOVR_EXCL_LINE -- open validates a non-NULL reader */
110 }
111 if (file_ofs >= sm->size) {
112 return 0U; /* GCOVR_EXCL_LINE -- miniz clamps reads within the archive size */
113 }
114 const uint64_t avail = sm->size - file_ofs;
115 const size_t want = ((uint64_t)n > avail) ? (size_t)avail : n;
116 return sm->read(sm->ctx, (uint64_t)file_ofs, buf, want);
117}

163RA8_INTERNAL
164static ra8_err_t s_add_entry(ra8_comic_t* c, mz_zip_archive* zip, mz_uint i)
165{
166 if (mz_zip_reader_is_file_a_directory(zip, i) != MZ_FALSE) {
167 return k_ra8_ok;
168 }
169 char nb[k_cbz_name_max] = {};
170 (void)mz_zip_reader_get_filename(zip, i, nb, (mz_uint)sizeof(nb));
171 nb[sizeof(nb) - 1U] = '\0';
172 const void* const z = memchr(nb, '\0', sizeof(nb));
173 const size_t nl = (z != nullptr) ? (size_t)((const char*)z - nb) : sizeof(nb);
174 if (!ra8_comic_is_page_name(nb, (uint16_t)nl)) {
175 return k_ra8_ok;
176 }
177 mz_zip_archive_file_stat st = {};
178 if (mz_zip_reader_file_stat(zip, i, &st) == MZ_FALSE) {
179 return k_ra8_ok; /* GCOVR_EXCL_LINE -- stat cannot fail on an index locate already found */
180 }
181 /* Decompression-limits guard: a page whose central-directory record
182 * declares an over-cap or bomb-ratio size rejects the whole archive
183 * before any inflation (the same policy every decoder enforces). */
184 const ra8_decomp_limits_t lim = ra8_decomp_limits_default();
185 const ra8_err_t derr =
186 ra8_decomp_check_declared(&lim, (uint64_t)st.m_comp_size, (uint64_t)st.m_uncomp_size);
187 if (derr != k_ra8_ok) {
188 return derr;
189 }
190 return ra8_comic_page_add(c,
191 nb,
192 (uint16_t)nl,
193 (uint64_t)st.m_uncomp_size,
194 1U,
195 (uint8_t)k_ra8_rar_method_store, /* method unused by CBZ */
196 (uint32_t)i,
197 0U,
198 0U);
199}

201ra8_err_t ra8_comic_cbz_open(ra8_comic_t* c)
202{
203 RA8_CHECK_NULL_PTR(c, s_tag_cbz, "cbz open: null c");
204 mz_zip_archive* zip = s_zip(c);
205 (void)memset(zip, 0, sizeof(*zip));
206 s_set_alloc(zip);
207 zip->m_pRead = s_stream_read;
208 zip->m_pIO_opaque = &c->stream;
209 if (mz_zip_reader_init(zip, (mz_uint64)c->size, 0U) == MZ_FALSE) {
210 return k_ra8_err_validation_failed;
211 }
212 c->zip_active = 1U;
213 const mz_uint num = mz_zip_reader_get_num_files(zip);
214 /* Decompression-limits guard: the many-tiny-entries bomb dies before
215 * any central-directory entry is touched. */
216 const ra8_decomp_limits_t lim = ra8_decomp_limits_default();
217 if ((uint64_t)num > (uint64_t)lim.max_entries) {
218 (void)ra8_comic_cbz_close(c);
219 return k_ra8_err_decomp_entries;
220 }
221 for (mz_uint i = 0U; i < num; ++i) { /* bound: finite central-directory count */
222 const ra8_err_t e = s_add_entry(c, zip, i);
223 if (e != k_ra8_ok) {
224 (void)ra8_comic_cbz_close(c);
225 return e;
226 }
227 }
228 return k_ra8_ok;
229}

231ra8_err_t ra8_comic_cbz_extract(ra8_comic_t* c,
232 const ra8_comic_page_t* p,
233 uint8_t* buf,
234 size_t cap,
235 size_t* got)
236{
237 RA8_CHECK_NULL_PTR(c, s_tag_cbz, "cbz extract: null c");
238 RA8_CHECK_NULL_PTR(p, s_tag_cbz, "cbz extract: null p");
239 RA8_CHECK_NULL_PTR(buf, s_tag_cbz, "cbz extract: null buf");
240 RA8_CHECK_NULL_PTR(got, s_tag_cbz, "cbz extract: null got");
241 *got = 0U;
242 if (c->zip_active == 0U) {
243 return k_ra8_err_invalid_state;
244 }
245 if ((uint64_t)cap < p->raw_size) {
246 return k_ra8_err_no_mem;
247 }
248 mz_zip_archive* zip = s_zip(c);
249 if (mz_zip_reader_extract_to_mem(zip, (mz_uint)p->zip_index, buf, cap, 0U) == MZ_FALSE) {
250 return k_ra8_err_validation_failed;
251 }
252 *got = (size_t)p->raw_size;
253 return k_ra8_ok;
254}
