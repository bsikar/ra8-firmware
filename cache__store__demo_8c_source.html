92static void demo_fill(uint8_t* buf, uint32_t len, uint8_t seed)
93{
94 if (buf == nullptr) {
95 return;
96 }
97 if (len == 0U) {
98 return;
99 }
100 for (uint32_t i = 0U; i < len; i++) {
101 buf[i] = (uint8_t)(seed + (uint8_t)i);
102 }

120static bool demo_matches(const uint8_t* buf, uint32_t len, uint8_t seed)
121{
122 if (buf == nullptr) {
123 return false;
124 }
125 if (len == 0U) {
126 return true;
127 }
128 for (uint32_t i = 0U; i < len; i++) {
129 if (buf[i] != (uint8_t)(seed + (uint8_t)i)) {
130 return false;
131 }
132 }
133 return true;
134}

191static ra8_err_t demo_put_and_verify(ra8_cache_store_t* st,
192 uint32_t key,
193 uint8_t seed,
194 uint32_t len,
195 const cache_store_demo_cfg_t* cfg)
196{
197 RA8_CHECK_NULL_PTR(st, "rcs_demo", "store is NULL");
198 RA8_CHECK_NULL_PTR(cfg, "rcs_demo", "cfg is NULL");
199 if (len > cfg->scratch_bytes) {
200 return k_ra8_err_invalid_size;
201 }
202 demo_fill(cfg->scratch, len, seed);
203 ra8_err_t rc = ra8_cache_store_put(st, key, cfg->scratch, len);
204 if (rc != k_ra8_ok) {
205 return rc;
206 }
207 ra8_cache_store_reader_t rd = {};
208 rc = ra8_cache_store_get(st, key, &rd);
209 if (rc != k_ra8_ok) {
210 return rc;
211 }
212 if (rd.byte_len != len) {
213 return k_ra8_err_invalid_state;
214 }
215 (void)memset(cfg->scratch, 0, (size_t)len);
216 rc = ra8_cache_store_read(&rd, 0U, cfg->scratch, len);
217 if (rc != k_ra8_ok) {
218 return rc;
219 }
220 if (!demo_matches(cfg->scratch, len, seed)) {
221 return k_ra8_err_invalid_state;
222 }
223 return k_ra8_ok;
224}

328{
329 RA8_CHECK_NULL_PTR(st, "rcs_demo", "store is NULL");
330 ra8_err_t rc = ra8_cache_store_pin(st, k_demo_key_cover, true);

334 rc = ra8_cache_store_evict(st, k_demo_key_tile_b);
335 if (rc != k_ra8_ok) {
336 return rc;
337 }
338 ra8_cache_store_reader_t rd = {};
339 if (ra8_cache_store_get(st, k_demo_key_tile_b, &rd) != k_ra8_err_not_found) {
340 return k_ra8_err_invalid_state; /* evicted entry must be a miss */

343 return k_ra8_err_invalid_state; /* survivor must resolve */

345 if (ra8_cache_store_get(st, k_demo_key_tile_c, &rd) != k_ra8_ok) {
346 return k_ra8_err_invalid_state; /* survivor must resolve */
347 }
348 if (ra8_cache_store_evict(st, k_demo_key_cover) != k_ra8_err_busy) {
349 return k_ra8_err_invalid_state; /* pinned entry must refuse eviction */

414static ra8_err_t demo_phase_remount(const cache_store_demo_cfg_t* cfg,
415 cache_store_demo_result_t* out)
416{
417 RA8_CHECK_NULL_PTR(cfg, "rcs_demo", "cfg is NULL");
418 RA8_CHECK_NULL_PTR(out, "rcs_demo", "out is NULL");
419 /* Reboot: control state is lost, the media survives. */
420 (void)memset(cfg->index, 0, (size_t)cfg->index_cap * sizeof(cfg->index[0]));
421 (void)memset(cfg->staging, 0, (size_t)cfg->staging_bytes);
422 ra8_cache_store_cfg_t cs2 = {};
423 if (demo_build_cfg(cfg, false, &cs2) != k_ra8_ok) {
424 out->status = k_cache_store_demo_err_remount;
425 return k_ra8_err_invalid_state;
426 }
427 ra8_cache_store_t st2 = {};
428 if (ra8_cache_store_init(&st2, &cs2) != k_ra8_ok) {
429 out->status = k_cache_store_demo_err_remount;
430 return k_ra8_err_invalid_state;
431 }
432 out->last_stage = k_cache_store_demo_stage_remounted;
433 if (demo_verify_survivors(&st2, cfg) != k_ra8_ok) {
434 out->status = k_cache_store_demo_err_persist;
435 return k_ra8_err_invalid_state;
436 }
437 out->survivors = k_demo_survivors;
438 ra8_cache_store_reader_t rd = {};
439 if (ra8_cache_store_get(&st2, k_demo_key_tile_b, &rd) != k_ra8_err_not_found) {
440 out->status = k_cache_store_demo_err_persist;
441 return k_ra8_err_invalid_state;
442 }
443 out->evicted_stayed_gone = true;
444 if (ra8_cache_store_evict(&st2, k_demo_key_cover) != k_ra8_err_busy) {
445 out->status = k_cache_store_demo_err_persist;
446 return k_ra8_err_invalid_state;
447 }
448 out->pin_survived = true;
449 (void)ra8_cache_store_close(&st2);
450 return k_ra8_ok;
451}

454{

456 RA8_CHECK_NULL_PTR(out, "rcs_demo", "out is NULL");

473 out->status = k_cache_store_demo_err_mount;

496 out->last_stage = k_cache_store_demo_stage_reused;
497
498 if (ra8_cache_store_close(&st) != k_ra8_ok) {
499 out->status = k_cache_store_demo_err_close;
500 return k_ra8_err_invalid_state;
501 }
502 out->last_stage = k_cache_store_demo_stage_closed;
503
504 if (demo_phase_remount(cfg, out) != k_ra8_ok) {
505 return k_ra8_err_invalid_state; /* demo_phase_remount set out->status */
506 }

508 out->status = k_cache_store_demo_ok;
