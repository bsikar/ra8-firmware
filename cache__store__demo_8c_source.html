200RA8_INTERNAL static ra8_err_t internal_demo_put_and_verify(ra8_cache_store_t* st,
201 uint32_t key,
202 uint8_t seed,
203 uint32_t len,
204 const cache_store_demo_cfg_t* cfg)
205{
206 RA8_CHECK_NULL_PTR(st, "rcs_demo", "store is NULL");
207 RA8_CHECK_NULL_PTR(cfg, "rcs_demo", "cfg is NULL");
208 if (len > cfg->scratch_bytes) {
209 return k_ra8_err_invalid_size;
210 }
211 internal_demo_fill(cfg->scratch, len, seed);
212 ra8_err_t rc = ra8_cache_store_put(st, key, cfg->scratch, len);
213 if (rc != k_ra8_ok) {
214 return rc;
215 }
216 ra8_cache_store_reader_t rd = {};
217 rc = ra8_cache_store_get(st, key, &rd);
218 if (rc != k_ra8_ok) {
219 return rc;
220 }
221 if (rd.byte_len != len) {
222 return k_ra8_err_invalid_state;
223 }
224 (void)memset(cfg->scratch, 0, (size_t)len);
225 rc = ra8_cache_store_read(&rd, 0U, cfg->scratch, len);
226 if (rc != k_ra8_ok) {
227 return rc;
228 }
229 if (!internal_demo_matches(cfg->scratch, len, seed)) {
230 return k_ra8_err_invalid_state;
231 }
232 return k_ra8_ok;

348{
349 RA8_CHECK_NULL_PTR(st, "rcs_demo", "store is NULL");

355 if (rc != k_ra8_ok) {
356 return rc;

358 ra8_cache_store_reader_t rd = {};

360 return k_ra8_err_invalid_state; /* evicted entry must be a miss */
361 }
362 if (ra8_cache_store_get(st, k_demo_key_tile_a, &rd) != k_ra8_ok) {
363 return k_ra8_err_invalid_state; /* survivor must resolve */

365 if (ra8_cache_store_get(st, k_demo_key_tile_c, &rd) != k_ra8_ok) {

368 if (ra8_cache_store_evict(st, k_demo_key_cover) != k_ra8_err_busy) {
369 return k_ra8_err_invalid_state; /* pinned entry must refuse eviction */

440RA8_INTERNAL static ra8_err_t internal_demo_phase_remount(const cache_store_demo_cfg_t* cfg,
441 cache_store_demo_result_t* out)
442{
443 RA8_CHECK_NULL_PTR(cfg, "rcs_demo", "cfg is NULL");
444 RA8_CHECK_NULL_PTR(out, "rcs_demo", "out is NULL");
445 /* Reboot: control state is lost, the media survives. */
446 (void)memset(cfg->index, 0, (size_t)cfg->index_cap * sizeof(cfg->index[0]));
447 (void)memset(cfg->staging, 0, (size_t)cfg->staging_bytes);
448 ra8_cache_store_cfg_t cs2 = {};
449 if (internal_demo_build_cfg(cfg, false, &cs2) != k_ra8_ok) {
450 out->status = k_cache_store_demo_err_remount;
451 return k_ra8_err_invalid_state;
452 }
453 ra8_cache_store_t st2 = {};
454 if (ra8_cache_store_init(&st2, &cs2) != k_ra8_ok) {
455 out->status = k_cache_store_demo_err_remount;
456 return k_ra8_err_invalid_state;
457 }
458 out->last_stage = k_cache_store_demo_stage_remounted;
459 if (internal_demo_verify_survivors(&st2, cfg) != k_ra8_ok) {
460 out->status = k_cache_store_demo_err_persist;
461 return k_ra8_err_invalid_state;
462 }
463 out->survivors = k_demo_survivors;
464 ra8_cache_store_reader_t rd = {};
465 if (ra8_cache_store_get(&st2, k_demo_key_tile_b, &rd) != k_ra8_err_not_found) {
466 out->status = k_cache_store_demo_err_persist;
467 return k_ra8_err_invalid_state;
468 }
469 out->evicted_stayed_gone = true;
470 if (ra8_cache_store_evict(&st2, k_demo_key_cover) != k_ra8_err_busy) {
471 out->status = k_cache_store_demo_err_persist;
472 return k_ra8_err_invalid_state;
473 }
474 out->pin_survived = true;
475 (void)ra8_cache_store_close(&st2);
476 return k_ra8_ok;
477}

479ra8_err_t cache_store_demo_run(const cache_store_demo_cfg_t* cfg, cache_store_demo_result_t* out)
480{
481 RA8_CHECK_NULL_PTR(cfg, "rcs_demo", "cfg is NULL");
482 RA8_CHECK_NULL_PTR(out, "rcs_demo", "out is NULL");
483 *out = (cache_store_demo_result_t){
484 .status = k_cache_store_demo_err_arg,
485 .last_stage = k_cache_store_demo_stage_none,
486 .survivors = 0U,
487 .evicted_key = k_demo_key_tile_b,
488 .evicted_stayed_gone = false,
489 .pin_survived = false,
490 };
491
492 ra8_cache_store_cfg_t cs = {};
493 if (internal_demo_build_cfg(cfg, true, &cs) != k_ra8_ok) {
494 out->status = k_cache_store_demo_err_mount;
495 return k_ra8_err_invalid_state;
496 }
497 ra8_cache_store_t st = {};
498 if (ra8_cache_store_init(&st, &cs) != k_ra8_ok) {
499 out->status = k_cache_store_demo_err_mount;
500 return k_ra8_err_invalid_state;
501 }
502 out->last_stage = k_cache_store_demo_stage_mounted;
503
504 if (internal_demo_phase_seed(&st, cfg) != k_ra8_ok) {
505 out->status = k_cache_store_demo_err_seed;
506 return k_ra8_err_invalid_state;
507 }
508 out->last_stage = k_cache_store_demo_stage_seeded;
509
510 if (internal_demo_phase_evict(&st) != k_ra8_ok) {
511 out->status = k_cache_store_demo_err_evict;
512 return k_ra8_err_invalid_state;
513 }
514 out->last_stage = k_cache_store_demo_stage_evicted;
515
516 /* Reuse: the sectors tile_b freed now hold tile_d. */
517 if (internal_demo_put_and_verify(&st,
518 k_demo_key_tile_d,
519 k_demo_seed_tile_d,
520 k_demo_len_tile_d,
521 cfg) != k_ra8_ok) {
522 out->status = k_cache_store_demo_err_reuse;
523 return k_ra8_err_invalid_state;
524 }
525 out->last_stage = k_cache_store_demo_stage_reused;
526
527 if (ra8_cache_store_close(&st) != k_ra8_ok) {
528 out->status = k_cache_store_demo_err_close;
529 return k_ra8_err_invalid_state;
530 }
531 out->last_stage = k_cache_store_demo_stage_closed;
532
533 if (internal_demo_phase_remount(cfg, out) != k_ra8_ok) {
534 return k_ra8_err_invalid_state; /* internal_demo_phase_remount set out->status */
535 }
536 out->last_stage = k_cache_store_demo_stage_persisted;
537 out->status = k_cache_store_demo_ok;
538 return k_ra8_ok;
539}
