4"""Gate: every first-party source file shall end in a trailing newline.
5
6POSIX defines a text line as ending in a newline; tools that read the last
7line (diff, cat, shell ``read``, parsers) behave better when the rule holds.
8``.editorconfig`` declares ``insert_final_newline`` but only editors honour
9it, and ``.clang-format``'s ``InsertNewlineAtEOF`` only covers C/C++ that
10reaches the formatter. Neither is a gate, and neither touches Python, shell,
11CMake, or YAML.
12
13This repo-wide backstop covers every first-party source file (C/C++, Python,
14shell, CMake, Make, YAML, linker scripts) and fails if any is non-empty and
15does not end in a newline byte. The whole-tree set is DERIVED from
16``git ls-files`` via ``lint_targets.first_party_paths`` rather than a hardcoded
17root list, so a new top-level directory is covered the day it lands; a hardcoded
18list -- which this used to carry, omitting ``docs/``, ``mk/``, ``infra/`` and
19``coprocessor/`` (#549) -- does not fail when it goes stale, it just reports
20success over a shrinking slice. Vendor trees and generated font tables are
21excluded. There is no grandfathering.
22
23Run::
24
25 check_final_newline.py # scan the whole tree
26 check_final_newline.py path/to/file ... # scan listed files
27 check_final_newline.py --selftest # prove both directions
28
29Exit 0 if every file ends in a newline, exit 1 (with a list) otherwise, exit 2
30when the whole-tree sweep collapses below FILE_FLOOR.
31"""

79# source files (the derived scope, up from 2843 when it was a hardcoded root

84def _ends_in_newline(path: Path) -> bool:
85 """Return True if `path` is empty or ends in a newline byte."""
86 try:
87 data = path.read_bytes()
88 except OSError:
89 return True # unreadable: not this gate's problem
90 return (not data) or data.endswith(b"\n")

93def _is_excluded(path: Path) -> bool:
94 return is_build_output_path(path) or any(frag in str(path) for frag in EXCLUDE_FRAGMENTS)

97def _is_source(path: Path) -> bool:
98 return path.suffix in SOURCE_SUFFIXES or path.name in SOURCE_NAMES

101def _rel(path: Path) -> str:
102 if path.is_relative_to(REPO_ROOT):
103 return str(path.relative_to(REPO_ROOT))
104 return str(path)

107def _derived_sources() -> list[Path]:
108 """Every first-party source file, derived from git rather than a root list.
109
110 Enumeration goes through ``lint_targets.first_party_paths`` -- the shared
111 derived-scope primitive -- so a newly added top-level directory (``docs/``,
112 ``mk/``, ``infra/`` and ``coprocessor/`` were the ones a hardcoded root
113 list had silently omitted, #549) is in scope the day it lands. The suffix
114 set is the sole language filter; ``SOURCE_NAMES`` catches the
115 extensionless-by-convention listfiles (``Makefile``, ``CMakeLists.txt``).
116 """
117 rels = set(first_party_paths(SOURCE_SUFFIXES))
118 for name in SOURCE_NAMES:
119 rels |= {rel for rel in first_party_paths((name,)) if Path(rel).name == name}
120 return [REPO_ROOT / rel for rel in sorted(rels)]

123def _enumerate_targets(arg_paths: Iterable[str]) -> list[Path]:
124 args = list(arg_paths)
125 if args:
126 out: list[Path] = []
127 for raw in args:
128 path = Path(raw)
129 if not path.is_absolute():
130 path = REPO_ROOT / path
131 if path.is_dir():
132 out.extend(c for c in path.rglob("*") if c.is_file() and _is_source(c))
133 elif _is_source(path):
134 out.append(path)
135 return [p for p in out if not _is_excluded(p)]
136
137 return [p for p in _derived_sources() if not _is_excluded(p)]

140def selftest() -> int:
141 """Prove the detector fires on a missing newline and that the scope is real.
142
143 Both directions plus a scope probe: a non-empty file with no trailing
144 newline must FIRE, a newline-terminated file and an empty file must stay
145 QUIET, the derived whole-tree scope must clear ``FILE_FLOOR``, and it must
146 actually reach the roots a hardcoded list had dropped (``mk/``, ``infra/``)
147 -- a clean run over a scope that never sees those roots proves nothing.
148
149 Returns:
150 0 when every assertion held in both directions, 1 otherwise.
151 """
152 failures: list[str] = []
153 with tempfile.TemporaryDirectory() as tmp:
154 root = Path(tmp)
155 good = root / "good.py"
156 good.write_text("x = 1\n", encoding="utf-8")
157 bad = root / "bad.py"
158 bad.write_bytes(b"x = 1")
159 empty = root / "empty.py"
160 empty.write_bytes(b"")
161 expect(_ends_in_newline(good), "MUST NOT FIRE: a newline-terminated file", failures)
162 expect(_ends_in_newline(empty), "MUST NOT FIRE: an empty file", failures)
163 expect(not _ends_in_newline(bad), "MUST FIRE: a file with no trailing newline", failures)
164
165 scope = _enumerate_targets([])
166 rels = {str(p.relative_to(REPO_ROOT)) for p in scope if p.is_relative_to(REPO_ROOT)}
167 expect(
168 len(scope) >= FILE_FLOOR,
169 f"derived scope sees {len(scope)} file(s) (floor {FILE_FLOOR})",
170 failures,
171 )
172 for root_name in ("mk", "infra"):
173 expect(
174 any(rel.startswith(root_name + "/") for rel in rels),
175 f"the derived scope reaches {root_name}/ (previously omitted)",
176 failures,
177 )
178 return report(failures)

181def main(argv: list[str]) -> int:
182 """Fail any first-party source file that does not end in a newline.
183
184 Exists because the two existing mechanisms are not gates: .editorconfig's
185 ``insert_final_newline`` is honoured only by editors, and clang-format's
186 ``InsertNewlineAtEOF`` only reaches C/C++ that goes through the formatter,
187 leaving Python, shell, CMake and YAML unenforced.
188
189 An empty target set exits 0 rather than FATAL only when a file list was
190 passed on argv: the caller there is the pre-commit hook handing over a
191 staged file list, which legitimately filters to nothing when a commit
192 touches only excluded paths. The WHOLE-TREE sweep gets the opposite
193 treatment -- FILE_FLOOR, exit 2 -- because nothing about this tree can
194 legitimately reduce it to a handful of files, and a sweep that read almost
195 nothing reports a clean tree for exactly the wrong reason.
196
197 Returns 1 with each offending path listed, 0 when clean or when an argv
198 list filtered to nothing, 2 when the whole-tree sweep enumerated too few
199 files to trust.
200 """
201 if "--selftest" in argv[1:]:
202 return selftest()
203 paths = argv[1:]
204 targets = _enumerate_targets(paths)
205 if not paths and len(targets) < FILE_FLOOR:
206 print(
207 f"check_final_newline.py: FATAL -- only {len(targets)} file(s) in scope, "
208 f"floor is {FILE_FLOOR}. A collapsed sweep reports a clean tree because "
209 "it scanned nothing.",
210 file=sys.stderr,
211 )
212 return 2
213 if not targets:
214 print("check_final_newline.py: no files to scan", file=sys.stderr)
215 return 0
216
217 missing = sorted(_rel(p) for p in targets if not _ends_in_newline(p))
218 if not missing:
219 print(f"check_final_newline.py: {len(targets)} file(s) scanned, all end in a newline.")
220 return 0
221
222 print(
223 f"check_final_newline.py: {len(missing)} file(s) missing a trailing newline:\n",
224 file=sys.stderr,
225 )
226 for path in missing:
227 print(f" {path}", file=sys.stderr)
228 print("\nAdd a single newline at end of file.", file=sys.stderr)
229 return 1
