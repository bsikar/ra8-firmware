4"""Gate: every first-party source file shall end in a trailing newline.
5
6POSIX defines a text line as ending in a newline; tools that read the last
7line (diff, cat, shell ``read``, parsers) behave better when the rule holds.
8``.editorconfig`` declares ``insert_final_newline`` but only editors honour
9it, and ``.clang-format``'s ``InsertNewlineAtEOF`` only covers C/C++ that
10reaches the formatter. Neither is a gate, and neither touches Python, shell,
11CMake, or YAML.
12
13This repo-wide backstop walks every first-party source file (C/C++, Python,
14shell, CMake, YAML, linker scripts) under the scanned roots and fails if any
15is non-empty and does not end in a newline byte. Vendor trees and generated
16font tables are excluded. There is no grandfathering.
17
18Run::
19
20 check_final_newline.py # scan the whole tree
21 check_final_newline.py path/to/file ... # scan listed files
22
23Exit 0 if every file ends in a newline, exit 1 (with a list) otherwise, exit 2
24when the whole-tree sweep collapses below FILE_FLOOR.
25"""
26

75def _ends_in_newline(path: Path) -> bool:
76 """Return True if `path` is empty or ends in a newline byte."""
77 try:
78 data = path.read_bytes()
79 except OSError:
80 return True # unreadable: not this gate's problem
81 return (not data) or data.endswith(b"\n")

84def _is_excluded(path: Path) -> bool:
85 return is_build_output_path(path) or any(frag in str(path) for frag in EXCLUDE_FRAGMENTS)

88def _is_source(path: Path) -> bool:
89 return path.suffix in SOURCE_SUFFIXES or path.name in SOURCE_NAMES

92def _rel(path: Path) -> str:
93 if path.is_relative_to(REPO_ROOT):
94 return str(path.relative_to(REPO_ROOT))

98def _enumerate_targets(arg_paths: Iterable[str]) -> list[Path]:
99 args = list(arg_paths)
100 if args:
101 out: list[Path] = []
102 for raw in args:
103 path = Path(raw)
104 if not path.is_absolute():
105 path = REPO_ROOT / path
106 if path.is_dir():
107 out.extend(c for c in path.rglob("*") if c.is_file() and _is_source(c))
108 elif _is_source(path):
109 out.append(path)
110 return [p for p in out if not _is_excluded(p)]
111
112 out = []
113 for root in SCAN_ROOTS:
114 base = REPO_ROOT / root
115 if base.is_dir():
116 out.extend(c for c in base.rglob("*") if c.is_file() and _is_source(c))
117 return [p for p in out if not _is_excluded(p)]

120def main(argv: list[str]) -> int:
121 """Fail any first-party source file that does not end in a newline.
122
123 Exists because the two existing mechanisms are not gates: .editorconfig's
124 ``insert_final_newline`` is honoured only by editors, and clang-format's
125 ``InsertNewlineAtEOF`` only reaches C/C++ that goes through the formatter,
126 leaving Python, shell, CMake and YAML unenforced.
127
128 An empty target set exits 0 rather than FATAL only when a file list was
129 passed on argv: the caller there is the pre-commit hook handing over a
130 staged file list, which legitimately filters to nothing when a commit
131 touches only excluded paths. The WHOLE-TREE sweep gets the opposite
132 treatment -- FILE_FLOOR, exit 2 -- because nothing about this tree can
133 legitimately reduce it to a handful of files, and a sweep that read almost
134 nothing reports a clean tree for exactly the wrong reason.
135
136 Returns 1 with each offending path listed, 0 when clean or when an argv
137 list filtered to nothing, 2 when the whole-tree sweep enumerated too few
138 files to trust.
139 """
140 paths = argv[1:]
141 targets = _enumerate_targets(paths)
142 if not paths and len(targets) < FILE_FLOOR:
143 print(
144 f"check_final_newline.py: FATAL -- only {len(targets)} file(s) in scope, "
145 f"floor is {FILE_FLOOR}. A collapsed sweep reports a clean tree because "
146 "it scanned nothing.",
147 file=sys.stderr,
148 )
149 return 2
150 if not targets:
151 print("check_final_newline.py: no files to scan", file=sys.stderr)
152 return 0
153
154 missing = sorted(_rel(p) for p in targets if not _ends_in_newline(p))
155 if not missing:
156 print(f"check_final_newline.py: {len(targets)} file(s) scanned, all end in a newline.")
157 return 0
158
159 print(
160 f"check_final_newline.py: {len(missing)} file(s) missing a trailing newline:\n",
161 file=sys.stderr,
162 )
163 for path in missing:
164 print(f" {path}", file=sys.stderr)
165 print("\nAdd a single newline at end of file.", file=sys.stderr)
166 return 1
