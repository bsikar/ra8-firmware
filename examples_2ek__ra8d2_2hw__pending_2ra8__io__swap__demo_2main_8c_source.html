319static ra8_err_t swap_run_one(const swap_backend_t* b, uint32_t len, ra8_io_stream_t* log)
320{
321 RA8_CHECK_NULL_PTR(b, s_tag, "backend");
322 RA8_CHECK_NULL_PTR(log, s_tag, "log");

324 (void)ra8_io_stream_puts(log, "swap[");
325 (void)ra8_io_stream_puts(log, b->name);
326 (void)ra8_io_stream_puts(log, "]: mount ");
327 (void)ra8_io_stream_puts(log, b->label);
328 RA8_RETURN_ON_ERROR(ra8_io_blockdev_as_fs_backend(b->bd, &s_be), s_tag, "fs bridge");

331 opts.label = b->label;
332 RA8_RETURN_ON_ERROR(ra8_fs_format(&s_be, &opts), s_tag, "format");
333 ra8_fs_mount_t* mnt = nullptr;
334 RA8_RETURN_ON_ERROR(ra8_fs_mount(&s_be, &mnt), s_tag, "mount");
335 RA8_RETURN_ON_ERROR(ra8_io_vfs_mount(b->name, mnt), s_tag, "vfs mount");
336
337 swap_fill(len);
338 RA8_RETURN_ON_ERROR(swap_vfs_write(b->file, len), s_tag, "write");
339 RA8_RETURN_ON_ERROR(swap_vfs_read_verify(b->file, len), s_tag, "verify");
340
341 RA8_RETURN_ON_ERROR(ra8_io_vfs_unmount(b->name), s_tag, "vfs unmount");
342 RA8_RETURN_ON_ERROR(ra8_fs_unmount(mnt), s_tag, "unmount");
343 (void)ra8_io_stream_puts(log, " -> ");
344 (void)ra8_io_stream_put_u32(log, len);
345 (void)ra8_io_stream_puts(log, " bytes ok\r\n");
346 return k_ra8_ok;

409static ra8_err_t swap_run_all(const char** out_failed)
410{
411 RA8_CHECK_NULL_PTR(out_failed, s_tag, "out_failed");
412 *out_failed = nullptr;
413
414 RA8_RETURN_ON_ERROR(ra8_io_blockdev_ram_init(&s_ram_bd,
415 &s_ram_state,
416 s_ram_disk,
417 (uint32_t)k_swap_ram_blocks,
418 false),
419 s_tag,
420 "ram bind");
421 RA8_RETURN_ON_ERROR(ra8_xspi_init((uint8_t)k_swap_xspi_inst, k_ra8_xspi_lio_1s1s1s),
422 s_tag,
423 "xspi init");
424 RA8_RETURN_ON_ERROR(ra8_io_blockdev_xspi_init(&s_xspi_bd,
425 &s_xspi_state,
426 (uint8_t)k_swap_xspi_inst,
427 (uint32_t)k_swap_xspi_base,
428 (uint32_t)k_swap_xspi_blocks,
429 false),
430 s_tag,
431 "xspi bind");
432
433 const swap_backend_t table[k_swap_backends] = {
434 {.bd = &s_ram_bd, .name = "ram", .label = "RAIORAM", .file = "ram:/DATA.BIN"},
435 {.bd = &s_xspi_bd, .name = "xs", .label = "RAIOXS", .file = "xs:/DATA.BIN"},
436 };
437
438 for (uint32_t i = 0U; i < (uint32_t)k_swap_backends; ++i) {
439 const ra8_err_t e = swap_run_one(&table[i], (uint32_t)k_swap_payload, &s_ram_log);
440 if (e != k_ra8_ok) {
441 *out_failed = table[i].name;
442 return e;
443 }
444 }
445 return k_ra8_ok;
446}
