61bool ra8_unarch_tar_probe(const uint8_t* block, size_t len)
62{
63 if (block == nullptr) {
64 return false;
65 }
66 if (len < (size_t)k_ra8_unarch_tar_block) {
67 return false;
68 }
69 if (!ra8_unarch_tar_magic_ok(block)) {
70 return false;
71 }
72 return ra8_unarch_tar_checksum_ok(block);

75ra8_err_t ra8_unarch_tar_open(ra8_unarch_tar_t* t,
76 ra8_unarch_read_fn read,
77 void* ctx,
78 uint64_t size,
79 const ra8_decomp_limits_t* limits)
80{
81 RA8_CHECK_NULL_PTR(t, s_tag_tar, "open: null t");
82 *t = (ra8_unarch_tar_t){};
83 RA8_CHECK_NULL_PTR((const void*)read, s_tag_tar, "open: null read");
84 if (size < (uint64_t)k_ra8_unarch_tar_block) {
85 return k_ra8_err_invalid_size;
86 }
87 const ra8_err_t berr = ra8_decomp_budget_init(&t->budget, limits);
88 if (berr != k_ra8_ok) {
89 return berr;
90 }
91 uint8_t block[k_ra8_unarch_tar_block] = {};
92 const size_t got = read(ctx, 0U, block, sizeof(block));
93 if (got != sizeof(block)) {
94 return k_ra8_err_invalid_size;
95 }
96 if (!ra8_unarch_tar_probe(block, sizeof(block))) {
97 return k_ra8_err_not_supported;
98 }
99 t->read = read;
100 t->ctx = ctx;
101 t->size = size;
102 t->live = true;
103 return k_ra8_ok;
104}

125{
126 const uint64_t need = (uint64_t)k_ra8_unarch_tar_block;
127 if (off > (t->size - need)) {
128 return k_ra8_err_validation_failed; /* partial trailing block */
129 }
130 const size_t got = t->read(t->ctx, off, block, (size_t)need);
131 if ((uint64_t)got != need) {
132 return k_ra8_err_validation_failed; /* backing truncated */
133 }

155RA8_INTERNAL
156static ra8_err_t s_next_off(uint64_t off, uint64_t dsize, uint64_t* next)
157{
158 const uint64_t blk = (uint64_t)k_ra8_unarch_tar_block;
159 if (dsize > (UINT64_MAX - (blk - 1U))) {
160 return k_ra8_err_validation_failed; /* rounding would wrap */
161 }
162 const uint64_t blocks = (dsize + (blk - 1U)) / blk;
163 if (blocks > ((UINT64_MAX / blk) - 1U)) {
164 return k_ra8_err_validation_failed; /* span would wrap */
165 }
166 const uint64_t span = (blocks + 1U) * blk;
167 if (off > (UINT64_MAX - span)) {
168 return k_ra8_err_validation_failed; /* next offset would wrap */
169 }
170 *next = off + span;
171 return k_ra8_ok;
172}

210RA8_INTERNAL
211static ra8_err_t s_meta_consume(const ra8_unarch_tar_t* t,
212 ra8_tar_type_t type,
213 uint64_t doff,
214 uint64_t dsize,
215 tar_meta_t* meta)
216{
217 if (dsize > (uint64_t)k_ra8_unarch_tar_pax_max) {
218 return k_ra8_err_validation_failed; /* meta flood: reject the member */
219 }
220 uint8_t* const data = s_tar_meta_scratch;
221 const size_t got = t->read(t->ctx, doff, data, (size_t)dsize);
222 if ((uint64_t)got != dsize) {
223 return k_ra8_err_validation_failed; /* backing truncated */
224 }
225 if (type == k_ra8_tar_type_pax) {
226 return ra8_unarch_tar_pax_parse(data,
227 (size_t)dsize,
228 meta->name_buf,
229 meta->name_cap,
230 &meta->name_len,
231 &meta->have_path,
232 &meta->size_ovr,
233 &meta->have_size);
234 }
235 /* GNU longname: the data is the member name, usually NUL-padded. */
236 size_t nlen = (size_t)dsize;
237 while (nlen > 0U) { /* bound: dsize */
238 if (data[nlen - 1U] != 0U) {
239 break;
240 }
241 nlen -= 1U;
242 }
243 const size_t want = (nlen > (size_t)meta->name_cap) ? (size_t)meta->name_cap : nlen;
244 if (want > 0U) {
245 (void)memcpy(meta->name_buf, data, want);
246 }
247 meta->name_len = (uint16_t)want;
248 meta->have_path = true;
249 return k_ra8_ok;
250}

267static void s_ustar_name(const uint8_t* block, tar_meta_t* meta)
268{
269 const void* nz = memchr(&block[k_ra8_tar_off_name], 0, (size_t)k_ra8_tar_len_name);
270 const size_t n_len = (nz != nullptr) ? (size_t)((const uint8_t*)nz - &block[k_ra8_tar_off_name])
271 : (size_t)k_ra8_tar_len_name;
272 const void* pz = memchr(&block[k_ra8_tar_off_prefix], 0, (size_t)k_ra8_tar_len_prefix);
273 const size_t p_len = (pz != nullptr) ? (size_t)((const uint8_t*)pz - &block[k_ra8_tar_off_prefix])
274 : (size_t)k_ra8_tar_len_prefix;
275 size_t used = 0U;
276 if (p_len > 0U) {
277 const size_t pw = (p_len > (size_t)meta->name_cap) ? (size_t)meta->name_cap : p_len;
278 if (pw > 0U) {
279 (void)memcpy(meta->name_buf, &block[k_ra8_tar_off_prefix], pw);
280 }
281 used = pw;
282 if (used < (size_t)meta->name_cap) {
283 meta->name_buf[used] = '/';
284 used += 1U;
285 }
286 }
287 const size_t room = (size_t)meta->name_cap - used;
288 const size_t nw = (n_len > room) ? room : n_len;
289 if (nw > 0U) {
290 (void)memcpy(&meta->name_buf[used], &block[k_ra8_tar_off_name], nw);
291 }
292 meta->name_len = (uint16_t)(used + nw);
293}

365static ra8_err_t s_finish_member(ra8_unarch_tar_t* t,
366 const uint8_t* block,
367 uint64_t off,
368 ra8_tar_type_t type,
369 uint64_t dsize,
370 tar_meta_t* meta,
371 ra8_unarch_tar_entry_t* out)
372{
373 uint64_t msize = dsize;
374 if (meta->have_size) {
375 msize = meta->size_ovr; /* pax size record overrides the header field */
376 }
377 uint64_t next = 0U;
378 const ra8_err_t aerr = s_next_off(off, msize, &next);
379 if (aerr != k_ra8_ok) {
380 return aerr;
381 }
382 const uint64_t doff = off + (uint64_t)k_ra8_unarch_tar_block;
383 if (msize > (t->size - doff)) {
384 return k_ra8_err_validation_failed; /* lying size: data overruns archive */
385 }
386 const ra8_err_t derr = ra8_decomp_check_declared(&t->budget.limits, msize, msize);
387 if (derr != k_ra8_ok) {
388 return derr;
389 }
390 const ra8_err_t eerr = ra8_decomp_budget_charge_entry(&t->budget);
391 if (eerr != k_ra8_ok) {
392 return eerr;
393 }
394 if (!meta->have_path) {
395 s_ustar_name(block, meta);
396 }
397 out->data_off = doff;
398 out->size = msize;
399 out->next_off = next;
400 out->name_len = meta->name_len;
401 out->is_file = (type == k_ra8_tar_type_file) ? 1U : 0U;
402 out->is_dir = (type == k_ra8_tar_type_dir) ? 1U : 0U;
403 return k_ra8_ok;
404}

406ra8_err_t ra8_unarch_tar_next(ra8_unarch_tar_t* t,
407 uint64_t off,
408 char* name_buf,
409 uint16_t name_cap,
410 ra8_unarch_tar_entry_t* out)
411{
412 RA8_CHECK_NULL_PTR(t, s_tag_tar, "next: null t");
413 RA8_CHECK_NULL_PTR(out, s_tag_tar, "next: null out");
414 *out = (ra8_unarch_tar_entry_t){};
415 if (!t->live) {
416 return k_ra8_err_invalid_state;
417 }
418 if (name_cap > 0U) {
419 RA8_CHECK_NULL_PTR(name_buf, s_tag_tar, "next: null name_buf");
420 }
421 tar_meta_t meta = {.name_buf = name_buf, .name_cap = name_cap};
422 uint64_t cur = off;
423 for (uint32_t m = 0U; m <= (uint32_t)k_ra8_unarch_tar_meta_max; ++m) { /* bound: meta cap */
424 const ra8_err_t ierr = ra8_decomp_budget_charge_iter(&t->budget);
425 if (ierr != k_ra8_ok) {
426 return ierr;
427 }
428 if (cur >= t->size) {
429 return k_ra8_err_not_found; /* clean EOF without an end marker */
430 }
431 uint8_t block[k_ra8_unarch_tar_block] = {};
432 const ra8_err_t rerr = s_read_block(t, cur, block);
433 if (rerr != k_ra8_ok) {
434 return rerr;
435 }
436 if (ra8_unarch_tar_block_zero(block)) {
437 return k_ra8_err_not_found; /* end-of-archive marker */
438 }
439 ra8_tar_type_t type = k_ra8_tar_type_other;
440 uint64_t dsize = 0U;
441 uint64_t next = 0U;
442 const ra8_err_t herr = s_decode_header(block, cur, &type, &dsize, &next);
443 if (herr != k_ra8_ok) {
444 return herr;
445 }
446 if ((type == k_ra8_tar_type_pax) || (type == k_ra8_tar_type_longname)) {
447 const uint64_t doff = cur + (uint64_t)k_ra8_unarch_tar_block;
448 if (dsize > (t->size - doff)) {
449 return k_ra8_err_validation_failed; /* meta data overruns archive */
450 }
451 const ra8_err_t merr = s_meta_consume(t, type, doff, dsize, &meta);
452 if (merr != k_ra8_ok) {
453 return merr;
454 }
455 cur = next;
456 continue;
457 }
458 if (type == k_ra8_tar_type_meta) {
459 cur = next; /* skipped meta ('g' global, 'K' longlink) */
460 continue;
461 }
462 const ra8_err_t ferr = s_finish_member(t, block, cur, type, dsize, &meta, out);
463 if (ferr != k_ra8_ok) {
464 *out = (ra8_unarch_tar_entry_t){};
465 return ferr;
466 }
467 return k_ra8_ok;
468 }
469 return k_ra8_err_validation_failed; /* meta prelude longer than the cap */
470}

472ra8_err_t ra8_unarch_tar_read(const ra8_unarch_tar_t* t,
473 const ra8_unarch_tar_entry_t* ent,
474 uint8_t* buf,
475 size_t cap,
476 size_t* got)
477{
478 RA8_CHECK_NULL_PTR(t, s_tag_tar, "read: null t");
479 RA8_CHECK_NULL_PTR(ent, s_tag_tar, "read: null ent");
480 RA8_CHECK_NULL_PTR(buf, s_tag_tar, "read: null buf");
481 RA8_CHECK_NULL_PTR(got, s_tag_tar, "read: null got");
482 *got = 0U;
483 if (!t->live) {
484 return k_ra8_err_invalid_state;
485 }
486 if (ent->is_file == 0U) {
487 return k_ra8_err_not_supported;
488 }
489 if ((uint64_t)cap < ent->size) {
490 return k_ra8_err_no_mem;
491 }
492 if (ent->data_off > t->size) {
493 return k_ra8_err_invalid_size; /* forged entry: data area outside archive */
494 }
495 if (ent->size > (t->size - ent->data_off)) {
496 return k_ra8_err_invalid_size; /* forged entry: data area overruns archive */
497 }
498 if (ent->size == 0U) {
499 return k_ra8_ok; /* empty member: nothing to fetch */
500 }
501 const size_t n = t->read(t->ctx, ent->data_off, buf, (size_t)ent->size);
502 if ((uint64_t)n != ent->size) {
503 return k_ra8_err_invalid_size; /* backing truncated */
504 }
505 *got = n;
506 return k_ra8_ok;
507}
