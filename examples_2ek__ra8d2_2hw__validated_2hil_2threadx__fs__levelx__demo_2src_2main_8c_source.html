298static void demo_read_test_file(void)
299{
300 ra8_fs_file_t* file = nullptr;
301 if (ra8_fs_open(s_fs_mount, s_demo_file_path, k_ra8_fs_mode_read, &file) != k_ra8_ok) {
302 demo_print("[fslx] ra8_fs_open(read) failed\r\n");
303 return;
304 }
305 demo_print("[fslx] readback: ");
306 uint8_t buf[k_demo_file_chunk];
307 uint32_t got = 0U;
308 do {
309 got = 0U;
310 if (ra8_fs_read(file, buf, (uint32_t)k_demo_file_chunk, &got) != k_ra8_ok) {
311 break;
312 }
313 if (got > 0U) {
314 (void)ra8_board_uart_console_write(buf, (size_t)got);
315 }
316 } while (got > 0U);
317 demo_print("\r\n");
318 (void)ra8_fs_close(file);
319}

331static void demo_thread_entry(ULONG thread_input)
332{
333 (void)thread_input;
334
335 demo_print("[fslx] booting xSPI flash\r\n");
336 /* The LevelX NOR driver owns OCTA bus bring-up: lx_nor_flash_format ->
337 * lx_nor_driver_ra8_xspi_initialize -> priv_bus_init_once routes the
338 * pins, runs the 8D/1S software-reset recovery, calls ra8_xspi_init,
339 * and probes RDID exactly once. Doing it here as well double-routes
340 * the PFS pins (the validator rejects the second route with
341 * k_ra8_err_gpio_conflict), so the driver's initialize bails before
342 * wiring the sector buffer and format returns LX_NO_MEMORY. Let the
343 * driver be the single owner. */
344 demo_print("[fslx] formatting + opening LevelX partition\r\n");
345 demo_lx_open_or_panic();
346
347 demo_print("[fslx] formatting + mounting FAT volume on LevelX\r\n");
348 demo_fs_format_or_panic();
349
350 demo_write_test_file();
351 demo_read_test_file();
352
353 (void)ra8_fs_unmount(s_fs_mount);
354 (void)lx_nor_flash_close(&s_nor_flash);
355 demo_print("[fslx] done\r\n");
356}
