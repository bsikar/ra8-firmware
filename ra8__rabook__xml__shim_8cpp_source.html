107const XMLElement* s_find_body(const XMLDocument& doc)
108{
109 const XMLElement* root = doc.RootElement();
110 if (root == nullptr) {
111 return nullptr;
112 }
113
114 uint16_t tries = 0U;
115 for (const XMLElement* e = root->FirstChildElement();
116 e != nullptr && tries < (uint16_t)k_xhtml_max_siblings;
117 e = e->NextSiblingElement(), ++tries) {
118 if (std::strcmp(e->Name(), "body") == 0) {
119 return e;
120 }
121 }
122 return root;
123}

143uint16_t s_collect_attrs(const XMLElement* elem, ra8_rabook_ctx_t* ctx, ra8_book_attr_t* out)
144{
145 uint16_t count = 0U;
146 const XMLAttribute* a = elem->FirstAttribute();
147 for (; a != nullptr && count < (uint16_t)k_xhtml_max_attrs; a = a->Next(), ++count) {
148 out[count].name_off = ra8_rabook_intern(ctx, a->Name());
149 out[count].value_off = ra8_rabook_intern(ctx, a->Value());
150 }
151 if (a != nullptr) {
152 /* More than k_xhtml_max_attrs attributes on one element: refuse to silently
153 * drop the overflow -- latch the builder failure so finalize reports it. */
154 ra8_log_error(s_tag, "element exceeds k_xhtml_max_attrs; refusing to truncate");
155 ctx->failed = true;
156 }
157 return count;
158}

179uint32_t s_emit_node(ra8_rabook_ctx_t* ctx, const XMLNode* node)
180{
181 const XMLElement* elem = node->ToElement();
182 if (elem != nullptr) {
183 ra8_book_attr_t attrs[k_xhtml_max_attrs] = {};
184 const uint16_t ac = s_collect_attrs(elem, ctx, attrs);
185 return ra8_rabook_add_element(ctx, ra8_rabook_intern(ctx, elem->Name()), attrs, ac);
186 }
187
188 const XMLText* text = node->ToText();
189 if (text != nullptr) {
190 /* tinyxml2 models CDATA as an XMLText with CData()==true; skip it like a
191 * comment/PI to match the desktop DomBuilder (CDATA is dropped). */
192 if (text->CData()) {
193 return k_ra8_book_nil;
194 }
195 const char* val = text->Value();
196 // mcdc-deactivated: DO-178C 6.4.4.3 -- both conditions are invariantly true on any parsed text node, so the decision has no false outcome to pair against. tinyxml2's XMLText::Value() never returns nullptr (it returns "" at worst), so `val != nullptr` cannot flip; and a text node reached here always carries at least one character -- an empty run produces no XMLText, and PEDANTIC_WHITESPACE preserves even whitespace-only runs -- so `val[0] != '\0'` cannot flip either. Both are defensive guards against a future tinyxml2 contract change.
197 if (val != nullptr && val[0] != '\0') {
198 return ra8_rabook_add_text(ctx, ra8_rabook_intern(ctx, val));
199 }
200 }
201 return k_ra8_book_nil;
202}

222uint16_t s_push_frame(ra8_xhtml_frame_t* stack,
223 uint16_t top,
224 const XMLNode* node,
225 uint32_t parent_idx,
226 uint32_t prev_sib_idx)
227{
228 // mcdc-deactivated: DO-178C 6.4.4.3 -- C2 (top < k_xhtml_max_stack) is invariantly true: this DFS never pushes a nil frame (see the node != nullptr guard), so the stack depth is bounded by 2 * tree_depth, and tinyxml2 rejects any document deeper than its element-depth cap (100) before the walk begins. 100 * 2 < k_xhtml_max_stack (512), so no document tinyxml2 accepts can fill the stack -- C2 is the NASA Rule 2 safety valve and cannot flip on any reachable input. C1 (node != nullptr) is the live condition and both its arms are covered.
229 if (node != nullptr && top < (uint16_t)k_xhtml_max_stack) {
230 stack[top] = {node, parent_idx, prev_sib_idx};
231 return (uint16_t)(top + 1U);
232 }
233 return top;
234}

253void s_walk_body_subtree(ra8_rabook_ctx_t* ctx, const XMLElement* body, uint32_t chapter_root)
254{
255 ra8_xhtml_frame_t stack[k_xhtml_max_stack] = {};
256 uint16_t top = 0U;
257 top = s_push_frame(stack, top, body->FirstChild(), chapter_root, k_ra8_book_nil);
258
259 while (top > 0U) {
260 const ra8_xhtml_frame_t frame = stack[--top];
261 const uint32_t new_idx = s_emit_node(ctx, frame.node);
262
263 if (new_idx != k_ra8_book_nil) {
264 /* Link into the parent's child list or the preceding sibling chain. */
265 if (frame.prev_sib_idx == k_ra8_book_nil) {
266 (void)ra8_rabook_link_child(ctx, frame.parent_idx, new_idx);
267 } else {
268 (void)ra8_rabook_link_sibling(ctx, frame.prev_sib_idx, new_idx);
269 }
270 }
271
272 /* The effective "previous sibling" for the next sibling frame. */
273 const uint32_t sib_prev = (new_idx != k_ra8_book_nil) ? new_idx : frame.prev_sib_idx;
274
275 /* Push next_sibling BEFORE first_child (LIFO: first_child popped first). */
276 top = s_push_frame(stack, top, frame.node->NextSibling(), frame.parent_idx, sib_prev);
277
278 const XMLElement* elem = frame.node->ToElement();
279 if (elem != nullptr && new_idx != k_ra8_book_nil) {
280 top = s_push_frame(stack, top, elem->FirstChild(), new_idx, k_ra8_book_nil);
281 }
282 }
283}

324ra8_err_t ra8_rabook_xml_parse_chapter(const uint8_t* xhtml_bytes,
325 size_t xhtml_len,
326 ra8_rabook_ctx_t* ctx,
327 const char* chapter_href,
328 const char* chapter_title)
329{
330 const ra8_err_t arg_err = s_check_parse_args(xhtml_bytes, ctx, chapter_href, chapter_title);
331 if (arg_err != k_ra8_ok) {
332 return arg_err;
333 }
334
335 /* PEDANTIC_WHITESPACE (#151): keep inter-element whitespace text nodes so the
336 * chapter DOM is byte-identical to the desktop reference DomBuilder, which
337 * preserves every text run. This retains significant inline whitespace such as
338 * the space between adjacent <abbr> elements ("<abbr>x</abbr> <abbr>y</abbr>"),
339 * which the default PRESERVE_WHITESPACE mode drops -- merging the words. Entity
340 * processing stays on (processEntities=true). Only chapter content opts in;
341 * ra8_epub keeps the default mode for OPF / container / TOC parsing. */
342 tinyxml2::XMLDocument doc(true, tinyxml2::PEDANTIC_WHITESPACE);
343 if (doc.Parse(reinterpret_cast<const char*>(xhtml_bytes), xhtml_len) != tinyxml2::XML_SUCCESS) {
344 ra8_log_error(s_tag, "tinyxml2 parse failed");
345 return k_ra8_err_no_mem;
346 }
347
348 const tinyxml2::XMLElement* body = s_find_body(doc);
349 if (body == nullptr) {
350 ra8_log_error(s_tag, "no root element in XHTML");
351 return k_ra8_err_no_mem;
352 }
353
354 /* Add the <body> element as the chapter root, then walk its subtree. */
355 ra8_book_attr_t body_attrs[k_xhtml_max_attrs] = {};
356 const uint16_t body_ac = s_collect_attrs(body, ctx, body_attrs);
357 const uint32_t chapter_root =
358 ra8_rabook_add_element(ctx, ra8_rabook_intern(ctx, body->Name()), body_attrs, body_ac);
359
360 s_walk_body_subtree(ctx, body, chapter_root);
361
362 const uint32_t title_off = ra8_rabook_intern(ctx, chapter_title);
363 const uint32_t href_off = ra8_rabook_intern(ctx, chapter_href);
364 const uint32_t chapter_idx = ra8_rabook_add_chapter(ctx, title_off, href_off, chapter_root);
365 return (chapter_idx != k_ra8_book_nil) ? k_ra8_ok : k_ra8_err_no_mem;
366}
