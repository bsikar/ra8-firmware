74/* The frame stack must cover every document tinyxml2 hands us: a full stack
75 * drops a frame, and a dropped frame silently truncates a whole subtree of the
76 * chapter. Bind the capacity to the vendored cap so a tinyxml2 re-vendor that
77 * raises TINYXML2_MAX_ELEMENT_DEPTH fails the BUILD rather than quietly
78 * shipping a parser that truncates deep chapters (#625). */
79static_assert(TINYXML2_MAX_ELEMENT_DEPTH - 2 <= (int)k_xhtml_max_stack,
80 "k_xhtml_max_stack must cover tinyxml2's element-depth cap");

126const XMLElement* s_find_body(const XMLDocument& doc)
127{
128 const XMLElement* root = doc.RootElement();
129 if (root == nullptr) {
130 return nullptr;
131 }
132
133 uint16_t tries = 0U;
134 for (const XMLElement* e = root->FirstChildElement();
135 e != nullptr && tries < (uint16_t)k_xhtml_max_siblings;
136 e = e->NextSiblingElement(), ++tries) {
137 if (std::strcmp(e->Name(), "body") == 0) {
138 return e;
139 }
140 }
141 return root;
142}

162uint16_t s_collect_attrs(const XMLElement* elem, ra8_rabook_ctx_t* ctx, ra8_book_attr_t* out)
163{
164 uint16_t count = 0U;
165 const XMLAttribute* a = elem->FirstAttribute();
166 for (; a != nullptr && count < (uint16_t)k_xhtml_max_attrs; a = a->Next(), ++count) {
167 out[count].name_off = ra8_rabook_intern(ctx, a->Name());
168 out[count].value_off = ra8_rabook_intern(ctx, a->Value());
169 }
170 if (a != nullptr) {
171 /* More than k_xhtml_max_attrs attributes on one element: refuse to silently
172 * drop the overflow -- latch the builder failure so finalize reports it. */
173 ra8_log_error(s_tag, "element exceeds k_xhtml_max_attrs; refusing to truncate");
174 ctx->failed = true;
175 }
176 return count;
177}

198uint32_t s_emit_node(ra8_rabook_ctx_t* ctx, const XMLNode* node)
199{
200 const XMLElement* elem = node->ToElement();
201 if (elem != nullptr) {
202 ra8_book_attr_t attrs[k_xhtml_max_attrs] = {};
203 const uint16_t ac = s_collect_attrs(elem, ctx, attrs);
204 return ra8_rabook_add_element(ctx, ra8_rabook_intern(ctx, elem->Name()), attrs, ac);
205 }
206
207 const XMLText* text = node->ToText();
208 if (text != nullptr) {
209 /* tinyxml2 models CDATA as an XMLText with CData()==true; skip it like a
210 * comment/PI to match the desktop DomBuilder (CDATA is dropped). */
211 if (text->CData()) {
212 return k_ra8_book_nil;
213 }
214 const char* val = text->Value();
215 // mcdc-deactivated: DO-178C 6.4.4.3 -- both conditions are invariantly true on any parsed text node, so the decision has no false outcome to pair against. tinyxml2's XMLText::Value() never returns nullptr (it returns "" at worst), so `val != nullptr` cannot flip; and a text node reached here always carries at least one character -- an empty run produces no XMLText, and PEDANTIC_WHITESPACE preserves even whitespace-only runs -- so `val[0] != '\0'` cannot flip either. Both are defensive guards against a future tinyxml2 contract change.
216 if (val != nullptr && val[0] != '\0') {
217 return ra8_rabook_add_text(ctx, ra8_rabook_intern(ctx, val));
218 }
219 }
220 return k_ra8_book_nil;
221}

246uint16_t s_push_frame(ra8_xhtml_frame_t* stack,
247 uint16_t top,
248 const XMLNode* node,
249 uint32_t parent_idx,
250 uint32_t prev_sib_idx)
251{
252 // mcdc-deactivated: DO-178C 6.4.4.3 -- C2 (top < k_xhtml_max_stack) is invariantly true. Each turn of the walk pops one frame and pushes at most two, so the stack grows only where the DFS descends into a first_child whose parent also had a next_sibling; a sibling step is net zero. The high-water mark is therefore the node-depth of the deepest node measured from `<body>` (depth + 1), NOT 2 * depth. tinyxml2 aborts the parse once nesting reaches TINYXML2_MAX_ELEMENT_DEPTH -- 500 in the vendored 11.0.0 snapshot, not the 100 this rationale claimed before #625 -- so the deepest node it admits is at absolute depth 499 and `<body>` occupies absolute depth >= 1, bounding top at 499 - 1 = 498 < k_xhtml_max_stack (512). The bound is measured, not asserted: test_deep_at_tinyxml2_cap compiles the deepest document tinyxml2 accepts in the shape that maximises the stack (top == 498) and checks all 996 nodes are emitted, and test_deep_beyond_tinyxml2_cap confirms one level deeper is rejected by the parser, so the shim reports an error instead of truncating. The static_assert on ra8_xhtml_limits_t binds this premise to the vendored constant, failing the build if a re-vendor raises the cap. C1 (node != nullptr) is the live condition and both its arms are covered.
253 if (node != nullptr && top < (uint16_t)k_xhtml_max_stack) {
254 stack[top] = {node, parent_idx, prev_sib_idx};
255 return (uint16_t)(top + 1U);
256 }
257 return top;
258}

277void s_walk_body_subtree(ra8_rabook_ctx_t* ctx, const XMLElement* body, uint32_t chapter_root)
278{
279 ra8_xhtml_frame_t stack[k_xhtml_max_stack] = {};
280 uint16_t top = 0U;
281 top = s_push_frame(stack, top, body->FirstChild(), chapter_root, k_ra8_book_nil);
282
283 while (top > 0U) {
284 const ra8_xhtml_frame_t frame = stack[--top];
285 const uint32_t new_idx = s_emit_node(ctx, frame.node);
286
287 if (new_idx != k_ra8_book_nil) {
288 /* Link into the parent's child list or the preceding sibling chain. */
289 if (frame.prev_sib_idx == k_ra8_book_nil) {
290 (void)ra8_rabook_link_child(ctx, frame.parent_idx, new_idx);
291 } else {
292 (void)ra8_rabook_link_sibling(ctx, frame.prev_sib_idx, new_idx);
293 }
294 }
295
296 /* The effective "previous sibling" for the next sibling frame. */
297 const uint32_t sib_prev = (new_idx != k_ra8_book_nil) ? new_idx : frame.prev_sib_idx;
298
299 /* Push next_sibling BEFORE first_child (LIFO: first_child popped first). */
300 top = s_push_frame(stack, top, frame.node->NextSibling(), frame.parent_idx, sib_prev);
301
302 const XMLElement* elem = frame.node->ToElement();
303 if (elem != nullptr && new_idx != k_ra8_book_nil) {
304 top = s_push_frame(stack, top, elem->FirstChild(), new_idx, k_ra8_book_nil);
305 }
306 }
307}

348ra8_err_t ra8_rabook_xml_parse_chapter(const uint8_t* xhtml_bytes,
349 size_t xhtml_len,
350 ra8_rabook_ctx_t* ctx,
351 const char* chapter_href,
352 const char* chapter_title)
353{
354 const ra8_err_t arg_err = s_check_parse_args(xhtml_bytes, ctx, chapter_href, chapter_title);
355 if (arg_err != k_ra8_ok) {
356 return arg_err;
357 }
358
359 /* PEDANTIC_WHITESPACE (#151): keep inter-element whitespace text nodes so the
360 * chapter DOM is byte-identical to the desktop reference DomBuilder, which
361 * preserves every text run. This retains significant inline whitespace such as
362 * the space between adjacent <abbr> elements ("<abbr>x</abbr> <abbr>y</abbr>"),
363 * which the default PRESERVE_WHITESPACE mode drops -- merging the words. Entity
364 * processing stays on (processEntities=true). Only chapter content opts in;
365 * ra8_epub keeps the default mode for OPF / container / TOC parsing. */
366 tinyxml2::XMLDocument doc(true, tinyxml2::PEDANTIC_WHITESPACE);
367 if (doc.Parse(reinterpret_cast<const char*>(xhtml_bytes), xhtml_len) != tinyxml2::XML_SUCCESS) {
368 ra8_log_error(s_tag, "tinyxml2 parse failed");
369 return k_ra8_err_no_mem;
370 }
371
372 const tinyxml2::XMLElement* body = s_find_body(doc);
373 if (body == nullptr) {
374 ra8_log_error(s_tag, "no root element in XHTML");
375 return k_ra8_err_no_mem;
376 }
377
378 /* Add the <body> element as the chapter root, then walk its subtree. */
379 ra8_book_attr_t body_attrs[k_xhtml_max_attrs] = {};
380 const uint16_t body_ac = s_collect_attrs(body, ctx, body_attrs);
381 const uint32_t chapter_root =
382 ra8_rabook_add_element(ctx, ra8_rabook_intern(ctx, body->Name()), body_attrs, body_ac);
383
384 s_walk_body_subtree(ctx, body, chapter_root);
385
386 const uint32_t title_off = ra8_rabook_intern(ctx, chapter_title);
387 const uint32_t href_off = ra8_rabook_intern(ctx, chapter_href);
388 const uint32_t chapter_idx = ra8_rabook_add_chapter(ctx, title_off, href_off, chapter_root);
389 return (chapter_idx != k_ra8_book_nil) ? k_ra8_ok : k_ra8_err_no_mem;
390}
