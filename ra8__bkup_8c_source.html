219[[nodiscard]] ra8_err_t ra8_bkup_init(const ra8_bkup_config_t* cfg)
220{
221 RA8_CHECK_NULL_PTR(cfg, s_bkup_tag, "cfg must not be nullptr");
222 const ra8_err_t v_err = internal_validate_cfg(cfg);
223 RA8_RETURN_ON_ERROR(v_err, s_bkup_tag, "bkup_init: cfg out of range"); /* GCOVR_EXCL_BR_LINE */
224
225 /* All of VBTBPCR1 / VBTBPCR2 / VBTBER / VBTBPSR / VBTADSR sit behind PRC1
226 * (HUM Ch 13.1 Table 13.1 p 521); one window covers the whole bring-up. */
227 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_lpm)
228 {
229 if (cfg->enable_switch) {
230 /* HUM Ch 12.2.11 "VBTBPCR1 : VBATT Battery Power Supply Control Register 1", p 507 */
231 *ra8_bkup_vbtbpcr1() = 0U;
232
233 /* HUM Ch 12.2.12 "VBTBPCR2 : VBATT Battery Power Supply Control Register 2", p 508 */
234 /* Program VDETLVL with VDETE = 0 first per the HUM stabilisation note. */
235 *ra8_bkup_vbtbpcr2() = (uint8_t)((uint8_t)cfg->vdet_level & k_ra8_bkup_vbtbpcr2_mask_lvl);
236
237 /* HUM Ch 12.2.12 "VBTBPCR2 : VBATT Battery Power Supply Control Register 2", p 508 */
238 /* Now set VDETE so VCC drop detection arms. */
239 *ra8_bkup_vbtbpcr2() = (uint8_t)(((uint8_t)cfg->vdet_level & k_ra8_bkup_vbtbpcr2_mask_lvl) |
240 k_ra8_bkup_vbtbpcr2_mask_vdete);
241 } else {
242 /* HUM Ch 12.2.11 "VBTBPCR1 : VBATT Battery Power Supply Control Register 1", p 507 */
243 *ra8_bkup_vbtbpcr1() = k_ra8_bkup_vbtbpcr1_mask_bpwswstp;
244 }
245
246 /* HUM Ch 12.2.6 "VBTBER : VBATT Backup Enable Register", p 504 */
247 if (cfg->enable_backup) {
248 /* HUM p 504: "You must write 1 to VBAE before accessing VBTBKR." */
249 *ra8_bkup_vbtber() = k_ra8_bkup_vbtber_mask_vbae;
250 /* HUM p 504: "wait for at least 500 ns after writing 1 to VBAE, and
251 * then access VBTBKR" -- arm the VBTBKRn window before it is used. */
252 internal_vbae_access_settle();
253 } else {
254 *ra8_bkup_vbtber() = 0U;
255 }
256
257 /* HUM Ch 12.2.13 "VBTBPSR : VBATT Battery Power Supply Status Register", p 509 */
258 *ra8_bkup_vbtbpsr() =
259 (uint8_t)k_ra8_bkup_status_clear_keep_mask; /* W0C: clear VBPORF, leave others. */
260
261 /* HUM Ch 12.2.14 "VBTADSR : VBATT Tamper Detection Status Register", p 509 */
262 *ra8_bkup_vbtadsr() = 0U;
263 }
264
265 s_bkup_initialized = true;
266 ra8_log_info(s_bkup_tag, "bkup_init");
267 return k_ra8_ok;

288[[nodiscard]] ra8_err_t ra8_bkup_cold_start_init(ra8_bkup_vdet_level_t level,
289 uint32_t timeout_iters)
290{
291 if ((uint16_t)level > k_ra8_bkup_max_vdet_level) {
292 return k_ra8_err_invalid_arg;
293 }
294 if (timeout_iters == 0U) {
295 return k_ra8_err_invalid_arg;
296 }
297
298 /* Cold-start step 1 (12.3.7.1 p 517): wait for VBPORM == 1. */
299 /* HUM Ch 12.2.13 "VBTBPSR : VBATT Battery Power Supply Status Register", p 509 */
300 bool vbporm_ok = false;
301 for (uint32_t i = 0U; i < timeout_iters; ++i) { /* GCOVR_EXCL_BR_LINE */
302 if ((*ra8_bkup_vbtbpsr() & k_ra8_bkup_vbtbpsr_mask_vbporm) != 0U) { /* GCOVR_EXCL_BR_LINE */
303 vbporm_ok = true;
304 break;
305 }
306 }
307 if (!vbporm_ok) {
308 return k_ra8_err_hw_timeout;
309 }
310
311 /* Steps 2-5 all target PRC1 registers (HUM Ch 13.1 Table 13.1 p 521). */
312 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_lpm)
313 {
314 /* Cold-start step 2 (12.3.7.1 p 517): clear VBPORF. */
315 /* HUM Ch 12.2.13 "VBTBPSR : VBATT Battery Power Supply Status Register", p 509 */
316 *ra8_bkup_vbtbpsr() = (uint8_t)k_ra8_bkup_status_clear_keep_mask;
317
318 /* Cold-start step 3 (12.3.7.1 p 517): programme VDETLVL with VDETE still 0. */
319 /* HUM Ch 12.2.12 "VBTBPCR2 : VBATT Battery Power Supply Control Register 2", p 508 */
320 *ra8_bkup_vbtbpcr2() = (uint8_t)((uint8_t)level & k_ra8_bkup_vbtbpcr2_mask_lvl);
321
322 /* Cold-start step 4 (12.3.7.1 p 517): external tDETWT wait -- caller's job. */
323
324 /* Cold-start step 5 (12.3.7.1 p 517): arm VDETE. */
325 /* HUM Ch 12.2.12 "VBTBPCR2 : VBATT Battery Power Supply Control Register 2", p 508 */
326 *ra8_bkup_vbtbpcr2() =
327 (uint8_t)(((uint8_t)level & k_ra8_bkup_vbtbpcr2_mask_lvl) | k_ra8_bkup_vbtbpcr2_mask_vdete);
328
329 /* HUM Ch 12.2.11 "VBTBPCR1 : VBATT Battery Power Supply Control Register 1", p 507 */
330 *ra8_bkup_vbtbpcr1() = 0U; /* Switch enabled. */
331 }
332
333 s_bkup_initialized = true;
334 ra8_log_info(s_bkup_tag, "bkup_cold_start_init");
335 return k_ra8_ok;

581[[nodiscard]] ra8_err_t ra8_bkup_isr_handle(void)
582{
583 if (!s_bkup_initialized) {
584 return k_ra8_err_not_initialized;
585 }
586
587 /* HUM Ch 12.4 "Interrupt Sources" Table 12.2 p 518: VBATTADI fires when
588 * any (VBTADFn AND VBTADIEn) is true. Read both, AND them, and
589 * dispatch the masked subset. */
590 /* HUM Ch 12.2.14 "VBTADSR : VBATT Tamper Detection Status Register", p 509 */
591 const uint8_t flags = (uint8_t)(*ra8_bkup_vbtadsr() & k_ra8_bkup_vbtadsr_mask_all);
592 /* HUM Ch 12.2.15 "VBTADCR1 : VBATT Tamper Detection Control Register 1", p 510 */
593 const uint8_t enables = (uint8_t)(*ra8_bkup_vbtadcr1() & k_ra8_bkup_vbtadcr1_mask_ie_all);
594 const uint8_t fired = (uint8_t)(flags & enables);
595
596 if (fired != 0U) {
597 /* HUM Ch 12.2.14 "VBTADSR : VBATT Tamper Detection Status Register", p 509 */
598 /* W0C: write 0 to the bits we are dispatching, leave others alone.
599 * VBTADSR is PRC1-protected (HUM Ch 13.1 Table 13.1 p 521); without the
600 * unlock the flag would never clear and the ISR would re-fire forever. */
601 ra8_bkup_internal_rmw8(ra8_bkup_vbtadsr(), fired, false, (uint16_t)k_ra8_prcr_unlock_lpm);
602 ra8_bkup_dispatch(fired);
603 }
604 return k_ra8_ok;
605}
