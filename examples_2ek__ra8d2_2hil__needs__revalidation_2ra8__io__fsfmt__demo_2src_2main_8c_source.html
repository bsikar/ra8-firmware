287RA8_INTERNAL static ra8_err_t
288internal_stub_read(void* file_ctx, void* buf, uint32_t bytes, uint32_t* out_read)
289{
290 if ((out_read == nullptr) || ((buf == nullptr) && (bytes != 0U))) {
291 return k_ra8_err_null_ptr;
292 }
293 if ((file_ctx != &s_demo_stub_file) || !s_demo_stub_file.open) {
294 return k_ra8_err_invalid_state;
295 }
296 const uint32_t size = (uint32_t)sizeof(s_demo_stub_payload) - 1U;
297 uint32_t count = size - s_demo_stub_file.offset;
298 if (bytes < count) {
299 count = bytes;
300 }
301 (void)memcpy(buf, &s_demo_stub_payload[s_demo_stub_file.offset], count);
302 s_demo_stub_file.offset += count;
303 *out_read = count;
304 return k_ra8_ok;
305}

548RA8_INTERNAL static ra8_err_t internal_demo_probe_fat(const ra8_io_fsfmt_t** out_fmt)
549{
550 RA8_CHECK_NULL_PTR(out_fmt, s_tag, "out_fmt");
551 RA8_RETURN_ON_ERROR(
552 ra8_io_blockdev_ram_init(&s_bd, &s_bstate, s_disk, (uint32_t)k_demo_disk_blocks, false),
553 s_tag,
554 "blockdev init");
555 RA8_RETURN_ON_ERROR(ra8_io_blockdev_as_fs_backend(&s_bd, &s_be), s_tag, "fs bridge");
556 ra8_fs_format_opts_t opts = {};
557 opts.type = k_ra8_fs_type_fat12;
558 opts.label = "RAIO";
559 RA8_RETURN_ON_ERROR(ra8_fs_format(&s_be, &opts), s_tag, "format");
560
561 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_init(), s_tag, "fsfmt init");
562 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_probe(&s_be, out_fmt), s_tag, "probe fat");
563 if (strcmp((*out_fmt)->name, "fat") != 0) {
564 return k_ra8_err_not_found;
565 }
566 if ((*out_fmt)->caps.read_only || !(*out_fmt)->caps.supports_streaming_write) {
567 return k_ra8_err_not_supported;
568 }
569 if ((*out_fmt)->caps.max_name_len != (uint16_t)k_ra8_io_fsfmt_fat_max_name_utf8) {
570 return k_ra8_err_invalid_size;
571 }
572 return k_ra8_ok;
573}

592RA8_INTERNAL static ra8_err_t internal_demo_probe_foreign(void)
593{
594 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_register(&s_demo_stub), s_tag, "register stub");
595
596 ra8_io_blockdev_t bd = {};
597 ra8_io_blockdev_ram_state_t state = {};
598 RA8_RETURN_ON_ERROR(

600 s_tag,
601 "stub blockdev");
602 uint8_t blk[(size_t)k_ra8_io_block_size_bytes] = {};
603 blk[0] = (uint8_t)k_demo_stub_magic;
604 RA8_RETURN_ON_ERROR(ra8_io_blockdev_write(&bd, 0, 1, blk), s_tag, "stub mark");
605
606 ra8_fs_backend_t be = {};
607 RA8_RETURN_ON_ERROR(ra8_io_blockdev_as_fs_backend(&bd, &be), s_tag, "stub bridge");
608 const ra8_io_fsfmt_t* fmt = nullptr;
609 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_probe(&be, &fmt), s_tag, "probe stub");
610 if (strcmp(fmt->name, "stub") != 0) {
611 return k_ra8_err_not_found;
612 }
613 if (!fmt->caps.read_only || !fmt->caps.case_sensitive) {
614 return k_ra8_err_not_supported;
615 }
616 return k_ra8_ok;
