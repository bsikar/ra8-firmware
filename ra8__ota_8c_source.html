119RA8_INTERNAL
120static ra8_err_t priv_drain(uint8_t* dst, uint32_t cap, uint32_t* out_n)
121{
122 uint32_t total = 0U;
123 /* Bounded loop: each iteration must consume >= 1 byte or hit EOF. */
124 for (uint32_t guard = 0U; guard < cap + 1U; ++guard) {
125 if (total >= cap) {
126 break;
127 }
128 uint32_t got = 0U;
129 const ra8_err_t e =
130 s_ra8_ota_cfg.net.read(s_ra8_ota_cfg.net.ctx, dst + total, cap - total, &got);
131 if (e != k_ra8_ok) {
132 return e;
133 }
134 if (got == 0U) {
135 break; /* EOF */
136 }
137 total += got;
138 }
139 *out_n = total;
140 return k_ra8_ok;
141}

287{

392RA8_INTERNAL
393static ra8_err_t priv_download_chunk(uint32_t addr_base, uint32_t* in_out_done, uint32_t total)
394{
395 const uint32_t remaining = total - *in_out_done;
396 const uint32_t want = (remaining < k_ra8_ota_chunk_bytes) ? remaining : k_ra8_ota_chunk_bytes;
397 uint32_t got = 0U;
398 ra8_err_t e = priv_drain(s_ra8_ota_buf, want, &got);
399 if (e != k_ra8_ok) {
400 return e;
401 }
402 if (got == 0U) {
403 return k_ra8_err_hw_error;
404 }
405 e = s_ra8_ota_cfg.crypto.sha256_update(s_ra8_ota_cfg.crypto.ctx, s_ra8_ota_buf, got);
406 if (e != k_ra8_ok) {
407 return e;
408 }
409 e = s_ra8_ota_cfg.flash.program(s_ra8_ota_cfg.flash.ctx,
410 addr_base + *in_out_done,
411 s_ra8_ota_buf,
412 got);
413 if (e != k_ra8_ok) {
414 return e;
415 }
416 *in_out_done += got;
417 ra8_ota_internal_set_state(k_ra8_ota_state_downloading, k_ra8_ok);
418 return k_ra8_ok;
419}

480RA8_INTERNAL
481static ra8_err_t priv_download_loop(const ra8_ota_manifest_t* manifest)
482{
483 const uint32_t max_chunks = (k_ra8_ota_max_image_bytes / k_ra8_ota_chunk_bytes) + 1U;
484 uint32_t chunks = 0U;
485 ra8_err_t e = k_ra8_ok;
486 while (s_bytes_done < manifest->image_size_bytes) {
487 /* Validation ensures image_size <= k_ra8_ota_max_image_bytes == 128 *
488 * k_ra8_ota_chunk_bytes, so chunk count never exceeds 128 < max_chunks. */
489 if (chunks >= max_chunks) {
490 e = k_ra8_err_hw_error; /* GCOVR_EXCL_LINE */
491 break; /* GCOVR_EXCL_LINE */
492 }
493 e = priv_download_chunk(s_ra8_ota_cfg.flash.inactive_bank_addr,
494 &s_bytes_done,
495 manifest->image_size_bytes);
496 if (e != k_ra8_ok) {
497 break;
498 }
499 ++chunks;
500 }
501 return e;
502}

533ra8_err_t ra8_ota_download_to_inactive_bank(const ra8_ota_manifest_t* manifest)
534{
535 if (!s_ra8_ota_initialized) {
536 return k_ra8_err_not_initialized;
537 }
538 RA8_CHECK_NULL_PTR(manifest, s_tag, "manifest");
539 if (ra8_ota_internal_download_state_invalid((uint32_t)k_ra8_ota_state_idle,
540 (uint32_t)k_ra8_ota_state_downloading,
541 (uint32_t)s_ra8_ota_state)) {
542 return k_ra8_err_invalid_state;
543 }
544 if (manifest->image_size_bytes > s_ra8_ota_cfg.flash.bank_size_bytes) {
545 ra8_ota_internal_set_state(k_ra8_ota_state_error, k_ra8_err_invalid_size);
546 return k_ra8_err_invalid_size;
547 }
548
549 if (s_bytes_done == 0U) {
550 const ra8_err_t e = priv_prepare_bank(manifest);
551 if (e != k_ra8_ok) {
552 ra8_ota_internal_set_state(k_ra8_ota_state_error, e);
553 return e;
554 }
555 }
556
557 uint32_t content_len = 0U;
558 ra8_err_t e = s_ra8_ota_cfg.net.open(s_ra8_ota_cfg.net.ctx, manifest->image_url, &content_len);
559 if (e != k_ra8_ok) {
560 ra8_ota_internal_set_state(k_ra8_ota_state_error, e);
561 return e;
562 }
563 ra8_ota_internal_set_state(k_ra8_ota_state_downloading, k_ra8_ok);
564
565 e = priv_download_loop(manifest);
566 (void)s_ra8_ota_cfg.net.close(s_ra8_ota_cfg.net.ctx);
567
568 if (e != k_ra8_ok) {
569 ra8_ota_internal_set_state(k_ra8_ota_state_error, e);
570 return e;
571 }
572 ra8_ota_internal_set_state(k_ra8_ota_state_verifying, k_ra8_ok);
573 return k_ra8_ok;
574}

647RA8_INTERNAL
648static ra8_err_t priv_step_dispatch(void)
649{
650 switch (s_ra8_ota_state) {
651 case k_ra8_ota_state_idle: {
652 /* If a previous run_step already fetched and validated the
653 * manifest, advance to download instead of re-fetching it. */
654 if (s_manifest_valid) {
655 return ra8_ota_download_to_inactive_bank(&s_manifest);
656 }
657 ra8_ota_manifest_t m;
658 return ra8_ota_check_for_update(&m);
659 }
660 /* checking and downloading are always resolved synchronously within a
661 * single API call; priv_step_dispatch is never entered while the SM
662 * holds one of these transient states in the host build. */
663 case k_ra8_ota_state_checking:
664 case k_ra8_ota_state_downloading:
665 if (s_manifest_valid) { /* GCOVR_EXCL_LINE */
666 return ra8_ota_download_to_inactive_bank(&s_manifest); /* GCOVR_EXCL_LINE */
667 }
668 return k_ra8_err_invalid_state; /* GCOVR_EXCL_LINE */
669 case k_ra8_ota_state_verifying:
670 return ra8_ota_verify_signature(&s_manifest);
671 case k_ra8_ota_state_committing:
672 return ra8_ota_commit_and_reboot();
673 case k_ra8_ota_state_done:
674 case k_ra8_ota_state_error:
675 case k_ra8_ota_state_count:
676 default:
677 return k_ra8_ok;
678 }
679}

739ra8_err_t ra8_ota_run_full_update(void)
740{
741 if (!s_ra8_ota_initialized) {
742 return k_ra8_err_not_initialized;
743 }
744 /* Bounded by the longest legal sequence (idle -> checking ->
745 * downloading -> verifying -> committing -> done). Six steps is
746 * the upper bound; pad to ``k_ra8_ota_state_count`` for safety. */
747 for (uint32_t i = 0U; i < (uint32_t)k_ra8_ota_state_count; ++i) {
748 if ((s_ra8_ota_state == k_ra8_ota_state_done) || (s_ra8_ota_state == k_ra8_ota_state_error)) {
749 break;
750 }
751 const ra8_err_t e = ra8_ota_run_step();
752 if (e != k_ra8_ok) {
753 return e;
754 }
755 }
756 return s_last_err;
757}
