73static volatile uint32_t* internal_wupen_ptr(uint8_t reg)
74{
75 const uintptr_t base = k_ra8_system_base_addr;
76 if (reg == 0U) {
77 return (volatile uint32_t*)(base + (uintptr_t)k_ra8_lpm_wupen0_off);
78 }
79 return (volatile uint32_t*)(base + (uintptr_t)k_ra8_lpm_wupen1_off);
80}

99static bool internal_decode_wake(ra8_pwr_wake_t source, uint8_t* out_reg, uint8_t* out_bit)
100{
101 const uint8_t reg = (uint8_t)(((uint16_t)source >> 8) & k_pwr_byte_mask);
102 const uint8_t bit = (uint8_t)((uint16_t)source & k_pwr_byte_mask);
103 if (reg >= k_ra8_pwr_wupen_count) {
104 return false;
105 }
106 if (bit >= k_ra8_pwr_wupen_bits) {
107 return false;
108 }
109 *out_reg = reg;
110 *out_bit = bit;
111 return true;
112}

139ra8_err_t ra8_pwr_init(void)
140{
141 ra8_log_info(s_tag, "ra8_pwr_init");
142
143 const ra8_err_t mst_err = ra8_mstp_init();
144 if (mst_err != k_ra8_ok) {
145 ra8_log_error_val(s_tag, "mstp init failed", (uint32_t)mst_err);
146 return mst_err;
147 }
148
149 /* HUM Ch 14.2.19 "WUPEN0 : Wake Up Interrupt Enable Register 0", p 550 */
150 *internal_wupen_ptr(0U) = 0U;
151 /* HUM Ch 14.2.20 "WUPEN1 : Wake Up Interrupt Enable Register 1", p 552 */
152 *internal_wupen_ptr(1U) = 0U;
153 return k_ra8_ok;
154}

166ra8_err_t ra8_pwr_set_wake_source(ra8_pwr_wake_t source)
167{
168 uint8_t reg = 0U;
169 uint8_t bit = 0U;
170 if (!internal_decode_wake(source, &reg, &bit)) {
171 ra8_log_error_val(s_tag, "set_wake: invalid source", (uint32_t)source);
172 return k_ra8_err_invalid_arg;
173 }
174 /* HUM Ch 14.2.19 "WUPEN0 : Wake Up Interrupt Enable Register 0", p 550 */
175 volatile uint32_t* p = internal_wupen_ptr(reg);
176 const uint32_t mask = (uint32_t)1U << bit;
177 *p = *p | mask;
178 return k_ra8_ok;
179}

181ra8_err_t ra8_pwr_clear_wake_source(ra8_pwr_wake_t source)

183 uint8_t reg = 0U;

190 volatile uint32_t* p = internal_wupen_ptr(reg);

196ra8_err_t ra8_pwr_wake_source_is_enabled(ra8_pwr_wake_t source, bool* out_enabled)
197{
198 RA8_CHECK_NULL_PTR(out_enabled, s_tag, "wake_source_is_enabled: out_enabled");
199 uint8_t reg = 0U;
200 uint8_t bit = 0U;
201 if (!internal_decode_wake(source, &reg, &bit)) {
202 return k_ra8_err_invalid_arg;
203 }
204 volatile const uint32_t* p = internal_wupen_ptr(reg);
205 const uint32_t mask = (uint32_t)1U << bit;
206 *out_enabled = ((*p & mask) != 0U);
207 return k_ra8_ok;
208}

215ra8_err_t ra8_pwr_enter_software_standby(void)
216{
217 /* Validate at least one wake source is armed -- entering Software
218 * Standby with WUPEN0 == WUPEN1 == 0 wedges the chip. */
219 /* HUM Ch 14.2.19 "WUPEN0", p 550 */
220 const uint32_t wupen0 = *internal_wupen_ptr(0U);
221 /* HUM Ch 14.2.20 "WUPEN1", p 552 */
222 const uint32_t wupen1 = *internal_wupen_ptr(1U);
223 if ((wupen0 == 0U) && (wupen1 == 0U)) {
224 ra8_log_error(s_tag, "software_standby with no wake source armed");
225 return k_ra8_err_invalid_state;
226 }
227
228 /* The full LPMD + SLEEPDEEP + SBYCR sequence is implemented in
229 * along with the RTC alarm bring-up. For the
230 * helper just issues WFI; the chip stays in normal sleep until
231 * the wake source fires. This keeps the host build a no-op and
232 * lets driver code that calls this function compile and link
233 * cleanly. */
234 internal_wfi();
235 return k_ra8_ok;
236}
