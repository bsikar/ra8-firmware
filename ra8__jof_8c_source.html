196 const size_t take = (len < avail) ? len : avail;

199 return k_ra8_ok;

223RA8_INTERNAL
224static ra8_err_t priv_parse_hdr_fields(const uint8_t* hdr, ra8_jof_info_t* info)
225{
226 info->width = priv_rd_u16(hdr, (uint8_t)k_ra8_jof_ofs_width);
227 info->height = priv_rd_u16(hdr, (uint8_t)k_ra8_jof_ofs_height);
228 info->tile_w = priv_rd_u16(hdr, (uint8_t)k_ra8_jof_ofs_tile_w);
229 info->tile_h = priv_rd_u16(hdr, (uint8_t)k_ra8_jof_ofs_tile_h);
230 info->bpp = hdr[k_ra8_jof_ofs_bpp];
231 info->codec = hdr[k_ra8_jof_ofs_codec];
232 if (info->width == 0U) {
233 return k_ra8_err_validation_failed;
234 }
235 if ((uint32_t)info->width > (uint32_t)k_ra8_jof_max_dim) {
236 return k_ra8_err_validation_failed;
237 }
238 if (info->height == 0U) {
239 return k_ra8_err_validation_failed;
240 }
241 if ((uint32_t)info->height > (uint32_t)k_ra8_jof_max_dim) {
242 return k_ra8_err_validation_failed;
243 }
244 if (info->tile_w == 0U) {
245 return k_ra8_err_validation_failed;
246 }
247 if (info->tile_h == 0U) {
248 return k_ra8_err_validation_failed;
249 }
250 /* bpp 1 (gray8), 3 (RGB888) or 4 (RGBA8888); 2 is not a defined layout. */
251 if ((info->bpp == 0U) || (info->bpp == 2U) ||
252 ((uint32_t)info->bpp > (uint32_t)k_ra8_jof_bpp_max)) {
253 return k_ra8_err_validation_failed;
254 }
255 if (info->codec > (uint8_t)k_ra8_jof_codec_deflate) {
256 return k_ra8_err_validation_failed;
257 }
258 return k_ra8_ok;
259}

309RA8_INTERNAL
310static ra8_err_t
311priv_check_cross(ra8_jof_info_t* info, uint32_t hdr_count, const uint8_t* ftr, uint64_t total_size)
312{
313 const uint32_t cols = priv_ceil_div(info->width, info->tile_w);
314 const uint32_t rows = priv_ceil_div(info->height, info->tile_h);
315 const uint32_t want = cols * rows;
316 if (want > (uint32_t)k_ra8_jof_max_tiles) {
317 return k_ra8_err_validation_failed;
318 }
319 if (hdr_count != want) {
320 return k_ra8_err_validation_failed;
321 }
322 const uint32_t ftr_count = priv_rd_u32(ftr, (uint8_t)k_ra8_jof_ftr_tile_count);
323 const uint32_t ftr_total = priv_rd_u32(ftr, (uint8_t)k_ra8_jof_ftr_total_size);
324 const uint32_t index_off = priv_rd_u32(ftr, (uint8_t)k_ra8_jof_ftr_index_off);
325 if (ftr_count != want) {
326 return k_ra8_err_validation_failed;
327 }
328 if ((uint64_t)ftr_total != total_size) {
329 return k_ra8_err_validation_failed;
330 }
331 if (index_off < (uint32_t)k_ra8_jof_hdr_bytes) {
332 return k_ra8_err_validation_failed;
333 }
334 const uint64_t index_end = (uint64_t)index_off +
335 ((uint64_t)want * (uint64_t)k_ra8_jof_index_entry) +
336 (uint64_t)k_ra8_jof_footer_bytes;
337 if (index_end != total_size) {
338 return k_ra8_err_validation_failed;
339 }
340 info->tile_cols = (uint16_t)cols;
341 info->tile_rows = (uint16_t)rows;
342 info->tile_count = want;
343 info->index_off = index_off;
344 info->total_size = ftr_total;
345 return k_ra8_ok;
346}

387ra8_err_t ra8_jof_parse(ra8_jof_pread_fn pread,
388 void* pread_ctx,
389 uint64_t total_size,
390 ra8_jof_info_t* out_info)
391{
392 RA8_CHECK_NULL_PTR(pread, s_tag, "pread must not be nullptr");
393 RA8_CHECK_NULL_PTR(out_info, s_tag, "out_info must not be nullptr");
394 const uint64_t min_size = (uint64_t)k_ra8_jof_hdr_bytes + (uint64_t)k_ra8_jof_footer_bytes;
395 if ((total_size < min_size) || (total_size > (uint64_t)UINT32_MAX)) {
396 return k_ra8_err_invalid_size;
397 }
398 uint8_t hdr[k_ra8_jof_hdr_bytes] = {};
399 ra8_err_t err = priv_parse_header_region(pread, pread_ctx, hdr, out_info);
400 if (err != k_ra8_ok) {
401 return err;
402 }
403 uint8_t ftr[k_ra8_jof_footer_bytes] = {};
404 err = priv_pread_exact(pread,
405 pread_ctx,
406 total_size - (uint64_t)k_ra8_jof_footer_bytes,
407 ftr,
408 sizeof(ftr));
409 if (err != k_ra8_ok) {
410 return err;
411 }
412 if (memcmp(&ftr[k_ra8_jof_ftr_magic], s_magic_ftr, sizeof(s_magic_ftr)) != 0) {
413 return k_ra8_err_validation_failed;
414 }
415 const uint32_t hdr_count = priv_rd_u32(hdr, (uint8_t)k_ra8_jof_ofs_tile_count);
416 return priv_check_cross(out_info, hdr_count, ftr, total_size);
417}
