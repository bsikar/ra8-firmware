117[[nodiscard]] static ra8_err_t dac_demo_one_triangle_period(void)
118{
119 const uint16_t step = (uint16_t)((uint32_t)k_dac_demo_max_code / (uint32_t)k_dac_demo_steps);
120 for (uint16_t i = 0U; i < (uint16_t)k_dac_demo_steps; ++i) {
121 const uint16_t code = (uint16_t)(i * step);
122 if (ra8_dac_b_write((uint8_t)k_dac_demo_channel, code) != k_ra8_ok) {
123 return k_ra8_err_hw_error;
124 }
125 ra8_delay_ms((uint32_t)k_dac_demo_step_ms);
126 }
127 for (uint16_t i = (uint16_t)k_dac_demo_steps; i > 0U; --i) {
128 const uint16_t code = (uint16_t)((i - 1U) * step);
129 if (ra8_dac_b_write((uint8_t)k_dac_demo_channel, code) != k_ra8_ok) {
130 return k_ra8_err_hw_error;
131 }
132 ra8_delay_ms((uint32_t)k_dac_demo_step_ms);
133 }
134 return k_ra8_ok;
135}
