170static uint32_t xspi_opcode(uint32_t cdt)
171{
172 const uint32_t cmdsize =
173 (cdt >> (uint32_t)k_xspi_cdt_pos_cmdsize) & (uint32_t)k_xspi_cdt_mask_cmdsize;
174 const uint32_t cmdfield = (cdt >> (uint32_t)k_xspi_cdt_pos_cmd) & (uint32_t)k_xspi_cdt_mask_cmd;
175 /* internal_make_cdt left-justifies the opcode in the 16-bit CMD field:
176 * shift = 8 * (2 - cmdsize). Recover the low opcode byte. */
177 const uint32_t shift =
178 (uint32_t)k_xspi_byte_bits * (2U - (cmdsize & (uint32_t)k_xspi_cdt_mask_cmdsize));
179 return (cmdfield >> shift) & (uint32_t)k_xspi_cdt_byte;
180}

183static void xspi_do_read(uint32_t addr, uint32_t n)
184{
185 uint32_t d0 = 0U;
186 uint32_t d1 = 0U;
187 for (uint32_t i = 0U; i < n; i++) {
188 const uint8_t b = xspi_flash_byte(addr + i);
189 if (i < 4U) {
190 d0 |= (uint32_t)b << (i * (uint32_t)k_xspi_byte_bits);
191 } else {
192 d1 |= (uint32_t)b << ((i - 4U) * (uint32_t)k_xspi_byte_bits);
193 }
194 }
195 s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_data0)] = d0;
196 s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_data1)] = d1;
197 s_xspi.reads++;
198 /* Console OSPI tab: one line per manual read command. */
199 char ln[k_xspi_console_line_cap];
200 (void)snprintf(ln, sizeof(ln), "OSPI rd @0x%06X %uB", (unsigned)addr, (unsigned)n);
201 board_console_push(k_board_console_ch_ospi, ln);
202}

205static void xspi_do_program(uint32_t addr, uint32_t n)
206{
207 const uint32_t d0 = s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_data0)];
208 const uint32_t d1 = s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_data1)];
209 for (uint32_t i = 0U; i < n; i++) {
210 if ((addr + i) >= (uint32_t)k_xspi_flash_size) {
211 continue;
212 }
213 const uint32_t w = (i < 4U) ? d0 : d1;
214 const uint32_t s = (i < 4U) ? i : (i - 4U);
215 const uint8_t b =
216 (uint8_t)((w >> (s * (uint32_t)k_xspi_byte_bits)) & (uint32_t)k_xspi_cdt_byte);
217 /* NOR program clears flash bits only: flash &= b. In the inverted store
218 * that is neg = ~(~neg & b) = neg | ~b. */
219 s_xspi_flash_neg[addr + i] |= (uint8_t)~b;
220 xspi_mark_dirty(addr + i + 1U);
221 }
222 s_xspi.writes++;
223 /* Console OSPI tab: one line per page-program command. */
224 char ln[k_xspi_console_line_cap];
225 (void)snprintf(ln, sizeof(ln), "OSPI pp @0x%06X %uB", (unsigned)addr, (unsigned)n);
226 board_console_push(k_board_console_ch_ospi, ln);
227}

230static void xspi_do_erase(uint32_t addr)
231{
232 const uint32_t base = addr & ~((uint32_t)k_xspi_sector_len - 1U);
233 if (base >= (uint32_t)k_xspi_flash_size) {
234 return;
235 }
236 /* Erased flash = 0xFF = all-zero bytes in the inverted store. */
237 (void)memset(&s_xspi_flash_neg[base], 0, (size_t)k_xspi_sector_len);
238 xspi_mark_dirty(base + (uint32_t)k_xspi_sector_len);
239 s_xspi.erases++;
240}

243static void xspi_exec_command(void)
244{
245 const uint32_t cdt = s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_cdt)];
246 const uint32_t addr = s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_addr)];
247 const uint32_t n =
248 (cdt >> (uint32_t)k_xspi_cdt_pos_datasize) & (uint32_t)k_xspi_cdt_mask_datasize;
249 switch (xspi_opcode(cdt)) {
250 case (uint32_t)k_xspi_op_rdid:
251 s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_data0)] = (uint32_t)k_xspi_jedec_cdd0;
252 break;
253 case (uint32_t)k_xspi_op_rdsr:
254 s_xspi.regs[xspi_cdbuf_word((uint32_t)k_xspi_cdbuf_data0)] = (uint32_t)k_xspi_status_wel;
255 break;
256 case (uint32_t)k_xspi_op_read:
257 xspi_do_read(addr, n);
258 break;
259 case (uint32_t)k_xspi_op_pp:
260 xspi_do_program(addr, n);
261 break;
262 case (uint32_t)k_xspi_op_erase:
263 xspi_do_erase(addr);
264 break;
265 default:
266 break; /* WREN, 8D/1S software reset, mode switches: no flash effect. */
267 }
268 s_xspi.regs[(uint32_t)(k_xspi_off_ints / 4U)] |= (uint32_t)k_xspi_ints_cmdcmp;
269}

272static void xspi_reset(void)
273{
274 (void)memset(s_xspi.regs, 0, sizeof(s_xspi.regs));
275 /* Only the dirty prefix ever left the erased state; sweeping the whole
276 * 64 MiB would needlessly commit resident pages in every emulator run. */
277 (void)memset(s_xspi_flash_neg, 0, (size_t)s_xspi_dirty_hi);
278 s_xspi_dirty_hi = 0U;
279 s_xspi.reads = 0U;
280 s_xspi.writes = 0U;
281 s_xspi.erases = 0U;
282}

297static void xspi_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
298{
299 (void)uc;
300 (void)size;
301 const uint64_t off = addr - (uint64_t)k_xspi_base;
302 if ((off / 4U) >= (uint64_t)k_xspi_words) {
303 return;
304 }
305 if (off == (uint64_t)k_xspi_off_intc) {
306 /* Write-1-to-clear against INTS. */
307 s_xspi.regs[(uint32_t)(k_xspi_off_ints / 4U)] &= ~(uint32_t)value;
308 return;
309 }
310 s_xspi.regs[off / 4U] = (uint32_t)value;
311 if ((off == (uint64_t)k_xspi_off_cdctl0) &&
312 (((uint32_t)value & (uint32_t)k_xspi_cdctl0_trreq) != 0U)) {
313 xspi_exec_command();
314 /* TRREQ self-clears on completion. */
315 s_xspi.regs[off / 4U] &= ~(uint32_t)k_xspi_cdctl0_trreq;
316 }
317}
