87uint32_t ra8_cache_dcache_line_bytes(void)
88{
89 /* Arm v8-M ARM: CTR.DminLine = log2 of the smallest D-cache line in words. */
90 const uint32_t ctr = *ra8_cache_reg(k_ra8_cache_ctr);
91 const uint32_t dmin =
92 (ctr >> (uint32_t)k_ra8_cache_ctr_dmin_shift) & (uint32_t)k_ra8_cache_ctr_dmin_mask;
93 return (uint32_t)k_ra8_cache_word_bytes << dmin;
94}

119static uint32_t
120ra8_cache_span_lines(uintptr_t addr, uint32_t size, uint32_t line, uintptr_t* out_start)
121{
122 const uintptr_t mask = (uintptr_t)line - 1U;
123 const uintptr_t start = addr & ~mask;
124 const uintptr_t last = (addr + (uintptr_t)size - 1U) & ~mask;
125 *out_start = start;
126 return (uint32_t)((last - start) / (uintptr_t)line) + 1U;
127}

151static ra8_err_t ra8_cache_maintain_range(const void* addr, uint32_t size, ra8_cache_reg_addr_t reg)
152{
153 RA8_CHECK_NULL_PTR(addr, s_tag, "maintain: addr");
154 if (size == 0U) {
155 return k_ra8_ok;
156 }
157 const uint32_t line = ra8_cache_dcache_line_bytes();
158 uintptr_t start = 0U;
159 const uint32_t lines = ra8_cache_span_lines((uintptr_t)addr, size, line, &start);
160 ra8_hw_dsb();
161 for (uint32_t i = 0U; i < lines; ++i) {
162 *ra8_cache_reg(reg) = (uint32_t)(start + ((uintptr_t)i * (uintptr_t)line));
163 }
164 ra8_hw_dsb();
165 ra8_hw_isb();
166 return k_ra8_ok;
167}

184void ra8_cache_dcache_invalidate_all(void)
185{
186 /* Arm v8-M ARM: select L1 data cache, read its geometry, then invalidate
187 * every set/way. CSSELR=0 selects level 0, data. */
188 *ra8_cache_reg(k_ra8_cache_csselr) = 0U;
189 ra8_hw_dsb();
190 const uint32_t ccsidr = *ra8_cache_reg(k_ra8_cache_ccsidr);
191 if ((ccsidr == 0U) || (ccsidr == UINT32_MAX)) {
192 return; /* geometry unavailable -- nothing safe to invalidate */
193 }
194 uint32_t sets =
195 (ccsidr >> (uint32_t)k_ra8_cache_ccsidr_sets_sh) & (uint32_t)k_ra8_cache_ccsidr_sets_msk;
196 do {
197 uint32_t ways =
198 (ccsidr >> (uint32_t)k_ra8_cache_ccsidr_assoc_sh) & (uint32_t)k_ra8_cache_ccsidr_assoc_mk;
199 do {
200 *ra8_cache_reg(k_ra8_cache_dcisw) = (sets << (uint32_t)k_ra8_cache_dcisw_set_shift) |
201 (ways << (uint32_t)k_ra8_cache_dcisw_way_shift);
202 } while (ways-- != 0U);
203 } while (sets-- != 0U);
204 ra8_hw_dsb();
205 ra8_hw_isb();
206}
