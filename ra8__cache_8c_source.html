96uint32_t ra8_cache_dcache_line_bytes(void)
97{
98 /* Arm v8-M ARM: CTR.DminLine = log2 of the smallest D-cache line in words. */
99 const uint32_t ctr = *ra8_cache_reg(k_ra8_cache_ctr);
100 const uint32_t dmin =
101 (ctr >> (uint32_t)k_ra8_cache_ctr_dmin_shift) & (uint32_t)k_ra8_cache_ctr_dmin_mask;
102 return (uint32_t)k_ra8_cache_word_bytes << dmin;
103}

128static uint32_t
129ra8_cache_span_lines(uintptr_t addr, uint32_t size, uint32_t line, uintptr_t* out_start)
130{
131 const uintptr_t mask = (uintptr_t)line - 1U;
132 const uintptr_t start = addr & ~mask;
133 const uintptr_t last = (addr + (uintptr_t)size - 1U) & ~mask;
134 *out_start = start;
135 return (uint32_t)((last - start) / (uintptr_t)line) + 1U;
136}

160static ra8_err_t ra8_cache_maintain_range(const void* addr, uint32_t size, ra8_cache_reg_addr_t reg)
161{
162 RA8_CHECK_NULL_PTR(addr, s_tag, "maintain: addr");
163 if (size == 0U) {
164 return k_ra8_ok;
165 }
166 const uint32_t line = ra8_cache_dcache_line_bytes();
167 uintptr_t start = 0U;
168 const uint32_t lines = ra8_cache_span_lines((uintptr_t)addr, size, line, &start);
169 ra8_hw_dsb();
170 for (uint32_t i = 0U; i < lines; ++i) {
171 *ra8_cache_reg(reg) = (uint32_t)(start + ((uintptr_t)i * (uintptr_t)line));
172 }
173 ra8_hw_dsb();
174 ra8_hw_isb();
175 return k_ra8_ok;
176}

217static void ra8_cache_setway_all(ra8_cache_reg_addr_t op_reg)
218{
219 /* Arm v8-M ARM: select L1 data cache, read its geometry, then apply the
220 * set/way op to every set/way. CSSELR=0 selects level 0, data. */
221 *ra8_cache_reg(k_ra8_cache_csselr) = 0U;
222 ra8_hw_dsb();
223 const uint32_t ccsidr = *ra8_cache_reg(k_ra8_cache_ccsidr);
224 if ((ccsidr == 0U) || (ccsidr == UINT32_MAX)) {
225 return; /* geometry unavailable -- nothing safe to touch */
226 }
227 uint32_t sets =
228 (ccsidr >> (uint32_t)k_ra8_cache_ccsidr_sets_sh) & (uint32_t)k_ra8_cache_ccsidr_sets_msk;
229 do {
230 uint32_t ways =
231 (ccsidr >> (uint32_t)k_ra8_cache_ccsidr_assoc_sh) & (uint32_t)k_ra8_cache_ccsidr_assoc_mk;
232 do {
233 *ra8_cache_reg(op_reg) = (sets << (uint32_t)k_ra8_cache_dcisw_set_shift) |
234 (ways << (uint32_t)k_ra8_cache_dcisw_way_shift);
235 } while (ways-- != 0U);
236 } while (sets-- != 0U);
237 ra8_hw_dsb();
238 ra8_hw_isb();
239}

246void ra8_cache_icache_invalidate_all(void)
247{
248 /* Arm v8-M ARM: ICIALLU invalidates the whole L1 instruction cache to the
249 * point of unification. Barriers bracket it so prior accesses retire and the
250 * pipeline refetches before execution continues. */
251 ra8_hw_dsb();
252 ra8_hw_isb();
253 *ra8_cache_reg(k_ra8_cache_iciallu) = 0U;
254 ra8_hw_dsb();
255 ra8_hw_isb();
256}

258void ra8_cache_icache_enable(void)
259{
260 ra8_cache_icache_invalidate_all();
261 /* Arm v8-M ARM: set SCB.CCR.IC (bit 17) to enable the L1 instruction cache.
262 * Read-modify-write preserves the other CCR controls (DIV_0_TRP, BP, ...). */
263 uint32_t ccr = *ra8_cache_reg(k_ra8_cache_ccr);
264 ccr |= (uint32_t)k_ra8_cache_ccr_ic_bit;
265 *ra8_cache_reg(k_ra8_cache_ccr) = ccr;
266 ra8_hw_dsb();
267 ra8_hw_isb();
268}

270void ra8_cache_icache_disable(void)
271{
272 /* Arm v8-M ARM: clear SCB.CCR.IC (bit 17), then invalidate so no stale line
273 * survives a later re-enable. The I-cache holds no dirty state. */
274 ra8_hw_dsb();
275 ra8_hw_isb();
276 uint32_t ccr = *ra8_cache_reg(k_ra8_cache_ccr);
277 ccr &= ~(uint32_t)k_ra8_cache_ccr_ic_bit;
278 *ra8_cache_reg(k_ra8_cache_ccr) = ccr;
279 *ra8_cache_reg(k_ra8_cache_iciallu) = 0U;
280 ra8_hw_dsb();
281 ra8_hw_isb();
282}

284void ra8_cache_dcache_enable(void)
285{
286 /* Invalidate every set/way first (existing primitive) so no random power-on
287 * line is treated as valid once the cache is on, THEN set CCR.DC. */
288 ra8_cache_dcache_invalidate_all();
289 /* Arm v8-M ARM: set SCB.CCR.DC (bit 16) to enable the L1 data cache. */
290 uint32_t ccr = *ra8_cache_reg(k_ra8_cache_ccr);
291 ccr |= (uint32_t)k_ra8_cache_ccr_dc_bit;
292 *ra8_cache_reg(k_ra8_cache_ccr) = ccr;
293 ra8_hw_dsb();
294 ra8_hw_isb();
295}

297void ra8_cache_dcache_disable(void)
298{
299 /* Arm v8-M ARM: clear SCB.CCR.DC (bit 16) to stop new allocations, then clean
300 * AND invalidate every set/way via DCCISW so any dirty line is written back to
301 * memory as the cache goes cold -- disable must not discard dirty data. */
302 uint32_t ccr = *ra8_cache_reg(k_ra8_cache_ccr);
303 ccr &= ~(uint32_t)k_ra8_cache_ccr_dc_bit;
304 *ra8_cache_reg(k_ra8_cache_ccr) = ccr;
305 ra8_cache_setway_all(k_ra8_cache_dccisw);
306}

308void ra8_cache_enable(void)
309{
310 /* Unified L1 bring-up: I-cache then D-cache, matching the order every
311 * system_init.c boot copy uses. Each half runs its own architectural
312 * invalidate before setting its CCR enable bit, so this is safe to call once
313 * from a cold cache at boot. */
314 ra8_cache_icache_enable();
315 ra8_cache_dcache_enable();
316}
