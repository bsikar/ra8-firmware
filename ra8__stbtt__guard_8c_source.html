181priv_table_internal_in_bounds(const uint8_t* data, uint64_t buf_len, uint32_t tag, uint64_t t_off)
182{
183 if (tag == (uint32_t)k_sfnt_tag_cmap) {
184 if ((t_off + (uint64_t)k_cmap_header_bytes) > buf_len) {
185 return false;
186 }
187 const uint32_t sub_tables =
188 priv_rd_be_u16(data + (size_t)(t_off + (uint64_t)k_cmap_num_tables_off));
189 const uint64_t sub_end = t_off + (uint64_t)k_cmap_header_bytes +
190 ((uint64_t)sub_tables * (uint64_t)k_cmap_record_bytes);
191 return sub_end <= buf_len;
192 }
193 if (tag == (uint32_t)k_sfnt_tag_head) {
194 return (t_off + (uint64_t)k_head_loc_format_end) <= buf_len;
195 }
196 if (tag == (uint32_t)k_sfnt_tag_maxp) {
197 return (t_off + (uint64_t)k_maxp_num_glyphs_end) <= buf_len;
198 }
199 return true;

202bool ra8_stbtt_sfnt_dir_in_bounds(const uint8_t* data, size_t len, uint32_t fontstart)
203{
204 if (data == nullptr) {
205 return false;
206 }
207
208 const uint64_t buf_len = (uint64_t)len;
209 const uint64_t start = (uint64_t)fontstart;
210
211 /* (1) The 12-byte offset table must lie within the buffer before we may
212 * read numTables from it. Computed in uint64_t so a hostile fontstart
213 * (up to UINT32_MAX) cannot wrap the sum. */
214 if ((start + (uint64_t)k_sfnt_offset_table_bytes) > buf_len) {
215 return false;
216 }
217 const uint32_t num_tables =
218 priv_rd_be_u16(data + (size_t)(start + (uint64_t)k_sfnt_num_tables_off));
219
220 /* (2) The whole table directory (num_tables records of 16 bytes) must fit.
221 * num_tables is a uint16 (<= 65535), so the product is bounded. */
222 const uint64_t dir_end = start + (uint64_t)k_sfnt_offset_table_bytes +
223 ((uint64_t)num_tables * (uint64_t)k_sfnt_table_record_bytes);
224 if (dir_end > buf_len) {
225 return false;
226 }
227
228 /* (3) Every table's declared [offset, offset + length) extent must fit, and
229 * (4) the reads stbtt_InitFont makes INSIDE cmap / head / maxp must also stay
230 * in-bounds -- the top-level extent alone does not bound cmap's internal
231 * 8 * numTables sub-directory walk. Both record fields are uint32, so the
232 * sum is at most 0x1FFFFFFFE and cannot overflow uint64_t. The record-header
233 * reads at rec+0 / rec+8 / rec+12 are proven in-bounds by check (2) above
234 * (rec + 16 <= dir_end <= len). */
235 for (uint32_t i = 0U; i < num_tables; ++i) {
236 const uint64_t rec = start + (uint64_t)k_sfnt_offset_table_bytes +
237 ((uint64_t)i * (uint64_t)k_sfnt_table_record_bytes);
238 const uint32_t tag = priv_rd_be_u32(data + (size_t)(rec + (uint64_t)k_sfnt_record_tag_off));
239 const uint64_t t_off =
240 (uint64_t)priv_rd_be_u32(data + (size_t)(rec + (uint64_t)k_sfnt_record_offset_off));
241 const uint64_t t_len =
242 (uint64_t)priv_rd_be_u32(data + (size_t)(rec + (uint64_t)k_sfnt_record_length_off));
243 if ((t_off + t_len) > buf_len) {
244 return false;
245 }
246 if (!priv_table_internal_in_bounds(data, buf_len, tag, t_off)) {
247 return false;
248 }
249 }
250
251 return true;
252}
