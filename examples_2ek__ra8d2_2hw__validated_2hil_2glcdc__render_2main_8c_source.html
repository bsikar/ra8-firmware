199static void gh_print_hex(uint32_t value)
200{
201 uint8_t buf[k_gh_hex_nibbles];
202 for (uint32_t i = 0U; i < (uint32_t)k_gh_hex_nibbles; i++) {
203 const uint32_t shift = ((uint32_t)k_gh_hex_nibbles - 1U - i) * (uint32_t)k_gh_nibble_bits;
204 const uint32_t nib = (value >> shift) & (uint32_t)k_gh_nibble_mask;
205 buf[i] = (uint8_t)((nib < (uint32_t)k_gh_dec_ten) ? ('0' + nib) : ('A' + (nib - k_gh_dec_ten)));
206 }
207 gh_print(buf, (uint32_t)k_gh_hex_nibbles);
208}

211static void gh_print_uint(uint32_t value)
212{
213 uint8_t buf[k_gh_dec_ten];
214 uint32_t n = 0U;
215 if (value == 0U) {
216 buf[n] = '0';
217 n++;
218 }
219 while ((value > 0U) && (n < (uint32_t)k_gh_dec_ten)) {
220 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_gh_dec_ten));
221 n++;
222 value /= (uint32_t)k_gh_dec_ten;
223 }
224 for (uint32_t i = 0U; i < n; i++) {
225 gh_print(&buf[n - 1U - i], 1U);
226 }
227}

243static void gh_draw_x(uint16_t color)
244{
245 for (uint16_t x = 0U; x < (uint16_t)k_gh_fb_w; x++) {
246 const int32_t y_down = ((int32_t)x * (int32_t)k_gh_fb_h) / (int32_t)k_gh_fb_w;
247 const int32_t y_up = ((int32_t)k_gh_fb_h - 1) - y_down;
248 for (int32_t dy = -(int32_t)k_gh_x_thickness; dy <= (int32_t)k_gh_x_thickness; dy++) {
249 const int32_t yy_down = y_down + dy;
250 const int32_t yy_up = y_up + dy;
251 if ((yy_down >= 0) && (yy_down < (int32_t)k_gh_fb_h)) {
252 s_framebuffer[((uint32_t)yy_down * (uint32_t)k_gh_fb_w) + (uint32_t)x] = color;
253 }
254 if ((yy_up >= 0) && (yy_up < (int32_t)k_gh_fb_h)) {
255 s_framebuffer[((uint32_t)yy_up * (uint32_t)k_gh_fb_w) + (uint32_t)x] = color;
256 }
257 }
258 }

262static void gh_draw_border(uint16_t color)
263{
264 const uint32_t last_row = ((uint32_t)k_gh_fb_h - 1U) * (uint32_t)k_gh_fb_w;

271 s_framebuffer[row] = color;

288static uint32_t gh_framebuffer_hash(void)
289{
290 const uint8_t* p = (const uint8_t*)s_framebuffer;
291 const size_t n = sizeof(s_framebuffer);
292 uint32_t hsh = (uint32_t)k_gh_fnv_offset;
293 for (size_t i = 0U; i < n; i++) {
294 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_gh_fnv_prime;
295 }
296 return hsh;
297}

327static bool gh_glcdc_programmed(void)
328{
329 display_handle_t* d = nullptr;
330 if (display_init(&k_gh_display_cfg, &d) != k_ra8_ok) {
331 return false;
332 }
333 if (d == nullptr) {
334 return false;
335 }

341 return false;

344 if (ra8_glcdc_get_status(&status_mask) != k_ra8_ok) {
345 return false;
346 }
347 g_glcdc_hil_status = status_mask;
348 return true;
