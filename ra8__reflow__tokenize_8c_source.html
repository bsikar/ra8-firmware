130RA8_INTERNAL
131static ra8_err_t priv_handle_cdata(tok_ctx_t* ctx, const uint8_t* buf, size_t* pi, size_t len)
132{
133 const size_t inner = *pi + strlen("<![CDATA[");
134 size_t close = inner;
135 while (((close + 3U) <= len) && (memcmp(&buf[close], "]]>", 3U) != 0)) {
136 ++close;
137 }
138 if ((ctx->sp > 0U) && !priv_suppressed(ctx)) {
139 uint32_t off = ctx->engine->text_pool_used;
140 bool last_ws = true;
141 for (size_t k = inner; k < close; ++k) {
142 if (!ra8_reflow_tok_feed(ctx->engine, (char)buf[k], &last_ws)) {
143 return k_ra8_err_no_mem;
144 }
145 }
146 const uint32_t tlen = ctx->engine->text_pool_used - off;
147 if ((tlen > 0U) && !ra8_reflow_tok_emit(ctx->engine,
148 k_ra8_reflow_tok_text,
149 k_ra8_reflow_tag_unknown,
150 ctx->style,
151 off,
152 tlen)) {
153 return k_ra8_err_no_mem;
154 }
155 if (tlen > 0U) {
156 ctx->engine->tokens[ctx->engine->token_count - 1U].reserved16 = (uint16_t)ctx->face_slot;
157 }
158 }
159 *pi = ((close + 3U) <= len) ? (close + 3U) : len;
160 return k_ra8_ok;
161}

182RA8_INTERNAL
183static ra8_err_t priv_handle_end(tok_ctx_t* ctx, const uint8_t* buf, size_t* pi, size_t len)
184{
185 size_t i = *pi + 2U; /* past "</" */
186 while ((i < len) && (buf[i] != '>')) {
187 ++i;
188 }
189 *pi = (i < len) ? (i + 1U) : len;
190 if (ctx->sp == 0U) {
191 return k_ra8_err_validation_failed;
192 }
193 ctx->sp--;
194 const ra8_reflow_html_tag_t tag = (ra8_reflow_html_tag_t)ctx->stack_el[ctx->sp].tag;
195 ctx->style = ctx->stack_style[ctx->sp];
196 ctx->active_link = ctx->stack_link[ctx->sp];
197 ctx->color = ctx->stack_color[ctx->sp];
198 ctx->css_font_px = ctx->stack_font[ctx->sp];
199 ctx->family_off = ctx->stack_fam_off[ctx->sp];
200 ctx->family_len = ctx->stack_fam_len[ctx->sp];
201 ctx->face_slot = ctx->stack_face[ctx->sp];
202 if (ra8_reflow_tok_is_block(tag) && !priv_suppressed(ctx) &&
203 !ra8_reflow_tok_emit(ctx->engine, k_ra8_reflow_tok_block_end, tag, ctx->style, 0U, 0U)) {
204 return k_ra8_err_no_mem;
205 }
206 /* Exited the display:none subtree once we pop back above its depth. */
207 if ((ctx->suppress_sp != 0U) && (ctx->sp < ctx->suppress_sp)) {
208 ctx->suppress_sp = 0U;
209 }
210 return k_ra8_ok;
211}

232RA8_INTERNAL
233static ra8_reflow_html_tag_t
234priv_parse_start(const uint8_t* buf, size_t* pi, size_t len, bool* selfclose)
235{
236 size_t i = *pi + 1U; /* past '<' */
237 const size_t nstart = i;
238 while ((i < len) && (buf[i] != '>') && (buf[i] != '/') &&
239 !ra8_reflow_tok_is_xml_whitespace((char)buf[i])) {
240 ++i;
241 }
242 const ra8_reflow_html_tag_t tag = ra8_reflow_tok_classify((const char*)&buf[nstart], i - nstart);
243 *selfclose = false;
244 while ((i < len) && (buf[i] != '>')) {
245 const char c = (char)buf[i];
246 if ((c == '"') || (c == '\'')) {
247 ++i;
248 while ((i < len) && (buf[i] != (uint8_t)c)) {
249 ++i;
250 }
251 } else if ((c == '/') && ((i + 1U) < len) && (buf[i + 1U] == '>')) {
252 *selfclose = true;
253 }
254 ++i;
255 }
256 *pi = (i < len) ? (i + 1U) : len;
257 return tag;
258}

283RA8_INTERNAL
284static ra8_err_t priv_open_attrs(tok_ctx_t* ctx,
285 const uint8_t* tag,
286 size_t span,
287 ra8_reflow_html_tag_t kind,
288 bool block,
289 const ra8_css_style_t* comp,
290 uint16_t css_font_px)
291{
292 if (priv_suppressed(ctx)) {
293 return k_ra8_ok; /* inside a display:none subtree -> emit nothing */
294 }
295 if (block) {
296 uint32_t id_off = 0U;
297 uint32_t id_len = 0U;
298 ra8_reflow_tok_capture_attr(ctx->engine, tag, span, "id", sizeof("id") - 1U, &id_off, &id_len);
299 if (!ra8_reflow_tok_emit(ctx->engine,
300 k_ra8_reflow_tok_block_start,
301 kind,
302 ctx->style,
303 id_off,
304 id_len)) {
305 return k_ra8_err_no_mem;
306 }
307 /* Stash the block's cascaded alignment + CSS font px on the block-start token
308 * (left + 0 = unstyled defaults, so unstyled blocks lay out byte-identically). */
309 const uint8_t align = ((comp->set & (uint8_t)k_ra8_css_set_align) != 0U)
310 ? comp->align
311 : (uint8_t)k_ra8_reflow_align_left;
312 ra8_reflow_token_t* bst = &ctx->engine->tokens[ctx->engine->token_count - 1U];
313 bst->reserved = align;
314 bst->css_font_px = css_font_px;
315 return k_ra8_ok;
316 }
317 if (kind == k_ra8_reflow_tag_a) {
318 uint32_t href_off = 0U;
319 uint32_t href_len = 0U;
320 ra8_reflow_tok_capture_attr(ctx->engine,
321 tag,
322 span,
323 "href",
324 sizeof("href") - 1U,
325 &href_off,
326 &href_len);
327 ctx->active_link = ra8_reflow_tok_intern_link(ctx->engine, href_off, href_len);
328 }
329 return k_ra8_ok;
330}

354static void priv_handle_link(tok_ctx_t* ctx, const uint8_t* buf, size_t tag_lt, size_t end)
355{
356 if (ctx->engine->css_loader == nullptr) {
357 return;
358 }
359 const uint8_t* tagbuf = &buf[tag_lt];
360 const size_t span = (end > tag_lt) ? (end - tag_lt) : 0U;
361 size_t rel_off = 0U;
362 size_t rel_len = 0U;
363 size_t href_off = 0U;
364 size_t href_len = 0U;
365 if (!ra8_reflow_tok_find_attr(tagbuf, span, "rel", sizeof("rel") - 1U, &rel_off, &rel_len) ||
366 !ra8_reflow_tok_find_attr(tagbuf, span, "href", sizeof("href") - 1U, &href_off, &href_len) ||
367 !ra8_reflow_tok_rel_is_stylesheet(&tagbuf[rel_off], rel_len)) {
368 return;
369 }
370 const uint8_t* css_bytes = nullptr;
371 size_t css_len = 0U;
372 if ((ctx->engine->css_loader(ctx->engine->css_loader_ctx,
373 (const char*)&tagbuf[href_off],
374 (uint32_t)href_len,
375 &css_bytes,
376 &css_len) == k_ra8_ok) &&
377 (css_bytes != nullptr) && (css_len > 0U)) {
378 (void)ra8_css_parse(&ctx->engine->css, (const char*)css_bytes, (uint32_t)css_len);
379 }
380}

407RA8_INTERNAL
408static bool priv_handle_void(tok_ctx_t* ctx,
409 const uint8_t* buf,
410 size_t tag_lt,
411 size_t end,
412 ra8_reflow_html_tag_t tag,
413 ra8_err_t* out_err)
414{
415 if (tag == k_ra8_reflow_tag_link) {
416 priv_handle_link(ctx, buf, tag_lt, end); /* external stylesheet (no token) */
417 *out_err = k_ra8_ok;
418 return true;
419 }
420 const bool is_void =
421 (tag == k_ra8_reflow_tag_br) || (tag == k_ra8_reflow_tag_hr) || (tag == k_ra8_reflow_tag_img);
422 if (is_void && priv_suppressed(ctx)) {
423 *out_err = k_ra8_ok; /* void tag inside a display:none subtree -> drop */
424 return true;
425 }
426 if (tag == k_ra8_reflow_tag_br) {
427 *out_err = ra8_reflow_tok_emit(ctx->engine, k_ra8_reflow_tok_break, tag, ctx->style, 0U, 0U)
428 ? k_ra8_ok
429 : k_ra8_err_no_mem;
430 return true;
431 }
432 if (tag == k_ra8_reflow_tag_hr) {
433 *out_err = ra8_reflow_tok_emit(ctx->engine, k_ra8_reflow_tok_rule, tag, ctx->style, 0U, 0U)
434 ? k_ra8_ok
435 : k_ra8_err_no_mem;
436 return true;
437 }
438 if (tag == k_ra8_reflow_tag_img) {
439 uint32_t src_off = 0U;
440 uint32_t src_len = 0U;
441 const size_t span = (end > tag_lt) ? (end - tag_lt) : 0U;
442 ra8_reflow_tok_capture_attr(ctx->engine,
443 &buf[tag_lt],
444 span,
445 "src",
446 sizeof("src") - 1U,
447 &src_off,
448 &src_len);
449 *out_err =
450 ra8_reflow_tok_emit(ctx->engine, k_ra8_reflow_tok_image, tag, ctx->style, src_off, src_len)
451 ? k_ra8_ok
452 : k_ra8_err_no_mem;
453 return true;
454 }
455 return false;
456}

477RA8_INTERNAL
478static uint16_t priv_css_font_px(const tok_ctx_t* ctx, const ra8_css_style_t* comp)
479{
480 if ((comp->set & (uint8_t)k_ra8_css_set_fontsize) == 0U) {
481 return ctx->css_font_px; /* inherit (0 = none -> UA default) */
482 }
483 const uint16_t parent = (ctx->css_font_px != 0U) ? ctx->css_font_px : ctx->engine->font_px;
484 uint32_t px =
485 ((ra8_css_font_unit_t)comp->font_unit == k_ra8_css_font_pct)
486 ? (((uint32_t)parent * (uint32_t)comp->font_val) / (uint32_t)k_ra8_reflow_pct_full)
487 : (uint32_t)comp->font_val;
488 if (px < (uint32_t)k_ra8_reflow_min_font_px) {
489 px = (uint32_t)k_ra8_reflow_min_font_px;
490 }
491 if (px > (uint32_t)k_ra8_reflow_max_font_px) {
492 px = (uint32_t)k_ra8_reflow_max_font_px;
493 }
494 return (uint16_t)px;
495}

524RA8_INTERNAL
525static ra8_err_t priv_open_styled(tok_ctx_t* ctx,
526 const uint8_t* tagbuf,
527 size_t span,
528 ra8_reflow_html_tag_t tag,
529 bool block)
530{
531 const uint8_t base = (uint8_t)(ctx->style | ra8_reflow_tok_style_for(tag));
532 const ra8_css_element_t el = ctx->stack_el[ctx->sp - 1U]; /* pushed by the caller */
533 ra8_css_style_t inh = {.set = (uint8_t)k_priv_style_mask, .style = base};
534 if (ctx->color != (uint32_t)k_ra8_reflow_color_inherit) {
535 inh.set = (uint8_t)(inh.set | (uint8_t)k_ra8_css_set_color);
536 inh.color = ctx->color;
537 }
538 /* font-family inherits (#109): seed the cascade with the parent's resolved
539 * family so a child without its own `font-family` keeps the ancestor's face. */
540 if (ctx->family_len != 0U) {
541 inh.set = (uint8_t)(inh.set | (uint8_t)k_ra8_css_set_family);
542 inh.family_off = ctx->family_off;
543 inh.family_len = ctx->family_len;
544 }
545 const ra8_css_style_t inl = ra8_reflow_tok_css_inline(tagbuf, span);
546 /* Ancestors = the open-element stack below this one (stack_el[0 .. sp-2]). */
547 const uint8_t n_anc = (uint8_t)(ctx->sp - 1U);
548 const ra8_css_style_t comp =
549 ra8_css_cascade_ctx(&ctx->engine->css, &el, inh, inl, ctx->stack_el, n_anc);
550 const uint16_t fpx = priv_css_font_px(ctx, &comp);
551 /* display:none (#140): begin suppressing this element + its subtree at the
552 * current depth; priv_open_attrs and the emit sites then drop every token
553 * until the matching close pops back above this depth. */
554 const bool hidden = ((comp.set & (uint8_t)k_ra8_css_set_display) != 0U) && (comp.display != 0U);
555 if (hidden && (ctx->suppress_sp == 0U)) {
556 ctx->suppress_sp = ctx->sp;
557 }
558 const ra8_err_t aerr = priv_open_attrs(ctx, tagbuf, span, tag, block, &comp, fpx);
559 if (aerr != k_ra8_ok) {
560 return aerr;
561 }
562 ctx->style = (uint8_t)(comp.style & (uint8_t)k_priv_style_mask);
563 ctx->color = ((comp.set & (uint8_t)k_ra8_css_set_color) != 0U)
564 ? comp.color
565 : (uint32_t)k_ra8_reflow_color_inherit;
566 ctx->css_font_px = fpx;
567 /* Resolve the embedded face for this element from its cascaded family +
568 * emphasis (#109). Children inherit the family slice; text runs stamp the
569 * resolved slot. Both are 0 / unchanged for content without `@font-face`. */
570 if ((comp.set & (uint8_t)k_ra8_css_set_family) != 0U) {
571 ctx->family_off = comp.family_off;
572 ctx->family_len = comp.family_len;
573 }
574 ctx->face_slot = ra8_reflow_tok_resolve_face_slot(ctx->engine, &comp);
575 return k_ra8_ok;
576}

601RA8_INTERNAL
602static ra8_err_t priv_handle_start(tok_ctx_t* ctx, const uint8_t* buf, size_t* pi, size_t len)
603{
604 const size_t tag_lt = *pi; /* index of '<' before parse */
605 bool selfclose = false;
606 const ra8_reflow_html_tag_t tag = priv_parse_start(buf, pi, len, &selfclose);
607 ctx->saw_element = true;
608
609 ra8_err_t verr = k_ra8_ok;
610 if (priv_handle_void(ctx, buf, tag_lt, *pi, tag, &verr)) {
611 return verr;
612 }
613
614 const bool block = ra8_reflow_tok_is_block(tag);
615 if (selfclose) {
616 if (block && !priv_suppressed(ctx)) {
617 if (!ra8_reflow_tok_emit(ctx->engine,
618 k_ra8_reflow_tok_block_start,
619 tag,
620 ctx->style,
621 0U,
622 0U)) {
623 return k_ra8_err_no_mem;
624 }
625 if (!ra8_reflow_tok_emit(ctx->engine, k_ra8_reflow_tok_block_end, tag, ctx->style, 0U, 0U)) {
626 return k_ra8_err_no_mem;
627 }
628 }
629 return k_ra8_ok;
630 }
631
632 if (ctx->sp >= (uint32_t)k_priv_max_depth) {
633 return k_ra8_err_no_mem;
634 }
635 const size_t span = (*pi > tag_lt) ? (*pi - tag_lt) : 0U;
636 /* Record the element's full identity (tag + id + class, aliasing the chapter
637 * buffer) so the cascade reads it back and descendant selectors can match it
638 * as an ancestor of a later element. */
639 ctx->stack_el[ctx->sp] = ra8_reflow_tok_css_element(tag, &buf[tag_lt], span);
640 ctx->stack_style[ctx->sp] = ctx->style;
641 ctx->stack_link[ctx->sp] = ctx->active_link;
642 ctx->stack_color[ctx->sp] = ctx->color;
643 ctx->stack_font[ctx->sp] = ctx->css_font_px;
644 ctx->stack_fam_off[ctx->sp] = ctx->family_off;
645 ctx->stack_fam_len[ctx->sp] = ctx->family_len;
646 ctx->stack_face[ctx->sp] = ctx->face_slot;
647 ctx->sp++;
648 return priv_open_styled(ctx, &buf[tag_lt], span, tag, block);
649}

672static bool priv_tag_is(const uint8_t* buf, size_t i, size_t len, const char* name)
673{
674 if (!ra8_reflow_tok_starts_with(buf, i, len, name)) {
675 return false;

745RA8_INTERNAL
746static ra8_err_t priv_handle_lt(tok_ctx_t* ctx, const uint8_t* buf, size_t* pi, size_t len)
747{
748 if (ra8_reflow_tok_starts_with(buf, *pi, len, "<!--")) {
749 *pi = ra8_reflow_tok_skip_past(buf, *pi + 4U, len, "-->");
750 return k_ra8_ok;
751 }
752 if (ra8_reflow_tok_starts_with(buf, *pi, len, "<![CDATA[")) {
753 return priv_handle_cdata(ctx, buf, pi, len);
754 }
755 if (ra8_reflow_tok_starts_with(buf, *pi, len, "<?")) {
756 *pi = ra8_reflow_tok_skip_past(buf, *pi + 2U, len, "?>");
757 return k_ra8_ok;
758 }
759 if (ra8_reflow_tok_starts_with(buf, *pi, len, "<!")) {
760 *pi = ra8_reflow_tok_skip_past(buf, *pi + 2U, len, ">");
761 return k_ra8_ok;
762 }
763 if (priv_tag_is(buf, *pi, len, "<style")) {
764 priv_handle_raw_text(ctx, buf, pi, len, "</style>", true);
765 return k_ra8_ok;
766 }
767 if (priv_tag_is(buf, *pi, len, "<script")) {
768 priv_handle_raw_text(ctx, buf, pi, len, "</script>", false);
769 return k_ra8_ok;
770 }
771 if (((*pi + 1U) < len) && (buf[*pi + 1U] == '/')) {
772 return priv_handle_end(ctx, buf, pi, len);
773 }
774 return priv_handle_start(ctx, buf, pi, len);
775}

798static bool priv_emit_text_run(tok_ctx_t* ctx, const uint8_t* buf, size_t run, size_t end)
799{
800 if ((ctx->sp == 0U) || priv_suppressed(ctx)) {
801 return true; /* outside any element, or inside display:none -> drop */
802 }
803 uint32_t off = 0U;
804 uint32_t tln = 0U;
805 if (!ra8_reflow_tok_stash_run(ctx->engine, buf, run, end, &off, &tln)) {
806 return false;
807 }
808 if (tln == 0U) {
809 return true;
810 }
811 if (!ra8_reflow_tok_emit(ctx->engine,
812 k_ra8_reflow_tok_text,
813 k_ra8_reflow_tag_unknown,
814 ctx->style,
815 off,
816 tln)) {
817 return false;
818 }
819 ctx->engine->tokens[ctx->engine->token_count - 1U].reserved = ctx->active_link;
820 ctx->engine->tokens[ctx->engine->token_count - 1U].color = ctx->color;
821 ctx->engine->tokens[ctx->engine->token_count - 1U].reserved16 = (uint16_t)ctx->face_slot;
822 return true;
823}

830ra8_err_t priv_reflow_xml_walk(ra8_reflow_t* engine, const uint8_t* xhtml_buf, size_t xhtml_len)
831{
832 if ((engine == nullptr) || (xhtml_buf == nullptr)) {
833 return k_ra8_err_null_ptr;
834 }
835 if (xhtml_len == 0U) {
836 return k_ra8_err_invalid_size;
837 }
838
839 (void)ra8_css_sheet_reset(&engine->css); /* fresh CSS rules per chapter (#111) */
840
841 /* The tokenizer scratch (per-open-element cascade stacks) is ~2 kB, which
842 * would push this frame past the project stack-usage budget. Hold it in
843 * module-static storage instead of on the stack and reset it on entry.
844 * Safe: priv_reflow_xml_walk has a single, non-recursive, single-threaded
845 * caller (priv_reflow_parse), so the shared instance never overlaps. */
846 static tok_ctx_t s_walk_ctx;
847 s_walk_ctx = (tok_ctx_t){};
848 tok_ctx_t* ctx = &s_walk_ctx;
849 ctx->engine = engine;
850 ctx->style = (uint8_t)k_ra8_reflow_style_normal;
851 ctx->color = (uint32_t)k_ra8_reflow_color_inherit;
852
853 size_t i = 0U;
854 while (i < xhtml_len) {
855 if (xhtml_buf[i] == '<') {
856 const ra8_err_t err = priv_handle_lt(ctx, xhtml_buf, &i, xhtml_len);
857 if (err != k_ra8_ok) {
858 return err;
859 }
860 continue;
861 }
862 const size_t run = i;
863 while ((i < xhtml_len) && (xhtml_buf[i] != '<')) {
864 ++i;
865 }
866 if (!priv_emit_text_run(ctx, xhtml_buf, run, i)) {
867 return k_ra8_err_no_mem;
868 }
869 }
870
871 if (!ctx->saw_element) {
872 return k_ra8_err_validation_failed;
873 }
874 if (ctx->sp != 0U) {
875 return k_ra8_err_validation_failed;
876 }
877 return k_ra8_ok;
878}
