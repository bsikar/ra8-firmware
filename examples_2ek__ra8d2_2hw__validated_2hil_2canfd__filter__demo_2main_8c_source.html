251[[nodiscard]] static ra8_err_t canfd_filter_one_round(uint32_t id)
252{
253 ra8_canfd_frame_t tx = {.id = id,
254 .dlc = (uint8_t)k_canfd_filter_dlc,
255 .is_extended = 0U,
256 .is_fd = 0U,
257 .is_brs = 0U,
258 .data = {(uint8_t)id,
259 (uint8_t)(id >> k_canfd_filter_byte_shift),
260 (uint8_t)k_canfd_filter_payload_b2,
261 (uint8_t)k_canfd_filter_payload_b3,
262 (uint8_t)k_canfd_filter_payload_b4,
263 (uint8_t)k_canfd_filter_payload_b5,
264 (uint8_t)k_canfd_filter_payload_b6,
265 (uint8_t)k_canfd_filter_payload_b7}};
266 if (ra8_canfd_transmit((uint8_t)k_canfd_filter_channel, &tx) != k_ra8_ok) {
267 return k_ra8_err_hw_error;
268 }
269 /* Poll for the loopback frame: 500 kbit/s + 8 data bytes is ~240 us
270 * round-trip, so wait up to ~10 ms before declaring no_data. */
271 ra8_canfd_frame_t rx = {};
272 for (uint32_t i = 0U; i < (uint32_t)k_canfd_filter_rx_spin; i++) {
273 if (ra8_canfd_receive((uint8_t)k_canfd_filter_channel, &rx) == k_ra8_ok) {
274 return k_ra8_ok;
275 }
276 }
277 return k_ra8_err_no_data;
278}

282int32_t main(void)
283{
284 canfd_filter_setup_or_halt();
285 ra8_isr_globals_enable();
286
287 while (1) {
288 /* Drain any stale frames left in RX FIFO 0 before this iteration's
289 * three sub-rounds. Without this, a late-arriving frame from the
290 * mask round can land in the FIFO while the no-match round is
291 * spinning on ra8_canfd_receive and report a false "match" for
292 * what should be a rejected frame. */
293 ra8_canfd_frame_t scratch = {};
294 for (uint8_t i = 0U; i < (uint8_t)k_canfd_filter_drain_max; i++) {
295 if (ra8_canfd_receive((uint8_t)k_canfd_filter_channel, &scratch) != k_ra8_ok) {
296 break;
297 }
298 }
299
300 /* Two IDs the filters accept and one they should reject. The
301 * loopback path mirrors every TX, but only filter-matched frames
302 * land in the RX FIFO. */
303 if (canfd_filter_one_round((uint32_t)k_canfd_filter_id_exact) == k_ra8_ok) {
304 (void)ra8_board_led_toggle(k_ra8_board_led1);
305 g_canfd_filter_exact_match += 1U;
306 g_canfd_filter_match += 1U;
307 } else {
308 g_canfd_filter_exact_mismatch += 1U;
309 g_canfd_filter_mismatch += 1U;
310 }
311 if (canfd_filter_one_round((uint32_t)k_canfd_filter_id_mask) == k_ra8_ok) {
312 (void)ra8_board_led_toggle(k_ra8_board_led1);
313 g_canfd_filter_mask_match += 1U;
314 g_canfd_filter_match += 1U;
315 } else {
316 g_canfd_filter_mask_mismatch += 1U;
317 g_canfd_filter_mismatch += 1U;
318 }
319 /* No-match: receive should report no_data; if it doesn't, latch LED2. */
320 if (canfd_filter_one_round((uint32_t)k_canfd_filter_id_nomatch) == k_ra8_ok) {
321 (void)ra8_board_led_toggle(k_ra8_board_led2);
322 g_canfd_filter_nomatch_mismatch += 1U;
323 g_canfd_filter_mismatch += 1U;
324 } else {
325 g_canfd_filter_nomatch_match += 1U;
326 g_canfd_filter_match += 1U;
327 }
328 ra8_delay_ms(k_canfd_filter_period_ms);
329 }
330 canfd_filter_panic_halt();
331 return 0;
332}
