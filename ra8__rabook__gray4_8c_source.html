83static uint8_t s_bilinear_sample(const uint8_t* src,
84 uint16_t src_w,
85 uint16_t src_h,
86 uint64_t sx_fp,
87 uint64_t sy_fp)
88{
89 assert(src != nullptr);
90 assert(src_w > 0U);
91 assert(src_h > 0U);
92
93 uint16_t sx0 = (uint16_t)(sx_fp >> k_fp_shift);
94 uint16_t sy0 = (uint16_t)(sy_fp >> k_fp_shift);
95 uint16_t sx1 = (uint16_t)((sx0 + 1U < (uint32_t)src_w) ? sx0 + 1U : src_w - 1U);
96 uint16_t sy1 = (uint16_t)((sy0 + 1U < (uint32_t)src_h) ? sy0 + 1U : src_h - 1U);
97 uint64_t fx = sx_fp & (uint64_t)k_fp_mask;
98 uint64_t fy = sy_fp & (uint64_t)k_fp_mask;
99 uint64_t ifx = (uint64_t)k_fp_unit - fx;
100 uint64_t ify = (uint64_t)k_fp_unit - fy;
101
102 uint64_t p00 = src[((uint32_t)sy0 * src_w) + sx0];
103 uint64_t p10 = src[((uint32_t)sy0 * src_w) + sx1];
104 uint64_t p01 = src[((uint32_t)sy1 * src_w) + sx0];
105 uint64_t p11 = src[((uint32_t)sy1 * src_w) + sx1];
106
107 uint64_t val = (p00 * ifx * ify) + (p10 * fx * ify) + (p01 * ifx * fy) + (p11 * fx * fy);
108 return (uint8_t)(val >> k_fp_norm_shift);
109}

134static void s_pack_nibbles(const uint8_t* gray_pixels, uint32_t n_pixels, uint8_t* out)
135{
136 for (uint32_t i = 0U; i < n_pixels; i++) {
137 uint8_t nib = (uint8_t)((gray_pixels[i] + (uint8_t)k_ra8_rabook_gray4_round_half) /
138 (uint8_t)k_ra8_rabook_gray4_quant_div);
139 if (nib > (uint8_t)k_ra8_rabook_gray4_nib_max) {
140 nib = (uint8_t)k_ra8_rabook_gray4_nib_max;
141 }
142
143 uint32_t byte_idx = i / k_ra8_rabook_gray4_nib_per_byte;
144 if ((i & (uint32_t)k_nib_lo_msk) == 0U) {
145 out[byte_idx] = (uint8_t)(nib << k_nib_shift);
146 } else {
147 out[byte_idx] |= nib;
148 }
149 }
150}

156void ra8_rabook_gray4_output_dims(uint16_t src_w,

173 if (longer <= max_edge) {
174 *out_w = src_w;
175 *out_h = src_h;

177 }

180 *out_w = (uint16_t)((((uint32_t)src_w * max_edge) + half_longer) / longer);
181 *out_h = (uint16_t)((((uint32_t)src_h * max_edge) + half_longer) / longer);
182
183 if (*out_w == 0U) {
184 *out_w = 1U;
185 }
186 if (*out_h == 0U) {
187 *out_h = 1U;

191ra8_err_t ra8_rabook_gray4_downscale(const uint8_t* src,

193 uint16_t src_h,
194 uint8_t* dst,
195 uint16_t dst_w,
196 uint16_t dst_h)
197{
198 RA8_CHECK_NULL_PTR(src, s_tag, "src");
199 RA8_CHECK_NULL_PTR(dst, s_tag, "dst");
200
201 if (dst_w == 0U || dst_h == 0U) {
202 ra8_log_error(s_tag, "dst_w or dst_h is zero");
203 return k_ra8_err_invalid_arg;
204 }
205
206 if (src_w == 0U || src_h == 0U) {
207 (void)memset(dst, 0, (size_t)dst_w * (size_t)dst_h);
208 return k_ra8_ok;
209 }
210
211 uint64_t x_step = ((uint64_t)src_w << k_fp_shift) / dst_w;
212 uint64_t y_step = ((uint64_t)src_h << k_fp_shift) / dst_h;
213
214 for (uint16_t dy = 0U; dy < dst_h; dy++) {
215 uint64_t sy_fp = (uint64_t)dy * y_step;
216
217 for (uint16_t dx = 0U; dx < dst_w; dx++) {
218 uint64_t sx_fp = (uint64_t)dx * x_step;
219 dst[((uint32_t)dy * dst_w) + dx] = s_bilinear_sample(src, src_w, src_h, sx_fp, sy_fp);
220 }
221 }
222
223 return k_ra8_ok;
224}
