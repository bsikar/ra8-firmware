154static ra8_err_t ra8_io_compress_drive(tdefl_compressor* d,
155 compress_ctx_t* ctx,
156 const uint8_t* src,
157 uint32_t src_len,
158 uint32_t* out_len)
159{
160 if (tdefl_init(d, compress_put, ctx, (int)TDEFL_DEFAULT_MAX_PROBES) != TDEFL_STATUS_OKAY) {
161 return k_ra8_fail;
162 }
163 const tdefl_status st = tdefl_compress_buffer(d, src, (size_t)src_len, TDEFL_FINISH);
164 if (ctx->overflow != 0U) {
165 return k_ra8_err_no_mem;
166 }
167 if (st != TDEFL_STATUS_DONE) {
168 return k_ra8_fail;
169 }
170 *out_len = ctx->len;
171 return k_ra8_ok;
172}
