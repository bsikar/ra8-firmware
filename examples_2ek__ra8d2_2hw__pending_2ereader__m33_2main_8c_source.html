142static void banner_append(char* dst, uint32_t* off, uint32_t cap, const char* src)
143{
144 if (dst == nullptr) {
145 return;
146 }
147 if (off == nullptr) {
148 return;
149 }
150 if (src == nullptr) {
151 return;
152 }
153 RA8_LOOP_BOUND(k_banner_cap);
154 for (uint32_t i = 0U; i < cap; i++) {
155 const char c = src[i];
156 if (c == '\0') {
157 break;
158 }
159 if (*off >= (cap - 1U)) {
160 break;
161 }
162 dst[*off] = c;
163 *off += 1U;
164 }

185static void banner_append_hex(char* dst, uint32_t* off, uint32_t cap, uint32_t value)
186{
187 if (dst == nullptr) {
188 return;
189 }
190 if (off == nullptr) {
191 return;
192 }
193 static const char digits[] = "0123456789ABCDEF";
194 RA8_LOOP_BOUND(k_hex_nibbles);
195 for (uint32_t i = 0U; i < (uint32_t)k_hex_nibbles; i++) {
196 if (*off >= (cap - 1U)) {
197 break;
198 }
199 const uint32_t shift = ((uint32_t)k_hex_nibbles - 1U - i) * (uint32_t)k_nibble_bits;
200 const uint32_t nibble = (value >> shift) & (uint32_t)k_nibble_mask;
201 dst[*off] = digits[nibble];
202 *off += 1U;
203 }
204}

228static void banner_append_u32(char* dst, uint32_t* off, uint32_t cap, uint32_t value)
229{
230 if (dst == nullptr) {
231 return;
232 }
233 if (off == nullptr) {
234 return;
235 }
236 char tmp[k_dec_digits_max];
237 uint32_t count = 0U;
238 uint32_t v = value;
239 RA8_LOOP_BOUND(k_dec_digits_max);
240 for (uint32_t i = 0U; i < (uint32_t)k_dec_digits_max; i++) {
241 tmp[count] = (char)('0' + (char)(v % (uint32_t)k_dec_radix));
242 count += 1U;
243 v /= (uint32_t)k_dec_radix;
244 if (v == 0U) {
245 break;
246 }
247 }
248 RA8_LOOP_BOUND(k_dec_digits_max);
249 for (uint32_t i = 0U; i < count; i++) {
250 if (*off >= (cap - 1U)) {
251 break;
252 }
253 dst[*off] = tmp[count - 1U - i];
254 *off += 1U;
255 }

738static bool run_handoff_cycle(volatile erm33_mailbox_t* mb)
739{
740 if (mb == nullptr) {
741 return false;
742 }
743 RA8_LOOP_BOUND(k_erm33_max_turns);
744 for (uint32_t turn = 1U; turn <= (uint32_t)k_erm33_max_turns; turn++) {
745 m85_gate_hoco(true);
746 const bool woke = m85_wait_turn(mb, turn);
747 m85_gate_hoco(false);
748 if (!woke) {
749 ra8_log_info_val("M85", "page-turn wake timed out at turn", turn);
750 return false;
751 }
752 [[maybe_unused]] const uint32_t work = m85_heavy_work(turn);
753 mb->turn_ack = turn;
754 __asm volatile("dsb" ::: "memory");
755 ra8_log_info_val("M85", "woke from low-power for page turn", turn);
756 ra8_log_info_val("M85", "M85 heavy next-page work result", work);
757 }
758 return true;
759}

819static void run_mode_switch(volatile erm33_mailbox_t* mb)
820{
821 if (mb == nullptr) {
822 return;
823 }
824 ra8_log_info("M85", "entering #150 mode-switch cycle; M33 holds the page + polls touch");
825 const bool cycled = run_handoff_cycle(mb);
826 bool done_ok = false;
827 if (cycled) {
828 done_ok = m85_wait_turn_done(mb, (uint32_t)k_erm33_max_turns);

830 bool verdict = false;

903int main(void)
904{
905 ra8_log_init();
906 ra8_log_info("M85", "==== RA8D2 ereader_m33 demo (#150 M85-park / M33-hold) ====");
907 ra8_log_info("M85", "Cortex-M85 primary core online");
908 ra8_log_info("M85", "M33 renders a held page into external SDRAM (0x68000000)");
909
910 volatile erm33_mailbox_t* mb = erm33_mailbox();
911 prep_mailbox(mb);
912
913 /* Arm the IPC0 receive IRQ + configure the LPM block BEFORE releasing the M33,
914 * so the page-turn wake path and the Sleep-mode posture are ready the moment
915 * the M33 starts running. A wake-arm failure is non-fatal: m85_wait_turn still
916 * bounds-polls the mailbox (its WFI then just sleeps to the next interrupt). */
917 if (arm_ipc_wake()) {
918 ra8_log_info("M85", "IPC0 receive IRQ armed -- M85 can WFI-wake on a page turn");
919 } else {
920 ra8_log_info("M85", "IPC wake arm failed -- page-turn WFI falls back to bounded poll");
921 }
922 m85_lpm_configure();
923
924 ra8_log_info("M85", "releasing Cortex-M33 secondary core ...");
925 const ra8_err_t err = ra8_cpu1_release(&g_ra8_ls_cpu1_mram_start, &g_ra8_ls_cpu1_stack_top);
926 ra8_log_info_val("M85", "ra8_cpu1_release rc (0 = ok)", (uint32_t)err);
927 if (err != k_ra8_ok) {
928 ra8_log_info("M85", "release FAILED -- halting");
929 park_forever();
930 }
931
932 if (wait_for_m33_sig(mb)) {
933 ra8_log_info("M85", "M33 reader is alive");
934 } else {
935 ra8_log_info("M85", "M33 signature not seen -- did it boot?");
936 }
937
938 ra8_log_info("M85", "M85 idle; waiting for the M33 to render the held page ...");
939 if (!wait_for_done(mb)) {
940 ra8_log_info("M85", "M33 done flag not seen -- timed out");
941 park_forever();
942 }
943 __asm volatile("dsb" ::: "memory");
944
945 ra8_log_info_val("M85", "M33 published fb_base", mb->fb_base);
946 ra8_log_info_val("M85", "M33 held-page glyphs", mb->glyph_count);
947 ra8_log_info_val("M85", "M33 held-page pixels CRC32 (decimal)", mb->fb_crc);
948 emit_verdict(mb->fb_crc, verify_page(mb));
949
950 /* #150 mode-switch: hand the live core to the slow M33, park, and wake on each
951 * scripted page turn to do the heavy next-page work, then park for good. */
952 run_mode_switch(mb);
953 park_low_power();
954}
