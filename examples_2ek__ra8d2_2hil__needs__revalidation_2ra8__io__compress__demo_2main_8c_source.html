232static ra8_err_t demo_roundtrip(const char* prefix, uint32_t* out_blob_len)
233{
234 RA8_CHECK_NULL_PTR(prefix, s_tag, "prefix");
235 RA8_CHECK_NULL_PTR(out_blob_len, s_tag, "out_blob_len");
236
237 char path[(size_t)k_ra8_io_vfs_name_max + sizeof("STORY.RBK") + 1U] = {};
238 (void)strncpy(path, prefix, sizeof(path) - 1U);
239 (void)strncat(path, "STORY.RBK", sizeof(path) - strlen(path) - 1U);
240
241 uint32_t blob_len = 0;
242 RA8_RETURN_ON_ERROR(ra8_io_vfs_write_compressed(path,
243 s_payload,
244 (uint32_t)k_demo_payload,
245 s_scratch,
246 (uint32_t)k_ra8_io_compress_scratch_bytes,
247 s_blob,
248 (uint32_t)k_demo_blob_cap,
249 &blob_len),
250 s_tag,
251 "write compressed");
252
253 uint32_t out_len = 0;
254 RA8_RETURN_ON_ERROR(ra8_io_vfs_read_compressed(path,
255 s_blob,
256 (uint32_t)k_demo_blob_cap,
257 s_restored,
258 (uint32_t)k_demo_payload,
259 &out_len),
260 s_tag,
261 "read compressed");
262 if (out_len != (uint32_t)k_demo_payload) {
263 return k_ra8_err_invalid_size;
264 }
265 if (memcmp(s_restored, s_payload, sizeof(s_payload)) != 0) {
266 return k_ra8_err_checksum_mismatch;
267 }
268 *out_blob_len = blob_len;
269 return k_ra8_ok;
270}

295static ra8_err_t demo_run(uint32_t* out_blob_len)
296{
297 RA8_CHECK_NULL_PTR(out_blob_len, s_tag, "out_blob_len");
298
299 for (uint32_t i = 0; i < (uint32_t)k_demo_payload; ++i) {
300 s_payload[i] = (uint8_t)((i * (uint32_t)k_demo_seed_mul) % (uint32_t)k_demo_pattern_mod);
301 }
302
303 /* Backend 1: RAM disk over an in-SRAM buffer, mounted as "ram". */
304 RA8_RETURN_ON_ERROR(
305 ra8_io_blockdev_ram_init(&s_bd_ram, &s_bstate_ram, s_disk, (uint32_t)k_demo_disk_blocks, false),
306 s_tag,
307 "ram blockdev init");
308 RA8_RETURN_ON_ERROR(demo_mount(&s_bd_ram, &s_be_ram, "ram"), s_tag, "ram mount");
309 uint32_t ram_blob_len = 0;
310 RA8_RETURN_ON_ERROR(demo_roundtrip("ram:/", &ram_blob_len), s_tag, "ram round-trip");
311
312 /* Backend 2: external SDRAM as a volatile ramdisk, mounted as "dr". */
313 RA8_RETURN_ON_ERROR(
314 ra8_io_blockdev_sdram_init(&s_bd_dr, &s_bstate_dr, (uint32_t)k_demo_disk_blocks),
315 s_tag,
316 "sdram blockdev init");
317 RA8_RETURN_ON_ERROR(demo_mount(&s_bd_dr, &s_be_dr, "dr"), s_tag, "dr mount");
318 uint32_t dr_blob_len = 0;
319 RA8_RETURN_ON_ERROR(demo_roundtrip("dr:/", &dr_blob_len), s_tag, "dr round-trip");
320
321 *out_blob_len = ram_blob_len;
322 return k_ra8_ok;
323}
