115static bool internal_button_on_input(ra8_widget_t* w, const ra8_widget_event_t* ev)
116{
117 ra8_widget_button_t* b = (ra8_widget_button_t*)w->ctx;
118 if (b == nullptr) {
119 return false;
120 }
121 if (ev->kind != k_ra8_widget_ev_touch) {
122 return false;
123 }
124 b->pressed = !b->pressed;
125 b->presses++;
126 (void)ra8_widget_invalidate(w, k_ra8_widget_refresh_fast);
127 if (b->on_press != nullptr) {
128 b->on_press(w);
129 }
130 return true;
131}
