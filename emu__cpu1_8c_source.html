67void emu_cpu1_notify_mmio_write(uint64_t mmio_abs, uint64_t value)
68{
69 if (mmio_abs == (uint64_t)k_cpu1_initvtor_addr) {
70 s_cpu1_initvtor = (uint32_t)value;
71 } else if (mmio_abs == (uint64_t)k_cpu1_actcsr_addr) {
72 /* HUM Ch 2.9.1.9 "CPU1ACTCSR" p 130 -- a write is honored only with
73 * KEY=0xA5 in bits [15:8]; the firmware writes 0xA501 (KEY | ACTREQ).
74 * Model the key gate so a keyless ACTREQ does not release cpu1. */
75 const uint32_t key =
76 ((uint32_t)value >> (uint32_t)k_cpu1_actcsr_key_shift) & (uint32_t)k_cpu1_actcsr_key_mask;
77 if ((key == (uint32_t)k_cpu1_actcsr_key) &&
78 (((uint32_t)value & (uint32_t)k_cpu1_actcsr_actreq) != 0U)) {
79 s_cpu1_release_req = true;
80 }
81 }
82}

91static bool cpu1_image_present(const uint8_t* elf)
92{
93 uint32_t phoff = 0U;
94 (void)memcpy(&phoff, elf + (uint32_t)k_elf_e_phoff_off, 4);
95 const uint16_t phentsize = (uint16_t)(elf[42] | (elf[43] << 8));
96 const uint16_t phnum = (uint16_t)(elf[44] | (elf[45] << 8));
97 for (uint16_t i = 0U; i < phnum; i++) {
98 const uint8_t* ph = elf + phoff + ((size_t)(uint32_t)i * phentsize);
99 uint32_t p_type;
100 uint32_t p_paddr;
101 (void)memcpy(&p_type, ph + 0, 4);
102 (void)memcpy(&p_paddr, ph + (uint32_t)k_elf_ph_paddr_off, 4);
103 if ((p_type == 1U) && (p_paddr >= (uint32_t)k_cpu1_mram_base) &&
104 (p_paddr < (uint32_t)k_cpu1_mram_end)) {
105 return true;
106 }
107 }
108 return false;

112static bool cpu1_map_regions(uc_engine* c1)
113{
114 uint32_t region_n = 0U;
115 const mem_region_t* regions = emu_memmap_regions(&region_n);
116 for (size_t i = 0U; i < (size_t)region_n; i++) {
117 uc_err mr = UC_ERR_OK;
118 if (regions[i].base == (uint64_t)k_sram_base) {
119 mr = uc_mem_map_ptr(c1,
120 regions[i].base,
121 (size_t)regions[i].size,
122 UC_PROT_ALL,
123 emu_memmap_sram_buf());
124 } else if (regions[i].base == (uint64_t)k_ns_sram2_base) {
125 /* SRAM2 Non-secure alias: the SAME physical bytes as 0x22100000, so back
126 * it with the shared SRAM host buffer at that offset (mirroring cpu0's
127 * map). cpu1 stamps its bring-up markers through this alias
128 * (cpu1_pingpong_ipc's cpu1_main writes 0x321002xx); a private mapping
129 * here would hide them from cpu0 and the --dump-sym probe. */
130 uint8_t* const ns_host =
131 emu_memmap_sram_buf() +
132 ((size_t)k_ns_sram2_base - (size_t)k_ns_alias_bit - (size_t)k_sram_base);
133 mr = uc_mem_map_ptr(c1, regions[i].base, (size_t)regions[i].size, UC_PROT_ALL, ns_host);
134 } else {
135 mr = uc_mem_map(c1, regions[i].base, (size_t)regions[i].size, UC_PROT_ALL);
136 }
137 if (mr != UC_ERR_OK) {
138 return false;
139 }
140 }
141 return true;
142}

145static bool cpu1_map_periph(uc_engine* c1)
146{
147 /* Map the Renesas peripheral space through the SAME board_periph models cpu0
148 * uses -- the peripherals are shared hardware, so the M33's MMIO (GPIO LED
149 * toggles, SCI, timers, ...) reaches the models and the board view instead of
150 * faulting on unmapped space. cpu0 maps this region separately via uc_mmio_map
151 * (it is absent from k_regions); cpu1 needs the identical mapping. */
152 if (uc_mmio_map(c1,
153 (uint64_t)k_periph_base,
154 (size_t)k_periph_size,
155 mmio_read,
156 nullptr,
157 mmio_write,
158 nullptr) != UC_ERR_OK) {
159 return false;
160 }
161 /* The M33 runs as a permanent non-secure bus controller (SECEXT off): the
162 * peripheral channels it owns -- e.g. the IPC unit it pokes to wake the M85 --
163 * are NS-attributed, so its accesses route through the bit[28] non-secure
164 * alias (0x5xxxxxxx) rather than the Secure 0x4xxxxxxx the M85 uses. Map that
165 * alias to the SAME models with the SAME hooks: the hooks rebuild the absolute
166 * address as k_periph_base + window-relative offset, so a 0x50020000 access
167 * (offset 0x20000) dispatches to 0x40020000 identically to a Secure one. */
168 return uc_mmio_map(c1,
169 (uint64_t)k_periph_base | (uint64_t)k_ns_alias_bit,
170 (size_t)k_periph_size,
171 mmio_read,
172 nullptr,
173 mmio_write,
174 nullptr) == UC_ERR_OK;
175}

203static uc_engine* cpu1_engine_init(const uint8_t* elf, long elf_len)
204{
205 if (!cpu1_image_present(elf)) {
206 return nullptr; /* single-core firmware -- no second engine. */
207 }
208 uc_engine* c1 = nullptr;
209 if (uc_open(UC_ARCH_ARM, (uc_mode)(UC_MODE_THUMB | UC_MODE_MCLASS), &c1) != UC_ERR_OK) {
210 return nullptr;
211 }
212 (void)uc_ctl_set_cpu_model(c1, UC_CPU_ARM_CORTEX_M33);
213 if (!cpu1_map_regions(c1) || !cpu1_map_periph(c1) || (load_elf(c1, elf, elf_len) != 0)) {
214 (void)uc_close(c1);
215 return nullptr;
216 }
217 (void)fprintf(stderr, " cpu1 engine : Cortex-M33, shared SRAM + peripherals (dual-core)\n");
218 return c1;
219}

228void emu_cpu1_step(void)
229{
230 if (s_cpu1_uc == nullptr) {
231 return;
232 }
233 if (s_cpu1_release_req && !s_cpu1_active) {
234 s_cpu1_release_req = false;
235 const uint32_t cpu1_sp = rd32(s_cpu1_uc, (uint64_t)s_cpu1_initvtor);
236 s_cpu1_pc = rd32(s_cpu1_uc, (uint64_t)s_cpu1_initvtor + 4U);
237 (void)uc_reg_write(s_cpu1_uc, UC_ARM_REG_SP, &cpu1_sp);
238 s_cpu1_active = true;
239 }
240 if (s_cpu1_active) {
241 const uc_err e1 =
242 uc_emu_start(s_cpu1_uc, (uint64_t)s_cpu1_pc | 1U, 0, 0, (size_t)k_cpu1_chunk_insns);
243 (void)uc_reg_read(s_cpu1_uc, UC_ARM_REG_PC, &s_cpu1_pc);
244 if (e1 != UC_ERR_OK) {
245 s_cpu1_active = false; /* cpu1 hit a fault -- halt it, cpu0 continues */
246 }
247 }
