90static void wdt_tick(uc_engine* uc)
91{
92 (void)uc;
93 if (!s_wdt.armed || s_wdt.fired) {
94 return;
95 }
96 if (s_wdt.counter > 0U) {
97 s_wdt.counter--;
98 return;
99 }
100 /* Underflow. Latch UNDFF; if configured for reset, request a warm reboot with
101 * the watchdog-reset cause and disarm so it fires exactly once. */
102 s_wdt.undff = (uint16_t)k_wdt_sr_undff;
103 if ((s_wdt.wdtrcr & (uint8_t)k_wdt_rcr_rstirqs) != 0U) {
104 s_wdt.fired = true;
105 s_wdt.armed = false;
106 board_periph_reset_request_reboot(true, false);
107 }
108}

117static uint64_t wdt_read(uc_engine* uc, uint64_t addr, unsigned size)
118{
119 (void)uc;
120 (void)size;
121 const uint64_t off = addr - (uint64_t)k_wdt_base;
122 if (off == (uint64_t)k_wdt_off_wdtcr) {
123 return s_wdt.wdtcr;
124 }
125 if (off == (uint64_t)k_wdt_off_wdtsr) {
126 const uint16_t cnt = (uint16_t)(s_wdt.counter & (uint32_t)k_wdt_sr_cntval);
127 return (uint64_t)(cnt | s_wdt.undff);
128 }
129 if (off == (uint64_t)k_wdt_off_wdtrcr) {
130 return s_wdt.wdtrcr;
131 }
132 return 0U;
133}

136static void wdt_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
137{
138 (void)uc;
139 (void)size;
140 const uint64_t off = addr - (uint64_t)k_wdt_base;
141 if (off == (uint64_t)k_wdt_off_wdtrr) {
142 const uint8_t b = (uint8_t)value;
143 /* The 0x00 then 0xFF sequence refreshes: arm (first time) and reload. */
144 if ((s_wdt.last_rr == (uint8_t)k_wdt_refresh_a) && (b == (uint8_t)k_wdt_refresh_b)) {
145 s_wdt.armed = true;
146 s_wdt.counter = (uint32_t)k_wdt_timeout_ticks;
147 s_wdt.undff = 0U;
148 s_wdt.refreshes++;
149 }
150 s_wdt.last_rr = b;
151 } else if (off == (uint64_t)k_wdt_off_wdtcr) {
152 s_wdt.wdtcr = (uint16_t)value;
153 } else if (off == (uint64_t)k_wdt_off_wdtrcr) {
154 s_wdt.wdtrcr = (uint8_t)value;
155 }
156}

159static void wdt_report(void)
160{
161 if ((s_wdt.refreshes == 0U) && !s_wdt.fired) {
162 return; /* Untouched: stay quiet. */
163 }
164 (void)fprintf(stderr,
165 " WDT0 : refreshes=%u armed=%u reset_fired=%u\n",
166 s_wdt.refreshes,
167 (unsigned)s_wdt.armed,
168 (unsigned)s_wdt.fired);
169}
