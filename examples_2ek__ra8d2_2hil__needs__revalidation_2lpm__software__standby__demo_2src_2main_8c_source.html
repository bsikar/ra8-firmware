185RA8_INTERNAL static void internal_lpm_swstd_add_offset(const ra8_rtc_datetime_t* now,
186 ra8_rtc_datetime_t* out)
187{
188 *out = *now;
189 const uint16_t s = (uint16_t)now->second + (uint16_t)k_lpm_swstd_alarm_offset_s;
190 out->second = (uint8_t)(s % (uint16_t)k_lpm_swstd_secs_per_min);
191 const uint16_t carm = (uint16_t)(s / (uint16_t)k_lpm_swstd_secs_per_min);
192 const uint16_t m = (uint16_t)now->minute + carm;
193 out->minute = (uint8_t)(m % (uint16_t)k_lpm_swstd_mins_per_hour);
194 const uint16_t carh = (uint16_t)(m / (uint16_t)k_lpm_swstd_mins_per_hour);
195 const uint16_t h = (uint16_t)now->hour + carh;
196 out->hour = (uint8_t)(h % (uint16_t)k_lpm_swstd_hours_per_day);
197}

226[[nodiscard]] RA8_INTERNAL static ra8_err_t
227internal_lpm_swstd_arm_wake(const ra8_rtc_datetime_t* now)
228{
229 ra8_rtc_datetime_t a = {};
230 internal_lpm_swstd_add_offset(now, &a);
231 ra8_err_t err = ra8_rtc_set_alarm(&a);
232 if (err != k_ra8_ok) {
233 return err;
234 }
235 err = ra8_rtc_set_irq_enable((uint8_t)k_ra8_rtc_irq_alarm);
236 if (err != k_ra8_ok) {
237 return err;
238 }
239 return ra8_lpm_arm_wupen0_bits((uint32_t)k_ra8_lpm_wupen0_rtcalm);
240}

242void main(void)
243{
244 internal_lpm_swstd_setup_or_halt();
245 ra8_isr_globals_enable();
246
247 /* Emit the boot banner *before* the standby entry so the HIL gate
248 * confirms the firmware booted even when the sub-clock crystal is
249 * silent (see the SOSC caveat in the file header). */
250 (void)ra8_board_uart_console_write(s_lpm_swstd_boot_msg,
251 (size_t)(sizeof(s_lpm_swstd_boot_msg) - 1U));
252
253 while (1) {
254 ra8_rtc_datetime_t now = {};
255 if (ra8_rtc_get(&now) != k_ra8_ok) {
256 break;
257 }
258 if (internal_lpm_swstd_arm_wake(&now) != k_ra8_ok) {
259 break;
260 }
261 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_software_std) != k_ra8_ok) {
262 break;
263 }
264 g_lpm_swstd_wake_count++;
265 /* Clear the alarm flag so the next cycle starts from a known
266 * state. Errors here are non-fatal -- the loop body will simply
267 * re-arm the alarm. */
268 (void)ra8_rtc_clear_status((uint8_t)k_ra8_rtc_irq_alarm);
269 }
270 internal_lpm_swstd_panic_halt();
271}
