97static void priv_add(ra8_kbd_layout_t* kb,
98 int32_t x,

100 int32_t y,
101 int32_t h,

186static void priv_place(ra8_kbd_layout_t* kb,
187 const char* lo,
188 const char* hi,
189 int32_t n,
190 int32_t hu0,
191 const ra8_ui_rect_t* f,
192 int32_t y,
193 int32_t rh)
194{

196 const int32_t x0 = priv_hx(f, hu0 + (i * (int32_t)k_kbd_key_hu));
197 const int32_t x1 = priv_hx(f, hu0 + ((i + 1) * (int32_t)k_kbd_key_hu));

200 x1 - x0,
201 y,
202 rh,
203 (char)lo[i],
204 (char)((hi != nullptr) ? hi[i] : lo[i]),

206 0U);

520static void priv_build_layer(ra8_kbd_layout_t* kb)
521{
522 kb->count = 0U;
523 const int32_t rh = kb->frame.h / (int32_t)k_ra8_kbd_rows;
524 if (kb->layer == (uint8_t)k_ra8_kbd_layer_numbers) {
525 priv_build_numbers(kb, &kb->frame, rh);
526 } else if (kb->layer == (uint8_t)k_ra8_kbd_layer_symbols) {
527 priv_build_symbols(kb, &kb->frame, rh);
528 } else {
529 priv_build_letters(kb, &kb->frame, rh);
530 }

618[[nodiscard]] ra8_err_t ra8_kbd_apply(ra8_kbd_text_t* t, ra8_kbd_layout_t* kb, uint8_t key_idx)
619{
620 RA8_CHECK_NULL_PTR(t, s_tag, "t must not be nullptr");
621 RA8_CHECK_NULL_PTR(kb, s_tag, "kb must not be nullptr");
622 if (key_idx >= kb->count) {
623 return k_ra8_ok; /* no-hit sentinel or out of range -> no-op */
624 }
625 const ra8_kbd_key_t* k = &kb->keys[key_idx];
626 switch (k->kind) {
627 case k_ra8_kbd_key_char:
628 priv_append(t, (char)(kb->shift ? k->ch_upper : k->ch_lower));
629 kb->shift = false; /* one-shot SHIFT clears after a character */
630 break;
631 case k_ra8_kbd_key_space:
632 priv_append(t, ' ');
633 kb->shift = false;
634 break;
635 case k_ra8_kbd_key_backspace:
636 if (t->len > 0U) {
637 t->len--;
638 t->buf[t->len] = '\0';
639 }
640 break;
641 case k_ra8_kbd_key_enter:
642 t->committed = true;
643 break;
644 case k_ra8_kbd_key_shift:
645 kb->shift = !kb->shift;
646 break;
647 case k_ra8_kbd_key_layer:
648 kb->layer = k->aux;
649 kb->shift = false;
650 priv_build_layer(kb); /* re-lay the grid for the new layer */
651 break;
652 default:
653 break;
654 }
655 return k_ra8_ok;
656}
