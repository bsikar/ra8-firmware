178static VOID selftest_host_worker(ULONG arg)
179{
180 (void)arg;
181
182 tx_thread_sleep(k_selftest_boot_wait_ticks);
183 for (;;) {
184 const ra8_err_t err = selftest_host_pass();
185 if (err == k_ra8_ok) {
186 break;
187 }
188 tx_thread_sleep(k_selftest_retry_ticks);
189 }
190 while (1) {
191 tx_thread_sleep(k_selftest_idle_ticks);
192 }
193}

280static void selftest_route_usb_or_halt(void)
281{

292 selftest_panic_halt();

296 selftest_panic_halt();

303 selftest_panic_halt();

326static void selftest_setup_or_halt(void)

339 selftest_panic_halt();

342 selftest_panic_halt();

345 selftest_panic_halt();

348 selftest_panic_halt();

351 selftest_panic_halt();
