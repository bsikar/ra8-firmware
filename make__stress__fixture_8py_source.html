4"""Generate epub_stress_fixture.h, a synthetic large-STRUCTURE EPUB3.
5
6Reproduces the on-device miniz+tinyxml2 shared-arena pressure of a big
7real-world book (#144 bug 1) without shipping a copyrighted 7 MB novel.
8
9The insight the fixture is built on: pool pressure during `ra8_epub_open()`
10comes from miniz's central directory, proportional to the archive's FILE COUNT,
11and from tinyxml2's DOM of the OPF, proportional to the number of
12manifest/spine items. It does NOT come from total byte size. So this packs many
13tiny files instead of a few large ones -- 60 chapters (the spine, deliberately
14under the k_ra8_epub_max_chapters=64 cap), 60 extra manifest-only resources, an
15NCX with 60 navPoints, and a cover. That is ~120 archive entries and a ~20 KB
16OPF, comparable to the 108-file, 41-chapter real book, in tens of KB total. It
17therefore bakes into MRAM and opens in memory like epub_parse.
18
19Raise the byte size and the fixture gets no more stressful; raise the entry
20count and it does.
21
22Output is pure 7-bit ASCII, 16 bytes per row inside a clang-format-off guard.
23
24Usage:
25 python3 make_stress_fixture.py
26"""
27

49def build_opf() -> str:
50 """Build the OPF package document -- the main tinyxml2 DOM pressure source.
51
52 Emits every chapter as both a manifest item and a spine itemref, plus the
53 manifest-only CSS resources that inflate the file count without adding
54 spine entries. The resulting element count is what the on-device tinyxml2
55 DOM has to hold, so this function's output size IS the stress being tested.
56
57 Declares EPUB3 `properties="cover-image"` and the legacy
58 `<meta name="cover">` both, so the fixture exercises either cover-resolution
59 path a reader might take.
60
61 Returns:
62 The complete OPF as ASCII-safe text.
63 """
64 items = [
65 ' <item id="ncx" href="toc.ncx" media-type="application/x-dtbncx+xml"/>',
66 ' <item id="cover" href="cover.png" media-type="image/png" properties="cover-image"/>',
67 ]
68 spine = []
69 for i in range(N_CHAPTERS):
70 items.append(
71 f' <item id="ch{i}" href="ch{i}.xhtml" media-type="application/xhtml+xml"/>'
72 )
73 spine.append(f' <itemref idref="ch{i}"/>')
74 items.extend(
75 f' <item id="res{i}" href="res{i}.css" media-type="text/css"/>'
76 for i in range(N_RESOURCES)
77 )
78 return (
79 '<?xml version="1.0" encoding="UTF-8"?>\n'
80 '<package xmlns="http://www.idpf.org/2007/opf" version="3.0" '
81 'unique-identifier="id">\n'
82 ' <metadata xmlns:dc="http://purl.org/dc/elements/1.1/">\n'
83 " <dc:title>Stress Structure Book</dc:title>\n"
84 " <dc:creator>Brighton Sikarskie</dc:creator>\n"
85 " <dc:language>en</dc:language>\n"
86 ' <dc:identifier id="id">urn:test:stress</dc:identifier>\n'
87 ' <meta name="cover" content="cover"/>\n'
88 " </metadata>\n"
89 " <manifest>\n" + "\n".join(items) + "\n </manifest>\n"
90 ' <spine toc="ncx">\n' + "\n".join(spine) + "\n </spine>\n"
91 "</package>\n"
92 )

95def build_ncx() -> str:
96 """Build the EPUB2 NCX table of contents, one navPoint per chapter.
97
98 An EPUB3 fixture does not need an NCX, and that is exactly why it is here:
99 real books ship one for backward compatibility, and it adds another
100 60-element XML document the parser must handle. `playOrder` is 1-based
101 while the chapter hrefs are 0-based, matching how real books number them.
102
103 Returns:
104 The complete NCX as ASCII-safe text.
105 """
106 points = [
107 f' <navPoint id="np{i}" playOrder="{i + 1}">\n'
108 f" <navLabel><text>Chapter {i + 1}</text></navLabel>\n"
109 f' <content src="ch{i}.xhtml"/>\n'
110 " </navPoint>"
111 for i in range(N_CHAPTERS)
112 ]
113 return (
114 '<?xml version="1.0" encoding="UTF-8"?>\n'
115 '<ncx xmlns="http://www.daisy.org/z3986/2005/ncx/" version="2005-1">\n'
116 ' <head><meta name="dtb:uid" content="urn:test:stress"/></head>\n'
117 " <docTitle><text>Stress Structure Book</text></docTitle>\n"
118 " <navMap>\n" + "\n".join(points) + "\n </navMap>\n"
119 "</ncx>\n"
120 )
121

139def build_epub() -> bytes:
140 """Assemble the whole EPUB archive in memory and return its bytes.
141
142 Byte-for-byte reproducible: every entry is written with a fixed 2026-01-01
143 timestamp and fixed permissions, so re-running the generator on a clean tree
144 produces an identical header and an empty diff. Without the pinned
145 timestamps the baked fixture would churn on every regeneration.
146
147 Two entries are ZIP_STORED rather than deflated. `mimetype` must be stored
148 first and uncompressed per the EPUB spec, and the 1x1 cover PNG is stored
149 because deflating an already-compressed PNG would only grow it.
150
151 Returns:
152 The complete .epub archive as bytes.
153 """
154 out = io.BytesIO()
155 fixed = (2026, 1, 1, 0, 0, 0)
156
157 def add(zf: zipfile.ZipFile, name: str, data: bytes | str, store: bool = False) -> None:
158 info = zipfile.ZipInfo(name, date_time=fixed)
159 info.compress_type = zipfile.ZIP_STORED if store else zipfile.ZIP_DEFLATED
160 info.external_attr = 0o600 << 16
161 zf.writestr(info, data)
162
163 with zipfile.ZipFile(out, "w") as zf:
164 add(zf, "mimetype", b"application/epub+zip", store=True)
165 add(zf, "META-INF/container.xml", CONTAINER_XML.encode("ascii"))
166 add(zf, "OEBPS/content.opf", build_opf().encode("ascii"))
167 add(zf, "OEBPS/toc.ncx", build_ncx().encode("ascii"))
168 add(zf, "OEBPS/cover.png", COVER_PNG, store=True)
169 for i in range(N_CHAPTERS):
170 body = (
171 f'<?xml version="1.0"?><html><body><h1>Chapter {i + 1}</h1>'
172 f"<p>Body text for chapter {i + 1}.</p></body></html>"
173 )
174 add(zf, f"OEBPS/ch{i}.xhtml", body.encode("ascii"))
175 for i in range(N_RESOURCES):
176 add(zf, f"OEBPS/res{i}.css", f".c{i} {{ margin: 0; }}\n".encode("ascii"))
177 return out.getvalue()

180def bake_header(epub: bytes) -> str:
181 """Render the EPUB bytes as a C header with a `static const uint8_t` table.
182
183 The array is sized by a generated `enum : size_t` rather than a `[]`, so the
184 firmware gets the length as a compile-time constant and the two can never
185 disagree.
186
187 The byte table is wrapped in `clang-format off`/`on`. That is load-bearing,
188 not cosmetic: the formatter would otherwise reflow tens of thousands of
189 bytes into its own line width and the generator would no longer own the
190 file's shape, making every regeneration a large spurious diff.
191
192 Args:
193 epub: The archive bytes to bake, emitted 16 per row as `0xNN`.
194
195 Returns:
196 The complete header source, pure 7-bit ASCII.
197 """
198 rows = []
199 for i in range(0, len(epub), 16):
200 chunk = epub[i : i + 16]
201 rows.append(" " + ", ".join(f"0x{b:02X}" for b in chunk) + ",")
202 body = "\n".join(rows)
203 return (
204 "/**\n"
205 " * @file epub_stress_fixture.h\n"
206 " * @brief Baked synthetic large-structure EPUB3 for the #144 pool-stress gate.\n"
207 " *\n"
208 f" * @details {N_CHAPTERS} chapters + {N_RESOURCES} manifest resources + an NCX\n"
209 f" * with {N_CHAPTERS} navPoints + a cover -- ~{N_CHAPTERS + N_RESOURCES + 5} archive\n"
210 " * entries / a large OPF, reproducing a big real book's miniz+tinyxml2 shared-\n"
211 " * arena pressure in tens of KB. Pure ASCII.\n"
212 " *\n"
213 " * @generated by make_stress_fixture.py -- do not edit by hand; the generator\n"
214 " * owns this file's length (a baked byte table).\n"
215 " *\n"
216 " * @copyright Copyright (c) 2026 Brighton Sikarskie\n"
217 " * SPDX-License-Identifier: MIT\n"
218 " */\n"
219 "#pragma once\n"
220 "\n"
221 "#include <stddef.h>\n"
222 "#include <stdint.h>\n"
223 "\n"
224 "/** @brief Length of the baked stress EPUB blob, bytes. */\n"
225 f"enum : size_t {{ k_epub_stress_fixture_len = {len(epub)}U }};\n"
226 "\n"
227 "/** @brief Baked synthetic large-structure EPUB3 byte stream. */\n"
228 "/* clang-format off */\n"
229 "static const uint8_t k_epub_stress_fixture[k_epub_stress_fixture_len] = {\n"
230 f"{body}\n"
231 "};\n"
232 "/* clang-format on */\n"
233 )

236def main() -> int:
237 """Regenerate epub_stress_fixture.h in the CURRENT working directory.
238
239 The output path is relative, so this must be run from the app directory that
240 owns the fixture -- running it elsewhere silently writes a stray header
241 there instead of updating the committed one.
242
243 Writes with `encoding="ascii"`, so any non-ASCII that crept into the
244 generated text raises here rather than producing a file the encoding gate
245 would later reject.
246 """
247 epub = build_epub()
248 with Path("epub_stress_fixture.h").open("w", encoding="ascii") as f:
249 f.write(bake_header(epub))
250 print(f"wrote epub_stress_fixture.h ({len(epub)} epub bytes)")
