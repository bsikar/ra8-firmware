81 /* From SCSTLCTL+1 (0x0B4) through CNDCTL+RESERVED13 ending at 0x14F is
82 * 0x9C bytes = 39 words. Covers SVTDLG0, STCTL, ATCTL, ATTRG, ATCCNTE,
83 * CNDCTL and the surrounding RESERVED10..RESERVED13 padding. None of
84 * these registers are touched by the current driver -- the higher-
85 * level peripheral-mode + sync-timing card-stack consumer can carve them
86 * out of the gap when it lands. */
87 k_ra8_i3c_pad_after_scstlctl = 37U,
88 /* The 0x158..0x17B gap is 8 reserved words (RESERVED14[8]) before NIBIQP. */
89 k_ra8_i3c_pad_after_ntdtbp0 = 8U,
90 /* IBI-status-queue port immediately follows NIBIQP -> NRSQP at 0x180. */
91 /* From NRSQP+1 (0x184) up to NTST at 0x1E0 is 0x5C bytes = 23 words.
92 * Covers HCMDQP, HRSPQP, HTDTBP, NQTHCTL, NTBTHCTL0, NRQTHCTL,
93 * HQTHCTL, HTBTHCTL, BST, BSTE, BIE, BSTFC and intervening
94 * reserved padding -- none touched by the current driver. */
