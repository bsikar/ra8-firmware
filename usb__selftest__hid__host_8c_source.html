166[[nodiscard]] static ra8_err_t hid_enum_hunt(uint8_t* desc)
167{
168 ra8_delay_ms(k_hid_vbus_settle_ms);
169 const uint32_t t0 = ra8_time_ms();
170 for (uint32_t spin = 0U; spin < (uint32_t)k_hid_attach_spin; spin++) {
171 if (ra8_usb_host_line_state(k_ra8_usb_speed_hs) != 0U) {
172 break;
173 }
174 if ((ra8_time_ms() - t0) > (uint32_t)k_hid_attach_to_ms) {
175 break;
176 }
177 }
178 ra8_delay_ms(k_hid_debounce_ms);
179 ra8_err_t err = k_ra8_err_hw_timeout;
180 for (uint8_t attempt = 0U; attempt < (uint8_t)k_hid_enum_tries; attempt++) {
181 (void)ra8_usb_host_bus_reset(k_ra8_usb_speed_hs, true);
182 ra8_delay_ms(k_hid_reset_hold_ms);
183 (void)ra8_usb_host_bus_reset(k_ra8_usb_speed_hs, false);
184 (void)ra8_usb_host_set_uact(k_ra8_usb_speed_hs, true);
185 ra8_delay_ms(k_hid_recovery_ms);
186 (void)ra8_usb_host_set_target(k_ra8_usb_speed_hs, 0U);
187 err = hid_ctrl_get_dev_desc(desc);
188 if (err == k_ra8_ok) {
189 return k_ra8_ok;
190 }
191 }
192 return err;
193}

408[[nodiscard]] static ra8_err_t hid_read_round(uint32_t round)
409{
410 static uint8_t s_expect[k_hid_read_buf] = {};
411 static uint8_t s_rx[k_hid_read_buf] = {};
412 hid_fill_report_body(s_expect, (uint32_t)k_hid_report_len);
413 uint16_t rx = 0U;
414 const ra8_err_t err = ra8_usb_host_bulk_in(k_ra8_usb_speed_hs,
415 (uint8_t)k_hid_pipe_in,
416 s_rx,
417 (uint16_t)k_hid_read_buf,
418 &rx);
419 if (err != k_ra8_ok) {
420 (void)hid_print_fail("report read", err);
421 return err;
422 }
423 if (rx != (uint16_t)k_hid_report_len) {
424 s_dbg_mismatch = round;
425 (void)hid_print_fail("report length", k_ra8_err_invalid_size);
426 return k_ra8_err_invalid_size;
427 }
428 const size_t body = (size_t)((uint32_t)k_hid_report_len - (uint32_t)k_hid_body_idx);
429 if (memcmp(&s_rx[k_hid_body_idx], &s_expect[k_hid_body_idx], body) != 0) {
430 s_dbg_mismatch = round;
431 (void)hid_print_fail("report pattern mismatch", k_ra8_err_invalid_state);
432 return k_ra8_err_invalid_state;
433 }
434 s_dbg_last_seq = (uint32_t)s_rx[k_hid_seq_idx];
435 return k_ra8_ok;

455[[nodiscard]] static ra8_err_t hid_host_pass(void)
456{
457 s_dbg_phase = (uint32_t)k_hid_phase_init;
458 ra8_err_t err = hid_print("ra8d2 hid: host up on USB-HS, probing the loop...\r\n");
459 if (err != k_ra8_ok) {
460 return err;
461 }
462 err = ra8_usb_host_init(k_ra8_usb_speed_hs);
463 if (err != k_ra8_ok) {
464 (void)hid_print_fail("host init", err);
465 return err;
466 }
467
468 s_dbg_phase = (uint32_t)k_hid_phase_enum;
469 uint32_t pid = 0U;
470 err = hid_enumerate(&pid);
471 if (err != k_ra8_ok) {
472 (void)ra8_usb_host_deinit(k_ra8_usb_speed_hs);
473 return err;
474 }
475 s_dbg_pid = pid;
476 err = hid_print_enum(pid);
477 if (err != k_ra8_ok) {
478 return err;
479 }
480
481 s_dbg_phase = (uint32_t)k_hid_phase_verify;
482 s_dbg_rounds_ok = 0U;
483 for (uint32_t r = 0U; r < (uint32_t)k_hid_rounds; r++) {
484 err = hid_read_round(r);
485 if (err != k_ra8_ok) {
486 (void)ra8_usb_host_deinit(k_ra8_usb_speed_hs);
487 return err;
488 }
489 s_dbg_rounds_ok++;
490 }
491
492 s_dbg_phase = (uint32_t)k_hid_phase_pass;
493 s_dbg_pass_count++;
494 err = hid_print_pass();
495 if (err != k_ra8_ok) {
496 return err;
497 }
498 (void)ra8_board_led_on(k_ra8_board_led2);
499 return k_ra8_ok;
500}
