204RA8_HW_REGISTER_ACCESS
205static inline volatile r_cnecc_regs_t* ra8_cnecc(uint8_t instance)
206{
207 if ((uint16_t)instance >= k_ra8_cnecc_instance_count) {
208 return nullptr;
209 }
210 const uintptr_t base =
211 k_ra8_cnecc0_base_addr + ((uintptr_t)instance * (uintptr_t)k_ra8_cnecc_stride);
212 return (volatile r_cnecc_regs_t*)base;
213}
