434static ra8_err_t selftest_ospi_write_pattern(void)
435{
436 static UCHAR chunk[k_ospi_erase_sector] = {};
437 const uint32_t sec_per_erase = (uint32_t)k_ospi_erase_sector / (uint32_t)k_selftest_block_size;
438 for (uint32_t e = 0U; e < (uint32_t)k_ospi_erase_count; e++) {
439 const uint32_t addr = (uint32_t)k_ospi_test_offset + (e * (uint32_t)k_ospi_erase_sector);
440 const ra8_err_t err = ra8_xspi_flash_erase_sector((uint8_t)k_ospi_instance, addr);
441 if (err != k_ra8_ok) {
442 return err;
443 }
444 }
445 for (uint32_t e = 0U; e < (uint32_t)k_ospi_erase_count; e++) {
446 for (uint32_t s = 0U; s < sec_per_erase; s++) {
447 selftest_pattern_fill((e * sec_per_erase) + s, &chunk[s * (uint32_t)k_selftest_block_size]);
448 }
449 const uint32_t addr = (uint32_t)k_ospi_test_offset + (e * (uint32_t)k_ospi_erase_sector);
450 const ra8_err_t err =
451 ra8_xspi_flash_program((uint8_t)k_ospi_instance, addr, chunk, (uint32_t)k_ospi_erase_sector);
452 if (err != k_ra8_ok) {
453 return err;
454 }
455 }
456 return k_ra8_ok;
457}

478static ra8_err_t selftest_ospi_provision(void)

493 s_dbg_ospi_id = id;
494 err = selftest_ospi_write_pattern();
495 s_dbg_ospi_prov = (uint32_t)err;
496 return err;

519static VOID selftest_device_worker(ULONG arg)
520{
521 (void)arg;
522
523 if (selftest_ospi_provision() != k_ra8_ok) {
524 return;
525 }
526 if (selftest_usbx_stack_up() != UX_SUCCESS) {
527 return;
528 }
529 if (selftest_msc_class_register() != UX_SUCCESS) {
530 return;
531 }
532 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs) != k_ra8_ok) {
533 return;
534 }
535 if (ra8_usb_device_attach(k_ra8_usb_speed_fs, true) != k_ra8_ok) {
536 return;
537 }
538
539 /* Idle. USBX runs the SCSI/BBB state machine on its own threads. */
540 while (1) {
541 tx_thread_sleep(k_selftest_idle_ticks);
542 }
543}

563static VOID selftest_host_worker(ULONG arg)
564{

566

575 while (1) {
576 tx_thread_sleep(k_selftest_idle_ticks);
