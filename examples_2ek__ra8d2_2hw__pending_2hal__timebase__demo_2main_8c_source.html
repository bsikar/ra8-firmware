140static void demo_emit_u32(uint32_t value)
141{
142 uint8_t buf[k_demo_u32_digits];
143 uint32_t idx = (uint32_t)k_demo_u32_digits;
144
145 if (value == 0U) {
146 idx--;
147 buf[idx] = (uint8_t)'0';
148 } else {
149 for (uint32_t d = 0U; d < (uint32_t)k_demo_u32_digits; d++) {
150 if (value == 0U) {
151 break;
152 }
153 idx--;
154 buf[idx] = (uint8_t)('0' + (uint8_t)(value % (uint32_t)k_demo_dec_base));
155 value /= (uint32_t)k_demo_dec_base;
156 }
157 }
158
159 demo_emit(&buf[idx], (size_t)((uint32_t)k_demo_u32_digits - idx));
160}

186static ra8_err_t demo_setup_or_halt(uint32_t* out_cpuclk_hz)
187{
188 if (out_cpuclk_hz == nullptr) {
189 return k_ra8_err_invalid_arg;
190 }
191
192 uint32_t cpuclk0_hz = 0U;
193 if (ra8_cgc_init() != k_ra8_ok) {
194 demo_panic_halt();
195 }
196 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
197 demo_panic_halt();
198 }
199 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
200 demo_panic_halt();
201 }
202 if (ra8_board_uart_console_init((uint32_t)k_demo_baud) != k_ra8_ok) {
203 demo_panic_halt();
204 }
205 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
206 demo_panic_halt();
207 }
208
209 /* The timebase primitive owns the DWT enable and the SysTick arithmetic. */
210 ra8_dwt_cyccnt_enable();
211
212 uint32_t reload = 0U;
213 if (ra8_systick_reload_for(cpuclk0_hz, (uint32_t)k_demo_tick_hz, &reload) != k_ra8_ok) {
214 demo_panic_halt();
215 }
216 /* Re-arm the SAME SysTick via the HAL path: same reload + control bits the
217 * raw ra8_time_init() programmed, so ra8_delay_ms() keeps ticking. */
218 if (ra8_systick_configure(reload, k_ra8_systick_clk_cpu, true) != k_ra8_ok) {
219 demo_panic_halt();
220 }
221
222 demo_emit(k_demo_lbl_reload, sizeof(k_demo_lbl_reload) - 1U);
223 demo_emit_u32(reload);
224 demo_emit(k_demo_crlf, sizeof(k_demo_crlf) - 1U);
225
226 *out_cpuclk_hz = cpuclk0_hz;
227 return k_ra8_ok;
228}

248static void demo_emit_measurement(uint32_t cpuclk_hz)
249{
250 uint32_t cycles_per_us = cpuclk_hz / (uint32_t)k_demo_hz_per_mhz;
251 if (cycles_per_us == 0U) {
252 cycles_per_us = 1U;
253 }
254
255 ra8_dwt_cyccnt_reset();
256 ra8_delay_ms((uint32_t)k_demo_period_ms);
257 const uint32_t cycles = ra8_dwt_cyccnt_read();
258 const uint32_t micros = cycles / cycles_per_us;
259
260 demo_emit(k_demo_lbl_ms, sizeof(k_demo_lbl_ms) - 1U);
261 demo_emit_u32((uint32_t)k_demo_period_ms);
262 demo_emit(k_demo_lbl_cycles, sizeof(k_demo_lbl_cycles) - 1U);
263 demo_emit_u32(cycles);
264 demo_emit(k_demo_lbl_us, sizeof(k_demo_lbl_us) - 1U);
265 demo_emit_u32(micros);
266 demo_emit(k_demo_crlf, sizeof(k_demo_crlf) - 1U);
267}
