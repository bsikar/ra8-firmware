275[[nodiscard]] static ra8_err_t dma_coh_run_copy(void)
276{
277 /* clean-before: dirty src lines -> SRAM so the DMAC reads fresh bytes. */
278 if (ra8_cache_dcache_clean_by_addr(s_src, (uint32_t)sizeof(s_src)) != k_ra8_ok) {
279 return k_ra8_err_hw_error;
280 }
281
282 const ra8_dmac_config_t cfg = {
283 .src = (uint32_t)(uintptr_t)s_src,
284 .dst = (uint32_t)(uintptr_t)s_dst,
285 .count = (uint16_t)k_dma_coh_buf_words,
286 .block_count = 1U,
287 .width = k_ra8_dmac_width_word,
288 .src_inc = true,
289 .dst_inc = true,
290 };
291 ra8_err_t err = ra8_dmac_start_block((uint8_t)k_dma_coh_channel, &cfg);
292 if (err != k_ra8_ok) {
293 return err;
294 }
295
296 volatile r_dmac_channel_regs_t* reg = ra8_dmac((uint8_t)k_dma_coh_channel);
297 if (reg == nullptr) {
298 (void)ra8_dmac_stop((uint8_t)k_dma_coh_channel);
299 return k_ra8_err_hw_error;
300 }
301
302 /* Software trigger: assert DMREQ.SWREQ to request the block transfer.
303 * HUM Ch 17.2.15 p 744: write SWREQ=1; the bit auto-clears when the
304 * controller accepts the request (CLRS=0 is the reset default). */
305 reg->DMREQ = (uint8_t)k_ra8_dmreq_swreq_mask;
306
307 /* Poll DMSTS.ACT until the controller goes idle.
308 * HUM Ch 17.2.16 p 745: ACT clears when the transfer completes. */
309 bool done = false;
310 for (uint32_t i = 0U; i < (uint32_t)k_dma_coh_poll_limit; ++i) {
311 if ((reg->DMSTS & (uint8_t)k_ra8_dmsts_act_mask) == 0U) {
312 done = true;
313 break;
314 }
315 }
316 err = ra8_dmac_stop((uint8_t)k_dma_coh_channel);
317 if (!done) {
318 return k_ra8_err_hw_timeout;
319 }
320 if (err != k_ra8_ok) {
321 return err;
322 }
323
324 /* invalidate-after: drop stale dst lines so the verify reads SRAM. */
325 if (ra8_cache_dcache_invalidate_by_addr(s_dst, (uint32_t)sizeof(s_dst)) != k_ra8_ok) {
326 return k_ra8_err_hw_error;
327 }
328 return k_ra8_ok;
329}

344static void dma_coh_report(uint8_t ok)
