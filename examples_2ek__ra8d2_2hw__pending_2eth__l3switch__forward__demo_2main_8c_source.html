137static uint32_t l3_u32_to_dec(uint8_t* buf, uint32_t val)
138{
139 if (val == 0U) {
140 buf[0] = (uint8_t)'0';
141 return 1U;
142 }
143 uint8_t tmp[k_l3_dec_u32_max];
144 uint32_t n = 0U;
145 uint32_t v = val;
146 while (v != 0U) {
147 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint32_t)k_l3_radix));
148 v = v / (uint32_t)k_l3_radix;
149 n++;
150 }
151 for (uint32_t i = 0U; i < n; i++) {
152 buf[i] = tmp[n - 1U - i];
153 }
154 return n;
155}

190static bool l3_program_forwarding(void)
191{
192 bool ok = true;
193 const uint8_t masks[k_l3_mask_entries] = {
194 (uint8_t)k_l3_fwd_mask_all,
195 (uint8_t)k_l3_fwd_mask_all,
196 (uint8_t)k_l3_fwd_mask_all,
197 };
198 if (ra8_eth_mfwd_set_forwarding_masks(masks) != k_ra8_ok) {
199 ok = false;
200 }
201 if (ra8_eth_mfwd_route_queue((uint8_t)k_l3_src_port, (uint8_t)k_l3_rx_queue) != k_ra8_ok) {
202 ok = false;
203 }
204 uint32_t sts = 0U;
205 const ra8_err_t err = ra8_eth_mfwd_get_status(&sts);
206 if (err != k_ra8_ok) {
207 ok = false;
208 }
209 l3_write(k_l3_fwd_prefix, (uint32_t)(sizeof(k_l3_fwd_prefix) - 1U));
210 l3_write_u32(sts);
211 l3_write(k_l3_crlf, (uint32_t)(sizeof(k_l3_crlf) - 1U));
212 return ok;
213}

234static bool l3_run_facade(void)
235{
236 bool ok = true;
237 const ra8_layer3_switch_cfg_t cfg = {
238 .port_count = (uint8_t)k_l3_port_count,
239 .mtu_bytes = (uint16_t)k_l3_mtu_bytes,
240 .promiscuous = (uint8_t)k_l3_promisc_off,
241 };
242 if (ra8_layer3_switch_open(&cfg) != k_ra8_ok) {
243 ok = false;
244 }
245 uint8_t is_open = 0U;
246 uint8_t promisc = 0U;
247 const ra8_err_t st_err = ra8_layer3_switch_status_get(&is_open, &promisc);
248 if (st_err != k_ra8_ok) {
249 ok = false;
250 }
251 l3_write(k_l3_open_prefix, (uint32_t)(sizeof(k_l3_open_prefix) - 1U));
252 l3_write_u32((uint32_t)is_open);
253 l3_write(k_l3_promisc_sep, (uint32_t)(sizeof(k_l3_promisc_sep) - 1U));
254 l3_write_u32((uint32_t)promisc);
255 l3_write(k_l3_crlf, (uint32_t)(sizeof(k_l3_crlf) - 1U));
256
257 const ra8_layer3_switch_route_t route = {
258 .dst_ip = (uint32_t)k_l3_route_ip,
259 .mask = (uint32_t)k_l3_route_mask,
260 .egress_port = (uint8_t)k_l3_egress_port,
261 };
262 const ra8_err_t route_err = ra8_layer3_switch_route_add(&route);
263 l3_write(k_l3_route_prefix, (uint32_t)(sizeof(k_l3_route_prefix) - 1U));
264 l3_write_u32((uint32_t)route_err);
265 l3_write(k_l3_route_sfx, (uint32_t)(sizeof(k_l3_route_sfx) - 1U));
266
267 if (ra8_layer3_switch_close() != k_ra8_ok) {
268 ok = false;
269 }
270 return ok;
271}

353int32_t main(void)
354{
355 l3_setup_or_halt();
356 ra8_isr_globals_enable();
357
358 if (l3_arm() != k_ra8_ok) {
359 l3_panic_halt();
360 }
361
362 while (1) {
363 const bool healthy = l3_run_cycle();
364 if (healthy) {
365 l3_write(k_l3_verdict_pass, (uint32_t)(sizeof(k_l3_verdict_pass) - 1U));
366 } else {
367 l3_write(k_l3_verdict_fail, (uint32_t)(sizeof(k_l3_verdict_fail) - 1U));
368 }
369 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
370 break;
371 }
372 ra8_delay_ms((uint32_t)k_l3_period_ms);
373 }
374 l3_panic_halt();
375 return 0;
376}
