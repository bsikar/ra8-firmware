119static void fc_print_uint(uint32_t value)
120{
121 enum : uint32_t {
122 k_fc_dec_base = 10U,
123 k_fc_dec_digits = 10U,
124 };
125 uint8_t buf[k_fc_dec_digits];
126 uint32_t n = 0U;
127 if (value == 0U) {
128 buf[n] = '0';
129 n++;
130 }
131 while ((value > 0U) && (n < (uint32_t)k_fc_dec_digits)) {
132 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_fc_dec_base));
133 n++;
134 value /= (uint32_t)k_fc_dec_base;
135 }
136 for (uint32_t i = 0U; i < n; i++) {
137 fc_print(&buf[n - 1U - i], 1U);
138 }
139}

153static void fc_print_hex32(uint32_t value)
154{
155 static const uint8_t k_hex[] = "0123456789ABCDEF";
156 const uint8_t prefix[] = {'0', 'x'};
157 fc_print(prefix, (uint32_t)sizeof(prefix));
158 for (uint32_t i = 0U; i < (uint32_t)k_fc_hex_digits; i++) {
159 const uint32_t shift = (uint32_t)((k_fc_hex_digits - 1U - i) * (uint32_t)k_fc_hex_nib_bits);
160 const uint32_t nib = (value >> shift) & (uint32_t)k_fc_hex_nib_mask;
161 fc_print(&k_hex[nib], 1U);
162 }
163}

282int32_t main(void)
283{
284 fc_setup_or_halt();
285 ra8_isr_globals_enable();
286 ra8_log_set_byte_sink(fc_log_sink, nullptr);
287 fc_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
288
289 /* Arm the persist hook so any subsequent real fault is captured, then
290 * read whatever a prior (warm-reset) boot left behind. */
291 ra8_crashlog_install();
292
293 ra8_crashlog_record_t rec = {};
294 if (ra8_crashlog_peek(&rec)) {
295 fc_print_record(&rec);
296 if (ra8_crashlog_safe_mode_requested()) {
297 fc_print(k_msg_safemode, (uint32_t)sizeof(k_msg_safemode) - 1U);
298 ra8_crashlog_claim(); /* clean claim: reset the loop guard */
299 while (1) {
300 __asm__ volatile("wfi");
301 }
302 }
303 } else {
304 fc_print(k_msg_none, (uint32_t)sizeof(k_msg_none) - 1U);
305 }
306
307 /* In-process write+decode proof (the only leg ra8_emulator can show): push
308 * a synthesised decoded snapshot through the installed write path, then
309 * read it back. Left in place so a bench warm reset shows survival. */
310 ra8_exception_last_t synth = {};
311 fc_make_synth(&synth);
312 ra8_crashlog_record_fault(&synth);
313
314 if (ra8_crashlog_peek(&rec)) {
315 fc_print(k_msg_recorded, (uint32_t)sizeof(k_msg_recorded) - 1U);
316 fc_print_uint(rec.fault.exc_number);
317 fc_print(k_msg_pc, (uint32_t)sizeof(k_msg_pc) - 1U);
318 fc_print_hex32(rec.fault.frame.pc);
319 fc_print(k_msg_loops, (uint32_t)sizeof(k_msg_loops) - 1U);
320 fc_print_uint(rec.boot_loops);
321 fc_print(k_msg_crlf, (uint32_t)sizeof(k_msg_crlf) - 1U);
322 } else {
323 fc_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
324 }
325
326 while (1) {
327 __asm__ volatile("wfi");
328 }
329 return 0;
330}
