73static bool rot_genuine_ok(void)
74{
75 const ra8_rot_trailer_t* trailer =
76 ra8_rot_trailer_after(k_rot_fixture_image, k_rot_fixture_body_len);
77 if (trailer == nullptr) {
78 return false;
79 }
80 return ra8_rot_verify_image(k_rot_fixture_image, k_rot_fixture_body_len, trailer) == k_ra8_ok;
81}

84static bool rot_tamper_rejected(void)
85{
86 memcpy(s_rot_tamper, k_rot_fixture_image, sizeof(s_rot_tamper));
87 s_rot_tamper[0] ^= (uint8_t)k_rot_tamper_bit;
88 const ra8_rot_trailer_t* trailer = ra8_rot_trailer_after(s_rot_tamper, k_rot_fixture_body_len);
89 if (trailer == nullptr) {
90 return false;
91 }
92 return ra8_rot_verify_image(s_rot_tamper, k_rot_fixture_body_len, trailer) != k_ra8_ok;
93}

134int32_t main(void)
135{
136 rot_setup_or_halt();

144 } else {
145 rot_write(k_rot_diag_no_gen, (size_t)(sizeof(k_rot_diag_no_gen) - 1U));
146 ok = false;
147 }
148
149 if (rot_tamper_rejected()) {
150 rot_write(k_rot_diag_tamper, (size_t)(sizeof(k_rot_diag_tamper) - 1U));
151 } else {
152 rot_write(k_rot_diag_no_tamp, (size_t)(sizeof(k_rot_diag_no_tamp) - 1U));
153 ok = false;
154 }
155
156 if (ok) {
157 rot_write(k_rot_msg_pass, (size_t)(sizeof(k_rot_msg_pass) - 1U));
158 (void)ra8_board_led_toggle(k_ra8_board_led1);
159 } else {
160 rot_write(k_rot_msg_fail, (size_t)(sizeof(k_rot_msg_fail) - 1U));
161 (void)ra8_board_led_toggle(k_ra8_board_led2);
162 }
163 ra8_delay_ms(k_rot_period_ms);
164 }
165 rot_panic_halt();
166 return 0;
167}
