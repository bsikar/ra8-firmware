115static uint32_t pdm_next_sample(uint32_t ch)
116{
117 pdm_channel_t* c = &s_pdm[ch];
118 const uint32_t ph = c->phase & (uint32_t)k_pdm_tone_qmask;
119 int32_t tri = 0;
120 if (ph < (uint32_t)k_pdm_tone_half) {
121 tri = -(int32_t)k_pdm_tone_amp + ((int32_t)k_pdm_tone_step * (int32_t)ph);
122 } else {
123 tri = (int32_t)k_pdm_tone_amp -
124 ((int32_t)k_pdm_tone_step * (int32_t)(ph - (uint32_t)k_pdm_tone_half));
125 }
126 c->lcg = (c->lcg * (uint32_t)k_pdm_lcg_mul) + (uint32_t)k_pdm_lcg_add;
127 const int32_t dither =
128 (int32_t)((c->lcg >> (uint32_t)k_pdm_lcg_shift) & (uint32_t)k_pdm_dither_mask) -
129 (int32_t)k_pdm_dither_half;
130 c->phase++;
131 c->samples++;
132 const int32_t sample = tri + dither;
133 return (uint32_t)sample & (uint32_t)k_pdm_pddrr_mask20;
134}

137static void pdm_console_maybe(uint32_t ch)
138{
139 const uint32_t n = s_pdm[ch].samples;
140 if ((n != 1U) && ((n % (uint32_t)k_pdm_console_every) != 0U)) {

149static bool pdm_channel_at(uint64_t off, uint32_t* ch, uint64_t* ch_off)
150{
151 if (off < (uint64_t)k_pdm_ch0_off) {
152 return false;
153 }
154 const uint64_t rel = off - (uint64_t)k_pdm_ch0_off;
155 *ch = (uint32_t)(rel / (uint64_t)k_pdm_ch_stride);
156 *ch_off = rel % (uint64_t)k_pdm_ch_stride;
157 return (*ch < (uint32_t)k_pdm_ch_count);
158}

161static uint32_t pdm_read_csr(void)
162{
163 uint32_t csr = 0U;
164 for (uint32_t i = 0U; i < (uint32_t)k_pdm_ch_count; i++) {
165 if (s_pdm[i].running) {

173static uint64_t pdm_read(uc_engine* uc, uint64_t addr, unsigned size)
174{
175 (void)uc;
176 (void)size;
177 const uint64_t off = addr - (uint64_t)k_pdm_base;
178 if (off >= (uint64_t)k_pdm_span) {
179 return 0U;
180 }
181 if (off == (uint64_t)k_pdm_off_csr) {
182 return pdm_read_csr();
183 }
184 uint32_t ch = 0U;
185 uint64_t ch_off = 0U;
186 if (pdm_channel_at(off, &ch, &ch_off)) {
187 if (ch_off == (uint64_t)k_pdm_ch_pddsr) {
188 return pdm_live(ch) ? (uint64_t)((uint32_t)k_pdm_fifo_depth & (uint32_t)k_pdm_pddsr_num) : 0U;
189 }
190 if (ch_off == (uint64_t)k_pdm_ch_pddrr) {
191 if (!pdm_live(ch)) {
192 return 0U;
193 }
194 const uint32_t sample = pdm_next_sample(ch);
195 pdm_console_maybe(ch);
196 return (uint64_t)sample;
197 }
198 }
199 return (uint64_t)s_pdm_reg[off >> 2U];
200}

203static void pdm_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
204{
205 (void)uc;
206 (void)size;
207 const uint64_t off = addr - (uint64_t)k_pdm_base;
208 if (off >= (uint64_t)k_pdm_span) {
209 return;
210 }
211 s_pdm_reg[off >> 2U] = (uint32_t)value;
212
213 if (off == (uint64_t)k_pdm_off_cstrtr) {
214 for (uint32_t i = 0U; i < (uint32_t)k_pdm_ch_count; i++) {
215 if (((uint32_t)value & (1U << i)) != 0U) {
216 s_pdm[i].running = true;
217 }
218 }
219 return;
220 }
221 if (off == (uint64_t)k_pdm_off_cstptr) {
222 for (uint32_t i = 0U; i < (uint32_t)k_pdm_ch_count; i++) {
223 if (((uint32_t)value & (1U << i)) != 0U) {
224 s_pdm[i].running = false;
225 }
226 }
227 return;
228 }
229 uint32_t ch = 0U;
230 uint64_t ch_off = 0U;
231 if (pdm_channel_at(off, &ch, &ch_off) && (ch_off == (uint64_t)k_pdm_ch_pddrcr)) {
232 s_pdm[ch].read_enable = (((uint32_t)value & (uint32_t)k_pdm_pddrcr_datre) != 0U);
233 }
234}
