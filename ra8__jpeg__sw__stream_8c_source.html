104static ra8_err_t js_refill(ra8_jpeg_stream_state_t* st)
105{
106 while ((st->eof == 0U) && (st->win_len < st->win_cap)) {
107 size_t got = 0U;
108 const ra8_err_t err =
109 st->pull(st->pull_ctx, &st->win[st->win_len], (size_t)(st->win_cap - st->win_len), &got);
110 if (err != k_ra8_ok) {
111 return err;
112 }
113 if (got == 0U) {
114 st->eof = 1U;
115 } else {
116 st->win_len += (uint32_t)got;
117 }
118 }
119 return k_ra8_ok;
120}

165static ra8_err_t js_bind_geometry(ra8_jpeg_stream_state_t* st, ra8_jpeg_sw_geom_fn on_geom)
166{
167 const ra8_jpeg_dec_ctx_t* d = &st->dec;
168 st->channels =
169 (d->ncomp == 1U) ? (uint8_t)k_ra8_jpeg_stream_gray_ch : (uint8_t)k_ra8_jpeg_stream_rgb_ch;
170 st->mcu_w = (uint16_t)((uint16_t)k_ra8_jpeg_block_dim * d->hmax);
171 st->mcu_h = (uint16_t)((uint16_t)k_ra8_jpeg_block_dim * d->vmax);
172 st->mcus_x = (uint16_t)((d->width + st->mcu_w - 1U) / st->mcu_w);
173 st->mcus_y = (uint16_t)((d->height + st->mcu_h - 1U) / st->mcu_h);

175 uint8_t* stripe = nullptr;
176 uint32_t stripe_cap = 0U;

179 if (err != k_ra8_ok) {
180 return err;

182 RA8_CHECK_NULL_PTR(stripe, s_tag, "geometry callback supplied a NULL stripe");

184 if (stripe_cap < need) {

187 st->stripe = stripe;

214static ra8_err_t
215js_parse_markers(ra8_jpeg_stream_state_t* st, ra8_jpeg_sw_geom_fn on_geom, uint32_t* out_scan_pos)
216{
217 bool got_sof = false;
218 bool geom_done = false;
219 for (uint16_t iter = 0U; iter < (uint16_t)k_ra8_jpeg_stream_max_markers; iter++) {
220 if (st->win_len == 0U) {
221 return k_ra8_err_protocol_error; /* stream ended before SOS */
222 }
223 st->dec.src = st->win;
224 st->dec.src_len = st->win_len;
225 st->dec.cursor = 0U;
226 ra8_jpeg_dec_marker_action_t action = k_ra8_jpeg_dec_continue;
227 ra8_err_t err = ra8_jpeg_sw_priv_dispatch(&st->dec, &got_sof, &action);
228 if (err != k_ra8_ok) {
229 return err;
230 }
231 if (got_sof && !geom_done) {
232 err = js_bind_geometry(st, on_geom);
233 if (err != k_ra8_ok) {
234 return err;
235 }
236 geom_done = true;
237 }
238 if (action == k_ra8_jpeg_dec_scan) {
239 *out_scan_pos = st->dec.cursor;
240 return k_ra8_ok;
241 }
242 if (action == k_ra8_jpeg_dec_eoi) {
243 return k_ra8_err_protocol_error; /* EOI before any scan */
244 }
245 err = js_slide(st, st->dec.cursor);
246 if (err != k_ra8_ok) {
247 return err;
248 }
249 }
250 return k_ra8_err_protocol_error; /* marker budget exhausted (hostile stream) */
251}

272static void js_emit_mcu(const ra8_jpeg_stream_state_t* st,
273 const uint8_t* y_tile,
274 const uint8_t* cb_tile,
275 const uint8_t* cr_tile,
276 uint16_t mx,
277 uint16_t rows)
278{
279 const ra8_jpeg_dec_ctx_t* d = &st->dec;
280 const uint32_t stride = (uint32_t)d->width * (uint32_t)st->channels;
281 for (uint16_t r = 0U; r < rows; r++) {
282 for (uint16_t c = 0U; c < st->mcu_w; c++) {
283 const uint32_t px = ((uint32_t)mx * (uint32_t)st->mcu_w) + (uint32_t)c;
284 if (px >= (uint32_t)d->width) {
285 break;
286 }
287 const int32_t y = (int32_t)y_tile[((uint32_t)r * (uint32_t)st->mcu_w) + (uint32_t)c];
288 if (st->channels == (uint8_t)k_ra8_jpeg_stream_gray_ch) {
289 st->stripe[((uint32_t)r * stride) + px] = (uint8_t)y;
290 continue;
291 }
292 const uint16_t cx = (uint16_t)(c / d->hmax);
293 const uint16_t cy = (uint16_t)(r / d->vmax);
294 const int32_t cb = (int32_t)cb_tile[((uint32_t)cy * (uint32_t)k_ra8_jpeg_block_dim) + cx];
295 const int32_t cr = (int32_t)cr_tile[((uint32_t)cy * (uint32_t)k_ra8_jpeg_block_dim) + cx];
296 const uint32_t idx = ((uint32_t)r * stride) + (px * (uint32_t)st->channels);
297 ra8_jpeg_sw_ycc_to_rgb(y,
298 cb,
299 cr,
300 &st->stripe[idx],
301 &st->stripe[idx + 1U],
302 &st->stripe[idx + 2U]);
303 }
304 }

405static ra8_err_t js_scan(ra8_jpeg_stream_state_t* st, uint32_t scan_pos)
406{
407 ra8_jpeg_dec_ctx_t* d = &st->dec;
408 ra8_jpeg_bitreader_t br = {
409 .buf = st->win,
410 .len = st->win_len,
411 .pos = scan_pos,
412 .acc = 0U,
413 .nbits = 0U,
414 .had_eoi = 0U,
415 };
416 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_max_comps; i++) {
417 d->comp_dc_pred[i] = 0;
418 }
419 uint8_t y_tile[(uint32_t)k_ra8_jpeg_mcu_max_dim * (uint32_t)k_ra8_jpeg_mcu_max_dim] = {};
420 uint8_t cb_tile[(uint32_t)k_ra8_jpeg_block_size] = {};
421 uint8_t cr_tile[(uint32_t)k_ra8_jpeg_block_size] = {};
422
423 for (uint16_t my = 0U; my < st->mcus_y; my++) {
424 const uint32_t y0 = (uint32_t)my * (uint32_t)st->mcu_h;
425 const uint32_t left = (uint32_t)d->height - y0;
426 const uint16_t rows = (uint16_t)((left < (uint32_t)st->mcu_h) ? left : (uint32_t)st->mcu_h);
427 for (uint16_t mx = 0U; mx < st->mcus_x; mx++) {
428 const ra8_err_t err = js_decode_mcu(st, &br, y_tile, cb_tile, cr_tile, mx, rows);
429 if (err != k_ra8_ok) {
430 return err;
431 }
432 }
433 const ra8_err_t err =
434 st->on_rows(st->cb_ctx, st->stripe, d->width, (uint16_t)y0, rows, st->channels);
435 if (err != k_ra8_ok) {
436 return err;
437 }
438 }
439 return k_ra8_ok;
440}
