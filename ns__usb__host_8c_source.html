159static void ns_host_pattern_fill(uint32_t round, uint8_t* out, uint32_t len)
160{
161 for (uint32_t i = 0U; i < len; i++) {
162 const uint32_t v = (round * (uint32_t)k_ns_pat_round_mul) + (i * (uint32_t)k_ns_pat_idx_mul) +
163 (uint32_t)k_ns_pat_bias;
164 out[i] = (uint8_t)(v & (uint32_t)k_ns_byte_mask);
165 }
166}

212[[nodiscard]] static ra8_err_t ns_host_enum_hunt(uint8_t* desc)
213{
214 ra8_delay_ms((uint32_t)k_ns_vbus_settle_ms);
215 const uint32_t t0 = ra8_time_ms();
216 for (uint32_t spin = 0U; spin < (uint32_t)k_ns_attach_spin; spin++) {
217 if (ra8_usb_host_line_state(k_ra8_usb_speed_hs) != 0U) {
218 break;
219 }
220 if ((ra8_time_ms() - t0) > (uint32_t)k_ns_attach_to_ms) {
221 break;
222 }
223 }
224 ra8_delay_ms((uint32_t)k_ns_debounce_ms);
225 ra8_err_t err = k_ra8_err_hw_timeout;
226 for (uint8_t attempt = 0U; attempt < (uint8_t)k_ns_enum_tries; attempt++) {
227 (void)ra8_usb_host_bus_reset(k_ra8_usb_speed_hs, true);
228 ra8_delay_ms((uint32_t)k_ns_reset_hold_ms);
229 (void)ra8_usb_host_bus_reset(k_ra8_usb_speed_hs, false);
230 (void)ra8_usb_host_set_uact(k_ra8_usb_speed_hs, true);
231 ra8_delay_ms((uint32_t)k_ns_recovery_ms);
232 (void)ra8_usb_host_set_target(k_ra8_usb_speed_hs, 0U);
233 err = ns_host_get_dev_desc(desc);
234 if (err == k_ra8_ok) {
235 return k_ra8_ok;
236 }
237 }
238 return err;
239}

368[[nodiscard]] static ra8_err_t ns_host_echo_round(uint32_t round)
369{
370 static uint8_t s_tx[k_ns_payload] = {};
371 static uint8_t s_rx[k_ns_echo_buf] = {};
372 ns_host_pattern_fill(round, s_tx, (uint32_t)k_ns_payload);
373 ra8_err_t err = ra8_usb_host_bulk_out(k_ra8_usb_speed_hs,
374 (uint8_t)k_ns_host_pipe_out,
375 s_tx,
376 (uint16_t)k_ns_payload);
377 if (err != k_ra8_ok) {
378 return err;
379 }
380 uint16_t rx = 0U;
381 err = ra8_usb_host_bulk_in(k_ra8_usb_speed_hs,
382 (uint8_t)k_ns_host_pipe_in,
383 s_rx,
384 (uint16_t)k_ns_echo_buf,
385 &rx);
386 if (err != k_ra8_ok) {
387 return err;
388 }
389 if (rx != (uint16_t)k_ns_payload) {
390 return k_ra8_err_invalid_size;
391 }
392 if (memcmp(s_rx, s_tx, (size_t)k_ns_payload) != 0) {
393 return k_ra8_err_invalid_state;
394 }
395 return k_ra8_ok;
396}

398VOID ns_host_worker(ULONG arg)
399{
400 (void)arg;
401 (void)tx_thread_sleep((ULONG)k_ns_boot_wait_tk);
402
403 uint32_t pid = 0U;
404 while (1) {
405 g_tz_usb_host_phase = 1U;
406 ra8_err_t err = ra8_usb_host_init(k_ra8_usb_speed_hs);
407 if (err != k_ra8_ok) {
408 g_tz_usb_host_err = (uint32_t)err;
409 (void)tx_thread_sleep((ULONG)k_ns_retry_tk);
410 continue;
411 }
412 g_tz_usb_host_phase = 2U;
413 err = ns_host_enumerate(&pid);
414 if (err != k_ra8_ok) {
415 g_tz_usb_host_err = (uint32_t)err;
416 (void)ra8_usb_host_deinit(k_ra8_usb_speed_hs);
417 (void)tx_thread_sleep((ULONG)k_ns_retry_tk);
418 continue;
419 }
420 g_tz_usb_host_pid = pid;
421 g_tz_usb_host_phase = 3U;
422 break;
423 }
424
425 /* Enumerated. Run bulk echo rounds forever; rounds_ok advancing is the
426 * end-to-end self-loop proof (host bulk-OUT -> device auto-echo -> host
427 * bulk-IN, byte-checked). */
428 g_tz_usb_host_phase = 4U;
429 uint32_t round = 0U;
430 while (1) {
431 const ra8_err_t err = ns_host_echo_round(round);
432 if (err == k_ra8_ok) {
433 g_tz_usb_host_rounds_ok += 1U;
434 } else if (g_tz_usb_host_err == 0U) {
435 g_tz_usb_host_err = (uint32_t)err;
436 }
437 round += 1U;
438 }
