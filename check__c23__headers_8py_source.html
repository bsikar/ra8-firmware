4"""check_c23_headers.py -- enforce C23 typed enums and #pragma once headers.
5
6Two rules that CLAUDE.md mandates but that had, until now, no automated
7checker -- they were convention only, so drift was invisible until a human
8happened to read the file:
9
101. **C23 typed enums.** Every ``enum`` / ``typedef enum`` in first-party code
11 MUST specify an explicit underlying type: ``enum : <type> { ... }`` (for
12 example ``typedef enum : uint8_t { ... } name_t;``). A bare ``enum {`` or a
13 tagged ``enum Name {`` with no ``: <type>`` FAILS. A fixed underlying type
14 pins the enum's size for ABI stability and debugger compatibility -- the
15 reason CLAUDE.md's "Constants and Macros" section makes it mandatory.
16
172. **#pragma once headers.** First-party headers (``.h`` / ``.hpp``) MUST use
18 ``#pragma once`` rather than a classic ``#ifndef`` / ``#define`` /
19 ``#endif`` include guard.
20
21Detection notes:
22
23* Only enum *definitions* are flagged -- a definition is ``enum`` followed by
24 an optional tag and then ``{``. A *reference* (``enum xz_ret ret``, a
25 ``sizeof(enum foo)``, a forward ``enum foo;``) has some other token after the
26 tag and is deliberately left alone. C++ scoped enums (``enum class``) do not
27 occur in this C tree and are not special-cased.
28* The header-guard rule is expressed as a *positive* requirement: a header
29 passes iff it contains a real ``#pragma once`` directive. This is
30 deliberately simpler and far less false-positive-prone than trying to
31 pattern-match an ``#ifndef`` / ``#define`` / ``#endif`` *triple*: an
32 ``#ifndef`` on its own is an ordinary feature-test (``#ifndef __cplusplus``)
33 and must not be mistaken for an include guard. Requiring ``#pragma once``
34 catches every guarded header (it lacks the pragma) without ever having to
35 decide whether a given ``#ifndef`` is a guard.
36* Comments and string / character literals are blanked before scanning, so an
37 ``enum`` or an ``#ifndef`` written inside a comment or a string is never
38 flagged, and a ``#pragma once`` mentioned inside a comment does not satisfy
39 the rule.
40
41Per-line / per-file opt-out: append ``C23HDR-OK: <reason>`` to waive a finding
42(mirrors the ``MAGIC-OK`` / ``CITES-OK`` / ``AI-OK`` markers the sibling gates
43use). For the enum rule the marker must sit on the offending ``enum`` line; for
44the header rule it may sit anywhere in the file. The one standing use is
45``port/mbedtls/inc/tf_psa_crypto_config.h``, whose ``#ifndef
46PSA_CRYPTO_CONFIG_H`` guard is not an ordinary include guard at all but a
47vendor-ABI sentinel: the vendored Mbed TLS SOUP tests ``#if
48defined(PSA_CRYPTO_CONFIG_H)`` to confirm the crypto config file was supplied,
49so the guard macro must keep being defined.
50
51Vendored trees (``libs/third_party/``), generated tables (``libs/ra8_fonts/``) and
52build output are skipped wholesale.
53
54Usage:
55 python3 scripts/checks/check_c23_headers.py # staged files
56 python3 scripts/checks/check_c23_headers.py FILE [FILE ...] # explicit list
57 python3 scripts/checks/check_c23_headers.py --all # all tracked
58 python3 scripts/checks/check_c23_headers.py --selftest # self-check
59
60Returns 0 on clean, 1 on findings, 2 on usage / enumeration error.
61"""

85# The typed-enum rule applies to every first-party C/C++ translation unit and
86# header; the pragma-once rule applies only to headers.
87ENUM_EXTENSIONS = frozenset({".c", ".h", ".cpp", ".hpp"})
88HEADER_EXTENSIONS = frozenset({".h", ".hpp"})
89
90# Per-line / per-file opt-out marker (written reason required by convention).
91OPT_OUT = "C23HDR-OK"
92
93# An enum introduced by the keyword ``enum``, then an OPTIONAL tag identifier,
94# then the first following non-space token. That token disambiguates:
95# ``{`` -> this is a DEFINITION body, and (having no ``:`` before it) it is
96# UNTYPED -> a violation.
97# ``:`` -> a fixed underlying type follows -> typed, compliant.
98# else -> a reference / use / forward declaration -> not our concern.
99# DOTALL lets the leading ``\s*`` swallow newlines so a ``typedef enum`` whose
100# opening ``{`` sits on the next line is still recognised as one definition.
101_ENUM_RE = re.compile(
102 r"\benum\b\s*(?:[A-Za-z_]\w*\s*)?(?P<next>.)",
103 re.DOTALL,
104)
105
106# A real ``#pragma once`` directive (in code, after comment/string blanking).
107_PRAGMA_ONCE_RE = re.compile(r"^\s*#\s*pragma\s+once\b", re.MULTILINE)

110def _strip_comments_and_strings(text: str) -> str:
111 """Blank comment and literal bytes to spaces, preserving every position.
112
113 Newlines survive so line numbers computed against the blanked view remain
114 valid for the original source. A naive char-by-char state machine is used
115 rather than a regex because nested-looking quote / comment interactions
116 make a single regex fragile.
117
118 Args:
119 text: The original source text of one file.
120
121 Returns:
122 The same text with the interior of every ``//`` / ``/* */`` comment and
123 every string / char literal replaced by spaces (newlines kept).
124 """
125 out: list[str] = []
126 i = 0
127 n = len(text)
128 while i < n:
129 c = text[i]
130 nxt = text[i + 1] if i + 1 < n else ""
131 if c == "/" and nxt == "/":
132 while i < n and text[i] != "\n":
133 i += 1
134 continue
135 if c == "/" and nxt == "*":
136 out.append(" ")
137 i += 2
138 while i < n and not (text[i] == "*" and i + 1 < n and text[i + 1] == "/"):
139 out.append("\n" if text[i] == "\n" else " ")
140 i += 1
141 if i < n:
142 out.append(" ")
143 i += 2
144 continue
145 if c in {'"', "'"}:
146 quote = c
147 out.append(" ")
148 i += 1
149 while i < n and text[i] != quote:
150 if text[i] == "\\" and i + 1 < n:
151 out.append(" ")
152 i += 2
153 continue
154 out.append("\n" if text[i] == "\n" else " ")
155 i += 1
156 if i < n:
157 out.append(" ")
158 i += 1
159 continue
160 out.append(c)
161 i += 1
162 return "".join(out)

178def enum_violations(text: str) -> list[tuple[int, str]]:
179 """Find every untyped ``enum`` definition in one file's text.
180
181 Only enum definitions (``enum [tag] {``) are considered; references and
182 forward declarations are skipped. A definition whose ``{`` is preceded by a
183 ``: <type>`` clause is compliant and is not reported.
184
185 Args:
186 text: The original source text of one file.
187
188 Returns:
189 A list of ``(line_no, snippet)`` for each untyped enum definition, with
190 opt-out (``C23HDR-OK``) lines already removed.
191 """
192 stripped = _strip_comments_and_strings(text)
193 orig_lines = text.splitlines()
194 violations: list[tuple[int, str]] = []
195 for match in _ENUM_RE.finditer(stripped):
196 if match.group("next") != "{":
197 continue # ``:`` is typed; any other token is a reference/use
198 line_no = _line_of(stripped, match.start())
199 raw = orig_lines[line_no - 1] if line_no - 1 < len(orig_lines) else ""
200 if OPT_OUT in raw:
201 continue
202 violations.append((line_no, raw.strip()))
203 return violations

206def header_lacks_pragma_once(text: str) -> bool:
207 """Return whether a header's text is missing a real ``#pragma once``.
208
209 The check runs against the comment/string-blanked view so a ``#pragma
210 once`` written inside a comment does not count. A file carrying the
211 ``C23HDR-OK`` opt-out anywhere is treated as compliant.
212
213 Args:
214 text: The original source text of a header file.
215
216 Returns:
217 True when the header has no ``#pragma once`` directive and is not
218 waived; False otherwise.
219 """
220 if OPT_OUT in text:
221 return False
222 stripped = _strip_comments_and_strings(text)
223 return _PRAGMA_ONCE_RE.search(stripped) is None

226def find_violations(path: Path) -> list[tuple[int, str]]:
227 """Report every C23 typed-enum / pragma-once violation in one file.
228
229 The enum rule is applied to any in-scope C/C++ file; the pragma-once rule
230 is applied only to headers.
231
232 Args:
233 path: The file to scan.
234
235 Returns:
236 A list of ``(line_no, message)`` findings; an unreadable file yields an
237 empty list rather than raising. A header missing ``#pragma once`` is
238 reported at line 1.
239 """
240 try:
241 text = path.read_text(encoding="utf-8", errors="replace")
242 except OSError:
243 return []
244 findings: list[tuple[int, str]] = []
245 if path.suffix.lower() in ENUM_EXTENSIONS:
246 for line_no, snippet in enum_violations(text):
247 findings.append(
248 (line_no, f"untyped enum -- add `: <type>` (e.g. `enum : uint8_t`): {snippet}")
249 )
250 if path.suffix.lower() in HEADER_EXTENSIONS and header_lacks_pragma_once(text):
251 findings.append((1, "header has no `#pragma once` (classic include guards are banned)"))
252 findings.sort()
253 return findings

268def _in_scope(path: Path) -> bool:
269 """Return whether a path is a first-party file this gate scans.
270
271 Args:
272 path: The candidate file path (absolute or repo-relative).
273
274 Returns:
275 True when the suffix is one the gate handles and the path is not
276 excluded.
277 """
278 if path.suffix.lower() not in ENUM_EXTENSIONS:
279 return False

284def _git_lines(args: list[str]) -> list[str]:
285 """Run a git command under the repo root and return its stdout lines.

334def _enum_cases() -> list[tuple[bool, str]]:
335 """Return the enum-rule self-check cases as ``(failed, message)`` pairs.
336
337 Each case maps a fragment through :func:`enum_violations` and records
338 whether the observed behaviour differs from the specified behaviour: the
339 untyped/tagged-untyped definitions must fire; typed forms, references and
340 commented-out enums must stay silent; the opt-out must be honoured.
341
342 Returns:
343 One ``(failed, message)`` tuple per assertion; ``failed`` is True when
344 the detector behaved wrongly.
345 """
346 fires = lambda frag: bool(enum_violations(frag)) # noqa: E731 -- local alias keeps the table terse
347 return [
348 (not fires("typedef enum { k_a = 0 } foo_t;"), "untyped `typedef enum {` not flagged"),
349 (not fires("enum bearer { k_a = 1 };"), "untyped tagged `enum bearer {` not flagged"),
350 (fires("typedef enum : uint8_t { k_a = 0 } foo_t;"), "typed `enum : uint8_t {` flagged"),
351 (fires("enum bearer : uint16_t { k_a = 1 };"), "typed tagged `enum ... :` flagged"),
352 (fires("static int map(enum xz_ret ret) { return ret; }"), "enum parameter ref flagged"),
353 (fires("size_t n = sizeof(enum foo);"), "enum sizeof reference flagged"),
354 (fires("/* typedef enum { k_a } t; */"), "untyped enum in a comment flagged"),
355 (not fires("enum { k_x = 1 }; /* real */"), "untyped enum w/ trailing comment missed"),
356 (fires("enum { k_x = 1 }; /* C23HDR-OK: reason */"), "enum opt-out not honoured"),
357 ]

360def _header_cases() -> list[tuple[bool, str]]:
361 """Return the header-rule self-check cases as ``(failed, message)`` pairs.
362
363 An ``#ifndef`` include-guarded header must fire; a ``#pragma once`` header
364 must not; a ``#pragma once`` buried in a comment must not satisfy the rule;
365 the opt-out must be honoured.
366
367 Returns:
368 One ``(failed, message)`` tuple per assertion; ``failed`` is True when
369 the detector behaved wrongly.
370 """
371 guarded = "#ifndef FOO_H\n#define FOO_H\nint foo(void);\n#endif\n"
372 lacks = header_lacks_pragma_once
373 return [
374 (not lacks(guarded), "`#ifndef` include-guarded header not flagged"),
375 (lacks("#pragma once\nint foo(void);\n"), "`#pragma once` header wrongly flagged"),
376 (not lacks("/* #pragma once */\nint foo(void);\n"), "commented `#pragma once` accepted"),
377 (lacks(guarded + "/* C23HDR-OK: reason */\n"), "header opt-out not honoured"),
378 ]

381def _selftest() -> int:
382 """Assert both rules fire on the real defect and stay quiet on legal forms.
383
384 Returns:
385 0 when every direction of both rules behaves as specified, 1 otherwise.
386 """
387 print("check_c23_headers.py --selftest")
388 failures = [msg for failed, msg in _enum_cases() + _header_cases() if failed]
389 if failures:
390 for msg in failures:
391 print(f" FAIL: {msg}", file=sys.stderr)
392 print(f"check_c23_headers.py --selftest: {len(failures)} failure(s)", file=sys.stderr)
393 return 1
394 print("check_c23_headers.py --selftest: OK")
395 return 0

398def _report(candidates: list[Path]) -> int:
399 """Scan ``candidates`` and print any findings.
400
401 Args:
402 candidates: The in-scope files to scan.
403
404 Returns:
405 1 if any violation was found, 0 otherwise.
406 """
407 total = 0
408 for path in candidates:
409 rel = path.relative_to(REPO_ROOT) if path.is_relative_to(REPO_ROOT) else path
410 for line, message in find_violations(path):
411 print(f"{rel}:{line}: {message}", file=sys.stderr)
412 total += 1
413 if total:
414 print(
415 f"\n{total} C23 header/enum violation(s). Every enum needs an explicit "
416 "underlying type (`enum : uint8_t { ... }`) and every header must use "
417 f"`#pragma once`. A genuine exception carries a `{OPT_OUT}: <reason>` "
418 "marker.",
419 file=sys.stderr,
420 )
421 return 1
422 print(f"check_c23_headers.py: {len(candidates)} file(s) scanned, 0 findings.")
423 return 0

426def main(argv: list[str]) -> int:
427 """Enforce C23 typed-enum underlying types and pragma-once headers.
428
429 ``--selftest`` proves the detector in both directions; ``--all`` sweeps the
430 tree; an explicit file list scans just those files; otherwise the staged
431 set is scanned (how the pre-commit hook stays fast).
432
433 Args:
434 argv: The full process argument vector (``sys.argv``).
435
436 Returns:
437 0 clean, 1 on findings, 2 on usage / enumeration error.
438 """
439 parser = argparse.ArgumentParser(description="C23 typed-enum + pragma-once checker")
440 parser.add_argument("--all", action="store_true", help="scan all tracked files")
441 parser.add_argument("--selftest", action="store_true", help="assert both rules, both ways")
442 parser.add_argument("files", nargs="*", type=Path, help="explicit file list (e.g. staged)")
443 args = parser.parse_args(argv[1:])
444
445 if args.selftest:
446 return _selftest()
447 if args.all:
448 candidates = all_files()
449 elif args.files:
450 candidates = [p if p.is_absolute() else REPO_ROOT / p for p in args.files]
451 candidates = [p for p in candidates if p.is_file() and _in_scope(p)]
452 else:
453 candidates = staged_files()
454
455 return _report(candidates)
