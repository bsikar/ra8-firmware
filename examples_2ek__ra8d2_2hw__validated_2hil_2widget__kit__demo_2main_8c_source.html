318static void wk_print_hex(uint32_t value)
319{
320 uint8_t buf[k_wk_hex_nibbles];
321 for (uint32_t i = 0U; i < (uint32_t)k_wk_hex_nibbles; i++) {
322 const uint32_t shift = ((uint32_t)k_wk_hex_nibbles - 1U - i) * (uint32_t)k_wk_nibble_bits;
323 const uint32_t nib = (value >> shift) & (uint32_t)k_wk_nibble_mask;
324 buf[i] = (uint8_t)((nib < (uint32_t)k_wk_dec_ten) ? ('0' + nib) : ('A' + (nib - k_wk_dec_ten)));
325 }
326 wk_print(buf, (uint32_t)k_wk_hex_nibbles);
327}

330static uint32_t wk_framebuffer_hash(void)
331{
332 const uint8_t* p = (const uint8_t*)s_framebuffer;
333 const size_t n = sizeof(s_framebuffer);
334 uint32_t hsh = (uint32_t)k_wk_fnv_offset;
335 for (size_t i = 0U; i < n; i++) {
336 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_wk_fnv_prime;
337 }
338 return hsh;
339}

594{

606#pragma GCC diagnostic push
607#pragma GCC diagnostic ignored "-Wmain"
620int32_t main(void)
621{
622 wk_setup_or_halt();
623 ra8_isr_globals_enable();
624 wk_print(k_wk_msg_boot, (uint32_t)sizeof(k_wk_msg_boot) - 1U);
625
626 if (!wk_panel_up()) {
627 wk_fail_halt(k_wk_msg_fpanl, (uint32_t)sizeof(k_wk_msg_fpanl) - 1U);
628 }
629 if (!wk_build_tree()) {
630 wk_fail_halt(k_wk_msg_ftree, (uint32_t)sizeof(k_wk_msg_ftree) - 1U);
631 }
632
633 uint32_t crc0 = 0U;
634 uint32_t crc1 = 0U;
635 if (!wk_selfcheck(&crc0, &crc1)) {
636 wk_fail_halt(k_wk_msg_fchk, (uint32_t)sizeof(k_wk_msg_fchk) - 1U);
637 }
638 wk_print_banner(crc0, crc1);
639
640 /* Land on a clean full composite and show it live on the panel. */
641 (void)ra8_gfx_clear((uint32_t)k_wk_col_clear);
642 wk_invalidate_all(k_ra8_widget_refresh_quality);
643 ra8_ui_rect_t dmg = {};
644 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
645 uint16_t nd = 0U;
646 if (wk_compose(&dmg, &hint, &nd) == k_ra8_ok) {
647 wk_flush_rect(&dmg, hint);
648 }
649
650 while (1) {
651 wk_tick_live();
652 ra8_delay_ms((uint32_t)k_wk_frame_ms);
653 }
654}
