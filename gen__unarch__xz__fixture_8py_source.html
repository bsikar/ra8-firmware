1#!/usr/bin/env python3
2"""gen_unarch_xz_fixture.py -- regenerate tests/unarch_xz_fixture.h.
3
4The XZ decoder tests (tests/test_ra8_unarch_xz.c) need real .xz streams
5with controlled properties: integrity-check type (CRC32 / CRC64 / SHA-256),
6LZMA2 dictionary size, and compression ratio. Those bytes cannot be built
7portably at test runtime (the C tree has no XZ *encoder*), so this script
8materialises them once with Python's lzma module and emits a committed
9C header of byte arrays. Re-run it only when a fixture needs to change:
10
11 python3 scripts/gen/gen_unarch_xz_fixture.py
12
13Fixture inventory (all payloads reproducible in C, see the header):
14 fx_xz_crc64_4k 4 KiB LCG payload, dict 64 KiB, CRC64 (the xz default)
15 -- stream > 512 B so the unwrap loop takes several
16 input-refill passes.
17 fx_xz_crc32 328 B text payload, dict 64 KiB, CRC32.
18 fx_xz_sha256 same payload, SHA-256 check -- xz-embedded is built
19 without SHA-256, so this must be REJECTED cleanly.
20 fx_xz_bigdict same payload, dict 8 MiB -- must be rejected by a
21 session whose scratch dictionary budget is smaller.
22 fx_xz_bomb_1m 1 MiB of zeros in a 284 B stream (~3690:1) -- breaches
23 the DEFAULT decompression-limits ratio bound
24 (1024:1 + 64 KiB grace) and must be rejected mid-decode.
25 fx_xz_cbt an XZ-wrapped ustar comic (two .png page members with
26 the payloads "PAGE-ONE" / "PAGE-TWO") for the
27 ra8_comic_open_wrapped .tar.xz path.
28
29Copyright (c) 2026 Brighton Sikarskie
30SPDX-License-Identifier: MIT
31"""

51def lcg_bytes(n: int, seed: int = LCG_SEED) -> bytes:
52 """The C-reproducible pseudo-random payload generator."""
53 out = bytearray()
54 s = seed
55 for _ in range(n):
56 s = (LCG_MUL * s + LCG_ADD) & 0xFFFFFFFF
57 out.append((s >> 16) & 0xFF)
58 return bytes(out)

61def xz(payload: bytes, check: int, dict_size: int) -> bytes:
62 """One .xz stream with an explicit check type and LZMA2 dictionary."""
63 filters = [{"id": lzma.FILTER_LZMA2, "preset": 6, "dict_size": dict_size}]
64 return lzma.compress(payload, format=lzma.FORMAT_XZ, check=check, filters=filters)

67def c_array(name: str, data: bytes) -> str:
68 """Emit one static C byte array (12 bytes per line, pure ASCII)."""
69 lines = [f"/** @brief {name}: {len(data)} fixture bytes (see file header). */"]
70 lines.append(f"static const uint8_t {name}[{len(data)}U] = {{")
71 for i in range(0, len(data), 12):
72 chunk = ", ".join(f"0x{b:02X}U" for b in data[i : i + 12])
73 lines.append(f" {chunk},")
74 lines.append("};")
75 return "\n".join(lines)

78def cbt_tar() -> bytes:
79 """A deterministic two-page ustar comic (the .tar.xz fixture core)."""
80 buf = io.BytesIO()
81 with tarfile.open(fileobj=buf, mode="w", format=tarfile.USTAR_FORMAT) as tf:

84 (b"page1.png", b"PAGE-ONE"),

87 info.size = len(payload)

94 """Emit the xz decoder fixtures covering each supported integrity check.
95
96 Generates one stream per check type (CRC32, CRC64, SHA256) and per
97 dictionary size, because the decoder's bounds handling differs by both --
98 a fixture set covering only the default would leave the fail-closed paths
99 unexercised.
100 """
101 small = b"hello xz stream, bounded and fail-closed\n" * 8
102 dict_64k = 1 << 16
103 dict_8m = 1 << 23
104
105 arrays = [
106 ("k_fx_xz_crc64_4k", xz(lcg_bytes(4096), lzma.CHECK_CRC64, dict_64k)),
107 ("k_fx_xz_crc32", xz(small, lzma.CHECK_CRC32, dict_64k)),
108 ("k_fx_xz_sha256", xz(small, lzma.CHECK_SHA256, dict_64k)),
109 ("k_fx_xz_bigdict", xz(small, lzma.CHECK_CRC64, dict_8m)),
110 ("k_fx_xz_bomb_1m", xz(b"\x00" * (1 << 20), lzma.CHECK_CRC64, dict_64k)),
111 ("k_fx_xz_cbt", xz(cbt_tar(), lzma.CHECK_CRC64, dict_64k)),
112 ]
113
114 body = "\n\n".join(c_array(n, d) for n, d in arrays)
115 header = f"""/**
116 * @file unarch_xz_fixture.h
117 * @brief Committed .xz byte fixtures for the XZ decoder tests.
118 *
119 * @details
120 * GENERATED FILE -- regenerate with:
121 * python3 scripts/gen/gen_unarch_xz_fixture.py

127 * @copyright Copyright (c) 2026 Brighton Sikarskie
128 * SPDX-License-Identifier: MIT

134{body}
