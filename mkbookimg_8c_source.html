69static uint8_t* read_file(const char* path, uint32_t* out_len)
70{
71 FILE* f = fopen(path, "rb");
72 if (f == nullptr) {
73 return nullptr;
74 }
75 (void)fseek(f, 0, SEEK_END);
76 const long n = ftell(f);
77 (void)fseek(f, 0, SEEK_SET);
78 if (n <= 0) {
79 (void)fclose(f);
80 return nullptr;
81 }
82 uint8_t* buf = (uint8_t*)malloc((size_t)n);
83 if ((buf == nullptr) || (fread(buf, 1U, (size_t)n, f) != (size_t)n)) {
84 free(buf);
85 (void)fclose(f);
86 return nullptr;
87 }
88 (void)fclose(f);
89 *out_len = (uint32_t)n;
90 return buf;
91}

119static int write_books(ra8_fs_mount_t* mnt, char** argv, int n_books)
120{
121 for (int i = 0; i < n_books; ++i) {
122 uint32_t len = 0U;
123 uint8_t* const data = read_file(argv[2 + i], &len);
124 if (data == nullptr) {
125 (void)fprintf(stderr, "mkbookimg: cannot read %s\n", argv[2 + i]);
126 return 1;
127 }
128 char name[k_name_len];
129 (void)snprintf(name, sizeof name, "BOOK%02d.%s", i + 1, book_ext(argv[2 + i]));
130 const ra8_err_t err = ra8_fs_write_file(mnt, name, data, len);
131 if (err == k_ra8_ok) {
132 (void)fprintf(stderr, "mkbookimg: + %s (%u bytes) <- %s\n", name, len, argv[2 + i]);
133 }
134 free(data);
135 if (err != k_ra8_ok) {
136 (void)fprintf(stderr, "mkbookimg: write %s failed\n", name);
137 return 1;
138 }
139 }
140 return 0;
141}

144static int dump_image(const char* path)
145{

157int main(int argc, char** argv)
158{
159 if (argc < 3) {
160 (void)fprintf(stderr, "usage: %s <out.img> <book1.rabook> [book2.rabook ...]\n", argv[0]);
161 return 2;
162 }
163 const int n_books = argc - 2;
164 if (n_books > (int)k_max_books) {
165 (void)fprintf(stderr, "mkbookimg: too many books (max %u)\n", k_max_books);
166 return 2;
167 }
168 s_disk.block_count = k_img_sectors;
169 s_disk.bytes = (uint8_t*)calloc(1U, (size_t)k_img_sectors * k_block_size);
170 if (s_disk.bytes == nullptr) {
171 (void)fprintf(stderr, "mkbookimg: out of memory\n");
172 return 1;
173 }
174 const ra8_fs_backend_t backend = {.read_block = mem_read,
175 .write_block = mem_write,
176 .get_capacity = mem_cap,
177 .ctx = &s_disk};
178 ra8_fs_mount_t* mnt = nullptr;
179 int rc = fs_format_mount(&backend, &mnt);
180 if (rc == 0) {
181 rc = write_books(mnt, argv, n_books);
182 (void)ra8_fs_unmount(mnt);
183 }
184 if (rc == 0) {
185 rc = dump_image(argv[1]);
186 }
187 free(s_disk.bytes);
188 if (rc == 0) {
189 (void)fprintf(stderr, "mkbookimg: wrote %s (%d book(s), 64 MiB FAT32)\n", argv[1], n_books);
190 }
191 return rc;
192}
