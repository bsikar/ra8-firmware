146static uint8_t* read_file(const char* path, uint32_t* out_len)
147{
148 FILE* f = fopen(path, "rb");
149 if (f == nullptr) {
150 return nullptr;
151 }
152 (void)fseek(f, 0, SEEK_END);
153 const long n = ftell(f);
154 (void)fseek(f, 0, SEEK_SET);
155 if (n <= 0) {
156 (void)fclose(f);
157 return nullptr;
158 }
159 uint8_t* buf = (uint8_t*)malloc((size_t)n);
160 if ((buf == nullptr) || (fread(buf, 1U, (size_t)n, f) != (size_t)n)) {
161 free(buf);
162 (void)fclose(f);
163 return nullptr;
164 }
165 (void)fclose(f);
166 *out_len = (uint32_t)n;
167 return buf;
168}

272static int write_books(ra8_fs_mount_t* mnt, char** argv, int n_books)
273{
274 for (int i = 0; i < n_books; ++i) {
275 char name[k_mkbookimg_name_cap];
276 if (!mkbookimg_dest_name(argv[2 + i], name, sizeof name)) {
277 (void)fprintf(stderr, "mkbookimg: unusable card name for %s\n", argv[2 + i]);
278 return 1;
279 }
280 if (dup_dest_name(argv, i)) {
281 (void)fprintf(stderr, "mkbookimg: duplicate card name %s (from %s)\n", name, argv[2 + i]);
282 return 1;
283 }
284 uint32_t len = 0U;
285 uint8_t* const data = read_file(argv[2 + i], &len);
286 if (data == nullptr) {
287 (void)fprintf(stderr, "mkbookimg: cannot read %s\n", argv[2 + i]);
288 return 1;
289 }
290 const ra8_err_t err = ra8_fs_write_file(mnt, name, data, len);
291 if (err == k_ra8_ok) {
292 (void)fprintf(stderr, "mkbookimg: + %s (%u bytes) <- %s\n", name, len, argv[2 + i]);
293 }
294 free(data);
295 if (err != k_ra8_ok) {
296 (void)fprintf(stderr, "mkbookimg: write %s failed\n", name);
297 return 1;
298 }
299 }
300 return 0;
301}

338int main(int argc, char** argv)
339{
340 if (argc < 3) {
341 (void)fprintf(stderr, "usage: %s <out.img> <book1.rabook> [book2.rabook ...]\n", argv[0]);
342 return 2;
343 }
344 const int n_books = argc - 2;
345 if (n_books > (int)k_max_books) {
346 (void)fprintf(stderr, "mkbookimg: too many books (max %u)\n", k_max_books);
347 return 2;
348 }
349 s_disk.block_count = k_img_sectors;
350 s_disk.bytes = (uint8_t*)calloc(1U, (size_t)k_img_sectors * k_block_size);
351 if (s_disk.bytes == nullptr) {
352 (void)fprintf(stderr, "mkbookimg: out of memory\n");
353 return 1;
354 }
355 const ra8_fs_backend_t backend = {.read_block = mem_read,
356 .write_block = mem_write,
357 .get_capacity = mem_cap,
358 .ctx = &s_disk};
359 ra8_fs_mount_t* mnt = nullptr;
360 int rc = fs_format_mount(&backend, &mnt);
361 if (rc == 0) {
362 rc = write_books(mnt, argv, n_books);
363 (void)ra8_fs_unmount(mnt);
364 }
365 if (rc == 0) {
366 rc = dump_image(argv[1]);
367 }
368 free(s_disk.bytes);
369 if (rc == 0) {
370 (void)fprintf(stderr, "mkbookimg: wrote %s (%d book(s), 64 MiB FAT32)\n", argv[1], n_books);
371 }
372 return rc;
373}
