207static ra8_err_t cam_sccb_write(uint16_t reg, uint8_t val)
208{
209 if (reg == 0U) {
210 return k_ra8_err_invalid_arg;
211 }
212 const uint8_t payload[3] = {
213 (uint8_t)((reg >> (uint16_t)k_cam_hi_byte_shift) & (uint16_t)k_cam_byte_mask),
214 (uint8_t)(reg & (uint16_t)k_cam_byte_mask),
215 val,
216 };
217 return ra8_i2c_write((uint8_t)k_cam_iic_ch,
218 s_sccb_addr,
219 payload,
220 (uint32_t)sizeof(payload),
221 true);
222}

241static ra8_err_t cam_sccb_read(uint16_t reg, uint8_t* out_val)
242{
243 RA8_CHECK_NULL_PTR(out_val, "cam", "sccb_read");
244 if (reg == 0U) {
245 return k_ra8_err_invalid_arg;
246 }
247 const uint8_t addr[2] = {
248 (uint8_t)((reg >> (uint16_t)k_cam_hi_byte_shift) & (uint16_t)k_cam_byte_mask),
249 (uint8_t)(reg & (uint16_t)k_cam_byte_mask),
250 };
251 return ra8_i2c_transfer((uint8_t)k_cam_iic_ch,
252 s_sccb_addr,
253 addr,
254 (uint32_t)k_cam_reg_addr_bytes,
255 out_val,
256 1U);
257}

274static ra8_err_t cam_read_chip_id(uint16_t* out_id)
275{
276 RA8_CHECK_NULL_PTR(out_id, "cam", "chip_id");
277 uint8_t hi = 0U;
278 uint8_t lo = 0U;
279 ra8_err_t err = cam_sccb_read((uint16_t)k_cam_reg_chip_id_hi, &hi);
280 if (err != k_ra8_ok) {
281 return err;
282 }
283 err = cam_sccb_read((uint16_t)k_cam_reg_chip_id_lo, &lo);
284 if (err != k_ra8_ok) {
285 return err;
286 }
287 *out_id = (uint16_t)(((uint16_t)hi << (uint16_t)k_cam_hi_byte_shift) | (uint16_t)lo);
288 return k_ra8_ok;
289}

296ra8_err_t cam_reset_sensor(void)
297{
298 const ra8_port_pin_t rst = RA8_PIN(k_ra8_port_7, k_ra8_pin_9);
299 ra8_err_t err = ra8_gpio_output_init(rst, k_ra8_level_low);
300 if (err != k_ra8_ok) {
301 return err;
302 }
303 ra8_delay_ms((uint32_t)k_cam_reset_low_ms);
304 err = ra8_gpio_write(rst, k_ra8_level_high);
305 if (err != k_ra8_ok) {
306 return err;
307 }
308 ra8_delay_ms((uint32_t)k_cam_reset_high_ms);
309 return k_ra8_ok;
310}

312ra8_err_t cam_configure_sensor(void)
313{
314 ra8_err_t err = cam_sccb_write((uint16_t)k_cam_reg_sw_reset, (uint8_t)k_cam_sw_reset_hold);
315 if (err != k_ra8_ok) {
316 return err;
317 }
318 ra8_delay_ms((uint32_t)k_cam_swreset_ms);
319 const uint32_t count = (uint32_t)(sizeof(s_ov5640_dvp_colorbar) / sizeof(cam_reg_t));
320 for (uint32_t i = 0U; i < count; i += 1U) {
321 err = cam_sccb_write(s_ov5640_dvp_colorbar[i].reg, s_ov5640_dvp_colorbar[i].val);
322 if (err != k_ra8_ok) {
323 return err;
324 }
325 }
326 err = cam_sccb_write((uint16_t)k_cam_reg_sw_reset, (uint8_t)k_cam_sw_reset_wake);
327 if (err != k_ra8_ok) {
328 return err;
329 }
330 ra8_delay_ms((uint32_t)k_cam_cfg_settle_ms);
331 return k_ra8_ok;
332}

334bool cam_probe_sensor(uint16_t* out_id)
335{
336 if (out_id == nullptr) {
337 return false;
338 }
339 *out_id = 0U;
340 const uint8_t addrs[2] = {(uint8_t)k_cam_sccb_addr_7b, (uint8_t)k_cam_sccb_addr_alt};
341 for (uint32_t i = 0U; i < (uint32_t)sizeof(addrs); i += 1U) {
342 s_sccb_addr = addrs[i];
343 uint16_t id = 0U;
344 ra8_err_t rerr = cam_read_chip_id(&id);
345 if (rerr == k_ra8_ok) {
346 *out_id = id;
347 if (id == (uint16_t)k_cam_chip_id_ov5640) {
348 return true;
349 }
350 }
351 }
352 s_sccb_addr = (uint8_t)k_cam_sccb_addr_7b;
353 return false;
354}
