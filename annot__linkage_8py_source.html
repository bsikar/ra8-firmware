3"""The linkage rule: every non-static function states why it has external linkage.
4
5Separate from the other rules because it is the only one defined over the
6*absence* of an annotation. Every rule in :mod:`annot_rules` starts from a tag
7someone wrote and checks a property of it; this one starts from the whole
8symbol table and asks which definitions nobody has classified at all. That
9makes it the rule most likely to look clean by simply not running -- so it also
10owns the largest share of the selftest.
11"""
12
13from __future__ import annotations
14
15import pathlib
16
17from annot_model import AnnotatedSymbol, Violation
18from annot_rulekeys import LINKAGE_ANNOTATIONS, parse_annotation
19from annot_scope import SOURCE_SUFFIXES, is_first_party, relative
20
21#: Header suffix that marks a declaration as library-private rather than
22#: published API. See CLAUDE.md, "Which linkage annotation to use".
23INTERNAL_HEADER_SUFFIX = "_internal.h"
24
25#: The ISO C program entry point. The startup code reaches `main` by name
26#: through the C runtime contract, so it has external linkage by
27#: definition and no first-party header declares it. Exactly one name --
28#: this is a language rule, not a naming convention.
29C_ENTRY_POINT = "main"
30
31
32def published_header(sym: AnnotatedSymbol) -> str | None:
33 """Return the header that publishes ``sym``, or None when none does.
34
35 A prototype in a header is an exported contract: a library's public
36 ``inc/`` header, an application's local header, a mock's header, or a
37 vendored SOUP header whose interface the function implements. A
38 prototype in a ``.c`` is a forward declaration and publishes nothing,
39 and an ``*_internal.h`` declaration is explicitly library-private --
40 both leave the function needing a linkage annotation.
41
42 "Header" is anything that is not a translation unit, rather than a
43 list of header suffixes: the C++ standard library headers that declare
44 the replaceable ``operator new`` / ``operator delete`` are spelled
45 ``<new>``, with no suffix at all.
46 """
47 for path in sorted(sym.decl_files):
48 name = pathlib.PurePath(path).name
49 if pathlib.PurePath(name).suffix in SOURCE_SUFFIXES:
50 continue
51 if name.endswith(INTERNAL_HEADER_SUFFIX):
52 continue
53 return path
54 return None
55
56
57def internal_header(sym: AnnotatedSymbol) -> str | None:
58 """Return the ``*_internal.h`` declaring ``sym``, or None."""
59 for path in sorted(sym.decl_files):
60 if pathlib.PurePath(path).name.endswith(INTERNAL_HEADER_SUFFIX):
61 return path
62 return None

65def _linkage_verdict(sym: AnnotatedSymbol, vector_entries: set[str]) -> Violation | None:
66 """Return the linkage violation for ``sym``, or None when it passes."""
67 if any(parse_annotation(a)[0] in LINKAGE_ANNOTATIONS for a in sym.annotations):
68 return None
69 if published_header(sym) is not None:
70 return None
71 if sym.usr in vector_entries or sym.name == C_ENTRY_POINT:
72 return None
73 internal = internal_header(sym)
74 if internal is not None:
75 return Violation(
76 "ra8_linkage",
77 sym.file,
78 sym.line,
79 f"'{sym.name}' is declared in {relative(internal)} but carries no "
80 f"linkage annotation; a symbol that is non-static only so one "
81 f"library's other TUs can reach it is RA8_PRIV (or RA8_TEST_HELPER "
82 f"when only tests call it)",
83 )
84 return Violation(
85 "ra8_linkage",
86 sym.file,
87 sym.line,
88 f"'{sym.name}' has external linkage but nothing publishes it: no header "
89 f"declares it and no vector table names it. Make it static and tag it "
90 f"RA8_INTERNAL, or declare it -- in the library's inc/ header if it is "
91 f"API, in an *_internal.h with RA8_PRIV if it is shared inside the library",
92 )

95def enforce_linkage(
96 symbols: dict[str, AnnotatedSymbol],
97 vector_entries: set[str],
98) -> list[Violation]:
99 """Every non-static function must state why it has external linkage.
100
101 CLAUDE.md, "Which linkage annotation to use", makes this a tree-wide
102 expectation rather than an opt-in: a function that is not `static`
103 either publishes a contract other code may call, or it is deliberately
104 non-`static` for one narrow reason that has to be written down.
105
106 A definition passes when any of the following holds.
107
108 * It carries `RA8_PRIV`, `RA8_INTERNAL` or `RA8_TEST_HELPER`.
109 * A header publishes it. A prototype in a `.h` is an exported
110 contract -- a library's public `inc/` header, an application's local
111 header, a mock's header, or the vendored SOUP header whose interface
112 the function implements (the NimBLE `ble_npl_*` porting layer, the
113 ThreadX `tx_application_define` hook, the USBX class entry points).
114 An `*_internal.h` prototype does not count: that header exists to
115 say "library-private", which is what `RA8_PRIV` marks.
116 * It is a hardware vector-table entry -- see `is_vector_table()`. The
117 CPU reaches these through VTOR with no C caller at all, so no
118 annotation describes them and no header can publish them: the only
119 thing that names an IRQ trampoline is the table slot itself. The
120 category is derived from the table's structure, so a handler that is
121 removed from the table stops being exempt the moment it is unwired.
122 * It is `main`, the ISO C entry point.
123
124 Anything else is a gap, and the two shapes need different fixes, so
125 they get different messages.
126
127 A verdict is reached per *definition site*, not per symbol name. The
128 coverage tests reach a module's `static` helpers by `#define`-ing a
129 function to a `_cov` spelling and then `#include`-ing the `.c`, so one
130 source construct shows up under two names with two USRs and the same
131 file and line. Judging the renamed copy on its own would report the
132 original function as an unpublished symbol in a file that never
133 declared it.
134 """
135 verdicts: dict[tuple[str, int], list[Violation | None]] = {}
136 for sym in symbols.values():
137 if not sym.is_defined or sym.is_static or not is_first_party(sym.file):
138 continue
139 verdicts.setdefault((sym.file, sym.line), []).append(_linkage_verdict(sym, vector_entries))
140 out: list[Violation] = []
141 for site in sorted(verdicts):
142 found = [v for v in verdicts[site] if v is not None]
143 if len(found) == len(verdicts[site]) and found:
144 out.append(found[0])
145 return out
