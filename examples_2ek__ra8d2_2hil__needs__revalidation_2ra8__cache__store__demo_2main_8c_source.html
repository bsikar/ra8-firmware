139static void rcs_print_u32(uint32_t value)
140{
141 char buf[k_rcs_u32_buf_size];
142 uint32_t i = (uint32_t)sizeof(buf);
143 buf[--i] = '\0';
144 if (value == 0U) {
145 buf[--i] = '0';
146 } else {
147 while ((value != 0U) && (i > 0U)) {
148 buf[--i] = (char)('0' + (value % (uint32_t)k_rcs_radix_dec));
149 value /= (uint32_t)k_rcs_radix_dec;
150 }
151 }
152 rcs_print(&buf[i]);
153}

167static void rcs_report(ra8_err_t rc, const cache_store_demo_result_t* res)

213int32_t main(void)
214{
215 rcs_setup_or_halt();
216 ra8_isr_globals_enable();
217 rcs_print("[rcs] cache_store demo: mounting RAM-backed LevelX...\r\n");
218
219 cache_store_demo_cfg_t cfg = rcs_build_cfg();
220 cache_store_demo_result_t res = {};
221 const ra8_err_t rc = cache_store_demo_run(&cfg, &res);
222 rcs_report(rc, &res);
223
224 /* Idle, re-emitting the verdict so the STOP_ON / scrape always sees the
225 * steady-state banner and the run reaches its budget cleanly. */
226 while (1) {
227 ra8_delay_ms((uint32_t)k_rcs_reemit_ms);
228 rcs_report(rc, &res);
229 }
230 return 0;
231}
