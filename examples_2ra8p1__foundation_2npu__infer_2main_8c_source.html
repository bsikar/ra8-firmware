304static ra8_err_t npu_infer_run_job_irq(void)
305{
306 ra8_npu_job_t job = {};
307 job.cmd_stream = s_npu_cmd_stream;
308 job.cmd_stream_bytes = (uint32_t)sizeof(s_npu_cmd_stream);
309 job.region_count = (uint8_t)k_npu_infer_region_count;
310 job.region_base[k_npu_infer_region_weights] = (uint64_t)(uintptr_t)s_npu_weights;
311 job.region_base[k_npu_infer_region_input] = (uint64_t)(uintptr_t)s_npu_input;
312 job.region_base[k_npu_infer_region_output] = (uint64_t)(uintptr_t)s_npu_output;
313
314 const ra8_err_t sub = ra8_npu_submit(&job);
315 if (sub != k_ra8_ok) {
316 return sub;
317 }
318 const ra8_err_t arm = ra8_npu_irq_arm();
319 if (arm != k_ra8_ok) {
320 return arm;
321 }
322 const ra8_err_t run = ra8_npu_run();
323 if (run != k_ra8_ok) {
324 return run;
325 }
326 return ra8_npu_wait_irq();
327}

339static bool npu_infer_verify(uint32_t* out_check)
340{
341 uint32_t check = (uint32_t)k_npu_infer_fnv_offset;
342 bool ok = true;
343 for (uint32_t i = 0U; i < (uint32_t)k_npu_infer_arena_bytes; i++) {
344 const int32_t expect_q =
345 ((int32_t)i & (int32_t)k_npu_infer_seed_mask) + (int32_t)k_npu_infer_zero_point;
346 const int32_t expect_out =
347 (expect_q + (int32_t)k_npu_infer_addk) & (int32_t)k_npu_infer_byte_mask;
348 const float expect_f =
349 s_infer_scale * (float)((int32_t)s_npu_output[i] - (int32_t)k_npu_infer_zero_point);
350 const float diff = (s_infer_output_f[i] > expect_f) ? (s_infer_output_f[i] - expect_f)
351 : (expect_f - s_infer_output_f[i]);
352 if ((int32_t)s_npu_input[i] != expect_q) {
353 ok = false; /* quantize stage */
354 }
355 if ((int32_t)s_npu_output[i] != expect_out) {
356 ok = false; /* NPU op stage (IRQ completion) */
357 }
358 if (diff > s_infer_deq_tol) {
359 ok = false; /* dequantize stage */
360 }
361 check = (check ^ (uint32_t)s_npu_output[i]) * (uint32_t)k_npu_infer_fnv_prime;
362 }
363 *out_check = check;
364 return ok;
365}

411static bool npu_infer_execute(uint32_t* out_id, bool* out_tflm, uint32_t* out_check)
412{
413 *out_id = 0U;
414 *out_tflm = (ra8_ethosu_kernel_available() == 1);
415 *out_check = 0U;
416
417 if (ra8_npu_init() != k_ra8_ok) {
418 return false;
419 }
420 (void)ra8_npu_read_id(out_id);
421
422 const ra8_err_t reg = ra8_isr_register((ra8_elc_event_t)k_ra8_npu_event_irq,
423 ra8_npu_irq_handler,
424 nullptr,
425 (uint8_t)k_ra8_isr_prio_default,
426 nullptr);
427 if (reg != k_ra8_ok) {
428 return false;
429 }
430 ra8_isr_globals_enable();
431
432 npu_infer_build_stream();
433 if (npu_infer_prepare_input() != k_ra8_ok) {
434 return false;
435 }
436 if (npu_infer_run_job_irq() != k_ra8_ok) {
437 return false;
438 }
439 if (ra8_npu_dequantize_u8(s_npu_output,
440 s_infer_output_f,
441 (size_t)k_npu_infer_arena_bytes,
442 s_infer_scale,
443 (int32_t)k_npu_infer_zero_point) != k_ra8_ok) {
444 return false;
445 }
446 return npu_infer_verify(out_check);
447}

460int32_t main(void)
461{
462 npu_infer_setup_or_halt();
463
464 uint32_t id = 0U;
465 bool tflm = false;
466 uint32_t check = 0U;
467 /* run_ok = init + IRQ-driven quantize/op/dequantize pipeline all matched. The
468 * TFLite-micro registration (tflm) is a separate leg reported on its own field;
469 * the verdict is PASS only when every leg holds. */
470 const bool run_ok = npu_infer_execute(&id, &tflm, &check);
471 const bool pass = run_ok && tflm && (id != 0U);
472
473 g_npu_infer_pass = pass ? 1U : 0U;
474 npu_infer_emit(id, tflm, run_ok, check, pass);
475
476 npu_infer_park();
477 return 0;
478}
