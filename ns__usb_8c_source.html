536static VOID ns_usb_worker(ULONG arg)
537{
538 (void)arg;
539
540 if (ns_usbx_stack_up() != UX_SUCCESS) {
541 return;
542 }
543 g_tz_usb_state = (uint32_t)k_tz_usb_state_stack_up;
544
545 if (ns_cdc_class_register() != UX_SUCCESS) {
546 return;
547 }
548 g_tz_usb_state = (uint32_t)k_tz_usb_state_cdc_ready;
549
550 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs) != k_ra8_ok) {
551 return;
552 }
553 g_tz_usb_state = (uint32_t)k_tz_usb_state_dcd_ready;
554
555 if (ra8_usb_device_attach(k_ra8_usb_speed_fs, true) != k_ra8_ok) {
556 return;
557 }
558 g_tz_usb_state = (uint32_t)k_tz_usb_state_attached;
559
560 /* Polled controller service: drives bus reset, chapter-9 SETUP, and the
561 * bulk auto-echo (all inside ra8_usb_dispatch -> internal_event_cb ->
562 * ux_dcd_ra8_usb_irq). Time-sliced against ::ns_host_worker at the same
563 * priority, so this yields each tick and the host's transfers land inside the
564 * dispatch service window. */
565 g_tz_usb_state = (uint32_t)k_tz_usb_state_dispatched;
566 while (1) {
567 ra8_usb_dispatch(k_ra8_usb_speed_fs);
568 g_tz_usb_intsts_or |= (uint32_t)ra8_usb_intsts0_snapshot(k_ra8_usb_speed_fs);
569 g_tz_usb_dispatch_count += 1U;
570 g_tz_nsc_cgc_usb_match += 1U;
571 }
572}
