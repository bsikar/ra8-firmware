219static ra8_err_t cam_put_hex(uint32_t value, uint8_t width)
220{
221 static const char k_hex[] = "0123456789ABCDEF";
222 RA8_CHECK_RANGE(width, 1U, 8U, k_ra8_err_invalid_arg);
223 char buf[8];
224 uint8_t i = width;
225 uint32_t v = value;
226 while (i > 0U) {
227 i = (uint8_t)(i - 1U);
228 buf[i] = k_hex[v & (uint32_t)k_cam_nibble_mask];
229 v >>= 4U;
230 }
231 return ra8_board_uart_console_write((const uint8_t*)buf, (size_t)width);
232}

249static ra8_err_t cam_put_u32(uint32_t value)
250{
251 if (value > (uint32_t)k_cam_dec_cap) {
252 return k_ra8_err_invalid_arg;
253 }
254 char buf[8];
255 uint8_t i = (uint8_t)sizeof(buf);
256 uint32_t v = value;
257 do {
258 i = (uint8_t)(i - 1U);
259 buf[i] = (char)('0' + (v % (uint32_t)k_cam_dec_base));
260 v /= (uint32_t)k_cam_dec_base;
261 } while ((v != 0U) && (i > 0U));
262 const uint32_t len = (uint32_t)((uint8_t)sizeof(buf) - i);
263 return ra8_board_uart_console_write((const uint8_t*)&buf[i], (size_t)len);
264}

343static bool cam_frame_is_plausible(void)
344{
345 const uint8_t* frame = cam_ceu_frame();
346 if (frame == nullptr) {
347 return false;
348 }
349 const uint32_t n = (uint32_t)k_cam_frame_bytes;
350 if (n == 0U) {
351 return false;
352 }
353 uint32_t lo = (uint32_t)k_cam_byte_max;
354 uint32_t hi = 0U;
355 uint32_t sum = 0U;
356 for (uint32_t i = 0U; i < n; i += 1U) {
357 const uint32_t b = (uint32_t)frame[i];
358 if (b < lo) {
359 lo = b;
360 }
361 if (b > hi) {
362 hi = b;
363 }
364 sum += b;
365 }
366 g_cam_min = lo;
367 g_cam_max = hi;
368 g_cam_mean = sum / n;
369 return hi != lo;
370}

485static ra8_err_t cam_bringup(void)
486{
487 uint32_t cpuclk0_hz = 0U;
488 ra8_err_t err = ra8_cgc_init();
489 if (err != k_ra8_ok) {
490 return err;
491 }
492 err = ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz);
493 if (err != k_ra8_ok) {
494 return err;
495 }
496 err = ra8_mstp_init();
497 if (err != k_ra8_ok) {
498 return err;
499 }
500 err = ra8_time_init(cpuclk0_hz);
501 if (err != k_ra8_ok) {
502 return err;
503 }
504 err = ra8_board_uart_console_init((uint32_t)k_cam_baud);
505 if (err != k_ra8_ok) {
506 return err;
507 }
508 err = cam_start_xclk();
509 if (err != k_ra8_ok) {
510 return err;
511 }
512 err = ra8_board_io_expander_apply_project_sw4_defaults();
513 if (err != k_ra8_ok) {
514 return err;
515 }
516 /* The board default is SW4-6 OFF (MIPI camera); flip it ON so the DVP
517 path + the P501 XVCLK reach the sensor. */
518 err = cam_select_parallel_camera();
519 if (err != k_ra8_ok) {
520 return err;
521 }
522 ra8_delay_ms((uint32_t)k_cam_mode_settle_ms);
523 return k_ra8_ok;
524}

541static void cam_capture_and_verdict(void)
542{
543 ra8_err_t err = cam_configure_sensor();
544 if (err == k_ra8_ok) {
545 err = cam_ceu_setup();
546 }
547 (void)cam_puts(" ceu=");
548 (void)cam_puts((err == k_ra8_ok) ? "OK" : "ERR");
549
550 bool frame_ok = false;
551 if (err == k_ra8_ok) {
552 frame_ok = (cam_capture_one() == k_ra8_ok);
553 }
554 g_cam_frame_ok = frame_ok ? 1U : 0U;
555 (void)cam_puts(" frame=");
556 (void)cam_puts(frame_ok ? "OK" : "TIMEOUT");
557
558 /* Emit the raw CEU event register (CETCR) so a timeout is diagnosable
559 from the banner alone: bit17=IGHS (HD cycle-count mismatch), bit20=VBP
560 (invalid VD), bit24/25=NHD/NVD (sync missing), bit0=CPE (frame done). */
561 uint32_t cetcr = 0U;
562 (void)cam_ceu_get_status(&cetcr);
563 (void)cam_puts(" cetcr=");
564 (void)cam_put_hex(cetcr, 8U);
565
566 bool plausible = false;
567 if (frame_ok) {
568 plausible = cam_frame_is_plausible();
569 }
570 (void)cam_puts(" min=");
571 (void)cam_put_u32(g_cam_min);
572 (void)cam_puts(" max=");
573 (void)cam_put_u32(g_cam_max);
574 (void)cam_puts(" mean=");
575 (void)cam_put_u32(g_cam_mean);
576
577 g_cam_verdict = plausible ? 1U : 0U;
578 (void)cam_puts(plausible ? " verdict=PASS\r\n" : " verdict=FAIL\r\n");
579}

595static void cam_run(void)
596{
597 (void)cam_puts((const char*)k_cam_tag);
598
599 /* Confirm the XCLK GPT is actually counting (a proxy that the P501
600 XVCLK square wave is live). The GTCNT wraps every few counts at the
601 ~24 MHz output rate, so sample it in a tight loop and call it live as
602 soon as any read differs from the first; a stuck GTCNT (every sample
603 identical) means the sensor never gets a clock. */
604 uint32_t cnt_ref = 0U;
605 bool gpt_running = false;
606 (void)ra8_gpt_read((uint8_t)k_cam_xclk_gpt_ch, &cnt_ref);
607 for (uint32_t i = 0U; i < (uint32_t)k_cam_gpt_probe_iters; i += 1U) {
608 uint32_t cnt = 0U;
609 (void)ra8_gpt_read((uint8_t)k_cam_xclk_gpt_ch, &cnt);
610 if (cnt != cnt_ref) {
611 gpt_running = true;
612 break;
613 }
614 }
615 (void)cam_puts(" gpt=");
616 (void)cam_puts(gpt_running ? "RUN" : "DEAD");
617
618 /* The OV5640 needs XVCLK (up in cam_bringup) and must be out of
619 hardware reset before it answers on SCCB, so route the DVP pins and
620 release RST (P709) FIRST, then probe. */
621 (void)cam_route_ceu_pins();
622 const ra8_err_t rst_err = cam_reset_sensor();
623
624 cam_bus_scan();
625
626 uint16_t chip = 0U;
627 const bool id_ok = cam_probe_sensor(&chip);
628 g_cam_chipid = (uint32_t)chip;
629 (void)cam_puts(" chipid=");
630 (void)cam_put_hex((uint32_t)chip, 4U);
631 (void)cam_puts(" xclk=OK rst=");
632 (void)cam_puts((rst_err == k_ra8_ok) ? "OK" : "ERR");
633 (void)cam_puts(" sccb=");
634 (void)cam_puts(id_ok ? "OK" : "ERR");
635
636 if (!id_ok) {
637 (void)cam_puts(" verdict=FAIL\r\n");
638 return;
639 }
640
641 cam_capture_and_verdict();
642}
