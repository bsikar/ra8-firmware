106ra8_err_t ra8_sdhi_init(uint8_t instance)
107{
108 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
109 RA8_CHECK_NULL_PTR(reg, s_tag, "instance out of range");
110
111 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B" p 444 */
112 const ra8_err_t mst_err = ra8_mstp_enable(s_sdhi_mstp_table[instance]);
113 RA8_RETURN_ON_ERROR(mst_err, s_tag, "sdhi_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
114
115 /* HUM Ch 47.2.15 "SD_INFO1 : SD Card Interrupt Flag Register 1" p 3140 */
116 /* Drop any stale flags before reset so the card-detection latch is
117 * not interpreted as an event by the consumer. */
118 reg->SD_INFO1 = 0U;
119
120 /* HUM Ch 47.2.31 "SOFT_RST : Software Reset Register" p 3173 */
121 /* Mirror FSP r_sdhi_hw_cfg(): assert reset (SDRST=0), release
122 * reset (SDRST=1). This brings every status flag and bus-state
123 * machine to a known idle state. */
124 reg->SOFT_RST = k_ra8_sdhi_soft_rst_assert;
125 reg->SOFT_RST = k_ra8_sdhi_soft_rst_release;
126
127 /* HUM Ch 47.2.18 "SD_CLK_CTRL : SD Clock Control Register" p 3145 */
128 /* Slowest CLKSEL (PCLKB/64) with CLKEN/CLKCTRLEN cleared; the
129 * card-stack consumer will retune for 25 MHz / 50 MHz once
130 * identification completes. */
131 reg->SD_CLK_CTRL = k_ra8_sdhi_clk_ctrl_default;
132
133 /* HUM Ch 47.2.27 "SDIO_MODE : SDIO Mode Control Register" p 3169 */
134 /* HUM Ch 47.2.30 "SD_DMAEN : DMA Mode Enable Register" p 3172 */
135 /* HUM Ch 47.2.32 "SDIF_MODE : SD Interface Mode Setting" p 3174 */
136 /* HUM Ch 47.2.33 "EXT_SWAP : Swap Control Register" p 3175 */
137 /* Disable SDIO mode, DMA, MMC test-CRC mask, and endian-swap so
138 * the FIFO interface starts in the canonical big-endian-aware
139 * configuration. */
140 reg->SDIO_MODE = 0U;
141 reg->SD_DMAEN = 0U;
142 reg->SDIF_MODE = 0U;
143 reg->EXT_SWAP = 0U;
144
145 /* HUM Ch 47.2.16 "SD_OPTION : SD Card Access Control Option Register" p 3139-3140 */
146 /* 1-bit-wide bus (WIDTH=1, WIDTH8=0), default timeout counters. The
147 * consumer flips WIDTH/WIDTH8 via ::ra8_sdhi_set_bus_width_4bit once
148 * the card acknowledges ACMD6. */
149 reg->SD_OPTION = (uint32_t)k_ra8_sdhi_option_bus_1bit;
150
151 /* HUM Ch 47.2.17 "SD_INFO2_MASK : SD Card Interrupt Mask 2" p 3144 */
152 /* HUM Ch 47.2.16 "SD_INFO1_MASK : SD Card Interrupt Mask 1" p 3142 */
153 /* All interrupts masked off until ::ra8_sdhi_attach_handler installs
154 * a callback. */
155 reg->SD_INFO1_MASK = 0U;
156 reg->SD_INFO2_MASK = 0U;
157 reg->SD_INFO2 = 0U;
158
159 ra8_log_info_val(s_tag, "sdhi_init inst", (uint32_t)instance);
160 return k_ra8_ok;
161}

182ra8_err_t ra8_sdhi_send_command(uint8_t instance, uint32_t cmd, uint32_t arg, uint32_t* out_rsp)
183{
184 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
185 RA8_CHECK_NULL_PTR(reg, s_tag, "instance out of range");
186
187 /* HUM Ch 47.2.2 "SD_ARG / SD_ARG1 : SD Command Argument" p 3128 */
188 /* HUM Ch 47.2.1 "SD_CMD : Command Type Register" p 3125 */
189 /* Load SD_ARG, then SD_CMD kicks the command out on the bus.
190 * FSP r_sdhi.c r_sdhi_command_send_no_wait() splits the argument
191 * into the legacy SD_ARG1 (upper 16 bits) plus SD_ARG (lower 16
192 * bits); on RA8D2 SD_ARG is a full 32-bit register so a single
193 * write is sufficient. */
194 reg->SD_ARG = arg;
195 reg->SD_CMD = cmd;
196
197 /* HUM Ch 47.2.15 "SD_INFO1 : SD Card Interrupt Flag Register 1" p 3140 */
198 /* Poll SD_INFO1.RSPEND (bit 0) with bounded spin budget. */
199 for (uint32_t i = 0U; i < k_ra8_sdhi_cmd_spin; ++i) { /* GCOVR_EXCL_BR_LINE */
200#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
201 if (ra8_fake_mmio_poll(&reg->SD_INFO1,
202 i,
203 (reg->SD_INFO1 & k_ra8_sdhi_info1_rspend_mask) !=
204 0U)) { /* GCOVR_EXCL_BR_LINE */
205#else
206 if ((reg->SD_INFO1 & k_ra8_sdhi_info1_rspend_mask) != 0U) { /* GCOVR_EXCL_BR_LINE */
207#endif
208 if (out_rsp != nullptr) {
209 /* HUM Ch 47.2.5 "SD_RSP10..SD_RSP76 : Response Registers" p 3132 */
210 out_rsp[0] = reg->SD_RSP10;
211 out_rsp[1] = reg->SD_RSP32;
212 out_rsp[2] = reg->SD_RSP54;
213 out_rsp[3] = reg->SD_RSP76;
214 }
215 reg->SD_INFO1 = reg->SD_INFO1 & ~k_ra8_sdhi_info1_rspend_mask;
216 return k_ra8_ok;
217 }
218 }
219 return k_ra8_err_hw_timeout;
220}

222ra8_err_t ra8_sdhi_set_clock(uint8_t instance, uint32_t divider)
223{
224 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
225 RA8_CHECK_NULL_PTR(reg, s_tag, "instance out of range");
226 /* HUM Ch 47.2.18 "SD_CLK_CTRL : SD Clock Control Register" p 3145 */
227 /* Preserve CLKEN (bit 8) and CLKCTRLEN (bit 9) -- caller controls
228 * only the CLKSEL[7:0] divider. FSP r_sdhi_set_clock() snapshots
229 * CLKCTRLEN, OR's CLKEN back in, and writes the merged value;
230 * mirror that semantics so a runtime divider change does not
231 * silently disable the SD bus clock. */
232 const uint32_t preserved =
233 reg->SD_CLK_CTRL & (k_ra8_sdhi_clk_ctrl_clkctrlen_mask | k_ra8_sdhi_clk_ctrl_clken_mask);
234 reg->SD_CLK_CTRL = (divider & k_ra8_sdhi_clk_ctrl_setting_mask) | preserved;
235 return k_ra8_ok;
236}

238ra8_err_t ra8_sdhi_set_bus_width(uint8_t instance, ra8_sdhi_bus_width_t width)
239{
240 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
241 RA8_CHECK_NULL_PTR(reg, s_tag, "set_bus_width: instance out of range");
242
243 /* Reject anything that is not one of the three valid lane counts.
244 * This compound decision (3 conditions) is MC/DC-tested in
245 * test_ra8_sdhi_width.c. */
246 if ((width != k_ra8_sdhi_bus_width_1bit) && (width != k_ra8_sdhi_bus_width_4bit) &&
247 (width != k_ra8_sdhi_bus_width_8bit)) {
248 return k_ra8_err_invalid_arg;
249 }
250
251 /* Map the lane count onto the SD_OPTION.WIDTH / WIDTH8 selector per
252 * the HUM Ch 47.2.16 truth table: 1-bit is WIDTH=1 (the initial
253 * value below), 4-bit is both bits clear, 8-bit is WIDTH8 only. */
254 uint32_t sel = (uint32_t)k_ra8_sdhi_option_width_bit;
255 if (width == k_ra8_sdhi_bus_width_4bit) {
256 sel = 0U;
257 } else if (width == k_ra8_sdhi_bus_width_8bit) {
258 sel = (uint32_t)k_ra8_sdhi_option_width8_bit;
259 }
260
261 /* HUM Ch 47.2.16 "SD_OPTION : SD Card Access Control Option Register" p 3139-3140 */
262 /* Read-modify-write: clear WIDTH+WIDTH8, OR in the new selector, and
263 * preserve TOP / CTOP / TOUTMASK / reserved bit 14. */
264 reg->SD_OPTION = (reg->SD_OPTION & ~(uint32_t)k_ra8_sdhi_option_width_mask) | sel;
265 return k_ra8_ok;
266}

268ra8_err_t ra8_sdhi_set_bus_width_4bit(uint8_t instance, uint16_t rca)
269{
270 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
271 RA8_CHECK_NULL_PTR(reg, s_tag, "set_bus_width_4bit: instance out of range");
272
273 uint32_t rsp[4] = {0U, 0U, 0U, 0U};
274 const uint32_t cmd55_arg = (uint32_t)rca << (uint32_t)k_ra8_sdhi_rca_arg_shift;
275
276 /* HUM Ch 47.2.1 "SD_CMD : Command Type Register" p 3125 */
277 /* CMD55 APP_CMD prefix (arg = RCA<<16) -- required before any ACMDxx. */
278 const ra8_err_t e55 =
279 ra8_sdhi_send_command(instance, (uint32_t)k_ra8_sdhi_cmd_app_cmd, cmd55_arg, rsp);
280 RA8_RETURN_ON_ERROR(e55, s_tag, "acmd6: CMD55 timeout"); /* GCOVR_EXCL_BR_LINE */
281 const uint32_t cmd55_rsp = rsp[0];
282
283 /* HUM Ch 47.2.2 "SD_ARG : SD Command Argument" p 3128 */
284 /* ACMD6 SET_BUS_WIDTH -- arg bits[1:0] = 0b10 selects the 4-bit bus. */
285 const ra8_err_t e6 = ra8_sdhi_send_command(instance,
286 (uint32_t)k_ra8_sdhi_cmd_set_bus_width,
287 (uint32_t)k_ra8_sdhi_acmd6_arg_4bit,
288 rsp);
289 RA8_RETURN_ON_ERROR(e6, s_tag, "acmd6: ACMD6 timeout"); /* GCOVR_EXCL_BR_LINE */
290 const uint32_t acmd6_rsp = rsp[0];
291
292 /* Card acknowledges only when CMD55 echoed APP_CMD AND the ACMD6 R1
293 * response carries no error bits. This compound decision (2
294 * conditions) is MC/DC-tested in test_ra8_sdhi_width.c. On any
295 * rejection the host stays at the 1-bit default. */
296 const bool app_cmd_ready = (cmd55_rsp & (uint32_t)k_ra8_sdhi_r1_app_cmd_mask) != 0U;
297 const bool acmd6_clean = (acmd6_rsp & (uint32_t)k_ra8_sdhi_r1_error_mask) == 0U;
298 if ((!app_cmd_ready) || (!acmd6_clean)) {
299 return k_ra8_err_not_supported;
300 }
301
302 /* Card switched to 4-bit -- widen the host side to match. */
303 return ra8_sdhi_set_bus_width(instance, k_ra8_sdhi_bus_width_4bit);
304}

564#endif

576 cursor += k_ra8_sdhi_fifo_word_bytes;
577 }

703ra8_err_t
704ra8_sdhi_write_block(uint8_t instance, uint32_t lba, const uint8_t* buf, uint32_t block_count)
705{
706 RA8_CHECK_NULL_PTR(buf, s_tag, "write_block: buf must not be nullptr");
707 if (block_count == 0U) {
708 return k_ra8_err_invalid_arg;
709 }
710 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
711 RA8_CHECK_NULL_PTR(reg, s_tag, "write_block: instance out of range");
712
713 internal_sdhi_setup_xfer(reg, block_count);
714
715 const uint32_t cmd = (block_count > 1U) ? (uint32_t)k_ra8_sdhi_cmd_write_multi_block
716 : (uint32_t)k_ra8_sdhi_cmd_write_single_block;
717 const ra8_err_t cmd_err = internal_sdhi_send(reg, cmd, lba);
718 RA8_RETURN_ON_ERROR(cmd_err, s_tag, "write_block: RSPEND timeout"); /* GCOVR_EXCL_BR_LINE */
719
720 /* HUM Ch 47.2.21 "SD_BUF0 : SD Buffer Register" p 3150 */
721 const uint32_t total_words = block_count * k_ra8_sdhi_words_per_block;
722 const ra8_err_t fill_err = internal_sdhi_fill(reg, buf, total_words);
723 RA8_RETURN_ON_ERROR(fill_err, s_tag, "write_block: BWE timeout"); /* GCOVR_EXCL_BR_LINE */
724
725 return internal_sdhi_finish_xfer(reg, block_count);
726}

728ra8_err_t ra8_sdhi_attach_dma(uint8_t instance, uint8_t enable)
729{
730 volatile r_sdhi_regs_t* reg = ra8_sdhi(instance);
731 RA8_CHECK_NULL_PTR(reg, s_tag, "attach_dma: instance out of range");
732
733 /* HUM Ch 47.2.30 "SD_DMAEN : DMA Mode Enable Register" p 3172 */
734 /* HUM Ch 47.2.17 "SD_INFO2_MASK : SD Card Interrupt Mask 2" p 3144 */
735 /* FSP r_sdhi_transfer_read / r_sdhi_transfer_write set the BREM
736 * and BWEM mask bits whenever DMAEN is asserted, so that the
737 * DMA-driven path does not race with the polled BRE / BWE wait
738 * the PIO path uses. Mirror that pairing here. */
739 if (enable != 0U) {
740 reg->SD_INFO2_MASK |= k_ra8_sdhi_info2_brem_bwem;
741 reg->SD_DMAEN = k_ra8_sdhi_dmaen_set;
742 } else {
743 reg->SD_DMAEN = 0U;
744 reg->SD_INFO2_MASK &= ~k_ra8_sdhi_info2_brem_bwem;
745 }
746 return k_ra8_ok;
747}
