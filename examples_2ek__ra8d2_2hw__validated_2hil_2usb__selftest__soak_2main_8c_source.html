177static VOID selftest_host_worker(ULONG arg)
178{
179 (void)arg;
180
181 tx_thread_sleep(k_selftest_boot_wait_ticks);
182 for (;;) {
183 const ra8_err_t err = selftest_host_pass();
184 if (err == k_ra8_ok) {
185 break;
186 }
187 tx_thread_sleep(k_selftest_retry_ticks);
188 }
189 while (1) {
190 tx_thread_sleep(k_selftest_idle_ticks);
191 }
192}

276static void selftest_route_usb_or_halt(void)
277{
278 /* FS port: device role. */
279 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_vbus, k_ra8_psel_usb_fs, "selftest.fs_vbus") !=
280 k_ra8_ok) {
281 selftest_panic_halt();
282 }
283 if (ra8_gpio_output_init(k_selftest_pin_fs_vbusen, k_ra8_level_low) != k_ra8_ok) {
284 selftest_panic_halt();
285 }
286 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_dp, k_ra8_psel_usb_fs, "selftest.fs_dp") !=
287 k_ra8_ok) {
288 selftest_panic_halt();
289 }
290 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_dm, k_ra8_psel_usb_fs, "selftest.fs_dm") !=
291 k_ra8_ok) {
292 selftest_panic_halt();
293 }
294 /* HS port: host role. */
295 if (ra8_board_io_expander_set_usbhs_host_mode() != k_ra8_ok) {
296 selftest_panic_halt();
297 }
298 if (ra8_gpio_output_init(k_selftest_pin_hs_pwr, k_ra8_level_high) != k_ra8_ok) {
299 selftest_panic_halt();
300 }
301 if (ra8_pfs_route_peripheral(k_selftest_pin_hs_vbus, k_ra8_psel_usb_hs, "selftest.hs_vbus") !=
302 k_ra8_ok) {
303 selftest_panic_halt();
304 }
305}

322static void selftest_setup_or_halt(void)
323{
324 uint32_t cpuclk0_hz = 0U;
325 if (ra8_cgc_init() != k_ra8_ok) {
326 selftest_panic_halt();
327 }
328 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
329 selftest_panic_halt();
330 }
331 if (ra8_cgc_usbhs_pll_enable() != k_ra8_ok) {
332 selftest_panic_halt();
333 }
334 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
335 selftest_panic_halt();
336 }
337 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
338 selftest_panic_halt();
339 }
340 if (ra8_board_uart_console_init((uint32_t)k_selftest_baud) != k_ra8_ok) {
341 selftest_panic_halt();
342 }
343 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
344 selftest_panic_halt();
345 }
346 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
347 selftest_panic_halt();
348 }
349 selftest_route_usb_or_halt();
350}
