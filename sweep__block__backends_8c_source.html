83static uint64_t cbs_rng(uint64_t* s)
84{
85 uint64_t x = *s;
86 x ^= x << (uint8_t)k_cbs_shift_a;
87 x ^= x >> (uint8_t)k_cbs_shift_b;
88 x ^= x << (uint8_t)k_cbs_shift_c;
89 *s = x;
90 return x;
91}

94void cbs_priv_fill_text(uint8_t* blob, uint32_t len)
95{
96 if ((blob == nullptr) || (len == 0U)) {
97 return;
98 }
99 const uint32_t nwords = (uint32_t)(sizeof(s_cbs_words) / sizeof(s_cbs_words[0]));
100 uint64_t s = (uint64_t)k_cbs_seed_fill;
101 uint32_t at = 0U;
102 uint32_t word = 0U;
103 while (at < len) {
104 const char* w = s_cbs_words[(uint32_t)(cbs_rng(&s) % (uint64_t)nwords)];
105 for (const char* p = w; (*p != '\0') && (at < len); ++p) {
106 blob[at] = (uint8_t)*p;
107 at++;
108 }
109 word++;
110 if (word == (uint32_t)k_cbs_words_per_dot) {
111 word = 0U;
112 if (at < len) {
113 blob[at] = (uint8_t)'.';
114 at++;
115 }
116 if (at < len) {
117 blob[at] = (uint8_t)'\n';
118 at++;
119 }
120 } else if (at < len) {
121 blob[at] = (uint8_t)' ';
122 at++;
123 }
124 }
125}

157 * Reason: bound as an ra8_vsource_read_fn, whose signature is

161static ra8_err_t cbs_memspan_read(void* ctx, uint64_t offset, uint8_t* buf, uint32_t len)
162{

165 return k_ra8_err_null_ptr;
166 }
167 if ((offset + (uint64_t)len) > sp->len) {
168 return k_ra8_err_out_of_range;
169 }
170 memcpy(buf, &sp->data[offset], (size_t)len);
171 return k_ra8_ok;
172}

252static ra8_err_t
253cbs_inflate(const void* src, size_t src_len, void* dst, size_t dst_cap, size_t* out_len)
254{
255 if ((src == nullptr) || (dst == nullptr) || (out_len == nullptr)) {
256 return k_ra8_err_null_ptr;
257 }
258 tinfl_init(&s_cbs_tinfl);
259 size_t in_n = src_len;
260 size_t out_n = dst_cap;
261 const tinfl_status st = tinfl_decompress(
262 &s_cbs_tinfl,
263 (const mz_uint8*)src,
264 &in_n,
265 (mz_uint8*)dst,
266 (mz_uint8*)dst,
267 &out_n,
268 (mz_uint32)(TINFL_FLAG_PARSE_ZLIB_HEADER | TINFL_FLAG_USING_NON_WRAPPING_OUTPUT_BUF));
269 if (st != TINFL_STATUS_DONE) {
270 return k_ra8_err_invalid_size;
271 }
272 *out_len = out_n;
273 return k_ra8_ok;
274}

318static int cbs_rbkc_pack(const uint8_t* blob,
319 uint32_t blob_bytes,
320 uint32_t block_bytes,
321 uint8_t* out,
322 uint64_t out_cap,
323 uint64_t* out_len)
324{
325 if ((blob == nullptr) || (out == nullptr) || (out_len == nullptr) || (block_bytes == 0U)) {
326 return 1;
327 }
328 const uint32_t count = (blob_bytes + block_bytes - 1U) / block_bytes;
329 const uint64_t table_off = (uint64_t)k_ra8_book_container_header_len;
330 const uint64_t payload_off =
331 table_off + (((uint64_t)count + 1U) * k_ra8_book_container_entry_len);
332 cbs_rbkc_header(out, block_bytes, (uint64_t)blob_bytes, count);
333 uint64_t cur = 0U;
334 memcpy(&out[table_off], &cur, sizeof(cur));
335 for (uint32_t i = 0U; i < count; ++i) {
336 const uint32_t at = i * block_bytes;
337 uint32_t span = blob_bytes - at;
338 if (span > block_bytes) {
339 span = block_bytes;
340 }
341 mz_ulong dst_len = (mz_ulong)(out_cap - (payload_off + cur));
342 const int rc = mz_compress2(&out[payload_off + cur],
343 &dst_len,
344 &blob[at],
345 (mz_ulong)span,
346 MZ_BEST_COMPRESSION);
347 if (rc != MZ_OK) {
348 return 1;
349 }
350 cur += (uint64_t)dst_len;
351 memcpy(&out[table_off + (((uint64_t)i + 1U) * k_ra8_book_container_entry_len)],
352 &cur,
353 sizeof(cur));
354 }
355 *out_len = payload_off + cur;
356 return 0;
357}

399static int
400cbs_rbkc_setup(cbs_backend_t* be, const uint8_t* blob, uint32_t blob_bytes, uint32_t block_bytes)
401{
402 if ((be == nullptr) || (blob == nullptr) || (blob_bytes == 0U) || (block_bytes == 0U)) {
403 return 1;
404 }
405 cbs_rbkc_t* rb = &s_cbs_rbkc;
406 *rb = (cbs_rbkc_t){};
407 const uint32_t count = (blob_bytes + block_bytes - 1U) / block_bytes;
408 const uint64_t bound = (uint64_t)mz_compressBound((mz_ulong)block_bytes);
409 const uint64_t cap = (uint64_t)k_ra8_book_container_header_len +
410 (((uint64_t)count + 1U) * k_ra8_book_container_entry_len) +
411 ((uint64_t)count * bound);
412 rb->container = (uint8_t*)malloc((size_t)cap);
413 rb->table = (uint64_t*)malloc(((size_t)count + 1U) * sizeof(uint64_t));
414 rb->staging = (uint8_t*)malloc((size_t)bound);
415 if ((rb->container == nullptr) || (rb->table == nullptr) || (rb->staging == nullptr)) {
416 cbs_rbkc_teardown(be);
417 return 1;
418 }
419 if (cbs_rbkc_pack(blob, blob_bytes, block_bytes, rb->container, cap, &rb->file_len) != 0) {
420 cbs_rbkc_teardown(be);
421 return 1;
422 }
423 rb->span = (cbs_memspan_t){.data = rb->container, .len = rb->file_len};
424 rb->file_meter = (cbs_meter_t){.inner = cbs_memspan_read, .inner_ctx = &rb->span};
425 const ra8_err_t err = ra8_book_chunked_open(&rb->rd,
426 cbs_priv_meter_read,
427 &rb->file_meter,
428 rb->file_len,
429 cbs_inflate,
430 rb->table,
431 count + 1U,
432 rb->staging,
433 (uint32_t)bound);
434 if (err != k_ra8_ok) {
435 (void)fprintf(stderr, "sweep-block: rbkc open failed (err=%d)\n", (int)err);
436 cbs_rbkc_teardown(be);
437 return 1;
438 }
439 be->read = ra8_book_chunked_read;
440 be->read_ctx = &rb->rd;
441 be->backing_bytes = rb->file_len;
442 be->src_bytes = &rb->file_meter.bytes;
443 return 0;
444}
