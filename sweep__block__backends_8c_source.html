102static uint64_t cbs_rng(uint64_t* s)
103{
104 uint64_t x = *s;
105 x ^= x << (uint8_t)k_cbs_shift_a;
106 x ^= x >> (uint8_t)k_cbs_shift_b;
107 x ^= x << (uint8_t)k_cbs_shift_c;
108 *s = x;
109 return x;
110}

113void cbs_priv_fill_text(uint8_t* blob, uint32_t len)
114{
115 if ((blob == nullptr) || (len == 0U)) {
116 return;
117 }
118 const uint32_t nwords = (uint32_t)(sizeof(s_cbs_words) / sizeof(s_cbs_words[0]));
119 uint64_t s = (uint64_t)k_cbs_seed_fill;
120 uint32_t at = 0U;
121 uint32_t word = 0U;
122 while (at < len) {
123 const char* w = s_cbs_words[(uint32_t)(cbs_rng(&s) % (uint64_t)nwords)];
124 for (const char* p = w; (*p != '\0') && (at < len); ++p) {
125 blob[at] = (uint8_t)*p;
126 at++;
127 }
128 word++;
129 if (word == (uint32_t)k_cbs_words_per_dot) {
130 word = 0U;
131 if (at < len) {
132 blob[at] = (uint8_t)'.';
133 at++;
134 }
135 if (at < len) {
136 blob[at] = (uint8_t)'\n';
137 at++;
138 }
139 } else if (at < len) {
140 blob[at] = (uint8_t)' ';
141 at++;
142 }
143 }
144}

205{

209 }
210 if ((offset + (uint64_t)len) > sp->len) {
211 return k_ra8_err_out_of_range;
212 }
213 memcpy(buf, &sp->data[offset], (size_t)len);
214 return k_ra8_ok;
215}

362static ra8_err_t
363cbs_inflate(const void* src, size_t src_len, void* dst, size_t dst_cap, size_t* out_len)
364{
365 if ((src == nullptr) || (dst == nullptr) || (out_len == nullptr)) {
366 return k_ra8_err_null_ptr;
367 }
368 tinfl_init(&s_cbs_tinfl);
369 size_t in_n = src_len;
370 size_t out_n = dst_cap;
371 const tinfl_status st = tinfl_decompress(
372 &s_cbs_tinfl,
373 (const mz_uint8*)src,
374 &in_n,
375 (mz_uint8*)dst,
376 (mz_uint8*)dst,
377 &out_n,
378 (mz_uint32)(TINFL_FLAG_PARSE_ZLIB_HEADER | TINFL_FLAG_USING_NON_WRAPPING_OUTPUT_BUF));
379 if (st != TINFL_STATUS_DONE) {
380 return k_ra8_err_invalid_size;
381 }
382 *out_len = out_n;
383 return k_ra8_ok;
384}

449static int cbs_rbkc_pack(const uint8_t* blob,
450 uint32_t blob_bytes,
451 uint32_t block_bytes,
452 uint8_t* out,
453 uint64_t out_cap,
454 uint64_t* out_len)
455{
456 if ((blob == nullptr) || (out == nullptr) || (out_len == nullptr) || (block_bytes == 0U)) {
457 return 1;
458 }
459 const uint32_t count = (blob_bytes + block_bytes - 1U) / block_bytes;
460 const uint64_t table_off = (uint64_t)k_ra8_book_container_header_len;
461 const uint64_t payload_off =
462 table_off + (((uint64_t)count + 1U) * k_ra8_book_container_entry_len);
463 cbs_rbkc_header(out, block_bytes, (uint64_t)blob_bytes, count);
464 uint64_t cur = 0U;
465 memcpy(&out[table_off], &cur, sizeof(cur));
466 for (uint32_t i = 0U; i < count; ++i) {
467 const uint32_t at = i * block_bytes;
468 uint32_t span = blob_bytes - at;
469 if (span > block_bytes) {
470 span = block_bytes;
471 }
472 mz_ulong dst_len = (mz_ulong)(out_cap - (payload_off + cur));
473 const int rc = mz_compress2(&out[payload_off + cur],
474 &dst_len,
475 &blob[at],
476 (mz_ulong)span,
477 MZ_BEST_COMPRESSION);
478 if (rc != MZ_OK) {
479 return 1;
480 }
481 cur += (uint64_t)dst_len;
482 memcpy(&out[table_off + (((uint64_t)i + 1U) * k_ra8_book_container_entry_len)],
483 &cur,
484 sizeof(cur));
485 }
486 *out_len = payload_off + cur;
487 return 0;
488}

549static int
550cbs_rbkc_setup(cbs_backend_t* be, const uint8_t* blob, uint32_t blob_bytes, uint32_t block_bytes)
551{
552 if ((be == nullptr) || (blob == nullptr) || (blob_bytes == 0U) || (block_bytes == 0U)) {
553 return 1;
554 }
555 cbs_rbkc_t* rb = &s_cbs_rbkc;
556 *rb = (cbs_rbkc_t){};
557 const uint32_t count = (blob_bytes + block_bytes - 1U) / block_bytes;
558 const uint64_t bound = (uint64_t)mz_compressBound((mz_ulong)block_bytes);
559 const uint64_t cap = (uint64_t)k_ra8_book_container_header_len +
560 (((uint64_t)count + 1U) * k_ra8_book_container_entry_len) +
561 ((uint64_t)count * bound);
562 rb->container = (uint8_t*)malloc((size_t)cap);
563 rb->table = (uint64_t*)malloc(((size_t)count + 1U) * sizeof(uint64_t));
564 rb->staging = (uint8_t*)malloc((size_t)bound);
565 if ((rb->container == nullptr) || (rb->table == nullptr) || (rb->staging == nullptr)) {
566 cbs_rbkc_teardown(be);
567 return 1;
568 }
569 if (cbs_rbkc_pack(blob, blob_bytes, block_bytes, rb->container, cap, &rb->file_len) != 0) {
570 cbs_rbkc_teardown(be);
571 return 1;
572 }
573 rb->span = (cbs_memspan_t){.data = rb->container, .len = rb->file_len};
574 rb->file_meter = (cbs_meter_t){.inner = cbs_memspan_read, .inner_ctx = &rb->span};
575 const ra8_err_t err = ra8_book_chunked_open(&rb->rd,
576 cbs_priv_meter_read,
577 &rb->file_meter,
578 rb->file_len,
579 cbs_inflate,
580 rb->table,
581 count + 1U,
582 rb->staging,
583 (uint32_t)bound);
584 if (err != k_ra8_ok) {
585 (void)fprintf(stderr, "sweep-block: rbkc open failed (err=%d)\n", (int)err);
586 cbs_rbkc_teardown(be);
587 return 1;
588 }
589 be->read = ra8_book_chunked_read;
590 be->read_ctx = &rb->rd;
591 be->backing_bytes = rb->file_len;
592 be->src_bytes = &rb->file_meter.bytes;
593 return 0;
594}
