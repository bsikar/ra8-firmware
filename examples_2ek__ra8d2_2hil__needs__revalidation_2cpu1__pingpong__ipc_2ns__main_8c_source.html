274[[gnu::section(".ns_text")]] static void ns_ipc_channel_reset(uintptr_t ch_base)
275{
276 /* HUM Ch 3.2.14 "IPC0CLR0" p 216 -- RST resets the FIFO and clears
277 * RDY/FULL atomically; RCLR/FCLR clear the sticky FIFO error bits;
278 * CLR7..CLR0 drop any latent IRQ event bits. */
279 ns_write32(ch_base + (uintptr_t)k_ns_ipc_off_clr, (uint32_t)k_ns_ipc_clr_all);
280}

296[[gnu::section(".ns_text")]] static void ns_ipc_send(uintptr_t ch_base, uint32_t msg)
297{
298 /* HUM Ch 3.2.12 "IPC0TXD0" p 215-216 -- writes push the word onto
299 * the FIFO; FERR is raised separately and is cleared by
300 * ns_ipc_channel_reset on init. */
301 ns_write32(ch_base + (uintptr_t)k_ns_ipc_off_txd, msg);
302}

320[[gnu::section(".ns_text")]] static uint32_t ns_ipc_recv(uintptr_t ch_base, uint32_t* out_word)
321{
322 if (out_word == ((void*)0)) {
323 return 1U;
324 }
325 for (uint32_t i = 0U; i < (uint32_t)k_ns_recv_poll_max; ++i) {
326 const uint32_t sta = ns_read32(ch_base + (uintptr_t)k_ns_ipc_off_sta);
327 if ((sta & (uint32_t)k_ns_ipc_sta_rdy) != 0U) {
328 /* HUM Ch 3.2.13 "IPC0RXD0" p 216 -- reading RXD pops one word
329 * from the FIFO and re-evaluates RDY for the next iteration. */
330 const uint32_t w = ns_read32(ch_base + (uintptr_t)k_ns_ipc_off_rxd);
331 *out_word = w;
332 return 0U;
333 }
334 }
335 return 1U;
336}

372[[gnu::section(".ns_text"), noreturn]] void ns_reset_handler(void)
373{
374 /* Zero the NS BSS region. NOLOAD means the linker does not populate
375 * initial values from MRAM, and cold-boot SRAM contents are
376 * undefined. The address comparison is via uintptr_t because the
377 * two linker-provided extern symbols are different objects from a
378 * static-analysis perspective even though the linker guarantees
379 * they bracket a contiguous BSS run. */
380 const uintptr_t bss_start = (uintptr_t)&g_ra8_ls_ns_bss_start;
381 const uintptr_t bss_end = (uintptr_t)&g_ra8_ls_ns_bss_end;
382 for (uintptr_t a = bss_start; a < bss_end; a += sizeof(uint32_t)) {
383 *(volatile uint32_t*)a = 0U;
384 }
385
386 /* Stamp the entry marker AFTER the bss zero so the bench can tell
387 * "BLXNS reached NS" apart from "marker happened to land back at 0". */
388 g_ns_pingpong_entry_marker = (uint32_t)k_ns_entry_marker;
389
390 g_ns_pingpong_step = (uint32_t)k_ns_step_bss_zeroed;
391
392 /* Cold-init the CPU0-owned channels. */
393 ns_ipc_channel_reset((uintptr_t)k_ns_ipc_ch0_addr);
394 ns_ipc_channel_reset((uintptr_t)k_ns_ipc_ch2_addr);
395 g_ns_pingpong_step = (uint32_t)k_ns_step_channels_up;
396
397 for (;;) {
398 g_ns_pingpong_step = (uint32_t)k_ns_step_tx_start;
399 ns_ipc_send((uintptr_t)k_ns_ipc_ch2_addr, (uint32_t)k_ns_magic_ping);
400 g_ns_pingpong_step = (uint32_t)k_ns_step_tx_done;
401
402 uint32_t got = 0U;
403 if (ns_ipc_recv((uintptr_t)k_ns_ipc_ch0_addr, &got) != 0U) {
404 g_ns_pingpong_step = (uint32_t)k_ns_step_rx_timeout;
405 g_ns_pingpong_mismatch += 1U;
406 continue;
407 }
408 g_ns_pingpong_step = (uint32_t)k_ns_step_rx_word;
409 g_ns_pingpong_last_rxd = got;
410 if (got != (uint32_t)k_ns_magic_pong) {
411 g_ns_pingpong_mismatch += 1U;
412 continue;
413 }
414 g_ns_pingpong_step = 4U;
415 g_ns_pingpong_match += 1U;
416 }
417}
