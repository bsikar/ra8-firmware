36/* =============================================================================
37 * Host-mode bulk-transfer engine (polled, synchronous)
38 *
39 * Built on the same host signals the control engine validated on hardware:
40 * BRDY (a packet landed in the pipe buffer), BEMP (the pipe buffer emptied
41 * onto the wire), and PIPECTR.PID for STALL detection. Pipes reuse the CFIFO
42 * port (CFIFOSEL.CURPIPE selects the pipe; the data direction is fixed by
43 * PIPECFG.DIR, so ISEL stays clear for non-DCP pipes).
44 * =============================================================================
45 */

78static ra8_err_t
79internal_host_wait_pipe(volatile r_usb_regs_t* reg, volatile const uint16_t* sts, uint8_t pipe_num)
80{
81 const uint16_t bit = (uint16_t)(1U << pipe_num);
82 const uint8_t idx = (uint8_t)(pipe_num - 1U);
83 for (uint32_t i = 0U; i < (uint32_t)k_ra8_usb_bulk_poll_limit; ++i) {
84 if ((*sts & bit) != 0U) {
85 return k_ra8_ok;
86 }
87 if ((reg->PIPECTR[idx] & (uint16_t)k_ra8_usb_pid_stall_bit) != 0U) {
88 /* STALL bit is SIE-async; callers force PID=BUF before this spin, the fake cannot raise it. */
89 return k_ra8_err_hw_error; /* GCOVR_EXCL_LINE */
90 }
91 }
92 return k_ra8_err_hw_timeout;
93}

199ra8_err_t ra8_usb_host_pipe_setup(ra8_usb_speed_t speed,
200 uint8_t pipe_num,
201 uint8_t dev_addr,
202 uint8_t ep_num,
203 bool device_to_host,
204 uint16_t max_packet)
205{
206 volatile r_usb_regs_t* reg = internal_pick(speed);
207 if (reg == nullptr) {
208 return k_ra8_err_invalid_arg;
209 }
210 const ra8_err_t arg_err = internal_host_pipe_args_ok(pipe_num, dev_addr, ep_num, max_packet);
211 if (arg_err != k_ra8_ok) {
212 return arg_err;
213 }
214 internal_pipe_quiesce(reg, pipe_num);
215 const ra8_usb_ep_dir_t cfg_dir = device_to_host ? k_ra8_usb_ep_dir_out : k_ra8_usb_ep_dir_in;
216 reg->PIPESEL = pipe_num;
217 reg->PIPECFG = internal_pipecfg_word(ep_num, cfg_dir, k_ra8_usb_ep_type_bulk, true);
218 reg->PIPEBUF = internal_pipebuf_word(pipe_num, max_packet);
219 reg->PIPEMAXP = (uint16_t)((uint16_t)((uint16_t)dev_addr << (uint16_t)k_ra8_usb_devsel_shift) |
220 (uint16_t)(max_packet & (uint16_t)k_ra8_usb_pipemaxp_mxps));
221 reg->PIPEPERI = 0U;
222 reg->PIPESEL = 0U;
223 internal_rmw16(&reg->PIPECTR[(uint8_t)(pipe_num - 1U)],
224 (uint16_t)(1U << k_ra8_dcpctr_bit_sqclr),
225 0U);
226 const uint16_t pipe_bit = (uint16_t)(1U << pipe_num);
227 reg->BRDYSTS = (uint16_t)~pipe_bit;
228 reg->NRDYSTS = (uint16_t)~pipe_bit;
229 reg->BEMPSTS = (uint16_t)~pipe_bit;
230 return k_ra8_ok;
231}

305RA8_INTERNAL
306static ra8_err_t internal_host_bulk_rx_packet(volatile r_usb_regs_t* reg,
307 uint8_t pipe_num,
308 uint8_t* dst,
309 uint16_t room,
310 uint16_t* out_dtln,
311 uint16_t* out_copied)
312{
313 const ra8_err_t werr = internal_host_wait_pipe(reg, &reg->BRDYSTS, pipe_num);
314 if (werr != k_ra8_ok) {
315 return werr;
316 }
317 /* Acknowledge the BRDY edge BEFORE draining the buffer: at high speed
318 * the device's next packet (e.g. the CSW right behind a full data
319 * packet) lands while the FIFO is being read, and clearing the status
320 * afterwards wipes that new edge (observed as the follow-up packet
321 * stuck with BSTS=1 while the wait times out). */
322 reg->BRDYSTS = (uint16_t)~(uint16_t)(1U << pipe_num);
323 internal_select_cfifo(reg, (uint16_t)pipe_num, false);
324 const ra8_err_t ferr = internal_wait_frdy(reg);
325 if (ferr != k_ra8_ok) {
326 return ferr;
327 }
328 const uint16_t dtln = (uint16_t)(reg->CFIFOCTR & (uint16_t)k_ra8_fifoctr_dtln);
329 uint16_t copy = dtln;
330 if (copy > room) {
331 copy = room;
332 }
333 if (copy > 0U) {
334 internal_fifo_read(reg, dst, copy);
335 }
336 if (copy < dtln) {
337 reg->CFIFOCTR = (uint16_t)k_ra8_fifoctr_bclr;
338 }
339 if (dtln == 0U) {
340 reg->CFIFOCTR = (uint16_t)k_ra8_fifoctr_bclr;
341 }
342 *out_dtln = dtln;
343 *out_copied = copy;
344 return k_ra8_ok;
345}

368RA8_INTERNAL
369static ra8_err_t internal_host_bulk_rx_loop(volatile r_usb_regs_t* reg,
370 uint8_t pipe_num,
371 uint8_t* buf,
372 uint16_t max_len,
373 uint16_t mps,
374 uint16_t* out_rx)
375{
376 uint16_t rx = 0U;
377 ra8_err_t err = k_ra8_ok;
378 bool done = false;
379 while (!done) {
380 uint16_t dtln = 0U;
381 uint16_t copied = 0U;
382 err = internal_host_bulk_rx_packet(reg,
383 pipe_num,
384 &buf[rx],
385 (uint16_t)(max_len - rx),
386 &dtln,
387 &copied);
388 if (err != k_ra8_ok) {
389 done = true;
390 } else {
391 rx = (uint16_t)(rx + copied);
392 if (dtln < mps) {
393 done = true;
394 }
395 if (rx >= max_len) {
396 done = true;
397 }
398 }
399 }
400 *out_rx = rx;
401 return err;
402}

425ra8_err_t ra8_usb_host_bulk_in(ra8_usb_speed_t speed,
426 uint8_t pipe_num,
427 uint8_t* buf,
428 uint16_t max_len,
429 uint16_t* out_received)
430{
431 RA8_CHECK_NULL_PTR(buf, s_tag, "host_bulk_in: buf");
432 RA8_CHECK_NULL_PTR(out_received, s_tag, "host_bulk_in: out_received");
433 volatile r_usb_regs_t* reg = internal_pick(speed);
434 if (reg == nullptr) {
435 return k_ra8_err_invalid_arg;
436 }
437 if (pipe_num == 0U) {
438 return k_ra8_err_invalid_arg;
439 }
440 if (pipe_num > (uint8_t)k_ra8_usb_max_pipe_num) {
441 return k_ra8_err_invalid_arg;
442 }
443 reg->PIPESEL = pipe_num;
444 const uint16_t mps = (uint16_t)(reg->PIPEMAXP & (uint16_t)k_ra8_usb_pipemaxp_mxps);
445 reg->PIPESEL = 0U;
446 if (mps == 0U) {
447 return k_ra8_err_invalid_state;
448 }
449 const uint16_t pipe_bit = (uint16_t)(1U << pipe_num);
450 /* Do NOT clear a pending BRDY here: at high speed the device can push
451 * the next packet (e.g. the CSW right behind a full data packet) into
452 * the pipe buffer while the previous transfer is winding down, and its
453 * BRDY is already latched -- clearing it would strand that packet in
454 * the buffer (observed as BSTS=1 with the wait timing out). */
455 reg->BRDYENB = (uint16_t)(reg->BRDYENB | pipe_bit);
456 internal_pipe_pid(reg, pipe_num, k_ra8_pid_buf);
457 uint16_t rx = 0U;
458 const ra8_err_t err = internal_host_bulk_rx_loop(reg, pipe_num, buf, max_len, mps, &rx);
459 internal_pipe_pid(reg, pipe_num, k_ra8_pid_nak);
460 *out_received = rx;
461 return err;
462}
