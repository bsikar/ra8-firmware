284static inline volatile r_dmac_channel_regs_t* ra8_dmac(uint8_t channel)
285{
286 if (channel >= k_ra8_dmac_channel_count) {
287 return nullptr;
288 }
289 return (
290 volatile r_dmac_channel_regs_t*)(k_ra8_dmac0_base_addr +
291 ((uintptr_t)channel * (uintptr_t)k_ra8_dmac_channel_stride));
292}
