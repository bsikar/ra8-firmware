98static ra8_err_t
99priv_read_one_chunk(ra8_fs_file_t* file, uint8_t* buf, uint32_t remaining, uint32_t* out_take)
100{
101 const uint32_t cluster_bytes = file->mount->sectors_per_cluster * k_ra8_fs_bytes_per_sector;
102 const uint32_t cluster_idx_now = file->offset / cluster_bytes;
103 uint32_t target = 0;
104 /* exFAT contiguous files (NoFatChain) have no valid FAT chain: clusters are
105 * sequential from the first. FAT files (no_fat_chain == 0) walk the chain. */
106 if (file->no_fat_chain != 0U) {
107 target = file->first_cluster + cluster_idx_now;
108 } else {
109 /* Read accelerator: a sequential read advances the offset one cluster at a
110 * time, so resume the FAT walk from the cached forward waypoint instead of
111 * re-walking from the chain head on every sector -- O(n) over a file, not the
112 * O(n^2) a per-sector walk-from-head would cost. The cache holds a valid chain
113 * position (walk_cache_cluster at walk_cache_idx); use it only when it is set
114 * (a real data cluster) AND the target is at or ahead of it, else walk from
115 * the head.
116 *
117 * @par MC/DC:
118 * Decision: `(walk_cache_cluster >= k_cluster_first_data) && (cluster_idx_now
119 * >= walk_cache_idx)` (2 conditions)
120 * - cache set, target ahead -> true (resume from waypoint)
121 * - cache unset (cluster < 2) -> false (first condition independent)
122 * - cache set, target behind -> false (second condition independent; a
123 * backward seek re-walks from the head)
124 */
125 uint32_t walk_from = file->first_cluster;
126 uint32_t walk_n = cluster_idx_now;
127 if ((file->walk_cache_cluster >= (uint32_t)k_cluster_first_data) &&
128 (cluster_idx_now >= file->walk_cache_idx)) {
129 walk_from = file->walk_cache_cluster;
130 walk_n = cluster_idx_now - file->walk_cache_idx;
131 }
132 ra8_err_t werr = priv_skip_clusters(file->mount, walk_from, walk_n, &target);
133 if (werr != k_ra8_ok) {
134 return werr;
135 }
136 file->walk_cache_idx = cluster_idx_now;
137 file->walk_cache_cluster = target;
138 }
139 file->cur_cluster = target;
140 const uint32_t off_in_cluster = file->offset % cluster_bytes;
141 const uint32_t sector_in_cluster = off_in_cluster / k_ra8_fs_bytes_per_sector;
142 const uint32_t off_in_sector = off_in_cluster % k_ra8_fs_bytes_per_sector;
143 const uint32_t lba = priv_cluster_to_lba(file->mount, file->cur_cluster) + sector_in_cluster;
144 uint8_t sec[k_ra8_fs_bytes_per_sector] = {};
145 const ra8_err_t err = priv_read_sector(file->mount, lba, sec);
146 if (err != k_ra8_ok) {
147 return err;
148 }
149 uint32_t take = k_ra8_fs_bytes_per_sector - off_in_sector;
150 if (take > remaining) {
151 take = remaining;
152 }
153 priv_byte_copy(buf, &sec[off_in_sector], take);
154 *out_take = take;
155 return k_ra8_ok;
156}

185ra8_err_t ra8_fs_read(ra8_fs_file_t* file, uint8_t* buf, uint32_t max_len, uint32_t* got_len)
186{
187 if (file == nullptr || buf == nullptr || got_len == nullptr) {
188 return k_ra8_err_null_ptr;
189 }
190 if (file->in_use == 0U) {
191 return k_ra8_err_invalid_state;
192 }
193 *got_len = 0;
194 if (file->offset >= file->size_bytes || max_len == 0U) {
195 return k_ra8_ok;
196 }
197 uint32_t remaining = file->size_bytes - file->offset;
198 if (remaining > max_len) {
199 remaining = max_len;
200 }
201 uint32_t produced = 0;
202 while (remaining > 0U) {
203 uint32_t take = 0;
204 ra8_err_t err = priv_read_one_chunk(file, &buf[produced], remaining, &take);
205 if (err != k_ra8_ok) {
206 return err;
207 }
208 produced += take;
209 file->offset += take;
210 remaining -= take;
211 }
212 *got_len = produced;
213 return k_ra8_ok;
214}

252priv_walk_grow(const ra8_fs_mount_t* m, uint32_t start, uint32_t idx, uint32_t* out_cluster)
253{
254 uint32_t cur = start;
255 for (uint32_t i = 0; i < idx; i++) {
256 uint32_t next = 0;
257 ra8_err_t err = priv_fat_get(m, cur, &next);
258 if (err != k_ra8_ok) {
259 return err;
260 }
261 if (priv_is_eoc(m, next) != 0U) {
262 uint32_t newc = 0;
263 err = priv_alloc_eoc_cluster(m, &newc);
264 if (err != k_ra8_ok) {
265 return err;
266 }
267 err = priv_fat_set(m, cur, newc);
268 if (err != k_ra8_ok) {
269 return err;
270 }
271 next = newc;
272 }
273 cur = next;
274 }
275 *out_cluster = cur;
276 return k_ra8_ok;
277}

345RA8_INTERNAL
346static ra8_err_t priv_write_stream(ra8_fs_file_t* file, const uint8_t* buf, uint32_t len)
347{
348 ra8_fs_mount_t* m = file->mount;
349 const uint32_t cluster_bytes = m->sectors_per_cluster * k_ra8_fs_bytes_per_sector;
350 uint32_t consumed = 0;
351 /* A write may allocate/grow the chain, invalidating any cached read waypoint. */
352 file->walk_cache_cluster = 0;
353 while (consumed < len) {
354 if (file->first_cluster < k_cluster_first_data) {
355 uint32_t c = 0;
356 ra8_err_t err = priv_alloc_eoc_cluster(m, &c);
357 if (err != k_ra8_ok) {
358 return err;
359 }
360 file->first_cluster = c;
361 file->cur_cluster = c;
362 }
363 const uint32_t cluster_idx_now = file->offset / cluster_bytes;
364 uint32_t cur = 0;
365 ra8_err_t err = priv_walk_grow(m, file->first_cluster, cluster_idx_now, &cur);
366 if (err != k_ra8_ok) {
367 return err;
368 }
369 file->cur_cluster = cur;
370 const uint32_t off_in_cluster = file->offset % cluster_bytes;
371 const uint32_t sector_in_cluster = off_in_cluster / k_ra8_fs_bytes_per_sector;
372 const uint32_t off_in_sector = off_in_cluster % k_ra8_fs_bytes_per_sector;
373 const uint32_t lba = priv_cluster_to_lba(m, file->cur_cluster) + sector_in_cluster;
374 uint32_t put = k_ra8_fs_bytes_per_sector - off_in_sector;
375 if (put > (len - consumed)) {
376 put = len - consumed;
377 }
378 err = priv_write_into_sector(m, lba, off_in_sector, &buf[consumed], put);
379 if (err != k_ra8_ok) {
380 return err;
381 }
382 consumed += put;
383 file->offset += put;
384 if (file->offset > file->size_bytes) {
385 file->size_bytes = file->offset;
386 }
387 }
388 return k_ra8_ok;
389}

416ra8_err_t ra8_fs_write(ra8_fs_file_t* file, const uint8_t* buf, uint32_t len)
417{
418 if (file == nullptr || buf == nullptr) {
419 return k_ra8_err_null_ptr;
420 }
421 if (file->in_use == 0U || file->mode == k_ra8_fs_mode_read) {
422 return k_ra8_err_invalid_state;
423 }
424 if (len == 0U) {
425 return k_ra8_ok;
426 }
427 ra8_err_t err = priv_write_stream(file, buf, len);
428 if (err != k_ra8_ok) {
429 return err;
430 }
431 ra8_fs_mount_t* m = file->mount;
432 uint8_t dirsec[k_ra8_fs_bytes_per_sector] = {};
433 err = priv_read_sector(m, file->dir_entry_lba, dirsec);
434 if (err != k_ra8_ok) {
435 return err;
436 }
437 priv_entry_set_cluster_size(&dirsec[file->dir_entry_idx], file->first_cluster, file->size_bytes);
438 return priv_write_sector(m, file->dir_entry_lba, dirsec);
439}
