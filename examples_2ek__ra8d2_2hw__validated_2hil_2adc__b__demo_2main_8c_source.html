73static uint32_t adc_b_demo_u16_to_dec(uint8_t* buf, uint16_t val)
74{
75 if (val == 0U) {
76 buf[0] = (uint8_t)'0';
77 return 1U;
78 }
79 uint8_t tmp[k_adc_b_demo_dec_digits];
80 uint32_t n = 0U;
81 uint16_t v = val;
82 while (v != 0U) {
83 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint16_t)k_adc_b_demo_radix));
84 v = (uint16_t)(v / (uint16_t)k_adc_b_demo_radix);
85 n++;
86 }
87 for (uint32_t i = 0U; i < n; i++) {
88 buf[i] = tmp[n - 1U - i];
89 }
90 return n;
91}

141int32_t main(void)
142{
143 adc_b_demo_setup_or_halt();
144 ra8_isr_globals_enable();
145
146 if (adc_b_demo_arm() != k_ra8_ok) {
147 adc_b_demo_panic_halt();
148 }
149
150 while (1) {
151 uint16_t raw = 0U;
152 if (ra8_adc_read_channel((uint8_t)k_adc_b_demo_channel, &raw) != k_ra8_ok) {
153 break;
154 }
155 const uint16_t mv = (uint16_t)(((uint32_t)raw * (uint32_t)k_adc_b_demo_vref_mv) /
156 (uint32_t)k_adc_b_demo_full_scale);
157
158 uint8_t line[k_adc_b_demo_line_max];
159 uint32_t pos = 0U;
160 for (uint32_t i = 0U; i < (sizeof(k_adc_b_demo_log_prefix) - 1U); i++) {
161 line[pos++] = k_adc_b_demo_log_prefix[i];
162 }
163 pos += adc_b_demo_u16_to_dec(&line[pos], raw);
164 for (uint32_t i = 0U; i < (sizeof(k_adc_b_demo_mv_sep) - 1U); i++) {
165 line[pos++] = k_adc_b_demo_mv_sep[i];
166 }
167 pos += adc_b_demo_u16_to_dec(&line[pos], mv);
168 line[pos++] = (uint8_t)k_adc_b_demo_cr;
169 line[pos++] = (uint8_t)k_adc_b_demo_lf;
170
171 (void)ra8_board_uart_console_write(line, (size_t)pos);
172 (void)ra8_board_uart_console_write(k_adc_b_demo_pass_msg,
173 (size_t)(sizeof(k_adc_b_demo_pass_msg) - 1U));
174 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
175 break;
176 }
177 ra8_delay_ms((uint32_t)k_adc_b_demo_period_ms);
178 }
179 adc_b_demo_panic_halt();
180 return 0;
181}
