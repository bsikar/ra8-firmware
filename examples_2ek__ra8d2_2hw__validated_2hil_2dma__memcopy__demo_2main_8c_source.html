172[[nodiscard]] static ra8_err_t dma_demo_run_copy(void)
173{
174 const ra8_dmac_config_t cfg = {
175 .src = (uint32_t)(uintptr_t)s_src,
176 .dst = (uint32_t)(uintptr_t)s_dst,
177 .count = (uint16_t)k_dma_demo_buf_words,
178 .block_count = 1U,
179 .width = k_ra8_dmac_width_word,
180 .src_inc = true,
181 .dst_inc = true,
182 };
183 const ra8_err_t err = ra8_dmac_start_block((uint8_t)k_dma_demo_channel, &cfg);
184 if (err != k_ra8_ok) {
185 return err;
186 }
187
188 /* Software trigger: assert DMREQ.SWREQ on the channel.
189 * HUM Ch 17.2.15 p 744: write SWREQ=1 to request one block-mode
190 * transfer. SWREQ auto-clears when the controller accepts the
191 * request (CLRS=0 is the reset default). */
192 volatile r_dmac_channel_regs_t* reg = ra8_dmac((uint8_t)k_dma_demo_channel);
193 if (reg == nullptr) {
194 return k_ra8_err_hw_error;
195 }
196 reg->DMREQ = (uint8_t)k_ra8_dmreq_swreq_mask;
197
198 /* Poll DMSTS.ACT until the controller goes idle.
199 * HUM Ch 17.2.16 p 745: "When data transfer in response to one
200 * transfer request is completed" the ACT flag clears. */
201 for (uint32_t i = 0U; i < (uint32_t)k_dma_demo_poll_limit; ++i) {
202 if ((reg->DMSTS & (uint8_t)k_ra8_dmsts_act_mask) == 0U) {
203 return ra8_dmac_stop((uint8_t)k_dma_demo_channel);
204 }
205 }
206 (void)ra8_dmac_stop((uint8_t)k_dma_demo_channel);
207 return k_ra8_err_hw_timeout;
208}

212int32_t main(void)
213{
214 dma_demo_setup_or_halt();
215 ra8_isr_globals_enable();
216
217 while (1) {
218 dma_demo_fill_buffers();
219 const ra8_err_t err = dma_demo_run_copy();
220 const uint8_t ok = (err == k_ra8_ok && dma_demo_verify() != 0U) ? 1U : 0U;
221 if (ok != 0U) {
222 (void)ra8_board_uart_console_write(k_dma_demo_ok_msg,
223 (size_t)(sizeof(k_dma_demo_ok_msg) - 1U));
224 (void)ra8_board_led_toggle(k_ra8_board_led1);
225 } else {
226 (void)ra8_board_uart_console_write(k_dma_demo_bad_msg,
227 (size_t)(sizeof(k_dma_demo_bad_msg) - 1U));
228 (void)ra8_board_led_toggle(k_ra8_board_led2);
229 }
230 ra8_delay_ms(k_dma_demo_period_ms);
231 }
232 dma_demo_panic_halt();
233 return 0;
234}
