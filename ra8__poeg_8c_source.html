82static uint32_t internal_cfg_to_poegg(const ra8_poeg_cfg_t* cfg)
83{
84 uint32_t v = 0U;
85 if (cfg->enable_pin) {
86 v |= k_ra8_poeg_en_pide;
87 }
88 if (cfg->enable_ioc) {
89 v |= k_ra8_poeg_en_iocen;
90 }
91 if (cfg->enable_osc_stop) {
92 v |= k_ra8_poeg_en_osten;
93 }
94 if (cfg->invert_input) {
95 v |= k_ra8_poeg_en_inv;
96 }
97 return v;
98}

126ra8_err_t ra8_poeg_init(uint8_t group, const ra8_poeg_cfg_t* cfg)
127{
128 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
129 volatile r_poeg_regs_t* reg = ra8_poeg(group);
130 RA8_CHECK_NULL_PTR(reg, s_tag, "group out of range");
131
132 /* HUM Ch 11.2.9 "MSTPCRD : Module Stop Control Register D", p 448 */
133 const ra8_err_t mst_err = ra8_mstp_enable(s_poeg_mstp_table[group]);
134 RA8_RETURN_ON_ERROR(mst_err, s_tag, "poeg_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
135
136 /* HUM Ch 21.2.1 "POEGG : POEG Group n Setting Register", p 872 */
137 reg->POEGG = internal_cfg_to_poegg(cfg);
138 ra8_log_info_val(s_tag, "init group", (uint32_t)group);
139 return k_ra8_ok;
140}

397RA8_ISR_SAFE
398void ra8_poeg_dispatch(uint8_t group)
399{
400 if (group >= (uint8_t)k_ra8_poeg_group_count) {
401 return;
402 }
403 volatile r_poeg_regs_t* reg = ra8_poeg(group);
404 uint32_t mask = 0U;
405 if (reg != nullptr) {
406 mask = reg->POEGG & k_ra8_poeg_status_all;
407 }
408 const ra8_poeg_event_fn_t fn = s_poeg_state[group].fn;
409 void* const ctx = s_poeg_state[group].ctx;
410 if (fn != nullptr) {
411 fn(ctx, mask);
412 }
413}
