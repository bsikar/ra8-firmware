158static ra8_err_t internal_wait_octacksrdy(uint8_t expected)
159{
160 /* SRDY (clock-source ready) is bit 7 of OCTACKCR. */
161 /* HUM Ch 9.2.45 "OCTACKCR.OCTACKSRDY" p 360 */
162 volatile uint8_t* const ckcr = ra8_sys_octackcr();
163 const uint8_t mask = (uint8_t)(1U << k_ra8_usbckcr_bit_srdy);
164 /* Bounded wait through ra8_hw_err.h: on host tests the ra8_fake_mmio
165 * seam decides the poll (first-poll success unless a test arms a
166 * fault), so the real timeout leg is reachable everywhere. */
167 if (expected != 0U) {
168 return ra8_hw_wait_flag_set8(ckcr, mask, (uint32_t)k_ra8_xspi_ckcr_spin);
169 }
170 return ra8_hw_wait_flag_clear8(ckcr, mask, (uint32_t)k_ra8_xspi_ckcr_spin);
171}

224RA8_INTERNAL
225static ra8_err_t internal_xspi_clock_block_init(void)
226{
227 static bool s_xspi_clock_inited = false;
228 if (s_xspi_clock_inited) {
229 return k_ra8_ok;
230 }
231 ra8_err_t err = k_ra8_ok;
232 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
233 {
234 /* HUM Ch 9.2.40 "OCTACKDIVCR" p 357 -- /1 keeps MOCO at its
235 * native rate (~8 MHz nominal). The IS25LX512M JEDEC bring-up
236 * runs comfortably below 50 MHz so /1 is conservative. */
237 *ra8_sys_octackdivcr() = 0U;
238
239 /* HUM Ch 9.2.45 "OCTACKCR.OCTACKSREQ" p 360 -- assert SREQ with
240 * the reset-default source (MOCO, OCTACKSEL = 0001b). */
241 const uint8_t sreq_mask = (uint8_t)(1U << k_ra8_usbckcr_bit_sreq);
242 const uint8_t src_moco = 0x01U;
243 *ra8_sys_octackcr() = (uint8_t)(src_moco | sreq_mask);
244
245 /* Step 3: wait for SRDY = 1 (chip acknowledges the request). */
246 err = internal_wait_octacksrdy(1U);
247 if (err != k_ra8_ok) {
248 ra8_log_error(s_tag, "xspi: OCTACKSRDY=1 timeout");
249 break;
250 }
251 /* Step 4: drop SREQ -- commits the (same) source selection. */
252 *ra8_sys_octackcr() = src_moco;
253 /* Step 5: wait for SRDY = 0 -- handshake done. */
254 err = internal_wait_octacksrdy(0U);
255 if (err != k_ra8_ok) {
256 ra8_log_error(s_tag, "xspi: OCTACKSRDY=0 timeout");
257 break;
258 }
259 }
260 if (err == k_ra8_ok) {
261 s_xspi_clock_inited = true;
262 ra8_log_info(s_tag, "octa block clock stable");
263 }
264 return err;
265}

373{
374 volatile r_xspi_regs_t* reg = ra8_xspi(instance);
375 RA8_CHECK_NULL_PTR(reg, s_tag, "instance out of range");
376 if (instance >= (uint8_t)(sizeof(s_xspi_mstp_table) / sizeof(s_xspi_mstp_table[0]))) {
377 return k_ra8_err_invalid_arg;
378 }

380 /* HUM Ch 11.2.7 "MSTPCRB" Note 3 p 444 -- MSTPB16/17 must be written
381 * AFTER OCTACLK is stable. Run the block-level OCTACKCR handshake
382 * first; the helper is idempotent so a second ra8_xspi_init call
383 * (e.g. for instance 1) skips it. */
384 const ra8_err_t clk_err = internal_xspi_clock_block_init();
385 if (clk_err != k_ra8_ok) {

391 RA8_RETURN_ON_ERROR(mst_err, s_tag, "xspi_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
392
393 internal_xspi_apply_config(reg, mode);
394 internal_xspi_reset_device(reg);
395
396 ra8_log_info_val(s_tag, "xspi_init inst", (uint32_t)instance);
397 return k_ra8_ok;
398}

400ra8_err_t ra8_xspi_direct_command(uint8_t instance, const uint8_t* cmd_buf, uint8_t len)
401{
402 RA8_CHECK_NULL_PTR(cmd_buf, s_tag, "cmd_buf must not be nullptr");
403 if (len > k_ra8_xspi_cmd_max_bytes) {
404 return k_ra8_err_invalid_size;
405 }
406 volatile r_xspi_regs_t* reg = ra8_xspi(instance);
407 if (reg == nullptr) {
408 return k_ra8_err_out_of_range;
409 }
410
411 /* HUM Ch 44 "Octal Serial Peripheral Interface (OSPI)" p 2986 */
412 /* Pack the caller's bytes into CDBUF slot 0 as little-endian
413 * words. Slot 0 is the full 16-byte CDBUF window (4 u32), so
414 * raw callers can use any of the four indices. */
415 uint32_t word = 0U;
416 for (uint8_t i = 0U; i < len; i++) {
417 word |= ((uint32_t)cmd_buf[i]) << ((i % 4U) * 8U);
418 if ((i % 4U) == 3U) {
419 reg->CDBUF[i / 4U] = word;
420 word = 0U;
421 }
422 }
423 if ((len % 4U) != 0U) {
424 reg->CDBUF[len / 4U] = word;
425 }
426
427 return k_ra8_ok;
428}

531ra8_err_t ra8_xspi_xip_enter(uint8_t instance, uint8_t enter_code, uint8_t exit_code)
532{
533 volatile r_xspi_regs_t* reg = ra8_xspi(instance);
534 RA8_CHECK_NULL_PTR(reg, s_tag, "instance out of range");
535
536 /* HUM Ch 44 "Octal Serial Peripheral Interface (OSPI)" p 2986 */
537 /* FSP r_ospi_b_xip(true) flow:
538 * 1. Stage XIP enter/exit codes in CMCTLCH for both channels.
539 * 2. Map the target window read-only via BMCTL0 = 0x55.
540 * 3. Set CMCTLCH.XIPEN to arm execute-in-place.
541 * The first read on the memory-mapped window then transmits the
542 * enter code. We omit the bus-bridge prefetch dance because the
543 * driver does not enable prefetch by default. */
544 const uint32_t code_word = ((uint32_t)enter_code << k_ra8_xspi_cmctlch_xipencode_pos) |
545 ((uint32_t)exit_code << k_ra8_xspi_cmctlch_xipexcode_pos);
546
547 reg->BMCTL0 = k_ra8_xspi_bmctl0_read_only;
548 reg->CMCTLCH[0] = code_word | k_ra8_xspi_cmctlch_xipen_mask;
549 reg->CMCTLCH[1] = code_word | k_ra8_xspi_cmctlch_xipen_mask;
550 return k_ra8_ok;
551}

615ra8_err_t ra8_xspi_set_xip_mode(uint8_t instance, bool enable, uint8_t read_cmd, uint8_t addr_bytes)
616{
617 volatile r_xspi_regs_t* reg = ra8_xspi(instance);
618 RA8_CHECK_NULL_PTR(reg, s_tag, "instance out of range");
619 if ((addr_bytes != k_ra8_xspi_addr_bytes_3) && (addr_bytes != k_ra8_xspi_addr_bytes_4)) {
620 return k_ra8_err_invalid_arg;
621 }
622
623 /* HUM Ch 44 "Octal Serial Peripheral Interface (OSPI)" p 2986 */
624 /* CMCFGCS slot 0 = (mode, read-cmd-word, write-cmd-word, addr-word).
625 * For XIP we only need read + addr; write opcode stays zero. */
626 const uint8_t base = 0U; /* slot 0 base index */
627 reg->CMCFGCS[base + k_ra8_xspi_cmcfgcs_word_read_cmd] = (uint32_t)read_cmd
628 << k_ra8_xspi_cmcfgcs_pos_cmd;
629 reg->CMCFGCS[base + k_ra8_xspi_cmcfgcs_word_addr] = (uint32_t)addr_bytes
630 << k_ra8_xspi_cmcfgcs_pos_addr_size;
631
632 if (enable) {
633 /* Mirror FSP r_ospi_b_xip(true): map read-only and arm XIPEN. */
634 reg->BMCTL0 = k_ra8_xspi_bmctl0_read_only;
635 reg->CMCTLCH[0] = k_ra8_xspi_cmctlch_xipen_mask;
636 reg->CMCTLCH[1] = k_ra8_xspi_cmctlch_xipen_mask;
637 } else {
638 /* FSP r_ospi_b_xip(false): clear XIPEN and re-open the bus. */
639 reg->CMCTLCH[0] = 0U;
640 reg->CMCTLCH[1] = 0U;
641 reg->BMCTL0 = k_ra8_xspi_bmctl0_read_write;
642 }
643 return k_ra8_ok;
644}

725RA8_INTERNAL
726static ra8_err_t
727internal_issue_reset_opcode(volatile r_xspi_regs_t* reg, uint8_t opcode, uint8_t cmd_bytes)
728{
729 /* HUM Ch 44 "Octal Serial Peripheral Interface (OSPI)" p 2986 */
730 /* Build the CMD half-word for either 1S (just the opcode) or 8D
731 * (opcode + complement). The complement form is what 8D-mode SPI
732 * NOR devices require so they can distinguish "real opcode" from
733 * "garbage on the bus". */
734 /* CMD is transmitted MSB-first from CDT[31:16]; a 1-byte opcode must
735 * be left-justified to CDT[31:24] (cmd_word << 8), a 2-byte 8D pair
736 * fills the full [31:16] field. Same rule as internal_make_cdt. */
737 uint16_t cmd_word = (uint16_t)(opcode << 8U);
738 if (cmd_bytes == (uint8_t)k_ra8_xspi_reset_cmd_bytes_8d) {
739 cmd_word = (uint16_t)(opcode | (((uint16_t)(uint8_t)~opcode) << 8U));
740 }
741 reg->CDBUF[k_ra8_xspi_cdbuf_idx_cdt] =
742 (((uint32_t)cmd_bytes & k_ra8_xspi_cdt_mask_cmdsize) << k_ra8_xspi_cdt_pos_cmdsize) |
743 (((uint32_t)k_ra8_xspi_cdt_addsize_0 & k_ra8_xspi_cdt_mask_addsize)
744 << k_ra8_xspi_cdt_pos_addsize) |
745 (((uint32_t)k_ra8_xspi_cdt_trtype_write & k_ra8_xspi_cdt_mask_trtype)
746 << k_ra8_xspi_cdt_pos_trtype) |
747 (((uint32_t)cmd_word) << k_ra8_xspi_cdt_pos_cmd);
748 reg->CDBUF[k_ra8_xspi_cdbuf_idx_addr] = 0U;
749 reg->CDBUF[k_ra8_xspi_cdbuf_idx_data0] = 0U;
750 reg->CDBUF[k_ra8_xspi_cdbuf_idx_data1] = 0U;
751 return ra8_xspi_kick_command(reg);
752}
