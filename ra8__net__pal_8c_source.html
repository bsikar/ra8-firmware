259ra8_err_t ra8_net_pal_init(const ra8_net_pal_mac_t* mac)
260{
261 const ra8_err_t eth_err = ra8_eth_init();
262 if (eth_err != k_ra8_ok) {
263 ra8_log_error_val(s_tag, "ra8_eth_init failed", (uint32_t)eth_err);
264 return k_ra8_err_hw_init_failed;
265 }
266
267 internal_zero_bytes(s_state.mac.bytes, k_ra8_net_pal_mac_addr_len);
268 s_state.link_state = k_ra8_net_pal_link_down;
269 s_state.event_fn = nullptr;
270 s_state.event_ctx = nullptr;
271 s_state.initialized = true;
272 internal_ring_reset();
273
274 if (mac != nullptr) {
275 internal_copy_bytes(s_state.mac.bytes, mac->bytes, k_ra8_net_pal_mac_addr_len);
276 }
277
278 const ra8_err_t att_err = ra8_eth_attach_handler(internal_eth_event, nullptr);
279 if (att_err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
280 /* GCOVR_EXCL_START */
281 ra8_log_error_val(s_tag, "ra8_eth_attach_handler failed", (uint32_t)att_err);
282 s_state.initialized = false;
283 (void)ra8_eth_deinit();
284 return k_ra8_err_hw_init_failed;
285 /* GCOVR_EXCL_STOP */
286 }
287
288 ra8_log_info(s_tag, "PAL ready");
289 return k_ra8_ok;
290}

420ra8_err_t ra8_net_pal_send_frame(const uint8_t* frame, uint16_t len)
421{
422 RA8_CHECK_NULL_PTR(frame, s_tag, "send_frame: frame");
423 if (!s_state.initialized) {
424 return k_ra8_err_invalid_state;
425 }
426 if ((len == 0U) || (len > k_ra8_net_pal_frame_max)) {
427 return k_ra8_err_invalid_arg;
428 }
429 if (s_state.count >= k_ra8_net_pal_ring_slots) {
430 return k_ra8_err_no_mem;
431 }
432 ra8_net_pal_slot_t* slot = &s_state.ring[s_state.tail];
433 internal_copy_bytes(slot->data, frame, len);
434 slot->len = len;
435 s_state.tail = (uint16_t)((s_state.tail + 1U) % k_ra8_net_pal_ring_slots);
436 ++s_state.count;
437 if ((s_state.event_fn != nullptr)) {
438 s_state.event_fn(s_state.event_ctx, k_ra8_net_pal_event_tx_done);
439 }
440 return k_ra8_ok;
441}

473ra8_err_t ra8_net_pal_recv_frame(uint8_t* out_buf, uint16_t* inout_len)
474{
475 RA8_CHECK_NULL_PTR(out_buf, s_tag, "recv_frame: out_buf");
476 RA8_CHECK_NULL_PTR(inout_len, s_tag, "recv_frame: inout_len");
477 if (!s_state.initialized) {
478 return k_ra8_err_invalid_state;
479 }
480 if (*inout_len < k_ra8_net_pal_frame_max) {
481 return k_ra8_err_invalid_arg;
482 }
483 if (s_state.count == 0U) {
484 return k_ra8_err_no_data;
485 }
486 ra8_net_pal_slot_t* slot = &s_state.ring[s_state.head];
487 const uint16_t n = slot->len;
488 internal_copy_bytes(out_buf, slot->data, n);
489 *inout_len = n;
490 slot->len = 0U;
491 s_state.head = (uint16_t)((s_state.head + 1U) % k_ra8_net_pal_ring_slots);
492 --s_state.count;
493 return k_ra8_ok;
494}
