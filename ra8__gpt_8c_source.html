334 * must be programmed BEFORE MSTPCR is cleared, or per-channel

336 internal_gpt_clock_block_init();
337 /* HUM Ch 11.2.10 "MSTPCRE : Module Stop Control Register E", p 449 */
338 const ra8_err_t mst_err = ra8_mstp_enable(s_gpt_mstp_table[channel]);

354 reg->GTSTR = k_ra8_gpt_gtstr_start;
355 }
356 reg->GTWP = k_ra8_gtwp_key_lock;
357
358 s_gpt_state[channel].configured = true;
359 ra8_log_info_val(s_tag, "init channel", (uint32_t)channel);

684 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
685 if (pin > k_ra8_gpt_pin_b) {

689 RA8_CHECK_NULL_PTR(reg, s_tag, "channel out of range");

692 reg->GTWP = k_ra8_gtwp_key_unlock;

694
695 if (pin == k_ra8_gpt_pin_a) {
696 /* Clear and re-pack the A-side fields only -- preserve B-side. */
697 gtior &=
698 ~(k_ra8_gpt_gtioa_mask | k_ra8_gpt_oadflt_mask | k_ra8_gpt_oae_mask | k_ra8_gpt_oadf_mask);
699 gtior |= (pattern & k_ra8_gpt_gtioa_mask) << k_ra8_gpt_gtioa_shift;
700 gtior |= ((uint32_t)cfg->stop_level << k_ra8_gpt_oadflt_shift) & k_ra8_gpt_oadflt_mask;
701 if (cfg->output_enable) {
702 gtior |= k_ra8_gpt_oae_mask;
703 }
704 gtior |= ((uint32_t)cfg->disable_on_fault << k_ra8_gpt_oadf_shift) & k_ra8_gpt_oadf_mask;
705 } else {
706 gtior &=
707 ~(k_ra8_gpt_gtiob_mask | k_ra8_gpt_obdflt_mask | k_ra8_gpt_obe_mask | k_ra8_gpt_obdf_mask);
708 gtior |= (pattern & (k_ra8_gpt_gtiob_mask >> k_ra8_gpt_gtiob_shift)) << k_ra8_gpt_gtiob_shift;
709 gtior |= ((uint32_t)cfg->stop_level << k_ra8_gpt_obdflt_shift) & k_ra8_gpt_obdflt_mask;
710 if (cfg->output_enable) {
711 gtior |= k_ra8_gpt_obe_mask;
712 }
713 gtior |= ((uint32_t)cfg->disable_on_fault << k_ra8_gpt_obdf_shift) & k_ra8_gpt_obdf_mask;
714 }
715 reg->GTIOR = gtior;
716 reg->GTWP = k_ra8_gtwp_key_lock;
717 return k_ra8_ok;
718}

744typedef struct {

750static ra8_gpt_three_phase_state_t s_three_phase;
751
771static ra8_err_t internal_three_phase_init_subs(const ra8_gpt_three_phase_cfg_t* cfg,
772 uint32_t* out_mask)
773{
774 const uint32_t initial_duties[k_ra8_gpt_three_phase_count] = {
775 cfg->initial_duty_u,
776 cfg->initial_duty_v,
777 cfg->initial_duty_w,
778 };
779 uint32_t mask = 0U;
780 for (uint8_t i = 0U; i < k_ra8_gpt_three_phase_count; ++i) {
781 const ra8_gpt_cfg_t per_ch_cfg = {
782 .mode = cfg->mode,
783 .prescaler = cfg->prescaler,
784 .period = cfg->period_counts,
785 .duty_a = initial_duties[i],
786 .duty_b = initial_duties[i],
787 .auto_start = false,
788 };
789 const ra8_err_t err = ra8_gpt_init(cfg->channels[i], &per_ch_cfg);
790 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
791 for (uint8_t j = 0U; j < i; ++j) { /* GCOVR_EXCL_LINE */
792 (void)ra8_gpt_deinit(cfg->channels[j]); /* GCOVR_EXCL_LINE */
793 } /* GCOVR_EXCL_LINE */
794 return err; /* GCOVR_EXCL_LINE */
795 }
796 mask |= (uint32_t)(1UL << cfg->channels[i]);
797 s_three_phase.channels[i] = cfg->channels[i];
798 }
799 *out_mask = mask;
800 return k_ra8_ok;
801}
802
803ra8_err_t ra8_gpt_three_phase_open(const ra8_gpt_three_phase_cfg_t* cfg)
804{
805 RA8_CHECK_NULL_PTR(cfg, s_tag, "three_phase cfg must not be nullptr");
806 if (s_three_phase.open) {
807 return k_ra8_err_invalid_state;
808 }
809 for (uint8_t i = 0U; i < k_ra8_gpt_three_phase_count; ++i) {
810 if (cfg->channels[i] >= (uint8_t)k_ra8_gpt_channel_count) {
811 return k_ra8_err_invalid_arg;
812 }
813 }
814
815 uint32_t mask = 0U;
816 const ra8_err_t err = internal_three_phase_init_subs(cfg, &mask);
817 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
818 return err; /* GCOVR_EXCL_LINE */
819 }
820
821 /* Synchronous start: a single GTSTR write to the U-channel slot
822 with all three CSTRTn bits set kicks all three counters on the
823 same PCLKD edge. HUM Ch 22.2.2 "GTSTR" p 886. */
824 volatile r_gpt_channel_regs_t* u_reg = ra8_gpt(cfg->channels[0]);
825 if (u_reg != nullptr) {
826 u_reg->GTWP = k_ra8_gtwp_key_unlock;
827 u_reg->GTSTR = mask;
828 u_reg->GTWP = k_ra8_gtwp_key_lock;
829 }
830
831 s_three_phase.channel_mask = mask;
832 s_three_phase.open = true;
833 ra8_log_info_val(s_tag, "three_phase open mask", mask);
834 return k_ra8_ok;
835}
836
837ra8_err_t ra8_gpt_three_phase_set_duty(uint32_t u_duty, uint32_t v_duty, uint32_t w_duty)
838{
839 if (!s_three_phase.open) {
840 return k_ra8_err_invalid_state;
841 }
842 const uint32_t duties[k_ra8_gpt_three_phase_count] = {u_duty, v_duty, w_duty};
843
844 /* Range check against the shared GTPR (read once from U). */
845 volatile r_gpt_channel_regs_t* u_reg = ra8_gpt(s_three_phase.channels[0]);
846 if (u_reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
847 return k_ra8_err_invalid_state; /* GCOVR_EXCL_LINE */
848 }
849 const uint32_t period = u_reg->GTPR;
850 for (uint8_t i = 0U; i < k_ra8_gpt_three_phase_count; ++i) {
851 if (duties[i] > period) {
852 return k_ra8_err_invalid_arg;
853 }
854 }
855
856 for (uint8_t i = 0U; i < k_ra8_gpt_three_phase_count; ++i) {
857 volatile r_gpt_channel_regs_t* reg = ra8_gpt(s_three_phase.channels[i]);
858 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
859 return k_ra8_err_invalid_state; /* GCOVR_EXCL_LINE */
860 }
861 reg->GTWP = k_ra8_gtwp_key_unlock;
862 reg->GTCCR[k_ra8_gpt_ccr_idx_c] = duties[i];
863 reg->GTCCR[k_ra8_gpt_ccr_idx_e] = duties[i];
864 reg->GTBER |= k_ra8_gpt_gtber_ccra_single | k_ra8_gpt_gtber_ccrb_single;
865 reg->GTWP = k_ra8_gtwp_key_lock;
866 }
867 return k_ra8_ok;
868}
869
870ra8_err_t ra8_gpt_three_phase_close(void)
871{
872 if (!s_three_phase.open) {
873 return k_ra8_err_invalid_state;
874 }
875 /* Synchronous stop: single GTSTP write covers all three channels. */
876 volatile r_gpt_channel_regs_t* u_reg = ra8_gpt(s_three_phase.channels[0]);
877 if (u_reg != nullptr) {
878 u_reg->GTWP = k_ra8_gtwp_key_unlock;
879 u_reg->GTSTP = s_three_phase.channel_mask;
880 u_reg->GTWP = k_ra8_gtwp_key_lock;
881 }
882 for (uint8_t i = 0U; i < k_ra8_gpt_three_phase_count; ++i) {
883 (void)ra8_gpt_deinit(s_three_phase.channels[i]);
884 }
885 s_three_phase.channel_mask = 0U;
886 s_three_phase.open = false;
887 return k_ra8_ok;
888}
