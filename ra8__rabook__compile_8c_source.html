76RA8_INTERNAL
77static uint32_t rabook_crc32(const uint8_t* data, uint32_t len)
78{
79 uint32_t crc = (uint32_t)k_rabook_crc_init;
80 for (uint32_t i = 0U; i < len; i++) {
81 crc ^= (uint32_t)data[i];
82 for (uint32_t bit = 0U; bit < (uint32_t)k_rabook_crc_byte_bits; bit++) {
83 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
84 crc = (crc >> 1U) ^ ((uint32_t)k_rabook_crc_poly & mask);
85 }
86 }
87 return crc ^ (uint32_t)k_rabook_crc_init;
88}

158uint32_t ra8_rabook_intern(ra8_rabook_ctx_t* ctx, const char* str)
159{
160 if (ctx == nullptr) {
161 return (uint32_t)k_ra8_book_nil;
162 }
163 if (str == nullptr) {
164 ctx->failed = true;
165 return (uint32_t)k_ra8_book_nil;
166 }
167 if (ctx->failed) {
168 return (uint32_t)k_ra8_book_nil;
169 }
170
171 const uint32_t len = (uint32_t)strlen(str);
172 /* De-dup: walk the pool string-by-string, return the first exact match. */
173 uint32_t scan = 0U;
174 while (scan < ctx->string_size) {
175 const char* cand = &ctx->buf.string_pool[scan];
176 const uint32_t cand_len = (uint32_t)strlen(cand);
177 if (cand_len == len) {
178 if (memcmp(cand, str, (size_t)len) == 0) {
179 return scan;
180 }
181 }
182 scan += cand_len + 1U; /* skip the candidate and its NUL */
183 }
184
185 /* Miss: append str + NUL, guarding the pool capacity. */
186 const uint32_t need = len + 1U;
187 if (need > (ctx->buf.string_cap - ctx->string_size)) {
188 ctx->failed = true;
189 return (uint32_t)k_ra8_book_nil;
190 }
191 const uint32_t off = ctx->string_size;
192 (void)memcpy(&ctx->buf.string_pool[off], str, (size_t)len);
193 ctx->buf.string_pool[off + len] = '\0';
194 ctx->string_size += need;
195 return off;
196}

488static ra8_err_t s_compute_layout(const ra8_rabook_ctx_t* ctx, ra8_rabook_layout_t* lay)
489{
490 const uint64_t chap_bytes = (uint64_t)ctx->chapter_count * (uint64_t)k_ra8_book_sizeof_chapter;
491 const uint64_t node_bytes = (uint64_t)ctx->node_count * (uint64_t)k_ra8_book_sizeof_node;
492 const uint64_t attr_bytes = (uint64_t)ctx->attr_count * (uint64_t)k_ra8_book_sizeof_attr;
493 const uint64_t style_bytes =
494 (uint64_t)ctx->stylesheet_count * (uint64_t)k_ra8_book_sizeof_stylesheet;
495 const uint64_t image_bytes = (uint64_t)ctx->image_count * (uint64_t)k_ra8_book_sizeof_image;
496
497 const uint64_t off_chap = (uint64_t)k_ra8_book_sizeof_header;
498 const uint64_t off_node = off_chap + chap_bytes;
499 const uint64_t off_attr = off_node + node_bytes;
500 const uint64_t off_style = off_attr + attr_bytes;
501 const uint64_t off_image = off_style + style_bytes;
502 const uint64_t off_string = off_image + image_bytes;
503 const uint64_t off_pool = off_string + (uint64_t)ctx->string_size;
504 const uint64_t total = off_pool + (uint64_t)ctx->image_pool_size;
505
506 if (total > (uint64_t)UINT32_MAX) {
507 ra8_log_error(s_tag_rabook, "finalize: layout overflows 32-bit offsets");
508 return k_ra8_err_invalid_size;
509 }
510
511 lay->off_chap = (uint32_t)off_chap;
512 lay->off_node = (uint32_t)off_node;
513 lay->off_attr = (uint32_t)off_attr;
514 lay->off_style = (uint32_t)off_style;
515 lay->off_image = (uint32_t)off_image;
516 lay->off_string = (uint32_t)off_string;
517 lay->off_pool = (uint32_t)off_pool;
518 lay->total = (uint32_t)total;
519 return k_ra8_ok;
520}

583{
584 RA8_CHECK_NULL_PTR(ctx, s_tag_rabook, "ctx");
585 RA8_CHECK_NULL_PTR(out_blob, s_tag_rabook, "out_blob");
586 RA8_CHECK_NULL_PTR(out_len, s_tag_rabook, "out_len");

594 if (lay_err != k_ra8_ok) {

598 ra8_log_error(s_tag_rabook, "finalize: blob exceeds output capacity");

600 }
601
602 s_write_blob(ctx->buf.out, ctx, &lay);
603
604 *out_blob = ctx->buf.out;
605 *out_len = lay.total;
606 return k_ra8_ok;
