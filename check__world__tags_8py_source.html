4"""check_world_tags.py -- enforce TrustZone world tags on Ring 3+ files.
5
6Every file in Ring 3 (HAL) and above carries two tags in its file
7header:
8
9 [Ring N / NAME]
10 {World: S | NS | NSC}
11
12This script:
13
14 1. Walks every first-party C file, derived from git ls-files via
15 lint_targets (#358) rather than a hardcoded ("libs","src","tests")
16 + example-app list that silently omitted tools/ and port/. Vendored
17 trees (libs/third_party/, port/threadx/) are dropped automatically.
18 2. For Ring 3+ files (anything outside Ring 1 BSP and Ring 2 Core),
19 requires both a [Ring N / ...] tag and a {World: ...} tag in
20 the first ~80 lines of the file.
21 3. Verifies that any file carrying {World: NSC} lives under
22 libs/ra8_nsc/ -- NSC veneers may not be defined anywhere else.
23 4. Verifies that no file outside libs/ra8_nsc/ uses
24 __attribute__((cmse_nonsecure_entry)) -- the SG-instruction
25 compiler attribute is the only legal way to mark a function
26 as a Non-Secure entry point, and the project requires that
27 only happen in NSC veneers.
28
29Host tooling under tools/ and vendored trees are enumerated but are NOT
30ring3+, so they require no World tag -- a documented scope decision (see
31file_is_in_ring3_plus and _select_targets), not an accident of a tuple. The
32NSC-location and cmse_nonsecure_entry bans (checks 3 and 4) apply to every
33file, everywhere.
34
35Modes:
36
37 --warn (default) -- exit 0, print findings
38 --strict (onward) -- exit 1 on any finding
39 --selftest -- prove the bans fire and the scope decision holds, then exit
40
41A finite LEGACY_RING3_EXEMPT_PREFIXES list grandfathers the pre-tag-system
42libs/ra8_hal/ and tests/ files: each leaves the exemption automatically the
43moment it grows either tag, so the exemption only ever shrinks.
44"""

78def discover_app_dirs() -> tuple[str, ...]:
79 """Every examples/**/ dir at ANY depth holding both main.c and CMakeLists.txt.
80
81 Requiring BOTH is what distinguishes a real app directory from a shared
82 subdirectory that merely contains sources. Discovery recurses on that pair
83 rather than assuming a fixed examples/<tier>/<app> layout: an earlier
84 TWO-LEVEL walk reached only the handful of one-level apps and silently
85 skipped the entire deeply-nested ek_ra8d2 board tree -- the same
86 under-scoping defect as #358, and the reason nine ek_ra8d2 app main.c files
87 went unclassified and untagged. cite_check.py already discovers apps this
88 way.
89 """
90 examples_root = REPO_ROOT / "examples"
91 if not examples_root.is_dir():
92 return ()
93 app_dirs: set[str] = set()
94 for cmake in examples_root.rglob("CMakeLists.txt"):
95 app_dir = cmake.parent
96 if any(
97 part in ("third_party", "_deps") or part.startswith("build") for part in app_dir.parts
98 ):
99 continue
100 if (app_dir / "main.c").is_file():
101 app_dirs.add(app_dir.relative_to(REPO_ROOT).as_posix())
102 return tuple(sorted(app_dirs))
103
104
105APP_DIRS = discover_app_dirs()

107# Files that lived in the tree before the world-tag system was
108# introduced (baseline). They are exempt from world-tag
109# enforcement until the wave that retrofits them. As soon as a file
110# under one of these prefixes gains its [Ring N / NAME] +
111# {World: ...} tag pair, it leaves the exemption automatically:
112# the script enforces consistency on any file that already carries
113# at least one of the two tags.
114#
115# Practical effect: starts with 0 findings; + adds tags
116# incrementally and the script catches any mismatches.
117LEGACY_RING3_EXEMPT_PREFIXES = (
118 "libs/ra8_hal/",
119 "tests/",
120)
121
122# Header-window size for tag scanning. The tags must appear in the
123# first N lines of the file (just inside the file-level Doxygen
124# block).
125HEADER_LINE_WINDOW = 80

127RING_RE = re.compile(r"\‍[\s*Ring\s+(\d)\s*/\s*([A-Za-z_]+)\s*\‍]")
128WORLD_RE = re.compile(r"\{\s*World\s*:\s*(S|NS|NSC|MIXED)\s*\}")
129NSC_ENTRY_RE = re.compile(r"__attribute__\s*\‍(\s*\‍(\s*cmse_nonsecure_entry\s*\‍)\s*\‍)")

132def is_legacy_exempt(rel_path: str) -> bool:
133 """Whether a path predates the World-tag requirement and is grandfathered.
134
135 A prefix list, deliberately finite and not extended: it records what was
136 already in the tree when the rule landed. New code has no route into it,
137 so the exemption shrinks as those files are tagged and never grows.
138 """
139 return any(rel_path.startswith(p) for p in LEGACY_RING3_EXEMPT_PREFIXES)

142def file_is_in_ring1_or_ring2(rel_path: str) -> bool:
143 """Whether a file sits in Ring 1 (BSP) or Ring 2 (Core).
144
145 Both rings are Secure-only by definition, so a ``{World: ...}`` tag would
146 be restating the ring rather than adding information -- which is why they
147 are exempt from the requirement rather than required to say "S".
148 """
149 if rel_path.startswith("libs/ra8_core/"):
150 return True
151 # Per-app boot files (vector_table.c, system_init.c,
152 # secure_exception.c, trustzone_init.c/h) are Ring 1 by virtue of
153 # the role they play, even though they live next to main.c.
154 for app_dir in APP_DIRS:
155 prefix = app_dir + "/"
156 if rel_path.startswith(prefix):
157 tail = rel_path[len(prefix) :]
158 if tail in APP_BOOT_FILES:
159 return True
160 return bool(rel_path.endswith("/linker_script.ld"))

163def file_is_in_ring3_plus(rel_path: str) -> bool:
164 """Whether a file is project-owned Ring 3+ code, where the World tag is required.
165
166 Covers libs/ra8_hal/, libs/ra8_*_pal/, libs/ra8_nsc/, src/secure_app/,
167 tests/, and per-app main.c (Ring 6 application code). These are the rings
168 that can run in either TrustZone world, which is precisely why each file
169 must declare which one it is written for.
170 """
171 if rel_path.startswith("libs/ra8_hal/"):
172 return True
173 if rel_path.startswith("libs/ra8_") and "_pal/" in rel_path:
174 return True
175 if rel_path.startswith("libs/ra8_nsc/"):
176 return True
177 if rel_path.startswith("src/secure_app/"):
178 return True
179 if rel_path.startswith("tests/"):
180 return True
181 # Per-app main.c is Ring 6.
182 return any(rel_path == f"{app_dir}/main.c" for app_dir in APP_DIRS)

185def iter_source_files(targets: Iterable[pathlib.Path]) -> Iterable[pathlib.Path]:
186 """Expand a mixed list of files and directories into source files.
187
188 A path that does not exist is skipped silently rather than raising, so a
189 stale entry in a caller's list cannot abort the sweep.
190 """
191 for t in targets:
192 if not t.exists():
193 continue
194 if t.is_file():
195 if t.suffix.lower() in SOURCE_SUFFIXES:
196 yield t
197 continue
198 for sub in t.rglob("*"):
199 if not sub.is_file():
200 continue
201 if sub.suffix.lower() not in SOURCE_SUFFIXES:
202 continue
203 parts = set(sub.parts)
204 if parts & EXCLUDED_PATH_PARTS:
205 continue
206 yield sub

222def check_file(path: pathlib.Path) -> list[str]:
223 """Report a missing or malformed ``{World: ...}`` tag in one file.
224
225 Ring membership decides whether the tag is required at all, so the ring
226 tests run before the tag is looked for -- a Ring 2 file with no tag is
227 correct, not a finding.
228
229 Returns one message per finding; an empty list means the file is fine or
230 out of scope.
231 """
232 findings: list[str] = []
233 rel = _to_repo_relative(path)
234
235 try:
236 text = path.read_text(encoding="utf-8", errors="replace")
237 except OSError as exc:
238 return [f"{rel}: read error: {exc}"]
239
240 head_lines = text.splitlines()[:HEADER_LINE_WINDOW]
241 head = "\n".join(head_lines)
242
243 ring_match = RING_RE.search(head)
244 world_match = WORLD_RE.search(head)
245
246 if file_is_in_ring3_plus(rel):
247 # A legacy-exempt file is only exempt while it carries
248 # NEITHER tag. As soon as it grows one, both are enforced.
249 legacy = is_legacy_exempt(rel)
250 has_any_tag = ring_match is not None or world_match is not None
251 if (not legacy) or has_any_tag:
252 if ring_match is None:
253 findings.append(f"{rel}: missing [Ring N / NAME] tag in file header")
254 if world_match is None:
255 findings.append(f"{rel}: missing {{World: S|NS|NSC}} tag in file header")
256
257 # Check 3: NSC tag must live under libs/ra8_nsc/
258 if (
259 world_match is not None
260 and world_match.group(1) == "NSC"
261 and not rel.startswith("libs/ra8_nsc/")
262 ):
263 findings.append(f"{rel}: {{World: NSC}} tag is only legal under libs/ra8_nsc/")
264
265 # Check 4: cmse_nonsecure_entry only inside libs/ra8_nsc/
266 for m in NSC_ENTRY_RE.finditer(text):
267 if rel.startswith("libs/ra8_nsc/"):
268 continue
269 line_no = text.count("\n", 0, m.start()) + 1
270 findings.append(
271 f"{rel}:{line_no}: cmse_nonsecure_entry attribute outside libs/ra8_nsc/ "
272 f"-- NSC veneers must live under that tree"
273 )
274
275 # Check 5: Ring 1/2 files must NOT carry a {World: NS} tag --
276 # they are Secure by definition. {World: S} is allowed for
277 # explicitness; missing tag is the default.
278 if (
279 file_is_in_ring1_or_ring2(rel)
280 and world_match is not None
281 and world_match.group(1) in ("NS", "NSC")
282 ):
283 findings.append(
284 f"{rel}: Ring 1/2 file carries {{World: {world_match.group(1)}}} "
285 f"-- Rings 1 and 2 are Secure-only by policy"
286 )
287
288 return findings

291# ---------------------------------------------------------------------------
292# Selftest -- both directions, plus scope/classification assertions. tools/ was
293# silently omitted until #358; it is enumerated now but is not ring3+, so the
294# selftest pins that decision in place rather than leaving it to a tuple.
295# ---------------------------------------------------------------------------
296def selftest() -> int:
297 """Prove the bans fire, plain code stays quiet, and the scope decision holds."""
298 print("check_world_tags.py --selftest")
299 failures: list[str] = []
300 with tempfile.TemporaryDirectory() as tmp:
301 bad = pathlib.Path(tmp) / "bad.c"
302 bad.write_text("void f(void) __attribute__((cmse_nonsecure_entry));\n", encoding="utf-8")
303 expect(bool(check_file(bad)), "cmse_nonsecure_entry outside libs/ra8_nsc/ fires", failures)
304 good = pathlib.Path(tmp) / "good.c"
305 good.write_text("void f(void) { /* plain host code, no TrustZone */ }\n", encoding="utf-8")
306 expect(not check_file(good), "plain code with no ring3+ role stays quiet", failures)
307
308 expect(
309 file_is_in_ring3_plus("libs/ra8_hal/ra8_gpio.c"),
310 "libs/ra8_hal/ is ring3+ (a World tag is required)",
311 failures,
312 )
313 expect(
314 not file_is_in_ring3_plus("tools/ra8_emulator/src/main.c"),
315 "tools/ host code is not ring3+ (no World tag required -- documented decision)",
316 failures,
317 )
318 scope = set(first_party_paths(SOURCE_SUFFIXES))
319 expect(
320 any(s.startswith("tools/") for s in scope),
321 "tools/ is enumerated (the scan-dir list omitted it before #358)",
322 failures,
323 )
324 return report(failures)

361def main(argv: list[str]) -> int:
362 """Check that every Ring 3+ source declares which TrustZone world it targets.
363
364 The tag exists because the same source can be compiled into the Secure or
365 the Non-secure image, and nothing in the file otherwise says which was
366 intended -- so a file that quietly ends up in the wrong world produces a
367 build that links and faults at runtime.
368
369 With no paths the scan covers the derived first-party C set; naming paths
370 narrows it for the pre-commit hook.
371
372 Returns 1 listing each untagged file, 0 when every in-scope file declares
373 a world.
374 """
375 args = _build_parser().parse_args(argv)
376
377 if args.selftest:
378 return selftest()
379
380 if args.warn and args.strict:
381 print(
382 "check_world_tags.py: --warn and --strict are mutually exclusive",
383 file=sys.stderr,
384 )
385 return 2
386
387 strict = args.strict and not args.warn
388 targets = _select_targets(args.paths)
389
390 findings: list[str] = []
391 file_count = 0
392 for f in iter_source_files(targets):
393 file_count += 1
394 findings.extend(check_file(f))
395
396 if findings:
397 for line in findings:
398 print(line, file=sys.stderr)
399 verdict = "strict" if strict else "warn"
400 print(
401 f"check_world_tags.py: {len(findings)} finding(s) "
402 f"across {file_count} file(s) [{verdict}]",
403 file=sys.stderr,
404 )
405 return 1 if strict else 0
406
407 print(
408 f"check_world_tags.py: 0 findings across {file_count} file(s)",
409 file=sys.stderr,
410 )
411 return 0
