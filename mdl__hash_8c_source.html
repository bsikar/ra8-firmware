53RA8_INTERNAL static bool hash_stream(FILE* fp, uint64_t* h)
54{
55 uint8_t buf[k_hash_chunk_bytes];
56 for (uint32_t chunk = 0U; chunk < (uint32_t)k_hash_max_chunks; ++chunk) {
57 const size_t n = fread(buf, 1U, sizeof(buf), fp);
58 if (n > 0U) {
59 *h = mdl_hash_bytes_seed(buf, n, *h);
60 }
61 if (n < sizeof(buf)) {
62 return ferror(fp) == 0; /* short read: EOF (ok) vs a real error */
63 }
64 }
65 return true; /* hit the (absurd) chunk ceiling: treat as complete */
66}
