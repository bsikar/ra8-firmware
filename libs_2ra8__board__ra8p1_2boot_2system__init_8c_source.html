250static void internal_enable_fault_handlers(void)
251{
252 /* ARMv8-M SCB->SHCSR: MEMFAULTENA[16], BUSFAULTENA[17], USGFAULTENA[18],
253 * SECUREFAULTENA[19] take MemManage/BusFault/UsageFault/SecureFault to
254 * their own handlers instead of escalating to HardFault, so
255 * ra8_exception_report sees the real class. SECUREFAULTENA is a
256 * Secure-bank bit (RES0 from NS); this boot executes in the Secure
257 * state, see the function contract for the scoping rationale. */
258 enum : uint32_t {
259 k_ra8_shcsr_memfaultena = 1UL << 16,
260 k_ra8_shcsr_busfaultena = 1UL << 17,
261 k_ra8_shcsr_usgfaultena = 1UL << 18,
262 k_ra8_shcsr_securefaultena = 1UL << 19,
263 };
264 uint32_t shcsr = ra8_boot_read32(k_ra8_scb_shcsr_addr);
265 shcsr |= k_ra8_shcsr_memfaultena | k_ra8_shcsr_busfaultena | k_ra8_shcsr_usgfaultena |
266 k_ra8_shcsr_securefaultena;
267 ra8_boot_write32(k_ra8_scb_shcsr_addr, shcsr);
268 ra8_boot_dsb();
269 ra8_boot_isb();
270}

301static void internal_enable_div0_trap(void)
302{
303 /* ARMv8-M SCB->CCR: DIV_0_TRP[4] promotes SDIV/UDIV-by-zero from
304 * "quotient reads as 0" to a UsageFault with CFSR.DIVBYZERO set.
305 * UNALIGN_TRP[3] stays 0 -- see the function contract for evidence. */

307 k_ra8_ccr_div_0_trp = 1UL << 4,

311 ra8_boot_write32(k_ra8_scb_ccr_addr, ccr);

313 ra8_boot_isb();

367/* RBAR attribute bits[4:0] = { SH[4:3], AP[2:1], XN[0] }. AP: 0b01<<1 = RW
368 * any-privilege, 0b11<<1 = RO any-privilege. The memory TYPE is NOT here --
369 * it is the RLAR AttrIndx (see internal_mpu_set_region). */
370enum : uint32_t {
371 k_ra8_rbar_attr_ro_x = 0x06U,
372 k_ra8_rbar_attr_rw_xn = 0x03U,
373 k_ra8_mpu_rlar_enable = 1UL << 0,
374};
375
376/* RLAR AttrIndx (bits[3:1]) selects which MAIR byte sets the region's memory
377 * type. The old code tried to encode "device" in RBAR (k_ra8_rbar_attr_device_rw
378 * = 0x23), which both spilled bit5 into the base address and left every RLAR at
379 * AttrIndx 0 = MAIR0[0] = Normal cacheable -- so peripheral MMIO was mapped
380 * cacheable. The type must be selected here instead. */
381enum : uint32_t {
382 k_ra8_mpu_rlar_attridx_shift = 1U,
383 k_ra8_mpu_attridx_normal_wbwa = 0U,
384 k_ra8_mpu_attridx_noncacheable = 1U,
385 k_ra8_mpu_attridx_device = 2U,
386};
387

423[[maybe_unused]] static void internal_mpu_init(void)
424{
425 enum : uint32_t {
426 k_ra8_mpu_ctrl_enable = 1UL << 0,
427 k_ra8_mpu_ctrl_privdefena = 1UL << 2,
428 k_ra8_mair0_default = 0x000444FFUL,
429 };
430
431 /* MAIR0: idx0 = Normal WB/WA, idx1 = Normal non-cacheable, idx2 = Device-nGnRE. */
432 ra8_boot_write32(k_ra8_mpu_mair0_addr, k_ra8_mair0_default);
433 ra8_boot_write32(k_ra8_mpu_mair1_addr, 0U);
434
435 /* Code = RO + executable cacheable; RAM = RW/NX cacheable; peripherals =
436 * Device-nGnRE (NOT cacheable -- the AttrIndx selects the type). */
437 internal_mpu_set_region(0U,
438 (k_ra8_mpu_mram_base | k_ra8_rbar_attr_ro_x),
439 (k_ra8_mpu_mram_limit | k_ra8_mpu_rlar_enable),
440 k_ra8_mpu_attridx_normal_wbwa);
441 internal_mpu_set_region(1U,
442 (k_ra8_mpu_sram_base | k_ra8_rbar_attr_rw_xn),
443 (k_ra8_mpu_sram_limit | k_ra8_mpu_rlar_enable),
444 k_ra8_mpu_attridx_normal_wbwa);
445 internal_mpu_set_region(2U,
446 (k_ra8_mpu_sdram_base | k_ra8_rbar_attr_rw_xn),
447 (k_ra8_mpu_sdram_limit | k_ra8_mpu_rlar_enable),
448 k_ra8_mpu_attridx_normal_wbwa);
449 internal_mpu_set_region(3U,
450 (k_ra8_mpu_peri_base | k_ra8_rbar_attr_rw_xn),
451 (k_ra8_mpu_peri_limit | k_ra8_mpu_rlar_enable),
452 k_ra8_mpu_attridx_device);
453 /* Shared M85<->M33 SRAM bank: Normal NON-cacheable so cross-core hand-offs
454 * (mailbox at 0x22100000, CPU1 RAM) stay coherent with the M85 D-cache on. */
455 internal_mpu_set_region(4U,
456 (k_ra8_mpu_shram_base | k_ra8_rbar_attr_rw_xn),
457 (k_ra8_mpu_shram_limit | k_ra8_mpu_rlar_enable),
458 k_ra8_mpu_attridx_noncacheable);
459
460 /* Enable MPU with PRIVDEFENA so unclassified privileged accesses
461 * still succeed through the default system memory map. */
462 ra8_boot_dsb();
463 ra8_boot_write32(k_ra8_mpu_ctrl_addr, k_ra8_mpu_ctrl_enable | k_ra8_mpu_ctrl_privdefena);
464 ra8_boot_dsb();
465 ra8_boot_isb();

468/* =============================================================================
469 * Public entry point
470 * =============================================================================
471 *
472 * Called from `Reset_Handler` before the .data copy and .bss zero.
473 * Must therefore not touch any global variables -- everything here
474 * runs on the stack and writes to CPU / SCB memory only.

477/* NOLINTNEXTLINE(readability-identifier-naming) -- CMSIS-mandated name. */
478void SystemInit(void)
479{
480 /* Mask global IRQs so the application gets a deterministic init
481 * window. main() must call ``ra8_isr_globals_enable`` once every
482 * driver / handler is wired up before any IRQ source can fire. */
483 ra8_boot_disable_irq();
484
485 internal_set_vtor();
486 internal_enable_fpu();
487 internal_enable_fpu_lazy_stack();
488 /* Always-on fault diagnostics: take MemManage/BusFault/UsageFault/
489 * SecureFault to their own decoded handlers and trap divide-by-zero
490 * as a UsageFault (independent of the cache / MPU build flag). */
491 internal_enable_fault_handlers();
492 internal_enable_div0_trap();
493#ifdef RA8_BOOT_ENABLE_CACHE_MPU
494 /* T4-06: MPU memory-attribute map FIRST (so the D-cache sees Device-nGnRE
495 * MMIO and the non-cacheable shared SRAM, not Normal cacheable), then the
496 * caches. Each *_enable does its own architectural invalidate before setting
497 * CCR. Region 4 (non-cacheable shared SRAM) keeps M85<->M33 hand-offs
498 * coherent with no software maintenance. */
499 internal_mpu_init();
500 internal_enable_icache();
501 internal_enable_dcache();
502 internal_enable_branch_predictor();
503#else
504 /* Default OFF: caches + MPU stay disabled -- no behaviour change. */
505 (void)internal_enable_icache;
506 (void)internal_enable_dcache;
507 (void)internal_enable_branch_predictor;
508 (void)internal_mpu_init;
509#endif
510 (void)ra8_trustzone_init; /* TrustZone has its own gate (RA8_TRUSTZONE_ENABLE). */
511 internal_set_priority_grouping();
512}
