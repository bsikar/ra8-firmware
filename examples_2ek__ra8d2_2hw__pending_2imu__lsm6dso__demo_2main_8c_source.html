132static uint32_t imu_demo_i32_to_dec(int32_t value, uint8_t* out)
133{
134 uint8_t tmp[k_imu_demo_int_str_cap] = {};
135 uint32_t n = 0U;
136 bool neg = false;
137 /* Two's complement edge case: INT16_MIN. We work in int32_t so we
138 * have headroom to negate. */
139 int32_t v = value;
140 if (v < 0) {
141 neg = true;
142 v = -v;
143 }
144 if (v == 0) {
145 tmp[n++] = (uint8_t)'0';
146 } else {
147 while (v > 0 && n < (uint32_t)k_imu_demo_int_str_cap) {
148 tmp[n++] = (uint8_t)('0' + (uint8_t)((uint32_t)v % k_imu_decimal_base));
149 v = (int32_t)((uint32_t)v / k_imu_decimal_base);
150 }
151 }
152 if (neg && n < (uint32_t)k_imu_demo_int_str_cap) {
153 tmp[n++] = (uint8_t)'-';
154 }
155 /* Reverse into the caller's buffer. */
156 for (uint32_t i = 0U; i < n; ++i) {
157 out[i] = tmp[n - 1U - i];
158 }
159 if (n < (uint32_t)k_imu_demo_int_str_cap) {
160 out[n] = 0U;
161 }
162 return n;
163}

219/* cppcheck-suppress constParameterCallback ; ctx must stay non-const to match

221static ra8_err_t imu_demo_iic_write(void* ctx, uint8_t reg, const uint8_t* buf, uint32_t len)
222{
223 if (len > (uint32_t)k_imu_demo_iic_tx_cap - 1U) {
224 return k_ra8_err_invalid_arg;

226 uint8_t scratch[k_imu_demo_iic_tx_cap] = {};

229 scratch[i + 1U] = buf[i];

232 return ra8_i3c_write(c->channel, c->addr7, scratch, len + 1U, false);

494int32_t main(void)
495{
496 imu_demo_setup_or_halt();
497 ra8_isr_globals_enable();
498
499 imu_demo_iic_ctx_t ctx = {
500 .channel = (uint8_t)k_imu_demo_iic_channel,
501 .addr7 = (uint8_t)k_lsm6dso_i2c_addr_sa0_high,
502 };
503 const ra8_lsm6dso_bus_t bus = {
504 .read_regs = imu_demo_iic_read,
505 .write_regs = imu_demo_iic_write,
506 .ctx = &ctx,
507 };
508 ra8_lsm6dso_t dev = {};
509 if (ra8_lsm6dso_init(&dev, &bus) != k_ra8_ok) {
510 imu_demo_tx(k_imu_demo_banner_err, (uint32_t)(sizeof(k_imu_demo_banner_err) - 1U));
511 (void)ra8_board_led_on(k_ra8_board_led2);
512 imu_demo_panic_halt();
513 }
514
515 imu_demo_check_who_am_i(&dev);
516 /* Configure: +-2 g, +-250 dps, 104 Hz. */
517 imu_demo_configure(&dev);
518
519 while (1) {
520 imu_demo_sample_and_emit(&dev);
521 (void)ra8_board_led_toggle(k_ra8_board_led1);
522 ra8_delay_ms(k_imu_demo_period_ms);
523 }
524
525 imu_demo_panic_halt();
526 return 0;
527}
