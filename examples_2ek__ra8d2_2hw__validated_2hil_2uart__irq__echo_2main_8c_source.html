244{
245 uint32_t cpuclk0_hz = 0U;

247
248 if (ra8_cgc_init() != k_ra8_ok) {

250 }
251 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
252 uart_irq_panic_halt();
253 }
254 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
255 uart_irq_panic_halt();
256 }
257 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
258 uart_irq_panic_halt();
259 }
260 if (uart_irq_pins_init() != k_ra8_ok) {
261 uart_irq_panic_halt();
262 }

264 const ra8_sci_cfg_t sci_cfg = {
265 .baud = k_uart_irq_baud,
266 .data_bits = k_ra8_sci_data_8,
267 .parity = k_ra8_sci_parity_none,
268 .stop_bits = k_ra8_sci_stop_1,
269 .pclk_hz = pclka_hz,
270 };
271 if (ra8_sci_init((uint8_t)k_uart_irq_sci_chan, &sci_cfg) != k_ra8_ok) {
272 uart_irq_panic_halt();
273 }
274 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
275 uart_irq_panic_halt();
276 }
277 if (ra8_isr_init() != k_ra8_ok) {
278 uart_irq_panic_halt();
279 }
280 if (uart_irq_register_isrs() != k_ra8_ok) {
281 uart_irq_panic_halt();
282 }
283}

296int32_t main(void)
297{
298 uart_irq_setup_or_halt();
299
300 ra8_isr_globals_enable();
301
302 /* Polled path: announce readiness (also exercises ra8_sci_write_polling). */
303 (void)ra8_sci_write_polling((uint8_t)k_uart_irq_sci_chan,
304 k_uart_irq_banner,
305 (uint32_t)(sizeof(k_uart_irq_banner) - 1U));
306
307 /* Attach the echo callback and arm the first interrupt-driven receive. */
308 (void)ra8_sci_attach_rx_handler((uint8_t)k_uart_irq_sci_chan, uart_irq_rx_cb, nullptr);
309 (void)ra8_sci_read((uint8_t)k_uart_irq_sci_chan, &s_uart_irq_rx_byte, 1U);
310
311 while (1) {
312 /* All echo work happens in the SCI ISRs; the loop only idles. */
313 ra8_delay_ms((uint32_t)k_uart_irq_loop_ms);
314 }
315
316 uart_irq_panic_halt();
317 return 0;
318}
