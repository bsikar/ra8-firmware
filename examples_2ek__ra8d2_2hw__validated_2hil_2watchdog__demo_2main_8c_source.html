99{

102 wdt_demo_panic_halt();

126int32_t main(void)
127{
128 wdt_demo_setup_or_halt();
129 ra8_isr_globals_enable();
130
131 ra8_reset_cause_t cause = k_ra8_reset_cause_unknown;
132 (void)ra8_reset_get_cause(&cause);
133 uint32_t msg_len = 0U;
134 const uint8_t* msg = wdt_demo_banner_for(cause, &msg_len);
135 (void)ra8_board_uart_console_write(msg, (size_t)msg_len);
136
137 /* Stage 1: refresh for ``alive_seconds`` seconds. */
138 const uint32_t refreshes_per_sec = k_ms_per_sec / (uint32_t)k_wdt_demo_refresh_ms;
139 const uint32_t total_refreshes = (uint32_t)k_wdt_demo_alive_seconds * refreshes_per_sec;
140 for (uint32_t i = 0U; i < total_refreshes; ++i) {
141 ra8_iwdt_refresh_deferred();
142 (void)ra8_board_led_toggle(k_ra8_board_led1);
143 ra8_delay_ms((uint32_t)k_wdt_demo_refresh_ms);
144 }
145
146 /* Stage 2: stop refreshing and let the IWDT underflow reset us. */
147 (void)ra8_board_uart_console_write(k_wdt_demo_msg_stop,
148 (size_t)(sizeof(k_wdt_demo_msg_stop) - 1U));
149 while (1) {
150 __asm__ volatile("wfi");
151 }
152}
