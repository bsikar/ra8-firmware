4"""Assert a viewer-dumped PPM actually rendered content.
5
6This exists because "the viewer exited 0 and wrote a file" is not evidence that
7anything was drawn. The viewer's failure modes both produce a perfectly valid
8PPM: an all-one-color frame (the blank or error fill) and a zero-size image. A
9real render must therefore have positive dimensions AND more than one distinct
10color, and this checks both.
11
12Distinct exit statuses let integration.sh tell a blank render (a real product
13failure that must fail the run) apart from a bad invocation or a missing
14Pillow, which must not be reported as a rendering bug:
15
16 0 frame has content
17 1 blank, degenerate, or unopenable frame
18 2 usage error
19 3 Pillow not installed
20
21The color test is a floor, not a fingerprint: it catches blank output, not
22wrong-but-colorful output.
23"""
24
25from __future__ import annotations
26
27import sys
28
29try:
30 from PIL import Image
31except ImportError:
32 sys.stderr.write("check_ppm.py: needs Pillow\n")
33 sys.exit(3)

35# Exit statuses (integration.sh distinguishes a blank render from a bad call).
36EXIT_OK = 0
37EXIT_BLANK = 1
38EXIT_USAGE = 2
39
40# argv layout.
41ARGV_MIN = 2
42ARGV_WITH_MIN_COLORS = 3
43
44# A frame must carry at least this many distinct colors to count as rendered.
45DEFAULT_MIN_COLORS = 2
46
47# getcolors() cap: past this it returns None, which is itself proof of a rich frame.
48COLOR_CAP = 1 << 20
49
50
51def main() -> int:
52 """Check the PPM named in argv and map the verdict onto an exit status.
53
54 Reads `FILE.ppm [MIN_COLORS]` from `sys.argv`. MIN_COLORS defaults to 2 --
55 "not a single flat color" -- and raising it is how a caller demands a
56 richer frame than merely non-blank.
57
58 An unopenable file returns EXIT_BLANK rather than a usage error, on purpose:
59 from integration.sh's point of view a frame the viewer never produced and a
60 frame it produced blank are the same product failure.
61
62 Color counting is capped at COLOR_CAP. Past that `getcolors()` returns None,
63 which is itself proof of a rich frame, so the count is reported as the cap.
64
65 Returns:
66 EXIT_OK, EXIT_BLANK, or EXIT_USAGE, for `sys.exit`.
67
68 Raises:
69 ValueError: MIN_COLORS was given but is not an integer.
70 """
71 if len(sys.argv) < ARGV_MIN:
72 sys.stderr.write("usage: check_ppm.py FILE.ppm [MIN_COLORS]\n")
73 return EXIT_USAGE
74 path = sys.argv[1]
75 min_colors = int(sys.argv[2]) if len(sys.argv) >= ARGV_WITH_MIN_COLORS else DEFAULT_MIN_COLORS
76 try:
77 img = Image.open(path).convert("RGB")
78 except OSError as exc:
79 sys.stderr.write(f"check_ppm: cannot open {path}: {exc}\n")
80 return EXIT_BLANK
81 w, h = img.size
82 if w <= 0 or h <= 0:
83 sys.stderr.write(f"check_ppm: degenerate size {w}x{h}\n")
84 return EXIT_BLANK
85 colors = img.getcolors(maxcolors=COLOR_CAP)
86 n = COLOR_CAP if colors is None else len(colors)
87 if n < min_colors:
88 sys.stderr.write(f"check_ppm: BLANK {path} ({w}x{h}, {n} color(s))\n")
89 return EXIT_BLANK
90 print(f"check_ppm: OK {path} ({w}x{h}, {n}+ colors)")
91 return EXIT_OK
92
93
94if __name__ == "__main__":
95 sys.exit(main())
