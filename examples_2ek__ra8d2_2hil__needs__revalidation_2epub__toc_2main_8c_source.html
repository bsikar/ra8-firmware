177static uint32_t etoc_crc32(const uint8_t* data, size_t len)
178{
179 uint32_t crc = (uint32_t)k_etoc_crc_init;
180 for (size_t i = 0U; i < len; i++) {
181 crc ^= (uint32_t)data[i];
182 for (uint32_t bit = 0U; bit < (uint32_t)k_etoc_crc_bits; bit++) {
183 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
184 crc = (crc >> 1U) ^ ((uint32_t)k_etoc_crc_poly & mask);
185 }
186 }
187 return ~crc;
188}

335static void etoc_read_toc(ra8_fs_mount_t* mount,
336 const char* path,
337 uint32_t stage,
338 const uint8_t* fmsg,
339 uint32_t fmsglen,
340 volatile uint32_t* kind,
341 volatile uint32_t* count,
342 volatile uint32_t* e0crc,
343 volatile uint32_t* ch0)
344{
345 if (ra8_epub_open_streamed_fs(mount, path, &s_epub_io, &s_book) != k_ra8_ok) {
346 etoc_fail(stage, fmsg, fmsglen);
347 }
348 uint8_t k = 0U;
349 uint16_t n = 0U;
350 if ((ra8_epub_get_toc_kind(&s_book, &k) != k_ra8_ok) ||
351 (ra8_epub_get_toc_count(&s_book, &n) != k_ra8_ok)) {
352 etoc_fail(stage, fmsg, fmsglen);
353 }
354 *kind = (uint32_t)k;
355 *count = (uint32_t)n;
356 *e0crc = 0U;
357 *ch0 = (uint32_t)k_etoc_no_chap;
358 if (n > 0U) {
359 ra8_epub_toc_entry_t entry = {};
360 uint16_t chap = (uint16_t)k_etoc_no_chap;
361 if ((ra8_epub_get_toc_entry(&s_book, 0U, &entry) != k_ra8_ok) ||
362 (ra8_epub_toc_entry_to_chapter(&s_book, 0U, &chap) != k_ra8_ok)) {
363 etoc_fail(stage, fmsg, fmsglen);
364 }
365 *e0crc = etoc_title_crc(entry.title);
366 *ch0 = (uint32_t)chap;
367 }
368 (void)ra8_epub_close_streamed_fs(&s_epub_io, &s_book);
369}

451int32_t main(void)
452{

454 etoc_setup_or_halt(&pclka_hz);
455 ra8_isr_globals_enable();

462

465 etoc_check_bad(mount);
466

468
469 /* Idle with a heartbeat that advances ONLY here -- every failure path parks in
470 * WFI without bumping it -- so a memprobe gate proves both TOC code paths plus
471 * the malformed-TOC fallback ran on real SD bytes. */
472 while (1) {
473 ra8_delay_ms(k_etoc_frame_ms);
474 g_etoc_heartbeat++;
475 }
