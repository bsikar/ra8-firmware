65RA8_INTERNAL static bool starts_with_ci(const char* s, const char* prefix)
66{
67 size_t i = 0U;
68 while (prefix[i] != '\0') {
69 const char a = s[i];
70 if (a == '\0') {
71 return false;
72 }
73 const char la = (char)(((a >= 'A') && (a <= 'Z')) ? (a + ('a' - 'A')) : a);
74 if (la != prefix[i]) {
75 return false;
76 }
77 ++i;
78 }
79 return true;
80}

91RA8_INTERNAL static mdl_addr_class_t classify_v4(const unsigned char* o)
92{
93 const unsigned o0 = o[k_v4_o0];
94 const unsigned o1 = o[k_v4_o1];
95 if (o0 == (unsigned)k_v4_zero_net) {
96 return k_mdl_addr_unknown;
97 }
98 if (o0 == (unsigned)k_v4_loopback_net) {
99 return k_mdl_addr_loopback;
100 }
101 if (o0 >= (unsigned)k_v4_multicast_min) {
102 return k_mdl_addr_unknown;
103 }
104 if (o0 == (unsigned)k_v4_linklocal) {
105 return (o1 == (unsigned)k_v4_linklocal_2) ? k_mdl_addr_linklocal : k_mdl_addr_public;
106 }
107 const bool priv_a = (o0 == (unsigned)k_v4_private_a);
108 const bool priv_b = (o0 == (unsigned)k_v4_private_b) && (o1 >= (unsigned)k_v4_private_b_lo) &&
109 (o1 <= (unsigned)k_v4_private_b_hi);
110 const bool priv_c = (o0 == (unsigned)k_v4_private_c) && (o1 == (unsigned)k_v4_private_c_2);
111 const bool cgnat = (o0 == (unsigned)k_v4_cgnat) && (o1 >= (unsigned)k_v4_cgnat_lo) &&
112 (o1 <= (unsigned)k_v4_cgnat_hi);
113 return (priv_a || priv_b || priv_c || cgnat) ? k_mdl_addr_private : k_mdl_addr_public;
114}

117RA8_INTERNAL static bool is_v4_mapped(const unsigned char* b)
118{
119 for (size_t i = 0U; i < (size_t)k_v6_mapped_ff_a; ++i) {

140RA8_INTERNAL static mdl_addr_class_t classify_v6(const unsigned char* b)
141{
142 if (is_v4_mapped(b)) {
143 return classify_v4(b + (size_t)k_v6_mapped_v4);
144 }
145 if (is_v6_loopback(b)) {
146 return k_mdl_addr_loopback;
147 }
148 if (b[k_v4_o0] == (unsigned char)k_v6_byte_ff) {
149 return k_mdl_addr_unknown; /* multicast */
150 }
151 if ((b[k_v4_o0] == (unsigned char)k_v6_ll_lead) &&
152 ((b[k_v4_o1] & (unsigned char)k_v6_ll_mask) == (unsigned char)k_v6_ll_value)) {
153 return k_mdl_addr_linklocal;
154 }
155 if ((b[k_v4_o0] & (unsigned char)k_v6_ula_mask) == (unsigned char)k_v6_ula_value) {
156 return k_mdl_addr_private;
157 }
158 return k_mdl_addr_public;
159}

161mdl_addr_class_t mdl_classify_ip(const char* ip)
162{
163 if (ip == nullptr) {
164 return k_mdl_addr_unknown;
165 }
166 unsigned char v4[sizeof(struct in_addr)] = {};
167 if (inet_pton(AF_INET, ip, v4) == 1) {
168 return classify_v4(v4);
169 }
170 unsigned char v6[sizeof(struct in6_addr)] = {};
171 if (inet_pton(AF_INET6, ip, v6) == 1) {
172 return classify_v6(v6);
173 }
174 return k_mdl_addr_unknown;
175}

199bool mdl_url_host(const char* url, char* out, size_t cap)
200{
201 if ((out == nullptr) || (cap == 0U)) {
202 return false;
203 }
204 out[0] = '\0';
205 if (url == nullptr) {
206 return false;
207 }
208 const char* sep = strstr(url, "://");
209 if (sep == nullptr) {
210 return false;
211 }
212 const char* host = sep + (sizeof("://") - 1U);
213 const char* at = strchr(host, '@');
214 const char* path = strpbrk(host, "/?#");
215 if (at != nullptr) {
216 const bool at_in_authority = (path == nullptr) || (at < path);
217 if (at_in_authority) {
218 host = at + 1U;
219 }
220 }
221 size_t n = 0U;
222 /* Keep the port: robots.txt is scoped per (scheme, host, port) authority, and
223 * a redirect to a different port is a different origin. Only userinfo is
224 * dropped (handled above). */
225 while ((host[n] != '\0') && (host[n] != '/') && (host[n] != '?') && (host[n] != '#')) {
226 if ((n + 1U) >= cap) {
227 out[0] = '\0';
228 return false;
229 }
230 const char c = host[n];
231 out[n] = (char)(((c >= 'A') && (c <= 'Z')) ? (c + ('a' - 'A')) : c);
232 ++n;
233 }
234 out[n] = '\0';
235 return n > 0U;
236}

238bool mdl_url_path(const char* url, char* out, size_t cap)
239{
240 if ((out == nullptr) || (cap == 0U)) {
241 return false;
242 }
243 out[0] = '\0';
244 if (url == nullptr) {
245 return false;
246 }
247 const char* sep = strstr(url, "://");
248 if (sep == nullptr) {
249 return false;
250 }
251 const char* host = sep + (sizeof("://") - 1U);
252 const char* path = strchr(host, '/');
253 if (path == nullptr) {
254 if (cap < 2U) {
255 return false;
256 }
257 out[0] = '/';
258 out[1] = '\0';
259 return true;
260 }
261 size_t n = 0U;
262 while ((path[n] != '\0') && (path[n] != '?') && (path[n] != '#')) {
263 if ((n + 1U) >= cap) {
264 out[0] = '\0';
265 return false;
266 }
267 out[n] = path[n];
268 ++n;
269 }
270 out[n] = '\0';
271 return true;
272}
