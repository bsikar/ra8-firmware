154{
155 internal_agt_periodic_setup_or_halt();
156 ra8_isr_globals_enable();
157
158 if (internal_agt_periodic_arm() != k_ra8_ok) {
159 internal_agt_periodic_panic_halt();

162 while (1) {
163 uint8_t status = 0U;
164 if (ra8_agt_get_status((uint8_t)k_agt_periodic_channel, &status) != k_ra8_ok) {
165 break;
166 }
167 if ((status & (uint8_t)k_agt_periodic_undf_bit) != 0U) {
168 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
169 break;
170 }
171 if (ra8_board_uart_console_write(s_agt_periodic_log_msg,
172 (size_t)(sizeof(s_agt_periodic_log_msg) - 1U)) != k_ra8_ok) {
173 break;
174 }
175 /* Re-arm: stop + start clears AGTCR.TUNDF on real silicon. */
176 if (ra8_agt_stop((uint8_t)k_agt_periodic_channel) != k_ra8_ok) {

179 if (internal_agt_periodic_arm() != k_ra8_ok) {
180 break;
181 }
182 }
183 ra8_delay_ms((uint32_t)k_agt_periodic_poll_ms);
184 }
185 internal_agt_periodic_panic_halt();
