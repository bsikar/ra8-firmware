248static void hs_cfifo_write(uint32_t value, unsigned size)
249{
250 const uint8_t p = hs_curpipe();
251 for (unsigned i = 0U; (i < size) && (s_hs.pout_len[p] < (uint16_t)k_usbhs_pkt_cap); i++) {
252 s_hs.pout[p][s_hs.pout_len[p]] =

255 }

259static uint32_t hs_cfifo_read(unsigned size)
260{
261 const uint8_t p = hs_curpipe();
262 uint8_t* data = (p == 0U) ? s_hs.din : s_hs.pin[p];
263 uint16_t* rd = (p == 0U) ? &s_hs.din_rd : &s_hs.pin_rd[p];
264 const uint16_t len = (p == 0U) ? s_hs.din_len : s_hs.pin_len[p];
265 uint32_t v = 0U;
266 for (unsigned i = 0U; (i < size) && (*rd < len); i++) {
267 v |= (uint32_t)((uint32_t)data[*rd] << (i * (unsigned)k_usbhs_byte_bits));
268 (*rd)++;
269 }
270 return v;
271}

279static void hs_bulk_out_commit(uc_engine* uc)
280{
281 const uint8_t p = hs_curpipe();
282 if (p == 0U) {
283 /* DCP OUT: a control-write data stage (e.g. a DFU_DNLOAD block). Deliver it
284 * to the device and gate BEMP on the device draining it. A zero-length DCP
285 * BVAL is instead the control-read OUT status ZLP, driven from the CCPL
286 * path -- ignore it here. */
287 if (s_hs.pout_len[0] > 0U) {
288 board_usb_bridge_dcp_out(uc, s_hs.pout[0], s_hs.pout_len[0]);
289 s_hs.pout_wait[0] = true;
290 s_hs.pout_len[0] = 0U;
291 }
292 return;
293 }
294 board_usb_bridge_bulk_out(uc, s_hs.pipe_ep[p], s_hs.pout[p], s_hs.pout_len[p]);
295 s_hs.bulk_out++;
296 /* DEFER the buffer-empty (BEMP) signal until the device firmware drains this
297 * packet from its CFIFO: raising it now lets the polled host push the next
298 * OUT packet immediately, overwriting the undrained bank (which wedged the
299 * multi-packet WRITE(10) data stage -- the device saw only the last chunk and
300 * never produced the CSW). hs_bempsts_read() raises BEMP on device drain. */
301 s_hs.pout_wait[p] = true;
302 s_hs.pout_len[p] = 0U;
303}

306static uint16_t hs_bempsts_read(void)
307{
308 if (s_hs.pout_wait[0] && board_usb_bridge_dcp_out_consumed()) {
309 s_hs.pout_wait[0] = false; /* DCP control-write data stage drained by device. */
310 hs_or((uint16_t)k_ra8_usb_off_bempsts, (uint16_t)k_usbhs_dcp_bit);
311 }
312 for (uint8_t p = 1U; p < (uint8_t)k_usbhs_pipes; p++) {
313 if (s_hs.pout_wait[p] && board_usb_bridge_bulk_out_consumed(s_hs.pipe_ep[p])) {
314 s_hs.pout_wait[p] = false;
315 hs_or((uint16_t)k_ra8_usb_off_bempsts, (uint16_t)(1U << p));
316 }
317 }
318 return hs_reg((uint16_t)k_ra8_usb_off_bempsts);
319}

359static uint16_t hs_brdysts_read(uc_engine* uc)
360{
361 if (s_hs.ctrl_status_pending && board_usb_bridge_dev_took_ccpl()) {
362 s_hs.ctrl_status_pending = false;
363 hs_ctrl_status_complete(uc);
364 }
365 if (!s_hs.din_ready && board_usb_bridge_dcp_in_ready()) {
366 s_hs.din_len = board_usb_bridge_dcp_in_take(s_hs.din, (uint16_t)k_usbhs_din_cap);
367 s_hs.din_rd = 0U;
368 s_hs.din_ready = true;
369 hs_or((uint16_t)k_ra8_usb_off_brdysts, (uint16_t)k_usbhs_dcp_bit);
370 }
371 for (uint8_t p = 1U; p < (uint8_t)k_usbhs_pipes; p++) {
372 const bool armed = (s_hs.pipe_pid[p] & (uint16_t)k_ra8_pid_mask) == (uint16_t)k_ra8_pid_buf;
373 if (!armed || s_hs.pin_ready[p]) {
374 continue;
375 }
376 if (board_usb_bridge_bulk_in_ready(s_hs.pipe_ep[p])) {
377 s_hs.pin_len[p] =
378 board_usb_bridge_bulk_in_take(uc, s_hs.pipe_ep[p], s_hs.pin[p], (uint16_t)k_usbhs_pkt_cap);
379 s_hs.pin_rd[p] = 0U;
380 s_hs.pin_ready[p] = true;
381 s_hs.bulk_in++;
382 hs_or((uint16_t)k_ra8_usb_off_brdysts, (uint16_t)(1U << p));
383 }
384 }
385 return hs_reg((uint16_t)k_ra8_usb_off_brdysts);
386}

389static void hs_brdysts_write(uint16_t value)
390{
391 const uint16_t before = hs_reg((uint16_t)k_ra8_usb_off_brdysts);
392 const uint16_t cleared = (uint16_t)(before & (uint16_t)~value); /* 1 -> 0 transitions. */
393 hs_set((uint16_t)k_ra8_usb_off_brdysts, (uint16_t)(before & value));
394 if ((cleared & (uint16_t)k_usbhs_dcp_bit) != 0U) {
395 s_hs.din_ready = false; /* host consumed the DCP IN packet; allow a re-latch. */
396 }
397 for (uint8_t p = 1U; p < (uint8_t)k_usbhs_pipes; p++) {
398 if ((cleared & (uint16_t)(1U << p)) != 0U) {
399 s_hs.pin_ready[p] = false; /* host acked this pipe's BRDY; allow the next. */
400 }
401 }

410static void hs_capture_setup(void)
411{
412 const uint16_t req = hs_reg((uint16_t)k_ra8_usb_off_usbreq);
413 const uint16_t val = hs_reg((uint16_t)k_ra8_usb_off_usbval);
414 const uint16_t indx = hs_reg((uint16_t)k_ra8_usb_off_usbindx);

417 s_hs.setup[k_usbhs_setup_breq] =

419 s_hs.setup[k_usbhs_setup_wval_lo] = (uint8_t)(val & (uint16_t)k_usbhs_byte_mask);
420 s_hs.setup[k_usbhs_setup_wval_hi] =

423 s_hs.setup[k_usbhs_setup_widx_hi] =

425 s_hs.setup[k_usbhs_setup_wlen_lo] = (uint8_t)(leng & (uint16_t)k_usbhs_byte_mask);
426 s_hs.setup[k_usbhs_setup_wlen_hi] =

431static void hs_setup_launch(uc_engine* uc)
432{
433 hs_capture_setup();
434 const uint16_t leng = hs_reg((uint16_t)k_ra8_usb_off_usbleng);
435 s_hs.ctrl_breq = s_hs.setup[k_usbhs_setup_breq];
436 s_hs.ctrl_read =
437 ((s_hs.setup[k_usbhs_setup_bmrt] & (uint8_t)k_usbhs_setup_dir_in) != 0U) && (leng != 0U);
438 s_hs.ctrl_status_pending = false; /* fresh transfer: no deferred status yet. */
439 /* Discard any device CCPL left over from the previous control transfer's
440 * status stage (a control read leaves it unconsumed). Otherwise a deferred
441 * no-data status stage would complete on that stale pulse instead of waiting
442 * for THIS request's completion, re-introducing the host-races-device bug. */
443 (void)board_usb_bridge_dev_took_ccpl();
444
445 s_hs.din_ready = false; /* stale DCP IN state cleared for the fresh transfer. */
446 s_hs.din_len = 0U;
447 s_hs.din_rd = 0U;
448 board_usb_bridge_deliver_setup(uc, s_hs.setup);
449 s_hs.setups++;
450
451 /* The SIE latches SACK when the device ACKs the SETUP; the host gates on it. */
452 hs_or((uint16_t)k_ra8_usb_off_intsts1, (uint16_t)(1U << k_ra8_int1_bit_sack));
453}

456static void hs_ccpl_status(uc_engine* uc)
457{
458 if (s_hs.ctrl_read) {
459 /* Control-read OUT status: let the device close its side, then report the
460 * buffer empty so the host's best-effort BEMP wait returns promptly. */
461 board_usb_bridge_ctrl_status(uc);
462 hs_or((uint16_t)k_ra8_usb_off_bempsts, (uint16_t)k_usbhs_dcp_bit);
463 return;
464 }
465 /* A host-to-device control write the DEVICE must apply -- every one except
466 * SET_ADDRESS, whose USBADDR latch and IN-ZLP status stage the SIE owns:
467 * DEFER the host's status-stage BRDY until the device firmware has actually
468 * processed the request and pulsed DCPCTR.CCPL. Completing it synchronously
469 * races the polled host past a request the device is still applying:
470 * - no-data (SET_CONFIGURATION): it would issue GET_MAX_LUN and the first
471 * BOT CBW before the MSC class had activated and armed its bulk endpoints;
472 * - with-data (DFU_DNLOAD): it would issue the next SETUP -- clearing the
473 * device DCP-OUT staging -- before the device had drained the block, so
474 * the firmware image never lands. The device pulses CCPL only after it
475 * drains the data stage and applies the request, which is exactly the
476 * completion to wait on. hs_brdysts_read() finishes the stage on it. */
477 if (s_hs.ctrl_breq != (uint8_t)k_usbhs_breq_set_addr) {
478 s_hs.ctrl_status_pending = true;
479 return;
480 }
481 /* SET_ADDRESS: SIE-owned, no device CCPL to wait on -- complete now. */
482 hs_ctrl_status_complete(uc);
483}

486static void hs_dcpctr_write(uc_engine* uc, uint16_t value)
487{
488 const uint16_t before = hs_reg((uint16_t)k_ra8_usb_off_dcpctr);
489 const uint16_t sureq = (uint16_t)(1U << k_ra8_dcpctr_bit_sureq);
490 const uint16_t ccpl = (uint16_t)(1U << k_ra8_dcpctr_bit_ccpl);
491 if ((value & sureq) != 0U) {
492 hs_setup_launch(uc);
493 }
494 if (((value & ccpl) != 0U) && ((before & ccpl) == 0U)) {
495 hs_ccpl_status(uc);
496 }
497 /* SUREQ self-clears once the SIE has delivered the token; never latch it. */
498 hs_set((uint16_t)k_ra8_usb_off_dcpctr, (uint16_t)(value & (uint16_t)~sureq));
499}

542static uint64_t hs_read(uc_engine* uc, uint64_t off, unsigned size)
543{
544 switch ((uint16_t)off) {
545 case (uint16_t)k_ra8_usbhs_off_pllsta:
546 return (uint64_t)k_usbhs_plllock; /* PHY PLL reported locked so init proceeds. */
547 case (uint16_t)k_ra8_usb_off_syssts0:
548 return board_usb_dev_attached() ? (uint64_t)k_usbhs_lnst_j : 0U;
549 case (uint16_t)k_ra8_usb_off_dvstctr0: {
550 uint16_t v = hs_reg((uint16_t)k_ra8_usb_off_dvstctr0);
551 if (board_usb_dev_attached()) {
552 v = (uint16_t)(v | (uint16_t)k_usbhs_rhst_fs); /* connected FS device speed. */
553 }
554 return (uint64_t)v;
555 }
556 case (uint16_t)k_ra8_usb_off_cfifo:
557 return (uint64_t)hs_cfifo_read(size);
558 case (uint16_t)k_ra8_usb_off_cfifoctr:
559 return (uint64_t)((uint16_t)k_ra8_fifoctr_frdy |
560 (hs_cfifo_dtln() & (uint16_t)k_ra8_fifoctr_dtln));
561 case (uint16_t)k_ra8_usb_off_brdysts:
562 return (uint64_t)hs_brdysts_read(uc);
563 case (uint16_t)k_ra8_usb_off_bempsts:
564 return (uint64_t)hs_bempsts_read();
565 case (uint16_t)k_ra8_usb_off_pipemaxp:
566 return (uint64_t)s_hs.pipe_maxp[hs_pipesel()];
567 default:
568 break;
569 }
570 if (hs_is_pipectr(off)) {
571 return (uint64_t)s_hs.pipe_pid[hs_pipectr_index(off)];
572 }
573 return (uint64_t)hs_reg((uint16_t)off);
574}

577static void hs_write(uc_engine* uc, uint64_t off, unsigned size, uint64_t value)
578{
579 const uint16_t v16 = (uint16_t)value;
580 switch ((uint16_t)off) {
581 case (uint16_t)k_ra8_usb_off_cfifo:
582 /* CFIFOH tail. */
583 case (uint16_t)((uint16_t)k_ra8_usb_off_cfifo + (uint16_t)k_usbhs_cfifo_h):
584 /* CFIFOHH tail. */
585 case (uint16_t)((uint16_t)k_ra8_usb_off_cfifo + (uint16_t)k_usbhs_cfifo_hh):
586 hs_cfifo_write((uint32_t)value, size); /* full width: HS CFIFO is 32-bit (MBW=32). */
587 return;
588 case (uint16_t)k_ra8_usb_off_cfifoctr:
589 hs_cfifoctr_write(uc, v16);
590 return;
591 case (uint16_t)k_ra8_usb_off_dcpctr:
592 hs_dcpctr_write(uc, v16);
593 return;
594 case (uint16_t)k_ra8_usb_off_dvstctr0:
595 hs_dvstctr0_write(uc, v16);
596 return;
597 case (uint16_t)k_ra8_usb_off_brdysts:
598 hs_brdysts_write(v16);
599 return;
600 case (uint16_t)k_ra8_usb_off_bempsts:
601 case (uint16_t)k_ra8_usb_off_nrdysts:
602 case (uint16_t)k_ra8_usb_off_intsts0:
603 case (uint16_t)k_ra8_usb_off_intsts1:
604 hs_set((uint16_t)off, (uint16_t)(hs_reg((uint16_t)off) & v16)); /* W0C. */
605 return;
606 case (uint16_t)k_ra8_usb_off_pipecfg:
607 s_hs.pipe_ep[hs_pipesel()] = (uint8_t)(v16 & (uint16_t)k_usbhs_epnum_mask);
608 hs_set((uint16_t)off, v16);
609 return;
610 case (uint16_t)k_ra8_usb_off_pipemaxp:
611 s_hs.pipe_maxp[hs_pipesel()] = v16;
612 return;
613 default:
614 break;
615 }
616 if (hs_is_pipectr(off)) {
617 s_hs.pipe_pid[hs_pipectr_index(off)] = v16;

638uint32_t board_usbhs_host_shadow_handoff(uint16_t* dst_words, uint32_t word_capacity)
639{
640 if (dst_words == nullptr) {
641 return 0U;
642 }
643 uint32_t n = (uint32_t)k_usbhs_words;
644 if (n > word_capacity) {
645 n = word_capacity;
646 }
647 (void)memcpy(dst_words, s_hs.reg, (size_t)n * sizeof(uint16_t));
648 /* The embedded-PHY page past the device-model span (LPSTS.SUSPENDM at 0x102)
649 * is per-WINDOW state, not per-role: it stays with this physical controller
650 * so the device firmware's PHY configuration survives the rebinding. */
651 uint16_t phy_page[k_usbhs_words] = {};
652 (void)memcpy(&phy_page[(uint32_t)k_usbhs_dev_span / 2U],
653 &s_hs.reg[(uint32_t)k_usbhs_dev_span / 2U],
654 ((size_t)k_usbhs_words - ((size_t)k_usbhs_dev_span / 2U)) * sizeof(uint16_t));
655 s_hs = (usbhs_state_t){}; /* fresh start behind the USBFS window (Config B). */
656 (void)memcpy(s_hs.reg, phy_page, sizeof(phy_page));
657 return n;
658}

680static uint64_t usbhs_block_read(uc_engine* uc, uint64_t addr, unsigned size)
681{
682 const uint64_t off = addr - (uint64_t)k_usbhs_base;
683 if ((uint16_t)off == (uint16_t)k_ra8_usbhs_off_pllsta) {
684 return (uint64_t)
685 k_usbhs_plllock; /* HUM Ch 37.2.4 "PLLSTA : PLL Status Register" p 2065 -- role-agnostic. */
686 }
687 if (board_usb_roles_swapped()) {
688 if (off >= (uint64_t)k_usbhs_dev_span) {
689 return (uint64_t)hs_reg((uint16_t)off); /* PHY page (LPSTS): local shadow. */
690 }
691 return board_usb_dev_reg_read(uc, off, size);
692 }
693 return hs_read(uc, off, size);
694}
695
710static void usbhs_block_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
711{
712 const uint64_t off = addr - (uint64_t)k_usbhs_base;
713 /* Role declaration (HUM Ch 37.2.1 SYSCFG.DPRPU, p 2060): only the device
714 * role pulls D+ up. A DPRPU=1 write to THIS instance pins it to the device
715 * model -- the FS controller must then be the loop's host (Config B). */
716 if (!board_usb_roles_swapped() && ((uint16_t)off == (uint16_t)k_ra8_usb_off_syscfg) &&
717 (((uint16_t)value & (uint16_t)(1U << k_ra8_syscfg_bit_dprpu)) != 0U)) {
718 board_usb_roles_swap(uc);
719 }
720 if (board_usb_roles_swapped()) {
721 if (off >= (uint64_t)k_usbhs_dev_span) {
722 hs_set((uint16_t)off, (uint16_t)value); /* PHY page (LPSTS): local shadow. */
723 return;
724 }
725 board_usb_dev_reg_write(uc, off, size, value);
726 return;
727 }
728 hs_write(uc, off, size, value);
729}
