765RA8_INTERNAL
766static uint16_t internal_dcp_in_drain(volatile r_usb_regs_t* reg, uint8_t* out, uint16_t max_len)
767{
768 if (max_len == 0U) {
769 return 0U;
770 }
771 /* HUM Ch 36.2.7 "CFIFOSEL" p 1976 */ /* pipe=0 (DCP), MBW=16, ISEL=1. */
772 uint16_t sel = (uint16_t)k_ra8_hhid_dcp_pipe_dcp & (uint16_t)k_ra8_fifosel_curpipe;
773 sel = (uint16_t)(sel | k_ra8_fifosel_mbw_16);
774 sel = (uint16_t)(sel | k_ra8_fifosel_isel);
775 reg->CFIFOSEL = sel;
776
777 /* HUM Ch 36.2.8 "CFIFOCTR" p 1979 */ /* bounded FRDY spin. */
778 uint16_t ready = 0U;
779 for (uint16_t i = 0U; i < k_ra8_hhid_fifo_poll_lim; ++i) { /* GCOVR_EXCL_BR_LINE */
780 if ((reg->CFIFOCTR & k_ra8_fifoctr_frdy) != 0U) { /* GCOVR_EXCL_BR_LINE */
781 ready = 1U;
782 break;
783 }
784 }
785 if (ready == 0U) {
786 return 0U;
787 }
788
789 const uint16_t available = (uint16_t)(reg->CFIFOCTR & k_ra8_fifoctr_dtln);
790 uint16_t take = (available < max_len) ? available : max_len;
791
792 /* HUM Ch 36.2.5 "CFIFO" p 1973 */ /* 16-bit LE drain. */
793 enum : uint8_t {
794 k_byte_bits = 8U,
795 k_byte_mask = 0xFFU,
796 };
797 const uint16_t even = (uint16_t)(take >> 1U);
798 for (uint16_t i = 0U; i < even; ++i) {
799 const uint16_t word = reg->CFIFO;
800 out[(2U * i) + 0U] = (uint8_t)(word & k_byte_mask);
801 out[(2U * i) + 1U] = (uint8_t)((word >> k_byte_bits) & k_byte_mask);
802 }
803 if ((take & 1U) != 0U) {
804 const uint16_t word = reg->CFIFO;
805 out[take - 1U] = (uint8_t)(word & k_byte_mask);
806 }
807 /* Release the buffer so the controller can ACK the IN data phase. */
808 reg->CFIFOCTR = k_ra8_fifoctr_bclr;
809 return take;
810}

812/* USB HID 1.11 sec 7.2.1 "Get_Report request":
813 * bmRequestType = 0xA1 (D2H | Class | Interface)
814 * bRequest = 0x01 (GET_REPORT)
815 * wValue.high = report type (1=Input / 2=Output / 3=Feature)
816 * wValue.low = report ID (0 if device uses a single unnamed report)
817 */
818ra8_err_t ra8_usb_hhid_get_report(ra8_usb_hhid_report_type_t target_report_type,
819 uint8_t target_report_id,
820 uint8_t* out_buf,
821 uint16_t max_len,
822 uint16_t* got_len)
823{
824 RA8_CHECK_NULL_PTR(out_buf, s_tag, "get_report: out_buf");
825 RA8_CHECK_NULL_PTR(got_len, s_tag, "get_report: got_len");
826 if (!s_state.initialized) {
827 return k_ra8_err_invalid_state;
828 }
829 if (!s_state.attached) {
830 return k_ra8_err_invalid_state;
831 }
832 if (!internal_report_type_ok(target_report_type)) {
833 return k_ra8_err_invalid_arg;
834 }
835 if (max_len == 0U) {
836 return k_ra8_err_invalid_arg;
837 }
838
839 const uint16_t value = (uint16_t)(((uint16_t)target_report_type << k_ra8_hhid_shift_byte1) |
840 (uint16_t)target_report_id);
841 const ra8_usb_setup_t setup = {
842 .bm_request_type = k_ra8_hhid_bm_class_iface_in,
843 .b_request = k_ra8_hhid_req_get_report,
844 .w_value = value,
845 .w_index = (uint16_t)s_state.device.interface_number,
846 .w_length = max_len,
847 };
848 *got_len = 0U;
849 const ra8_err_t setup_err = ra8_usb_host_setup_request(s_state.speed, &setup);
850 if (setup_err != k_ra8_ok) {
851 return setup_err;
852 }
853
854 /* Wire the IN data phase: after the SETUP request lands the
855 * controller drives an IN token on EP0 and the device's response
856 * ends up in the DCP CFIFO. Drain it into the caller's buffer. */
857 volatile r_usb_regs_t* reg = internal_pick_regs(s_state.speed);
858 if (reg != nullptr) {
859 *got_len = internal_dcp_in_drain(reg, out_buf, max_len);
860 }
861 return k_ra8_ok;
862}

864/* USB HID 1.11 sec 7.2.2 "Set_Report request":
865 * bmRequestType = 0x21 (H2D | Class | Interface)
866 * bRequest = 0x09 (SET_REPORT)
