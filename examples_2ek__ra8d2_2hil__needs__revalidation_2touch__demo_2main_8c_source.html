96static void td_print_uint(uint32_t value)
97{
98 uint8_t buf[k_td_dec_ten];
99 uint32_t n = 0U;
100 if (value == 0U) {
101 buf[n] = '0';
102 n++;
103 }
104 while ((value > 0U) && (n < (uint32_t)k_td_dec_ten)) {
105 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_td_dec_ten));
106 n++;
107 value /= (uint32_t)k_td_dec_ten;
108 }
109 for (uint32_t i = 0U; i < n; i++) {
110 td_print(&buf[n - 1U - i], 1U);
111 }
112}

127static uint8_t td_poll_points(uint16_t* out_x, uint16_t* out_y)
128{
129 ra8_touch_point_t pts[k_td_max_points] = {};
130 uint8_t seen = 0U;
131 *out_x = 0U;
132 *out_y = 0U;
133 for (uint32_t i = 0U; (i < (uint32_t)k_td_poll_max) && (seen == 0U); i++) {
134 uint8_t got = 0U;
135 if ((ra8_touch_read(pts, (uint8_t)k_td_max_points, &got) == k_ra8_ok) && (got >= 1U)) {
136 *out_x = pts[0].x;
137 *out_y = pts[0].y;
138 seen = got;
139 }
140 }
141 return seen;
142}

193 td_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
194 }
195 if (ra8_io_i2c_bus_as_ops(&s_touch_bus, &bus_ops) != k_ra8_ok) {
196 td_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
197 }
198
199 const ra8_touch_cfg_t cfg = {.bus = bus_ops,
200 .target_7b = (uint8_t)k_td_gt911_addr,
201 .irq_pin = (uint8_t)k_ra8_touch_irq_pin_unset,
202 .max_points = (uint8_t)k_td_max_points};

218
219 while (1) {
220 __asm__ volatile("wfi");
221 }
222}
