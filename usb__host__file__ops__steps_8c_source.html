151[[nodiscard]] ra8_err_t fileops_print_dec(uint32_t value)
152{
153 uint8_t scratch[k_fileops_dec_chars_u32] = {};
154 uint8_t out[k_fileops_dec_chars_u32] = {};
155 uint8_t count = 0U;
156 uint32_t v = value;
157 if (v == 0U) {
158 out[0] = (uint8_t)'0';
159 return fileops_sci_write(out, 1U);
160 }
161 while ((v != 0U) && (count < (uint8_t)k_fileops_dec_chars_u32)) {
162 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_fileops_dec_radix));
163 v = v / k_fileops_dec_radix;
164 count++;
165 }
166 for (uint8_t i = 0U; i < count; i++) {
167 out[i] = scratch[count - 1U - i];
168 }
169 return fileops_sci_write(out, (uint32_t)count);
170}

172[[nodiscard]] ra8_err_t fileops_print_hex(uint32_t value, uint8_t digits)
173{
174 uint8_t out[k_fileops_hex_chars_u32] = {};
175 uint8_t width = digits;
176 if (width > (uint8_t)k_fileops_hex_chars_u32) {
177 width = (uint8_t)k_fileops_hex_chars_u32;
178 }
179 for (uint8_t i = 0U; i < width; i++) {
180 const uint8_t shift = (uint8_t)((width - 1U - i) * k_fileops_nibble_bits);
181 out[i] = fileops_nibble_to_hex((value >> shift) & k_fileops_nibble_mask);
182 }
183 return fileops_sci_write(out, (uint32_t)width);
184}

486[[nodiscard]] static ra8_err_t fileops_read_back(ra8_fs_mount_t* mount, const char* path)
487{
488 static uint8_t s_read_buf[k_fileops_sector_bytes] = {};
489
490 const uint32_t payload_len = (uint32_t)(sizeof(k_fileops_payload) - 1U);
491 ra8_fs_file_t* file = nullptr;
492 ra8_err_t err = ra8_fs_open(mount, path, k_ra8_fs_mode_read, &file);
493 if (err != k_ra8_ok) {
494 (void)fileops_print_fail("open for read-back", err);
495 return err;
496 }
497 uint32_t got = 0U;
498 err = ra8_fs_read(file, s_read_buf, (uint32_t)sizeof(s_read_buf), &got);
499 (void)ra8_fs_close(file);
500 if (err != k_ra8_ok) {
501 (void)fileops_print_fail("read", err);
502 return err;
503 }
504 if (got != payload_len) {
505 (void)fileops_print_fail("read-back length", k_ra8_err_invalid_size);
506 return k_ra8_err_invalid_size;
507 }
508 for (uint32_t i = 0U; i < payload_len; i++) {
509 if (s_read_buf[i] != k_fileops_payload[i]) {
510 (void)fileops_print_fail("read-back content", k_ra8_err_invalid_state);
511 return k_ra8_err_invalid_state;
512 }
513 }
514 return k_ra8_ok;
515}
