115static void prcr_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
116{
117 (void)uc;
118 (void)addr;
119 (void)size;
120 const uint16_t word = (uint16_t)value;
121 if ((uint16_t)(word & (uint16_t)k_prcr_key_mask) != (uint16_t)k_prcr_key_value) {
122 s_prcr.bad_key++;
123 return;
124 }
125 s_prcr.groups = (uint16_t)(word & (uint16_t)k_prcr_group_mask);
126 if (s_prcr.groups != 0U) {
127 s_prcr.unlocks++;
128 }
129}

132static void prcr_report(void)
133{
134 if (s_prcr.bad_key != 0U) {
135 /* Loud: a PRCR write without the 0xA5 key unlocks nothing on silicon. */
136 (void)fprintf(stderr,
137 " SYSC-PRCR : unlocks=%u REJECTED=%u (missing 0xA5 key in PRKEY[7:0])\n",
138 s_prcr.unlocks,
139 s_prcr.bad_key);
140 return;
141 }
142 if (s_prcr.unlocks == 0U) {
143 return; /* Untouched: stay quiet. */
144 }
145 (void)fprintf(stderr, " SYSC-PRCR : unlocks=%u\n", s_prcr.unlocks);
146}
