75 void* user)
76{
77 (void)uc;
78 (void)type;
79 (void)addr;
80 (void)size;
81 (void)user;
82 const char c = (char)((uint32_t)value & 0xFFU); /* MAGIC-OK: low-byte mask */
83 if (c == '\r') {
84 return;
85 }
86 if ((c == '\n') || (s_itm_len >= (uint32_t)k_itm_line_max)) {
87 s_itm_line[s_itm_len] = '\0';
88 /* Emit one ra8_log line as `[itm] <line>` -- the CoreSight ITM/SWO-trace
89 * analog of the `[uart] SCI8:` console echo (see the ITM model block above). */
90 (void)fprintf(stdout, "[itm] %s\n", s_itm_line);
91 (void)fflush(stdout);
92 /* Also route it to the board_console ITM channel so the tabbed board-view
93 * console shows ITM/SWO trace in its own tab (a different endpoint than the
94 * UART line). The stdout echo above is unchanged -- this is purely additive. */
95 board_console_push(k_board_console_ch_itm, s_itm_line);
96 s_itm_len = 0U;
97 if (c == '\n') {
98 return;
99 }
100 }
101 s_itm_line[s_itm_len] = c;
102 s_itm_len++;
103}

162void console_tx_sink(uint8_t channel, uint8_t byte)
163{
164 if (byte == (uint8_t)'\n') {
165 console_flush_line(channel);
166 return;
167 }
168 if ((byte != (uint8_t)'\r') && (s_uart_line_len < (uint32_t)(k_uart_line_max - 1U))) {
169 s_uart_line[s_uart_line_len++] = (char)byte;
170 }
171 if (s_uart_line_len == (uint32_t)(k_uart_line_max - 1U)) {
172 console_flush_line(channel); /* avoid an unbounded line on a stream with no LF */
173 }
174}

176uint32_t decode_escapes(const char* in, uint8_t* out, uint32_t cap)
177{
178 uint32_t n = 0U;
179 for (uint32_t i = 0U; (in[i] != '\0') && (n < cap); i++) {
180 char c = in[i];
181 if ((c == '\\') && (in[i + 1U] != '\0')) {
182 i++;
183 switch (in[i]) {
184 case 'n':
185 c = '\n';
186 break;
187 case 'r':
188 c = '\r';
189 break;
190 case 't':
191 c = '\t';
192 break;
193 case '0':
194 c = '\0';
195 break;
196 default:
197 c = in[i]; /* \\ -> \, and any other escape is taken literally */
198 break;
199 }
200 }
201 out[n++] = (uint8_t)c;
202 }
203 return n;
204}
