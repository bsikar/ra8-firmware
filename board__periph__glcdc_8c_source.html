204static bool glcdc_layer_decode(const glcdc_layer_t* layer, uint8_t num, board_glcdc_fb_t* out)
205{
206 if (!layer->flmrd || !glcdc_addr_is_ram(layer->saddr)) {
207 return false;
208 }
209 const uint32_t stride =
210 (layer->flm3 >> (uint32_t)k_glcdc_stride_shift) & (uint32_t)k_glcdc_stride_mask;
211 const uint32_t lnnum =
212 (layer->flm5 >> (uint32_t)k_glcdc_lnnum_shift) & (uint32_t)k_glcdc_lnnum_mask;
213 const uint8_t fmt =
214 (uint8_t)((layer->fmt >> (uint32_t)k_glcdc_fmt_shift) & (uint32_t)k_glcdc_fmt_mask);
215 const uint32_t bpp = glcdc_bpp_for_format(fmt);
216 const uint32_t height = lnnum + 1U;
217 const uint32_t width = stride / bpp;
218 if ((stride == 0U) || (width == 0U) || (width > (uint32_t)k_glcdc_max_dim) ||
219 (height > (uint32_t)k_glcdc_max_dim)) {
220 return false;
221 }
222 out->base = layer->saddr;
223 out->width = width;
224 out->height = height;
225 out->stride = stride;
226 out->format = fmt;
227 out->layer = num;
228 out->enabled = s_glcdc.bg_on;
229 return true;
230}

246static void glcdc_snoop(uint64_t off, uint32_t value)
247{
248 switch (off) {
249 /* HUM Ch 63 "BG_EN" p 3759 */ /* background-plane operation enable. */
250 case (uint64_t)k_glcdc_off_bg_en:
251 s_glcdc.bg_on = ((value & (uint32_t)k_glcdc_bg_en_bit) != 0U);
252 break;
253
254 /* HUM Ch 63 "GR1_FLM2.BASE" p 3744 */ /* layer-1 framebuffer base address. */
255 case (uint64_t)k_glcdc_off_gr1_saddr:
256 s_glcdc.gr1.saddr = value;
257 break;
258 /* HUM Ch 63 "GR1_FLM3.LNOFF" p 3744 */ /* layer-1 line stride (bytes). */
259 case (uint64_t)k_glcdc_off_gr1_flm3:
260 s_glcdc.gr1.flm3 = value;
261 break;
262 /* HUM Ch 63 "GR1_FLM5.LNNUM" p 3744 */ /* layer-1 line count minus one. */
263 case (uint64_t)k_glcdc_off_gr1_line:
264 s_glcdc.gr1.flm5 = value;
265 break;
266 /* HUM Ch 63 "GR1_FLM6.FORMAT" p 3744 */ /* layer-1 pixel format code. */
267 case (uint64_t)k_glcdc_off_gr1_fmt:
268 s_glcdc.gr1.fmt = value;
269 break;
270 /* HUM Ch 63 "GR1_FLMRD.RENB" p 3744 */ /* layer-1 framebuffer read enable. */
271 case (uint64_t)k_glcdc_off_gr1_flmrd:
272 s_glcdc.gr1.flmrd = ((value & (uint32_t)k_glcdc_flmrd_renb_bit) != 0U);
273 break;
274
275 /* HUM Ch 63 "GR2_FLM2.BASE" p 3744 */ /* layer-2 framebuffer base address. */
276 case (uint64_t)k_glcdc_off_gr2_saddr:
277 s_glcdc.gr2.saddr = value;
278 break;
279 /* HUM Ch 63 "GR2_FLM3.LNOFF" p 3744 */ /* layer-2 line stride (bytes). */
280 case (uint64_t)k_glcdc_off_gr2_flm3:
281 s_glcdc.gr2.flm3 = value;
282 break;
283 /* HUM Ch 63 "GR2_FLM5.LNNUM" p 3744 */ /* layer-2 line count minus one. */
284 case (uint64_t)k_glcdc_off_gr2_line:
285 s_glcdc.gr2.flm5 = value;
286 break;
287 /* HUM Ch 63 "GR2_FLM6.FORMAT" p 3744 */ /* layer-2 pixel format code. */
288 case (uint64_t)k_glcdc_off_gr2_fmt:
289 s_glcdc.gr2.fmt = value;
290 break;
291 /* HUM Ch 63 "GR2_FLMRD.RENB" p 3744 */ /* layer-2 framebuffer read enable. */
292 case (uint64_t)k_glcdc_off_gr2_flmrd:
293 s_glcdc.gr2.flmrd = ((value & (uint32_t)k_glcdc_flmrd_renb_bit) != 0U);
294 break;
295
296 default:
297 break; /* Other GLCDC registers: shadowed by the caller's sparse model. */
298 }

331static bool glcdc_hash_framebuffer(const board_glcdc_fb_t* fb, uint32_t* out_crc)
332{
333 if (s_glcdc.uc == nullptr) {
334 return false;
335 }
336 const uint32_t bpp = glcdc_bpp_for_format(fb->format);
337 const uint64_t line_bytes = (uint64_t)fb->width * (uint64_t)bpp;
338 const uint64_t total = line_bytes * (uint64_t)fb->height;
339 if ((total == 0U) || (total > (uint64_t)k_glcdc_max_fb_bytes)) {
340 return false;
341 }
342 uint32_t hsh = (uint32_t)k_glcdc_fnv_offset;
343 uint8_t buf[k_glcdc_read_chunk];
344 /* Hash the visible pixels row by row (skipping any inter-line stride padding)
345 * so the witness matches a contiguous width*bpp*height framebuffer hash. */
346 for (uint32_t row = 0U; row < fb->height; ++row) {
347 const uint64_t row_base = (uint64_t)fb->base + ((uint64_t)row * (uint64_t)fb->stride);
348 uint64_t done = 0U;
349 while (done < line_bytes) {
350 uint64_t want = line_bytes - done;
351 if (want > (uint64_t)k_glcdc_read_chunk) {
352 want = (uint64_t)k_glcdc_read_chunk;
353 }
354 if (uc_mem_read(s_glcdc.uc, row_base + done, buf, (size_t)want) != UC_ERR_OK) {
355 return false;
356 }
357 for (uint64_t i = 0U; i < want; ++i) {
358 hsh = (hsh ^ (uint32_t)buf[i]) * (uint32_t)k_glcdc_fnv_prime;
359 }
360 done += want;
361 }
362 }
363 *out_crc = hsh;
364 return true;
365}

387static void glcdc_report(void)
388{
389 board_glcdc_fb_t fb = {};
390 if (!board_periph_glcdc_get_framebuffer(&fb)) {
391 return; /* No graphics layer programmed: stay quiet. */
392 }
393 uint32_t crc = 0U;
394 const bool hashed = glcdc_hash_framebuffer(&fb, &crc);
395 const char* enab = fb.enabled ? "scanning" : "idle";
396 if (hashed) {
397 (void)fprintf(stderr,
398 " GLCDC : layer%u fb=0x%08X %ux%u %s stride=%u crc=%08X (%s)\n",
399 (unsigned)fb.layer,
400 fb.base,
401 fb.width,
402 fb.height,
403 glcdc_format_name(fb.format),
404 fb.stride,
405 crc,
406 enab);
407 } else {
408 (void)fprintf(stderr,
409 " GLCDC : layer%u fb=0x%08X %ux%u %s stride=%u (%s)\n",
410 (unsigned)fb.layer,
411 fb.base,
412 fb.width,
413 fb.height,
414 glcdc_format_name(fb.format),
415 fb.stride,
416 enab);
417 }
