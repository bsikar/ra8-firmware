174int32_t main(void)
175{
176 combo_setup_or_halt();
177 ra8_isr_globals_enable();
178
179 while (1) {
180 /* RIIC ch1 -> U15: a real ACK proves the ra8_i2c controller + device. */
181 bool i2c_acked = false;
182 const ra8_err_t i2c_err =
183 ra8_i2c_scan((uint8_t)k_combo_i2c_channel, (uint8_t)k_combo_probe_addr, &i2c_acked);
184
185 /* I3C ch0 -> J27 (no device on a bare EVM); the scan result is
186 * informational -- reaching here already proves ra8_i3c_init worked. */
187 bool i3c_acked = false;
188 (void)ra8_i3c_scan((uint8_t)k_combo_i3c_channel, (uint8_t)k_combo_probe_addr, &i3c_acked);
189
190 const uint8_t* msg = k_combo_msg_err;
191 uint32_t msg_len = (uint32_t)(sizeof(k_combo_msg_err) - 1U);
192 if (i2c_err == k_ra8_ok) {
193 if (i2c_acked) {
194 msg = k_combo_msg_ok;
195 msg_len = (uint32_t)(sizeof(k_combo_msg_ok) - 1U);
196 } else {
197 msg = k_combo_msg_i2c;
198 msg_len = (uint32_t)(sizeof(k_combo_msg_i2c) - 1U);
199 }
200 }
201 if (ra8_board_uart_console_write(msg, (size_t)msg_len) != k_ra8_ok) {
202 break;
203 }
204 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
205 break;
206 }
207 ra8_delay_ms(k_combo_period_ms);
208 }
209
210 combo_panic_halt();
211 return 0;
212}
