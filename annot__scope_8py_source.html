3"""Which files ``check_annotations.py`` analyses, and what module each belongs to.
4
5Every other annotation module asks this one "is this path mine to judge?".
6Keeping that question in one place matters more than it looks: the rules
7disagree about scope on purpose -- the linkage rule judges all first-party
8code, NASA Rule 3 judges firmware only, and RA8_PRIV is defined over module
9boundaries -- and each of those predicates has been wrong at least once by
10being restated at a call site instead of asked for here.
11
12The repo root is a variable, not a constant
13-------------------------------------------
14``run_selftest()`` builds a synthetic tree in a temporary directory and needs
15every scope predicate to resolve against *that* root rather than the real
16checkout. It used to do this by rebinding a module-global ``REPO_ROOT`` in
17``check_annotations``. That worked only while every predicate lived in the
18same module: once the checker is split, a rebinding in one module is invisible
19to the ``from ... import REPO_ROOT`` copies in the others, and the selftest
20would quietly assert against the real tree -- passing, but proving nothing.
21
22So the root lives here, behind :func:`repo_root`, and is overridden through
23:func:`override_repo_root`. Consumers call the function; there is no name to
24import a stale copy of.
25"""
26
27from __future__ import annotations

29import contextlib
30import pathlib
31from collections.abc import Iterator

33#: The real checkout root. Never read directly outside `repo_root()` -- the
34#: selftest override would not be visible through a bare import of it.
35_REAL_REPO_ROOT = pathlib.Path(__file__).resolve().parents[2]
36
37#: Active root; `override_repo_root()` swaps this for the duration of a block.
38_repo_root = _REAL_REPO_ROOT
39
40#: Every first-party source root. CLAUDE.md ("Scope") holds `tools/` to the
41#: same bar as the firmware -- "a file being a host tool or just an emulator
42#: is NOT a reason to relax the rules" -- but `tools/` was absent here, so
43#: ra8_emulator, media_dl, ra8_viewer and the rest were never annotation-checked
44#: at all. `scripts/` holds no C. Vendored SOUP under `libs/third_party/` is
45#: dropped by is_excluded(), not by omission from this tuple.
46SCAN_DIRS = ("libs", "src", "examples", "tests", "port", "tools")
47
48EXCLUDED_PATH_PARTS = {
49 "build",
50 "_deps",
51 "third_party",
52 "build-cov",
53 "build-bench",
54 "build-scan",
55 "build-mcdc",
56}
57
58SOURCE_SUFFIXES = {".c", ".cpp"}
59
60#: Directory names that only ever hold build output. Distinct from
61#: EXCLUDED_PATH_PARTS, which also drops vendored trees: those must stay
62#: off the *analysis* list but stay on the *include* path.
63BUILD_OUTPUT_PARTS = frozenset(
64 {"build", "_deps", "build-cov", "build-bench", "build-scan", "build-mcdc"}
65)
66
67#: Source roots that are compiled by the host toolchain and never cross-compiled
68#: into a firmware image. `tests/` is the host unit-test suite; `tools/` is the
69#: host-side tooling (ra8_emulator, media_dl, ra8_fmt, ...), which `scripts/ci.sh`
70#: builds with ``CC=clang-18`` and which no `cmake/toolchain-ra8d2.cmake` target
71#: ever references.
72HOST_ONLY_ROOTS = frozenset({"tests", "tools"})
73
74#: Roots whose immediate child directory is one module for RA8_PRIV purposes.
75#: `libs/<module>` is the obvious one. `tools/<tool>` is the same shape: each
76#: tool is one module split across several TUs with its own `*_internal.h`
77#: (ra8_fmt says so in that header's own file comment), and ra8_emulator calling
78#: ra8_fmt's private helper is the same boundary violation as one library
79#: calling another's. Without this, an RA8_PRIV tag under tools/ is decorative
80#: -- module_of() returned None, the rule hit `if callee_mod is None: continue`
81#: and never compared anything.
82MODULE_ROOTS = ("libs", "tools")

85def repo_root() -> pathlib.Path:
86 """Return the root every scope predicate resolves against."""
87 return _repo_root

90@contextlib.contextmanager
91def override_repo_root(root: pathlib.Path) -> Iterator[None]:
92 """Resolve scope against ``root`` for the duration of the block.
93
94 Used only by the selftest, which parses a synthetic tree in a temporary
95 directory. Restores the previous root even when the body raises, so a
96 failing assertion cannot leave the process judging the real tree against
97 a directory that no longer exists.
98 """
99 global _repo_root # noqa: PLW0603 # single-homed override; see module docstring
100 previous = _repo_root
101 _repo_root = root
102 try:
103 yield
104 finally:
105 _repo_root = previous

108def is_build_output(path: pathlib.Path) -> bool:
109 """True when ``path`` sits inside a build-output directory."""
110 return any(part in BUILD_OUTPUT_PARTS for part in path.parts)

113def is_excluded(path: pathlib.Path) -> bool:
114 """True when ``path`` must not be analysed (build output or vendored)."""
115 return any(part in EXCLUDED_PATH_PARTS for part in path.parts)

118def _root_part(path: str) -> str | None:
119 """Return ``path``'s first repo-relative path component, or None."""
120 if not path:
121 return None
122 try:
123 rel = pathlib.Path(path).resolve().relative_to(repo_root())
124 except (ValueError, OSError):
125 return None
126 if not rel.parts or is_excluded(rel):
127 return None
128 return rel.parts[0]

131def is_first_party(path: str) -> bool:
132 """True when ``path`` is hand-written source this project owns.
133
134 Definitions reached through the include path are not automatically in
135 scope. Parsing the ``.cpp`` translation units as C++ pulls in
136 libstdc++, whose headers define hundreds of non-static inline
137 functions; vendored trees under ``libs/third_party/`` are SOUP. Only
138 files under the scan roots are ours to hold to the linkage rule.
139 """
140 return _root_part(path) in SCAN_DIRS

143def is_test_path(path: str) -> bool:
144 """True when ``path`` is a host unit-test translation unit."""
145 return "/tests/" in path.replace("\\", "/")

148def is_host_only_path(path: str) -> bool:
149 """True when ``path`` is host-only code, i.e. not part of any firmware image.
150
151 NASA Power of 10 Rule 3 is a claim about *firmware*: CLAUDE.md states it
152 as "zero dynamic memory after initialization (zero malloc/free in
153 firmware)". The hazard it guards -- heap fragmentation and an allocator
154 failing unpredictably in a long-running image with no operator -- does not
155 exist for a host program that runs for a moment on Linux and exits.
156
157 So the rule's real question is "is this translation unit firmware", and
158 this predicate is where that gets decided. It used to be decided by a bare
159 ``"/tests/" in path`` substring at the one call site that needed it. That
160 was the right intent expressed too narrowly: when `tools/` came into scope
161 it added 216 findings telling a CPU emulator and a libcurl downloader not
162 to call ``malloc``, none of which a developer can act on. A gate that
163 cries wolf gets switched off, so the predicate is stated in terms of what
164 actually distinguishes the code -- which toolchain compiles it -- and is
165 matched on the repo-relative root rather than by substring, so a directory
166 named ``tests`` nested anywhere else cannot silently claim the exemption.
167
168 This narrows nothing for firmware: `libs/`, `src/`, `port/` and
169 `examples/` are all still held to Rule 3, and in fact carry zero
170 ``RA8_NASA_RULE_3_OK`` waivers tree-wide because the firmware genuinely
171 does not allocate.
172 """
173 return _root_part(path) in HOST_ONLY_ROOTS

176def module_of(path: str) -> str | None:
177 """libs/<module>/... or tools/<tool>/... -> the module name; else None."""
178 parts = pathlib.Path(path).parts
179 for root in MODULE_ROOTS:
180 try:
181 idx = parts.index(root)
182 except ValueError:
183 continue
184 if idx + 1 < len(parts):
185 return f"{root}/{parts[idx + 1]}"
186 return None

189def relative(path: str) -> str:
190 """Return ``path`` relative to the repo root when it lies inside it."""
191 with contextlib.suppress(ValueError):
192 return str(pathlib.Path(path).resolve().relative_to(repo_root()))
193 return path

196def discover_translation_units() -> list[pathlib.Path]:
197 """Return every .c/.cpp file under SCAN_DIRS, excluding vendored trees."""
198 out: list[pathlib.Path] = []
199 for top in SCAN_DIRS:
200 root = repo_root() / top
201 if not root.is_dir():
202 continue
203 out.extend(
204 path
205 for path in root.rglob("*")
206 if path.suffix in SOURCE_SUFFIXES and not is_excluded(path)
207 )
208 return sorted(out)
