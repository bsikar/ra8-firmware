154RA8_INTERNAL
155static ra8_err_t internal_enum_hunt(uint8_t* desc, uint8_t* out_addr)
156{
157 ra8_delay_ms(k_ra8_hmsc_vbus_settle_ms);
158 const uint32_t t0 = ra8_time_ms();
159 for (uint32_t spin = 0U; spin < (uint32_t)k_ra8_hmsc_attach_spin_limit; spin++) {
160 if (ra8_usb_host_line_state(s_usb_hmsc_state.speed) != 0U) {
161 break;
162 }
163 if ((ra8_time_ms() - t0) > (uint32_t)k_ra8_hmsc_attach_to_ms) {
164 break;
165 }
166 }
167 ra8_delay_ms(k_ra8_hmsc_debounce_ms);
168 ra8_err_t err = k_ra8_err_hw_timeout;
169 for (uint8_t attempt = 0U; attempt < (uint8_t)k_ra8_hmsc_enum_tries; attempt++) {
170 if (attempt >= (uint8_t)k_ra8_hmsc_no_reset_tries) {
171 (void)ra8_usb_host_bus_reset(s_usb_hmsc_state.speed, true);
172 ra8_delay_ms(k_ra8_hmsc_reset_hold_ms);
173 (void)ra8_usb_host_bus_reset(s_usb_hmsc_state.speed, false);
174 }
175 (void)ra8_usb_host_set_uact(s_usb_hmsc_state.speed, true);
176 ra8_delay_ms(k_ra8_hmsc_recovery_ms);
177 const uint8_t addr = (uint8_t)(attempt & (uint8_t)k_ra8_hmsc_addr_alt_mask);
178 (void)ra8_usb_host_set_target(s_usb_hmsc_state.speed, addr);
179 err = internal_enum_read_dev_desc(desc);
180 if (err == k_ra8_ok) {
181 *out_addr = addr;
182 return k_ra8_ok;
183 }
184 }
185 return err;
186}

318static ra8_err_t internal_enum_walk_cfg(const uint8_t* cfg, uint16_t len)
319{
320 uint16_t off = 0U;
321 bool in_msc = false;
322 while (off < len) {
323 const uint8_t dlen = cfg[off + (uint16_t)k_ra8_hmsc_off_dlen];
324 if (dlen == 0U) {
325 break;
326 }
327 const uint8_t dtype = cfg[off + (uint16_t)k_ra8_hmsc_off_dtype];
328 if (dtype == (uint8_t)k_ra8_hmsc_desc_interface) {
329 in_msc = internal_enum_iface_is_msc(&cfg[off]);
330 if (in_msc) {
331 s_usb_hmsc_state.device.interface_number = cfg[off + (uint16_t)k_ra8_hmsc_off_iface_num];
332 }
333 }
334 if (dtype == (uint8_t)k_ra8_hmsc_desc_endpoint) {
335 if (in_msc) {
336 internal_enum_note_endpoint(&cfg[off]);
337 }
338 }
339 off = (uint16_t)(off + dlen);
340 }
341 if (s_usb_hmsc_state.device.bulk_in_ep == 0U) {
342 return k_ra8_err_hw_error;
343 }
344 if (s_usb_hmsc_state.device.bulk_out_ep == 0U) {
345 return k_ra8_err_hw_error;
346 }
347 return k_ra8_ok;
348}
