264 const uint8_t* mram = (const uint8_t*)(uintptr_t)((uint32_t)k_mram_base_addr + offset);
265 if (memcmp(s_burst, mram, (size_t)k_selftest_burst_bytes) != 0) {
266 s_dbg_mismatch_off = offset;
267 (void)selftest_print_fail("content mismatch at 0x", (ra8_err_t)offset);
268 return k_ra8_err_invalid_state;
269 }
270 s_dbg_verified_bytes = offset + (uint32_t)k_selftest_burst_bytes;
271 }
272 s_dbg_verify_ms = ra8_time_ms() - t0;

517[[nodiscard]] ra8_err_t selftest_host_pass(void)
518{
519 ra8_usb_hmsc_device_t device = {};
520 ra8_err_t err = selftest_host_enumerate(&device);
521 if (err != k_ra8_ok) {
522 return err;
523 }
524 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
525 ra8_fs_mount_t* mount = nullptr;
526 err = selftest_mount_volume(&mount);
527 if (err != k_ra8_ok) {
528 (void)ra8_usb_hmsc_close();
529 return err;
530 }
531 (void)ra8_fs_unmount(mount);
532
533 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
534 uint64_t total_bytes = 0U;
535 uint64_t total_ms = 0U;
536 for (uint32_t it = 0U; it < (uint32_t)k_selftest_soak_iters; it++) {
537 err = selftest_verify_mram_raw();
538 if (err != k_ra8_ok) {
539 (void)ra8_usb_hmsc_close();
540 return err;
541 }
542 total_bytes += (uint64_t)s_dbg_verified_bytes;
543 total_ms += (uint64_t)s_dbg_verify_ms;
544 s_dbg_pass_count = it + 1U;
545 }
546
547 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
548 err = selftest_write_protect_probe();
549 if (err == k_ra8_ok) {
550 err = selftest_print_soak_verdict(total_bytes, total_ms);
551 }
552 if (err != k_ra8_ok) {
553 (void)ra8_usb_hmsc_close();
554 return err;
555 }
556 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
557 err = selftest_print("ra8d2 soak: USB SELFTEST SOAK PASS\r\n");
558 if (err != k_ra8_ok) {
559 return err;
560 }
561 (void)ra8_board_led_on(k_ra8_board_led2);
