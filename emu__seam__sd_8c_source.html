25/* =============================================================================
26 * --fast-sd: serve whole SD blocks in C instead of clocking 512 SPI bytes each.
27 * =============================================================================
28 *
29 * The firmware reads an SD card over SCI0 in Simple-SPI mode one byte at a time:
30 * ra8_sci_spi_xfer8() issues a FIXED five MMIO accesses per byte (poll TDRE, write
31 * TDR, poll RDRF, read RDR, clear RDRF), and a 512-byte block is 512 of those. A
32 * book-sized read is millions of MMIO callbacks -- tens of wall-clock seconds in
33 * Unicorn (QEMU-TCG) -- versus ~0.1 s on the real 25 MHz bus, so the cost is a
34 * pure emulation artifact, not firmware behaviour.
35 *
36 * The right granularity to skip is the BLOCK, not the byte: redirecting a hooked
37 * function back to its caller needs a uc_emu_stop/relaunch, which is far more
38 * expensive than an MMIO callback, so a per-byte hook is a net loss (millions of
39 * relaunches). Opt-in --fast-sd instead installs a UC_HOOK_CODE at the entry of
40 * ra8_sdmmc_spi_read_block(lba, buf): with a card attached it copies the 512-byte
41 * block straight from the image via ::board_sd_read_block -- the byte-identical
42 * data the CMD17 path would stream -- writes it to @c buf, returns k_ra8_ok and
43 * jumps to LR. One relaunch per sector (hundreds per book) instead of millions
44 * per byte, so the rendered framebuffer is byte-for-byte identical while the read
45 * drops from tens of seconds to well under one.
46 *
47 * Faithfulness: this serves block DATA directly, bypassing the SD-over-SPI block
48 * protocol (CMD17 / 0xFE token / CRC16 / the per-byte SPI loop) for the data
49 * path. The FAT parse, inflate, and render all still run on the real firmware and
50 * see identical bytes; the bypassed block protocol is covered by the host unit
51 * test (tests/test_ra8_sdmmc_card_reflow.c) and by hardware. It is OFF by default:
52 * HIL gates and the default run exercise the full handshake; turn it on only to
53 * load a large book fast for an interactive or recorded capture. A card-absent
54 * call, an out-of-range LBA, or a firmware without the symbol falls through to
55 * the real driver, so nothing else is affected.
56 */

98static void on_sdmmc_read_block(uc_engine* uc, uint64_t address, uint32_t size, void* user)
99{
100 (void)address;
101 (void)size;
102 (void)user;
103 uint32_t lba = 0U;
104 uint32_t buf_ptr = 0U;
105 (void)uc_reg_read(uc, UC_ARM_REG_R0, &lba);
106 (void)uc_reg_read(uc, UC_ARM_REG_R1, &buf_ptr);
107 uint8_t blk[k_fast_sd_block];
108 /* No card, a null buffer, or an out-of-range LBA -> run the real driver so the
109 * card-absent / error path stays exactly as on hardware. */
110 if (!board_sd_attached() || (buf_ptr == 0U) || !board_sd_read_block(lba, blk)) {
111 return;
112 }
113 (void)uc_mem_write(uc, (uint64_t)buf_ptr, blk, sizeof blk);
114 /* Relaunch as a zero-time seam (no chunk / SysTick cost) so a whole book-sized
115 * read drains within one settle window. */
116 emu_seam_request_relaunch();
117 eth_hook_return(uc, 0U); /* k_ra8_ok: set r0 + jump to LR, skipping the body. */
118}
