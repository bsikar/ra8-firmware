69RA8_PRIV ra8_err_t ra8_jpeg_sw_priv_parse_dqt(ra8_jpeg_dec_ctx_t* d)
70{
71 if (d->cursor + 2U > d->src_len) {
72 return k_ra8_err_protocol_error;
73 }
74 uint16_t len = read_be16(&d->src[d->cursor]);
75 if (len < 2U || (uint32_t)len > d->src_len - d->cursor) {
76 return k_ra8_err_protocol_error;
77 }
78 uint32_t end = d->cursor + len;
79 d->cursor += 2U;
80 while (d->cursor < end) {
81 uint8_t pq_tq = d->src[d->cursor];
82 d->cursor++;
83 uint8_t pq = (uint8_t)(pq_tq >> k_ra8_jpeg_nibble_shift);
84 uint8_t tq = (uint8_t)(pq_tq & k_ra8_jpeg_nibble_mask);
85 if (tq >= (uint8_t)k_ra8_jpeg_quant_tabs || pq != 0U) {
86 return k_ra8_err_not_supported; /* 16-bit precision not supported. */
87 }
88 if (d->cursor + (uint32_t)k_ra8_jpeg_block_size > end) {
89 return k_ra8_err_protocol_error;
90 }
91 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_block_size; i++) {
92 d->qtab[tq][s_zigzag[i]] = d->src[d->cursor];
93 d->cursor++;
94 }
95 }
96 return k_ra8_ok;
97}

125static ra8_err_t dec_parse_dht_one(ra8_jpeg_dec_ctx_t* d, uint32_t end)
126{
127 uint8_t tc_th = d->src[d->cursor];
128 d->cursor++;
129 uint8_t tc = (uint8_t)(tc_th >> k_ra8_jpeg_nibble_shift);
130 uint8_t th = (uint8_t)(tc_th & k_ra8_jpeg_nibble_mask);
131 if (tc >= (uint8_t)k_ra8_jpeg_huff_classes || th >= (uint8_t)k_ra8_jpeg_huff_ids) {
132 return k_ra8_err_not_supported;
133 }
134 if (d->cursor + (uint32_t)k_ra8_jpeg_huff_lengths > end) {
135 return k_ra8_err_protocol_error;
136 }
137 ra8_jpeg_htab_t* h = (tc == 0U) ? &d->hdc[th] : &d->hac[th];
138 uint16_t total = 0U;
139 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_huff_lengths; i++) {
140 h->bits[i] = d->src[d->cursor];
141 d->cursor++;
142 total = (uint16_t)(total + h->bits[i]);
143 }
144 if (total > (uint16_t)k_ra8_jpeg_huff_max) {
145 return k_ra8_err_protocol_error;
146 }
147 if (d->cursor + total > end) {
148 return k_ra8_err_protocol_error;
149 }
150 for (uint16_t i = 0U; i < total; i++) {
151 h->vals[i] = d->src[d->cursor];
152 d->cursor++;
153 }
154 ra8_jpeg_sw_htab_build(h);
155 return k_ra8_ok;
156}

199static void dec_parse_sof0_components(ra8_jpeg_dec_ctx_t* d, uint32_t* s)
200{
201 d->hmax = 0U;
202 d->vmax = 0U;
203 for (uint8_t i = 0U; i < d->ncomp; i++) {
204 d->comp_id[i] = d->src[*s];
205 (*s)++;
206 uint8_t hv = d->src[*s];
207 (*s)++;
208 d->comp_h[i] = (uint8_t)(hv >> k_ra8_jpeg_nibble_shift);
209 d->comp_v[i] = (uint8_t)(hv & k_ra8_jpeg_nibble_mask);
210 d->comp_qid[i] = d->src[*s];
211 (*s)++;
212 if (d->comp_h[i] > d->hmax) {
213 d->hmax = d->comp_h[i];
214 }
215 if (d->comp_v[i] > d->vmax) {
216 d->vmax = d->comp_v[i];
217 }
218 }

244static ra8_err_t dec_check_chroma_layout(const ra8_jpeg_dec_ctx_t* d)
245{
246 /* Only 4:4:4 (1,1,1) and 4:2:0 (2,1,1) accepted. */
247 if (d->ncomp == 3U) {
248 bool is_444 = (d->hmax == 1U && d->vmax == 1U);
249 bool is_420 = (d->hmax == 2U && d->vmax == 2U && d->comp_h[1] == 1U && d->comp_v[1] == 1U &&
250 d->comp_h[2] == 1U && d->comp_v[2] == 1U);
251 if (!is_444 && !is_420) {
252 return k_ra8_err_not_supported;
253 }
254 }
255 return k_ra8_ok;

259RA8_PRIV ra8_err_t ra8_jpeg_sw_priv_parse_sof0(ra8_jpeg_dec_ctx_t* d)
260{
261 if (d->cursor + 2U > d->src_len) {
262 return k_ra8_err_protocol_error;
263 }
264 uint16_t len = read_be16(&d->src[d->cursor]);
265 if (len < 8U || (uint32_t)len > d->src_len - d->cursor) {
266 return k_ra8_err_protocol_error;
267 }
268 uint32_t s = d->cursor + 2U;
269 uint8_t precision = d->src[s];
270 s++;
271 if (precision != 8U) {
272 return k_ra8_err_not_supported;
273 }
274 d->height = read_be16(&d->src[s]);
275 s += 2U;
276 d->width = read_be16(&d->src[s]);
277 s += 2U;
278 d->ncomp = d->src[s];
279 s++;
280 if (d->ncomp != 1U && d->ncomp != 3U) {
281 return k_ra8_err_not_supported;
282 }
283 if (((uint32_t)d->ncomp * 3U) + (s - d->cursor) > (uint32_t)len) {
284 return k_ra8_err_protocol_error;
285 }
286 dec_parse_sof0_components(d, &s);
287 d->cursor += len;
288 return dec_check_chroma_layout(d);

292RA8_PRIV ra8_err_t ra8_jpeg_sw_priv_parse_sos(ra8_jpeg_dec_ctx_t* d)
293{
294 if (d->cursor + 2U > d->src_len) {
295 return k_ra8_err_protocol_error;
296 }
297 uint16_t len = read_be16(&d->src[d->cursor]);
298 if (len < 6U || (uint32_t)len > d->src_len - d->cursor) {
299 return k_ra8_err_protocol_error;
300 }
301 uint32_t s = d->cursor + 2U;
302 uint8_t ns = d->src[s];
303 s++;
304 if (ns != d->ncomp) {
305 return k_ra8_err_not_supported;
306 }
307 for (uint8_t i = 0U; i < ns; i++) {
308 uint8_t cs = d->src[s];
309 s++;
310 uint8_t tdta = d->src[s];
311 s++;
312 uint8_t idx = i;
313 for (uint8_t j = 0U; j < d->ncomp; j++) {
314 if (d->comp_id[j] == cs) {
315 idx = j;
316 break;
317 }
318 }
319 d->comp_dc_id[idx] = (uint8_t)(tdta >> k_ra8_jpeg_nibble_shift);
320 d->comp_ac_id[idx] = (uint8_t)(tdta & k_ra8_jpeg_nibble_mask);
321 if (d->comp_dc_id[idx] >= (uint8_t)k_ra8_jpeg_huff_ids ||
322 d->comp_ac_id[idx] >= (uint8_t)k_ra8_jpeg_huff_ids) {
323 return k_ra8_err_not_supported;
324 }
325 }
326 /* Skip Ss/Se/AhAl (3 bytes). */
327 d->cursor += len;
328 return k_ra8_ok;
329}

359static ra8_err_t
360dec_block_ac(ra8_jpeg_dec_ctx_t* d, ra8_jpeg_bitreader_t* br, uint8_t ci, int32_t* outblk)
361{
362 uint8_t k = 1U;
363 while (k < (uint8_t)k_ra8_jpeg_block_size) {
364 int32_t rs = ra8_jpeg_sw_htab_decode(br, &d->hac[d->comp_ac_id[ci]]);
365 if (rs < 0) {
366 return k_ra8_err_protocol_error;
367 }
368 uint8_t rrrr = (uint8_t)(rs >> k_ra8_jpeg_nibble_shift);
369 uint8_t ssss = (uint8_t)(rs & k_ra8_jpeg_nibble_mask);
370 if (ssss == 0U) {
371 if (rrrr == k_jpeg_nibble_mask) {
372 k = (uint8_t)(k + 16U); /* ZRL: 16 zero coeffs. */
373 continue;
374 }
375 break; /* EOB. */
376 }
377 k = (uint8_t)(k + rrrr);
378 if (k >= (uint8_t)k_ra8_jpeg_block_size) {
379 return k_ra8_err_protocol_error;
380 }
381 int32_t v = ra8_jpeg_sw_br_get_bits(br, ssss);
382 if (v < 0) {
383 return k_ra8_err_protocol_error;
384 }
385 v = ra8_jpeg_sw_huff_extend(v, ssss);
386 outblk[s_zigzag[k]] = v * (int32_t)d->qtab[d->comp_qid[ci]][s_zigzag[k]];
387 k++;
388 }
389 return k_ra8_ok;
390}

393RA8_PRIV ra8_err_t ra8_jpeg_sw_priv_block(ra8_jpeg_dec_ctx_t* d,
394 ra8_jpeg_bitreader_t* br,
395 uint8_t ci,
396 int32_t* outblk)
397{
398 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_block_size; i++) {
399 outblk[i] = 0;
400 }
401 /* DC. */
402 int32_t t = ra8_jpeg_sw_htab_decode(br, &d->hdc[d->comp_dc_id[ci]]);
403 if (t < 0) {
404 return k_ra8_err_protocol_error;
405 }
406 int32_t r = ra8_jpeg_sw_br_get_bits(br, (uint8_t)t);
407 /* mcdc-deactivated: ra8_jpeg_sw_br_get_bits returns -1 only when the bitstream
408 * is exhausted; reaching this branch with t != 0 requires a
409 * truncated entropy-coded segment after every parser stage has
410 * succeeded -- the public-API contract (well-formed JFIF stream
411 * with EOI) makes this branch unreachable. Defensive guard for
412 * fault injection. */
413 // mcdc-deactivated: ra8_jpeg_sw_priv_block ra8_jpeg_sw_br_get_bits exhaustion guard; well-formed JFIF streams (public-API contract) cannot exhaust the bitstream mid-coefficient with t != 0 because every parser stage upstream has validated the segment-length budget.
414 if (r < 0 && t != 0) {
415 return k_ra8_err_protocol_error;
416 }
417 int32_t diff = ra8_jpeg_sw_huff_extend(r, (uint8_t)t);
418 d->comp_dc_pred[ci] += diff;
419 outblk[0] = d->comp_dc_pred[ci] * (int32_t)d->qtab[d->comp_qid[ci]][0];
420
421 /* AC. */
422 return dec_block_ac(d, br, ci, outblk);
423}

701static ra8_err_t dec_decode_scan(ra8_jpeg_dec_ctx_t* d, uint8_t* out_buf, uint32_t out_buf_len)
702{
703 uint32_t need = (uint32_t)d->width * (uint32_t)d->height * (uint32_t)k_ra8_jpeg_rgb_components;
704 if (out_buf_len < need) {
705 return k_ra8_err_invalid_size;
706 }
707
708 ra8_jpeg_bitreader_t br;
709 dec_scan_begin(d, &br);
710
711 /*
712 * Cross-function invariant: ra8_jpeg_sw_priv_parse_sof0() only returns
713 * success for the 4:4:4 and 4:2:0 chroma layouts (and grayscale), all
714 * of which set hmax and vmax to a non-zero value.
715 * The clang static analyzer (scan-build, core.DivideZero) cannot follow
716 * the cross-function invariant, so re-state it here as an assertion to
717 * make the property locally provable to the analyzer and to NASA Power-
718 * of-10 Rule 5 readers.
719 */
720 assert(d->hmax > 0U);
721 assert(d->vmax > 0U);
722
723 uint16_t mcu_w_px = (uint16_t)((uint16_t)k_ra8_jpeg_block_dim * d->hmax);
724 uint16_t mcu_h_px = (uint16_t)((uint16_t)k_ra8_jpeg_block_dim * d->vmax);
725 uint16_t mcus_x = (uint16_t)(((d->width + mcu_w_px) - 1U) / mcu_w_px);
726 uint16_t mcus_y = (uint16_t)(((d->height + mcu_h_px) - 1U) / mcu_h_px);
727
728 uint8_t y_tile[(uint32_t)k_ra8_jpeg_mcu_max_dim * (uint32_t)k_ra8_jpeg_mcu_max_dim];
729 uint8_t cb_tile[(uint32_t)k_ra8_jpeg_block_size];
730 uint8_t cr_tile[(uint32_t)k_ra8_jpeg_block_size];
731
732 for (uint16_t my = 0U; my < mcus_y; my++) {
733 for (uint16_t mx = 0U; mx < mcus_x; mx++) {
734 ra8_err_t e =
735 dec_decode_mcu(d, &br, y_tile, cb_tile, cr_tile, mx, my, mcu_w_px, mcu_h_px, out_buf);
736 if (e != k_ra8_ok) {
737 return e;
738 }
739 }
740 }
741 d->cursor = br.pos;
742 return k_ra8_ok;
743}

772static ra8_err_t dec_dispatch_tail(ra8_jpeg_dec_ctx_t* d,
773 uint16_t mk,
774 bool got_sof,
775 ra8_jpeg_dec_marker_action_t* action)
776{
777 if (mk == (uint16_t)k_ra8_jpeg_marker_dqt) {
778 return ra8_jpeg_sw_priv_parse_dqt(d);
779 }
780 if (mk == (uint16_t)k_ra8_jpeg_marker_dht) {
781 return ra8_jpeg_sw_priv_parse_dht(d);
782 }
783 if (mk == (uint16_t)k_ra8_jpeg_marker_sos) {
784 if (!got_sof) {
785 return k_ra8_err_protocol_error;
786 }
787 ra8_err_t e = ra8_jpeg_sw_priv_parse_sos(d);
788 if (e != k_ra8_ok) {
789 return e;
790 }
791 *action = k_ra8_jpeg_dec_scan;
792 return k_ra8_ok;
793 }
794 if (mk == (uint16_t)k_ra8_jpeg_marker_eoi) {
795 *action = k_ra8_jpeg_dec_eoi;
796 return k_ra8_ok;
797 }
798 if (mk >= (uint16_t)k_ra8_jpeg_marker_rst0 && mk <= (uint16_t)k_ra8_jpeg_marker_rst7) {
799 /* Standalone marker, no length. */
800 return k_ra8_ok;
801 }
802 return ra8_jpeg_sw_priv_skip_segment(d);
803}

806RA8_PRIV ra8_err_t ra8_jpeg_sw_priv_dispatch(ra8_jpeg_dec_ctx_t* d,
807 bool* got_sof,
808 ra8_jpeg_dec_marker_action_t* action)
809{
810 *action = k_ra8_jpeg_dec_continue;
811 if (d->src[d->cursor] != (uint8_t)k_ra8_jpeg_marker_byte) {
812 return k_ra8_err_protocol_error;
813 }
814 while (d->cursor < d->src_len && d->src[d->cursor] == (uint8_t)k_ra8_jpeg_marker_byte) {
815 d->cursor++;
816 }
817 if (d->cursor >= d->src_len) {
818 return k_ra8_err_protocol_error;
819 }
820 uint8_t m = d->src[d->cursor];
821 d->cursor++;
822 uint16_t mk = (uint16_t)(k_jpeg_marker_ff00 | m);
823
824 if (mk == (uint16_t)k_ra8_jpeg_marker_sof0) {
825 ra8_err_t e = ra8_jpeg_sw_priv_parse_sof0(d);
826 if (e != k_ra8_ok) {
827 return e;
828 }
829 *got_sof = true;
830 return k_ra8_ok;
831 // mcdc-deactivated: ra8_jpeg_sw_priv_dispatch unsupported-SOFn detector (SOF1..SOFF excluding DHT/SOF8); identical co-dependence rationale as the SOF0 detector decision in ra8_jpeg_sw_get_dimensions -- markers >= 0xFFC1 in the JPEG spec are always <= 0xFFCF.
832 }
833 if (mk >= k_jpeg_marker_sof1 && mk <= k_jpeg_marker_sof_hi &&
834 mk != (uint16_t)k_ra8_jpeg_marker_dht && mk != k_jpeg_marker_jpg) {
835 return k_ra8_err_not_supported;
836 }
837 return dec_dispatch_tail(d, mk, *got_sof, action);
838}

868static ra8_err_t dec_run(ra8_jpeg_dec_ctx_t* d,
869 uint8_t* out_buf,
870 uint32_t out_buf_len,
871 uint16_t* out_w,
872 uint16_t* out_h)
873{
874 bool got_sof = false;
875 while (d->cursor < d->src_len) {
876 ra8_jpeg_dec_marker_action_t action = k_ra8_jpeg_dec_continue;
877 ra8_err_t e = ra8_jpeg_sw_priv_dispatch(d, &got_sof, &action);
878 if (e != k_ra8_ok) {
879 return e;
880 }
881 if (action == k_ra8_jpeg_dec_scan) {
882 *out_w = d->width;
883 *out_h = d->height;
884 return dec_decode_scan(d, out_buf, out_buf_len);
885 }
886 if (action == k_ra8_jpeg_dec_eoi) {
887 break;
888 }
889 }
890 return k_ra8_err_protocol_error;
891}

893ra8_err_t ra8_jpeg_sw_decode(const uint8_t* jpeg_buf,
894 uint32_t jpeg_len,
895 uint8_t* out_buf,
896 uint32_t out_buf_len,
897 uint16_t* out_w,
898 uint16_t* out_h)
899{
900 RA8_CHECK_NULL_PTR(jpeg_buf, s_tag, "jpeg_buf is NULL");
901 RA8_CHECK_NULL_PTR(out_buf, s_tag, "out_buf is NULL");
902 RA8_CHECK_NULL_PTR(out_w, s_tag, "out_w is NULL");
903 RA8_CHECK_NULL_PTR(out_h, s_tag, "out_h is NULL");
904 if (jpeg_len < 4U) {
905 return k_ra8_err_invalid_size;
906 }
907
908 /* Decoder context is large (4 Huffman tables + 2 quant tables);
909 * allocate as `static` so it doesn't blow the stack budget the
910 * project enforces via `-Wstack-usage`. The codec is documented
911 * as not thread-safe, so the static is fine. */
912 static ra8_jpeg_dec_ctx_t s_d;
913 ra8_jpeg_dec_ctx_t* d = &s_d;
914 memset(d, 0, sizeof(*d));
915 d->src = jpeg_buf;
916 d->src_len = jpeg_len;
917
918 if (read_be16(jpeg_buf) != (uint16_t)k_ra8_jpeg_marker_soi) {
919 return k_ra8_err_protocol_error;
920 }
921 d->cursor = 2U;
922
923 return dec_run(d, out_buf, out_buf_len, out_w, out_h);
924}
