180[[nodiscard]] static ra8_err_t selftest_print_dec(uint32_t value)
181{
182 uint8_t scratch[k_selftest_dec_chars_u32] = {};
183 uint8_t out[k_selftest_dec_chars_u32] = {};
184 uint8_t count = 0U;
185 uint32_t v = value;
186 if (v == 0U) {
187 out[0] = (uint8_t)'0';
188 return selftest_sci_write(out, 1U);
189 }
190 while (v != 0U) {
191 if (count >= (uint8_t)k_selftest_dec_chars_u32) {
192 break;
193 }
194 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_selftest_dec_radix));
195 v = v / k_selftest_dec_radix;
196 count++;
197 }
198 for (uint8_t i = 0U; i < count; i++) {
199 out[i] = scratch[count - 1U - i];
200 }
201 return selftest_sci_write(out, (uint32_t)count);
202}

223[[nodiscard]] static ra8_err_t selftest_print_hex(uint32_t value, uint8_t digits)
224{
225 uint8_t out[k_selftest_hex_chars_u32] = {};
226 uint8_t width = digits;
227 if (width > (uint8_t)k_selftest_hex_chars_u32) {
228 width = (uint8_t)k_selftest_hex_chars_u32;
229 }
230 for (uint8_t i = 0U; i < width; i++) {
231 const uint8_t shift = (uint8_t)((width - 1U - i) * k_selftest_nibble_bits);
232 out[i] = selftest_nibble_to_hex((value >> shift) & k_selftest_nibble_mask);
233 }
234 return selftest_sci_write(out, (uint32_t)width);
235}

493[[nodiscard]] static ra8_err_t selftest_verify_mram_raw(void)
494{
495 static uint8_t s_burst[k_selftest_burst_bytes] = {};
496
497 s_dbg_verified_bytes = 0U;
498 s_dbg_mismatch_off = (uint32_t)k_selftest_no_mismatch;
499 const uint32_t t0 = ra8_time_ms();
500 for (uint32_t blk = 0U; blk < (uint32_t)k_fat_mram_clusters;
501 blk += (uint32_t)k_selftest_burst_blocks) {
502 const uint32_t lba = (uint32_t)k_fat_data_lba + blk;
503 ra8_err_t err = ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun,
504 lba,
505 (uint16_t)k_selftest_burst_blocks,
506 s_burst);
507 if (err != k_ra8_ok) {
508 (void)selftest_print_fail("READ(10) burst", err);
509 return err;
510 }
511 const uint32_t offset = blk * (uint32_t)k_selftest_block_size;
512 const uint8_t* mram = (const uint8_t*)(uintptr_t)((uint32_t)k_mram_base_addr + offset);
513 if (memcmp(s_burst, mram, (size_t)k_selftest_burst_bytes) != 0) {
514 s_dbg_mismatch_off = offset;
515 (void)selftest_print_fail("content mismatch at 0x", (ra8_err_t)offset);
516 return k_ra8_err_invalid_state;
517 }
518 s_dbg_verified_bytes = offset + (uint32_t)k_selftest_burst_bytes;
519 }
520 s_dbg_verify_ms = ra8_time_ms() - t0;
521 return k_ra8_ok;
522}

690[[nodiscard]] ra8_err_t selftest_host_pass(void)
691{
692 ra8_usb_hmsc_device_t device = {};
693 ra8_err_t err = selftest_host_enumerate(&device);
694 if (err != k_ra8_ok) {
695 return err;
696 }
697
698 /* Mount proves the host can PARSE the FAT16 volume over the loop;
699 * then unmount and do the heavy integrity check with raw multi-block
700 * READ(10) (ra8_fs's per-cluster metadata re-reads are far too slow
701 * for a 1 MiB sweep). */
702 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
703 ra8_fs_mount_t* mount = nullptr;
704 err = selftest_mount_volume(&mount);
705 if (err != k_ra8_ok) {
706 (void)ra8_usb_hmsc_close();
707 return err;
708 }
709 (void)ra8_fs_unmount(mount);
710
711 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
712 err = selftest_verify_mram_raw();
713 if (err == k_ra8_ok) {
714 err = selftest_print_verify_verdict();
715 }
716 if (err == k_ra8_ok) {
717 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
718 err = selftest_write_protect_probe();
719 }
720 if (err != k_ra8_ok) {
721 (void)ra8_usb_hmsc_close();
722 return err;
723 }
724
725 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
726 s_dbg_pass_count++;
727 err = selftest_print("ra8d2 selftest: USB SELFTEST CONFIG A PASS\r\n");
728 if (err != k_ra8_ok) {
729 return err;
730 }
731 (void)ra8_board_led_on(k_ra8_board_led2);
732 return k_ra8_ok;
733}
