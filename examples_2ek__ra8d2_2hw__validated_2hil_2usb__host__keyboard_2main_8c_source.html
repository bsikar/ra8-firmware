193static char hid_keycode_to_ascii(uint8_t kc)
194{
195 if ((kc >= (uint8_t)k_hid_kc_a) && (kc <= (uint8_t)k_hid_kc_z)) {
196 return (char)('A' + (int)(kc - (uint8_t)k_hid_kc_a));
197 }
198 if ((kc >= (uint8_t)k_hid_kc_1) && (kc < (uint8_t)k_hid_kc_0)) {
199 return (char)('1' + (int)(kc - (uint8_t)k_hid_kc_1));
200 }
201 if (kc == (uint8_t)k_hid_kc_0) {
202 return '0';
203 }
204 return '?';
205}

303[[nodiscard]] static ra8_err_t hid_enum_hunt(uint8_t* desc)
304{
305 ra8_delay_ms(k_hid_vbus_settle_ms);
306 const uint32_t t0 = ra8_time_ms();
307 for (uint32_t spin = 0U; spin < (uint32_t)k_hid_attach_spin; spin++) {
308 if (ra8_usb_host_line_state(k_ra8_usb_speed_hs) != 0U) {
309 break;
310 }
311 if ((ra8_time_ms() - t0) > (uint32_t)k_hid_attach_to_ms) {
312 break;
313 }
314 }
315 ra8_delay_ms(k_hid_debounce_ms);
316 ra8_err_t err = k_ra8_err_hw_timeout;
317 for (uint8_t attempt = 0U; attempt < (uint8_t)k_hid_enum_tries; attempt++) {
318 (void)ra8_usb_host_bus_reset(k_ra8_usb_speed_hs, true);
319 ra8_delay_ms(k_hid_reset_hold_ms);
320 (void)ra8_usb_host_bus_reset(k_ra8_usb_speed_hs, false);
321 (void)ra8_usb_host_set_uact(k_ra8_usb_speed_hs, true);
322 ra8_delay_ms(k_hid_recovery_ms);
323 (void)ra8_usb_host_set_target(k_ra8_usb_speed_hs, 0U);
324 err = hid_ctrl_get_dev_desc(desc);
325 if (err == k_ra8_ok) {
326 return k_ra8_ok;
327 }
328 }
329 return err;
330}

551[[nodiscard]] static ra8_err_t hid_read_round(uint32_t round)
552{
553 static uint8_t s_expect[k_hid_read_buf] = {};
554 static uint8_t s_rx[k_hid_read_buf] = {};
555 hid_fill_report_body(s_expect, (uint32_t)k_hid_report_len);
556 uint16_t rx = 0U;
557 const ra8_err_t err = ra8_usb_host_bulk_in(k_ra8_usb_speed_hs,
558 (uint8_t)k_hid_pipe_in,
559 s_rx,
560 (uint16_t)k_hid_read_buf,
561 &rx);
562 if (err != k_ra8_ok) {
563 (void)hid_print_fail("report read", err);
564 return err;
565 }
566 if (rx != (uint16_t)k_hid_report_len) {
567 s_dbg_mismatch = round;
568 (void)hid_print_fail("report length", k_ra8_err_invalid_size);
569 return k_ra8_err_invalid_size;
570 }
571 const size_t body = (size_t)((uint32_t)k_hid_report_len - (uint32_t)k_hid_body_idx);
572 if (memcmp(&s_rx[k_hid_body_idx], &s_expect[k_hid_body_idx], body) != 0) {
573 s_dbg_mismatch = round;
574 (void)hid_print_fail("report pattern mismatch", k_ra8_err_invalid_state);
575 return k_ra8_err_invalid_state;
576 }
577 s_dbg_last_seq = (uint32_t)s_rx[k_hid_seq_idx];
578 /* Decode the boot-keyboard keycodes (bytes 2..) back to ASCII for the verdict.
579 * The report is constant, so doing it each round is idempotent. */
580 for (uint32_t i = 0U; i < (uint32_t)k_hid_nkeys; i++) {
581 s_typed[i] = hid_keycode_to_ascii(s_rx[(uint32_t)k_hid_key0_idx + i]);
582 }
583 s_typed[(uint32_t)k_hid_nkeys] = '\0';
584 return k_ra8_ok;
585}

604[[nodiscard]] static ra8_err_t hid_host_pass(void)
605{
606 s_dbg_phase = (uint32_t)k_hid_phase_init;
607 ra8_err_t err = hid_print("ra8d2 hid: host up on USB-HS, probing the loop...\r\n");
608 if (err != k_ra8_ok) {
609 return err;
610 }
611 err = ra8_usb_host_init(k_ra8_usb_speed_hs);
612 if (err != k_ra8_ok) {
613 (void)hid_print_fail("host init", err);
614 return err;
615 }
616
617 s_dbg_phase = (uint32_t)k_hid_phase_enum;
618 uint32_t pid = 0U;
619 err = hid_enumerate(&pid);
620 if (err != k_ra8_ok) {
621 (void)ra8_usb_host_deinit(k_ra8_usb_speed_hs);
622 return err;
623 }
624 s_dbg_pid = pid;
625 err = hid_print_enum(pid);
626 if (err != k_ra8_ok) {
627 return err;
628 }
629
630 s_dbg_phase = (uint32_t)k_hid_phase_verify;
631 s_dbg_rounds_ok = 0U;
632 for (uint32_t r = 0U; r < (uint32_t)k_hid_rounds; r++) {
633 err = hid_read_round(r);
634 if (err != k_ra8_ok) {
635 (void)ra8_usb_host_deinit(k_ra8_usb_speed_hs);
636 return err;
637 }
638 s_dbg_rounds_ok++;
639 }
640
641 s_dbg_phase = (uint32_t)k_hid_phase_pass;
642 s_dbg_pass_count++;
643 err = hid_print_pass();
644 if (err != k_ra8_ok) {
645 return err;
646 }
647 (void)ra8_board_led_on(k_ra8_board_led2);
648 return k_ra8_ok;
649}
