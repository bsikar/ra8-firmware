4"""check_line_citations.py -- Reject in-tree source citations with line numbers.
5
6Per CLAUDE.md "Code Style / Comment citations":
7
8 Comments must NOT reference files in this repo by line number
9 (e.g. `libs/foo.c:776`). Line numbers go stale on the next reformat.
10 Reference the function / symbol name instead.
11
12 External / vendor citations (HUM, FSP, RFC, datasheet) remain
13 MANDATORY for any HAL register access, ISR, or driver path.
14
15Scope (derived, not a hardcoded root list -- #358):
16 Scans C / C++ source comments in every first-party C file (via
17 lint_targets, so tools/ -- which the old SCAN_ROOTS tuple
18 silently omitted -- is covered). Flags tokens matching
19 `<file>.<ext>:<line>` inside `// ...` or `/* ... */` comments.
20
21 Also scans every first-party Markdown (`.md`) / plain-text (`.txt`)
22 doc, wherever it lives (tools/mcp, examples/**/README.md, .claude/ agent
23 prompts, the repo root), not just docs/. The same regex applies; in docs
24 the whole line is treated as the "comment" (no comment-span extraction).
25
26Exemptions:
27 * docs/reference/* paths (HUM PDFs etc).
28 * libs/third_party/* (SOUP -- not our citations to manage).
29 * Any line containing `CITES-OK: <reason>` (reason text required).
30 * CHANGELOG-style "moved from <file>:NNN to ..." historical notes.
31 * `// SPDX-License-Identifier:` headers and `#include` directives.
32 * In docs: lines inside fenced code blocks whose info-string mentions
33 a tool name (`cppcheck`, `clang`, `clang-tidy`, `llvm-cov`, `gdb`,
34 `objdump`, `readelf`) -- these are tool transcripts, not citations.
35 * In docs: inline-code spans (backticked) whose first token is one
36 of the same tool names.
37"""

126def _line_is_tool_exempt(line: str) -> bool:
127 """Heuristic: line looks like tool transcript output."""
128 line.lstrip()
129 # Inline-code span starting with a known tool token.
130 # Examples: `clang foo.c:12: warning: ...`
131 # " `cppcheck libs/x.c:99 ...`"
132 for tok in TOOL_OUTPUT_TOKENS:
133 if tok in line:
134 # Require the token to appear *before* the first file:line
135 # match on the line (i.e. it owns/produced that citation).
136 tok_pos = line.find(tok)
137 m = CITATION_RE.search(line)
138 if m and tok_pos < m.start():
139 return True
140 return False

143def scan_doc_file(path: Path) -> list[tuple[int, str, str]]:
144 """Scan a markdown or plain-text doc for stale-prone line citations.
145
146 Returns a list of ``(line_no, matched_text, snippet)`` violations.
147
148 Docs need far more exemptions than source does, because a doc legitimately
149 QUOTES tool output that contains ``file:line`` -- which is a transcript,
150 not a citation the reader is meant to follow.
151
152 Exemptions:
153 * Fenced code blocks whose info-string contains a tool name.
154 * Lines where a tool token precedes the citation (inline tool
155 transcript).
156 * `CITES-OK: <reason>` opt-out.
157 * `moved from <file>:NN to ...` historical notes.
158 * Anchor links / heading IDs of the form `(file:NN)` are still
159 considered violations -- that is exactly what we want to ban.
160 """
161 try:
162 text = path.read_text(encoding="utf-8", errors="replace")
163 except OSError:
164 return []
165 violations: list[tuple[int, str, str]] = []
166 in_fence = False
167 fence_is_tool = False
168 fence_marker = ""
169 for line_no, raw in enumerate(text.splitlines(), start=1):
170 stripped = raw.lstrip()
171 # Fence open/close tracking (``` or ~~~).
172 if not in_fence and (stripped.startswith(("```", "~~~"))):
173 in_fence = True
174 fence_marker = stripped[:3]
175 info = stripped[3:].strip().lower()
176 fence_is_tool = any(tok in info for tok in TOOL_OUTPUT_TOKENS)
177 continue
178 if in_fence and stripped.startswith(fence_marker):
179 in_fence = False
180 fence_is_tool = False
181 fence_marker = ""
182 continue
183 if in_fence and fence_is_tool:
184 continue
185 for m in CITATION_RE.finditer(raw):
186 matched = m.group(0)
187 if DOCS_REFERENCE_RE.search(matched) or DOCS_REFERENCE_RE.search(raw):
188 continue
189 if THIRD_PARTY_RE.search(matched) or THIRD_PARTY_RE.search(raw):
190 continue
191 cok = CITES_OK_RE.search(raw)
192 if cok and cok.group(1).strip():
193 continue
194 if MOVED_FROM_RE.search(raw):
195 continue
196 if _line_is_tool_exempt(raw):
197 continue
198 snippet = raw.strip()
199 if len(snippet) > MAX_SNIPPET_LEN:
200 snippet = snippet[:SNIPPET_TRUNCATE_LEN] + "..."
201 violations.append((line_no, matched, snippet))
202 return violations

205def line_text(text: str, line_no: int) -> str:
206 """The text of a 1-based line, or "" when the number is out of range.
207
208 Returns empty rather than raising so a finding reported at a line past
209 EOF (possible on a truncated read) still prints instead of aborting the
210 sweep.
211 """
212 lines = text.splitlines()
213 if 1 <= line_no <= len(lines):
214 return lines[line_no - 1]
215 return ""

218def scan_file(path: Path) -> list[tuple[int, str, str]]:
219 """Return list of (line_no, matched_text, comment_snippet) violations."""
220 try:
221 text = path.read_text(encoding="utf-8", errors="replace")
222 except OSError:
223 return []
224 violations: list[tuple[int, str, str]] = []
225 seen: set[tuple[int, str]] = set()
226 # Two citation-bearing regions: C/C++ comments, and the string reason of an
227 # RA8_MCDC_DEACTIVATED(...) annotation. The reason is a string literal that
228 # find_comment_spans skips, but docs/ANNOTATIONS.md promises this gate scans
229 # it -- a deactivation reason anchored to a line number is DO-178C evidence
230 # that rots silently (#547). Both regions share one exemption cascade, so a
231 # CITES-OK on the line excuses either; dedup keeps a reason that happens to
232 # sit inside a doc comment from being reported twice.
233 spans = find_comment_spans(text) + find_mcdc_reason_spans(text)
234 for start, end in spans:
235 region = text[start:end]
236 # Per-line check: walk each match, validate against the
237 # specific physical line it sits on (so CITES-OK: on the same
238 # line excuses it).
239 for m in CITATION_RE.finditer(region):
240 abs_off = start + m.start()
241 line_no = line_of_offset(text, abs_off)
242 matched = m.group(0)
243 line = line_text(text, line_no)
244 if is_exempt(matched, line):
245 continue
246 if (line_no, matched) in seen:
247 continue
248 seen.add((line_no, matched))
249 snippet = line.strip()
250 if len(snippet) > MAX_SNIPPET_LEN:
251 snippet = snippet[:SNIPPET_TRUNCATE_LEN] + "..."
252 violations.append((line_no, matched, snippet))
253 return violations

269def _report_violations(
270 repo_root: Path,
271 paths: list[str],
272 scan: Callable[[Path], list[tuple[int, str, str]]],
273 per_file_counts: dict[str, int],
274) -> int:
275 """Print every violation ``scan`` finds under ``paths``; return the count.
276
277 Takes the scanner as a parameter because source files and documentation
278 are lexed differently but reported identically -- two copies of the
279 reporting loop is how the two report formats drift apart.
280 """
281 total = 0
282 for f in paths:
283 path = repo_root / f
284 if not path.is_file():
285 continue
286 viols = scan(path)
287 if not viols:
288 continue
289 per_file_counts[f] = len(viols)
290 for line_no, matched, snippet in viols:
291 print(f"{f}:{line_no}: line-citation found ('{matched}'): {snippet}")
292 print(" fix: replace with function/symbol name, or add `// CITES-OK: <reason>`")
293 total += 1
294 return total

316def _selftest_mcdc_reason_cases(tmp: Path, failures: list[str]) -> None:
317 """Assert both directions of the RA8_MCDC_DEACTIVATED reason scan (#547).
318
319 Extracted from :func:`selftest` so that function stays under the NASA Rule 4
320 line cap; the assertions are unchanged. The macro's reason is a string
321 literal, outside comment spans, so this proves a file:line inside one fires
322 and that a symbol-only or CITES-OK reason stays quiet -- the enforcement
323 docs/ANNOTATIONS.md promises.
324
325 Args:
326 tmp: A writable temporary directory for the fixture files.
327 failures: The accumulator each assertion records into.
328 """
329 bad_mcdc = tmp / "bad_mcdc.c"
330 bad_mcdc.write_text(
331 'RA8_MCDC_DEACTIVATED("guard justified in libs/foo.c:123")\n'
332 "static inline bool internal_guard(const void* p);\n",
333 encoding="utf-8",
334 )
335 expect(
336 bool(scan_file(bad_mcdc)),
337 "a file:line inside an RA8_MCDC_DEACTIVATED reason fires (#547)",
338 failures,
339 )
340 good_mcdc = tmp / "good_mcdc.c"
341 good_mcdc.write_text(
342 'RA8_MCDC_DEACTIVATED("guard: ra8_pin_validator_check asserts non-null")\n'
343 'RA8_MCDC_DEACTIVATED("legacy libs/foo.c:1 CITES-OK: historical note")\n',
344 encoding="utf-8",
345 )
346 expect(
347 not scan_file(good_mcdc),
348 "a symbol-only reason and a CITES-OK reason stay quiet (source)",
349 failures,
350 )

353def selftest() -> int:
354 """Prove a file:line citation fires, legal forms stay quiet, and scope holds."""
355 print("check_line_citations.py --selftest")
356 failures: list[str] = []
357 with tempfile.TemporaryDirectory() as tmp:
358 bad_src = Path(tmp) / "bad.c"
359 bad_src.write_text("/* see libs/foo.c:123 for the layout */\n", encoding="utf-8")
360 expect(bool(scan_file(bad_src)), "a file:line citation in a C comment fires", failures)
361 good_src = Path(tmp) / "good.c"
362 good_src.write_text(
363 "/* see ra8_foo(); moved from x.c:12 to here */\n"
364 "// libs/y.c:9 CITES-OK: illustrative example\n",
365 encoding="utf-8",
366 )
367 expect(
368 not scan_file(good_src),
369 "symbol ref / moved-from / CITES-OK stays quiet (source)",
370 failures,
371 )
372 _selftest_mcdc_reason_cases(Path(tmp), failures)
373 bad_doc = Path(tmp) / "bad.md"
374 bad_doc.write_text("See `ra8_ipc_regs.h:267` for the bit.\n", encoding="utf-8")
375 expect(bool(scan_doc_file(bad_doc)), "a file:line citation in a doc fires", failures)
376 good_doc = Path(tmp) / "good.md"
377 good_doc.write_text(
378 "See ra8_ipc_regs.h SAIPCIR2. libs/z.c:3 <!-- CITES-OK: illustrative -->\n",
379 encoding="utf-8",
380 )
381 expect(not scan_doc_file(good_doc), "doc CITES-OK stays quiet", failures)
382
383 _selftest_scope(failures)
384 return report(failures)

387def main() -> int:
388 """Reject in-tree citations that name a file by line number.
389
390 The rule exists because ``libs/foo.c:123`` goes stale the moment anything
391 above line 123 changes, and nothing detects that it has: the reference
392 still parses, still looks precise, and now points at the wrong line.
393 Function and symbol names survive edits, so they are what must be cited.
394
395 External HUM citations are unaffected -- the manual has stable page
396 numbers, and citing them is mandatory elsewhere in the tree.
397
398 Returns 1 listing each stale-prone citation, 0 when the scanned set is
399 clean.
400 """
401 if "--selftest" in sys.argv[1:]:
402 return selftest()
403
404 repo_root = Path(
405 subprocess.run(
406 ["git", "rev-parse", "--show-toplevel"], # noqa: S607 # trusted: fixed git argv
407 check=True,
408 capture_output=True,
409 text=True,
410 ).stdout.strip()
411 )
412
413 files = _files_to_scan()
414 total_violations = 0
415 per_file_counts: dict[str, int] = {}
416 for scan, paths in (
417 (scan_file, [f for f in files if is_in_scope(f)]),
418 (scan_doc_file, [f for f in files if is_doc_in_scope(f)]),
419 ):
420 total_violations += _report_violations(repo_root, paths, scan, per_file_counts)
421
422 if total_violations == 0:
423 return 0
424
425 print(file=sys.stderr)
426 print(
427 f"check_line_citations: {total_violations} violation(s) across "
428 f"{len(per_file_counts)} file(s).",
429 file=sys.stderr,
430 )
431 if WARN_ONLY_MODE:
432 print(
433 "check_line_citations: WAVE 0 -- warn-only, not blocking commit.",
434 file=sys.stderr,
435 )
436 return 0
437 return 1
