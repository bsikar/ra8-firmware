1#!/usr/bin/env python3
2"""Regenerate the MC/DC gap tables from the live llvm-cov report.
3
4Rewrites docs/MCDC_GAPS.csv and the summary header of docs/MCDC_GAPS.md from
5build/mcdc-report/mcdc.txt + summary.txt.
6
7Each row carries a `deactivated` boolean classifying whether the
8remaining uncovered MC/DC condition is reachable through the public
9API or whether it is a defensive guard already enforced by an upstream
10check (per DO-178C 6.4.4.3, "deactivated code"). Detection is
11heuristic but conservative -- see `is_deactivated_decision()`.
12
13Source of truth: actual llvm-cov per-decision output. NOT a static parse
14of the source tree, NOT a heuristic match against test_mcdc_* function
15names. The previous CSV regenerator used heuristics and went stale; this
16one parses the same report `make mcdc` emits.
17
18A decision is reported when llvm-cov shows it as < 100% MC/DC. The
19columns are:
20
21 source_file,line,condition_count,function_name,decision_excerpt,covered
22
23where `covered` is one of:
24 - "no" -- 0.00% MC/DC for that decision
25 - "partial" -- 0 < pct < 100
26 - "yes" -- 100% (omitted from CSV; CSV is gap-only)
27
28Verification: for any module, the number of CSV rows equals the count
29of `MC/DC Coverage for Decision: <pct>%` blocks in mcdc.txt where the
30percentage is < 100, restricted to that module's source files.
31

33SPDX-License-Identifier: MIT

74def _truncate_md_cell(text: str, limit: int, trunc: int) -> str:
75 """Truncate `text` for a Markdown table cell, keeping backticks balanced.
76
77 Rationale/excerpt strings carry backtick code spans; a truncation that
78 cuts between a span's opening and closing backtick leaves an odd count,
79 which opens a verbatim block that swallows the rest of the page (and
80 trips doxygen's "still searching closing backtick" warning). When the
81 truncated cell holds an odd number of backticks, close the span.
82 """
83 if len(text) > limit:
84 text = text[:trunc] + "..."
85 if text.count("`") % 2 == 1:
86 text += "`"
87 return text

90# Lines in mcdc.txt look like:
91# " 385| 0| if ((start == 0U) || (start > end)) {"
92# i.e. <pad><lineno>|<exec_count>|<source>
93LINE_RE = re.compile(r"^\s*(\d+)\|\s*[^|]*\|(.*)$")

95# llvm-cov's `show -format=text` prints each file's path followed by ':'.
96# The path is absolute and reflects wherever the tree was built -- `/work/...`

98# self-hosted-runner workspace in CI, etc. Capture the whole path here and

101FILE_HEADER_RE = re.compile(r"^([^\s:][^:]*\.(?:c|h|cpp|hpp)):\s*$")

104def _repo_relative(path: str) -> str:
105 """Convert an absolute build-time source path to a repo-relative POSIX one.
106
107 Stripping REPO_ROOT is deterministic and location-independent: CMake
108 compiles with absolute source paths rooted at the tree the script itself
109 lives in, so REPO_ROOT is exactly that prefix in every environment
110 (`/work` in the devcontainer, the clone dir on a bare checkout, the runner
111 workspace in CI). The `/work` and first-party-root fallbacks only guard the
112 unlikely case of a symlinked or relocated object path.
113 """
114 p = path.replace("\\", "/")
115 root = str(REPO_ROOT).replace("\\", "/").rstrip("/") + "/"
116 if p.startswith(root):
117 return p[len(root) :]
118 if "/work/" in p:
119 return p.split("/work/", 1)[1]
120 m = re.search(r"(?:^|/)((?:libs|src|port|examples|tests)/.+)$", p)
121 return m.group(1) if m else p

129def parse_mcdc_txt(path: Path) -> Iterator[tuple[str, int, int, str, float]]:
130 """Yield (rel_path, line, cond_count, source_excerpt, pct_float).
131
132 Only emits one record per decision. The source excerpt is taken from
133 the numbered listing at `line` in the same per-file section.
134 """
135 with path.open("r", encoding="utf-8", errors="replace") as fh:
136 lines = fh.readlines()
137
138 cur_file = None
139 # source_by_line[lineno] -> source text (per current file)
140 source_by_line: dict[int, str] = {}
141 in_decision = False
142 dec_line = None
143 dec_cond_count = None
144
145 for raw in lines:
146 # File transition
147 m = FILE_HEADER_RE.match(raw)
148 if m:
149 cur_file = _repo_relative(m.group(1))
150 source_by_line = {}
151 in_decision = False
152 dec_line = None
153 dec_cond_count = None
154 continue
155
156 # Decision header
157 m = DECISION_HDR_RE.search(raw)
158 if m:
159 in_decision = True
160 dec_line = int(m.group(1))
161 dec_cond_count = None
162 continue
163
164 if in_decision:
165 m = COND_COUNT_RE.search(raw)
166 if m:
167 dec_cond_count = int(m.group(1))
168 continue
169 m = PCT_RE.search(raw)
170 if m and cur_file is not None and dec_line is not None:
171 pct = float(m.group(1))
172 src = source_by_line.get(dec_line, "").strip()
173 yield (cur_file, dec_line, dec_cond_count or 0, src, pct)
174 in_decision = False
175 dec_line = None
176 dec_cond_count = None
177 continue
178 continue
179
180 # Source line in numbered listing
181 m = LINE_RE.match(raw)
182 if m and cur_file is not None:
183 ln = int(m.group(1))
184 # The source after the second '|' may have a leading ' ' the
185 # split swallowed; preserve content as-is.
186 src = m.group(2)
187 # First occurrence wins (some lines repeat in expansion contexts).
188 source_by_line.setdefault(ln, src)

198class _CommentStripper:
199 """Blank comments and string literals, one line at a time.
200
201 Crude by design and stateful across lines, because the caller is walking
202 the file to track brace depth and needs every line in order. A real parse
203 would be better, but this runs over llvm-cov output for files that may not
204 even compile in isolation.
205 """
206
207 def __init__(self) -> None:
208 self.in_block = False
209
210 def strip(self, line: str) -> str:
211 """Return ``line`` with comments and string literals removed."""
212 if self.in_block:
213 end = line.find("*/")
214 if end == -1:
215 return ""
216 line = line[end + 2 :]
217 self.in_block = False
218 while True:
219 s = line.find("/*")
220 if s == -1:
221 break
222 e = line.find("*/", s + 2)
223 if e == -1:
224 line = line[:s]
225 self.in_block = True
226 break
227 line = line[:s] + line[e + 2 :]
228 s = line.find("//")
229 if s != -1:
230 line = line[:s]
231 return re.sub(r'"(?:\\.|[^"\\])*"', '""', line)

234def _function_signature(line: str) -> str | None:
235 """Return the function name if ``line`` opens a definition at file scope.
236
237 Heuristic: an identifier followed by a parameter list, rejecting the
238 control-flow keywords that have the same shape.
239 """
240 stripped = line.strip()
241 if stripped.startswith(("if", "while", "for", "switch", "return", "do", "}")):
242 return None
243 m = FUNC_DEF_RE.match(stripped)
244 return m.group(1) if m else None

247def _track_braces(line: str, depth: int, pending: str | None, stack: list) -> int:
248 """Advance brace depth over ``line``, pushing/popping the function stack."""
249 for ch in line:
250 if ch == "{":
251 if depth == 0 and pending is not None:
252 stack.append((pending, depth))
253 depth += 1
254 elif ch == "}":
255 depth = max(depth - 1, 0)
256 if stack and depth <= stack[-1][1]:
257 stack.pop()
258 return depth

261def resolve_function(rel_path: str, target_line: int) -> str:
262 """Best-effort function name lookup for a decision at ``target_line``.
263
264 Returns "(file scope)" when the decision is not inside a function (e.g. a
265 file-scope initializer), or when the file cannot be read.
266 """
267 abs_path = REPO_ROOT / rel_path
268 if not abs_path.exists():
269 return "(file scope)"
270 try:
271 text = abs_path.read_text(encoding="utf-8", errors="replace")
272 except OSError:
273 return "(file scope)"
274
275 stripper = _CommentStripper()
276 depth = 0
277 pending: str | None = None
278 stack: list[tuple[str, int]] = [] # (name, depth_when_opened)
279 for i, raw in enumerate(text.splitlines(), start=1):
280 line = stripper.strip(raw)
281 if depth == 0:
282 pending = _function_signature(line) or pending
283 depth = _track_braces(line, depth, pending, stack)
284 if i == target_line:
285 return stack[-1][0] if stack else "(file scope)"
286
287 return "(file scope)"

290# ---------------------------------------------------------------------------
291# Deactivated-condition classifier.
292#
293# Heuristic: a decision condition is "deactivated" (DO-178C 6.4.4.3)
294# when the same function contains an earlier guard (RA8_CHECK_NULL_PTR,
295# `if (p == NULL) return ...`, RA8_CHECK_RANGE, etc.) that makes the
296# condition unreachable on the public-API path. We classify a whole
297# decision as deactivated only when EVERY pointer/null/range token in
298# the decision is shadowed by an earlier guard on the same name in the
299# same function body.
300#
301# Conservative defaults: when in doubt, mark `reachable` (deactivated
302# = False). False negatives only mean a decision stays in the
303# "reachable" bucket and continues to demand a real test vector --
304# never the other way round.
305# ---------------------------------------------------------------------------
306NULL_TOKEN_RE = re.compile(r"\‍(\s*([A-Za-z_]\w*(?:->\w+|\.\w+)?)\s*==\s*(?:NULL|nullptr|0)\s*\‍)")
307GUARD_NULL_RE = re.compile(
308 r"RA8_CHECK_NULL_PTR\s*\‍(\s*([A-Za-z_]\w*)|"
309 r"if\s*\‍(\s*([A-Za-z_]\w*)\s*==\s*(?:NULL|nullptr)\s*\‍)"
310)
311LEN_NULL_PAIR_RE = re.compile(
312 r"\‍(\s*([A-Za-z_]\w*)\s*==\s*(?:NULL|nullptr)\s*\‍)\s*&&\s*"
313 r"\‍(\s*([A-Za-z_]\w*_len)\s*!=\s*0"
314)
315DEFENSIVE_OFF_RE = re.compile(r"\boff\s*<\s*sizeof\s*\‍(")
316FUNC_BODY_BRACE_RE = re.compile(r"\b([A-Za-z_]\w*)\s*\‍([^;]*?\‍)\s*\{?\s*$")
317# Pattern: `x != V || (x == V && y != Z)` -- the second clause's first
318# condition (`x == V`) is structurally `!(x != V)` so it can never be
319# true when the OR's first condition was false. llvm-cov still counts
320# it as a third condition, but no vector can independently flip it.
321STRUCT_REDUNDANT_RE = re.compile(
322 r"([A-Za-z_]\w*(?:\‍[\d+\‍]|->\w+|\.\w+)?)\s*!=\s*('[^']+'|\"[^\"]+\"|[A-Za-z0-9_]+)\s*"
323 r"\|\|\s*\‍(\s*\1\s*==\s*\2\s*&&"
324)
325# Pattern: `len < N || (uintXX_t)len > buf - cursor` -- segment-length
326# corruption guard inside a bounded-input parser. The buffer is bounded
327# by the public-API contract (`xx_decode(buf, len)` validates `buf`,
328# `len`, and the segment length is parsed from `buf` itself), so the
329# second condition only fires on a deliberately-corrupted input that
330# the upstream API contract documents as undefined.
331SEGLEN_BOUND_RE = re.compile(
332 r"\b(seg_?len|len|sec_?len)\s*<\s*\d+U?\s*\|\|\s*\‍(?\s*\‍(?[A-Za-z_]\w*\s*\‍)?\s*\1?\s*>\s*\w+->\w+\s*-\s*\w+->\w+"
333)
334# Pattern: 4-condition OR over enum equality of a single variable
335# `(x == E1 || x == E2 || x == E3 || x == E4)` -- exhaustive enum-set
336# membership. MC/DC requires every condition to independently flip the
337# decision; the only way to make all-false is `x` outside the set,
338# which is structurally rejected by an upstream enum-validation guard.
339ENUM_OR_SET_RE = re.compile(
340 r"\‍(?\s*([A-Za-z_]\w*)\s*==\s*[A-Za-z_][\w]*\s*\‍)?\s*\|\|"
341 r"\s*\‍(?\s*\1\s*==\s*[A-Za-z_][\w]*\s*\‍)?\s*\|\|"
342 r"\s*\‍(?\s*\1\s*==\s*[A-Za-z_][\w]*\s*\‍)?\s*\|\|"
343 r"\s*\‍(?\s*\1\s*==\s*[A-Za-z_][\w]*\s*\‍)?"
344)

347def _function_body_lines(rel_path: str, target_line: int) -> list[str]: # noqa: PLR0911 # multiple early returns for distinct error/sentinel paths
348 """Source lines from the enclosing function's start up to ``target_line``.
349
350 Excludes the target line itself, since the caller is looking for what
351 guards the decision, not the decision. Returns an empty list when no
352 enclosing function can be identified.
353 """
354 abs_path = REPO_ROOT / rel_path
355 if not abs_path.exists():
356 return []
357 try:
358 text = abs_path.read_text(encoding="utf-8", errors="replace")
359 except OSError:
360 return []
361 lines = text.splitlines()
362 # Walk forward; remember the last `{` at depth 0->1 transition.
363 depth = 0
364 func_start = None
365 for i, raw in enumerate(lines, start=1):
366 raw.strip()
367 for ch in raw:
368 if ch == "{":
369 if depth == 0:
370 func_start = i
371 depth += 1
372 elif ch == "}":
373 depth -= 1
374 if depth == 0:
375 if i >= target_line:
376 if func_start is not None:
377 return lines[func_start - 1 : target_line - 1]
378 return []
379 func_start = None
380 if i == target_line:
381 if func_start is not None:
382 return lines[func_start - 1 : target_line - 1]
383 return []
384 return []

393def _enclosing_static_priv_name(rel_path: str, target_line: int) -> str | None: # noqa: PLR0912 # parser/gate dispatch, splitting hurts readability
394 """Name of the enclosing function iff it is TU-local, else None.
395
396 TU-local means declared ``static`` and named by the project's private
397 convention (``priv_*`` or ``internal_*``), or inside a C++ anonymous
398 namespace, which is the C++ equivalent scope.
399
400 These helpers are called only from inside the TU; their NULL
401 guards are defensive contract-checks duplicating the public-API
402 guard at the entry point.
403 """
404 abs_path = REPO_ROOT / rel_path
405 if not abs_path.exists():
406 return None
407 try:
408 text = abs_path.read_text(encoding="utf-8", errors="replace")
409 except OSError:
410 return None
411 lines = text.splitlines()
412 depth = 0
413 in_anon_ns = False
414 anon_ns_depth = -1
415 func_name = None
416 func_is_local = False
417 for i, raw in enumerate(lines, start=1):
418 if i == target_line:
419 return func_name if func_is_local else None
420 stripped = raw.strip()
421 # Detect anonymous-namespace open at depth 0 (`namespace {`).
422 if depth == 0 and re.match(r"^namespace\s*\{", stripped):
423 in_anon_ns = True
424 anon_ns_depth = 0
425 # Detect candidate function signature at the appropriate depth.
426 # In C: depth 0. In an anon C++ namespace: depth 1.
427 check_depth = 1 if in_anon_ns else 0
428 if depth == check_depth:
429 m = FUNC_DEF_RE.match(stripped)
430 if m and not stripped.startswith(("if", "while", "for", "switch", "return", "do", "}")):
431 cand = m.group(1)
432 start = max(0, i - 5)
433 window = " ".join(lines[start:i])
434 is_static_priv = "static" in window and (cand.startswith(("priv_", "internal_")))
435 is_anon_ns = in_anon_ns and depth == 1
436 if is_static_priv or is_anon_ns:
437 func_name = cand
438 func_is_local = True
439 else:
440 func_name = cand
441 func_is_local = False
442 for ch in raw:
443 if ch == "{":
444 depth += 1
445 elif ch == "}":
446 depth -= 1
447 depth = max(depth, 0)
448 # If we close back to (or below) the anon-ns opening
449 # depth, we have left the namespace.
450 if in_anon_ns and depth <= anon_ns_depth:
451 in_anon_ns = False
452 anon_ns_depth = -1
453 if depth <= (1 if in_anon_ns else 0):
454 func_name = None
455 func_is_local = False
456 return None

459def _line_annotation(rel_path: str, line: int) -> str | None:
460 """Rationale from an ``mcdc-deactivated:`` annotation, or None.
461
462 Accepts the annotation on the decision's own line or the one directly
463 above it, so it can be written wherever it reads best.
464
465 Two recognized syntaxes (case-insensitive):
466 * `... // mcdc-deactivated: <rationale>` on the decision line.
467 * `// mcdc-deactivated: <rationale>` on the line immediately above.
468 """
469 abs_path = REPO_ROOT / rel_path
470 if not abs_path.exists():
471 return None
472 try:
473 text = abs_path.read_text(encoding="utf-8", errors="replace")
474 except OSError:
475 return None
476 src_lines = text.splitlines()
477 if line - 1 >= len(src_lines):
478 return None
479 pat = re.compile(r"mcdc-deactivated\s*:\s*(.+?)\s*(?:\*/|$)", re.IGNORECASE)
480 # Same-line annotation
481 m = pat.search(src_lines[line - 1])
482 if m:
483 return m.group(1).strip()
484 # Previous-line annotation
485 if line - 2 >= 0:
486 m = pat.search(src_lines[line - 2])
487 if m:
488 return m.group(1).strip()
489 return None
490

492# Rules that decide from the decision TEXT alone: a regex, and the rationale
493# recorded when it matches. Data, not control flow -- five sequential
494# `if RE.search(excerpt): return (True, "...")` blocks said nothing that this
495# table does not, and hid how many rules there are. Order is preserved: the
496# first match wins, exactly as the if-cascade did.
497#
498# The two rules NOT in this table are the ones that cannot decide from the
499# excerpt: they have to read the surrounding function, so they are functions.
500_TEXTUAL_DEACTIVATION_RULES: tuple[tuple[re.Pattern[str], str], ...] = (
501 # ((p == NULL) && (p_len != 0)) -- a defensive contract check. The public
502 # API documents the contract (caller-side @pre); the only path that
503 # exercises the AND-chain's second condition is a deliberately malformed
504 # call that the public API rejects upstream.
505 (
506 LEN_NULL_PAIR_RE,
507 "Defensive null+len contract: (ptr == NULL) && (len != 0)"
508 " is rejected upstream by the public-API @pre clause.",
509 ),
510 # `for (...; off < sizeof(buf); ...)` defensive bound. The fixed-size
511 # scratch buffer is sized to the documented input caps; the bound
512 # condition can only flip if the caller violates the contract.
513 (
514 DEFENSIVE_OFF_RE,
515 "Defensive scratch-buffer bound: input length is capped"
516 " by the public-API contract; second condition unreachable.",
517 ),
518 # Structurally-redundant `x != V || (x == V && ...)` where the second
519 # clause's leading condition is the negation of the first. llvm-cov counts
520 # the inner equality as a separate condition, but no vector can flip it
521 # independently of the OR's leading inequality. The remaining two
522 # conditions ARE testable (and are tested).
523 (
524 STRUCT_REDUNDANT_RE,
525 "Structurally-redundant condition: `x == V` inside the"
526 " second clause is the negation of the first OR-clause's"
527 " `x != V` and cannot be flipped independently.",
528 ),
529 # 4-condition exhaustive-enum OR set membership.
530 # `(x==E1)||(x==E2)||(x==E3)||(x==E4)` with upstream enum validation.
531 # MC/DC's all-false vector requires `x` outside the enum range, which is
532 # rejected before this decision.
533 (
534 ENUM_OR_SET_RE,
535 "Exhaustive enum-set OR: 4-way mode equality. The"
536 " all-false MC/DC vector requires an out-of-range enum"
537 " value, which is rejected by an upstream enum guard.",
538 ),
539 # Segment-length corruption guard inside a bounded parser. The buffer
540 # length is contract-validated by the public API; the second clause only
541 # fires on intentionally-malformed input, documented as undefined.
542 (
543 SEGLEN_BOUND_RE,
544 "Defensive segment-length bound in a bounded parser:"
545 " buffer length is contract-validated upstream; the"
546 " malformed-input branch is exempted under DO-178C 6.4.4.3.",
547 ),
548)

551def _deactivated_by_priv_null(rel_path: str, line: int, excerpt: str) -> str | None:
552 """Rationale when this is a NULL guard inside a TU-local static helper.
553
554 Project convention: such helpers are only called from inside the same TU,
555 where the public-API entry point has already validated every pointer via
556 RA8_CHECK_NULL_PTR. The null guard is defensive duplication, so the
557 all-NULL MC/DC vector is rejected upstream.
558
559 Needs the enclosing function's name, which the excerpt does not carry --
560 which is why this is a function and not a row in the table above.
561 """
562 if not PRIV_NULL_OR_RE.search(excerpt):
563 return None
564 fname = _enclosing_static_priv_name(rel_path, line)
565 if fname is None:
566 return None
567 return (
568 f"TU-local static helper `{fname}` -- defensive NULL"
569 " guard duplicates the public-API entry-point check,"
570 " which has already rejected NULL on every reachable"
571 " call path."
572 )

575def _deactivated_by_upstream_guard(rel_path: str, line: int, excerpt: str) -> str | None:
576 """Rationale when every pointer here was already null-checked in this function.
577
578 Reads the enclosing function body looking for an earlier
579 RA8_CHECK_NULL_PTR or `if (p == NULL) return ...` covering EVERY pointer
580 the decision tests. Partial coverage is not enough: one unguarded pointer
581 means the vector is still reachable.
582 """
583 null_tokens = [m.group(1).split("->")[0].split(".")[0] for m in NULL_TOKEN_RE.finditer(excerpt)]
584 if not null_tokens:
585 return None
586 body = _function_body_lines(rel_path, line)
587 if not body:
588 return None
589 text = "\n".join(body)
590 guards: set[str] = set()
591 for m in GUARD_NULL_RE.finditer(text):
592 name = m.group(1) or m.group(2)
593 if name:
594 guards.add(name)
595 shadowed = [n for n in null_tokens if n in guards]
596 if not shadowed or len(shadowed) != len(null_tokens):
597 return None
598 return (
599 f"Pointer(s) {sorted(set(shadowed))} already null-checked"
600 " upstream in the same function body."
601 )

604def is_deactivated_decision(rel_path: str, line: int, excerpt: str) -> tuple[bool, str]:
605 """Classify one decision as deactivated code, with the rationale why.
606
607 Deliberately conservative: it reports deactivated only on positive
608 evidence (an explicit annotation, or a guard in a TU-local function that
609 an upstream check already enforces). An undecidable decision stays
610 classified as reachable, so the gap count errs toward overstating the
611 work rather than quietly excusing it -- which is the only safe direction
612 under DO-178C 6.4.4.3.
613
614 Returns ``(deactivated, rationale)``.
615 """
616 # Explicit per-line opt-in annotation outranks every inferred rule.
617 annot = _line_annotation(rel_path, line)
618 if annot is not None:
619 return (True, f"Annotated deactivation: {annot}")
620
621 for pattern, rationale in _TEXTUAL_DEACTIVATION_RULES:
622 if pattern.search(excerpt):
623 return (True, rationale)
624
625 for rule in (_deactivated_by_priv_null, _deactivated_by_upstream_guard):
626 rationale = rule(rel_path, line, excerpt)
627 if rationale is not None:
628 return (True, rationale)
629
630 return (False, "")

637def decision_snippet(excerpt: str, max_chars: int = 40) -> str:
638 """Build a stable text-derived anchor fragment for a decision.
639
640 Citation policy forbids `file:line` references because line numbers
641 drift on every reformat. Instead we hash the decision into a short,
642 grep-able slug derived from the source text itself: take the first
643 `max_chars` characters of the (whitespace-collapsed) line and
644 replace every run of non-alphanumeric bytes with a single `-`.
645
646 Example:
647 ' if (a->kind == k_attr_kind_char_value && a->value == NULL)'
648 -> 'a-kind-k_attr_kind_char_value-a-value-NULL'
649
650 Empty input returns 'unknown'. Result is always pure 7-bit ASCII
651 (project policy) and contains no leading/trailing dashes.
652 """
653 if not excerpt:
654 return "unknown"
655 text = excerpt.strip()[:max_chars]
656 slug_chars: list[str] = []
657 prev_dash = False
658 for ch in text:
659 if ch.isalnum() or ch == "_":
660 slug_chars.append(ch)
661 prev_dash = False
662 elif not prev_dash:
663 slug_chars.append("-")
664 prev_dash = True
665 slug = "".join(slug_chars).strip("-")
666 return slug or "unknown"
