134int32_t main(void)
135{
136 gpt_pwm_demo_setup_or_halt();
137 ra8_isr_globals_enable();
138
139 if (gpt_pwm_demo_arm() != k_ra8_ok) {
140 gpt_pwm_demo_panic_halt();
141 }
142
143 uint32_t duty = 0U;
144 bool up = true;
145 uint32_t prev_count = 0U;
146 bool have_prev = false;
147 while (1) {
148 if (ra8_gpt_set_duty((uint8_t)k_gpt_pwm_demo_channel, k_ra8_gpt_ccr_a, duty) != k_ra8_ok) {
149 g_gpt_pwm_mismatch += 1U;
150 break;
151 }
152 uint32_t now_count = 0U;
153 if (ra8_gpt_read((uint8_t)k_gpt_pwm_demo_channel, &now_count) != k_ra8_ok) {
154 g_gpt_pwm_mismatch += 1U;
155 } else if (have_prev && (now_count == prev_count)) {
156 /* 20 ms passed but GTCNT did not move -- timer is wedged. */
157 g_gpt_pwm_mismatch += 1U;
158 prev_count = now_count;
159 } else {
160 g_gpt_pwm_match += 1U;
161 prev_count = now_count;
162 have_prev = true;
163 }
164 if (up) {
165 if (duty + (uint32_t)k_gpt_pwm_demo_step_pct >= (uint32_t)k_gpt_pwm_demo_period) {
166 duty = (uint32_t)k_gpt_pwm_demo_period;
167 up = false;
168 } else {
169 duty += (uint32_t)k_gpt_pwm_demo_step_pct;
170 }
171 } else {
172 if (duty <= (uint32_t)k_gpt_pwm_demo_step_pct) {
173 duty = 0U;
174 up = true;
175 } else {
176 duty -= (uint32_t)k_gpt_pwm_demo_step_pct;
177 }
178 }
179 ra8_delay_ms((uint32_t)k_gpt_pwm_demo_step_ms);
180 }
181 gpt_pwm_demo_panic_halt();
182 return 0;
183}
