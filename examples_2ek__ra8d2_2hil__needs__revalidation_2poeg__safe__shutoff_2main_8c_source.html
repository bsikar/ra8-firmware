249[[nodiscard]] static ra8_err_t poeg_demo_cycle(uint8_t* out_ok)
250{
251 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
252 *out_ok = 0U;
253
254 /* RUN: sample the free-running PWM counter. */
255 uint32_t before = 0U;
256 const ra8_err_t rd0 = ra8_gpt_read((uint8_t)k_poeg_demo_gpt_channel, &before);
257 if (rd0 != k_ra8_ok) {
258 return rd0;
259 }
260
261 /* SHUTOFF: assert the software output-disable request and read ST back. */
262 const ra8_err_t terr = ra8_poeg_trigger_stop((uint8_t)k_poeg_demo_group);
263 if (terr != k_ra8_ok) {
264 return terr;
265 }
266 uint32_t asserted = 0U;
267 const ra8_err_t serr = ra8_poeg_get_status((uint8_t)k_poeg_demo_group, &asserted);
268 if (serr != k_ra8_ok) {
269 return serr;
270 }
271 g_poeg_last_status = asserted;
272 const uint8_t shutoff = ((asserted & (uint32_t)k_ra8_poeg_status_st) != 0U) ? 1U : 0U;
273
274 /* Hold the shutoff briefly; the GPT counter keeps advancing meanwhile. */
275 ra8_delay_ms((uint32_t)k_poeg_demo_settle_ms);
276
277 /* RE-ENABLE: clear the request and confirm ST fell back to 0. */
278 const ra8_err_t cerr =
279 ra8_poeg_clear_status((uint8_t)k_poeg_demo_group, (uint32_t)k_ra8_poeg_status_ssf);
280 if (cerr != k_ra8_ok) {
281 return cerr;
282 }
283 uint32_t cleared = 0U;
284 const ra8_err_t cs = ra8_poeg_get_status((uint8_t)k_poeg_demo_group, &cleared);
285 if (cs != k_ra8_ok) {
286 return cs;
287 }
288 const uint8_t reenabled = ((cleared & (uint32_t)k_ra8_poeg_status_st) == 0U) ? 1U : 0U;
289
290 /* RUN confirm: the PWM counter must have advanced across the hold. */
291 uint32_t after = 0U;
292 const ra8_err_t rd1 = ra8_gpt_read((uint8_t)k_poeg_demo_gpt_channel, &after);
293 if (rd1 != k_ra8_ok) {
294 return rd1;
295 }
296 const uint8_t running = (after != before) ? 1U : 0U;
297
298 *out_ok = (shutoff != 0U && reenabled != 0U && running != 0U) ? 1U : 0U;
299 return k_ra8_ok;
300}

304int32_t main(void)
305{
306 poeg_demo_setup_or_halt();
307 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the SysTick
308 * path (ra8_emulator does not advance DWT_CYCCNT). No NVIC sources are armed. */
309 ra8_isr_globals_enable();
310
311 if (poeg_demo_arm_gpt() != k_ra8_ok) {
312 poeg_demo_panic_halt();
313 }
314 if (poeg_demo_arm_poeg() != k_ra8_ok) {
315 poeg_demo_panic_halt();
316 }
317
318 while (1) {
319 uint8_t ok = 0U;
320 const ra8_err_t err = poeg_demo_cycle(&ok);
321 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
322 if (good != 0U) {
323 (void)ra8_board_uart_console_write(k_poeg_demo_ok_msg,
324 (size_t)(sizeof(k_poeg_demo_ok_msg) - 1U));
325 (void)ra8_board_led_toggle(k_ra8_board_led1);
326 ++g_poeg_shutoffs;
327 } else {
328 (void)ra8_board_uart_console_write(k_poeg_demo_bad_msg,
329 (size_t)(sizeof(k_poeg_demo_bad_msg) - 1U));
330 (void)ra8_board_led_toggle(k_ra8_board_led2);
331 ++g_poeg_mismatch;
332 }
333 ++g_poeg_heartbeat;
334 ra8_delay_ms((uint32_t)k_poeg_demo_period_ms);
335 }
336 poeg_demo_panic_halt();
337 return 0;
338}
