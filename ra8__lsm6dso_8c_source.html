136static uint8_t internal_lsm6dso_g_fs_bits(ra8_lsm6dso_g_fs_t fs)
137{
138 /* DS12140 Table 47: FS_125 takes priority over FS_G[1:0]. */
139 if (fs == k_lsm6dso_g_fs_125dps) {
140 return (uint8_t)((uint8_t)1U << (uint8_t)k_lsm6dso_shift_fs_125);
141 }
142 /* Numerically, _250dps == 1, _500dps == 2, _1000dps == 3, _2000dps == 4.
143 * Map back to the FS_G[1:0] bit field by subtracting one. */
144 const uint8_t fs_g_field = (uint8_t)((uint8_t)fs - (uint8_t)k_lsm6dso_g_fs_250dps);
145 return (uint8_t)((uint8_t)(fs_g_field & 0x03U) << (uint8_t)k_lsm6dso_shift_fs_g);
146}

296 /* DS12140 sec 9.12 CTRL1_XL: read-modify-write FS_XL[3:2]. */

397static ra8_err_t internal_lsm6dso_read_xyz(ra8_lsm6dso_t* dev, uint8_t reg, ra8_lsm6dso_xyz_t* out)
398{
399 uint8_t bytes[6] = {};
400 const ra8_err_t r =
401 dev->bus.read_regs(dev->bus.ctx, reg, bytes, (uint32_t)k_lsm6dso_xyz_burst_bytes);
402 if (r != k_ra8_ok) {
403 return r;
404 }
405 /* LSM6DSO XYZ samples are little-endian two's complement
406 * (DS12140 sec 9.29 .. 9.40). */
407 out->x = (int16_t)(((uint16_t)bytes[k_lsm6dso_xyz_off_x_h] << 8) |
408 (uint16_t)bytes[k_lsm6dso_xyz_off_x_l]);
409 out->y = (int16_t)(((uint16_t)bytes[k_lsm6dso_xyz_off_y_h] << 8) |
410 (uint16_t)bytes[k_lsm6dso_xyz_off_y_l]);
411 out->z = (int16_t)(((uint16_t)bytes[k_lsm6dso_xyz_off_z_h] << 8) |
412 (uint16_t)bytes[k_lsm6dso_xyz_off_z_l]);
413 return k_ra8_ok;
414}

434ra8_err_t ra8_lsm6dso_read_temp(ra8_lsm6dso_t* dev, int32_t* out_centi_c)
435{
436 RA8_CHECK_NULL_PTR(dev, k_lsm6dso_tag, "read_temp: dev");
437 RA8_CHECK_NULL_PTR(out_centi_c, k_lsm6dso_tag, "read_temp: out_centi_c");
438 RA8_VALIDATE_INIT(dev->initialized, k_lsm6dso_tag, "read_temp: not initialized");
439
440 /* DS12140 sec 9.27 OUT_TEMP_L (20h) + sec 9.28 OUT_TEMP_H (21h). */
441 uint8_t bytes[2] = {};
442 const ra8_err_t r = dev->bus.read_regs(dev->bus.ctx,
443 (uint8_t)k_lsm6dso_reg_out_temp_l,
444 bytes,
445 (uint32_t)k_lsm6dso_temp_burst_bytes);
446 if (r != k_ra8_ok) {
447 return r;
448 }
449 const int16_t raw =
450 (int16_t)(((uint16_t)bytes[k_lsm6dso_idx_high] << 8) | (uint16_t)bytes[k_lsm6dso_idx_low]);
451 /* DS12140 sec 4.3 "Temperature sensor characteristics":
452 * T[degC] = raw / 256 + 25
453 * Scale into centi-deg C to keep integer arithmetic: */
454 *out_centi_c = (int32_t)((int32_t)raw * (int32_t)k_lsm6dso_temp_scale_num /
455 (int32_t)k_lsm6dso_temp_scale_den) +
456 (int32_t)k_lsm6dso_temp_offset_centi_c;
457 return k_ra8_ok;
458}

582 uint8_t* out_buf,
583 uint32_t max_words,
584 uint32_t* out_words)
585{
586 /* Set the output count early so callers can rely on it on failure. */
587 if (out_words != nullptr) {
588 *out_words = 0U;
589 }

591 if (chk != k_ra8_ok) {
592 return chk;
593 }
594
595 uint32_t live = 0U;
596 const ra8_err_t rs = internal_lsm6dso_read_fifo_depth(dev, &live);
597 if (rs != k_ra8_ok) {
598 return rs;
599 }
600 const uint32_t to_read = (live > max_words) ? max_words : live;
601 if (to_read == 0U) {
602 return k_ra8_ok;
603 }
604 const ra8_err_t rd = internal_lsm6dso_burst_fifo(dev, out_buf, to_read);
605 if (rd != k_ra8_ok) {
606 return rd;
607 }
608 *out_words = to_read;
