196static void sd_demo_setup_or_halt(uint32_t* out_pclka_hz)
197{
198 uint32_t cpuclk0_hz = 0U;
199 uint32_t pclka_hz = 0U;
200 if (ra8_cgc_init() != k_ra8_ok) {
201 sd_demo_panic_halt();
202 }
203 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
204 sd_demo_panic_halt();
205 }
206 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
207 sd_demo_panic_halt();
208 }
209 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
210 sd_demo_panic_halt();
211 }
212 if (ra8_board_uart_console_init((uint32_t)k_sd_demo_uart_baud) != k_ra8_ok) {
213 sd_demo_panic_halt();
214 }
215 *out_pclka_hz = pclka_hz;
216}

239static void sd_demo_init_card_or_halt(uint32_t pclka_hz)
240{
241 const ra8_sdmmc_spi_sci_pins_t pins = {
242 .sck = k_sd_demo_pin_sck,
243 .cipo = k_sd_demo_pin_cipo,
244 .copi = k_sd_demo_pin_copi,
245 .cs = k_sd_demo_pin_cs,
246 };
247 ra8_sdmmc_spi_transport_t transport = {};
248 if (ra8_sdmmc_spi_transport_sci((uint8_t)k_sd_demo_spi_channel, pclka_hz, &pins, &transport) !=
249 k_ra8_ok) {
250 SD_DEMO_PUTS(k_msg_init_fail);
251 sd_demo_panic_halt();
252 }
253 if (ra8_sdmmc_spi_init(&transport) != k_ra8_ok) {
254 SD_DEMO_PUTS(k_msg_init_fail);
255 sd_demo_panic_halt();
256 }
257 SD_DEMO_PUTS(k_msg_card_ok);
258}

287[[nodiscard]] static ra8_err_t sd_demo_roundtrip(void)

321int main(void)
322{
323 uint32_t pclka_hz = 0U;
324 sd_demo_setup_or_halt(&pclka_hz);
325 ra8_isr_globals_enable();
326 ra8_log_init();
327 SD_DEMO_PUTS(k_msg_boot);
328
329 sd_demo_init_card_or_halt(pclka_hz);
330
331 const ra8_err_t r = sd_demo_roundtrip();
332 if (r != k_ra8_ok) {
333 SD_DEMO_PUTS(k_msg_fail);
334 sd_demo_panic_halt();
335 }
336 SD_DEMO_PUTS(k_msg_pass);
337 (void)ra8_board_uart_console_flush();
338
339 while (true) {
340 __asm__ volatile("wfi");
341 }
342}
