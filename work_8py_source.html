130def _ws_root(value: str | None = None) -> Path:
131 """Return the absolute lexical canonical workspace root."""
132 selected = value or os.environ.get(WS_ROOT_ENV) or str(Path.home() / "ra8-ws")
133 if not selected.isascii() or not selected.isprintable():
134 message = "workspace root must contain only printable single-line ASCII"
135 raise WorkError(message)
136 # ``resolve`` would follow a hostile symlink before the claim validator can reject it.
137 return Path(
138 os.path.abspath( # noqa: PTH100 -- preserve symlink evidence for claim validation
139 os.fspath(Path(selected).expanduser())
140 )

149def _writability(path: Path) -> tuple[bool, Path]:
150 """Return whether the nearest existing ancestor is writable."""
151 probe = path

153 probe = probe.parent
154 return os.access(probe, os.W_OK), probe

162def _doctor_checks(paths: RepoPaths, cwd: Path) -> list[Check]:
163 """Build read-only readiness checks, including emitted-script dependencies."""
164 version = ".".join(str(part) for part in sys.version_info[:3])
165 python_state = CHECK_OK if sys.version_info[:2] >= MIN_PYTHON else CHECK_FAIL
166 git_version = run_process([git_executable(), "--version"], cwd=cwd).stdout.strip()
167 root = _ws_root()
168 writable, ancestor = _writability(root)
169 script = _workspace_script(paths)
170 jq = shutil.which("jq")
171 jq_state = CHECK_OK if jq is not None else CHECK_FAIL
172 jq_detail = run_process([jq, "--version"], cwd=cwd).stdout.strip() if jq else "missing"
173 kind = "linked worktree" if paths.is_linked_worktree else "main worktree"
174 return [
175 Check("python", python_state, f"{version} (minimum {MIN_PYTHON[0]}.{MIN_PYTHON[1]})"),
176 Check("git", CHECK_OK, git_version),
177 Check("jq", jq_state, jq_detail),
178 Check("repository", CHECK_OK, f"{paths.toplevel} ({kind})"),
179 Check(
180 "canonical workspace tool",
181 CHECK_OK if script.is_file() else CHECK_FAIL,
182 str(script),
183 ),
184 Check(
185 "workspace root",
186 CHECK_OK if writable else CHECK_FAIL,
187 f"{root} (nearest ancestor {ancestor} is {'writable' if writable else 'NOT writable'})",
188 ),
189 _probe_to_check(probe_version()),
190 _probe_to_check(probe_auth()),
191 _probe_to_check(probe_project_query()),
192 ]

195def cmd_doctor(_options: argparse.Namespace) -> int:
196 """Report local readiness; ``gh auth status`` may perform a read-only API probe."""
197 cwd = Path.cwd()
198 paths = discover_repo(cwd)
199 checks = _doctor_checks(paths, cwd)
200 width = max(len(check.name) for check in checks)
201 for check in checks:
202 _print(f" {check.state:<12} {check.name:<{width}} {check.detail}")
203 for check in checks:
204 if check.state in (STATE_DEGRADED, STATE_UNAVAILABLE):
205 _print(f"notice: {check.name} is {check.state.lower()} -- {check.detail}")
206 failed = [check for check in checks if check.state == CHECK_FAIL]
207 if failed:
208 _fail(f"work doctor: {len(failed)} check(s) FAILED")
209 return EXIT_FAIL
210 return EXIT_OK

213def _reject_nonregular_destination(path: Path) -> None:
214 """Reject an existing destination unless ``lstat`` proves it is regular."""
215 try:
216 mode = path.lstat().st_mode
217 except FileNotFoundError:
218 return
219 if not stat.S_ISREG(mode):
220 message = f"refusing to replace destination that is not a regular file: {path}"
221 raise WorkError(message)

224def _write_text_atomic(path: Path, text: str) -> None:
225 """Write derived output atomically without opening a nonregular destination."""
226 _reject_nonregular_destination(path)
227 path.parent.mkdir(parents=True, exist_ok=True)
228 descriptor, temporary = tempfile.mkstemp(prefix=f".{path.name}.", dir=path.parent)
229 try:
230 with os.fdopen(descriptor, "w", encoding="ascii") as handle:
231 handle.write(text)
232 handle.flush()
233 os.fsync(handle.fileno())
234 _reject_nonregular_destination(path)
235 Path(temporary).replace(path)
236 finally:
237 with contextlib.suppress(FileNotFoundError):
238 Path(temporary).unlink()

255def cmd_plan(options: argparse.Namespace) -> int:
256 """Validate notes and emit deterministic artifacts."""
257 if problem := _plan_output_problem(options):
258 _fail(f"work plan: {problem}")
259 return EXIT_CONFIG
260 notes = Path(options.notes).expanduser()
261 try:
262 plan = load_plan(notes)
263 except PlanError as exc:
264 for problem in exc.problems:
265 _fail(f"{notes}: {problem}")
266 _fail(f"{notes}: {len(exc.problems)} problem(s); nothing was emitted")
267 return EXIT_FAIL
268 emitted = False
269 if options.json is not None:
270 text = plan_to_json(plan)
271 if options.json == "-":
272 sys.stdout.write(text)
273 else:
274 _write_text_atomic(Path(options.json), text)
275 _notice(f"wrote {options.json}")
276 emitted = True
277 if options.emit_commands:
278 sys.stdout.write(render_commands(plan))
279 emitted = True
280 if options.summary or not emitted:
281 sys.stdout.write(render_summary(plan))
282 return EXIT_OK

285def _build_start_plan(paths: RepoPaths, cwd: Path, options: argparse.Namespace) -> StartPlan:
286 """Build a read-only preview; canonical creation repeats checks under its lock."""
287 root = _ws_root(options.ws_root)
288 identifier = options.identifier
289 name = workspace_name(identifier)
290 branch = branch_name(identifier)
291 target = root / name
292 base = resolve_commit(options.ref, cwd=cwd)
293 if base is not None:
294 reject_executable_attributes(cwd, base)
295 refusals: list[str] = []
296 if base is None:
297 refusals.append(f"start ref does not resolve locally: {options.ref}")
298 if target.exists() or target.is_symlink():
299 refusals.append(f"workspace path already exists: {target}")
300 claim_path = metadata_path(root, identifier)
301 if claim_path.exists() or claim_path.is_symlink():
302 refusals.append(f"canonical workspace metadata already exists for {name}")
303 if branch_exists(branch, cwd=cwd):
304 refusals.append(f"branch already exists: {branch}")
305 if target.parent != root:
306 refusals.append("derived workspace is not a direct child of the canonical root")
307 if not _workspace_script(paths).is_file():
308 refusals.append("canonical workspace script is absent")
309 return StartPlan(identifier, name, branch, target, root, options.ref, base, tuple(refusals))

325def _execute_start(paths: RepoPaths, start: StartPlan) -> int:
326 """Delegate creation to the sole workspace lock/state authority."""
327 if start.base_commit is None:
328 return EXIT_FAIL
329 argv = [
330 "/bin/bash",
331 "-p",

343 environment["RA8_WS_UPSTREAM"] = str(paths.toplevel)

378def cmd_status(options: argparse.Namespace) -> int:
379 """Report only work-owned records from canonical workspace metadata."""
380 paths = discover_repo(Path.cwd())
381 root = _ws_root(options.ws_root)
382 records = list_claims(root)
383 if not records:
384 _print(f"work status: no work-owned canonical metadata under {metadata_dir(root)}")
385 return EXIT_OK
386 forged = False
387 for path, value in records:
388 if isinstance(value, ClaimError):
389 _print(f" {FORGED:<9} {path.name:<24} {value}")
390 forged = True
391 continue
392 verdict = classify(value, paths.toplevel)
393 _print(f" {verdict:<9} {value.identifier:<24} {value.branch:<28} {value.worktree}")
394 forged = forged or verdict == FORGED
395 if verdict in (STALE, FOREIGN):
396 command = recovery_command(value, paths.toplevel, stale=verdict == STALE)
397 _print(f" after human review: {command}")
398 return EXIT_FAIL if forged else EXIT_OK

401def _remote_ci(paths: RepoPaths, workspace: Path, head: str) -> tuple[str, str]:
402 """Read the shared monitor's cached verdict without polling GitHub."""
403 monitor = paths.toplevel / "scripts/ci/monitor.sh"
404 if not monitor.is_file():
405 return "UNKNOWN", "shared CI monitor script is absent"
406 result = run_process(
407 ["/bin/bash", "-p", str(monitor), "status", "--sha", head],
408 cwd=workspace,
409 timeout=30,
410 )
411 state = {0: "PASS", 1: "FAIL", 3: "UNKNOWN"}.get(result.returncode, "UNKNOWN")
412 detail = next(
413 (line for line in result.stdout.splitlines() if line.strip()), "no monitor detail"
414 )
415 return state, detail

418def _workspace_report(claim: Claim, phase: str) -> tuple[str | None, list[str]]:
419 """Print shared workspace facts and return its HEAD plus dirty paths."""
420 dirty = porcelain_status(claim.worktree)
421 head = resolve_commit("HEAD", cwd=claim.worktree)
422 if head is None:
423 _fail(f"work {phase}: workspace HEAD does not resolve")
424 return None, dirty
425 _print(f"work {phase}: {claim.identifier}")
426 _print(f" branch {claim.branch}")
427 _print(f" worktree {claim.worktree}")
428 _print(f" base {claim.base_ref} ({claim.base_commit})")
429 _print(f" claimed {claim.created} by {claim.creator}")
430 _print(f" HEAD {head}")
431 _print(f" working set {len(dirty)} changed path(s)")
432 _print(" diffstat vs base:")
433 for line in diff_stat(claim.worktree, claim.base_commit).splitlines() or ["(empty)"]:
434 _print(f" {line}")
435 return head, dirty

438def _clean_claim(options: argparse.Namespace, phase: str) -> tuple[RepoPaths, Claim] | None:
439 """Load one exact claim and require a clean committed working tree."""
440 if not is_identifier(options.identifier):
441 _fail(f"work {phase}: refusing invalid identifier {options.identifier!r}")
442 return None
443 paths = discover_repo(Path.cwd())
444 root = _ws_root(options.ws_root)
445 try:
446 claim = _claim_for(options.identifier, paths, root)
447 except ClaimError as exc:
448 _fail(f"work {phase}: {exc}")
449 return None
450 _head, dirty = _workspace_report(claim, phase)
451 if dirty:
452 _fail(f"work {phase}: REFUSE -- commit or intentionally discard the working set first")
453 _fail("just ci validates committed HEAD, not these uncommitted bytes")
454 return None
455 return paths, claim

458def cmd_ready(options: argparse.Namespace) -> int:
459 """Run exact local CI for a clean committed work claim before the sole push."""
460 loaded = _clean_claim(options, "ready")
461 if loaded is None:
462 return EXIT_FAIL
463 _paths, claim = loaded
464 if not options.run_ci:
465 _fail("work ready: REFUSE -- pass --run-ci to record exact local gate evidence")
466 return EXIT_FAIL
467 just = shutil.which("just")
468 if just is None:
469 _fail("work ready: just is not installed")
470 return EXIT_FAIL
471 _print(" local CI RUNNING: just ci")
472 result = run_process([just, "ci"], cwd=claim.worktree, timeout=7200)
473 _print_block(result.stdout)
474 if not result.ok:
475 _print_block(result.stderr, error=True)
476 _fail(f"work ready: local CI did not PASS (exit {result.returncode})")
477 return EXIT_FAIL
478 head = resolve_commit("HEAD", cwd=claim.worktree)
479 _print(f" local CI PASS: just ci at {head}")
480 _print("work ready: review/squash this tree, then perform the one authorized push")
481 return EXIT_OK

484def cmd_landed(options: argparse.Namespace) -> int:
485 """Require content-equivalent pushed dev and cached PASS after the push."""
486 loaded = _clean_claim(options, "landed")
487 if loaded is None:
488 return EXIT_FAIL
489 paths, claim = loaded
490 claim_head = resolve_commit("HEAD", cwd=claim.worktree)
491 dev_head = resolve_commit("origin/dev", cwd=claim.worktree)
492 claim_tree = resolve_tree(claim_head, cwd=claim.worktree) if claim_head else None
493 dev_tree = resolve_tree(dev_head, cwd=claim.worktree) if dev_head else None
494 if dev_head is None or claim_tree is None or claim_tree != dev_tree:
495 _fail("work landed: origin/dev is absent or not content-equivalent to this work claim")
496 return EXIT_FAIL
497 ci_state, ci_detail = _remote_ci(paths, claim.worktree, dev_head)
498 _print(f" remote CI {ci_state}: {ci_detail}")
499 if ci_state != "PASS":
500 _fail("work landed: remote CI is not PASS; Landed would be false")
501 return EXIT_FAIL
502 _print(f"work landed: origin/dev {dev_head} is content-equivalent and remotely green")
503 _print(
504 " now set the board card to Landed, close citing the squash SHA, "
505 "then review branch cleanup"
506 )
507 return EXIT_OK

549def run_selftest() -> int:
550 """Run tests and fail if discovery silently collapses below its floor."""
551 tests = Path(__file__).resolve().parents[1] / "tests"
552 suite = unittest.TestLoader().discover(str(tests), top_level_dir=str(tests))
553 result = unittest.TextTestRunner(verbosity=2).run(suite)
554 if result.testsRun < SELFTEST_MINIMUM:
555 _fail(f"work selftest: discovered {result.testsRun} tests, below floor {SELFTEST_MINIMUM}")
556 return EXIT_FAIL
557 return EXIT_OK if result.wasSuccessful() else EXIT_FAIL

560def main(argv: list[str]) -> int:
561 """Dispatch one command with redacted failures."""
562 parser = build_parser()
563 options = parser.parse_args(argv)
564 if options.selftest:
565 return run_selftest()
566 if options.command is None:
567 parser.print_help()
568 return EXIT_CONFIG

570 return HANDLERS[options.command](options)
