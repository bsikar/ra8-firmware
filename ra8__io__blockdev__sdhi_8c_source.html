148 uint32_t blocks = 0;

152 }
153 out->block_count = blocks;
154 out->erase_unit_blocks = (uint32_t)k_ra8_io_sdhi_erase_unit_blocks;

157 out->erase_value = (uint8_t)k_ra8_io_erase_value_zero;
158 out->must_erase_before_write = false;
159 out->read_only = false;
160 return k_ra8_ok;
