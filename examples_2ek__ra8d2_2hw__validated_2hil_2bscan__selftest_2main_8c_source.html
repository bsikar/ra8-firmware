93static void bs_print_hex(uint32_t value)
94{
95 uint8_t buf[k_bs_hex_nibbles];
96 for (uint32_t i = 0U; i < (uint32_t)k_bs_hex_nibbles; i++) {
97 const uint32_t shift = ((uint32_t)k_bs_hex_nibbles - 1U - i) * (uint32_t)k_bs_nibble_bits;
98 const uint32_t nib = (value >> shift) & (uint32_t)k_bs_nibble_mask;
99 buf[i] = (uint8_t)((nib < (uint32_t)k_bs_dec_ten) ? ('0' + nib) : ('A' + (nib - k_bs_dec_ten)));
100 }
101 bs_print(buf, (uint32_t)k_bs_hex_nibbles);
102}

105static void bs_print_uint(uint32_t value)
106{
107 uint8_t buf[k_bs_dec_ten];
108 uint32_t n = 0U;
109 if (value == 0U) {
110 buf[n] = '0';
111 n++;
112 }
113 while ((value > 0U) && (n < (uint32_t)k_bs_dec_ten)) {
114 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_bs_dec_ten));
115 n++;
116 value /= (uint32_t)k_bs_dec_ten;
117 }
118 for (uint32_t i = 0U; i < n; i++) {
119 bs_print(&buf[n - 1U - i], 1U);
120 }
121}

135static uint32_t bs_run_checks(uint32_t* out_idcode)
136{
137 uint32_t id = 0U;
138 ra8_bscan_status_t st = {};
139
140 bs_require(out_idcode != nullptr); /* 1 */
141 bs_require(ra8_bscan_init() == k_ra8_ok); /* 2 */
142 bs_require(ra8_bscan_get_idcode(nullptr) == k_ra8_err_null_ptr); /* 3 */
143 bs_require(ra8_bscan_get_idcode(&id) == k_ra8_ok); /* 4 */
144 bs_require(id == (uint32_t)k_ra8_bscan_jtidr_reset); /* 5 */
145 bs_require(ra8_bscan_get_status(nullptr) == k_ra8_err_null_ptr); /* 6 */
146 bs_require(ra8_bscan_get_status(&st) == k_ra8_ok); /* 7 */
147 bs_require(st.initialized && (st.expected_idcode == (uint32_t)k_ra8_bscan_jtidr_reset)); /* 8 */
148 bs_require(st.last_instruction == k_ra8_bscan_instr_bypass); /* 9 */
149 bs_require(ra8_bscan_set_instruction(k_ra8_bscan_instr_idcode) == k_ra8_ok); /* 10 */
150 bs_require(ra8_bscan_get_status(&st) == k_ra8_ok); /* 11 */
151 bs_require(st.last_instruction == k_ra8_bscan_instr_idcode); /* 12 */
152 bs_require(ra8_bscan_set_instruction((ra8_bscan_instr_t)k_bs_reserved_op) ==
153 k_ra8_err_invalid_arg); /* 13 */
154 bs_require(ra8_bscan_clear_status((uint32_t)k_bs_bad_mask) == k_ra8_err_invalid_arg); /* 14 */
155 bs_require(ra8_bscan_clear_status((uint32_t)k_bs_clear_none) == k_ra8_ok); /* 15 */
156 bs_require(ra8_bscan_get_status(&st) == k_ra8_ok); /* 16 */
157 bs_require(st.last_instruction == k_ra8_bscan_instr_bypass); /* 17 */
158
159 *out_idcode = id;
160 return (uint32_t)k_bs_checks_total;
161}

164static void bs_setup_or_halt(void)
165{
166 uint32_t cpuclk0_hz = 0U;
167 if ((ra8_cgc_init() != k_ra8_ok) || (ra8_mstp_init() != k_ra8_ok)) {
168 bs_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
169 }
170 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
171 bs_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
172 }
173 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
174 bs_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
175 }
176 if (ra8_board_uart_console_init((uint32_t)k_bs_uart_baud) != k_ra8_ok) {
177 bs_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
178 }
179}

194{

197 bs_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);

201

203 bs_print_hex(idcode);
204 bs_print(k_msg_mid, (uint32_t)sizeof(k_msg_mid) - 1U);
205 bs_print_uint(checks);
206 bs_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
207
208 while (1) {
209 __asm__ volatile("wfi");
210 }
211}
