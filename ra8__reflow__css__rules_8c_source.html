107static bool priv_intern_name(ra8_css_sheet_t* sheet, const char* s, size_t len, uint16_t* off)
108{
109 /* mcdc-deactivated: every caller guards the length before interning -- (nlen == 0U) reject at the selector site, (n > 0U) at the @font-face/font-family sites, and priv_extract_url rejects an empty url() -- so (len == 0U) is unreachable; only (len > k_ra8_css_name_max) varies. */

117 (void)memcpy(&sheet->names[sheet->names_used], s, len);
118 sheet->names_used = (uint16_t)(sheet->names_used + len);
119 return true;
120}

151static bool priv_parse_sel_type(const char* s, size_t len, size_t* i, ra8_css_rule_t* rule)
152{
153 if (s[*i] == '*') {
154 ++(*i); /* universal: no type constraint */
155 return true;
156 }
157 size_t start = *i;
158 /* Bounded: i advances over name chars, capped by len. */
159 while ((*i < len) && priv_is_name_char(s[*i])) {
160 ++(*i);
161 }
162 const ra8_reflow_html_tag_t tag = ra8_reflow_tok_classify(&s[start], *i - start);
163 if (tag == k_ra8_reflow_tag_unknown) {
164 return false; /* unrecognised tag -> drop the rule */
165 }
166 rule->sel_tag = (uint8_t)tag;
167 return true;
168}

202 const char* s,
203 size_t len,
204 size_t* i,

206{

262static bool
263priv_parse_selector(ra8_css_sheet_t* sheet, const char* s, size_t len, ra8_css_rule_t* rule)
264{
265 if (len == 0U) {
266 return false;
267 }
268 size_t i = 0U;
269 bool any = false;
270 if (priv_is_name_char(s[0]) || (s[0] == '*')) {
271 if (!priv_parse_sel_type(s, len, &i, rule)) {
272 return false;
273 }
274 any = true;
275 }
276 /* Then a run of `.class` / `#id` parts (at most one of each). */
277 while (i < len) {
278 if (((s[i] != '.') && (s[i] != '#')) || !priv_parse_sel_part(sheet, s, len, &i, rule)) {
279 return false; /* combinator / pseudo / dup / bad name -> unsupported */
280 }
281 any = true;
282 }
283 return any;
284}

315RA8_INTERNAL
316static int32_t priv_split_compounds(const char* s, size_t len, const char** part_p, size_t* part_n)
317{
318 size_t nparts = 0U;
319 size_t i = 0U;
320 /* Bounded: i advances to len; each pass consumes >=1 char after the ws skip. */
321 while (i < len) {
322 /* mcdc-deactivated: priv_split_compounds is only ever called with a whitespace-trimmed selector (ra8_reflow_css_trim in priv_parse_selector_list and the block dispatcher), so there is no trailing whitespace for this inner skip to consume up to len; the (i < len) false arm is unreachable on any public path. */
323 while ((i < len) && ra8_reflow_css_is_ws(s[i])) {
324 ++i;
325 }
326 if (i >= len) {
327 break;
328 }
329 if (nparts > (size_t)k_ra8_css_max_anc) {
330 return -1; /* more than one subject + k_ra8_css_max_anc ancestor parts */
331 }
332 const size_t start = i;
333 while ((i < len) && !ra8_reflow_css_is_ws(s[i])) {
334 ++i;
335 }
336 part_p[nparts] = &s[start];
337 part_n[nparts] = i - start;
338 ++nparts;
339 }
340 return (int32_t)nparts;
341}

371RA8_INTERNAL
372static bool
373priv_parse_complex_selector(ra8_css_sheet_t* sheet, const char* s, size_t len, ra8_css_rule_t* rule)
374{
375 const char* part_p[(size_t)k_ra8_css_max_anc + 1U] = {};
376 size_t part_n[(size_t)k_ra8_css_max_anc + 1U] = {};
377 const int32_t nparts = priv_split_compounds(s, len, part_p, part_n);
378 if (nparts <= 0) {
379 return false;
380 }
381 const size_t np = (size_t)nparts;
382 /* Last part = subject compound. */
383 if (!priv_parse_selector(sheet, part_p[np - 1U], part_n[np - 1U], rule)) {
384 return false;
385 }
386 /* Earlier parts = ancestor constraints (outermost first). */
387 rule->anc_count = (uint8_t)(np - 1U);
388 for (size_t a = 0U; (a + 1U) < np; ++a) {
389 ra8_css_rule_t tmp = {};
390 if (!priv_parse_selector(sheet, part_p[a], part_n[a], &tmp)) {
391 return false;
392 }
393 rule->anc[a].tag = tmp.sel_tag;
394 rule->anc[a].class_off = tmp.class_off;
395 rule->anc[a].class_len = tmp.class_len;
396 rule->anc[a].id_off = tmp.id_off;
397 rule->anc[a].id_len = tmp.id_len;
398 }
399 return true;
400}

529RA8_INTERNAL
530static bool priv_extract_url(const char* val, size_t vlen, const char** url, size_t* ulen)
531{
532 /* Bounded: at most (vlen - 3) windows; i advances by 1 each step. */
533 for (size_t i = 0U; (i + (size_t)k_priv_url_len) <= vlen; ++i) {
534 if (!ra8_reflow_css_ci_eq(&val[i], (size_t)k_priv_url_len, "url(")) {
535 continue;
536 }
537 size_t a = i + (size_t)k_priv_url_len;
538 size_t b = a;
539 while ((b < vlen) && (val[b] != ')')) {
540 ++b;
541 }
542 size_t n = b - a;
543 const char* p = ra8_reflow_css_trim(&val[a], &n);
544 p = priv_strip_quotes(p, &n);
545 if (n == 0U) {
546 return false;
547 }
548 *url = p;
549 *ulen = n;
550 return true;
551 }
552 return false;
553}

643static void priv_face_apply(ra8_css_sheet_t* sheet,
644 const char* prop,
645 size_t plen,
646 const char* val,
647 size_t vlen,

649{

652 const char* fam = priv_strip_quotes(val, &n);
653 uint16_t off = 0U;
654 if ((n > 0U) && priv_intern_name(sheet, fam, n, &off)) {
655 face->family_off = off;

657 }
658 } else if (ra8_reflow_css_ci_eq(prop, plen, "font-weight")) {
659 face->weight_bold = priv_is_bold_kw(val, vlen) ? 1U : 0U;
660 } else if (ra8_reflow_css_ci_eq(prop, plen, "font-style")) {
661 face->style_italic = priv_is_italic_kw(val, vlen) ? 1U : 0U;
662 } else if (ra8_reflow_css_ci_eq(prop, plen, "src")) {
663 const char* url = nullptr;
664 size_t ulen = 0U;
665 uint16_t off = 0U;
666 if (priv_extract_url(val, vlen, &url, &ulen) && priv_intern_name(sheet, url, ulen, &off)) {
667 face->src_off = off;
668 face->src_len = (uint16_t)ulen;
669 }
670 } else {
671 /* Other @font-face descriptors (unicode-range, ...) -> ignored. */
672 }
673}

703RA8_INTERNAL
704static void priv_for_each_decl(const char* s, size_t len, priv_decl_fn cb, void* ctx)
705{
706 size_t i = 0U;
707 /* Bounded: each pass advances past the next ';' (or to len). */
708 while (i < len) {
709 size_t semi = i;
710 while ((semi < len) && (s[semi] != ';')) {
711 ++semi;
712 }
713 size_t colon = i;
714 while ((colon < semi) && (s[colon] != ':')) {
715 ++colon;
716 }
717 if (colon < semi) {
718 size_t plen = colon - i;
719 const char* prop = ra8_reflow_css_trim(&s[i], &plen);
720 size_t vlen = semi - (colon + 1U);
721 const char* val = ra8_reflow_css_trim(&s[colon + 1U], &vlen);
722 if ((plen > 0U) && (vlen > 0U)) {
723 cb(ctx, prop, plen, val, vlen);
724 }
725 }
726 i = semi + 1U;
727 }
