272{

318static uint32_t drw_mix(uint32_t src, uint32_t dst, uint32_t fs, uint32_t fd)
319{
320 const uint32_t num = (src * fs) + (dst * fd) + (uint32_t)k_drw_alpha_round;
321 const uint32_t out = num / (uint32_t)k_drw_alpha_full;
322 return (out > (uint32_t)k_drw_byte_mask) ? (uint32_t)k_drw_byte_mask : out;
323}

326static uint32_t drw_out_alpha(uint32_t src_a, uint32_t dst_a)
327{
328 if ((s_drw.control2 & (uint32_t)k_drw_c2_useacb) != 0U) {
329 /* HUM Ch 62.6.5.2 "Alpha Channel Blending" p 3734 */
330 const uint32_t fsa = drw_factor((s_drw.control2 & (uint32_t)k_drw_c2_bsfa) != 0U,
331 (s_drw.control2 & (uint32_t)k_drw_c2_bsia) != 0U,
332 src_a);
333 const uint32_t fda = drw_factor((s_drw.control2 & (uint32_t)k_drw_c2_bdfa) != 0U,
334 (s_drw.control2 & (uint32_t)k_drw_c2_bdia) != 0U,
335 src_a);
336 return drw_mix(src_a, dst_a, fsa, fda);
337 }
338 /* HUM Ch 62.2.2 "WRITEALPHA[1:0]" p 3694 -- Figure 62.23 p 3734 */
339 const uint32_t code =
340 (s_drw.control2 >> (uint32_t)k_drw_c2_walpha_pos) & (uint32_t)k_drw_c2_walpha_mask;
341 switch (code) {
342 case (uint32_t)k_drw_walpha_color2:
343 return drw_chan(s_drw.color2, (uint32_t)k_drw_a_shift);
344 case (uint32_t)k_drw_walpha_pixel_cov:
345 return src_a;
346 case (uint32_t)k_drw_walpha_zero:
347 return 0U;
348 default:
349 return dst_a;
350 }

354static uint32_t drw_shade(uint32_t dst)
355{
356 const uint32_t src_a = drw_chan(s_drw.color1, (uint32_t)k_drw_a_shift);
357 const uint32_t fs = drw_factor((s_drw.control2 & (uint32_t)k_drw_c2_bsf) != 0U,
358 (s_drw.control2 & (uint32_t)k_drw_c2_bsi) != 0U,
359 src_a);
360 const uint32_t fd = drw_factor((s_drw.control2 & (uint32_t)k_drw_c2_bdf) != 0U,
361 (s_drw.control2 & (uint32_t)k_drw_c2_bdi) != 0U,
362 src_a);
363
364 const uint32_t r = drw_mix(drw_chan(s_drw.color1, (uint32_t)k_drw_r_shift),
365 drw_chan(dst, (uint32_t)k_drw_r_shift),
366 fs,
367 fd);
368 const uint32_t g = drw_mix(drw_chan(s_drw.color1, (uint32_t)k_drw_g_shift),
369 drw_chan(dst, (uint32_t)k_drw_g_shift),
370 fs,
371 fd);
372 const uint32_t b = drw_mix(drw_chan(s_drw.color1, 0U), drw_chan(dst, 0U), fs, fd);
373 const uint32_t a = drw_out_alpha(src_a, drw_chan(dst, (uint32_t)k_drw_a_shift));
374
375 return (a << (uint32_t)k_drw_a_shift) | (r << (uint32_t)k_drw_r_shift) |
376 (g << (uint32_t)k_drw_g_shift) | b;
377}

380static uint32_t drw_pack(uint32_t argb, uint32_t bpp)
381{
382 if (bpp == (uint32_t)k_drw_bpp_32) {
383 return argb;
384 }
385 const uint32_t a = drw_chan(argb, (uint32_t)k_drw_a_shift);
386 if (bpp == (uint32_t)k_drw_bpp_8) {
387 return a;
388 }
389 const uint32_t r = drw_chan(argb, (uint32_t)k_drw_r_shift);
390 const uint32_t g = drw_chan(argb, (uint32_t)k_drw_g_shift);
391 const uint32_t b = drw_chan(argb, 0U);
392 const uint32_t lo =
393 (s_drw.control2 >> (uint32_t)k_drw_c2_wfmt_lo_pos) & (uint32_t)k_drw_c2_wfmt_lo_mask;
394 if (lo == (uint32_t)k_drw_wfmt_argb4444) {
395 return ((a >> (uint32_t)k_drw_nibble_shift) << (uint32_t)k_drw_argb4444_a_pos) |
396 ((r >> (uint32_t)k_drw_nibble_shift) << (uint32_t)k_drw_g_shift) |
397 ((g >> (uint32_t)k_drw_nibble_shift) << (uint32_t)k_drw_nibble_shift) |
398 (b >> (uint32_t)k_drw_nibble_shift);
399 }
400 return ((r >> (uint32_t)k_drw_r565_shift) << (uint32_t)k_drw_rgb565_r_pos) |
401 ((g >> (uint32_t)k_drw_g565_shift) << (uint32_t)k_drw_rgb565_g_pos) |
402 (b >> (uint32_t)k_drw_r565_shift);
403}

419static bool drw_render_modelled(uint32_t w, uint32_t h)
420{
421 if (!board_pdctr_graphics_powered()) {
422 return false; /* Dark domain: writes are lost on silicon. */
423 }
424 if ((w == 0U) || (h == 0U) || (s_drw.pitch == 0U) || (s_drw.origin == 0U)) {
425 return false; /* Nothing programmed yet -- e.g. the init ORIGIN write. */
426 }
427 if ((s_drw.control & (uint32_t)k_drw_control_limiter_mask) != 0U) {
428 return false; /* Limiter semantics uncharacterised -- see file header. */
429 }
430 if ((s_drw.cachectl & (uint32_t)k_drw_cachectl_cenablefx) != 0U) {
431 return false; /* FB cache holds the pixels -- see file header. */
432 }
433 const uint32_t sources = (uint32_t)k_drw_c2_patternenable | (uint32_t)k_drw_c2_textureenable;
434 return (s_drw.control2 & sources) == 0U;
435}

438static void drw_render(uc_engine* uc)
439{
440 const uint32_t w = s_drw.size & (uint32_t)k_drw_size_w_mask;
441 const uint32_t h = (s_drw.size >> (uint32_t)k_drw_size_h_pos) & (uint32_t)k_drw_size_w_mask;
442
443 if (!drw_render_modelled(w, h)) {
444 if ((w != 0U) && (h != 0U) && (s_drw.origin != 0U)) {
445 s_drw.skipped++;
446 }
447 return;
448 }
449
450 const uint32_t bpp = drw_bpp();
451 for (uint32_t row = 0U; row < h; ++row) {
452 const uint64_t line =
453 (uint64_t)s_drw.origin + ((uint64_t)row * (uint64_t)s_drw.pitch * (uint64_t)bpp);
454 for (uint32_t col = 0U; col < w; ++col) {
455 const uint64_t addr = line + ((uint64_t)col * (uint64_t)bpp);
456 uint32_t dst = 0U;
457 (void)uc_mem_read(uc, addr, &dst, (size_t)bpp);
458 const uint32_t out = drw_pack(drw_shade(dst), bpp);
459 (void)uc_mem_write(uc, addr, &out, (size_t)bpp);
460 }
461 }
462 s_drw.renders++;
463 s_drw.last_w = w;
464 s_drw.last_h = h;
465}

468static uint64_t drw_read(uc_engine* uc, uint64_t addr, unsigned size)
469{
470 (void)uc;
471 (void)size;
472 const uint64_t off = addr - (uint64_t)k_drw_base;
473 switch (off) {
474 case (uint64_t)k_drw_off_control:
475 /* HUM Ch 62.2.5 "STATUS: Status Control Register" p 3695. The model
476 * rasterizes synchronously on the ORIGIN write, so the engine is never
477 * observably busy and latches no IRQ or bus error: STATUS reads
478 * all-zero and ra8_drw_wait_idle succeeds on its first poll.
479 *
480 * This MUST return explicitly rather than fall through to HWREVISION:
481 * the two registers are read aliases at different offsets carrying
482 * different values, and sharing an arm made ra8_drw_wait_idle poll
483 * STATUS, see the revision stamp, never observe idle, and time out. */
484 return 0U;
485 case (uint64_t)k_drw_off_control2:
486 /* HUM Ch 62.2.6 "HWREVISION: Hardware Revision Register" p 3696.
487 * Reads 0 while the graphics power domain is gated (HUM Ch 11.2.14
488 * p 452) and 0x0FBE0107 once it is powered -- both bench-captured, so
489 * firmware that skips the power-domain bring-up sees the dead register
490 * here exactly as it does on the bench. */
491 if (!board_pdctr_graphics_powered()) {
492 return 0U;
493 }
494 return (uint64_t)k_drw_hwrevision_value;
495 default:
496 /* Every other DRW register is write-only on silicon (HUM Ch 62.2) and
497 * the driver shadows what it needs, so a deterministic zero is both
498 * harmless and faithful. */
499 return 0U;
500 }
501}

504static bool drw_latch(uint64_t off, uint32_t val)
505{
506 switch (off) {
507 case (uint64_t)k_drw_off_control:
508 s_drw.control = val;
509 return true;
510 case (uint64_t)k_drw_off_control2:
511 s_drw.control2 = val;
512 return true;
513 case (uint64_t)k_drw_off_color1:
514 s_drw.color1 = val;
515 return true;
516 case (uint64_t)k_drw_off_color2:
517 s_drw.color2 = val;
518 return true;
519 case (uint64_t)k_drw_off_size:
520 s_drw.size = val;
521 return true;
522 case (uint64_t)k_drw_off_pitch:
523 s_drw.pitch = val;
524 return true;
525 case (uint64_t)k_drw_off_cachectl:
526 s_drw.cachectl = val;
527 return true;
528 default:
529 /* Limiter, texture, CLUT and IRQ registers are accepted and discarded:
530 * the modelled render path does not consume them. */
531 return false;
532 }

563static void drw_run_dlist(uc_engine* uc, uint32_t dlist_addr)
564{
565 uint32_t addr = dlist_addr;
566 for (uint32_t fetched = 0U; fetched < (uint32_t)k_drw_dlr_max_words; ++fetched) {
567 uint32_t tag = 0U;
568 (void)uc_mem_read(uc, (uint64_t)addr, &tag, sizeof(tag));
569 addr += (uint32_t)k_drw_dlr_bytes_per_word;
570
571 if ((tag & (uint32_t)k_drw_dlr_b1_boundary) != 0U) {
572 /* One-index entry: byte 0 is a register, one value word follows. */
573 const uint32_t index = tag & (uint32_t)k_drw_dlr_index_mask;
574 if (index == (uint32_t)k_drw_dlr_idx_dliststart) {
575 return; /* display-list jump / stop */
576 }
577 uint32_t val = 0U;
578 (void)uc_mem_read(uc, (uint64_t)addr, &val, sizeof(val));
579 addr += (uint32_t)k_drw_dlr_bytes_per_word;
580 drw_dlr_exec_reg(uc, index, val);
581 continue;
582 }
583 if ((tag & (uint32_t)k_drw_dlr_index_mask) == (uint32_t)k_drw_dlr_eol_low) {
584 /* End-of-list word: byte 1 is the argument. The builder emits only the
585 * pipeline+cache wait (2, a synchronous no-op here); every other value
586 * terminates. */
587 const uint32_t arg =
588 (tag >> (uint32_t)k_drw_dlr_eol_arg_pos) & (uint32_t)k_drw_dlr_index_mask;
589 if (arg == (uint32_t)k_drw_dlr_arg_wait) {
590 continue;
591 }
592 return; /* terminate */
593 }
594 return; /* unmodelled multi-index packing: stop, do not guess */
595 }
596}

599static void drw_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
600{
601 (void)size;
602 const uint64_t off = addr - (uint64_t)k_drw_base;
603 const uint32_t val = (uint32_t)value;
604
605 /* While the graphics power domain is gated the block is unpowered and every
606 * write is LOST on silicon (HUM Ch 11.2.14 p 452), so the shadow must not
607 * latch it either -- otherwise firmware that skips the bring-up would
608 * render here and not on the bench. */
609 if (!board_pdctr_graphics_powered()) {
610 return;
611 }
612
613 if (off == (uint64_t)k_drw_off_origin) {
614 /* HUM Ch 62.2.31 p 3705: writing ORIGIN both anchors the bounding box and
615 * TRIGGERS the render. */
616 s_drw.origin = val;
617 drw_render(uc);
618 return;
619 }
620 if (off == (uint64_t)k_drw_off_dliststart) {
621 /* HUM Ch 62.2.32 p 3705: writing DLISTSTART kicks the display-list reader,
622 * which executes the list from RAM (issue #247 loop-stable clear+fill). */
623 drw_run_dlist(uc, val);
624 return;
625 }
626 (void)drw_latch(off, val);
627}

636static void drw_report(void)
637{
638 if ((s_drw.renders == 0U) && (s_drw.skipped == 0U)) {
639 return;
640 }
641 if (s_drw.skipped != 0U) {
642 /* Loud: an app reached a DRW configuration this model declines to
643 * reproduce, so the emulated framebuffer is NOT what the bench shows. */
644 (void)fprintf(stderr,
645 " DRW : %u boxes rasterized (last %ux%u), "
646 "%u DECLINED (limiter / FB-cache / textured path unmodelled)\n",
647 s_drw.renders,
648 s_drw.last_w,
649 s_drw.last_h,
650 s_drw.skipped);
651 return;
652 }
653 (void)fprintf(stderr,
654 " DRW : %u boxes rasterized (last %ux%u)\n",
655 s_drw.renders,
656 s_drw.last_w,
657 s_drw.last_h);
658}
