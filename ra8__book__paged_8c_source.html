113static ra8_err_t
114priv_book_src_read_paged(const ra8_book_src_t* src, uint32_t off, void* dst, uint32_t len)
115{
116 uint8_t* d = (uint8_t*)dst;
117 uint32_t cur = off;
118 uint32_t remaining = len;
119 const uint32_t max_iter = (len / src->frame_bytes) + (uint32_t)k_ra8_book_paged_span_pad;
120 for (uint32_t it = 0U; (it < max_iter) && (remaining > 0U); ++it) {
121 void* page = nullptr;
122 const ra8_err_t ge = ra8_vmem_get(src->vm, src->object_id, (uint64_t)cur, &page);
123 if (ge != k_ra8_ok) {
124 return ge;
125 }
126 const uint32_t in_frame = cur % src->frame_bytes;
127 const uint32_t avail = src->frame_bytes - in_frame;
128 const uint32_t n = (remaining < avail) ? remaining : avail;
129 (void)memcpy(d, (const uint8_t*)page + in_frame, n);
130 const ra8_err_t pe = ra8_vmem_put(src->vm, page);
131 if (pe != k_ra8_ok) {
132 return pe;
133 }
134 d += n;
135 cur += n;
136 remaining -= n;
137 }
138 return (remaining == 0U) ? k_ra8_ok : k_ra8_err_out_of_range;
139}

221RA8_INTERNAL
222static ra8_err_t priv_book_image_row(const ra8_book_src_t* src,
223 const ra8_book_image_t* img,
224 uint32_t x,
225 uint32_t ry,
226 uint32_t w,
227 uint8_t* out)
228{
229 const uint32_t row_flat = (ry * (uint32_t)img->width) + x;
230 uint8_t packed[(k_ra8_book_rect_chunk_px / k_ra8_book_gray4_ppb) + 1U];
231 uint32_t done = 0U;
232 const uint32_t max_iter =
233 (w / (uint32_t)k_ra8_book_rect_chunk_px) + (uint32_t)k_ra8_book_rect_chunk_pad;
234 for (uint32_t it = 0U; it < max_iter; ++it) {
235 if (done >= w) {
236 break; /* whole run unpacked; the count bound above is the loop invariant */
237 }
238 const uint32_t remain = w - done;
239 const uint32_t cpx =
240 (remain < (uint32_t)k_ra8_book_rect_chunk_px) ? remain : (uint32_t)k_ra8_book_rect_chunk_px;
241 const uint32_t flat0 = row_flat + done;
242 const uint32_t first = flat0 >> 1U;
243 const uint32_t last = (flat0 + cpx - 1U) >> 1U;
244 const uint32_t n = (last - first) + 1U;
245 const uint64_t off =
246 (uint64_t)src->hdr.image_pool_off + (uint64_t)img->data_off + (uint64_t)first;
247 if ((off + (uint64_t)n) > (uint64_t)src->size) {
248 return k_ra8_err_out_of_range; /* descriptor / geometry points past the blob */
249 }
250 const ra8_err_t re = ra8_book_src_read(src, (uint32_t)off, packed, n);
251 if (re != k_ra8_ok) {
252 return re;
253 }
254 for (uint32_t i = 0U; i < cpx; ++i) {
255 const uint32_t flat = flat0 + i;
256 const uint8_t byte = packed[(flat >> 1U) - first];
257 const uint8_t nib = ((flat & 1U) != 0U) ? (uint8_t)(byte & (uint8_t)k_ra8_book_gray4_nib_lo)
258 : (uint8_t)(byte >> (uint8_t)k_ra8_book_gray4_nib_sh);
259 out[done + i] = (uint8_t)((nib << (uint8_t)k_ra8_book_gray4_nib_sh) | nib);
260 }
261 done += cpx;
262 }
263 return (done == w) ? k_ra8_ok : k_ra8_err_out_of_range;
264}

329ra8_err_t ra8_book_src_image_rect(const ra8_book_src_t* src,
330 const ra8_book_image_t* img,
331 uint32_t x,
332 uint32_t y,
333 uint32_t w,
334 uint32_t h,
335 uint8_t* out,
336 uint32_t out_stride)
337{
338 RA8_CHECK_NULL_PTR(src, s_tag_paged, "image_rect: null src");
339 RA8_CHECK_NULL_PTR(img, s_tag_paged, "image_rect: null img");
340 RA8_CHECK_NULL_PTR(out, s_tag_paged, "image_rect: null out");
341 /* Readable raster of a known depth (one decision, single exit -- MISRA 15.5):
342 * SVG has no pixel grid, and only the two grayscale depths are unpackable --
343 * gray4 nibble-unpacks 2px/byte, gray8 copies 1px/byte verbatim (the retained
344 * full-resolution continuous-tone source, #343/#476). ra8_book_validate()
345 * already refuses any other depth; this also guards a corrupt descriptor. */
346 const ra8_book_image_pixfmt_t pf = ra8_book_image_pixfmt(img);
347 if ((img->format != (uint8_t)k_ra8_book_image_gray4) ||
348 ((pf != k_ra8_book_pixfmt_gray4) && (pf != k_ra8_book_pixfmt_gray8))) {
349 return k_ra8_err_invalid_arg;
350 }
351 if ((w == 0U) || (h == 0U) || (out_stride < w)) {
352 return k_ra8_err_invalid_arg;
353 }
354 if ((((uint64_t)x + (uint64_t)w) > (uint64_t)img->width) ||
355 (((uint64_t)y + (uint64_t)h) > (uint64_t)img->height)) {
356 return k_ra8_err_out_of_range; /* sub-rect not fully inside the image */
357 }
358 const bool is_gray8 = (pf == k_ra8_book_pixfmt_gray8);
359 for (uint32_t row = 0U; row < h; ++row) {
360 uint8_t* const dst = &out[(size_t)row * (size_t)out_stride];
361 const ra8_err_t re = is_gray8 ? priv_book_image_row_gray8(src, img, x, y + row, w, dst)
362 : priv_book_image_row(src, img, x, y + row, w, dst);
363 if (re != k_ra8_ok) {
364 return re;
365 }
366 }
367 return k_ra8_ok;
368}

502RA8_INTERNAL
503static bool ra8_book_paged_emit_run(const ra8_book_src_t* src,
504 uint32_t abs_off,
505 char* out,
506 size_t cap,
507 size_t* pos,
508 bool* at_break,
509 ra8_err_t* io_err)
510{
511 char buf[k_ra8_book_paged_strbuf];
512 uint32_t off = abs_off;
513 const uint32_t max_iter =
514 (src->size / (uint32_t)(k_ra8_book_paged_strbuf - 1U)) + (uint32_t)k_ra8_book_paged_pad;
515 for (uint32_t it = 0U; it < max_iter; ++it) {
516 if (off >= src->size) {
517 return true; /* Ran to blob end without a NUL (degenerate); treat as done. */
518 }
519 const uint32_t remain = src->size - off;
520 const uint32_t want = (uint32_t)(k_ra8_book_paged_strbuf - 1U);
521 const uint32_t chunk = (remain < want) ? remain : want;
522 const ra8_err_t re = ra8_book_src_read(src, off, buf, chunk);
523 if (re != k_ra8_ok) {
524 *io_err = re;
525 return false;
526 }
527 uint32_t nlen = 0U;
528 while ((nlen < chunk) && (buf[nlen] != '\0')) {
529 ++nlen;
530 }
531 buf[nlen] = '\0';
532 if (!ra8_book_emit_text(out, cap, pos, buf, at_break)) {
533 return false; /* Output overflow. */
534 }
535 if (nlen < chunk) {
536 return true; /* Hit the real NUL terminator. */
537 }
538 off += nlen; /* nlen == chunk: advance and read the next chunk. */
539 }
540 return true;
541}

575RA8_INTERNAL
576static bool priv_paged_visit_node(const ra8_book_src_t* src,
577 uint32_t n,
578 char* out,
579 size_t cap,
580 size_t* pos,
581 bool* at_break,
582 uint32_t* stack,
583 uint32_t* sp,
584 ra8_err_t* io_err)
585{
586 ra8_book_node_t node = {};
587 const ra8_err_t ne = ra8_book_paged_node(src, n, &node);
588 if (ne != k_ra8_ok) {
589 *io_err = ne;
590 return false;
591 }
592 if (*sp >= k_ra8_book_xhtml_stack) {
593 return false;
594 }
595 stack[(*sp)++] = node.next_sibling; /* sibling chain after this subtree */
596 if (node.kind == (uint8_t)k_ra8_book_node_text) {
597 return ra8_book_paged_emit_run(src,
598 src->hdr.string_off + node.text_off,
599 out,
600 cap,
601 pos,
602 at_break,
603 io_err);
604 }
605 char tag[k_ra8_book_paged_tagbuf] = {};
606 const ra8_err_t te = ra8_book_paged_str_short(src,
607 src->hdr.string_off + node.name_off,
608 tag,
609 (uint32_t)k_ra8_book_paged_tagbuf);
610 if (te != k_ra8_ok) {
611 *io_err = te;
612 return false;
613 }
614 bool ok = true;
615 if (ra8_book_is_block(tag)) {
616 ok = ra8_book_emit_break(out, cap, pos, at_break);
617 }
618 if (ok && (*sp < k_ra8_book_xhtml_stack)) {
619 stack[(*sp)++] = node.first_child; /* descend, pre-order */
620 }
621 return ok;
622}

652RA8_INTERNAL
653static bool ra8_book_walk_text_paged(const ra8_book_src_t* src,
654 uint32_t root,
655 uint32_t node_count,
656 char* out,
657 size_t cap,
658 size_t* pos,
659 ra8_err_t* io_err)
660{
661 /* Explicit DFS stack (2 KiB) kept in module-static storage so this frame
662 * stays within the stack-usage budget; iterative (no recursion) and
663 * single-threaded, so the shared buffer never overlaps. */
664 static uint32_t s_paged_stack[k_ra8_book_xhtml_stack];
665 uint32_t* stack = s_paged_stack;
666 uint32_t sp = 0U;
667 bool ok = true;
668 bool at_break = true;
669 stack[sp++] = root;
670
671 const uint32_t max_iter = (node_count * k_ra8_book_xhtml_iter_x) + k_ra8_book_xhtml_stack;
672 uint32_t guard = 0U;

674 ++guard;

679 ok = priv_paged_visit_node(src, n, out, cap, pos, &at_break, stack, &sp, io_err);
