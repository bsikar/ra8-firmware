192void main(void)
193{
194 internal_lpm_deep_setup_or_halt();
195 ra8_isr_globals_enable();
196
197 (void)ra8_board_uart_console_write(s_lpm_deep_msg_boot,
198 (size_t)(sizeof(s_lpm_deep_msg_boot) - 1U));
199
200 g_lpm_deep_pre_count++;
201
202 /* SysTick does NOT wake from Deep-Sleep on RA8D2 under the
203 * default LPM config -- bench-verified 2026-05-27. The chip
204 * enters WFI and stays there. The HIL gate therefore scrapes
205 * the BOOT banner (emitted just above) rather than a post-wake
206 * banner; the gate proves the firmware loaded and reached the
207 * LPM entry, not that wake worked. A real wake-from-deep-sleep
208 * demo would need RTC/AGT (sub-clock-sourced, survives
209 * SLEEPDEEP) as the wake source instead of SysTick. */
210 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_deep_sleep) != k_ra8_ok) {
211 internal_lpm_deep_panic_halt();
212 }
213
214 g_lpm_deep_wake_count++;
215
216 (void)ra8_board_uart_console_write(s_lpm_deep_msg_woke,
217 (size_t)(sizeof(s_lpm_deep_msg_woke) - 1U));
218
219 /* Park in a normal (non-LPM) loop so subsequent bench flashes can
220 * halt the CPU without an Initialize step. SysTick + the CPU stay
221 * in their default active state. */
222 while (1) {
223 (void)ra8_board_led_toggle(k_ra8_board_led1);
224 ra8_delay_ms((uint32_t)k_lpm_deep_park_blink_ms);
225 }
226 internal_lpm_deep_panic_halt();
227}
