221void ra8_ceu_program_geometry(const ra8_ceu_config_t* cfg)

226 *ra8_ceu_reg32(k_ra8_ceu_off_cmcyr) = cmcyr;
227
228 /* HUM Ch 60.2.5 "CAMOR : Camera Interface Offset Register" p 3641 */
229 const uint32_t camor =

231 *ra8_ceu_reg32(k_ra8_ceu_off_camor) = camor;
232
233 /* HUM Ch 60.2.6 "CAPWR : Camera Interface Width Register" p 3643 */
234 uint16_t hwdth = cfg->width_px;
235 if (cfg->x_capture_px != 0U) {
236 hwdth = cfg->x_capture_px;
237 }
238 uint16_t vwdth = cfg->height_px;
239 if (cfg->y_capture_lines != 0U) {
240 vwdth = cfg->y_capture_lines;
241 }
242 const uint32_t capwr = (((uint32_t)vwdth) << k_ra8_ceu_capwr_shift_vwdth) | (uint32_t)hwdth;
243 *ra8_ceu_reg32(k_ra8_ceu_off_capwr) = capwr;
