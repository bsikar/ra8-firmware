4"""Pre-commit gate: ban AI attribution anywhere in the codebase.
5
6Policy (see ``docs/AI_ATTRIBUTION_POLICY.md`` and ``CLAUDE.md``):
7
8The literal token ``Claude`` is reserved EXCLUSIVELY for the project's own
9``CLAUDE.md`` filename. No file in ``libs/``, ``src/``, ``tests/``,
10``examples/``, ``port/``, ``scripts/``, ``docs/`` -- nor any other tracked
11file -- may attribute code, tests, docs, or any other artifact to an AI tool
12(Claude, GPT, Anthropic, OpenAI, Copilot, etc.).
13
14Per-line opt-out: append ``AI-OK: <reason>`` (case-sensitive) to the line
15that legitimately needs to mention the banned token (e.g. policy text in
16``CLAUDE.md`` itself or ``CONTRIBUTING.md``).
17
18Exit status:
19 * 0 -- no violations found
20 * 1 -- one or more violations (printed in ``<path>:<line>: ...`` form)
21"""

99# ---------------------------------------------------------------------------
100# Detection rules
101# ---------------------------------------------------------------------------
102
103# 1. ``claude`` -- case-insensitive word match, BUT allow:
104# - the literal filename ``CLAUDE.md`` (any case)
105# - paths that include ``CLAUDE.md``
106# - the ``.claude`` config directory (e.g. ``.claude``, ``./.claude/``)
107RX_CLAUDE = re.compile(r"\bclaude\b", re.IGNORECASE)
108RX_CLAUDE_MD_OK = re.compile(r"\bclaude\.md\b", re.IGNORECASE)
109RX_DOT_CLAUDE_OK = re.compile(r"(?<![A-Za-z0-9_])\.claude(?![A-Za-z0-9_])", re.IGNORECASE)
110
111# 2. ``gpt`` -- only flag suspicious AI-style usages. Renesas RA8D2 has
112# a GPT (general PWM timer) peripheral with channels ``GPT0``..``GPT13``
113# plus PDG sub-channels ``GPT320``..``GPT323``, so bare ``GPT``,
114# ``GPTn`` and ``ra8_gpt*`` are all legitimate hardware references.
115# Flag only the OpenAI product family (``ChatGPT``, ``GPT-3``,
116# ``GPT-4``, ``GPT-3.5``, ``GPT-4o``, ``OpenAI GPT``). Keep the dash
117# requirement -- ``GPT3`` without a separator is the Renesas channel.
118RX_GPT_BAD = re.compile(
119 r"\b(?:chatgpt|openai\s+gpt|gpt-[0-9](?:\.[0-9]+)?[a-z]*)\b",
120 re.IGNORECASE,
121)
122
123# 3. ``anthropic`` -- always banned (no legitimate use).
124RX_ANTHROPIC = re.compile(r"\banthropic\b", re.IGNORECASE)
125
126# 4. ``openai`` -- always banned.
127RX_OPENAI = re.compile(r"\bopenai\b", re.IGNORECASE)
128
129# 5. ``copilot`` -- always banned.
130RX_COPILOT = re.compile(r"\bcopilot\b", re.IGNORECASE)

189# 6b. AI-bot email-address domains and usernames. Matches both
190# ``Author: ... <foo@anthropic.com>`` and bare email mentions in
191# prose / commit bodies. The ``[bot]`` user-suffix is the GitHub
192# convention for app-installed bots (``copilot[bot]``, etc.) and is
193# a strong signal that the commit was authored by an AI assistant.

219# Match ANY ``[bot]`` suffix. The GitHub convention is that app-
220# installed accounts append ``[bot]`` to their login name; legitimate
221# CI bots (``github-actions[bot]``, ``dependabot[bot]``) and AI bots
222# (``claude[bot]``, ``copilot[bot]``, ``cursor[bot]``) both use this
223# suffix. Project policy: zero bot authorship. Per-line ``AI-OK:``
224# opt-out is the escape hatch when a bot commit is genuinely desired.
225RX_AI_BOT_USER = re.compile(r"\‍[bot\‍]\b")

235 return OPT_OUT_TAG in line

238def _claude_violation(line: str) -> str | None:
239 """Return matched token or None.
240
241 A line containing ``claude`` only via ``CLAUDE.md`` (filename) or the
242 ``.claude`` config directory is allowed.
243 """
244 if not RX_CLAUDE.search(line):
245 return None
246 # Strip the allowed forms and re-check.
247 stripped = RX_CLAUDE_MD_OK.sub("", line)
248 stripped = RX_DOT_CLAUDE_OK.sub("", stripped)
249 m = RX_CLAUDE.search(stripped)
250 return m.group(0) if m else None

253def _scan_line(line: str) -> list[str]:
254 """Return list of offending tokens found on this line."""
255 hits: list[str] = []
256 tok = _claude_violation(line)
257 if tok:
258 hits.append(tok)
259 for rx in (
260 RX_GPT_BAD,
261 RX_ANTHROPIC,
262 RX_OPENAI,
263 RX_COPILOT,
264 RX_OTHER_BRANDS,
265 RX_AI_EMAIL,
266 RX_AI_BOT_USER,
267 RX_COAUTH,
268 RX_GENERATED,
269 ):
270 m = rx.search(line)
271 if m:
272 hits.append(m.group(0))
273 return hits

276def _in_scan_scope(rel: str) -> bool:
277 """Whether repo-relative ``rel`` is a text file this ban inspects.
278
279 Pure and total, so the selftest can assert scope on synthetic paths without
280 touching the tree: a re-narrowing that drops ``tools/`` fails the selftest
281 instead of passing silently.
282
283 ``first_party_paths`` has already removed vendored SOUP and build output;
284 this adds the extension filter and the generated-doc-tree skip that git
285 itself may not exclude.
286 """
287 if Path(rel).suffix.lower() not in TEXT_EXTS:
288 return False
289 wrapped = "/" + rel + "/"
290 return not any(frag in wrapped for frag in SKIP_FRAGMENTS)

293def _iter_files() -> Iterable[Path]:
294 """Every tracked first-party text file, derived from git ls-files.
295
296 The universal AI-attribution ban has no directory allowlist by design: the
297 scope is the whole tree minus vendored SOUP, generated tables and build
298 output. ``respect_language_excludes=False`` keeps a vendored tree's *build
299 glue* (e.g. a first-party CMakeLists under port/threadx/) in view; the file
300 the ban actually cares about is any tracked prose, wherever it lives.
301 """
302 text_suffixes = tuple(sorted(TEXT_EXTS))
303 for rel in first_party_paths(text_suffixes, respect_language_excludes=False):
304 if _in_scan_scope(rel):
305 yield REPO_ROOT / rel

308def _scan_file(path: Path) -> list[tuple[int, str, str]]:
309 """Return list of ``(lineno, token, excerpt)`` violations."""
310 out: list[tuple[int, str, str]] = []
311 try:
312 with path.open("r", encoding="utf-8", errors="replace") as fh:
313 for i, line in enumerate(fh, start=1):
314 if _line_has_opt_out(line):
315 continue
316 hits = _scan_line(line)
317 out.extend((i, tok, line.rstrip("\n")) for tok in hits)
318 except OSError:
319 return []
320 return out

343def _scan_text(text: str) -> list[str]:
344 """Every offending token in ``text``, honouring the per-line AI-OK opt-out."""
345 hits: list[str] = []
346 for line in text.splitlines():
347 if _line_has_opt_out(line):
348 continue
349 hits.extend(_scan_line(line))
350 return hits

353def selftest() -> int:
354 """Prove the ban fires, stays quiet on legal prose, and now enumerates tools/."""
355 print("check_no_ai_attribution.py --selftest")
356 failures: list[str] = []
357 expect(bool(_scan_text(_BAD_FIXTURE)), "bad fixture reports AI attribution", failures)
358 expect(not _scan_text(_GOOD_FIXTURE), "legal-but-tricky fixture stays silent", failures)
359
360 expect(
361 _in_scan_scope("tools/ra8_emulator/src/main.c"),
362 "tools/ is in scope (SCAN_DIRS omitted it before #358)",
363 failures,
364 )
365 expect(
366 not _in_scan_scope("libs/third_party/miniz/miniz.c"),
367 "vendored SOUP stays out of scope",
368 failures,
369 )
370 scanned = {str(p.relative_to(REPO_ROOT)) for p in _iter_files()}
371 expect(
372 any(s.startswith("tools/") for s in scanned),
373 "the live enumeration actually reaches tools/",
374 failures,
375 )
376 return report(failures)

379def main(argv: list[str]) -> int:
380 """Scan every in-scope file for AI-attribution tokens and report each hit.
381
382 Exemption works at two levels, and they are not interchangeable. A handful
383 of files -- the policy document, this checker -- must spell the banned
384 tokens on nearly every line, so they are exempted WHOLE, by resolved path,
385 to avoid littering them with per-line tags. Everywhere else the only
386 escape is a per-line ``AI-OK: <reason>`` on the quoting line itself.
387
388 Paths are compared after ``resolve()`` so a symlinked or relative spelling
389 of an exempt file still matches; comparing the raw path would let the same
390 file be scanned or skipped depending on how it was reached.
391
392 Returns 1 with each hit printed as ``path:line``, 0 when the tree is clean.
393 """
394 if "--selftest" in argv[1:]:
395 return selftest()
396 # Self-exempt files legitimately spell the banned tokens; they are
397 # matched explicitly so every internal line need not carry AI-OK.
398 self_exempt_resolved = {q.resolve() for q in SELF_EXEMPT_FILES}
399
400 violations = 0
401 for p in _iter_files():
402 if p.resolve() in self_exempt_resolved:
403 continue
404 for lineno, tok, excerpt in _scan_file(p):
405 rel = p.relative_to(REPO_ROOT)
406 # Trim long excerpts.
407 display = (
408 excerpt
409 if len(excerpt) <= EXCERPT_MAX_LEN
410 else excerpt[:EXCERPT_TRUNCATE_LEN] + "..."
411 )
412 print(f"{rel}:{lineno}: AI attribution found ('{tok}'): {display}")
413 violations += 1
414
415 if violations:
416 print(f"\n[FAIL] {violations} AI-attribution violation(s).", file=sys.stderr)
417 print(" See docs/AI_ATTRIBUTION_POLICY.md. Use a per-line", file=sys.stderr)
418 print(" 'AI-OK: <reason>' tag only when quoting policy text.", file=sys.stderr)
419 return 1
420 return 0
