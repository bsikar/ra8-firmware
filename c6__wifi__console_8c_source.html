65void c6_wifi_put_u32(uint32_t value)
66{
67 uint8_t digits[k_c6_wifi_dec_digits] = {};
68 uint32_t count = 0U;
69 uint32_t rest = value;
70 do {
71 digits[count] = (uint8_t)('0' + (uint8_t)(rest % (uint32_t)k_c6_wifi_dec_radix));
72 rest = rest / (uint32_t)k_c6_wifi_dec_radix;
73 count++;
74 } while ((rest != 0U) && (count < (uint32_t)k_c6_wifi_dec_digits));
75
76 uint8_t out[k_c6_wifi_dec_digits] = {};
77 for (uint32_t i = 0U; i < count; i++) {
78 out[i] = digits[count - 1U - i];
79 }
80 (void)ra8_board_uart_console_write(out, (size_t)count);
81}

94void c6_wifi_put_hex(uint32_t value, uint8_t digits)
95{
96 if ((digits == 0U) || (digits > (uint8_t)k_c6_wifi_hex_digits)) {
97 return;
98 }
99 uint8_t out[k_c6_wifi_hex_digits] = {};
100 for (uint8_t i = 0U; i < digits; i++) {
101 const uint8_t shift = (uint8_t)((digits - 1U - i) * (uint8_t)k_c6_wifi_hex_bits);
102 const uint8_t nibble = (uint8_t)((value >> shift) & (uint32_t)k_c6_wifi_hex_mask);
103 out[i] = (nibble < (uint8_t)k_c6_wifi_hex_alpha)
104 ? (uint8_t)('0' + nibble)
105 : (uint8_t)('a' + (uint8_t)(nibble - (uint8_t)k_c6_wifi_hex_alpha));
106 }
107 (void)ra8_board_uart_console_write(out, (size_t)digits);
108}

111void c6_wifi_put_text(const char* text, size_t len)
112{
113 if ((text == nullptr) || (len == 0U)) {
114 return;
115 }
116 const size_t take = (len < (size_t)k_c6_wifi_text_max) ? len : (size_t)k_c6_wifi_text_max;
117 uint8_t out[k_c6_wifi_text_max] = {};
118 for (size_t i = 0U; i < take; i++) {
119 const uint8_t byte = (uint8_t)text[i];
120 const bool printable =
121 (byte >= (uint8_t)k_c6_wifi_ascii_low) && (byte <= (uint8_t)k_c6_wifi_ascii_high);
122 out[i] = printable ? byte : (uint8_t)k_c6_wifi_ascii_sub;
123 }
124 (void)ra8_board_uart_console_write(out, take);
125}
