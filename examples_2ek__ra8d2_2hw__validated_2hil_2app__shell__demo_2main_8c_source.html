522static bool app_shell_register(void)
523{
524 if (ra8_app_registry_init(&s_reg, s_slots, (uint16_t)k_app_reg_cap) != k_ra8_ok) {
525 return false;
526 }
527 if (ra8_app_nav_init(&s_nav, &s_reg, s_nav_stack, (uint16_t)k_app_nav_cap) != k_ra8_ok) {
528 return false;
529 }
530 bool ok =
531 app_shell_add(&s_library_app, &s_library_ctx, (uint16_t)k_app_id_library, "library", false);
532 if (ok) {
533 ok = app_shell_add(&s_reader_app, &s_reader_ctx, (uint16_t)k_app_id_reader, "reader", false);
534 }
535#if APP_SHELL_SETTINGS
536 if (ok) {
537 ok = app_shell_add(&s_settings_app,
538 &s_settings_ctx,
539 (uint16_t)k_app_id_settings,
540 "settings",
541 true);
542 }
543#endif
544 return ok;
545}

855static bool app_shell_selfcheck(void)
856{
857 if (!app_shell_launch_library()) {
858 return false;
859 }

861 return false;

904static void app_shell_banner(void)
905{
906 uint16_t n = 0U;
907 (void)ra8_app_count(&s_reg, &n);
908 ra8_log_info_val("app_shell", "apps", (uint32_t)n);
909 ra8_log_info_val("app_shell", "library enters", s_library_ctx.enters);
910 ra8_log_info("app_shell", "app_shell_demo PASS");
911 /* Additive HIL banner: the Pi rig has no ITM/SWO capture, so mirror the PASS

929int32_t main(void)
930{
931 ra8_log_init();
932 ra8_log_info("app_shell", "boot");
933
934 /* Bring up the VCOM console so the success path can emit the HIL banner.
935 * Best-effort: a failure is narrated over ITM and the demo continues. */
936 if (!app_shell_hil_console_init()) {
937 ra8_log_info("app_shell", "VCOM console init failed -- HIL banner unavailable");
938 }
939
940 bool ok = app_shell_register();
941 if (!ok) {
942 ra8_log_info("app_shell", "FAIL register");
943 }
944 if (ok) {
945 ok = app_shell_selfcheck();
946 if (!ok) {
947 ra8_log_info("app_shell", "FAIL selfcheck");
948 }
949 }
950 if (ok) {
951 app_shell_banner();
952 }
953
954 while (1) {
955 __asm__ volatile("wfi");
956 }
957}
