106static size_t priv_skip_comment(const char* css, size_t len, size_t start)
107{
108 const char open_a = '/';
109 const char open_b = '*';
110 size_t j = start + (size_t)k_priv_cmt_marker;
111 /* Bounded: j strictly increases each pass; capped at len. */
112 while (((j + 1U) < len) && !((css[j] == open_b) && (css[j + 1U] == open_a))) {
113 ++j;
114 }
115 const size_t past = j + (size_t)k_priv_cmt_marker;
116 return (past <= len) ? past : len;
117}

144static bool
145priv_find_block(const char* css, size_t len, size_t i, size_t* out_open, size_t* out_close)
146{
147 const char open_c = '{';
148 const char close_c = '}';
149 size_t brace = i;
150 /* Bounded: brace strictly increases each pass; capped at len. */
151 while ((brace < len) && (css[brace] != open_c)) {
152 ++brace;
153 }
154 if (brace >= len) {
155 return false;
156 }
157 size_t close = brace + 1U;
158 /* Bounded: close strictly increases each pass; capped at len. */
159 while ((close < len) && (css[close] != close_c)) {
160 ++close;
161 }
162 *out_open = brace;
163 *out_close = close;
164 return true;
165}

167ra8_err_t ra8_css_parse(ra8_css_sheet_t* sheet, const char* css, uint32_t len)
168{
169 if ((sheet == nullptr) || (css == nullptr)) {
170 return k_ra8_err_null_ptr;
171 }
172 const char open_a = '/';
173 const char open_b = '*';
174 size_t i = 0U;
175 /* Bounded: each iteration advances past a block or breaks at EOF. */
176 while (i < (size_t)len) {
177 if (((i + 1U) < (size_t)len) && (css[i] == open_a) && (css[i + 1U] == open_b)) {
178 i = priv_skip_comment(css, (size_t)len, i);
179 continue;
180 }
181 if (ra8_reflow_css_is_ws(css[i])) {
182 ++i;
183 continue;
184 }
185 size_t brace = 0U;
186 size_t close = 0U;
187 if (!priv_find_block(css, (size_t)len, i, &brace, &close)) {
188 break; /* no block -> done */
189 }
190 ra8_reflow_css_parse_one_block(sheet,
191 &css[i],
192 brace - i,
193 &css[brace + 1U],
194 close - (brace + 1U));
195 i = (close < (size_t)len) ? (close + 1U) : (size_t)len;
196 }
197 return k_ra8_ok;
198}

232static bool priv_class_list_has(const char* list, size_t list_len, const char* name, size_t nlen)
233{
234 size_t i = 0U;
235 /* Bounded: each pass skips >=0 ws then a token, advancing i to list_len. */
236 while (i < list_len) {
237 while ((i < list_len) && ra8_reflow_css_is_ws(list[i])) {
238 ++i;
239 }
240 size_t start = i;
241 while ((i < list_len) && !ra8_reflow_css_is_ws(list[i])) {
242 ++i;
243 }
244 const size_t tlen = i - start;
245 if ((tlen == nlen) && (memcmp(&list[start], name, nlen) == 0)) {
246 return true;
247 }
248 }
249 return false;
250}

253 const ra8_css_element_t* el,

255{

261 return false;

273 (memcmp(el->id, nm, rule->id_len) != 0)) {
274 return false;

363RA8_INTERNAL
364static bool priv_rule_matches_ctx(const ra8_css_rule_t* rule,
365 const ra8_css_element_t* el,
366 const ra8_css_element_t* ancestors,
367 uint8_t n_anc,
368 const ra8_css_sheet_t* sheet)
369{
370 if (!ra8_css_rule_matches(rule, el, sheet)) {
371 return false;
372 }
373 if (rule->anc_count == 0U) {
374 return true;
375 }
376 /* Greedy right-to-left: match each ancestor part (innermost first) walking up
377 * the stack from the parent. The descendant combinator allows any depth. */
378 int32_t ai = (int32_t)rule->anc_count - 1;
379 int32_t si = (int32_t)n_anc - 1;
380 /* Bounded: si strictly decreases each pass; ends when si < 0 or ai < 0. */
381 while ((ai >= 0) && (si >= 0)) {
382 if (priv_anc_matches(&rule->anc[ai], &ancestors[si], sheet)) {
383 --ai;
384 }
385 --si;
386 }
387 return ai < 0;
388}

415RA8_INTERNAL
416static uint16_t priv_rule_rank(const ra8_css_rule_t* rule)
417{
418 uint16_t spec = 0U;
419 if (rule->sel_tag != (uint8_t)k_ra8_reflow_tag_unknown) {
420 spec = (uint16_t)(spec + (uint16_t)k_priv_spec_type);
421 }
422 if (rule->class_len != 0U) {
423 spec = (uint16_t)(spec + (uint16_t)k_priv_spec_class);
424 }
425 if (rule->id_len != 0U) {
426 spec = (uint16_t)(spec + (uint16_t)k_priv_spec_id);
427 }
428 /* Each descendant ancestor part adds to specificity per CSS. */
429 for (uint8_t a = 0U; a < rule->anc_count; ++a) {
430 if (rule->anc[a].tag != (uint8_t)k_ra8_reflow_tag_unknown) {
431 spec = (uint16_t)(spec + (uint16_t)k_priv_spec_type);
432 }
433 if (rule->anc[a].class_len != 0U) {
434 spec = (uint16_t)(spec + (uint16_t)k_priv_spec_class);
435 }
436 if (rule->anc[a].id_len != 0U) {
437 spec = (uint16_t)(spec + (uint16_t)k_priv_spec_id);
438 }
439 }
440 return spec;
441}

450static const ra8_css_style_t* priv_resolve(uint8_t setbit,
451 const ra8_css_style_t* inherited,
452 const ra8_css_sheet_t* sheet,
453 const bool* matched,
454 const ra8_css_style_t* inl)
455{
456 const ra8_css_style_t* win = nullptr;
457 uint16_t best_rank = 0U;
458 uint16_t best_order = 0U;
459 bool have = false;
460 if ((inherited->set & setbit) != 0U) {
461 win = inherited;
462 best_rank = (uint16_t)k_priv_rank_inherited;
463 have = true;
464 }
465 /* Bounded: rule_count <= k_ra8_css_max_rules; i advances by 1 each step. */
466 for (uint16_t i = 0U; i < sheet->rule_count; ++i) {
467 if (!matched[i] || ((sheet->rules[i].decl.set & setbit) == 0U)) {
468 continue;
469 }
470 const uint16_t rank = priv_rule_rank(&sheet->rules[i]);
471 /* mcdc-deactivated: rules[].order is assigned monotonically in source order (priv_push_rule increments next_order) and this loop scans rules in that same order, so a later same-rank rule always has order > best_order; (order >= best_order) is invariantly true and its false arm is unreachable. */
472 if ((!have) || (rank > best_rank) ||
473 ((rank == best_rank) && (sheet->rules[i].order >= best_order))) {
474 win = &sheet->rules[i].decl;
475 best_rank = rank;
476 best_order = sheet->rules[i].order;
477 have = true;
478 }
479 }
480 if ((inl->set & setbit) != 0U) {
481 win = inl;
482 }
483 return win;
484}

753int16_t ra8_css_match_face(const ra8_css_sheet_t* sheet,
754 const char* family,
755 uint16_t family_len,
756 bool want_bold,
757 bool want_italic)
758{
759 if ((sheet == nullptr) || (family == nullptr) || (family_len == 0U)) {
760 return (int16_t)k_ra8_css_no_face;
761 }
762 int16_t fallback = (int16_t)k_ra8_css_no_face;
763 /* Bounded: face_count <= k_ra8_css_max_faces; i advances by 1 each step. */
764 for (uint16_t i = 0U; i < sheet->face_count; ++i) {
765 const ra8_css_fontface_t* f = &sheet->faces[i];
766 if (!priv_family_eq(sheet, f, family, (size_t)family_len)) {
767 continue;
768 }
769 if (((f->weight_bold != 0U) == want_bold) && ((f->style_italic != 0U) == want_italic)) {
770 return (int16_t)i;
771 }
772 if ((f->weight_bold == 0U) && (f->style_italic == 0U) && (fallback < 0)) {
773 fallback = (int16_t)i;
774 }
775 }
776 return fallback;
777}
