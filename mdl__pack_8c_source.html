159RA8_INTERNAL static size_t internal_pack_dir_output(mdl_storage_t* storage,

182size_t mdl_pack_one_meta(mdl_storage_t* storage,
183 mdl_format_t format,
184 const char* series_dir,
185 const char* chap_id,
186 const mdl_export_meta_t* meta,
187 mdl_export_workspace_t* ws,
188 ra8_io_stream_t* output,
189 ra8_io_stream_t* diagnostic)
190{
191 const char* ext = mdl_format_ext(format);
192 char dir[k_pack_dir_bytes];
193 if (!mdl_path_join(series_dir, chap_id, dir, sizeof(dir))) {
194 (void)internal_pack_text3(diagnostic, " export ", chap_id, ".");
195 (void)internal_pack_text3(diagnostic, "", ext, " path rejected, skipped\n");
196 return 1U;
197 }
198
199 const mdl_export_meta_t m = internal_pack_metadata(storage, meta, dir);
200
201 if (mdl_format_is_dir_output(format)) {
202 return internal_pack_dir_output(storage, format, dir, chap_id, ext, &m, ws, output, diagnostic);
203 }
204 char leaf[k_pack_leaf_bytes];
205 const int ln = snprintf(leaf, sizeof(leaf), "%s.%s", chap_id, ext);
206 char out[k_pack_dir_bytes];
207 if (!internal_pack_snprintf_fit(ln, sizeof(leaf)) ||
208 !mdl_path_join(series_dir, leaf, out, sizeof(out))) {
209 ra8_err_t output_error = internal_pack_text3(diagnostic, " export ", chap_id, ".");
210 output_error = priv_mdl_stream_text(output_error, diagnostic, ext);
211 (void)priv_mdl_stream_text(output_error, diagnostic, " path rejected, skipped\n");
212 return 1U;
213 }
214 const ra8_err_t rc = mdl_export_chapter_meta_ws(storage, format, dir, out, &m, ws);
215 if (rc != k_ra8_ok) {
216 internal_pack_report_failure(diagnostic, chap_id, ext, rc);
217 return 1U;
218 }
219 ra8_err_t output_error = internal_pack_text3(output, " packaged ", chap_id, ".");
220 output_error = priv_mdl_stream_text(output_error, output, ext);
221 output_error = priv_mdl_stream_text(output_error, output, "\n");
222 return (output_error == k_ra8_ok) ? 0U : 1U;
223}

386size_t mdl_pack_combined_meta(mdl_storage_t* storage,
387 mdl_format_t format,
388 bool allow_incomplete,
389 const char* series_dir,
390 const char* combined_rel,
391 const mdl_fetch_stats_t* stats,
392 const mdl_export_meta_t* meta,
393 mdl_export_workspace_t* ws,
394 ra8_io_stream_t* output,
395 ra8_io_stream_t* diagnostic)
396{
397 if (stats->chapters_completed == 0U) {
398 return 0U; /* nothing was fetched to package */
399 }
400 const bool incomplete = priv_mdl_fetch_run_incomplete(stats);
401 if (incomplete) {
402 if (!allow_incomplete) {
403 ra8_err_t output_error = priv_mdl_stream_text(k_ra8_ok,
404 diagnostic,
405 " combine: NOT packaged -- run is "
406 "incomplete (");
407 output_error = priv_mdl_stream_u64(output_error, diagnostic, stats->chapters_failed);
408 output_error = priv_mdl_stream_text(output_error, diagnostic, " chapter(s) / ");
409 output_error = priv_mdl_stream_u64(output_error, diagnostic, stats->pages_failed);
410 output_error = priv_mdl_stream_text(output_error,
411 diagnostic,
412 " page(s) failed); pass --allow-incomplete to force "
413 "a marked archive\n");
414 return (output_error == k_ra8_ok) ? 0U : 1U;
415 }
416 }
417 return internal_pack_combined_dir(storage,
418 format,
419 series_dir,
420 combined_rel,
421 incomplete,
422 meta,
423 ws,
424 output,
425 diagnostic);
426}
