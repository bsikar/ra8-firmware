33size_t mdl_pack_one(mdl_format_t format, const char* series_dir, const char* chap_id)
34{
35 const char* ext = mdl_format_ext(format);
36 char dir[k_pack_dir_bytes];
37 if (!mdl_path_join(series_dir, chap_id, dir, sizeof(dir))) {
38 (void)fprintf(stderr, " export %s.%s path rejected, skipped\n", chap_id, ext);
39 return 1U;
40 }
41 if (mdl_format_is_dir_output(format)) {
42 /* JOF writes per-page `.jof` siblings into the chapter dir; report that dir,
43 * never a single-container name that was not created. */
44 const ra8_err_t drc = mdl_export_chapter(format, dir, dir);
45 if (drc != k_ra8_ok) {
46 (void)fprintf(stderr, " export %s .%s FAILED (err 0x%X)\n", chap_id, ext, (unsigned)drc);
47 return 1U;
48 }
49 (void)printf(" converted %s -> %s/*.%s\n", chap_id, dir, ext);
50 return 0U;
51 }
52 char leaf[k_pack_leaf_bytes];
53 const int ln = snprintf(leaf, sizeof(leaf), "%s.%s", chap_id, ext);
54 char out[k_pack_dir_bytes];
55 if (!snprintf_fit(ln, sizeof(leaf)) || !mdl_path_join(series_dir, leaf, out, sizeof(out))) {
56 (void)fprintf(stderr, " export %s.%s path rejected, skipped\n", chap_id, ext);
57 return 1U;
58 }
59 const ra8_err_t rc = mdl_export_chapter(format, dir, out);
60 if (rc != k_ra8_ok) {
61 (void)fprintf(stderr, " export %s.%s FAILED (err 0x%X)\n", chap_id, ext, (unsigned)rc);
62 return 1U;
63 }
64 (void)printf(" packaged %s.%s\n", chap_id, ext);
65 return 0U;
66}

69RA8_INTERNAL static size_t pack_combined_dir(mdl_format_t format,
70 const char* series_dir,
71 const char* combined_rel,
72 bool incomplete)
73{
74 const char* ext = mdl_format_ext(format);
75 const char* mark = incomplete ? " (INCOMPLETE)" : "";
76 char dir[k_pack_dir_bytes];
77 if (!mdl_path_join(series_dir, combined_rel, dir, sizeof(dir))) {
78 (void)fprintf(stderr, " combine export path rejected under %s\n", series_dir);
79 return 1U;
80 }
81 if (mdl_format_is_dir_output(format)) {
82 /* JOF: the combined pages become `.jof` siblings inside the combined dir. */
83 const ra8_err_t drc = mdl_export_chapter(format, dir, dir);
84 if (drc != k_ra8_ok) {
85 (void)fprintf(stderr, " combine export FAILED (err 0x%X)\n", (unsigned)drc);
86 return 1U;
87 }
88 (void)printf(" combined%s -> %s/*.%s\n", mark, dir, ext);
89 return 0U;
90 }
91 char leaf[k_pack_leaf_bytes];
92 const int ln = incomplete ? snprintf(leaf, sizeof(leaf), "%s.INCOMPLETE.%s", combined_rel, ext)
93 : snprintf(leaf, sizeof(leaf), "%s.%s", combined_rel, ext);
94 char out[k_pack_dir_bytes];
95 if (!snprintf_fit(ln, sizeof(leaf)) || !mdl_path_join(series_dir, leaf, out, sizeof(out))) {
96 (void)fprintf(stderr, " combine export path rejected under %s\n", series_dir);
97 return 1U;
98 }
99 const ra8_err_t rc = mdl_export_chapter(format, dir, out);
100 if (rc != k_ra8_ok) {
101 (void)fprintf(stderr, " combine export FAILED (err 0x%X)\n", (unsigned)rc);
102 return 1U;
103 }
104 (void)printf(" combined%s -> %s\n", mark, out);
105 return 0U;
106}

108size_t mdl_pack_combined(mdl_format_t format,
109 bool allow_incomplete,
110 const char* series_dir,
111 const char* combined_rel,
112 const mdl_fetch_stats_t* stats)
113{
114 if (stats->chapters_completed == 0U) {
115 return 0U; /* nothing was fetched to package */
116 }
117 const bool incomplete = mdl_fetch_run_incomplete(stats);
118 if (incomplete) {
119 if (!allow_incomplete) {
120 (void)fprintf(stderr,
121 " combine: NOT packaged -- run is incomplete "
122 "(%zu chapter(s) / %zu page(s) failed); pass "
123 "--allow-incomplete to force a marked archive\n",
124 stats->chapters_failed,
125 stats->pages_failed);
126 return 0U;
127 }
128 }
129 return pack_combined_dir(format, series_dir, combined_rel, incomplete);
130}
