539static void demo_build_jiggle(demo_phase_t phase, UCHAR* report)
540{
541 int8_t dx = k_demo_jiggle_zero;
542 int8_t dy = k_demo_jiggle_zero;
543 switch (phase) {
544 case k_demo_phase_right:
545 dx = k_demo_jiggle_step;
546 break;
547 case k_demo_phase_down:
548 dy = k_demo_jiggle_step;
549 break;
550 case k_demo_phase_left:
551 dx = k_demo_jiggle_neg;
552 break;
553 case k_demo_phase_up:
554 dy = k_demo_jiggle_neg;
555 break;
556 case k_demo_phase_count:
557 default:
558 break;
559 }
560 report[k_demo_hid_idx_buttons] = 0U;
561 report[k_demo_hid_idx_dx] = (UCHAR)dx;
562 report[k_demo_hid_idx_dy] = (UCHAR)dy;
563}

640static void demo_jiggle_send(demo_phase_t* phase)
641{
642 UX_SLAVE_CLASS_HID_EVENT hid_event;
643 (void)memset(&hid_event, 0, sizeof(hid_event));
644 hid_event.ux_device_class_hid_event_length = (ULONG)k_demo_hid_report_bytes;
645 hid_event.ux_device_class_hid_event_report_id = 0UL;
646 hid_event.ux_device_class_hid_event_report_type = (ULONG)UX_DEVICE_CLASS_HID_REPORT_TYPE_INPUT;
647 demo_build_jiggle(*phase, hid_event.ux_device_class_hid_event_buffer);
648
649 if (_ux_device_class_hid_event_set(s_hid_class, &hid_event) == UX_SUCCESS) {
650 (void)ra8_board_led_toggle(k_ra8_board_led1);
651 *phase = (demo_phase_t)(((uint8_t)(*phase + 1U)) % k_demo_phase_count);
652 g_usb_hid_match += 1U;
653 } else {
654 g_usb_hid_mismatch += 1U;
655 }
656}

658static VOID demo_worker(ULONG arg)
659{
660 (void)arg;
661
662 if (demo_usbx_stack_up() != UX_SUCCESS) {
663 return;
664 }
665 if (demo_hid_class_register() != UX_SUCCESS) {
666 return;
667 }
668 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs) != k_ra8_ok) {
669 return;
670 }
671 if (ra8_usb_device_attach(k_ra8_usb_speed_fs, true) != k_ra8_ok) {
672 return;
673 }
674
675 /* Jiggle loop. INTSTS0 is drained by the bridge's own polled-dispatch
676 * worker (spawned inside ux_dcd_ra8_usb_initialize BEFORE the
677 * ra8_usb_device_attach(true) above raised DPRPU), so this thread
678 * only needs to push HID reports once per period. */
679 demo_phase_t phase = k_demo_phase_right;
680 while (1) {
681 if (s_hid_class == UX_NULL) {
682 tx_thread_sleep(k_demo_jiggle_period_ticks);
683 continue;
684 }
685 demo_jiggle_send(&phase);
686 tx_thread_sleep(k_demo_jiggle_period_ticks);
687 }
688}

754[[nodiscard]] static ra8_err_t demo_pins_init(void)
755{
756 ra8_err_t err = ra8_pfs_route_peripheral(k_demo_pin_vbus, k_ra8_psel_usb_fs, "usb_hid.vbus");
757 if (err != k_ra8_ok) {
758 return err;
759 }
760 /* VBUSEN as GPIO output LOW for USB device mode. Peripheral routing
761 * forces VBUSEN HIGH (host mode) which blocks device enumeration. */
762 err = ra8_gpio_output_init(k_demo_pin_vbusen, k_ra8_level_low);
763 if (err != k_ra8_ok) {
764 return err;
765 }
766 err = ra8_pfs_route_peripheral(k_demo_pin_dp, k_ra8_psel_usb_fs, "usb_hid.dp");
767 if (err != k_ra8_ok) {
768 return err;
769 }
770 return ra8_pfs_route_peripheral(k_demo_pin_dm, k_ra8_psel_usb_fs, "usb_hid.dm");
771}

805 demo_panic_halt();
