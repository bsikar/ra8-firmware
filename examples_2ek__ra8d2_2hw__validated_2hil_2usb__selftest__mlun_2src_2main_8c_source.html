388static UINT mlun_msc_read(VOID* storage,
389 ULONG lun,
390 UCHAR* data_pointer,
391 ULONG number_blocks,
392 ULONG lba,
393 ULONG* media_status)
394{
395 (void)storage;
396 s_dbg_read_calls++;
397 if ((lba + number_blocks) > (ULONG)k_mlun_sectors) {
398 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
399 k_scsi_asc_lba_out_of_range,
400 k_scsi_ascq_none);
401 return UX_ERROR;
402 }
403 for (ULONG i = 0UL; i < number_blocks; i++) {
404 mlun_pattern_fill((uint32_t)lun,
405 (uint32_t)(lba + i),
406 &data_pointer[i * (ULONG)k_mlun_block_size]);
407 }
408 *media_status = 0UL;
409 (void)ra8_board_led_toggle(k_ra8_board_led1);
410 return UX_SUCCESS;
411}

613static VOID mlun_device_worker(ULONG arg)
614{
615 (void)arg;
616
617 UINT ux = mlun_usbx_stack_up();
618 if (ux != UX_SUCCESS) {
619 s_dbg_dev_err = (uint32_t)ux;
620 return;
621 }
622 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_stack;
623 ux = mlun_class_register();
624 if (ux != UX_SUCCESS) {
625 s_dbg_dev_err = (uint32_t)ux;
626 return;
627 }
628 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_class;
629 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
630 if (e != k_ra8_ok) {
631 s_dbg_dev_err = (uint32_t)e;
632 return;
633 }
634 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_dcd;
635 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
636 if (e != k_ra8_ok) {
637 s_dbg_dev_err = (uint32_t)e;
638 return;
639 }
640 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_attach;
641
642 while (1) {
643 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_parked;
644 tx_thread_sleep(k_mlun_idle_ticks);
645 }
646}
