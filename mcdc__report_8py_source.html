3"""The five artefacts the MC/DC audit emits, and the counts they share.
4
5Split out of ``regen_mcdc_gaps.py`` (#359): parsing llvm-cov's report and
6classifying each decision is one job; rendering that into a CSV, two Markdown
7documents and two JSON roll-ups is another, and it is by far the larger half.
8
9Every artefact quotes the same headline counts, so :func:`headline` computes
10them once and each writer takes the result. That is the reason these live
11together rather than one file per artefact -- the numbers in the CSV, the
12report and the gate JSON must agree, and the only way to guarantee that is for
13them to be the same numbers.
14"""

42def write_gap_csv(classified: list) -> None:
43 """Write the gap-only CSV.
44
45 The `line` column is a text-derived snippet, not a line number: line
46 numbers drift on every reformat and produce stale anchors, which project
47 citation policy bans.
48 """
49 with CSV_OUT.open("w", encoding="ascii", newline="") as fh:
50 w = csv.writer(fh, lineterminator="\n")
51 w.writerow(
52 [
53 "source_file",
54 "decision_text_snippet",
55 "condition_count",
56 "function_name",
57 "decision_excerpt",
58 "covered",
59 "deactivated",
60 "deactivation_rationale",
61 ]
62 )
63 for src, _ln, n, func, excerpt, covered, deact, rationale in classified:
64 w.writerow(
65 [
66 src,
67 decision_snippet(excerpt),
68 n,
69 func,
70 excerpt,
71 covered,
72 "true" if deact else "false",
73 rationale,
74 ]
75 )

78def module_rows(all_decisions: list) -> list:
79 """Per-module (total, covered, partial, uncovered), worst first."""
80 per_module: dict[str, list[float]] = defaultdict(list)
81 for src, _ln, _n, _e, pct in all_decisions:
82 per_module[module_of(src)].append(pct)
83
84 rows = []
85 for mod, pcts in per_module.items():
86 total = len(pcts)
87 covered = sum(1 for p in pcts if p >= MCDC_FULL_PCT)
88 partial = sum(1 for p in pcts if MCDC_ZERO_PCT < p < MCDC_FULL_PCT)
89 uncov = sum(1 for p in pcts if p == MCDC_ZERO_PCT)
90 rows.append((mod, total, covered, partial, uncov))
91
92 # Sort by uncovered+partial desc, then total desc, then name.
93 rows.sort(key=lambda r: (-(r[3] + r[4]), -r[1], r[0]))
94 return rows

97def headline(all_decisions: list, classified: list) -> dict:
98 """The counts every generated artefact quotes, computed once."""
99 total_dec = len(all_decisions)
100 yes_dec = sum(1 for d in all_decisions if d[4] >= MCDC_FULL_PCT)
101 partial_dec = sum(1 for d in all_decisions if MCDC_ZERO_PCT < d[4] < MCDC_FULL_PCT)
102 no_dec = sum(1 for d in all_decisions if d[4] == MCDC_ZERO_PCT)
103 files_seen = len({d[0] for d in all_decisions})
104 rate = (100.0 * yes_dec / total_dec) if total_dec else 0.0
105
106 deactivated_rows = [r for r in classified if r[6]]
107 reachable_rows = [r for r in classified if not r[6]]
108 # Reachable-only MC/DC: treat deactivated decisions as if they were
109 # at 100% (they are exempted under DO-178C 6.4.4.3 with documented
110 # rationale in docs/MCDC_DEACTIVATIONS.md). The denominator is
111 # unchanged; the numerator counts every covered decision plus every
112 # documented-deactivated decision.
113 deact_count = len(deactivated_rows)
114 reachable_total = total_dec - deact_count
115 reachable_covered = yes_dec
116 reachable_rate = (100.0 * reachable_covered / reachable_total) if reachable_total else 100.0
117 return {
118 "total_dec": total_dec,
119 "yes_dec": yes_dec,
120 "partial_dec": partial_dec,
121 "no_dec": no_dec,
122 "files_seen": files_seen,
123 "rate": rate,
124 "deactivated_rows": deactivated_rows,
125 "reachable_rows": reachable_rows,
126 "deact_count": deact_count,
127 "reachable_total": reachable_total,
128 "reachable_covered": reachable_covered,
129 "reachable_rate": reachable_rate,
130 }

211def _md_gap_tables(h: dict) -> list[str]:
212 """The reachable-gap and deactivated-gap tables."""
213 deactivated_rows = h["deactivated_rows"]
214 reachable_rows = h["reachable_rows"]
215 md_lines: list[str] = []
216 md_lines.append("## Reachable gaps (require new MC/DC test vectors)")
217 md_lines.append("")
218 md_lines.append("| File | Conds | Function | Excerpt | Status |")
219 md_lines.append("|------|------:|----------|---------|--------|")
220 for src, _ln, n, func, excerpt, covered, _deact, _rat in reachable_rows[:TABLE_ROW_CAP]:
221 ex = _truncate_md_cell(
222 excerpt.replace("|", "\\|"), EXCERPT_MAX_REACHABLE, EXCERPT_TRUNC_REACHABLE
223 )
224 md_lines.append(f"| {src} | {n} | {func} | `{ex}` | {covered} |")
225 if len(reachable_rows) > TABLE_ROW_CAP:
226 overflow = len(reachable_rows) - TABLE_ROW_CAP
227 md_lines.append(f"| ... | | | | *({overflow} more rows in CSV)* | |")
228 md_lines.append("")
229 md_lines.append("## Deactivated gaps (DO-178C 6.4.4.3 exempted)")
230 md_lines.append("")
231 md_lines.append(
232 "These conditions are unreachable on any public-API path and"
233 " are therefore exempted from the MC/DC gate. Each row carries"
234 " the rationale used by the auto-classifier; humans may extend"
235 " the per-condition narrative in `docs/MCDC_DEACTIVATIONS.md`."
236 )
237 md_lines.append("")
238 md_lines.append("| File | Conds | Function | Excerpt | Rationale |")
239 md_lines.append("|------|------:|----------|---------|-----------|")
240 for src, _ln, n, func, excerpt, _covered, _deact, rationale in deactivated_rows:
241 ex = _truncate_md_cell(
242 excerpt.replace("|", "\\|"), EXCERPT_MAX_DEACTIVATED, EXCERPT_TRUNC_DEACTIVATED
243 )
244 rt = _truncate_md_cell(
245 rationale.replace("|", "\\|"), EXCERPT_MAX_DEACTIVATED, EXCERPT_TRUNC_DEACTIVATED
246 )
247 md_lines.append(f"| {src} | {n} | {func} | `{ex}` | {rt} |")
248 md_lines.append("")
249 return md_lines

252def _md_module_tables(rows: list) -> list[str]:
253 """The per-module roll-up tables and the report footer."""
254 md_lines: list[str] = []
255 md_lines.append("## Per-module gap counts (full table)")
256 md_lines.append("")
257 md_lines.append("Sorted by (uncovered + partial) descending, then total descending.")
258 md_lines.append("")
259 md_lines.append("| Module | Total | Covered | Partial | Uncovered |")
260 md_lines.append("|--------|------:|--------:|--------:|----------:|")
261 for mod, total, covered, partial, uncov in rows:
262 md_lines.append(f"| {mod} | {total} | {covered} | {partial} | {uncov} |")
263 md_lines.append("")
264 md_lines.append("## Top 30 modules with at least one uncovered decision")
265 md_lines.append("")
266 md_lines.append("| Module | Uncovered | Partial | Covered | Total |")
267 md_lines.append("|--------|----------:|--------:|--------:|------:|")
268 uncov_only = [r for r in rows if r[4] > 0]
269 uncov_only.sort(key=lambda r: (-r[4], -r[3], r[0]))
270 for mod, total, covered, partial, uncov in uncov_only[:30]:
271 md_lines.append(f"| {mod} | {uncov} | {partial} | {covered} | {total} |")
272 md_lines.append("")
273 md_lines.append("---")
274 md_lines.append("")
275 md_lines.append(
276 "*Regenerated from the live `make mcdc` report. See"
277 " `docs/MCDC_GAPS.csv` for the full per-decision table"
278 " including decision-text snippets and excerpts.*"
279 )
280 md_lines.append("")
281
282 return md_lines

331def _deact_entries(deactivated_rows: list[tuple]) -> list[str]:
332 """One catalog section per deactivated condition, keyed by a stable anchor.
333
334 The anchor is ``file::function::snippet`` rather than a line number
335 precisely because line numbers drift on every reformat, which would
336 silently detach a hand-written justification from its condition.
337 """
338 if not deactivated_rows:
339 return ["(no deactivated conditions detected)", ""]
340 out: list[str] = []
341 for src, _ln, n, func, excerpt, covered, _deact, rationale in deactivated_rows:
342 anchor = f"{src}::{func}::{decision_snippet(excerpt)}"
343 out.extend(
344 [
345 f"### {anchor}",
346 "",
347 f"- **Function**: `{func}`",
348 f"- **Conditions in decision**: {n}",
349 f"- **Current llvm-cov status**: {covered}",
350 f"- **Source line**: `{excerpt.strip()}`",
351 f"- **Rationale**: {rationale}",
352 "- **DO-178C 6.4.4.3 basis**: defensive guard whose"
353 " upstream contract is enforced on every public-API"
354 " entry.",
355 "",
356 ]
357 )
358 return out

414def write_per_file_json(all_decisions: list, classified: list) -> None:
415 """Per-file decision roll-up for check_mcdc_floor.py."""
416 per_file: dict[str, dict[str, int]] = defaultdict(
417 lambda: {"total": 0, "covered": 0, "deactivated": 0}
418 )
419 for src, _ln, _n, _e, pct in all_decisions:
420 per_file[src]["total"] += 1
421 if pct >= MCDC_FULL_PCT:
422 per_file[src]["covered"] += 1
423 for row in classified:
424 if row[6]: # deactivated
425 per_file[row[0]]["deactivated"] += 1
426
427 per_file_json = REPO_ROOT / "build" / "mcdc-report" / "mcdc_per_file.json"
428 per_file_entries = [
429 {
430 "file": src,
431 "total_decisions": rec["total"],
432 "covered_decisions": rec["covered"],
433 "deactivated_decisions": rec["deactivated"],
434 }
435 for src, rec in sorted(per_file.items())
436 ]
437 with contextlib.suppress(OSError):
438 per_file_json.write_text(
439 json.dumps({"files": per_file_entries}, indent=2) + "\n",
440 encoding="ascii",
441 )
