316[[nodiscard]] static ra8_err_t microsd_sd_spi_open(void)
317{
318 ra8_err_t err =
319 ra8_pfs_route_peripheral(k_microsd_pin_sd_sck, k_ra8_psel_sci_async, "microsd.sck");
320 if (err != k_ra8_ok) {
321 return err;
322 }
323 err = ra8_pfs_route_peripheral(k_microsd_pin_sd_cipo, k_ra8_psel_sci_async, "microsd.cipo");
324 if (err != k_ra8_ok) {
325 return err;
326 }
327 err = ra8_pfs_route_peripheral(k_microsd_pin_sd_copi, k_ra8_psel_sci_async, "microsd.copi");
328 if (err != k_ra8_ok) {
329 return err;
330 }
331 err = ra8_gpio_output_init(k_microsd_pin_sd_cs, k_ra8_level_high);
332 if (err != k_ra8_ok) {
333 return err;
334 }
335 const ra8_sci_spi_cfg_t spi_cfg = {
336 .baud_hz = (uint32_t)k_ra8_sdmmc_spi_clock_init_hz,
337 .pclk_hz = s_pclka_hz,
338 .mode = k_ra8_spi_mode_0,
339 .lsb_first = false,
340 };
341 return ra8_sci_spi_init((uint8_t)k_microsd_sd_spi_channel, &spi_cfg);
342}

366[[nodiscard]] static ra8_err_t microsd_sd_snapshot(void)
367{
368 ra8_err_t err = microsd_sd_spi_open();
369 if (err != k_ra8_ok) {
370 s_dbg_sd_err = (uint32_t)err;
371 return err;
372 }
373 const ra8_sdmmc_spi_transport_t transport = {
374 .set_clock = microsd_spi_set_clock,
375 .cs = microsd_spi_cs,
376 .xfer = microsd_spi_xfer,
377 .ctx = &s_pclka_hz,
378 };
379 err = ra8_sdmmc_spi_init(&transport);
380 if (err != k_ra8_ok) {
381 s_dbg_sd_err = (uint32_t)err;
382 return err;
383 }
384 uint32_t blocks = 0U;
385 (void)ra8_sdmmc_spi_get_capacity(&blocks);
386 s_dbg_sd_blocks = blocks;
387 for (uint32_t lba = 0U; lba < (uint32_t)k_microsd_sectors; lba++) {
388 err = ra8_sdmmc_spi_read_block(
389 lba,
390 &s_usb_selftest_microsd_sd_snapshot[lba * (uint32_t)k_microsd_block_size]);
391 if (err != k_ra8_ok) {
392 s_dbg_sd_err = (uint32_t)err;
393 return err;
394 }
395 }
396 const bool sig =
397 (s_usb_selftest_microsd_sd_snapshot[k_microsd_bootsig_off] == (UCHAR)k_microsd_bootsig_lo) &&
398 (s_usb_selftest_microsd_sd_snapshot[k_microsd_bootsig_off + 1U] == (UCHAR)k_microsd_bootsig_hi);
399 s_usb_selftest_microsd_dbg_bootsig = sig ? 1U : 0U;
400 s_dbg_sd_err = 0U;
401 return k_ra8_ok;
402}

531static void microsd_setup_or_halt(void)
532{
533 uint32_t cpuclk0_hz = 0U;
534 uint32_t pclka_hz = 0U;
535 if (ra8_cgc_init() != k_ra8_ok) {
536 microsd_panic_halt();
537 }
538 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
539 microsd_panic_halt();
540 }
541 if (ra8_cgc_usbhs_pll_enable() != k_ra8_ok) {
542 microsd_panic_halt();
543 }
544 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
545 microsd_panic_halt();
546 }
547 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
548 microsd_panic_halt();
549 }
550 s_pclka_hz = pclka_hz;
551 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
552 microsd_panic_halt();
553 }
554 if (ra8_board_uart_console_init((uint32_t)k_microsd_baud) != k_ra8_ok) {
555 microsd_panic_halt();
556 }
557 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
558 microsd_panic_halt();
559 }
560 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
561 microsd_panic_halt();
562 }
563 /* Snapshot SD LBA 0..63 into RAM (read-only) before the kernel starts.
564 * On failure s_dbg_sd_err records it and the host verifier reports the
565 * missing boot signature -- the board does not hang. */
566 (void)microsd_sd_snapshot();
567 microsd_route_usb_or_halt();
568}
