257static inline uint32_t ra8_npu_blob_read_word(const uint8_t* buf, uint32_t byte_off)
258{
259 uint32_t value = 0U;
260 for (uint32_t i = 0U; i < k_ra8_npu_blob_word_bytes; i++) {
261 value |= (uint32_t)buf[byte_off + i] << (i * 8U);
262 }
263 return value;
264}
