1#!/usr/bin/env python3
2# SPDX-License-Identifier: MIT

4"""Gate: no function shall exist only to satisfy the linker.
5
6A *silent stub* is a function whose body does no work -- it discards its
7arguments and hands back a canned answer -- yet whose presence lets the build
8link and the program claim a capability it does not have. The failure mode is
9worse than a missing feature: the tool links clean, advertises support, and
10fails at runtime (or, worse, silently succeeds having done nothing).
11
12``tools/rabook_imagepack/webp_stub.c`` and ``tools/media_dl/webp_stub.c`` were the
13motivating case. Each defined the real symbol
14``ra8_jof_priv_webp_transcode()``, discarded both arguments and returned
15``k_ra8_err_not_supported`` -- purely so the JOF producer would link
16without compiling the vendored libwebp decoder. A complete WebP decoder was
17already vendored, wrapped, tested and fuzzed in this very repository; the tools
18just were not compiling it. Both tools shipped a WebP feature that could never
19work.
20
21Calibration
22-----------
23Most no-op bodies in this tree are *legitimate* -- an early revision of this
24gate flagged 27 candidates of which 24 were correct by design:
25
26 * platform alternatives (``tools/ra8_emulator/src/display/board_view_stub.c`` is the
27 headless stand-in for the Cocoa window layer on Linux CI; it reports
28 failure honestly so callers take their headless branch);
29 * the fail-closed ``#else`` half of the placeholder-crypto guard, which
30 ``check_stub_crypto_guarded.py`` *requires* to return a hard error;
31 * callbacks matching a vtable / registry signature that genuinely have
32 nothing to do (USBX activate / deactivate hooks, an empty ISR completion
33 callback, a ra8_emulator MMIO write handler for a deliberately inert
34 peripheral);
35 * MMIO read handlers that return module state rather than a constant.
36
37A gate that fires on those is noise, and a noisy gate gets disabled -- which is
38worse than no gate. So this one does not ask "is the body empty?". It asks
39the two much narrower questions that separate the webp stubs from all 24:
40
41Rule SHADOW
42 A no-op body that provides a *second* definition of a symbol which is also
43 defined for real elsewhere in first-party code. This is always a defect:
44 whichever definition the linker picks, one of them is a lie, and the build
45 silently disables working code that exists in the tree. This is exactly
46 the webp case, and it has no legitimate form -- a platform alternative
47 supplies the *only* definition in its build, never a competing one.
48
49Rule CANNED
50 A function that returns an explicit "unsupported / unimplemented" error
51 constant, discards every parameter, and has no other statement. Such a
52 function claims an operation was attempted and refused, when in fact no
53 implementation exists at all. Callbacks returning ``void``/``bool``/a
54 pointer, and handlers returning module state, are outside this rule by
55 construction.
56
57Hardware waiver
58---------------
59A capability whose *hardware does not physically exist yet* may legitimately
60have no implementation. Such a function is waived by carrying an explicit
61marker naming the missing part, in the form CLAUDE.md mandates::
62
63 TODO(ESP32-C6 radio module not yet on the bench): ...
64
65The marker must name something -- a bare ``TODO`` or an empty ``TODO()`` is
66rejected, so the waiver cannot become a catch-all. It is keyed on the marker
67and not on a filename pattern, so it waives exactly the function that carries
68it and nothing else in the file. Note that a waiver is *not* available to
69Rule SHADOW: if a real implementation exists in the tree, the hardware plainly
70exists too.
71
72Run::
73
74 check_no_silent_stubs.py # scan the whole tree
75 check_no_silent_stubs.py FILE ... # scan listed files
76 check_no_silent_stubs.py --selftest # prove the detector both fires and
77 # stays silent on the right inputs
78
79Exit 0 if no silent stub is found, 1 otherwise.
80"""

91# First-party roots. libs/third_party (SOUP) and libs/ra8_fonts (generated) are out
92# of scope, matching every other repo gate.
93ROOTS = ("libs", "src", "tools", "examples", "port")
94EXCLUDED = ("libs/third_party", "libs/ra8_fonts")

96# Error constants that mean "this operation has no implementation". A function
97# whose entire body hands one of these back, having discarded its arguments,
98# has not refused an operation -- it never had one.
99CANNED_ERRORS = frozenset(
100 {
101 "k_ra8_err_not_supported",
102 "k_ra8_err_not_implemented",
103 "k_ra8_err_unsupported",
104 "k_ra8_err_unimplemented",
105 }
106)

108# A waiver must name the missing hardware: TODO(<something>). Bare TODO or
109# TODO() is deliberately not accepted.
110WAIVER_RE = re.compile(r"TODO\‍(\s*([^)]*?)\s*\‍)")
111
112# Keywords that can precede a parenthesised block but are not function
113# definitions.
114NOT_A_FUNCTION = frozenset(
115 {"if", "for", "while", "switch", "return", "sizeof", "do", "else", "catch"}
116)
117
118# A function definition: an optional return type, a name, a parameter list with
119# no nested braces or semicolons, then an opening brace. Newlines are allowed
120# between the parameter list and the brace -- the house style puts the brace of
121# a definition on its own line, and missing that made an early revision of this
122# detector report zero findings on a file that was a known stub.
123FUNCTION_RE = re.compile(
124 r"(?:^|\n)[ \t]*(?:[A-Za-z_][\w \t\*\n]*?)\b(\w+)[ \t\n]*\‍(([^;{}]*)\‍)[ \t\n]*\{",
125 re.DOTALL,
126)
127
128DISCARD_RE = re.compile(r"\‍(\s*void\s*\‍)\s*(\w+)\Z")
129RETURN_CONST_RE = re.compile(r"return\s+([A-Za-z_]\w*|-?\d+|nullptr|NULL|true|false)\Z")
130PARAM_NAME_RE = re.compile(r"(\w+)\s*(?:\‍[\s*\‍])?\s*\Z")

133def blank_noncode(text: str) -> str:
134 """Blank out comments and string/char literals, preserving offsets.
135
136 Offsets are preserved (and newlines kept) so that reported line numbers and
137 brace matching still refer to the original file.
138 """
139 out: list[str] = []
140 i, n = 0, len(text)
141 while i < n:
142 c = text[i]
143 if c == "/" and i + 1 < n and text[i + 1] == "*":
144 end = text.find("*/", i + 2)
145 end = n if end < 0 else end + 2
146 out.append(re.sub(r"[^\n]", " ", text[i:end]))
147 i = end
148 elif c == "/" and i + 1 < n and text[i + 1] == "/":
149 end = text.find("\n", i)
150 end = n if end < 0 else end
151 out.append(" " * (end - i))
152 i = end
153 elif c in "\"'":
154 j = i + 1
155 while j < n and text[j] != c:
156 j += 2 if text[j] == "\\" else 1
157 j = min(j + 1, n)
158 out.append(re.sub(r"[^\n]", " ", text[i:j]))
159 i = j
160 else:
161 out.append(c)
162 i += 1
163 return "".join(out)

166def body_span(text: str, brace_idx: int) -> tuple[int, int] | None:
167 """Return the (start, end) offsets of a function body by brace matching."""
168 depth = 0
169 for i in range(brace_idx, len(text)):
170 if text[i] == "{":
171 depth += 1
172 elif text[i] == "}":
173 depth -= 1
174 if depth == 0:
175 return (brace_idx + 1, i)
176 return None

179def param_names(params: str) -> set[str]:
180 """Extract the declared parameter NAMES from a C parameter list.
181
182 Feeds the discard test in ``classify_body``, which has to know whether a
183 ``(void)x;`` names a real parameter or some unrelated local -- so what is
184 wanted here is the identifiers, with the types thrown away.
185
186 ``void`` as the whole list yields the empty set rather than a name, which
187 is what makes a zero-parameter function fall out as "discards everything
188 it was given" and stay eligible for the CANNED rule.
189 """
190 names: set[str] = set()
191 for raw_part in params.split(","):
192 part = raw_part.strip()
193 if not part or part == "void":
194 continue
195 m = PARAM_NAME_RE.search(part)
196 if m:
197 names.add(m.group(1))
198 return names

201def classify_body(body: str, params: str) -> str | None:
202 """Return the returned constant if the body is a pure discard-and-return.
203
204 Returns None when the body does real work. A body qualifies only when every
205 statement is either a ``(void)param;`` discard or a single ``return
206 <constant>;``, AND at least one discarded name is an actual parameter --
207 that last condition is what separates "ignores its inputs" from a handler
208 that legitimately returns module state.
209 """
210 statements = [s.strip() for s in body.split(";") if s.strip()]
211 if not statements:
212 return None # a genuinely empty body is a callback shape, not a stub
213 discards: list[str] = []
214 returns: list[str] = []
215 for stmt in statements:
216 m = DISCARD_RE.match(stmt)
217 if m:
218 discards.append(m.group(1))
219 continue
220 m = RETURN_CONST_RE.match(stmt)
221 if m:
222 returns.append(m.group(1))
223 continue
224 return None # any other statement means real work
225 if len(returns) != 1:
226 return None
227 declared = param_names(params)
228 # Takes parameters: at least one must actually be discarded. That is what
229 # separates "ignores its inputs" from a handler that legitimately returns
230 # module state.
231 if declared and not (set(discards) & declared):
232 return None
233 # Takes NO parameters, so "discards every parameter" is vacuously true and
234 # the discard test above cannot speak. Only a canned-error return qualifies
235 # here: `ra8_widget_calibrate(void) { return k_ra8_err_not_supported; }` is
236 # every bit the stub its one-argument form is, but a bare `return s_state;`
237 # getter is module state and must stay silent. Requiring an empty discard
238 # list keeps a body that pokes at file-scope names out of the "pure canned
239 # return" shape.
240 if not declared and (discards or returns[0] not in CANNED_ERRORS):
241 return None
242 return returns[0]

245def waiver_in(text: str) -> str | None:
246 """Return the named missing dependency from a TODO(...) marker, if any."""
247 for m in WAIVER_RE.finditer(text):
248 named = m.group(1).strip()
249 if named:
250 return named
251 return None

254def preceding_doc(text: str, start: int) -> str:
255 """Return the ~40 lines before a definition (its comment block)."""
256 head = text[:start]
257 return "\n".join(head.splitlines()[-40:])

260def in_failclosed_crypto_guard(raw: str, offset: int) -> bool:
261 """True if the offset sits in the #else half of the placeholder-crypto guard.
262
263 ``check_stub_crypto_guarded.py`` REQUIRES those bodies to return a hard
264 error so a production image cannot ship fake crypto. Firing on them would
265 pit one gate against another.
266 """
267 guard = "defined(RA8_INSECURE_STUB_CRYPTO)"
268 if guard not in raw[:offset]:
269 return False
270 idx = raw.rfind(guard, 0, offset)
271 between = raw[idx:offset]
272 return "#else" in between and between.count("#endif") == 0

275def scan_text(raw: str, path: str) -> list[dict]:
276 """Find discard-and-return functions in one translation unit."""
277 code = blank_noncode(raw)
278 found: list[dict] = []
279 for m in FUNCTION_RE.finditer(code):
280 name, params = m.group(1), m.group(2)
281 if name in NOT_A_FUNCTION:
282 continue
283 span = body_span(code, m.end() - 1)
284 if span is None:
285 continue
286 returned = classify_body(code[span[0] : span[1]], params)
287 if returned is None:
288 continue
289 if in_failclosed_crypto_guard(raw, m.start()):
290 continue
291 region = preceding_doc(raw, m.start()) + raw[m.start() : span[1]]
292 found.append(
293 {
294 "path": path,
295 "line": code[: m.start()].count("\n") + 1,
296 "name": name,
297 "returns": returned,
298 "waiver": waiver_in(region),
299 }
300 )
301 return found

304def real_definitions(files: list[Path]) -> dict[str, list[str]]:
305 """Map symbol -> files defining it with a body that does real work."""
306 defs: dict[str, list[str]] = {}
307 for path in files:
308 raw = path.read_text(errors="replace")
309 code = blank_noncode(raw)
310 for m in FUNCTION_RE.finditer(code):
311 name = m.group(1)
312 if name in NOT_A_FUNCTION:
313 continue
314 span = body_span(code, m.end() - 1)
315 if span is None:
316 continue
317 if classify_body(code[span[0] : span[1]], m.group(2)) is None:
318 defs.setdefault(name, []).append(str(path))
319 return defs

322def first_party_sources(explicit: list[str]) -> list[Path]:
323 """Enumerate the tracked first-party .c files under ROOTS.
324
325 Tracked, not globbed. A bare rglob also sweeps in build output -- every
326 configured app leaves a CMake compiler-probe TU at
327 ``<app>/build/CMakeFiles/*/CompilerIdC/CMakeCCompilerId.c`` -- so the set
328 scanned depended on whether the caller had built, and generated code got
329 held to a first-party rule. CI never saw it (it runs against a clean
330 snapshot of committed ``HEAD``) but the pre-commit hook runs in the working
331 tree, which is exactly where a spurious finding costs the most trust.
332 ``git ls-files`` is also what the copyright and @since gates enumerate with.
333 """
334 if explicit:
335 return [Path(p) for p in explicit]
336 try:
337 pathspec = [f"{root}/**/*.c" for root in ROOTS]
338 listed = subprocess.run( # noqa: S603 # fixed argv, no shell
339 ["git", "ls-files", "-z", "--", *pathspec], # noqa: S607 # trusted: fixed git argv
340 capture_output=True,
341 text=True,
342 check=True,
343 ).stdout
344 except (OSError, subprocess.CalledProcessError) as exc:
345 sys.exit(
346 f"check_no_silent_stubs.py: FATAL -- cannot list tracked sources: {exc}\n"
347 " This gate enumerates via git and must not fall back to a glob:\n"
348 " a glob silently scans build output and changes the verdict."
349 )
350 return [
351 Path(name)
352 for name in listed.split("\0")
353 if name and not any(name.startswith(x) for x in EXCLUDED)
354 ]

357def analyse(files: list[Path]) -> list[tuple[str, dict]]:
358 """Return (rule, finding) violations across the given files."""
359 defs = real_definitions(files)
360 violations: list[tuple[str, dict]] = []
361 for path in files:
362 for f in scan_text(path.read_text(errors="replace"), str(path)):
363 # Rule SHADOW: a real definition of the same symbol exists elsewhere.
364 shadowed = [d for d in defs.get(f["name"], []) if d != f["path"]]
365 if shadowed:
366 f["shadows"] = shadowed
367 violations.append(("SHADOW", f))
368 continue
369 # Rule CANNED: claims an operation was refused when none exists.
370 if f["returns"] in CANNED_ERRORS and not f["waiver"]:
371 violations.append(("CANNED", f))
372 return violations

534def selftest(tmp: Path) -> int:
535 """Assert the detector fires on stubs and stays silent on legitimate code."""
536 failures: list[str] = []
537 groups: dict[str, list[tuple[str, str, bool]]] = {}
538 for label, body, should_fire, fname in SELFTEST_CASES:
539 groups.setdefault(label, []).append((body, fname, should_fire))
540
541 for label, cases in groups.items():
542 d = tmp / re.sub(r"\W+", "_", label)
543 d.mkdir(parents=True, exist_ok=True)
544 files = []
545 expect: dict[str, bool] = {}
546 for body, fname, should_fire in cases:
547 p = d / fname
548 p.write_text(body)
549 files.append(p)
550 expect[str(p)] = should_fire
551 fired = {v[1]["path"] for v in analyse(files)}
552 for path, should_fire in expect.items():
553 did = path in fired
554 if did != should_fire:
555 verb = "did not fire" if should_fire else "fired"
556 failures.append(f" {label} [{Path(path).name}]: gate {verb} (unexpected)")
557
558 if failures:
559 print("check_no_silent_stubs.py --selftest: FAILED\n", file=sys.stderr)
560 print("\n".join(failures), file=sys.stderr)
561 return 1
562 total = len(SELFTEST_CASES)
563 fires = sum(1 for c in SELFTEST_CASES if c[2])
564 print(
565 f"check_no_silent_stubs.py --selftest: PASS "
566 f"({total} cases: {fires} must fire, {total - fires} must stay silent)"
567 )
568 return 0

571def _scan_set_is_usable(files: list[Path]) -> bool:
572 """Whether the resolved scan set can be trusted; prints FATAL when not.
573
574 Fail loudly rather than silently passing on a broken scan: a gate that
575 reports success because it looked at nothing is worse than no gate. The
576 roots resolve relative to the current directory, so running this from
577 anywhere but the repository root finds nothing.
578 """
579 if not files:
580 print(
581 "check_no_silent_stubs.py: FATAL -- no first-party sources found.\n"
582 f"Expected .c files under {', '.join(ROOTS)} relative to the current\n"
583 "directory. Run this from the repository root.",
584 file=sys.stderr,
585 )
586 return False
587 missing = [str(p) for p in files if not p.is_file()]
588 if missing:
589 print(
590 "check_no_silent_stubs.py: FATAL -- these paths do not exist:\n "
591 + "\n ".join(missing),
592 file=sys.stderr,
593 )
594 return False
595 return True

598def _report_violations(violations: list[tuple[str, dict]]) -> None:
599 """List every stub found, then explain what each rule means and how to fix it.
600
601 The trailing prose is long on purpose: both rules reject code that
602 compiles, links and looks deliberate, so a bare file:line would leave a
603 reader with no idea why the gate objects.
604 """
605 print(
606 f"check_no_silent_stubs.py: {len(violations)} silent stub(s) found:\n",
607 file=sys.stderr,
608 )
609 for rule, f in sorted(violations, key=lambda v: (v[1]["path"], v[1]["line"])):
610 print(f" [{rule}] {f['path']}:{f['line']} {f['name']}()", file=sys.stderr)
611 if rule == "SHADOW":
612 for other in f["shadows"]:
613 print(f" real implementation lives in {other}", file=sys.stderr)
614 else:
615 print(f" discards its arguments, returns {f['returns']}", file=sys.stderr)
616 print(
617 "\n[SHADOW] A second, do-nothing definition of a symbol that is really\n"
618 "implemented elsewhere in this tree. Whichever one the linker picks, the\n"
619 "build silently disables working code. Compile the real implementation\n"
620 "instead of redefining the symbol -- if it did not link, fix the build\n"
621 "recipe, do not fake the symbol.\n"
622 "\n[CANNED] The function reports that an operation was refused when no\n"
623 "implementation exists at all. Implement it, or delete it and update\n"
624 "every call site in the same change.\n"
625 "\nIf -- and only if -- the capability is blocked on hardware that does\n"
626 "not physically exist yet, mark it with a TODO naming the missing part:\n"
627 " TODO(ESP32-C6 radio module ordered, not yet on the bench)\n"
628 "A bare TODO with no named dependency is not a waiver, and no waiver is\n"
629 "available when a real implementation already exists in the tree.",
630 file=sys.stderr,
631 )

634def main(argv: list[str]) -> int:
635 """Scan first-party C for SHADOW and CANNED stubs, or run the detector selftest.
636
637 An empty source set is FATAL rather than clean, and the message says to run
638 from the repository root: the roots are resolved relative to the current
639 directory, so invoking this from elsewhere finds nothing and would
640 otherwise report a stub-free tree it never opened.
641
642 CI runs ``--selftest`` before the scan for the same reason in the other
643 direction -- a detector whose patterns stopped matching would also report
644 a clean tree, and only an assertion in both directions can tell the two
645 apart.
646
647 Returns 0 when no stub is found, 1 on a finding, on an empty source set,
648 on a named file that does not exist, or on a failing selftest.
649 """
650 ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
651 ap.add_argument("files", nargs="*", help="specific files to scan")
652 ap.add_argument(
653 "--selftest",
654 action="store_true",
655 help="prove the detector fires on stubs and not on legitimate code",
656 )
657 args = ap.parse_args(argv[1:])
658
659 if args.selftest:
660 with tempfile.TemporaryDirectory() as td:
661 return selftest(Path(td))
662
663 files = first_party_sources(args.files)
664 if not _scan_set_is_usable(files):
665 return 1
666
667 violations = analyse(files)
668 if not violations:
669 print(f"check_no_silent_stubs.py: OK ({len(files)} files scanned, no silent stubs)")
