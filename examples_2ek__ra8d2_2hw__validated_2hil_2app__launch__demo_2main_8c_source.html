399static bool app_launch_register(void)
400{
401 if (ra8_app_registry_init(&s_reg, s_slots, (uint16_t)k_app_reg_cap) != k_ra8_ok) {
402 return false;
403 }
404 if (ra8_app_nav_init(&s_nav, &s_reg, s_nav_stack, (uint16_t)k_app_nav_cap) != k_ra8_ok) {
405 return false;

407 s_reader_app = (ra8_app_t){.vt = &k_app_stub_vt,
408 .ctx = &s_reader_ctx,
409 .id = (uint16_t)k_app_id_reader,
410 .name = "reader",
411 .removable = false};

418 .removable = true};
419 if (ok) {
420 ok = (ra8_app_register(&s_reg, &s_settings_app) == k_ra8_ok);

446static bool app_launch_enter_reader(void)
447{
448 uint16_t depth = 1U;
449 if (ra8_app_nav_go(&s_nav, (uint16_t)k_app_id_reader) != k_ra8_ok) {
450 return false;
451 }
452 if (ra8_app_render(&s_reg) != k_ra8_ok) {
453 return false;
454 }
455 if (s_reader_ctx.enters != 1U) {
456 return false;
457 }
458 if (ra8_app_nav_depth(&s_nav, &depth) != k_ra8_ok) {
459 return false;
460 }
461 return (depth == 0U);

484static bool app_launch_route_back(void)
485{
486 const ra8_widget_event_t back = {.kind = k_ra8_widget_ev_button,
487 .button_id = (uint16_t)k_app_btn_back};
488 bool handled = false;
489 if (ra8_app_route_input(&s_reg, &back, &handled) != k_ra8_ok) {
490 return false;
491 }
492 if (!handled) {
493 return false;
494 }
495 return (s_reader_ctx.inputs == 1U);

570static bool app_launch_selfcheck(void)
571{
572 if (!app_launch_enter_reader()) {
573 return false;
574 }
575 if (!app_launch_route_back()) {
576 return false;
577 }
578#if APP_LAUNCH_SETTINGS
579 if (!app_launch_switch_and_back()) {
580 return false;
581 }
582#endif
583 ra8_app_t* act = nullptr;
584 if (ra8_app_active(&s_reg, &act) != k_ra8_ok) {
585 return false;
586 }
587 if (act == nullptr) {
588 return false;
589 }
590 return (act->id == (uint16_t)k_app_id_reader);
591}

608static void app_launch_banner(void)
609{
610 uint16_t n = 0U;
611 (void)ra8_app_count(&s_reg, &n);
612 (void)n; /* used only by the (level-gated) log lines below */
613 ra8_log_info_val("app_launch", "apps", (uint32_t)n);
614 ra8_log_info_val("app_launch", "reader enters", s_reader_ctx.enters);
615 ra8_log_info("app_launch", "demo PASS");
616 /* Additive HIL banner: the Pi rig has no ITM/SWO capture, so mirror the PASS
617 * verdict to the SCI8 / J-Link OB VCOM console for uart_scrape to gate. */
618 app_launch_hil_emit_pass();
619}

634int32_t main(void)
635{
636 ra8_log_init();
637 ra8_log_info("app_launch", "boot");
638
639 /* Bring up the VCOM console so the success path can emit the HIL banner.
640 * Best-effort: a failure is narrated over ITM and the demo continues. */
641 if (!app_launch_hil_console_init()) {
642 ra8_log_info("app_launch", "VCOM console init failed -- HIL banner unavailable");
643 }
644
645 bool ok = app_launch_register();
646 if (!ok) {
647 ra8_log_info("app_launch", "FAIL register");
648 }
649 if (ok) {
650 ok = app_launch_selfcheck();
651 if (!ok) {
652 ra8_log_info("app_launch", "FAIL selfcheck");
653 }
654 }
655 if (ok) {
656 app_launch_banner();
657 }
658
659 while (1) {
660 __asm__ volatile("wfi");
661 }
662}
