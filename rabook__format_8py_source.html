33# .rabook chunked container ("RBKC"; keep in sync with ra8_book_container_t in
34# libs/ra8_book/inc/ra8_book.h):
35# "RBKC" + <I chunk_bytes + <Q inflated_total + <I chunk_count + <I reserved(0)
36# + <Q offset[chunk_count + 1] (payload-relative stream offsets)
37# + chunk_count concatenated zlib streams, one per chunk_bytes slice of the
38# flat blob (last slice short).
39# Every chunk inflates independently, so the device can either inflate all of
40# them into SDRAM (resident open) or inflate single chunks on demand into
41# ra8_vmem cache frames (multi-GB books). chunk_bytes must equal the reader's
42# cache frame size; 64 KiB is the current firmware default.
43CONTAINER_MAGIC = b"RBKC"
44CONTAINER_CHUNK_BYTES = 65536

47def wrap_container(blob: bytes, chunk_bytes: int = CONTAINER_CHUNK_BYTES) -> bytes:
48 """Wrap a flat RABOOK1 blob in the chunked RBKC container."""
49 if not blob:
50 msg = "empty blob"
51 raise ValueError(msg)
52 if chunk_bytes <= 0:
53 msg = "chunk_bytes must be positive"
54 raise ValueError(msg)
55 count = (len(blob) + chunk_bytes - 1) // chunk_bytes
56 streams = [
57 zlib.compress(blob[i * chunk_bytes : (i + 1) * chunk_bytes], 9) for i in range(count)
58 ]
59 offsets = [0]
60 for stream in streams:
61 offsets.append(offsets[-1] + len(stream))
62 header = CONTAINER_MAGIC + struct.pack("<IQII", chunk_bytes, len(blob), count, 0)
63 table = b"".join(struct.pack("<Q", off) for off in offsets)
64 return header + table + b"".join(streams)
