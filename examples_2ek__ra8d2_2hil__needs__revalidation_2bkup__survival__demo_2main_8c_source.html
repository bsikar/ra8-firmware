140static void bkup_demo_setup_or_halt(void)
141{
142 uint32_t cpuclk0_hz = 0U;
143
144 if (ra8_cgc_init() != k_ra8_ok) {
145 bkup_demo_panic_halt();
146 }
147 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
148 bkup_demo_panic_halt();
149 }
150 if (ra8_mstp_init() != k_ra8_ok) {
151 bkup_demo_panic_halt();
152 }
153 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
154 bkup_demo_panic_halt();
155 }
156 if (ra8_board_uart_console_init((uint32_t)k_bkup_demo_baud) != k_ra8_ok) {
157 bkup_demo_panic_halt();
158 }
159 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
160 bkup_demo_panic_halt();
161 }
162 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
163 bkup_demo_panic_halt();
164 }

166 /* Bring the VBATT backup-register block up before any VBTBKRn touch. Two
167 * silicon preconditions must both hold or every VBTBKRn write is dropped:
168 *
169 * 1. Voltage monitor 0 (LVD0) reset must be enabled -- OFS1.PVDAS = 0
170 * (HUM Ch 12.1.3 p 499: "It is necessary to enable voltage monitor 0
171 * resets to use the battery backup function"; Ch 12.3.2 p 514). This
172 * is an option byte set in CMakeLists.txt (OFS1 = 0xFFFFFFF0); without
173 * it the VBATT area stays held in VBATT_POR reset (VBPORF = 1) and the
174 * whole block -- VBTBPCR1, VBTBKRn -- rejects writes. This is the
175 * dominant #131 root cause, verified on the bench by debugger reads.
176 * 2. VBTBER.VBAE must be 1 before access -- HUM Ch 12.2.6 p 504 ("You
177 * must write 1 to VBAE before accessing VBTBKR", then "wait for at
178 * least 500 ns"). VBAE resets to 1, but ra8_bkup_init writes it and
179 * performs the mandatory settle so the sequence is explicit/correct.
180 *
181 * On the EK-RA8D2 VBATT is tied to VCC, so the battery power-supply switch
182 * stays stopped (enable_switch = false). The window stays open for the life
183 * of the demo (a J-Link/SYSRESETREQ reset keeps VCC/VBATT powered, so
184 * leaving VBAE = 1 does not lose data). */
185 const ra8_bkup_config_t bkup_cfg = {
186 .vdet_level = k_ra8_bkup_vdet_2p80v,
187 .enable_switch = false,
188 .enable_backup = true,
189 };
190 if (ra8_bkup_init(&bkup_cfg) != k_ra8_ok) {
191 bkup_demo_panic_halt();
192 }
193}

211[[nodiscard]] static ra8_err_t bkup_demo_rw_check(uint8_t* out_ok)
212{
213 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
214
215 for (uint8_t i = 0U; i < (uint8_t)k_bkup_demo_data_words; ++i) {
216 const ra8_err_t werr = ra8_bkup_write_word(i, bkup_demo_pattern(i));
217 if (werr != k_ra8_ok) {
218 return werr;
219 }
220 }
221 uint8_t ok = 1U;
222 for (uint8_t i = 0U; i < (uint8_t)k_bkup_demo_data_words; ++i) {
223 uint32_t got = 0U;
224 const ra8_err_t rerr = ra8_bkup_read_word(i, &got);
225 if (rerr != k_ra8_ok) {
226 return rerr;
227 }
228 if (got != bkup_demo_pattern(i)) {
229 ok = 0U;
230 }
231 }
232 *out_ok = ok;
233 return k_ra8_ok;
234}

256[[nodiscard]] static ra8_err_t bkup_demo_survival_check(void)
257{
258 uint32_t sentinel = 0U;
259 const ra8_err_t serr = ra8_bkup_read_word((uint8_t)k_bkup_demo_idx_sentinel, &sentinel);
260 if (serr != k_ra8_ok) {
261 return serr;
262 }
263 uint32_t boot = 0U;
264 if (sentinel == (uint32_t)k_bkup_demo_sentinel) {
265 g_bkup_survived = 1U;
266 const ra8_err_t rerr = ra8_bkup_read_word((uint8_t)k_bkup_demo_idx_boot, &boot);
267 if (rerr != k_ra8_ok) {
268 return rerr;
269 }
270 boot += 1U;
271 } else {
272 g_bkup_survived = 0U;
273 boot = 1U;
274 const ra8_err_t merr =
275 ra8_bkup_write_word((uint8_t)k_bkup_demo_idx_sentinel, (uint32_t)k_bkup_demo_sentinel);
276 if (merr != k_ra8_ok) {
277 return merr;
278 }
279 }
280 const ra8_err_t cerr = ra8_bkup_write_word((uint8_t)k_bkup_demo_idx_boot, boot);
281 if (cerr != k_ra8_ok) {
282 return cerr;
283 }
284 g_bkup_boot_count = boot;
285 return k_ra8_ok;
286}

289static void bkup_demo_report_survival(void)
290{
291 const uint8_t* tag = (g_bkup_survived != 0U) ? k_bkup_demo_surv_y : k_bkup_demo_surv_n;
292 /* Both suffixes are the same length ("survived=Y " / "survived=N "). */
293 static_assert(sizeof(k_bkup_demo_surv_y) == sizeof(k_bkup_demo_surv_n),
294 "survived=Y / survived=N suffixes must be equal length");
295 const uint32_t len = (uint32_t)(sizeof(k_bkup_demo_surv_y) - 1U);
296 (void)ra8_board_uart_console_write(tag, (size_t)len);
297}

301int32_t main(void)
302{
303 bkup_demo_setup_or_halt();
304 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
305 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No NVIC sources
306 * are armed by this demo. */
307 ra8_isr_globals_enable();
308
309 if (bkup_demo_survival_check() != k_ra8_ok) {
310 bkup_demo_panic_halt();
311 }
312
313 while (1) {
314 uint8_t rw = 0U;
315 const ra8_err_t err = bkup_demo_rw_check(&rw);
316 const uint8_t good = (err == k_ra8_ok && rw != 0U) ? 1U : 0U;
317 g_bkup_rw_ok = (uint32_t)good;
318
319 if (good != 0U) {
320 (void)ra8_board_uart_console_write(k_bkup_demo_ok_msg,
321 (size_t)(sizeof(k_bkup_demo_ok_msg) - 1U));
322 (void)ra8_board_led_toggle(k_ra8_board_led1);
323 } else {
324 (void)ra8_board_uart_console_write(k_bkup_demo_bad_msg,
325 (size_t)(sizeof(k_bkup_demo_bad_msg) - 1U));
326 (void)ra8_board_led_toggle(k_ra8_board_led2);
327 }
328 bkup_demo_report_survival();
329 (void)ra8_board_uart_console_write(k_bkup_demo_crlf, (size_t)(sizeof(k_bkup_demo_crlf) - 1U));
330 ++g_bkup_heartbeat;
331 ra8_delay_ms(k_bkup_demo_period_ms);
332 }
333 bkup_demo_panic_halt();
334 return 0;
335}
