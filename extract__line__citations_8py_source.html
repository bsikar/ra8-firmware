4"""extract_line_citations.py -- Emit a CSV of in-tree line citations.
5
6Companion to scripts/checks/check_line_citations.py. Walks the same set
7of files (libs/, src/, tests/, examples/, port/), finds every
8`<file>.<ext>:<line>` token inside C/C++ comments, and emits one row
9per violation:
10
11 file,line_number,matched_citation,enclosing_function,
12 suggested_replacement,full_comment_snippet
13
14`enclosing_function` is the most recent function definition signature
15above the citation in the same file (or "(file scope)" if none).
16
17`suggested_replacement` resolves the citation target -- opens
18`<file>` at `<line>` and reports `<basename>::<func_name>` so the
19caller can rewrite `libs/foo/src/bar.c:776` as `bar.c::priv_foo`.
20
21Output: CSV on stdout. One-shot agent tool, not wired into pre-commit.
22"""

47# Heuristic: a function definition has a return type, name, params, and
48# an opening brace either on the signature line or on the next line.
49FUNC_DEF_RE = re.compile(r"^[A-Za-z_][\w\s\*]*?\b([A-Za-z_]\w*)\s*\‍([^;]*?\‍)\s*\{?\s*$")
50# Common keywords that look like function defs but aren't.
51FUNC_DEF_DENYLIST = {
52 "if",
53 "for",
54 "while",
55 "switch",
56 "return",
57 "sizeof",
58 "static_assert",
59 "typeof",
60 "do",
61}

64def enclosing_function(lines: list[str], target_line: int) -> str:
65 """Walk backward from target_line looking for a function definition."""
66 for idx in range(min(target_line - 1, len(lines) - 1), -1, -1):
67 line = lines[idx]
68 # Skip lines inside comments / preprocessor / blank
69 stripped = line.strip()
70 if not stripped or stripped.startswith(("//", "/*", "*", "#")):
71 continue
72 m = FUNC_DEF_RE.match(line)
73 if not m:
74 continue
75 name = m.group(1)
76 if name in FUNC_DEF_DENYLIST:
77 continue
78 # Sanity: avoid matching variable declarations like `int x = foo();`
79 if "=" in line.split("(")[0]:
80 continue
81 return name
82 return "(file scope)"

85_FILE_CACHE: dict[Path, list[str]] = {}

88def file_lines(path: Path) -> list[str] | None:
89 """Cached line list for a file, or None when it cannot be read.
90
91 Cached because resolving the enclosing function for each citation re-reads
92 the same few files repeatedly; the cache turns that quadratic re-read into
93 one read per file.
94 """
95 if path in _FILE_CACHE:
96 return _FILE_CACHE[path]
97 try:
98 text = path.read_text(encoding="utf-8", errors="replace")
99 except OSError:
100 return None
101 lines = text.splitlines()
102 _FILE_CACHE[path] = lines
103 return lines

106def resolve_target(repo_root: Path, citation_path: str, citation_line: int) -> str:
107 """Try to find the target file and report basename::func."""
108 # Try as repo-relative
109 candidates: list[Path] = []
110 p = repo_root / citation_path
111 if p.is_file():
112 candidates.append(p)
113 else:
114 # Search by basename across tracked files
115 base = Path(citation_path).name
116 for f in all_tracked_files():
117 if Path(f).name == base and is_in_scope(f):
118 candidates.append(repo_root / f)
119 break
120 if not candidates:
121 return f"{Path(citation_path).name}::(unknown)"
122 target = candidates[0]
123 lines = file_lines(target)
124 if lines is None:
125 return f"{target.name}::(unreadable)"
126 func = enclosing_function(lines, citation_line)
127 return f"{target.name}::{func}"

130def is_in_scope(path: str) -> bool:
131 """Whether a source path is subject to the ban, with this tool's exclusions."""
132 return _lex_in_scope(path, EXCLUDE_PREFIXES)

135def _rows_for_file(repo_root: Path, rel: str) -> Iterator[list[object]]:
136 """Yield one CSV row per non-exempt line-citation in ``rel``.
137
138 Exemptions come from line_citation_lex.is_exempt -- the same predicate the
139 gate applies -- so this aid cannot list work the gate does not want, nor
140 miss work it does. It used to carry its own hand-copied cascade.
141 """
142 path = repo_root / rel
143 if not path.is_file():
144 return
145 try:
146 text = path.read_text(encoding="utf-8", errors="replace")
147 except OSError:
148 return
149 lines = text.splitlines()
150 for start, end in find_comment_spans(text):
151 for m in CITATION_RE.finditer(text[start:end]):
152 line_no = line_of_offset(text, start + m.start())
153 matched = m.group(0)
154 line = lines[line_no - 1] if 1 <= line_no <= len(lines) else ""
155 if is_exempt(matched, line):
156 continue
157 yield [
158 rel,
159 line_no,
160 matched,
161 enclosing_function(lines, line_no),
162 resolve_target(repo_root, m.group(1), int(m.group(2))),
163 line.strip(),
164 ]

167def main() -> int:
168 """List every in-tree ``file:line`` citation with the function that encloses it.
169
170 A migration aid, not a gate: check_line_citations is what FAILS on these,
171 and this prints the same set together with the symbol name each one should
172 be rewritten to cite. Exits 0 regardless of what it finds.
173 """
174 repo_root = Path(
175 subprocess.run(
176 ["git", "rev-parse", "--show-toplevel"], # noqa: S607 # trusted: fixed git argv
177 check=True,
178 capture_output=True,
179 text=True,
180 ).stdout.strip()
181 )
182
183 files = [f for f in all_tracked_files() if is_in_scope(f)]
184
185 writer = csv.writer(sys.stdout)
186 writer.writerow(
187 [
188 "file",
189 "line_number",
190 "matched_citation",
191 "enclosing_function",
192 "suggested_replacement",
193 "full_comment_snippet",
194 ]
195 )
196
197 rows = 0
198 for f in files:
199 for row in _rows_for_file(repo_root, f):
200 writer.writerow(row)
201 rows += 1
202
203 print(f"# {rows} rows", file=sys.stderr)
204 return 0
