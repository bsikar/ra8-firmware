234[[nodiscard]] static ra8_err_t selftest_print_dec(uint32_t value)
235{
236 uint8_t scratch[k_selftest_dec_chars_u32] = {};
237 uint8_t out[k_selftest_dec_chars_u32] = {};
238 uint8_t count = 0U;
239 uint32_t v = value;
240 if (v == 0U) {
241 out[0] = (uint8_t)'0';
242 return selftest_sci_write(out, 1U);
243 }
244 while (v != 0U) {
245 if (count >= (uint8_t)k_selftest_dec_chars_u32) {
246 break;
247 }
248 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_selftest_dec_radix));
249 v = v / k_selftest_dec_radix;
250 count++;
251 }
252 for (uint8_t i = 0U; i < count; i++) {
253 out[i] = scratch[count - 1U - i];
254 }
255 return selftest_sci_write(out, (uint32_t)count);
256}

277[[nodiscard]] static ra8_err_t selftest_print_hex(uint32_t value, uint8_t digits)
278{
279 uint8_t out[k_selftest_hex_chars_u32] = {};
280 uint8_t width = digits;
281 if (width > (uint8_t)k_selftest_hex_chars_u32) {
282 width = (uint8_t)k_selftest_hex_chars_u32;
283 }
284 for (uint8_t i = 0U; i < width; i++) {
285 const uint8_t shift = (uint8_t)((width - 1U - i) * k_selftest_nibble_bits);
286 out[i] = selftest_nibble_to_hex((value >> shift) & k_selftest_nibble_mask);
287 }
288 return selftest_sci_write(out, (uint32_t)width);
289}

530[[nodiscard]] static ra8_err_t selftest_verify_mram_raw(void)
531{
532 static uint8_t s_burst[k_selftest_burst_bytes] = {};
533
534 s_dbg_verified_bytes = 0U;
535 s_dbg_mismatch_off = (uint32_t)k_selftest_no_mismatch;
536 const uint32_t t0 = ra8_time_ms();
537 for (uint32_t blk = 0U; blk < (uint32_t)k_fat_mram_clusters;
538 blk += (uint32_t)k_selftest_burst_blocks) {
539 const uint32_t lba = (uint32_t)k_fat_data_lba + blk;
540 ra8_err_t err = ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun,
541 lba,
542 (uint16_t)k_selftest_burst_blocks,
543 s_burst);
544 if (err != k_ra8_ok) {
545 (void)selftest_print_fail("READ(10) burst", err);
546 return err;
547 }
548 const uint32_t offset = blk * (uint32_t)k_selftest_block_size;
549 const uint8_t* mram = (const uint8_t*)(uintptr_t)((uint32_t)k_mram_base_addr + offset);
550 if (memcmp(s_burst, mram, (size_t)k_selftest_burst_bytes) != 0) {
551 s_dbg_mismatch_off = offset;
552 (void)selftest_print_fail("content mismatch at 0x", (ra8_err_t)offset);
553 return k_ra8_err_invalid_state;
554 }
555 s_dbg_verified_bytes = offset + (uint32_t)k_selftest_burst_bytes;
556 }
557 s_dbg_verify_ms = ra8_time_ms() - t0;
558 return k_ra8_ok;
559}

745[[nodiscard]] static ra8_err_t selftest_browse_root(void)
746{
747 static uint8_t s_dir[k_selftest_block_size] = {};
748
749 ra8_err_t err =
750 ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun, (uint32_t)k_fat_root_lba, 1U, s_dir);
751 if (err != k_ra8_ok) {
752 (void)selftest_print_fail("READ(10) root dir", err);
753 return err;
754 }
755 const uint8_t* entry = &s_dir[k_dir_entry_bytes]; /* entry 0 = volume label */
756 char name[(uint32_t)k_dir_name_bytes + 1U];
757 for (uint32_t i = 0U; i < (uint32_t)k_dir_name_bytes; i++) {
758 name[i] = (char)entry[i];
759 }
760 name[(uint32_t)k_dir_name_bytes] = '\0';
761 uint32_t size = 0U;
762 for (uint32_t i = 0U; i < 4U; i++) {
763 size |= (uint32_t)entry[(uint32_t)k_dir_off_size + i] << (i * (uint32_t)k_byte_shift);
764 }
765 err = selftest_print("ra8d2 host: browsed root, file=");
766 if (err == k_ra8_ok) {
767 err = selftest_print(name);
768 }
769 if (err == k_ra8_ok) {
770 err = selftest_print(" size=");
771 }
772 if (err == k_ra8_ok) {
773 err = selftest_print_dec(size);
774 }
775 if (err == k_ra8_ok) {
776 err = selftest_print(" bytes\r\n");
777 }
778 return err;

783[[nodiscard]] ra8_err_t selftest_host_pass(void)
784{
785 ra8_usb_hmsc_device_t device = {};
786 ra8_err_t err = selftest_host_enumerate(&device);
787 if (err != k_ra8_ok) {
788 return err;
789 }
790
791 /* Mount proves the host can PARSE the FAT16 volume over the loop;
792 * then unmount and do the heavy integrity check with raw multi-block
793 * READ(10) (ra8_fs's per-cluster metadata re-reads are far too slow
794 * for a 1 MiB sweep). */
795 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
796 ra8_fs_mount_t* mount = nullptr;
797 err = selftest_mount_volume(&mount);
798 if (err != k_ra8_ok) {
799 (void)ra8_usb_hmsc_close();
800 return err;
801 }
802 (void)ra8_fs_unmount(mount);
803
804 /* Browse the FAT directory of the fake peripheral (the host-side
805 * directory walk this app adds on top of the raw integrity check). */
806 err = selftest_browse_root();
807 if (err != k_ra8_ok) {
808 (void)ra8_usb_hmsc_close();
809 return err;
810 }
811
812 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
813 err = selftest_verify_mram_raw();
814 if (err == k_ra8_ok) {
815 err = selftest_print_verify_verdict();
816 }
817 if (err == k_ra8_ok) {
818 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
819 err = selftest_write_protect_probe();
820 }
821 if (err != k_ra8_ok) {
822 (void)ra8_usb_hmsc_close();
823 return err;
824 }
825
826 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
827 s_dbg_pass_count++;
828 err = selftest_print("ra8d2 host: USB HOST MSC BROWSE PASS\r\n");
829 if (err != k_ra8_ok) {
830 return err;
831 }
832 (void)ra8_board_led_on(k_ra8_board_led2);
833 return k_ra8_ok;
834}
