234[[nodiscard]] static ra8_err_t selftest_print_dec(uint32_t value)
235{
236 uint8_t scratch[k_selftest_dec_chars_u32] = {};
237 uint8_t out[k_selftest_dec_chars_u32] = {};
238 uint8_t count = 0U;
239 uint32_t v = value;
240 if (v == 0U) {
241 out[0] = (uint8_t)'0';
242 return selftest_sci_write(out, 1U);
243 }
244 while (v != 0U) {
245 if (count >= (uint8_t)k_selftest_dec_chars_u32) {
246 break;
247 }
248 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_selftest_dec_radix));
249 v = v / k_selftest_dec_radix;
250 count++;
251 }
252 for (uint8_t i = 0U; i < count; i++) {
253 out[i] = scratch[count - 1U - i];
254 }
255 return selftest_sci_write(out, (uint32_t)count);
256}

277[[nodiscard]] static ra8_err_t selftest_print_hex(uint32_t value, uint8_t digits)
278{
279 uint8_t out[k_selftest_hex_chars_u32] = {};
280 uint8_t width = digits;
281 if (width > (uint8_t)k_selftest_hex_chars_u32) {
282 width = (uint8_t)k_selftest_hex_chars_u32;
283 }
284 for (uint8_t i = 0U; i < width; i++) {
285 const uint8_t shift = (uint8_t)((width - 1U - i) * k_selftest_nibble_bits);
286 out[i] = selftest_nibble_to_hex((value >> shift) & k_selftest_nibble_mask);
287 }
288 return selftest_sci_write(out, (uint32_t)width);
289}

547[[nodiscard]] static ra8_err_t selftest_verify_mram_raw(void)
548{
549 static uint8_t s_burst[k_selftest_burst_bytes] = {};
550
551 s_dbg_verified_bytes = 0U;
552 s_dbg_mismatch_off = (uint32_t)k_selftest_no_mismatch;
553 const uint32_t t0 = ra8_time_ms();
554 for (uint32_t blk = 0U; blk < (uint32_t)k_fat_mram_clusters;
555 blk += (uint32_t)k_selftest_burst_blocks) {
556 const uint32_t lba = (uint32_t)k_fat_data_lba + blk;
557 ra8_err_t err = ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun,
558 lba,
559 (uint16_t)k_selftest_burst_blocks,
560 s_burst);
561 if (err != k_ra8_ok) {
562 (void)selftest_print_fail("READ(10) burst", err);
563 return err;
564 }
565 const uint32_t offset = blk * (uint32_t)k_selftest_block_size;
566 const uint8_t* mram = (const uint8_t*)(uintptr_t)((uint32_t)k_mram_base_addr + offset);
567 if (memcmp(s_burst, mram, (size_t)k_selftest_burst_bytes) != 0) {
568 s_dbg_mismatch_off = offset;
569 (void)selftest_print_fail("content mismatch at 0x", (ra8_err_t)offset);
570 return k_ra8_err_invalid_state;
571 }
572 s_dbg_verified_bytes = offset + (uint32_t)k_selftest_burst_bytes;
573 }
574 s_dbg_verify_ms = ra8_time_ms() - t0;
575 return k_ra8_ok;
576}

762[[nodiscard]] static ra8_err_t selftest_browse_root(void)
763{
764 static uint8_t s_dir[k_selftest_block_size] = {};
765
766 ra8_err_t err =
767 ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun, (uint32_t)k_fat_root_lba, 1U, s_dir);
768 if (err != k_ra8_ok) {
769 (void)selftest_print_fail("READ(10) root dir", err);
770 return err;
771 }
772 const uint8_t* entry = &s_dir[k_dir_entry_bytes]; /* entry 0 = volume label */
773 char name[(uint32_t)k_dir_name_bytes + 1U];
774 for (uint32_t i = 0U; i < (uint32_t)k_dir_name_bytes; i++) {
775 name[i] = (char)entry[i];
776 }
777 name[(uint32_t)k_dir_name_bytes] = '\0';
778 uint32_t size = 0U;
779 for (uint32_t i = 0U; i < 4U; i++) {
780 size |= (uint32_t)entry[(uint32_t)k_dir_off_size + i] << (i * (uint32_t)k_byte_shift);
781 }
782 err = selftest_print("ra8d2 host: browsed root, file=");
783 if (err == k_ra8_ok) {
784 err = selftest_print(name);
785 }
786 if (err == k_ra8_ok) {
787 err = selftest_print(" size=");
788 }
789 if (err == k_ra8_ok) {
790 err = selftest_print_dec(size);
791 }
792 if (err == k_ra8_ok) {
793 err = selftest_print(" bytes\r\n");
794 }
795 return err;
796}

800[[nodiscard]] ra8_err_t selftest_host_pass(void)
801{
802 ra8_usb_hmsc_device_t device = {};
803 ra8_err_t err = selftest_host_enumerate(&device);
804 if (err != k_ra8_ok) {
805 return err;
806 }
807
808 /* Mount proves the host can PARSE the FAT16 volume over the loop;
809 * then unmount and do the heavy integrity check with raw multi-block
810 * READ(10) (ra8_fs's per-cluster metadata re-reads are far too slow
811 * for a 1 MiB sweep). */
812 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
813 ra8_fs_mount_t* mount = nullptr;
814 err = selftest_mount_volume(&mount);
815 if (err != k_ra8_ok) {
816 (void)ra8_usb_hmsc_close();
817 return err;
818 }
819 (void)ra8_fs_unmount(mount);
820
821 /* Browse the FAT directory of the fake peripheral (the host-side
822 * directory walk this app adds on top of the raw integrity check). */
823 err = selftest_browse_root();
824 if (err != k_ra8_ok) {
825 (void)ra8_usb_hmsc_close();
826 return err;
827 }
828
829 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
830 err = selftest_verify_mram_raw();
831 if (err == k_ra8_ok) {
832 err = selftest_print_verify_verdict();
833 }
834 if (err == k_ra8_ok) {
835 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
836 err = selftest_write_protect_probe();
837 }
838 if (err != k_ra8_ok) {
839 (void)ra8_usb_hmsc_close();
840 return err;
841 }
842
843 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
844 s_dbg_pass_count++;
845 err = selftest_print("ra8d2 host: USB HOST MSC BROWSE PASS\r\n");
846 if (err != k_ra8_ok) {
847 return err;
848 }
849 (void)ra8_board_led_on(k_ra8_board_led2);
850 return k_ra8_ok;
851}
