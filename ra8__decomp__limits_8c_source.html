99RA8_INTERNAL static uint64_t s_ratio_bound(const ra8_decomp_limits_t* lim, uint64_t in_total)
100{
101 const uint64_t ratio = (uint64_t)lim->max_ratio;
102 if (in_total > (UINT64_MAX / ratio)) {
103 return UINT64_MAX; /* product would wrap: saturate (output cap governs) */
104 }
105 const uint64_t product = in_total * ratio;
106 const uint64_t grace = (uint64_t)lim->ratio_grace_bytes;
107 if (product > (UINT64_MAX - grace)) {
108 return UINT64_MAX; /* sum would wrap: saturate (output cap governs) */
109 }
110 return product + grace;
111}

129ra8_decomp_budget_charge_output(ra8_decomp_budget_t* b, uint64_t in_total, uint64_t out_delta)
130{
131 RA8_CHECK_NULL_PTR(b, s_tag_decomp, "charge output: null budget");
132 if (out_delta > (UINT64_MAX - b->out_bytes)) {
133 b->out_bytes = UINT64_MAX; /* saturate: any wrap-sized delta is a breach */
134 } else {
135 b->out_bytes += out_delta;
136 }
137 b->in_bytes = in_total;
138 if (b->out_bytes > b->limits.max_output_bytes) {
139 return k_ra8_err_decomp_output_cap;
140 }
141 if (b->out_bytes > s_ratio_bound(&b->limits, in_total)) {
142 return k_ra8_err_decomp_ratio;
143 }
144 return k_ra8_ok;
145}
