172int32_t main(void)
173{
174 lpm_deep_setup_or_halt();
175 ra8_isr_globals_enable();
176
177 (void)ra8_board_uart_console_write(k_lpm_deep_msg_boot,
178 (size_t)(sizeof(k_lpm_deep_msg_boot) - 1U));
179
180 g_lpm_deep_pre_count++;
181
182 /* SysTick does NOT wake from Deep-Sleep on RA8D2 under the
183 * default LPM config -- bench-verified 2026-05-27. The chip
184 * enters WFI and stays there. The HIL gate therefore scrapes
185 * the BOOT banner (emitted just above) rather than a post-wake
186 * banner; the gate proves the firmware loaded and reached the
187 * LPM entry, not that wake worked. A real wake-from-deep-sleep
188 * demo would need RTC/AGT (sub-clock-sourced, survives
189 * SLEEPDEEP) as the wake source instead of SysTick. */
190 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_deep_sleep) != k_ra8_ok) {
191 lpm_deep_panic_halt();
192 }
193
194 g_lpm_deep_wake_count++;
195
196 (void)ra8_board_uart_console_write(k_lpm_deep_msg_woke,
197 (size_t)(sizeof(k_lpm_deep_msg_woke) - 1U));
198
199 /* Park in a normal (non-LPM) loop so subsequent bench flashes can
200 * halt the CPU without an Initialize step. SysTick + the CPU stay
201 * in their default active state. */
202 while (1) {
203 (void)ra8_board_led_toggle(k_ra8_board_led1);
204 ra8_delay_ms((uint32_t)k_lpm_deep_park_blink_ms);
205 }
206 lpm_deep_panic_halt();
207 return 0;
208}
