52static bool idle_back_edge(uc_engine* uc, uint32_t at, uint16_t hw, uint32_t aligned, bool* done)
53{
54 *done = false;
55 if ((hw & (uint16_t)k_op_bn_mask) != (uint16_t)k_op_bn_base) {
56 return false; /* not an unconditional b.n -- still inside the loop body */
57 }
58 *done = true;
59 /* Unconditional b.n: target = at + 4 + sign_extend(imm11) * 2. */
60 const uint32_t imm11 = (uint32_t)(hw & (uint16_t)k_op_bn_imm);
61 const int32_t off = (int32_t)(imm11 << (uint32_t)k_bn_imm_sext_shl) >> (int32_t)k_bn_imm_sext_shr;
62 if (off >= 0) {
63 return false; /* forward branch -- not a spin-in-place back-edge */
64 }
65 const uint32_t target = at + (uint32_t)k_thumb2_insn_bytes + (uint32_t)off;
66 if ((target > aligned) || ((aligned - target) > (uint32_t)k_idle_loop_max)) {
67 return false; /* back-edge does not bracket pc in a tight loop */
68 }
69 /* pc is enclosed by [target, at]: require a wait (cpsie/wfi) in the body. */
70 for (uint32_t k = target; k <= at; k += (uint32_t)k_thumb_hw_bytes) {
71 uint16_t bhw = 0U;
72 if (uc_mem_read(uc, (uint64_t)k, &bhw, sizeof(bhw)) != UC_ERR_OK) {
73 return false;
74 }
75 if ((bhw == (uint16_t)k_op_cpsie_i) || (bhw == (uint16_t)k_op_wfi)) {
76 return true;
77 }
78 }
79 return false; /* tight loop but no wait signature -- a busy loop, not idle */
80}

118bool idle_spin_at(uc_engine* uc, uint32_t pc)
119{
120 const uint32_t aligned = pc & ~1U;
121 uint16_t hw0 = 0U;
122 if (uc_mem_read(uc, (uint64_t)aligned, &hw0, sizeof(hw0)) != UC_ERR_OK) {
123 return false;
124 }
125 /* Case 1: the core is literally on a halt instruction. */
126 if ((hw0 == (uint16_t)k_op_branch_self) || (hw0 == (uint16_t)k_op_wfi)) {
127 return true;
128 }
129
130 /* Case 2: scan forward for the loop's unconditional backward back-edge. */
131 for (uint32_t j = 0U; j < (uint32_t)k_idle_scan_fwd; j++) {
132 const uint32_t at = aligned + (j * (uint32_t)k_thumb_hw_bytes);
133 uint16_t hw = 0U;
134 if (uc_mem_read(uc, (uint64_t)at, &hw, sizeof(hw)) != UC_ERR_OK) {
135 return false;
136 }
137 bool done = false;
138 const bool verdict = idle_back_edge(uc, at, hw, aligned, &done);
139 if (done) {
140 return verdict;
141 }
142 }
143 return false;
144}
