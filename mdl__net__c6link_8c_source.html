91RA8_INTERNAL static ra8_err_t
92internal_c6_output_write(void* ctx, const uint8_t* data, uint16_t len, uint16_t* written)
93{
94 mdl_c6_output_t* output = (mdl_c6_output_t*)ctx;
95 *written = 0U;
96 if (output->kind == k_mdl_c6_output_buffer) {
97 if ((output->length >= output->cap) || ((size_t)len > (output->cap - output->length - 1U))) {
98 return k_ra8_err_no_mem;
99 }
100 (void)memcpy(&output->buffer[output->length], data, len);
101 } else {
102 uint32_t accepted = 0U;
103 const ra8_err_t sink_result = output->sink->write(output->sink->ctx, data, len, &accepted);
104 if (sink_result != k_ra8_ok) {
105 return sink_result;
106 }
107 if (accepted != len) {
108 return k_ra8_err_invalid_size;
109 }
110 }
111 output->length += len;
112 *written = len;
113 return k_ra8_ok;
114}

153RA8_INTERNAL static ra8_err_t internal_c6_output_abort(void* ctx)
154{
155 mdl_c6_output_t* output = (mdl_c6_output_t*)ctx;
156 ra8_err_t result = k_ra8_ok;
157 if (output->kind == k_mdl_c6_output_buffer) {
158 output->buffer[0] = '\0';
159 } else {
160 result = output->sink->reset(output->sink->ctx);
161 }
162 if (result == k_ra8_ok) {
163 output->length = 0U;
164 }
165 return result;
166}

190RA8_INTERNAL static ra8_err_t internal_c6_get(mdl_net_c6link_t* backend,
191 const char* url,
192 const mdl_net_req_t* req,
193 mdl_c6_output_t* output,
194 size_t* out_len,
195 mdl_net_resp_t* resp)
196{
197 const ra8_mdl_transfer_config_t config = {
198 .storage = {.begin = internal_c6_output_begin,
199 .write = internal_c6_output_write,
200 .validate = nullptr,
201 .commit = internal_c6_output_commit,
202 .abort = internal_c6_output_abort,
203 .ctx = output},
204 .sha256 = backend->sha256,
205 .format = k_mdl_format_loose,
206 .http = {.user_agent = req->user_agent,
207 .referer = req->referer,
208 .if_none_match = req->if_none_match,
209 .if_modified_since = req->if_modified_since,
210 .timeout_ms = req->timeout_ms},
211 .chunk_bytes = backend->chunk_bytes,
212 .max_chunks = backend->max_chunks,
213 };
214 ra8_mdl_transfer_result_t transfer = {};
215 const ra8_err_t result =
216 ra8_c6link_mdl_transfer(backend->link, url, "c6-response", &config, &transfer);
217 if (result == k_ra8_ok) {
218 if (transfer.bytes_stored > SIZE_MAX) {
219 return k_ra8_err_invalid_size;
220 }
221 if (resp != nullptr) {
222 resp->status = transfer.response.status;
223 (void)memcpy(resp->retry_after, transfer.response.retry_after, sizeof(resp->retry_after));
224 (void)memcpy(resp->etag, transfer.response.etag, sizeof(resp->etag));
225 (void)memcpy(resp->last_modified,
226 transfer.response.last_modified,
227 sizeof(resp->last_modified));
228 (void)memcpy(resp->content_type, transfer.response.content_type, sizeof(resp->content_type));
229 }
230 const ra8_err_t classified = priv_mdl_net_classify_http(transfer.response.status);
231 if (classified != k_ra8_ok) {
232 (void)internal_c6_output_abort(output);
233 return classified;
234 }
235 if (out_len != nullptr) {
236 *out_len = (size_t)transfer.bytes_stored;
237 }
238 }
239 return result;
240}
