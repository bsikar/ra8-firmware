3"""Regression tests for the annotation checker itself, on a synthetic tree.
4
5Every defect guarded here is one this gate has actually shipped, and they share
6a shape: the checker kept reporting a number, the number got smaller, and
7smaller read as better. So each assertion below is made in BOTH directions --
8the rule fires on the broken fixture *and* stays quiet on the correct one --
9because a rule can be "fixed" by defanging it and no single-direction test can
10tell the difference.
11
12The fixtures are laid out like ``libs/<module>/`` and ``tools/<tool>/`` so
13``module_of()`` and ``is_first_party()`` resolve against them, with the
14annotations spelled the way ``ra8_attributes.h`` lowers them so no project
15header is needed. Scope is pointed at the temporary tree through
16:func:`annot_scope.override_repo_root`, never by rebinding a module global:
17a rebinding is invisible to other modules' imports, which would leave this
18suite asserting against the real checkout and passing without proving anything.
19"""

37#: Insertion order is load-bearing for the namesake case: it reproduces the
38#: walk order that once took dev red. The namesake `static` is walked first,
39#: so under a name-keyed table its USR was the one latched into the single
40#: merged entry while that entry's `file` was overwritten by the RA8_PRIV
41#: module's definition -- and a module's call to its own file-local helper

218#: What run_selftest() expects the linkage rule to report, by symbol name.
219#: ``handler_untabled`` is here because it is byte-identical to a handler
220#: the table does name: table membership is the only thing separating them.
221#: ``host_unpublished`` lives under ``tools/`` and is the scope assertion:
222#: the linkage rule only judges files is_first_party() accepts, so this name
223#: goes missing the moment ``tools`` drops out of SCAN_DIRS.
224_SELFTEST_LINKAGE_EXPECTED = frozenset(
225 {"link_internal_declared", "link_undeclared", "handler_untabled", "host_unpublished"}
226)

228#: Definitions the linkage rule must leave alone -- one per passing shape,
229#: plus the un-tabled handler's twin that proves the exemption is keyed on
230#: table membership rather than on what the function looks like.
231_SELFTEST_LINKAGE_CLEAN = (
232 "link_public_api",
233 "link_internal_annotated",
234 "link_test_hook",
235 "link_file_local",
236 "main",
237 "handler_tabled",
238)

245def _selftest_parse(root: pathlib.Path) -> _SelftestParse:
246 """Write the synthetic tree under ``root`` and walk every TU in it."""
247 tu_paths: list[pathlib.Path] = []
248 for rel, body in _SELFTEST_SOURCES.items():
249 path = root / rel
250 path.parent.mkdir(parents=True, exist_ok=True)
251 path.write_text(body)
252 if path.suffix in SOURCE_SUFFIXES:
253 tu_paths.append(path)
254
255 args = [
256 "-std=c23",
257 "-x",
258 "c",
259 f"-I{root}/libs/mod_link/inc",
260 f"-I{root}/libs/mod_link/src",
261 ]
262 symbols: dict[str, AnnotatedSymbol] = {}
263 calls: list[CallSite] = []
264 vector_entries: set[str] = set()
265 index = cindex.Index.create()
266 for path in tu_paths:
267 tu = index.parse(
268 str(path),
269 args=args,
270 options=cindex.TranslationUnit.PARSE_DETAILED_PROCESSING_RECORD,
271 )
272 walk_tu(tu, path, symbols, calls, ParseStats(), vector_entries)
273 return symbols, calls, vector_entries

276def _names_matching(violations: list[Violation], rule: str, pattern: str) -> set[str]:
277 """Pull the symbol names out of every ``rule`` finding matching ``pattern``."""
278 out: set[str] = set()
279 for v in violations:
280 if v.rule != rule:
281 continue
282 m = re.search(pattern, v.message)
283 if m:
284 out.add(m.group(1))
285 return out

288def _check_priv_namesakes(
289 violations: list[Violation], symbols: dict[str, AnnotatedSymbol]
290) -> list[str]:
291 """RA8_PRIV must separate namesakes by USR without going toothless."""
292 offenders = {pathlib.Path(v.file).name for v in violations if v.rule == "ra8_priv"}
293 failures: list[str] = []
294 if "other.c" in offenders:
295 failures.append(
296 "namesake regression: a module's call to its own file-local static "
297 "'shared_helper' was reported as a cross-module RA8_PRIV call"
298 )
299 if "priv.c" in offenders:
300 failures.append(
301 "same-module regression: mod_priv calling its own RA8_PRIV symbol was reported"
302 )
303 if "stranger.c" not in offenders:
304 failures.append(
305 "ra8_priv went toothless: the genuine cross-module call to RA8_PRIV "
306 "'shared_helper' from mod_stranger was NOT reported"
307 )
308 if "other_host.c" not in offenders:
309 failures.append(
310 "ra8_priv is blind under tools/: mod_other_host calls mod_host's "
311 "RA8_PRIV 'host_priv_helper' across a module boundary and it was NOT "
312 "reported -- module_of() is not resolving tools/<tool> as a module, so "
313 "every RA8_PRIV tag under tools/ is decorative"
314 )
315
316 # The merged symbol table is the underlying defect, so assert its shape
317 # directly too: the namesakes must stay distinct entries.
318 namesakes = [s for s in symbols.values() if s.name == "shared_helper"]
319 expected_namesakes = 2 # one external (mod_priv) + one static (mod_other)
320 if len(namesakes) != expected_namesakes:
321 failures.append(
322 f"symbol table merged namesakes: expected {expected_namesakes} distinct "
323 f"'shared_helper' entries, found {len(namesakes)}"
324 )
325 return failures

328def _check_linkage(violations: list[Violation]) -> list[str]:
329 """The linkage rule must catch both gap shapes and exempt only tabled handlers."""
330 linkage = _names_matching(violations, "ra8_linkage", r"'([^']+)'")
331 failures = [
332 f"ra8_linkage went toothless: '{name}' has external linkage that "
333 f"nothing justifies, and the rule did not report it"
334 for name in sorted(_SELFTEST_LINKAGE_EXPECTED - linkage)
335 ]
336 failures.extend(
337 f"ra8_linkage false positive: '{name}' is a justified definition but the rule reported it"
338 for name in sorted(linkage - _SELFTEST_LINKAGE_EXPECTED)
339 )
340 if "handler_untabled" not in linkage:
341 failures.append(
342 "vector-table exemption over-matches: 'handler_untabled' is byte-identical "
343 "to a tabled handler but appears in no table, so it must still be reported"
344 )
345 return failures

348def _check_rule3(violations: list[Violation]) -> list[str]:
349 """NASA P10 Rule 3, both axes: the waiver and the firmware/host boundary."""
350 allocators = _names_matching(violations, "ra8_nasa_rule_3_ok", r"from '([^']+)'")
351 failures: list[str] = []
352 if "fw_untagged_allocator" not in allocators:
353 failures.append(
354 "ra8_nasa_rule_3_ok went toothless: firmware function "
355 "'fw_untagged_allocator' calls malloc/free with no waiver and was not reported"
356 )
357 if "fw_waived_allocator" in allocators:
358 failures.append(
359 "ra8_nasa_rule_3_ok false positive: 'fw_waived_allocator' carries "
360 "RA8_NASA_RULE_3_OK, which is exactly the documented waiver"
361 )
362 if "host_allocator" in allocators:
363 failures.append(
364 "ra8_nasa_rule_3_ok false positive: 'host_allocator' is under tools/, "
365 "which the host toolchain compiles and no firmware image contains -- "
366 "Rule 3 is a claim about firmware"
367 )
368 return failures

382def run_selftest() -> int:
383 """Regression-test the checker itself. Returns a process exit code.
384
385 Four classes of defect are guarded, all of which this gate has shipped:
386
387 * **Namesake merging.** Keying the symbol table by bare name merged
388 distinct same-named functions into one entry, so a module calling its
389 own file-local ``static`` was reported for calling another module's
390 RA8_PRIV symbol.
391 * **A rule that cannot fire.** The linkage rule turns "nothing declares
392 this" into a failure, and a rule which reports nothing looks exactly
393 like a clean tree. The synthetic module contains one definition of
394 every passing shape and one of every failing shape.
395 * **A root silently out of scope.** ``tools/`` was absent from SCAN_DIRS,
396 so ra8_emulator, media_dl and ra8_viewer were never checked and the gate
397 reported a clean tree over code it had not read. The ``tools/`` fixture
398 pins the scope from inside the rules rather than by reading the
399 constant: ``host_unpublished`` is only reachable if is_first_party()
400 accepts the root.
401 * **NASA P10 Rule 3 scope and waiver.** Rule 3 is a claim about firmware,
402 so it is asserted along both axes -- untagged firmware allocation
403 fires, the documented waiver does not, and host-only code under
404 ``tools/`` does not. Getting the second axis wrong is what turns a real
405 gate into 216 findings nobody can act on.
406 """
407 with tempfile.TemporaryDirectory() as td:
408 root = pathlib.Path(td).resolve()
409 with override_repo_root(root):
410 symbols, calls, vector_entries = _selftest_parse(root)
411 violations = enforce_rules(symbols, calls, vector_entries)
412
413 failures = [
414 *_check_priv_namesakes(violations, symbols),
415 *_check_linkage(violations),
416 *_check_rule3(violations),
417 *_check_fixtures_parsed(symbols),
418 # The loop-bound scan is textual and libclang-free, so it self-tests on
419 # synthetic source strings rather than the parsed synthetic tree.
420 *run_loopbound_selftest(),
421 ]
422 if failures:
423 for f in failures:
424 sys.stderr.write(f"[FAIL] check_annotations selftest: {f}\n")
425 sys.stderr.write(f"check_annotations selftest: {len(failures)} failure(s)\n")
426 return 1
427 print(
428 "check_annotations selftest: OK (namesakes resolved by USR; linkage rule "
429 "catches both gap shapes, reaches tools/, and exempts only tabled handlers; "
430 "NASA rule 3 fires on untagged firmware allocation and stays quiet on the "
431 "documented waiver and on host-only code; loop-bound scan fires on a "
432 "mis-attached marker and a stale RA8_BOUNDED_LOOP statement, stays quiet on "
433 "correct markers and on #define/comment/string mentions)"
434 )
435 return 0
