4"""Measure raw bulk throughput to a CDC-ACM device, bypassing cdc_acm.
5
6The right tool for asking "what is the device firmware actually capable of,
7separate from kernel driver overhead?". ``stream_bench.py`` measures the same
8link through /dev/ttyACM*, so comparing the two attributes any shortfall to
9cdc_acm rather than to the firmware.
10
11How it works:
12 1. Detach the cdc_acm kernel driver from the bulk interface.
13 2. Claim the data interface via libusb.
14 3. Write and read bulk transfers back-to-back so the firmware never sees
15 the host idle.
16 4. Measure wall time for the whole round-trip.
17
18Re-attaches cdc_acm on exit via a finally block, so the device comes back as
19/dev/ttyACMx even if the benchmark fails partway -- leaving it detached would
20strand the port until replug.
21
22Requires root, since detaching a kernel driver does.
23
24Usage:
25 sudo python3 libusb_bench.py --vidpid 1209:000c [--bytes 1048576]
26"""

39CDC_DATA_INTERFACE_CLASS = 0x0A # USB CDC data class (bInterfaceClass)
40USB_TRANSFER_TYPE_BULK = 0x02 # bmAttributes transfer-type field value
41USB_TRANSFER_TYPE_MASK = 0x03 # mask to extract transfer type from bmAttributes
42USB_EP_DIR_IN_BIT = 0x80 # bEndpointAddress direction bit: 1 = IN

45def find_bulk_endpoints(dev: usb.core.Device) -> tuple[int, int, int, int] | None:
46 """Locate the first interface carrying a bulk IN/OUT pair.
47
48 Searches only CDC DATA interfaces (class 0x0A). That restriction is the
49 point: CDC ACM splits the function across a comm interface and a data
50 interface, and the bulk pair lives on the data one -- scanning every
51 interface would match the comm interface's interrupt endpoint first.
52
53 An interface offering only one direction is skipped rather than returned
54 half-filled, since the echo bench needs both.
55
56 Returns ``(cfg_value, interface_num, ep_out_addr, ep_in_addr)``, or None
57 when the device exposes no such interface.
58 """
59 cfg = dev.get_active_configuration()
60 for intf in cfg:
61 if intf.bInterfaceClass != CDC_DATA_INTERFACE_CLASS:
62 continue
63 ep_out = ep_in = None
64 for ep in intf:
65 attr = ep.bmAttributes & USB_TRANSFER_TYPE_MASK
66 if attr != USB_TRANSFER_TYPE_BULK:
67 continue # Not bulk
68 if (ep.bEndpointAddress & USB_EP_DIR_IN_BIT) == 0:
69 ep_out = ep.bEndpointAddress
70 else:
71 ep_in = ep.bEndpointAddress
72 if ep_out is not None and ep_in is not None:
73 return cfg.bConfigurationValue, intf.bInterfaceNumber, ep_out, ep_in
74 return None

77def _open_device(vidpid: str) -> tuple[usb.core.Device, int, int, int]:
78 """Find the device and its CDC-data bulk endpoints, or exit(2)."""
79 vid_s, pid_s = vidpid.split(":")
80 dev = usb.core.find(idVendor=int(vid_s, 16), idProduct=int(pid_s, 16))
81 if dev is None:
82 print(f"No device with vid:pid {vidpid}")
83 sys.exit(2)
84 print(
85 f"Found {dev.manufacturer or '?'} / {dev.product or '?'} bus {dev.bus} addr {dev.address}"
86 )
87 info = find_bulk_endpoints(dev)
88 if info is None:
89 print("No CDC-data bulk endpoints found.")
90 sys.exit(2)
91 cfg_val, intf_num, ep_out, ep_in = info
92 print(f"cfg={cfg_val} intf={intf_num} ep_out=0x{ep_out:02x} ep_in=0x{ep_in:02x}")
93 return dev, intf_num, ep_out, ep_in

96def _detach_kernel_driver(dev: usb.core.Device, intf_num: int) -> None:
97 """Unbind cdc_acm from the comm+data interface pair so libusb can claim it."""
98 for i in (intf_num - 1, intf_num):
99 try:
100 if dev.is_kernel_driver_active(i):
101 print(f"Detaching kernel driver from interface {i}")
102 dev.detach_kernel_driver(i)
103 except (NotImplementedError, usb.core.USBError) as e: # noqa: PERF203 # short 2-iter loop; inline try is clearest
104 print(f"Kernel-driver detach on intf {i} ignored: {e}")

107def _pump(
108 dev: usb.core.Device, ep_out: int, ep_in: int, payload: bytes, urb_size: int
109) -> tuple[int, bytearray, float]:
110 """Chunked write+read until both directions have moved the whole payload."""
111 total_bytes = len(payload)
112 sent = 0
113 recv = bytearray()
114 start = time.monotonic()
115 deadline = start + 30.0
116 while sent < total_bytes or len(recv) < total_bytes:
117 if sent < total_bytes:
118 with contextlib.suppress(usb.core.USBTimeoutError):
119 sent += dev.write(ep_out, payload[sent : sent + urb_size], timeout=1000)
120 with contextlib.suppress(usb.core.USBTimeoutError):
121 got = dev.read(ep_in, urb_size, timeout=50)
122 if got:
123 recv += bytes(got)
124 if time.monotonic() > deadline:
125 print("TIMEOUT")
126 break
127 return sent, recv, time.monotonic() - start

130def _report(sent: int, recv: bytes | bytearray, payload: bytes, elapsed: float) -> bool:
131 """Print throughput and integrity; return True when the echo matched."""
132 total_bytes = len(payload)
133 oneway = total_bytes / elapsed if elapsed > 0 else 0.0
134 ok = bytes(recv) == payload[: len(recv)] and len(recv) == total_bytes
135 print(f"sent={sent} recv={len(recv)} elapsed={elapsed:.3f}s")
136 print(f"one-way throughput : {oneway / 1024:9.1f} KB/s ({oneway / 1e6 * 8:.2f} Mbps)")
137 print(f"aggregate : {2 * oneway / 1024:9.1f} KB/s ({2 * oneway / 1e6 * 8:.2f} Mbps)")
138 print(f"integrity : {'OK' if ok else 'FAIL'}")
139 if not ok:
140 for i in range(min(len(recv), len(payload))):
141 if recv[i] != payload[i]:
142 lo, hi = max(0, i - 8), min(len(recv), i + 16)
143 print(f" first mismatch at offset {i}")
144 print(f" expected: {payload[lo:hi].hex()}")
145 print(f" received: {bytes(recv[lo:hi]).hex()}")
146 break
147 return ok

150def bench(
151 vidpid: str,
152 total_bytes: int,
153 urb_size: int,
154 parallel_out: int, # noqa: ARG001 # accepted, unused: reserved for async batching
155 parallel_in: int, # noqa: ARG001 # accepted, unused: reserved for async batching
156) -> NoReturn:
157 """Run one synchronous bulk echo benchmark over raw libusb, then exit.
158
159 Never returns -- exits 0 when the echo matched, 1 when it did not.
160
161 The kernel driver is re-attached in a finally block, so an exception or a
162 failed run still leaves the device usable as /dev/ttyACMx; without that a
163 crash here would strand the port until it was physically replugged.
164
165 ``parallel_out`` / ``parallel_in`` are accepted and deliberately unused:
166 the transfer loop is single-buffered today, and the parameters exist so
167 the async variant can land without a CLI change.
168 """
169 dev, intf_num, ep_out, ep_in = _open_device(vidpid)
170 _detach_kernel_driver(dev, intf_num)
171 try:
172 usb.util.claim_interface(dev, intf_num)
173 payload = bytes((i * 131 + 17) & 0xFF for i in range(total_bytes))
174 sent, recv, elapsed = _pump(dev, ep_out, ep_in, payload, urb_size)
175 sys.exit(0 if _report(sent, recv, payload, elapsed) else 1)
176 finally:
177 with contextlib.suppress(Exception):
178 usb.util.release_interface(dev, intf_num)
179 # Re-attach the kernel driver so the device comes back as ttyACM.
180 for i in (intf_num - 1, intf_num):
181 with contextlib.suppress(Exception):
182 dev.attach_kernel_driver(i)
