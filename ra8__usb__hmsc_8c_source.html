292/* Pack a uint32 into 4 little-endian bytes -- see surrounding code and HUM citations. */
293RA8_INTERNAL
294static void internal_pack_u32_le(uint32_t value, uint8_t* dst)
295{
296 dst[0] = (uint8_t)((value >> k_ra8_hmsc_shift_byte0) & k_ra8_hmsc_byte_mask);
297 dst[1] = (uint8_t)((value >> k_ra8_hmsc_shift_byte1) & k_ra8_hmsc_byte_mask);
298 dst[2] = (uint8_t)((value >> k_ra8_hmsc_shift_byte2) & k_ra8_hmsc_byte_mask);
299 dst[3] = (uint8_t)((value >> k_ra8_hmsc_shift_byte3) & k_ra8_hmsc_byte_mask);
300}

306 return ((uint32_t)src[0] << k_ra8_hmsc_shift_byte0) |

384ra8_err_t ra8_usb_hmsc_build_cbw(uint8_t target_lun,
385 uint32_t data_transfer_length,
386 bool data_in,
387 const uint8_t* cdb,
388 uint8_t cdb_len,
389 uint8_t* out_cbw)
390{
391 RA8_CHECK_NULL_PTR(cdb, s_tag, "build_cbw: cdb");
392 RA8_CHECK_NULL_PTR(out_cbw, s_tag, "build_cbw: out_cbw");
393 if (target_lun > k_ra8_hmsc_lun_field_mask) {
394 return k_ra8_err_invalid_arg;
395 }
396 if ((cdb_len == 0U) || (cdb_len > k_ra8_hmsc_cdb_max_len)) {
397 return k_ra8_err_invalid_arg;
398 }
399
400 /* Zero the entire CBW so reserved nibbles stay 0 -- USB MSC BBB
401 * rev 1.0 sec 5.1 requires zero in the high nibble of bCBWLUN /
402 * the high 3 bits of bCBWCBLength. */
403 internal_zero_bytes(out_cbw, k_ra8_hmsc_cbw_len);
404
405 /* dCBWSignature = 'USBC' little-endian. */
406 internal_pack_u32_le(k_ra8_hmsc_cbw_signature, &out_cbw[k_ra8_hmsc_cbw_off_signature]);
407
408 /* dCBWTag = monotonic. */
409 const uint32_t tag = internal_next_tag();
410 internal_pack_u32_le(tag, &out_cbw[k_ra8_hmsc_cbw_off_tag]);
411
412 /* dCBWDataTransferLength. */
413 internal_pack_u32_le(data_transfer_length, &out_cbw[k_ra8_hmsc_cbw_off_data_length]);
414
415 /* bmCBWFlags. */
416 out_cbw[k_ra8_hmsc_cbw_off_flags] =
417 data_in ? k_ra8_hmsc_cbw_flag_data_in : k_ra8_hmsc_cbw_flag_data_out;
418
419 /* bCBWLUN (low 4 bits) / bCBWCBLength (low 5 bits). */
420 out_cbw[k_ra8_hmsc_cbw_off_lun] = (uint8_t)(target_lun & k_ra8_hmsc_lun_field_mask);
421 out_cbw[k_ra8_hmsc_cbw_off_cdb_length] = (uint8_t)(cdb_len & k_ra8_hmsc_cdb_field_mask);
422
423 /* CBWCB[16]. */
424 internal_copy_bytes(&out_cbw[k_ra8_hmsc_cbw_off_cdb], cdb, (uint16_t)cdb_len);
425
426 return k_ra8_ok;
427}

429ra8_err_t ra8_usb_hmsc_decode_csw(const uint8_t* csw,
430 uint32_t expected_tag,
431 ra8_usb_hmsc_csw_status_t* out_status)
432{
433 RA8_CHECK_NULL_PTR(csw, s_tag, "decode_csw: csw");
434 RA8_CHECK_NULL_PTR(out_status, s_tag, "decode_csw: out_status");
435
436 const uint32_t signature = internal_unpack_u32_le(&csw[k_ra8_hmsc_csw_off_signature]);
437 if (signature != k_ra8_hmsc_csw_signature) {
438 return k_ra8_err_invalid_arg;
439 }
440 const uint32_t tag = internal_unpack_u32_le(&csw[k_ra8_hmsc_csw_off_tag]);
441 if (tag != expected_tag) {
442 return k_ra8_err_invalid_arg;
443 }
444
445 const uint8_t status_byte = csw[k_ra8_hmsc_csw_off_status];
446 if ((status_byte != k_ra8_hmsc_csw_status_passed) &&
447 (status_byte != k_ra8_hmsc_csw_status_failed) &&
448 (status_byte != k_ra8_hmsc_csw_status_phase_error)) {
449 return k_ra8_err_invalid_arg;
450 }
451
452 *out_status = (ra8_usb_hmsc_csw_status_t)status_byte;
453 return k_ra8_ok;
454}

461ra8_err_t ra8_usb_hmsc_init(ra8_usb_speed_t speed)
462{
463 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
464 return k_ra8_err_invalid_arg;
465 }
466 const ra8_err_t usb_err = ra8_usb_host_init(speed);
467 if (usb_err != k_ra8_ok) {
468 /* GCOVR_EXCL_START -- host-unreachable: ra8_usb_host_init fails only on an
469 * MSTP-enable or HS-PLL bring-up fault; under the fake MSTP is plain
470 * RAM (always enables) and the FRDY/bring-up polls short-circuit, so
471 * usb_err is always k_ra8_ok here. */
472 ra8_log_error_val(s_tag, "ra8_usb_host_init failed", (uint32_t)usb_err);
473 return k_ra8_err_hw_init_failed;
474 /* GCOVR_EXCL_STOP */
475 }
476
477 s_usb_hmsc_state.speed = speed;
478 s_usb_hmsc_state.attached = false;
479 s_usb_hmsc_state.attach_cb = nullptr;
480 s_usb_hmsc_state.attach_ctx = nullptr;
481 s_usb_hmsc_state.device = (ra8_usb_hmsc_device_t){};
482 s_usb_hmsc_state.next_cbw_tag = k_ra8_hmsc_initial_tag;
483 s_usb_hmsc_state.initialized = true;
484
485 ra8_log_info_val(s_tag, "host-MSC ready", (uint32_t)speed);
486 return k_ra8_ok;
487}

498 s_usb_hmsc_state.attached = false;
499 s_usb_hmsc_state.attach_cb = nullptr;
500 s_usb_hmsc_state.attach_ctx = nullptr;

502}

723RA8_INTERNAL
724static ra8_err_t internal_run_data_out(uint8_t target_lun,
725 const uint8_t* cdb,
726 uint8_t cdb_len,
727 const uint8_t* in_buf,
728 uint16_t push_len)
729{
730 uint32_t tag = 0U;
731 const ra8_err_t cbw_err =
732 internal_issue_cbw(target_lun, (uint32_t)push_len, false, cdb, cdb_len, &tag);
733 RA8_RETURN_ON_ERROR(cbw_err, s_tag, "run_data_out: issue cbw"); /* GCOVR_EXCL_BR_LINE */

735 * DEVICE's enumerated bulk-OUT wMaxPacketSize -- NOT the host
736 * controller's native ceiling. On a HS host driving an FS device (the
737 * self-loop) the controller speed is HS (512) but the endpoint is FS
738 * (64); chunking at 512 advances the offset 8x faster than the 64-byte
739 * packet actually sent, so the device receives only 1/8 of the data
740 * and the WRITE wedges. The pipe's PIPEMAXP is already this value
741 * (see the enum pipe setup). Fall back to the speed ceiling only if
742 * enumeration left it unset. */
743 /* GCOVR_EXCL_START -- host-unreachable: the per-packet data-OUT chunk loop
744 * and the trailing CSW read run only after the CBW push
745 * (internal_issue_cbw above) succeeds; under the fake that push
746 * always times out, so this run always returns on the cbw_err leg above. */
747 uint16_t mps = s_usb_hmsc_state.device.bulk_out_max_packet;
748 if (mps == 0U) {
749 mps = internal_bulk_max_packet(s_usb_hmsc_state.speed);
750 }
751 uint16_t offset = 0U;
752 while (offset < push_len) {
753 uint16_t chunk = (uint16_t)(push_len - offset);
754 if (chunk > mps) {
755 chunk = mps;
756 }
757 const ra8_err_t werr = ra8_usb_host_bulk_out(s_usb_hmsc_state.speed,
758 k_ra8_hmsc_pipe_bulk_out,
759 &in_buf[offset],
760 chunk);
761 RA8_RETURN_ON_ERROR(werr, s_tag, "run_data_out: data chunk"); /* GCOVR_EXCL_BR_LINE */
762 offset = (uint16_t)(offset + chunk);
763 }
764 return internal_read_csw(tag);
765}
766/* GCOVR_EXCL_STOP */
767
768/* GCOVR_EXCL_START -- host-unreachable: called only from ra8_usb_hmsc_inquiry
769 * to decode the 36-byte INQUIRY response, which arrives only after a
770 * completed BOT command; the fake bulk-OUT CBW push always times out,
771 * so the INQUIRY response is never received and never decoded on the host. */
792RA8_INTERNAL
793static void internal_decode_inquiry(const uint8_t* raw, ra8_usb_hmsc_inquiry_response_t* response)
794{
795 internal_zero_bytes((uint8_t*)response, (uint16_t)sizeof(*response));
796 const uint8_t b0 = raw[k_ra8_hmsc_inq_off_dev_type];
797 const uint8_t b1 = raw[k_ra8_hmsc_inq_off_removable];
798 response->peripheral_qualifier = (uint8_t)(b0 >> k_ra8_hmsc_inq_qual_shift);
799 response->peripheral_device_type = (uint8_t)(b0 & k_ra8_hmsc_inq_dev_type_mask);
800 response->removable =
801 (uint8_t)((b1 >> k_ra8_hmsc_inq_removable_shift) & k_ra8_hmsc_inq_removable_mask);
802 response->version = raw[k_ra8_hmsc_inq_off_version];
803 internal_copy_bytes(response->vendor_id,
804 &raw[k_ra8_hmsc_inq_off_vendor_id],
805 (uint16_t)sizeof(response->vendor_id));
806 internal_copy_bytes(response->product_id,
807 &raw[k_ra8_hmsc_inq_off_product_id],
808 (uint16_t)sizeof(response->product_id));
809 internal_copy_bytes(response->product_revision,
810 &raw[k_ra8_hmsc_inq_off_product_rev],
811 (uint16_t)sizeof(response->product_revision));
812}
813/* GCOVR_EXCL_STOP */
