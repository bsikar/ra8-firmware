92RA8_INTERNAL static ra8_err_t internal_audio_bits_to_word(uint8_t bit_depth,
93 ra8_ssie_data_word_t* out_dwl,
94 ra8_ssie_system_word_t* out_swl)
95{
96 switch ((ra8_audio_bit_depth_t)bit_depth) {
97 case k_ra8_audio_bits_8:
98 *out_dwl = k_ra8_ssie_dwl_8;
99 *out_swl = k_ra8_ssie_swl_8;
100 return k_ra8_ok;
101 case k_ra8_audio_bits_16:
102 *out_dwl = k_ra8_ssie_dwl_16;
103 *out_swl = k_ra8_ssie_swl_16;
104 return k_ra8_ok;
105 case k_ra8_audio_bits_18:
106 *out_dwl = k_ra8_ssie_dwl_18;
107 *out_swl = k_ra8_ssie_swl_24;
108 return k_ra8_ok;
109 case k_ra8_audio_bits_20:
110 *out_dwl = k_ra8_ssie_dwl_20;
111 *out_swl = k_ra8_ssie_swl_24;
112 return k_ra8_ok;
113 case k_ra8_audio_bits_22:
114 *out_dwl = k_ra8_ssie_dwl_22;
115 *out_swl = k_ra8_ssie_swl_24;
116 return k_ra8_ok;
117 case k_ra8_audio_bits_24:
118 *out_dwl = k_ra8_ssie_dwl_24;
119 *out_swl = k_ra8_ssie_swl_24;
120 return k_ra8_ok;
121 case k_ra8_audio_bits_32:
122 *out_dwl = k_ra8_ssie_dwl_32;
123 *out_swl = k_ra8_ssie_swl_32;

132{

143 {(ra8_port_pin_t)k_ra8_board_audio_pin_wclk, k_ra8_psel_ssie, "ra8_board.audio.wclk"},

204ra8_err_t ra8_board_audio_init(uint32_t sample_rate_hz, uint8_t bit_depth, uint8_t channels)
205{
206 if (sample_rate_hz == 0U) {
207 return k_ra8_err_invalid_arg;
208 }
209 // mcdc-deactivated: ra8_board_audio_init channel-validation guard; both conditions are exercised independently by tests/test_ra8_board_audio_validation, but llvm-cov gates require an N+1 vector that holds one condition true while the other is false -- only the all-valid (mono/stereo) inputs are reachable in production wiring.
210 if (channels != (uint8_t)k_ra8_audio_channels_mono &&
211 channels != (uint8_t)k_ra8_audio_channels_stereo) {
212 return k_ra8_err_invalid_arg;
213 }
214 ra8_ssie_data_word_t dwl = k_ra8_ssie_dwl_16;
215 ra8_ssie_system_word_t swl = k_ra8_ssie_swl_16;
216 ra8_err_t berr = internal_audio_bits_to_word(bit_depth, &dwl, &swl);
217 if (berr != k_ra8_ok) {
218 return berr;
219 }
220
221 /* Step 1: route the four DAI pins to SSIE0 + the I2C control pair
222 * to IIC1 (UM Table 32 p 38). */
223 ra8_err_t rerr = internal_audio_route_pins();
224 if (rerr != k_ra8_ok) {
225 return rerr;
226 }
227
228 /* Step 2: bring up SSIE0 in I2S controller mode. */
229 (void)sample_rate_hz;
230 const ra8_ssie_cfg_t cfg = internal_audio_build_ssie_cfg(channels, dwl, swl);
231 const ra8_err_t serr = ra8_ssie_init((uint8_t)k_ra8_board_audio_ssie_channel, &cfg);
232 if (serr != k_ra8_ok) {
233 return k_ra8_err_hw_init_failed;
234 }
235 s_audio_initialized = true;
236 return k_ra8_ok;
237}

239ra8_err_t ra8_board_audio_play_sample_block(const int16_t* buf, uint32_t len)
240{
241 if (buf == nullptr) {
242 return k_ra8_err_invalid_arg;
243 }
244 if (len == 0U || (len % (uint32_t)k_ra8_audio_samples_per_word) != 0U) {
245 return k_ra8_err_invalid_arg;
246 }
247 if (!s_audio_initialized) {
248 return k_ra8_err_not_initialized;
249 }
250 /* Two int16_t samples (one stereo frame) pack into one 32-bit SSIE
251 * FIFO word; len is guaranteed even by the check above. The buffer
252 * is a contiguous PCM stream the SSIE FIFO consumes 32 bits at a
253 * time. Reinterpret via uintptr_t so we neither (a) double-hop
254 * through void* (bugprone-casting-through-void) nor (b) trip
255 * cast-align by going int16_t* -> uint32_t* directly. The caller
256 * contract requires ``buf`` to be 32-bit aligned (one stereo frame
257 * naturally aligns); see ra8_board_audio_play_sample_block docs. */
258 const uint32_t words = len / (uint32_t)k_ra8_audio_samples_per_word;
259 const uintptr_t addr = (uintptr_t)buf;
260 const uint32_t* const packed = (const uint32_t*)
261 addr; /* NOLINT(performance-no-int-to-ptr) -- alignment-safe reinterpret documented above. */
262 uint16_t written = 0U;
263 const ra8_err_t err = ra8_ssie_write_buffer((uint8_t)k_ra8_board_audio_ssie_channel,
264 packed,
265 (uint16_t)words,
266 &written);
267 if (err != k_ra8_ok) {
268 return err;
269 }
270 if (written != (uint16_t)words) {
271 return k_ra8_err_hw_timeout;
272 }
273 return k_ra8_ok;
274}

547RA8_INTERNAL static ra8_err_t internal_io_expander_bus_recover(void)
548{
549 const ra8_port_pin_t scl = k_ra8_board_io_expander_pin_scl;
550 const ra8_port_pin_t sda = k_ra8_board_io_expander_pin_sda;
551 ra8_err_t err = ra8_gpio_output_init(scl, k_ra8_level_high);
552 if (err != k_ra8_ok) {
553 return err;
554 }
555 err = ra8_gpio_input_init(sda, k_ra8_pull_up);
556 if (err != k_ra8_ok) {
557 (void)ra8_gpio_release(scl);
558 return err;
559 }
560 for (uint32_t i = 0U; i < (uint32_t)k_ra8_board_i2c_recover_pulses; i++) {
561 ra8_level_t sda_lvl = k_ra8_level_low;
562 const ra8_err_t rd = ra8_gpio_read(sda, &sda_lvl);
563 if (rd == k_ra8_ok) {
564 if (sda_lvl == k_ra8_level_high) {
565 break; /* peripheral released SDA -- bus is free */
566 }
567 }
568 (void)ra8_gpio_write(scl, k_ra8_level_low);
569 internal_io_expander_bus_settle();
570 (void)ra8_gpio_write(scl, k_ra8_level_high);
571 internal_io_expander_bus_settle();
572 }
573 /* Frame a STOP (SDA low->high while SCL high) to end any partial frame,
574 * then release both pins for the IIC mux to re-claim. */
575 (void)ra8_gpio_release(sda);
576 (void)ra8_gpio_output_init(sda, k_ra8_level_low);
577 internal_io_expander_bus_settle();
578 (void)ra8_gpio_write(scl, k_ra8_level_high);
579 (void)ra8_gpio_write(sda, k_ra8_level_high);
580 internal_io_expander_bus_settle();
581 (void)ra8_gpio_release(sda);
582 (void)ra8_gpio_release(scl);
583 return k_ra8_ok;
584}

714RA8_INTERNAL static ra8_err_t internal_io_expander_apply(uint8_t output_byte)
715{
716 /* Step -1: clock out any wedged peripheral (stuck SDA) before touching
717 * the bus, so a prior aborted transfer cannot deadlock every START. */
718 ra8_err_t err = internal_io_expander_bus_recover();
719 if (err != k_ra8_ok) {
720 return err;
721 }
722
723 /* Step 0: drive P109/P311 high so the SCL1/SDA1 bus has pull-ups
724 * (EK-RA8D2 v1 UM Table 23). Without this the open-drain bus floats
725 * and U15 cannot ACK. */
726 err = internal_io_expander_enable_pullups();
727 if (err != k_ra8_ok) {
728 return err;
729 }
730
731 /* Step 1: route P512 -> SCL1 and P511 -> SDA1 with NCODR on both
732 * (HUM Ch 20.6 PSEL table p 859). */
733 s_io_expander_probe = (uint32_t)k_io_exp_probe_pre_pfs;
734 err = internal_io_expander_route_pins();
735 if (err != k_ra8_ok) {
736 return err;
737 }
738
739 /* Step 2: bring RIIC channel 1 up at 100 kHz. ra8_i2c_init ungates
740 * the per-channel MSTP gate (IIC1 = MSTPB8) itself, so no separate
741 * MSTP enable is needed here. */
742 s_io_expander_probe = (uint32_t)k_io_exp_probe_pre_init;
743 const ra8_i2c_cfg_t cfg = {
744 .bus_hz = (uint32_t)k_ra8_board_io_expander_bus_hz,
745 .pclkb_hz = (uint32_t)k_ra8_board_io_expander_pclkb_hz,
746 };
747 err = ra8_i2c_init((uint8_t)k_ra8_board_io_expander_iic_channel, &cfg);
748 if (err != k_ra8_ok) {
749 return err;
750 }
751
752 /* Step 4: program U15 in FSP-reference order. */
753 err = internal_io_expander_program_u15(output_byte);
754 if (err != k_ra8_ok) {
755 return err;
756 }
757 s_io_expander_probe = (uint32_t)k_io_exp_probe_success;
758 return k_ra8_ok;
759}

883 * pins, so any other packed value lands outside the enumerator set
884 * and trips clang-analyzer EnumCastOutOfRange; suppress in the same
885 * pattern used elsewhere in this file for board-specific pins. */

887 const ra8_port_pin_t pd07 = (ra8_port_pin_t)RA8_PIN(k_ra8_port_13, k_ra8_pin_7);
888 /* NOLINTEND(clang-analyzer-optin.core.EnumCastOutOfRange) */
889 const ra8_err_t err = ra8_gpio_output_init(pd07, k_ra8_level_low);
890 s_usbhs_role_pin_err = (uint32_t)err;
891 s_usbhs_role_pin_probe = (uint32_t)k_usbhs_role_probe_post_init;

893 s_usbhs_role_pin_probe = (uint32_t)k_usbhs_role_probe_success;
894 }

898ra8_err_t ra8_board_usbhs_device_init(void)
899{
900 /* HUM Ch 9 (CGC) USBCKCR p 365 + HUM Ch 11 (MSTP) MSTPCRB12 p 469
901 * stage the PHY clock and ungate the controller block; then the
902 * generic device-mode entry in libs/ra8_hal/inc/ra8_usb.h drives
903 * SYSCFG.SCKE / USBE / HSE and arms the device IRQ set.
904 *
905 * EK-RA8D2 v1 UM Rev 1.01 Section 6.2 p 34 ("USB High Speed"): J7's
906 * role is selected by the MCU GPIO PD07 -- LOW = Device, HIGH = Host.
907 * Drive PD07 low directly (no U15 I/O-expander dependency required;
908 * U15 is only an SW4 override path, see Section 4.3.4 p 16 / Table 23
909 * pullup-config p 31). The U15 helper is invoked best-effort below
910 * for boards where PD07 alone is insufficient (e.g. SW4-8 mechanical
911 * position contradicts the firmware intent), but its failure is
912 * non-fatal because PD07 already strapped the role line. */
913 const ra8_err_t pd07_err = internal_usbhs_role_select_device();
914 if (pd07_err != k_ra8_ok) {
915 return pd07_err;
916 }
917
918 /* Best-effort U15 override (logged via s_io_expander_probe). The U15
919 * I2C path may NACK if SW4-5 is OFF (I2C/I3C-Select strap routes the
920 * shared bus elsewhere) -- non-fatal because PD07 already strapped
921 * the role. */
922 const ra8_err_t io_err = ra8_board_io_expander_set_usbhs_device_mode();
923 (void)io_err;
924 ra8_err_t err = internal_usbhs_clock_and_mstp();
925 if (err != k_ra8_ok) {
926 return err;
927 }
928 s_usbhs_probe = (uint32_t)k_usbhs_probe_pre_usb_dev_init;
929 ra8_err_t rc_dev = ra8_usb_device_init(k_ra8_usb_speed_hs);
930 s_usbhs_probe = (uint32_t)k_usbhs_probe_post_usb_dev_init;
931 return rc_dev;
932}

934ra8_err_t ra8_board_usbhs_host_init(void)
935{
936 /* Symmetric to ra8_board_usbhs_device_init. EK-RA8D2 v1 UM Rev 1.01
937 * Table 28 lists USBHS_VBUSEN / USBHS_OVRCUR as PHY-controller
938 * pins not routed to RA8D2 port pins, so VBUS sourcing is left to
939 * the on-board USB-PD controller. ra8_usb_host_init drives
940 * SYSCFG.DCFM=1 / DRPD=1 / USBE=1 plus the DCP setup. */
941 ra8_err_t err = internal_usbhs_clock_and_mstp();
942 if (err != k_ra8_ok) {
943 return err;
944 }
945 return ra8_usb_host_init(k_ra8_usb_speed_hs);
946}
