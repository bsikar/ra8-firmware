106static void iv_print_hex(uint32_t value)
107{
108 uint8_t buf[k_iv_hex_nibbles];
109 for (uint32_t i = 0U; i < (uint32_t)k_iv_hex_nibbles; i++) {
110 const uint32_t shift = ((uint32_t)k_iv_hex_nibbles - 1U - i) * (uint32_t)k_iv_nibble_bits;
111 const uint32_t nib = (value >> shift) & (uint32_t)k_iv_nibble_mask;
112 buf[i] = (uint8_t)((nib < (uint32_t)k_iv_dec_ten) ? ('0' + nib) : ('A' + (nib - k_iv_dec_ten)));
113 }
114 iv_print(buf, (uint32_t)k_iv_hex_nibbles);
115}

118static void iv_print_uint(uint32_t value)
119{
120 uint8_t buf[k_iv_dec_ten];
121 uint32_t n = 0U;
122 if (value == 0U) {
123 buf[n] = '0';
124 n++;
125 }
126 while ((value > 0U) && (n < (uint32_t)k_iv_dec_ten)) {
127 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_iv_dec_ten));
128 n++;
129 value /= (uint32_t)k_iv_dec_ten;
130 }
131 for (uint32_t i = 0U; i < n; i++) {
132 iv_print(&buf[n - 1U - i], 1U);
133 }
134}

142static uint32_t iv_framebuffer_hash(void)
143{
144 const uint8_t* p = (const uint8_t*)s_framebuffer;
145 const size_t n = sizeof(s_framebuffer);
146 uint32_t hsh = (uint32_t)k_iv_fnv_offset;
147 for (size_t i = 0U; i < n; i++) {
148 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_iv_fnv_prime;
149 }
150 return hsh;
151}

169 iv_panic_halt();
