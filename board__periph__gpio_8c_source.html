101static void port_trace_leds(uint32_t port_idx, uint16_t before, uint16_t after)
102{
103 for (uint32_t i = 0U; i < (uint32_t)k_board_led_count; i++) {
104 if (k_led_map[i].port != (uint8_t)port_idx) {
105 continue;
106 }
107 const uint16_t mask = (uint16_t)(1U << k_led_map[i].pin);
108 const uint32_t was = ((before & mask) != 0U) ? 1U : 0U;
109 const uint32_t now = ((after & mask) != 0U) ? 1U : 0U;
110 if (was != now) {
111 s_led_level[i] = now;
112 s_led_transitions[i]++;
113 /* Console GPIO tab: one line per pin-level edge (bounded -- only the mapped
114 * board LEDs, and only on a 0<->1 change, so a blink loop cannot flood). */
115 char ln[k_gpio_console_line_cap];
116 (void)snprintf(ln, sizeof(ln), "%s -> %s", k_led_map[i].name, now ? "ON" : "OFF");
117 board_console_push(k_board_console_ch_gpio, ln);
118 if (board_periph_trace()) {
119 (void)fprintf(stderr, " [trace] %s -> %s\n", k_led_map[i].name, now ? "ON" : "OFF");
120 }
121 }
122 }
123}

136static uint64_t port_read(uc_engine* uc, uint64_t addr, unsigned size)
137{
138 (void)uc;
139 (void)size;
140 const uint32_t idx = (uint32_t)((addr - (uint64_t)k_port_base) / (uint64_t)k_port_stride);
141 const uint64_t off = (addr - (uint64_t)k_port_base) % (uint64_t)k_port_stride;
142 if (idx >= (uint32_t)k_port_count) {
143 return 0U;
144 }
145 const port_state_t* p = &s_port[idx];
146 if (off == (uint64_t)k_port_pcntr1) {
147 return ((uint32_t)p->podr << (uint32_t)k_half_shift) | (uint32_t)p->pdr;
148 }
149 if (off == (uint64_t)k_port_pcntr2) {
150 /* PIDR reads the live pin level: an output pin reads back its driven latch
151 * (so a "read what I drove" check is honest); an input pin reads any
152 * externally-injected level (user buttons -- see board_periph_gpio_set_input),
153 * else 0. */
154 const uint16_t driven = (uint16_t)(p->podr & p->pdr);
155 const uint16_t inputs = (uint16_t)((uint16_t)(~p->pdr) & p->in_ovr & p->in_lvl);
156 return (uint32_t)(uint16_t)(driven | inputs);
157 }
158 return 0U; /* PCNTR3 is write-only; PCNTR4 unmodelled -> 0. */
159}

162static void port_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
163{
164 (void)uc;
165 (void)size;
166 const uint32_t idx = (uint32_t)((addr - (uint64_t)k_port_base) / (uint64_t)k_port_stride);
167 const uint64_t off = (addr - (uint64_t)k_port_base) % (uint64_t)k_port_stride;
168 if (idx >= (uint32_t)k_port_count) {
169 return;
170 }
171 if (off == (uint64_t)k_port_pcntr1) {
172 s_port[idx].pdr = (uint16_t)(value & (uint32_t)k_half_mask);
173 port_set_podr(idx,
174 (uint16_t)(((uint32_t)value >> (uint32_t)k_half_shift) & (uint32_t)k_half_mask));
175 } else if (off == (uint64_t)k_port_pcntr3) {
176 const uint16_t posr = (uint16_t)((uint32_t)value & (uint32_t)k_half_mask);
177 const uint16_t porr =
178 (uint16_t)(((uint32_t)value >> (uint32_t)k_half_shift) & (uint32_t)k_half_mask);
179 uint16_t podr = s_port[idx].podr;
180 podr |= posr; /* atomic set */
181 podr &= (uint16_t)~porr; /* atomic clear */
182 port_set_podr(idx, podr);
183 }
184}

187static void port_reset(void)
188{
189 for (uint32_t i = 0U; i < (uint32_t)k_port_count; i++) {
190 s_port[i] = (port_state_t){};
191 }
192 /* EK-RA8D2 user switches are active-low with pull-ups: idle (released) reads
193 * high. Seed P009/P008 (SW1/SW2) high so a firmware poll sees "not pressed"
194 * until board_periph_gpio_set_input() drives them low (a --button press). */
195 const uint16_t sw_mask = (uint16_t)((1U << (uint32_t)k_sw1_pin) | (1U << (uint32_t)k_sw2_pin));
196 s_port[k_sw_port].in_ovr |= sw_mask;
197 s_port[k_sw_port].in_lvl |= sw_mask;
198 for (uint32_t i = 0U; i < (uint32_t)k_board_led_count; i++) {
199 s_led_level[i] = 0U;
200 s_led_transitions[i] = 0U;
201 }
202 /* An attached --eink IT8951 controller re-arms its HRDY "ready" GPIO input
203 * high here, AFTER the ports are cleared, so the firmware's ra8_epaper HRDY
204 * poll reads ready (a no-op when no controller is attached). Mirrors the
205 * user-switch seeding above. */
206 board_eink_apply_gpio_defaults();
207}

209void board_periph_gpio_set_input(uint8_t port, uint8_t pin, bool level)
210{
211 if ((uint32_t)port >= (uint32_t)k_port_count || (uint32_t)pin >= (uint32_t)k_pins_per_port) {
212 return;
213 }
214 const uint16_t bit = (uint16_t)(1U << (uint32_t)pin);
215 s_port[port].in_ovr |= bit;
216 if (level) {
217 s_port[port].in_lvl |= bit;
218 } else {
219 s_port[port].in_lvl &= (uint16_t)~bit;
220 }
221}

223bool board_periph_gpio_get_input(uint8_t port, uint8_t pin)
224{
225 if ((uint32_t)port >= (uint32_t)k_port_count || (uint32_t)pin >= (uint32_t)k_pins_per_port) {
226 return false;
227 }
228 const uint16_t bit = (uint16_t)(1U << (uint32_t)pin);
229 return (s_port[port].in_lvl & bit) != 0U;
230}
