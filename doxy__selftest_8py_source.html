56#include "mod_doc.h"
57
58/**
59 * @brief Doubles a value.
60 * @details Multiplies the input by two and returns it, saturating nothing.
61 * @param[in] a Value to double.
62 * @return The doubled value.
63 * @retval 0 The input was zero.
64 * @pre @p a is initialised.
65 * @pre The caller is single-threaded.
66 * @post No global state is mutated.
67 * @post The result is exactly 2*a.

69 * @since 0.1.0

71static int good_full_block(int a)
72{
73 return a * 2;
74}

76/* A static forward prototype: an ordering device, not a second contract.
77 Demanding a block here too would force the identical block to be written
78 twice, which check_doc_attachment.py rejects as a duplicate. */
79static int good_fwd_declared(int a);
80
81/**
82 * @brief Triples a value.
83 * @details Multiplies the input by three and returns it.
84 * @param[in] a Value to triple.
85 * @return The tripled value.
86 * @retval 0 The input was zero.
87 * @pre @p a is initialised.
88 * @pre The caller is single-threaded.
89 * @post No global state is mutated.
90 * @post The result is exactly 3*a.
91 * @note Pure; thread-safe.
92 * @since 0.1.0
93 */
94static int good_fwd_declared(int a)
95{
96 return a * 3;
97}
98
99/* Definition-site policy: a non-static definition in a .c carries no block --
100 the header owns the contract. */
101int hdr_undocumented(int a)
102{
103 int total = 0;
104 /* `else if (...)` is the shape NON_FUNC_NAMES exists for: the regex sees
105 `else` as a return-type token and `if` as the function name. Same for
106 `__asm__ volatile("...")`, which parses as a function named `volatile`.
107 A bare `if (...)` would NOT exercise this -- it has no type token in
108 front of it, so FUNC_RE never matches it in the first place. */
109 if (a > 0)
110 {
111 total = good_full_block(a);
112 }
113 else if (a < 0)
114 {
115 total = 0 - a;
116 }
117 while (total > 100) {
118 total = total - 1;
119 }
120 __asm__ volatile("nop");
121 return total + good_fwd_declared(a);
122}
123""",

173def _audit_synthetic(root: Path) -> tuple[list, list]:
174 """Write the fixture tree under ``root`` and audit it in both modes."""
175 for rel, body in _SELFTEST_SOURCES.items():
176 path = root / rel
177 path.parent.mkdir(parents=True, exist_ok=True)
178 path.write_text(body, encoding="ascii")
179 func_rows, member_rows = [], []
180 for rel in _SELFTEST_SOURCES:
181 path = root / rel
182 func_rows.extend(audit_file(path))
183 member_rows.extend(audit_members_file(path))
184 return func_rows, member_rows

187def _check_function_mode(func_rows: list) -> list[str]:
188 """The function gate must report every gap and spare every legal form.
189
190 The legal-but-tricky forms are the load-bearing half. A `.c` definition of
191 a function the header already documents must stay clean (CLAUDE.md,
192 "Definition-site comments"), a static forward prototype whose definition
193 carries the block must stay clean (otherwise this gate and
194 check_doc_attachment.py contradict each other and no file can satisfy
195 both), and `if` / `while` / `switch` must never be mistaken for
196 declarations.
197 """
198 gaps = {(r[0], r[2]) for r in func_rows if r[3]}
199 gap_names = {name for _f, name in gaps}
200 failures = [
201 f"function gate went toothless: '{name}' is undocumented and was not reported"
202 for name in sorted(_SELFTEST_FUNC_EXPECTED - gap_names)
203 ]
204 failures.extend(
205 f"function gate false positive: '{name}' in {path} is a legal form "
206 f"(fully documented, forward prototype, or a definition whose header "
207 f"owns the contract) but was reported"
208 for path, name in sorted(gaps)
209 if name in _SELFTEST_FUNC_CLEAN and path.endswith("good.c")
210 )
211 parsed = {r[2] for r in func_rows}
212 failures.extend(
213 f"selftest fixture did not parse: '{name}' never reached the auditor, "
214 f"so its shape was never exercised"
215 for name in sorted(_SELFTEST_FUNC_EXPECTED | _SELFTEST_FUNC_CLEAN)
216 if name not in parsed
217 )
218 # Keywords are not declarations. `else if (...)` and `__asm__ volatile(...)`
219 # both match FUNC_RE structurally -- the keyword in front reads as a return
220 # type -- and are rejected only by NON_FUNC_NAMES. If that filter erodes,
221 # the gate buries real findings under noise until somebody switches it off.
222 failures.extend(
223 f"function gate false positive: keyword '{bogus}' was parsed as a "
224 f"function declaration (NON_FUNC_NAMES no longer filters it)"
225 for bogus in ("if", "volatile")
226 if bogus in parsed
227 )
228 return failures

231def _check_member_mode(member_rows: list) -> list[str]:
232 """The member gate must report undocumented members and spare documented ones."""
233 member_hits = {(r[2], r[3]) for r in member_rows}
234 member_names = {name for _k, name in member_hits}
235 failures = [
236 f"member gate went toothless: undocumented {kind} '{name}' was not reported"
237 for kind, name in sorted(_SELFTEST_MEMBER_EXPECTED)
238 if name not in member_names
239 ]
240 failures.extend(
241 f"member gate false positive: '{name}' carries a doc comment but was reported"
242 for name in sorted(_SELFTEST_MEMBER_CLEAN & member_names)
243 )
244 return failures

322def _check_style_mode() -> list[str]:
323 """The style gate must fire on each defect and spare each legal spelling."""
324 failures = []
325 for rel, source, expected in _STYLE_FIXTURES:
326 rows, _seen = audit_text(rel, source)
327 codes = {row[2] for row in rows}
328 failures.extend(
329 f"style gate went toothless: {rel} should report {code} and did not"
330 for code in sorted(expected - codes)
331 )
332 failures.extend(
333 f"style gate false positive: {rel} is a legal form but reported {code}"
334 for code in sorted(codes - expected)
335 )
336 return failures

339def _check_style_ratchet() -> list[str]:
340 """The @details ratchet must excuse frozen debt, and only frozen debt."""
341 rows = [
342 ("libs/frozen.c", 1, "DETAILS_MISSING", "no @details"),
343 ("libs/fresh.c", 1, "DETAILS_MISSING", "no @details"),
344 ("libs/fresh.c", 9, "PARAM_NO_DIRECTION", "plain @param"),
345 ]
346 violations, stale = partition(rows, {"libs/frozen.c", "libs/written.c"})
347 offenders = {(row[0], row[2]) for row in violations}
348 failures = []
349 if ("libs/frozen.c", "DETAILS_MISSING") in offenders:
350 failures.append("style ratchet: frozen @details debt was reported as a new violation")
351 if ("libs/fresh.c", "DETAILS_MISSING") not in offenders:
352 failures.append("style ratchet: a file outside the baseline escaped the @details rule")
353 if ("libs/fresh.c", "PARAM_NO_DIRECTION") not in offenders:
354 failures.append("style ratchet: the baseline excused a rule it does not cover")
355 if stale != ["libs/written.c"]:
356 failures.append(f"style ratchet: stale baseline entry not reported (got {stale})")
357 return failures

360def _check_style_floor() -> list[str]:
361 """The vacuity floors must fire on a collapsed scan and pass a real one."""
362 failures = []
363 if _floor_failure([], 0) is None:
364 failures.append("style gate: an EMPTY file list did not trip the vacuity floor")
365 if _floor_failure(["x.c"] * 9999, 0) is None:
366 failures.append("style gate: ZERO @param tags did not trip the vacuity floor")
367 if _floor_failure(["x.c"] * 9999, 99999) is not None:
368 failures.append("style gate: a plausible scan was rejected by the vacuity floor")
369 return failures

372def run_selftest() -> int:
373 """Regression-test the auditor itself. Returns a process exit code.
374
375 ``--check``, ``--members --check`` and ``--style`` are all enforcing gates,
376 and a parser-driven gate has two ways to fail silently. It can stop
377 recognising a construct, in which case the offenders inside it vanish and
378 the tree looks documented; or it can start matching things that are not
379 declarations at all, in which case it reports noise until somebody
380 switches it off. Both directions are asserted, for all three modes.
381 """
382 with tempfile.TemporaryDirectory() as td:
383 root = Path(td).resolve()
384 with override_repo_root(root):
385 func_rows, member_rows = _audit_synthetic(root)
386
387 failures = [
388 *_check_function_mode(func_rows),
389 *_check_member_mode(member_rows),
390 *_check_style_mode(),
391 *_check_style_ratchet(),
392 *_check_style_floor(),
393 ]
394 if failures:
395 for f in failures:
396 sys.stderr.write(f"[FAIL] doxy_audit selftest: {f}\n")
397 sys.stderr.write(f"doxy_audit selftest: {len(failures)} failure(s)\n")
398 return 1
399 print(
400 "doxy_audit selftest: OK (function gate reports bare and thin blocks and "
401 "spares definition-site, forward-prototype and control-flow forms; member "
402 "gate reports undocumented macros, enum values and struct members and "
403 "spares documented ones; style gate reports a missing/stale file header, "
404 "a missing @brief/@details and a directionless @param, spares the "
405 "backslash, continued-name and full-path spellings, excuses only frozen "
406 "@details debt, and refuses a vacuous scan)"
407 )
408 return 0
