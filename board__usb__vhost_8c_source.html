69void usb_detect_class(const uint8_t* d, uint16_t len)
70{
71 if (s_dev_class != (uint8_t)k_usb_class_unknown) {
72 return;
73 }
74 uint16_t i = 0U;
75 while (((uint32_t)i + 2U) <= (uint32_t)len) {
76 const uint8_t blen = d[i];
77 const uint8_t btype = d[i + 1U];
78 if (blen == 0U) {
79 break;
80 }
81 if ((btype == (uint8_t)k_usb_dt_interface) && (((uint32_t)i + 6U) <= (uint32_t)len)) {
82 const uint8_t icls = d[i + k_iface_class_off];
83 if (icls == (uint8_t)k_usb_iclass_hid) {
84 s_dev_class = (uint8_t)k_usb_class_hid;
85 return;
86 }
87 if (icls == (uint8_t)k_usb_iclass_msc) {
88 s_dev_class = (uint8_t)k_usb_class_msc;
89 return;
90 }
91 if ((icls == (uint8_t)k_usb_iclass_cdc_comm) || (icls == (uint8_t)k_usb_iclass_cdc_data)) {
92 s_dev_class = (uint8_t)k_usb_class_cdc;
93 return;
94 }
95 if (icls == (uint8_t)k_usb_iclass_printer) {
96 s_dev_class = (uint8_t)k_usb_class_printer;
97 return;
98 }
99 if (icls == (uint8_t)k_usb_iclass_vendor) {
100 s_dev_class = (uint8_t)k_usb_class_vendor;
101 return;
102 }
103 }
104 i = (uint16_t)((uint32_t)i + (uint32_t)blen);
105 }
106}

245void host_deliver_setup(uc_engine* uc, const usb_setup_step_t* s)
246{
247 /* Clear any stale DCP state from the previous transfer. */
248 s_usb.dcp_in.len = 0U;
249 s_usb.dcp_in.valid = false;
250 s_usb.dcp_out.len = 0U;
251 s_usb.dcp_out.rd = 0U;
252 s_usb.dcp_out.ready = false;
253 /* For an OUT-data control transfer, stage placeholder bytes the device can
254 * drain through the DCP if its class handler reads them. Skipped under the
255 * external-host bridge: there the REAL host payload arrives via
256 * board_usb_bridge_dcp_out(), and a placeholder would let the device drain
257 * zeros before it lands. */
258 if (!s_external_host && ((s->bm_request_type & (uint8_t)k_usb_dir_device_to_host) == 0U) &&
259 (s->w_length != 0U)) {
260 const uint16_t n =
261 (s->w_length < (uint16_t)k_usb_dcp_mps) ? s->w_length : (uint16_t)k_usb_dcp_mps;
262 s_usb.dcp_out.len = n;
263 s_usb.dcp_out.rd = 0U;
264 s_usb.dcp_out.ready = true;
265 (void)memset(s_usb.dcp_out.data, 0, sizeof(s_usb.dcp_out.data));
266 }
267 host_latch_setup(s);
268 s_usb.setup_valid = true;
269 s_usb.ctsq = host_setup_ctsq(s);
270 usb_intsts0_set((uint8_t)k_ra8_int0_bit_ctrt);
271 char line[k_usb_log_width];
272 (void)snprintf(line, sizeof(line), "SETUP[%u] %s -> CTRT raised", (unsigned)s_host_step, s->name);
273 usb_log_line(line);
274 usb_raise_irq(uc);
275}

294bool host_take_ccpl(void)
295{
296 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_dcpctr);
297 const uint16_t ccpl = (uint16_t)(1U << k_ra8_dcpctr_bit_ccpl);
298 const bool seen = (s_usb.reg[w] & ccpl) != 0U;
299 if (seen) {

324static void host_step_deliver(uc_engine* uc, const usb_setup_step_t* s)
325{
326 host_deliver_setup(uc, s);
327 s_host_wait = 0U;
328 if ((s->bm_request_type & (uint8_t)k_usb_dir_device_to_host) != 0U) {
329 s_host_substate = (uint8_t)k_sub_wait_in; /* control read: wait for data. */
330 } else if (s->b_request == (uint8_t)k_usb_req_set_address) {
331 host_apply_no_data(uc, s);
332 s_host_substate = (uint8_t)k_sub_next; /* SIE-handled; no device CCPL. */
333 } else {
334 s_host_substate = (uint8_t)k_sub_wait_ack; /* no/OUT-data: wait for CCPL. */
335 }
336}

339static void host_step_wait_in(void)
340{
341 if (s_usb.dcp_in.valid && host_dcp_pid_buf()) {
342 host_drain_in();
343 s_host_substate = (uint8_t)k_sub_status;
344 s_host_wait = 0U;
345 return;
346 }
347 s_host_wait++;
348 if (s_host_wait > (uint32_t)k_usb_step_timeout) {
349 usb_log_count("STALLED waiting for control-IN data at step", (unsigned)s_host_step);
350 s_host_substate = (uint8_t)k_sub_status; /* push on; report the stall. */
351 s_host_wait = 0U;
352 }

414void host_run_setup_phase(uc_engine* uc)
415{
416 if (s_host_step >= host_script_len()) {
417 s_host_phase = (uint8_t)k_phase_configured;
418 return;
419 }
420 const usb_setup_step_t* s = &k_enum_script[s_host_step];
421 switch ((usb_host_sub_t)s_host_substate) {
422 case k_sub_deliver:
423 host_step_deliver(uc, s);
424 break;
425 case k_sub_wait_in:
426 host_step_wait_in();
427 break;
428 case k_sub_status:
429 host_step_status(uc);
430 break;
431 case k_sub_wait_ack:
432 host_step_wait_ack();
433 break;
434 case k_sub_next:
435 default:
436 host_step_next(uc, s);
437 break;
438 }

475static void host_echo_send_out(uc_engine* uc)
476{
477 const uint32_t remaining = s_echo_out_len - s_echo_out_sent;
478 const uint32_t chunk = (remaining < (uint32_t)k_usb_pipe_mps) ? remaining : k_usb_pipe_mps;

487 usb_intsts0_set((uint8_t)k_ra8_int0_bit_brdy);
488 s_echo_out_sent += chunk;

504void host_echo_read_in(uc_engine* uc)
505{
506 usb_in_buf_t* b = &s_usb.pipe_in[k_usb_bulk_in_pipe];
507 if (b->valid && (b->len > 0U)) {
508 if (s_dev_class == (uint8_t)k_usb_class_hid) {
509 usb_hid_decode_report(b->data, b->len); /* interrupt-IN mouse report. */
510 } else {
511 s_echo_in_got += b->len;
512 usb_log_count("bulk IN: read echoed bytes from data pipe", (unsigned)b->len);
513 }
514 b->len = 0U;
515 b->valid = false;
516 /* Acknowledge the IN transfer (BEMP) so the device can queue the next. */
517 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_bempsts);
518 s_usb.reg[w] = (uint16_t)(s_usb.reg[w] | (uint16_t)(1U << k_usb_bulk_in_pipe));
519 usb_intsts0_set((uint8_t)k_ra8_int0_bit_bemp);
520 usb_raise_irq(uc);
521 } else if (b->valid) {
522 b->valid = false; /* drained ZLP terminator. */
523 }

575static void host_msc_build_cdb(uint8_t cmd, uint8_t* cdb, uint8_t* cdb_len, uint32_t* data_len)
576{
577 for (uint32_t i = 0U; i < 16U; i++) {
578 cdb[i] = 0U;
579 }
580 switch ((msc_script_cmd_t)cmd) {
581 case k_msc_cmd_inquiry: /* INQUIRY: standard data, 36-byte allocation. */
582 cdb[0] = (uint8_t)k_scsi_inquiry;
583 cdb[4] = (uint8_t)k_inquiry_len;
584 *cdb_len = 6U;
585 *data_len = (uint32_t)k_inquiry_len;
586 break;
587 case k_msc_cmd_read_capacity: /* READ CAPACITY (10): last LBA + block size. */
588 cdb[0] = (uint8_t)k_scsi_read_capacity;
589 *cdb_len = (uint8_t)k_cdb10_len;
590 *data_len = 8U;
591 break;
592 case k_msc_cmd_read10: /* READ (10): LBA 0, one block (512 bytes). */
593 default:
594 cdb[0] = (uint8_t)k_scsi_read10;
595 cdb[8] = 1U; /* transfer length = 1 block (big-endian low byte). */
596 *cdb_len = (uint8_t)k_cdb10_len;
597 *data_len = (uint32_t)k_sector_bytes;
598 break;

603static void host_msc_send_cbw(uc_engine* uc)
604{
605 uint8_t cdb[16];
606 uint8_t cdb_len = 0U;
607 uint32_t data_len = 0U;
608 host_msc_build_cdb(s_msc_cmd, cdb, &cdb_len, &data_len);
609 s_msc_data_len = data_len;
610 s_msc_data_got = 0U;
611 s_msc_tag++;
612
613 usb_out_buf_t* b = &s_usb.pipe_out[k_usb_bulk_out_pipe];
614 uint8_t* d = b->data;
615 d[0] = (uint8_t)'U'; /* dCBWSignature 'USBC' (LE 0x43425355). */
616 d[1] = (uint8_t)'S';
617 d[2] = (uint8_t)'B';
618 d[3] = (uint8_t)'C';
619 /* dCBWTag / dCBWDataTransferLength are little-endian u32 (USB MSC BOT 5.1). */
620 d[k_cbw_tag_off + k_le_lane_b0] = (uint8_t)(s_msc_tag & (uint32_t)k_usb_byte_mask);
621 d[k_cbw_tag_off + k_le_lane_b1] = (uint8_t)((s_msc_tag >> 8) & (uint32_t)k_usb_byte_mask);
622 d[k_cbw_tag_off + k_le_lane_b2] = (uint8_t)((s_msc_tag >> 16) & (uint32_t)k_usb_byte_mask);
623 d[k_cbw_tag_off + k_le_lane_b3] =
624 (uint8_t)((s_msc_tag >> (uint32_t)k_usb_shift24) & (uint32_t)k_usb_byte_mask);
625 d[k_cbw_dtl_off + k_le_lane_b0] = (uint8_t)(data_len & (uint32_t)k_usb_byte_mask);
626 d[k_cbw_dtl_off + k_le_lane_b1] = (uint8_t)((data_len >> 8) & (uint32_t)k_usb_byte_mask);
627 d[k_cbw_dtl_off + k_le_lane_b2] = (uint8_t)((data_len >> 16) & (uint32_t)k_usb_byte_mask);
628 d[k_cbw_dtl_off + k_le_lane_b3] =
629 (uint8_t)((data_len >> (uint32_t)k_usb_shift24) & (uint32_t)k_usb_byte_mask);
630 d[k_cbw_flags_off] = (uint8_t)k_msc_flag_in; /* all scripted commands read. */
631 d[k_cbw_lun_off] = 0U; /* LUN 0. */
632 d[k_cbw_cdblen_off] = cdb_len;
633 for (uint32_t i = 0U; i < 16U; i++) {
634 d[k_cbw_cdb_off + i] = cdb[i];
635 }
636 b->len = (uint16_t)k_msc_cbw_len;
637 b->rd = 0U;
638 b->ready = true;
639 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_brdysts);
640 s_usb.reg[w] = (uint16_t)(s_usb.reg[w] | (uint16_t)(1U << k_usb_bulk_out_pipe));
641 usb_intsts0_set((uint8_t)k_ra8_int0_bit_brdy);
642 usb_raise_irq(uc);
643}

646static uint16_t host_msc_take_in(uc_engine* uc, uint8_t* out, uint16_t cap)
647{
648 usb_in_buf_t* b = &s_usb.pipe_in[k_usb_bulk_in_pipe];
649 if (!b->valid) {
650 return 0U;
651 }
652 const uint16_t n = (b->len < cap) ? b->len : cap;
653 for (uint16_t i = 0U; i < n; i++) {
654 out[i] = b->data[i];
655 }
656 b->len = 0U;
657 b->valid = false;
658 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_bempsts);
659 s_usb.reg[w] = (uint16_t)(s_usb.reg[w] | (uint16_t)(1U << k_usb_bulk_in_pipe));
660 usb_intsts0_set((uint8_t)k_ra8_int0_bit_bemp);
661 usb_raise_irq(uc);
662 return n;
663}

666static void host_msc_parse_capacity(const uint8_t* d, uint16_t n)
667{
668 if (n < 8U) {
669 return;
670 }
671 /* Both parameter-block fields are big-endian u32 (SCSI READ CAPACITY(10)). */
672 const uint32_t last_lba =
673 ((uint32_t)d[k_cap10_last_lba_off + k_be_lane_b3] << (uint32_t)k_usb_shift24) |
674 ((uint32_t)d[k_cap10_last_lba_off + k_be_lane_b2] << 16) |
675 ((uint32_t)d[k_cap10_last_lba_off + k_be_lane_b1] << 8) |
676 (uint32_t)d[k_cap10_last_lba_off + k_be_lane_b0];
677 s_msc_block_len = ((uint32_t)d[k_cap10_blocklen_off + k_be_lane_b3] << (uint32_t)k_usb_shift24) |
678 ((uint32_t)d[k_cap10_blocklen_off + k_be_lane_b2] << 16) |
679 ((uint32_t)d[k_cap10_blocklen_off + k_be_lane_b1] << 8) |
680 (uint32_t)d[k_cap10_blocklen_off + k_be_lane_b0];
681 s_msc_blocks = last_lba + 1U;
682}

731static void host_msc_phase_data(uc_engine* uc)
732{
733 uint8_t buf[k_usb_in_cap];
734 const uint16_t n = host_msc_take_in(uc, buf, (uint16_t)sizeof(buf));
735 if (n == 0U) {
736 s_msc_wait++;
737 } else {
738 host_msc_record_data(buf, n);
739 s_msc_data_got += n;
740 s_msc_wait = 0U;
741 }
742 if ((s_msc_data_got >= s_msc_data_len) || (s_msc_wait > (uint32_t)k_usb_step_timeout)) {
743 s_msc_phase = (uint8_t)k_msc_csw;
744 s_msc_wait = 0U;
745 }
746}

753 if ((n >= (uint16_t)k_msc_csw_len) || (s_msc_wait > (uint32_t)k_usb_step_timeout)) {
754 s_msc_cmd++;
755 s_msc_phase = (uint8_t)k_msc_send;
756 s_msc_wait = 0U;

758 s_msc_wait++;

763static void host_msc_drive(uc_engine* uc)
764{
765 switch ((msc_phase_t)s_msc_phase) {
766 case k_msc_send:
767 host_msc_phase_send(uc);
768 break;
769 case k_msc_data:
770 host_msc_phase_data(uc);
771 break;
772 case k_msc_csw:
773 host_msc_phase_csw(uc);
774 break;
775 case k_msc_done:
776 default:
777 break;
778 }

782void host_run_configured_phase(uc_engine* uc)
783{
784 if (s_dev_class == (uint8_t)k_usb_class_msc) {
785 host_msc_drive(uc); /* run the BOT/SCSI script against the RAM disk. */
786 return;
787 }
788 host_echo_read_in(uc);
789 if (s_dev_class == (uint8_t)k_usb_class_hid) {
790 return; /* keep polling the HID interrupt-IN pipe; reports keep flowing. */
791 }
792 if (s_echo_out_len == 0U) {
793 s_host_phase = (uint8_t)k_phase_done;
794 return;
795 }
796 /* Pace one OUT packet, then let the device echo it back before the next. */
797 if (s_echo_out_sent < s_echo_out_len) {
798 if (s_host_wait == 0U) {
799 host_echo_send_out(uc);
800 }
801 s_host_wait++;
802 if (s_host_wait >= (uint32_t)k_usb_reset_settle) {
803 s_host_wait = 0U;
804 }
805 return;
806 }
807 /* All OUT bytes sent; drain the tail echo for a settle window, then finish. */
808 s_host_wait++;
809 if (s_host_wait >= (uint32_t)k_usb_post_cfg_idle) {
810 s_host_phase = (uint8_t)k_phase_done;
811 }
