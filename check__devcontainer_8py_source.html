4"""Lint and format-check the devcontainer: the Dockerfile and the zshrc.
5
6WHY THESE TWO FILES, TOGETHER
7=============================
8`.devcontainer/Dockerfile` pins every tool version CI resolves -- clang-format
922, the ARM toolchain by sha256, ruff, shellcheck, shfmt, cmake-format,
10yamllint, actionlint. A defect in it changes what every other gate in this
11repository runs. It was, until #371, linted and formatted by nothing at all.
12`.devcontainer/zshrc` is copied to ~/.zshrc at image build time and is the
13other half of the same artefact. They are checked as one unit because they are
14one deliverable: the container.
15
16THE TOOLS, AND WHY EACH ONE
17---------------------------
18Dockerfile -> hadolint (pinned 2.14.0). The obvious and only serious candidate:
19 a single static binary, the same provisioning shape actionlint and shfmt
20 already use. It found a real defect on its first run -- DL4006, eight
21 `curl -fsSL <url> | tar -x` pipelines running under `/bin/sh -c`, where the
22 pipeline's status is that of `tar` alone. A failed download produced an empty
23 tarball, tar succeeded on it, and the image was built with the pinned tool
24 silently absent. That is precisely the class check_errexit_masking.py exists
25 to catch in shell, and it was live in the file that provisions CI. Fixed by
26 the `SHELL ["/bin/bash", "-o", "pipefail", "-c"]` directive. Rule
27 configuration and the one deliberate ignore live in .hadolint.yaml.
28
29zshrc -> `zsh -n` (zsh 5.9, already in the image; no new provisioning).
30 There is no zsh linter and no zsh formatter, and this was checked rather than
31 assumed:
32
33 * ShellCheck refuses zsh outright ("ShellCheck only supports sh/bash/dash/
34 ksh"). Running it as `--shell=bash` is worse than not running it: this
35 very file uses `${(%):-%n}` and `plugins=(...)`, zsh parameter-expansion
36 flags and array syntax that bash cannot parse, so bash-mode ShellCheck
37 reports errors that are not errors. A linter that is wrong about the
38 file is not partial coverage, it is noise that trains people to ignore
39 the gate. That is why lint_coverage_rules.py classifies `.zsh` as its
40 own class rather than folding it into `shell`.
41 * shfmt likewise handles sh/bash/mksh only.
42 * `zsh -n` parses the file with the real zsh grammar and reports syntax
43 errors without executing anything. It is a genuine check, it is the only
44 one that exists, and it is what runs here.
45
46 Being honest about the limit: `zsh -n` catches syntax, not semantics. It
47 would not notice a misspelled plugin name. That is the ceiling of what is
48 available for the language, and it is still strictly more than the nothing
49 this file had before.
50
51FORMAT ROLE
52-----------
53Neither tool rewrites its input, so the formatter half is a canonical-form
54checker -- the same arrangement lint_coverage_rules.py already documents for
55Makefiles and linker scripts, where nothing in the ecosystem rewrites the file
56either. DC001 rejects the layout deviations that a formatter would otherwise
57fix: non-ASCII bytes, CRLF, a missing final newline, tab indentation and
58trailing whitespace.
59
60Run with --selftest to prove both linters fire on deliberately broken input and
61stay quiet on the real files.
62"""

103 def __init__(self, rel: str, line: int, code: str, msg: str) -> None:
104 """Record one finding; all four fields are required and none is derived."""
105 self.rel, self.line, self.code, self.msg = rel, line, code, msg

108 """Render as ``rel:line: [CODE] message`` -- editor-jumpable."""
109 return f"{self.rel}:{self.line}: [{self.code}] {self.msg}"

112def classify(rel: str) -> str | None:
113 """The shared class of `rel`, by exact name then extension."""
114 name = rel.rsplit("/", 1)[-1]
115 if name in NAME_CLASS:
116 return NAME_CLASS[name]
117 if "." in name[1:]:
118 suffix = "." + name.rsplit(".", 1)[-1]
119 return EXT_CLASS.get(suffix.lower())
120 return None

144def check_format(rel: str, raw: bytes) -> list[Finding]:
145 """DC001 -- the canonical-form rules, on raw bytes."""
146 findings: list[Finding] = []
147
148 def add(line: int, msg: str) -> None:
149 """Append one DC001 finding, closing over this file's relative path."""
150 findings.append(Finding(rel, line, "DC001", msg))
151
152 try:
153 text = raw.decode("ascii")
154 except UnicodeDecodeError as exc:
155 add(raw[: exc.start].count(b"\n") + 1, f"non-ASCII byte 0x{raw[exc.start]:02x}")
156 text = raw.decode("ascii", errors="replace")
157 if b"\r\n" in raw:
158 add(raw.split(b"\r\n")[0].count(b"\n") + 1, "CRLF line ending")
159 if raw and not raw.endswith(b"\n"):
160 add(text.count("\n") + 1, "no final newline")
161 for num, line in enumerate(text.splitlines(), start=1):
162 if line.startswith("\t"):
163 add(num, "tab indentation (use spaces)")
164 if line != line.rstrip():
165 add(num, "trailing whitespace")
166 return findings

169def run_hadolint(rel: str, path: Path) -> list[Finding]:
170 """DC002 -- hadolint over one Dockerfile. Config comes from .hadolint.yaml.
171
172 hadolint prints ``<file>:<line> <CODE> <severity>: <message>``. That leading
173 ``<file>:<line>`` is re-parsed rather than passed through, so a finding reads
174 like every other checker here instead of carrying a second, absolute copy of
175 the path inside the message.
176 """
177 proc = subprocess.run( # noqa: S603 -- fixed argv
178 ["hadolint", "--no-color", str(path)], # noqa: S607 -- provisioned on PATH
179 cwd=REPO_ROOT,
180 capture_output=True,
181 text=True,
182 check=False,
183 )
184 findings: list[Finding] = []
185 prefix = f"{path}:"
186 for raw in (proc.stdout + proc.stderr).splitlines():
187 line = raw.strip()
188 if not line:
189 continue
190 num = 0
191 if line.startswith(prefix):
192 rest = line[len(prefix) :]
193 head, _, tail = rest.partition(" ")
194 if head.isdigit():
195 num, line = int(head), tail
196 findings.append(Finding(rel, num, "DC002", line))
197 return findings

200def run_zsh_syntax(rel: str, path: Path) -> list[Finding]:
201 """DC003 -- `zsh -n` over one zsh file. Parses only; never executes."""
202 proc = subprocess.run( # noqa: S603 -- fixed argv
203 ["zsh", "-n", str(path)], # noqa: S607 -- zsh is in the image
204 cwd=REPO_ROOT,
205 capture_output=True,
206 text=True,
207 check=False,
208 )
209 if proc.returncode == 0:
210 return []
211 detail = (proc.stderr + proc.stdout).strip() or f"zsh -n exited {proc.returncode}"
212 # zsh reports `<file>:<line>: <message>`; drop its copy of the path for the
213 # same reason as hadolint's above.
214 num = 0
215 prefix = f"{path}:"
216 if detail.startswith(prefix):
217 rest = detail[len(prefix) :]
218 head, _, tail = rest.partition(":")
219 if head.isdigit():
220 num, detail = int(head), tail.strip()
221 return [Finding(rel, num, "DC003", f"zsh syntax error: {detail}")]

224def check_one(rel: str, root: Path = REPO_ROOT) -> list[Finding]:
225 """Every finding for one devcontainer file."""
226 path = root / rel
227 findings = check_format(rel, path.read_bytes())
228 cls = classify(rel)
229 if cls == "dockerfile":
230 findings.extend(run_hadolint(rel, path))
231 elif cls == "zsh":
232 findings.extend(run_zsh_syntax(rel, path))
233 return findings

236def require_tools() -> int:
237 """Fail loudly when a linter is absent. A gate must never degrade to a no-op."""
238 missing = []
239 if shutil.which("hadolint") is None:

244 missing.append("zsh -- apt-get install zsh (already in the devcontainer image)")
245 if missing:
246 print("check_devcontainer.py: FATAL -- required tool(s) absent:", file=sys.stderr)
247 for item in missing:
248 print(f" {item}", file=sys.stderr)
249 print(
250 " This gate FAILS rather than skipping: a checker whose tool is\n"

252 file=sys.stderr,
253 )
254 return 1
255 return 0

279def _assert_dockerfile(root: Path, failures: list[str]) -> None:
280 """Assert hadolint fires on a bad Dockerfile and is silent on a clean one."""
281 (root / ".devcontainer/Dockerfile").write_bytes(GOOD_DOCKERFILE)
282 got = check_one(".devcontainer/Dockerfile", root)
283 expect(not got, f"a clean Dockerfile yields no findings (got {got})", failures)
284
285 (root / ".devcontainer/Dockerfile").write_bytes(BAD_DOCKERFILE)
286 got = check_one(".devcontainer/Dockerfile", root)
287 blob = " ".join(f.msg for f in got)
288 expect(any(f.code == "DC002" for f in got), "hadolint fires on a bad Dockerfile", failures)
289 expect("DL3020" in blob, " ... reporting DL3020 (ADD instead of COPY)", failures)
290 expect("DL4006" in blob, " ... reporting DL4006 (pipe without pipefail)", failures)

293def _assert_zshrc(root: Path, failures: list[str]) -> None:
294 """Assert zsh -n fires on a broken zshrc and accepts zsh-only syntax.
295
296 The negative case is the whole reason this file's class is ``zsh`` and not
297 ``shell``: bash cannot parse either of those lines, so checking them with
298 bash would report a syntax error in a correct file.
299 """
300 (root / ".devcontainer/zshrc").write_bytes(GOOD_ZSHRC)
301 got = check_one(".devcontainer/zshrc", root)
302 expect(not got, f"a clean zshrc yields no findings (got {got})", failures)
303
304 (root / ".devcontainer/zshrc").write_bytes(BAD_ZSHRC)
305 got = check_one(".devcontainer/zshrc", root)
306 expect(
307 any(f.code == "DC003" for f in got),
308 "zsh -n fires on a zshrc with a syntax error",
309 failures,
310 )
311
312 (root / ".devcontainer/zshrc").write_bytes(
313 b'H=${(%):-%n}\nplugins=(git gh docker)\nprint -r -- "$H ${#plugins}"\n'
314 )
315 got = check_one(".devcontainer/zshrc", root)
316 expect(
317 not got,
318 f"zsh-specific syntax is accepted, not flagged (got {got})",
319 failures,
320 )

323def _assert_format_rules(failures: list[str]) -> None:
324 """Assert each DC001 byte-level formatting rule fires on its own payload."""
325 for payload, want, label in (
326 (b"FROM x\n\tRUN y\n", "tab indentation", "tab indentation fires DC001"),
327 (b"FROM x \n", "trailing whitespace", "trailing whitespace fires DC001"),
328 (b"FROM x", "no final newline", "a missing final newline fires DC001"),
329 (b"FROM x\r\ny\n", "CRLF", "a CRLF ending fires DC001"),
330 (b"FROM \xc2\xa9\n", "non-ASCII", "a non-ASCII byte fires DC001"),
331 ):
332 got = check_format("f", payload)
333 expect(any(want in f.msg for f in got), label, failures)

370def main(argv: list[str]) -> int:
371 """Lint and format-check the devcontainer definition.
372
373 The devcontainer is what gives the Mac an Ubuntu userland to run the CI
374 suite in, so a broken definition does not fail loudly -- it makes `make
375 ci` unable to run at all, on the one platform that cannot verify natively.
376
377 Returns 0 when clean, 1 on any finding or a failing selftest.
378 """
379 ap = argparse.ArgumentParser(description="Lint and format-check the devcontainer files")
380 ap.add_argument("--selftest", action="store_true", help="assert both directions")
381 ap.add_argument("--list-files", action="store_true", help="print the scanned file list")
382 args = ap.parse_args(argv[1:])
383
384 if args.selftest:
385 return selftest()
386
387 files = targets()
388 if args.list_files:
389 print("\n".join(files))
390 return 0
391
392 if require_tools() != 0:
393 return 2
394 if len(files) < FILE_FLOOR:
395 print(
396 f"check_devcontainer.py: FATAL -- {len(files)} file(s) found, floor is "
397 f"{FILE_FLOOR}. An empty scan reports success because it saw nothing.",
398 file=sys.stderr,
399 )
400 return 2
401
402 findings: list[Finding] = []
403 for rel in files:
404 findings.extend(check_one(rel))
405 if findings:
406 print(
407 f"\n{len(findings)} finding(s) in {len(files)} devcontainer file(s):\n", file=sys.stderr
408 )
409 for finding in findings:
410 print(f" {finding}", file=sys.stderr)
411 return 1
412 print(f"check_devcontainer.py: {len(files)} file(s), no findings.")
413 return 0
