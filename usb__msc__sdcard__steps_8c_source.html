268static UINT sdmsc_msc_read(VOID* storage,
269 ULONG lun,
270 UCHAR* data_pointer,
271 ULONG number_blocks,
272 ULONG lba,
273 ULONG* media_status)
274{
275 (void)storage;
276 (void)lun;
277 s_dbg_read_calls++;
278 if ((lba + number_blocks) > (ULONG)s_usb_msc_sdcard_blocks) {
279 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
280 k_scsi_asc_lba_out_of_range,
281 k_scsi_ascq_none);
282 return UX_ERROR;
283 }
284 const ra8_err_t err =
285 ra8_sdmmc_spi_read_blocks((uint32_t)lba, data_pointer, (uint32_t)number_blocks);
286 if (err != k_ra8_ok) {
287 s_dbg_media_err = (uint32_t)err;
288 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_medium_error,
289 k_scsi_asc_unrecovered_read,
290 k_scsi_ascq_none);
291 return UX_ERROR;
292 }
293 s_dbg_read_blocks += (uint32_t)number_blocks;
294 *media_status = 0UL;
295 (void)ra8_board_led_toggle(k_ra8_board_led1);
296 return UX_SUCCESS;
297}

518VOID sdmsc_device_worker(ULONG arg)
519{
520 (void)arg;
521
522 if (s_usb_msc_sdcard_blocks == 0U) {
523 (void)sdmsc_print_fail("no card -- not attaching USB", k_ra8_err_invalid_state);
524 return;
525 }
526 UINT ux = sdmsc_usbx_stack_up();
527 if (ux != UX_SUCCESS) {
528 s_dbg_dev_err = (uint32_t)ux;
529 (void)sdmsc_print_fail("usbx stack", (ra8_err_t)ux);
530 return;
531 }
532 s_dbg_dev_step = (uint32_t)k_sdmsc_dev_step_stack;
533 ux = sdmsc_class_register();
534 if (ux != UX_SUCCESS) {
535 s_dbg_dev_err = (uint32_t)ux;
536 (void)sdmsc_print_fail("msc class", (ra8_err_t)ux);
537 return;
538 }
539 s_dbg_dev_step = (uint32_t)k_sdmsc_dev_step_class;
540 ra8_err_t err = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
541 if (err != k_ra8_ok) {
542 s_dbg_dev_err = (uint32_t)err;
543 (void)sdmsc_print_fail("dcd init", err);
544 return;
545 }
546 s_dbg_dev_step = (uint32_t)k_sdmsc_dev_step_dcd;
547 err = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
548 if (err != k_ra8_ok) {
549 s_dbg_dev_err = (uint32_t)err;
550 (void)sdmsc_print_fail("device attach", err);
551 return;
552 }
553 s_dbg_dev_step = (uint32_t)k_sdmsc_dev_step_attach;
554 (void)sdmsc_print_ready();
555
556 while (1) {
557 s_dbg_dev_step = (uint32_t)k_sdmsc_dev_step_parked;
558 tx_thread_sleep(k_sdmsc_idle_ticks);
559 }
560}
