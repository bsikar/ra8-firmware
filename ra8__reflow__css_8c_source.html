202RA8_INTERNAL
203static bool priv_ci_contains(const char* s, size_t len, const char* sub)
204{
205 const size_t sl = strlen(sub);
206 /* mcdc-deactivated: the sole caller passes priv_ci_contains(val, vlen, "underline") -- val is non-NULL (from ra8_reflow_css_trim) and sub is a compile-time literal so sl == 9 (never 0); (s == nullptr) and (sl == 0U) are unreachable, only (sl > len) varies. */
207 if ((s == nullptr) || (sl == 0U) || (sl > len)) {
208 return false;
209 }
210 /* Bounded: at most (len - sl + 1) windows; i advances by 1 each step. */
211 for (size_t i = 0U; (i + sl) <= len; ++i) {
212 bool hit = true;
213 /* Bounded: j < sl, the NUL-terminated literal's strlen. */
214 for (size_t j = 0U; j < sl; ++j) {
215 if (ra8_reflow_css_lower(s[i + j]) != ra8_reflow_css_lower(sub[j])) {
216 hit = false;
217 break;
218 }
219 }
220 if (hit) {
221 return true;
222 }
223 }
224 return false;
225}

228const char* ra8_reflow_css_trim(const char* s, size_t* len)
229{
230 size_t a = 0U;
231 size_t b = *len;
232 while ((a < b) && ra8_reflow_css_is_ws(s[a])) {
233 ++a;
234 }
235 while ((b > a) && ra8_reflow_css_is_ws(s[b - 1U])) {
236 --b;
237 }
238 *len = b - a;
239 return &s[a];

308static uint32_t priv_parse_hex_color(const char* s, size_t len)
309{
310 uint32_t rgb = 0U;
311 if (len == (size_t)k_priv_hex6_len) {
312 /* Bounded: exactly k_priv_hex6_len digits. */
313 for (size_t i = 0U; i < len; ++i) {
314 const uint8_t v = priv_hex_val(s[i]);
315 if (v >= (uint8_t)k_priv_hex_base) {
316 return (uint32_t)k_priv_col_invalid;
317 }
318 rgb = (rgb << (uint32_t)k_priv_hex_nib) | (uint32_t)v;
319 }
320 return rgb;
321 }
322 if (len == (size_t)k_priv_hex3_len) {
323 /* Bounded: exactly k_priv_hex3_len digits; each expands d -> dd. */
324 for (size_t i = 0U; i < len; ++i) {
325 const uint8_t v = priv_hex_val(s[i]);
326 if (v >= (uint8_t)k_priv_hex_base) {
327 return (uint32_t)k_priv_col_invalid;
328 }
329 rgb = (rgb << (uint32_t)k_priv_hex_chan) | ((uint32_t)v << (uint32_t)k_priv_hex_nib) |
330 (uint32_t)v;
331 }
332 return rgb;
333 }
334 return (uint32_t)k_priv_col_invalid;
335}

429RA8_INTERNAL
430static uint32_t priv_scan_hundredths(const char* s, size_t len, size_t* i, bool* any)
431{
432 uint32_t hund = 0U;
433 /* Bounded: integer digits, i advances by 1 each step, capped by len. */
434 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9')) {
435 hund = (hund * (uint32_t)k_priv_fs_dec) + (uint32_t)(s[*i] - '0');
436 *any = true;
437 ++(*i);
438 }
439 hund *= (uint32_t)k_priv_fs_pct1; /* integer part -> hundredths */
440 if ((*i < len) && (s[*i] == '.')) {
441 ++(*i);
442 uint32_t place = (uint32_t)k_priv_fs_pct1 / (uint32_t)k_priv_fs_dec; /* tenths */
443 size_t fd = 0U;
444 /* Bounded: at most k_priv_fs_frac fractional digits. */
445 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9') && (fd < (size_t)k_priv_fs_frac)) {
446 hund += (uint32_t)(s[*i] - '0') * place;
447 place /= (uint32_t)k_priv_fs_dec;
448 *any = true;
449 ++fd;
450 ++(*i);
451 }
452 /* Bounded: skip any remaining fractional digits, i capped by len. */
453 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9')) {
454 ++(*i);
455 }
456 }
457 return hund;
458}

520static bool priv_parse_fontsize(const char* s, size_t len, uint16_t* out_val, uint8_t* out_unit)
521{
522 size_t i = 0U;
523 bool any = false;
524 uint32_t hund = priv_scan_hundredths(s, len, &i, &any);
525 if (!any) {
526 return false;
527 }
528 const size_t rem = len - i;
529 if ((rem == (size_t)k_priv_fs_ulen) && ra8_reflow_css_ci_eq(&s[i], rem, "px")) {
530 *out_val = priv_fs_whole(hund);
531 *out_unit = (uint8_t)k_ra8_css_font_px;
532 return true;
533 }
534 if ((rem == 1U) && (s[i] == '%')) {
535 *out_val = priv_fs_whole(hund);
536 *out_unit = (uint8_t)k_ra8_css_font_pct;
537 return true;
538 }
539 if ((rem == (size_t)k_priv_fs_ulen) && ra8_reflow_css_ci_eq(&s[i], rem, "em")) {
540 /* 1em = 100%; `hund` is value*100, which is already the percent. */
541 *out_val = (uint16_t)((hund > (uint32_t)k_priv_fs_max) ? (uint32_t)k_priv_fs_max : hund);
542 *out_unit = (uint8_t)k_ra8_css_font_pct;
543 return true;
544 }
545 return false;
546}

729 uint16_t fv = 0U;

732 out->set = (uint8_t)(out->set | (uint8_t)k_ra8_css_set_fontsize);
733 out->font_val = fv;
734 out->font_unit = fu;
735 }
736 } else if (ra8_reflow_css_ci_eq(prop, plen, "display")) {
737 out->set = (uint8_t)(out->set | (uint8_t)k_ra8_css_set_display);
738 out->display = ra8_reflow_css_ci_eq(val, vlen, "none") ? 1U : 0U;

767void ra8_reflow_css_parse_decls(const char* s, size_t len, ra8_css_style_t* out)
768{
769 size_t i = 0U;
770 /* Bounded: each iteration consumes at least one byte via the ';' advance. */
771 while (i < len) {
772 size_t semi = i;
773 while ((semi < len) && (s[semi] != ';')) {
774 ++semi;
775 }
776 size_t colon = i;
777 while ((colon < semi) && (s[colon] != ':')) {
778 ++colon;
779 }
780 if (colon < semi) {
781 size_t plen = colon - i;
782 const char* prop = ra8_reflow_css_trim(&s[i], &plen);
783 size_t vlen = semi - (colon + 1U);
784 const char* val = ra8_reflow_css_trim(&s[colon + 1U], &vlen);
785 if ((plen > 0U) && (vlen > 0U)) {
786 priv_apply_decl(prop, plen, val, vlen, out);
787 }
788 }
789 i = semi + 1U;
790 }
791}
