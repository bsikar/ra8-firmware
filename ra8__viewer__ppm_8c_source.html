68uint16_t ra8_viewer_pack565(uint8_t rr, uint8_t gg, uint8_t bb)
69{
70 const uint16_t r5 = (uint16_t)(rr >> (uint16_t)k_rgb565_r5_drop);
71 const uint16_t g6 = (uint16_t)(gg >> (uint16_t)k_rgb565_g6_drop);
72 const uint16_t b5 = (uint16_t)(bb >> (uint16_t)k_rgb565_r5_drop);
73 return (uint16_t)((r5 << (uint16_t)k_rgb565_r_shift) | (g6 << (uint16_t)k_rgb565_g_shift) | b5);
74}

93static void viewer_unpack565(uint16_t px, uint8_t rgb[k_ppm_channels])

95 const uint32_t r5 = ((uint32_t)px >> (uint32_t)k_rgb565_r_shift) & (uint32_t)k_rgb565_r5_mask;

107ra8_err_t ra8_viewer_write_ppm565(const uint16_t* px, uint32_t w, uint32_t h, const char* path)
108{
109 if ((px == nullptr) || (path == nullptr)) {
110 return k_ra8_err_null_ptr;
111 }
112 if ((w == 0U) || (h == 0U)) {
113 return k_ra8_err_invalid_size;
114 }
115 FILE* fp = fopen(path, "wb");
116 if (fp == nullptr) {
117 return k_ra8_err_not_found;
118 }
119 (void)fprintf(fp, "P6\n%u %u\n255\n", (unsigned)w, (unsigned)h);
120 const size_t pixels = (size_t)w * (size_t)h;
121 for (size_t i = 0U; i < pixels; ++i) {
122 uint8_t rgb[k_ppm_channels];
123 viewer_unpack565(px[i], rgb);
124 (void)fwrite(rgb, 1U, sizeof(rgb), fp);
125 }
126 const int closed = fclose(fp);
127 return (closed == 0) ? k_ra8_ok : k_ra8_err_not_found;
128}
