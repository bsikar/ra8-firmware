294ra8_err_t
295ra8_lvd_set_filter(ra8_lvd_channel_t channel, ra8_lvd_loco_div_t filter_div, bool filter_en)
296{
297 uint8_t idx = 0U;
298 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
299 RA8_RETURN_ON_ERROR(map_err, s_tag, "lvd_set_filter: bad channel"); /* GCOVR_EXCL_BR_LINE */
300
301 const ra8_err_t div_err = ra8_lvd_internal_validate_div(filter_div);
302 RA8_RETURN_ON_ERROR(div_err, s_tag, "lvd_set_filter: bad div"); /* GCOVR_EXCL_BR_LINE */
303
304 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
305
306 /* HUM Ch 8.2.4 "PVDmCR0" p 305 / HUM Ch 8.2.5 "PVDnCR0" p 306
307 * -- FSAMP can only be modified while DFDIS = 1. */
308 ra8_lvd_internal_cr0_rmw(&map, 0U, k_ra8_lvd_cr0_mask_dfdis);
309
310 const uint8_t fsamp_bits =
311 (uint8_t)(((uint8_t)filter_div << k_ra8_lvd_cr0_shift_fsamp) & k_ra8_lvd_cr0_mask_fsamp);
312 ra8_lvd_internal_cr0_rmw(&map, k_ra8_lvd_cr0_mask_fsamp, fsamp_bits);
313
314 if (filter_en) {
315 /* Drop DFDIS only after FSAMP has landed. */
316 ra8_lvd_internal_cr0_rmw(&map, k_ra8_lvd_cr0_mask_dfdis, 0U);
317 }
318 return k_ra8_ok;
319}

346ra8_err_t ra8_lvd_set_hysteresis_mode(ra8_lvd_channel_t channel, ra8_lvd_hysteresis_t hyst)
347{
348 uint8_t idx = 0U;
349 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
350 RA8_RETURN_ON_ERROR(map_err,
351 s_tag,
352 "lvd_set_hysteresis_mode: bad channel"); /* GCOVR_EXCL_BR_LINE */
353
354 if ((uint8_t)hyst > k_ra8_lvd_hysteresis_hvd) {
355 return k_ra8_err_invalid_arg;
356 }
357
358 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
359
360 /* HUM Ch 8.2.8 "PVDmFCR" p 308 last bullet -- "RHSEL must not be set
361 * to 1 when PVDmCR0.RI = 0." Enforce only on m channels (n channels
362 * have no RI bit; their RHSEL is always legal). */
363 if (map.has_irq && (hyst == k_ra8_lvd_hysteresis_hvd)) {
364 if (ra8_lvd_internal_read_ri(&map) == 0U) {
365 return k_ra8_err_invalid_state;
366 }
367 }
368
369 /* HUM Ch 8.2.8 / 8.2.9 -- RHSEL can only be modified when every
370 * PVDE is 0; preserve and restore this channel's PVDE. */
371 const uint8_t prev_cmpcr = *ra8_lvd_reg8(map.cmpcr);
372 const uint8_t pvde_was = (uint8_t)(prev_cmpcr & k_ra8_lvd_cmpcr_mask_pvde);
373 *ra8_lvd_reg8(map.cmpcr) = (uint8_t)(prev_cmpcr & (uint8_t)~k_ra8_lvd_cmpcr_mask_pvde);
374
375 /* HUM Ch 8.2.8 "PVDmFCR" p 308 */
376 *ra8_lvd_reg8(map.fcr) = (uint8_t)((uint8_t)hyst & k_ra8_lvd_fcr_mask_rhsel);
377
378 /* Restore PVDE if it had been set. */
379 *ra8_lvd_reg8(map.cmpcr) = (uint8_t)(*ra8_lvd_reg8(map.cmpcr) | pvde_was);
380 return k_ra8_ok;
381}

407ra8_err_t ra8_lvd_set_negate_mode(ra8_lvd_channel_t channel, ra8_lvd_negate_t negate)
408{
409 uint8_t idx = 0U;
410 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
411 RA8_RETURN_ON_ERROR(map_err, s_tag, "lvd_set_negate_mode: bad channel"); /* GCOVR_EXCL_BR_LINE */
412
413 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
414 if (!map.has_irq) {
415 return k_ra8_err_not_supported;
416 }
417 if ((uint8_t)negate > k_ra8_lvd_negate_after_assert) {
418 return k_ra8_err_invalid_arg;
419 }
420
421 /* HUM Ch 8.2.4 "PVDmCR0" p 306 */
422 /* HUM Ch 8.2.8 "PVDmFCR" p 308 */
423 if (negate == k_ra8_lvd_negate_after_assert) {
424 const uint8_t fcr = *ra8_lvd_reg8(map.fcr);
425 if ((fcr & k_ra8_lvd_fcr_mask_rhsel) != 0U) {
426 return k_ra8_err_invalid_state;
427 }
428 }
429
430 /* HUM Ch 8.2.4 "PVDmCR0" p 305 */
431 ra8_lvd_internal_cr0_rmw(&map,
432 k_ra8_lvd_cr0_mask_rn,
433 (negate == k_ra8_lvd_negate_after_assert) ? k_ra8_lvd_cr0_mask_rn : 0U);
434 return k_ra8_ok;
435}
