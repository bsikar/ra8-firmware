185static bool gpt_dma_demo_run_one(void)
186{
187 s_dma_done = 0U;
188 uint8_t dma_chan = 0U;
189 /* ra8_gpt_write_dma succeeding proves: ra8_dma_init worked, a DMAC
190 * channel was allocated, the source / dest registers were
191 * programmed, the transfer was kicked. The completion-callback
192 * path additionally requires the per-app vector table to route
193 * the DMAC interrupt to ra8_dma's dispatcher (not wired in this
194 * app's boilerplate), so the IRQ-driven s_dma_done flag is a
195 * best-effort signal, not the gate. */
196 if (ra8_gpt_write_dma((uint8_t)k_gpt_dma_demo_channel,
197 s_period_table,
198 (uint16_t)k_gpt_dma_demo_period_count,
199 gpt_dma_demo_on_complete,
200 nullptr,
201 &dma_chan) != k_ra8_ok) {
202 return false;
203 }
204 /* Best-effort wait -- on real IRQ wiring the callback fires fast;
205 * without it the bounded wait simply elapses. */
206 enum : uint32_t { k_wait_iters = 5U };
207 for (uint32_t i = 0U; i < k_wait_iters; ++i) {
208 if (s_dma_done != 0U) {
209 break;
210 }
211 ra8_delay_ms(1U);
212 }
213 /* Manually release the channel because completion IRQ is not wired
214 * in this app's vector_table; without this each iteration leaks a
215 * channel and the 8-slot pool fills after 8 transfers. */
216 (void)ra8_dma_release(dma_chan);
217 return true;
218}
