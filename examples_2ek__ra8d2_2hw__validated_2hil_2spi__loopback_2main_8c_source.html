151static bool spi_demo_round_trip_ok(void)
152{
153 for (uint8_t i = 0U; i < (uint8_t)k_spi_demo_pattern_len; i++) {
154 const uint8_t tx = (uint8_t)((uint8_t)k_spi_demo_pattern_base + i);
155 uint8_t rx = 0U;
156 if (ra8_spi_xfer8((uint8_t)k_spi_demo_spi_channel, tx, &rx) != k_ra8_ok) {
157 return false;
158 }
159 if (rx != tx) {
160 return false;
161 }
162 }
163 return true;
164}
