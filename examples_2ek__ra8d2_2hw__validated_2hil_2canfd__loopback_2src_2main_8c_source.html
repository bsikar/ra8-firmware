226[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_canfd_demo_one_round_trip(uint8_t seq)
227{
228 ra8_canfd_frame_t tx = {
229 .id = (uint32_t)k_canfd_demo_id,
230 .dlc = (uint8_t)k_canfd_demo_dlc,
231 .is_extended = 0U,
232 .is_fd = 0U,
233 .is_brs = 0U,
234 .data = {seq,
235 (uint8_t)k_canfd_demo_payload_b1,
236 (uint8_t)k_canfd_demo_payload_b2,
237 (uint8_t)k_canfd_demo_payload_b3,
238 (uint8_t)k_canfd_demo_payload_b4,
239 (uint8_t)k_canfd_demo_payload_b5,
240 (uint8_t)k_canfd_demo_payload_b6,
241 (uint8_t)k_canfd_demo_payload_b7},
242 };
243 if (ra8_canfd_transmit((uint8_t)k_canfd_demo_channel, &tx) != k_ra8_ok) {
244 return k_ra8_err_hw_error;
245 }
246 /* Poll for the loopback frame: 500 kbit/s + 8 data bytes ~ 240 us
247 * round-trip, so wait up to ~10 ms before declaring no_data. */
248 ra8_canfd_frame_t rx = {};
249 for (uint32_t i = 0U; i < (uint32_t)k_canfd_demo_rx_spin; i++) {
250 if (ra8_canfd_receive((uint8_t)k_canfd_demo_channel, &rx) == k_ra8_ok) {
251 return k_ra8_ok;
252 }
253 }
254 return k_ra8_err_no_data;
255}

257void main(void)
258{
259 internal_canfd_demo_setup_or_halt();
260 ra8_isr_globals_enable();
261
262 uint8_t seq = 0U;
263 while (1) {
264 if (internal_canfd_demo_one_round_trip(seq) == k_ra8_ok) {
265 (void)ra8_board_led_toggle(k_ra8_board_led1);
266 g_canfd_match += 1U;
267 } else {
268 (void)ra8_board_led_toggle(k_ra8_board_led2);
269 g_canfd_mismatch += 1U;
270 }
271 seq++;
272 ra8_delay_ms(k_canfd_demo_period_ms);
273 }
274 internal_canfd_demo_panic_halt();
275}
