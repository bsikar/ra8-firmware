188RA8_INTERNAL static void internal_demo_clocks_or_halt(void)
189{
190 uint32_t cpuclk0_hz = 0U;
191
192 if (ra8_cgc_init() != k_ra8_ok) {
193 internal_demo_panic_halt();
194 }
195 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
196 internal_demo_panic_halt();

198 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
199 internal_demo_panic_halt();

202 internal_demo_panic_halt();

205 internal_demo_panic_halt();

285RA8_INTERNAL static void internal_demo_thread_entry(ULONG arg)
286{
287 (void)arg;
288
289 internal_demo_log("[nimble] starting NimBLE peripheral\r\n");
290 internal_demo_ble_or_halt();
291 internal_demo_log("[nimble] advertising as ");
292 internal_demo_log(s_demo_local_name);
293 internal_demo_log("\r\n");
294
295 while (1) {
296 /* Pump any inbound HCI events / ACL frames through the bridge. */
297 (void)ra8_ble_dispatch();
298 /* Sleep one full tick window before the next battery decrement. */
299 (void)tx_thread_sleep((ULONG)k_demo_tick_ms);
300 internal_demo_tick_battery();
301 }
302}
