100/* =============================================================================
101 * Peripheral IRQ handlers
102 * =============================================================================
103 *
104 * The RA Interrupt Control Unit exposes 112 routable interrupt lines
105 * to the NVIC (per HUM table 13.x).
106 *
107 * Each peripheral IRQ vector forwards to ra8_isr_dispatch(slot) so any
108 * (handler, ctx) registered through the substrate (ra8_isr_register)
109 * runs in NVIC handler-mode. Slots that were never registered land in
110 * the dispatcher's bounds-checked no-op path. The trampolines are
111 * weak so a driver can still install a fully-custom IRQ handler by
112 * defining a non-weak ``IRQ<n>_Handler``. HUM Ch 13 NVIC + Ch 14 ICU
113 * IELSR.
114 *
115 * NB: a prior revision aliased every IRQn_Handler directly to
116 * Default_Handler (which contains ``bkpt #0``). On debug-disabled
117 * silicon a stray bkpt escalates to HardFault, so the first USB IRQ
118 * after ra8_isr_register armed the NVIC line halted the chip -- this NSC
119 * demo routes USBFS_INT to NVIC slot 0 (ra8_usb_device_attach), so it hit
120 * exactly that. The working tz_secure_only_usb_fs / threadx_usbx_cdc_demo
121 * apps use this dispatching form; this app now matches.
122 */

417/* =============================================================================
418 * Reset handler: copy .data from MRAM to SRAM, zero .bss, call main()
419 * =============================================================================
420 */
421
422/* NOLINTBEGIN(clang-analyzer-security.ArrayBound,misc-use-internal-linkage) -- the .data/.bss copy loops walk linker-symbol bounds the analyzer cannot model, and Reset_Handler must be extern for the vector table. */
423void Reset_Handler(void)
424{
425 /* Step 1: initialise the C runtime FIRST. SystemInit() below performs the
426 * full Secure bring-up (clock tree + SAU + `ra8_trustzone_init`), and
427 * ra8_trustzone_init() BLXNS-es into the Non-Secure world and never returns on
428 * hardware. So the .data copy MUST run before SystemInit() -- if it ran after
429 * (the classic CMSIS order) it would be dead code here, and every Secure
430 * driver would execute against an uninitialised .data image (e.g. log-tag
431 * pointers read as NULL, which the pin validator rejects as a null owner). */
432
433 /* Copy initialized data from flash/MRAM to SRAM. */
434 uint32_t* src = &g_ra8_ls_sidata;
435 uint32_t* dst = &g_ra8_ls_sdata;
436 while (dst < &g_ra8_ls_edata) {
437 *dst++ = *src++;
438 }
439
440 /* Zero BSS. */
441 dst = &g_ra8_ls_sbss;
442 while (dst < &g_ra8_ls_ebss) {
443 *dst++ = 0U;
444 }
445
446 /* Step 2: core-level init (VTOR, FPU, priority grouping, interrupts masked)
447 * followed by the Secure bring-up and the BLXNS into the NS image. Safe to run
448 * after the .data copy: the core steps touch no .data globals, and the Secure
449 * drivers now see a fully initialised .data image. Does not return on hardware. */
450 SystemInit();
451
452 /* Step 3: SystemInit() does not return once it BLXNS-es into NS. Reaching here
453 * means the NS transition was bypassed (e.g. a host / off-target build): fall through to
454 * the Secure fallback main(), which parks the CPU. */
455 (void)main();
456
457 /* main() should never return; if it does, halt. */
458 while (1) {
459 __asm__ volatile("wfi");
460 }
461}
