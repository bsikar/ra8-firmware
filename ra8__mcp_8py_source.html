4"""A Model Context Protocol (MCP) server for this firmware repo.
5
6This server gives an MCP-aware assistant live, structured context about the
7ra8-firmware tree -- the firmware app catalogue, build / test / quality
8workflows, the hardware-in-the-loop (HIL) rig, code search, and the project's
9authoritative docs -- so the assistant can reason about what the repo can
10actually *do* instead of guessing from loose markdown files.
11
12Design goals
13------------
14 * Zero third-party dependencies. It speaks the MCP stdio transport
15 (newline-delimited JSON-RPC 2.0) using only the Python standard library,
16 so it runs anywhere ``python3`` exists with nothing to ``pip install``.
17 That matches the repo's hand-written, minimal-dependency ethos.
18 * Read-only by default. Every tool that only inspects the tree (catalogue,
19 search, docs) runs freely. Every tool that touches real hardware (flash,
20 HIL) is gated behind an explicit ``confirm`` flag and otherwise returns a
21 dry-run preview of the exact command it would run.
22 * Honest output. Build / test / gate tools shell out to the repo's real
23 ``make`` targets and helper scripts, capture the combined output, and
24 return the tail verbatim with the real exit status.
25
26Transport
27---------
28MCP stdio framing is one JSON-RPC message per line on stdin / stdout. All
29human-readable logging goes to stderr so it never corrupts the protocol
30stream. Run ``python3 tools/mcp/ra8_mcp.py --selftest`` to exercise the
31dispatcher in-process without a client (used by ``make mcp``).
32
33The methods implemented are the subset a tools+resources server needs:
34``initialize``, ``ping``, ``tools/list``, ``tools/call``, ``resources/list``,
35``resources/read``, plus empty ``prompts/list`` /
36``resources/templates/list`` replies for clients that probe them.
37"""

63def serve() -> int:
64 """Run the newline-delimited JSON-RPC loop on stdin/stdout until EOF."""
65 log(f"serving {SERVER_NAME} {SERVER_VERSION} on stdio (root {REPO_ROOT})")
66 out = sys.stdout
67 for raw in sys.stdin:
68 line = raw.strip()
69 if not line:
70 continue
71 try:
72 request = json.loads(line)
73 except json.JSONDecodeError as exc:
74 out.write(json.dumps(_error(None, ERR_PARSE, f"parse error: {exc}")) + "\n")
75 out.flush()
76 continue
77 if not isinstance(request, dict):
78 out.write(
79 json.dumps(_error(None, ERR_INVALID_REQUEST, "request must be an object")) + "\n"
80 )
81 out.flush()
82 continue
83 response = dispatch(request)
84 if response is not None:
85 out.write(json.dumps(response) + "\n")
86 out.flush()
87 log("stdin closed, exiting")
88 return 0

91def main(argv: list[str]) -> int:
92 """Run the server, or the in-process self-test with `--selftest`.
93
94 With no flags this blocks in `serve()` reading JSON-RPC from stdin until
95 EOF, which is the normal mode: an MCP client spawns this as a subprocess and
96 speaks the stdio transport to it. Running it from a terminal looks like a
97 hang -- it is waiting for a request.
98
99 `--selftest` exercises the dispatcher without a client or a board, so it is
100 safe to run anywhere and is what CI invokes.
101
102 Args:
103 argv: Argument list WITHOUT the program name (callers pass
104 `sys.argv[1:]`).
105
106 Returns:
107 Process exit status: 0 on clean shutdown or a passing self-test, 1 on a
108 failing one.
109 """
110 parser = argparse.ArgumentParser(description="MCP server for ra8-firmware.")
111 parser.add_argument(
112 "--selftest", action="store_true", help="run the in-process dispatcher self-test and exit"
113 )
114 options = parser.parse_args(argv)
115 if options.selftest:
116 return selftest()
117 return serve()
