4"""check_coverage.py -- gate plain statement+branch coverage against a baseline.
5
6Per CLAUDE.md "IEC 61508 SIL 3 / DO-178C Level B Qualification":
7
8 Statement and branch coverage are necessary but not sufficient
9 for the highest safety integrity level (MC/DC is the gold
10 standard). We still maintain a separate statement+branch
11 baseline because it is fast, cheap, and provides a useful
12 ratchet against accidental regressions.
13
14This script reads `build/coverage/coverage.xml` (the Cobertura XML
15produced by `bash scripts/report/coverage_report.sh`) and compares
16the overall line-rate / branch-rate against the two numbers stored
17in `.github/coverage-baseline.txt`. It fails (exit 1) if either
18metric drops below baseline by more than the slack tolerance.
19
20Modeled on scripts/checks/check_mcdc_block.py. Enforcing: a
21regression below baseline (minus the slack band) FAILS the gate.
22The baseline has been stable across CI runs, so there is no
23warn-only mode -- a coverage regression blocks the run.
24
25Copyright (c) 2026 Brighton Sikarskie
26SPDX-License-Identifier: MIT
27"""
28
29from __future__ import annotations
30
31import sys
32import xml.etree.ElementTree as ET
33from pathlib import Path
34
35SLACK_PCT = 0.5
36BASELINE_COLUMN_COUNT = 2 # baseline file has two numeric columns: statement% branch%
37"""Allow a 0.5pp regression band before failing. gcovr is deterministic
38across runs on a fixed source tree, but the test selection in CI may
39differ slightly between Ubuntu image versions."""
40
41REPO_ROOT = Path(__file__).resolve().parents[2]
42COVERAGE_XML = REPO_ROOT / "build" / "coverage" / "coverage.xml"
43BASELINE_FILE = REPO_ROOT / ".github" / "coverage-baseline.txt"
44
45
46def parse_baseline(path: Path) -> tuple[float, float]:
47 """Read the (statement %, branch %) pair from the baseline file.
48
49 Comment lines are skipped so the baseline can carry a note explaining why
50 a number is where it is -- which is the difference between a ratchet and
51 an unexplained constant.
52 """
53 for line in path.read_text(encoding="utf-8").splitlines():
54 s = line.strip()
55 if not s or s.startswith("#"):
56 continue
57 parts = s.split()
58 if len(parts) >= BASELINE_COLUMN_COUNT:
59 return float(parts[0]), float(parts[1])
60 msg = f"baseline file {path} has no numeric line"
61 raise RuntimeError(msg)
62
63
64def parse_cobertura(path: Path) -> tuple[float, float]:
65 """Read line-rate and branch-rate from the Cobertura XML root element.
66
67 Both attributes are 0..1 floats in the report and are converted to percent
68 here, so every comparison downstream is in one unit.
69 """
70 tree = ET.parse(path) # noqa: S314 # trusted CI-generated Cobertura XML, not user input
71 root = tree.getroot()
72 line_rate = float(root.attrib["line-rate"]) * 100.0
73 branch_rate = float(root.attrib["branch-rate"]) * 100.0
74 return line_rate, branch_rate
75
76
77def main() -> int:
78 """Gate coverage against the committed baseline.
79
80 A missing baseline or a missing coverage XML each exit 1 rather than being
81 treated as "nothing to compare": either one means the measurement did not
82 happen, and passing on absent evidence is what this gate exists to
83 prevent.
84
85 Returns 0 when coverage meets the baseline, 1 otherwise.
86 """
87 if not BASELINE_FILE.exists():
88 print(f"[ERROR] baseline file missing: {BASELINE_FILE}", file=sys.stderr)
89 return 1
90
91 if not COVERAGE_XML.exists():
92 msg = (
93 f"[ERROR] coverage XML missing: {COVERAGE_XML}\n"
94 f" Run `make coverage` (or "
95 f"`bash scripts/report/coverage_report.sh`) first."
96 )
97 print(msg, file=sys.stderr)
98 return 1
99
100 base_line, base_branch = parse_baseline(BASELINE_FILE)
101 line_pct, branch_pct = parse_cobertura(COVERAGE_XML)
102
103 print(f"check_coverage.py: baseline statement={base_line:.1f}% branch={base_branch:.1f}%")
104 print(f"check_coverage.py: measured statement={line_pct:.1f}% branch={branch_pct:.1f}%")
105
106 failures: list[str] = []
107 if line_pct + SLACK_PCT < base_line:
108 failures.append(
109 f" statement coverage {line_pct:.1f}% < baseline "
110 f"{base_line:.1f}% (slack {SLACK_PCT}pp)"
111 )
112 if branch_pct + SLACK_PCT < base_branch:
113 failures.append(
114 f" branch coverage {branch_pct:.1f}% < baseline "
115 f"{base_branch:.1f}% (slack {SLACK_PCT}pp)"
116 )
117
118 if failures:
119 print("[FAIL] coverage regressed below baseline:")
120 for f in failures:
121 print(f)
122 print(" Either add tests to restore coverage, or, if the")
123 print(" regression is intentional, update")
124 print(" .github/coverage-baseline.txt.")
125 return 1
126
127 # If we beat the baseline by more than 1pp on both metrics, suggest
128 # ratcheting. This is a hint; it does not modify the file.
129 if line_pct >= base_line + 1.0 and branch_pct >= base_branch + 1.0:
130 print("check_coverage.py: HINT -- coverage exceeds baseline by >=1pp.")
131 print(
132 f" Consider ratcheting .github/coverage-baseline.txt to "
133 f"{line_pct:.1f} {branch_pct:.1f}."
134 )
135
136 print("check_coverage.py: PASS")
137 return 0

140if __name__ == "__main__":
141 sys.exit(main())
