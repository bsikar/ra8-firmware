142static void lpm_dpsby3_add_offset(const ra8_rtc_datetime_t* now, ra8_rtc_datetime_t* out)
143{
144 *out = *now;
145 const uint16_t s = (uint16_t)now->second + (uint16_t)k_lpm_dpsby3_alarm_offset_s;
146 out->second = (uint8_t)(s % (uint16_t)k_lpm_dpsby3_secs_per_min);
147 const uint16_t carm = (uint16_t)(s / (uint16_t)k_lpm_dpsby3_secs_per_min);
148 const uint16_t m = (uint16_t)now->minute + carm;
149 out->minute = (uint8_t)(m % (uint16_t)k_lpm_dpsby3_mins_per_hour);
150 const uint16_t carh = (uint16_t)(m / (uint16_t)k_lpm_dpsby3_mins_per_hour);
151 const uint16_t h = (uint16_t)now->hour + carh;
152 out->hour = (uint8_t)(h % (uint16_t)k_lpm_dpsby3_hours_per_day);
153}

175[[nodiscard]] static ra8_err_t lpm_dpsby3_arm_wake(const ra8_rtc_datetime_t* now)
176{
177 ra8_rtc_datetime_t a = {};
178 lpm_dpsby3_add_offset(now, &a);
179 ra8_err_t err = ra8_rtc_set_alarm(&a);
180 if (err != k_ra8_ok) {
181 return err;
182 }
183 err = ra8_rtc_set_irq_enable((uint8_t)k_ra8_rtc_irq_alarm);
184 if (err != k_ra8_ok) {
185 return err;
186 }
187 err = ra8_lpm_arm_dpsier(k_ra8_lpm_dpsier_idx_2, (uint8_t)k_ra8_lpm_dpsier2_drtcaie_mask);
188 if (err != k_ra8_ok) {
189 return err;
190 }
191 return ra8_lpm_arm_wupen0_bits((uint32_t)k_ra8_lpm_wupen0_rtcalm);
192}

196int32_t main(void)
197{
198 lpm_dpsby3_setup_or_halt();
199 ra8_isr_globals_enable();
200
201 /* Emit boot banner before the deep-standby entry so the HIL gate
202 * confirms the build + bring-up path even if the sub-clock crystal
203 * is silent (see SOSC caveat in the file header). */
204 (void)ra8_board_uart_console_write(k_lpm_dpsby3_boot_msg,
205 (size_t)(sizeof(k_lpm_dpsby3_boot_msg) - 1U));
206
207 while (1) {
208 ra8_rtc_datetime_t now = {};
209 if (ra8_rtc_get(&now) != k_ra8_ok) {
210 break;
211 }
212 if (lpm_dpsby3_arm_wake(&now) != k_ra8_ok) {
213 break;
214 }
215 /* On hardware this never returns -- Deep Software Standby resets
216 * the CPU on wake, so control re-enters Reset_Handler. On the
217 * host (RA8_OFF_TARGET) WFI is a no-op so the loop simply
218 * iterates. */
219 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_deep_standby_3) != k_ra8_ok) {
220 break;
221 }
222 }
223 lpm_dpsby3_panic_halt();
224 return 0;
225}
