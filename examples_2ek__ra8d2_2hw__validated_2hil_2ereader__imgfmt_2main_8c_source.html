93static uint32_t ef_framebuffer_hash(void)
94{
95 const uint8_t* p = (const uint8_t*)s_framebuffer;
96 const size_t n = sizeof(s_framebuffer);
97 uint32_t hsh = (uint32_t)k_ef_fnv_offset;
98 for (size_t i = 0U; i < n; i++) {
99 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_ef_fnv_prime;
100 }
101 return hsh;
102}

105static void ef_print_hex(uint32_t value)
106{
107 uint8_t buf[k_ef_hex_nibbles];
108 for (uint32_t i = 0U; i < (uint32_t)k_ef_hex_nibbles; i++) {
109 const uint32_t shift = ((uint32_t)k_ef_hex_nibbles - 1U - i) * (uint32_t)k_ef_nibble_bits;
110 const uint32_t nib = (value >> shift) & (uint32_t)k_ef_nibble_mask;
111 buf[i] = (uint8_t)((nib < (uint32_t)k_ef_dec_ten) ? ('0' + nib) : ('A' + (nib - k_ef_dec_ten)));
112 }
113 ef_print(buf, (uint32_t)k_ef_hex_nibbles);
114}

150ef_decode_and_hash(const uint8_t* bytes, size_t len, const uint8_t* fail_msg, uint32_t fail_len)
151{
152 (void)ra8_gfx_clear((uint32_t)k_ef_col_bg);
153 ra8_img_arena_t arena = {.base = s_img_arena,
154 .cap = (size_t)k_ef_arena_bytes,
155 .offset = 0U,
156 .live = 0U};
157 int32_t out_w = 0;
158 int32_t out_h = 0;
159 if (ra8_img_decode_blit(&arena,
160 bytes,
161 len,
162 0,
163 0,
164 (int32_t)k_ef_fb_w,
165 (int32_t)k_ef_fb_h,
166 &out_w,
167 &out_h) != k_ra8_ok) {
168 ef_panic_halt(fail_msg, fail_len);
169 }
170 return ef_framebuffer_hash();

196 }
197
198 const uint32_t bmp_crc =
199 ef_decode_and_hash(k_bmp, (size_t)k_bmp_len, k_msg_dbmp, (uint32_t)sizeof(k_msg_dbmp) - 1U);
200 const uint32_t gif_crc =
201 ef_decode_and_hash(k_gif, (size_t)k_gif_len, k_msg_dgif, (uint32_t)sizeof(k_msg_dgif) - 1U);
202
203 ef_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
204 ef_print_hex(bmp_crc);
205 ef_print(k_msg_gif, (uint32_t)sizeof(k_msg_gif) - 1U);
206 ef_print_hex(gif_crc);
207 ef_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
208
209 while (1) {
210 __asm__ volatile("wfi");
211 }
212}
