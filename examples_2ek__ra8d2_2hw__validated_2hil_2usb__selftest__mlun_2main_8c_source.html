387static UINT mlun_msc_read(VOID* storage,
388 ULONG lun,
389 UCHAR* data_pointer,
390 ULONG number_blocks,
391 ULONG lba,
392 ULONG* media_status)
393{
394 (void)storage;
395 s_dbg_read_calls++;
396 if ((lba + number_blocks) > (ULONG)k_mlun_sectors) {
397 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
398 k_scsi_asc_lba_out_of_range,
399 k_scsi_ascq_none);
400 return UX_ERROR;
401 }
402 for (ULONG i = 0UL; i < number_blocks; i++) {
403 mlun_pattern_fill((uint32_t)lun,
404 (uint32_t)(lba + i),
405 &data_pointer[i * (ULONG)k_mlun_block_size]);
406 }
407 *media_status = 0UL;
408 (void)ra8_board_led_toggle(k_ra8_board_led1);
409 return UX_SUCCESS;
410}

614static VOID mlun_device_worker(ULONG arg)
615{
616 (void)arg;
617
618 UINT ux = mlun_usbx_stack_up();
619 if (ux != UX_SUCCESS) {
620 s_dbg_dev_err = (uint32_t)ux;
621 return;
622 }
623 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_stack;
624 ux = mlun_class_register();
625 if (ux != UX_SUCCESS) {
626 s_dbg_dev_err = (uint32_t)ux;
627 return;
628 }
629 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_class;
630 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
631 if (e != k_ra8_ok) {
632 s_dbg_dev_err = (uint32_t)e;
633 return;
634 }
635 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_dcd;
636 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
637 if (e != k_ra8_ok) {
638 s_dbg_dev_err = (uint32_t)e;
639 return;
640 }
641 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_attach;
642
643 while (1) {
644 s_dbg_dev_step = (uint32_t)k_mlun_dev_step_parked;
645 tx_thread_sleep(k_mlun_idle_ticks);
646 }
647}
