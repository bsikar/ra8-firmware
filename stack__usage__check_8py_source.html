4"""Aggregate gcc ``-fstack-usage`` .su files into a project-wide stack report.
5
6Background
7----------
8
9NASA Power-of-10 Rule 3 forbids dynamic memory allocation after init.
10The silent equivalent is unbounded stack growth: a deep call chain or a
11single oversized frame can blow past the linker-reserved stack region
12and corrupt adjacent RAM. IEC 61508 SIL 3 and DO-178C Level B both
13require the project to *demonstrate* a worst-case stack bound, not
14merely assume one.
15
16When compiled with ``-fstack-usage``, gcc emits a ``<file>.su`` file
17next to each ``.o``. Each line has the form::
18
19 path/to/file.c:LINE:COL:function_name<TAB>FRAME_BYTES<TAB>QUALIFIER
20
21where QUALIFIER is one of ``static``, ``dynamic``, ``bounded``, or
22combinations like ``dynamic,bounded``.
23
24What this script does
25---------------------
26
271. Walks every ``examples/*/*/build*/**/*.su`` file.
282. For each function: records TU, function name, frame size, qualifier.
293. Flags any function with frame > ``--frame-limit`` (default 2048) or
30 with a ``dynamic`` qualifier.
314. Writes ``build/stack_usage.csv`` and a per-app
32 ``build/stack_usage_<app>.txt``.
335. Exits non-zero if a critical-path module
34 (ra8_isr, ra8_check, ra8_err, ra8_mpu, ra8_cgc, ra8_pfs) has any function
35 with frame > 256 bytes or any ``dynamic`` qualifier anywhere.
36
37This script is report-only by default for the rest of the codebase --
38the per-target ``-Wstack-usage=N`` warning (in cmake/ra8_warnings.cmake)
39is the build-time gate; this script is the project-wide aggregator.
40"""
41

63# --- Enumeration floor (issue #386) ------------------------------------------
64#
65# A scan that parsed ZERO functions -- no .su files at all, or .su files that
66# decode to nothing -- used to exit 0, byte-for-byte identical to a genuine
67# clean pass. A build-flag change, a moved build directory, or a compiler swap
68# silently produces zero .su files; the gate then reported success over a stack
69# budget it never measured. On a bare-metal target a blown frame is memory
70# corruption rather than an exception, so this is the highest-consequence
71# instance of the "green while measuring nothing" defect this tree keeps
72# finding, and it left every RA8_MAX_STACK annotation unverified while it
73# appeared enforced.
74#
75# The floor makes a degraded enumeration read as FAILURE, never as an
76# improvement -- the same shape as check_lint_coverage.py's FILE_FLOOR and
77# check_annotations.py's MIN_CALL_RESOLUTION. Below it the gate exits non-zero
78# and says the sweep collapsed.
79#
80# Calibrated against a full cross-build: the `build-cross` gate runs
81# scripts/builders/all_examples.sh (218 apps at time of writing), then the
82# `stack-usage` gate aggregates its .su output. That build parses 90,140
83# first-party-plus-SOUP functions from 7552 .su files. The floor sits far below
84# that (~5.5%) so any legitimate build clears it with an 18x margin, while the
85# collapse modes above -- which drive the count to zero or a handful -- trip it
86# unambiguously. --allow-empty (the pre-commit path) skips the floor, because a
87# single-app local build legitimately parses far fewer than a full sweep.
88DEFAULT_MIN_FUNCTIONS = 5000
89
90# Process exit codes. Distinct so a caller (and a reader of the CI log) can tell
91# "the code has an over-budget frame" from "this gate could not measure its
92# subject", mirroring the convention in check_lint_coverage.py.
93RC_OK = 0
94RC_VIOLATION = 1
95RC_ENUMERATION_BROKE = 2

97# --- Strict-mode partitioning -------------------------------------------------
98#
99# In `--strict` mode the soft per-app frame limit is promoted to a hard
100# gate, but only for *first-party* code (everything outside
101# THIRD_PARTY_PATH_FRAGMENTS). Third-party SOUP is exempt because:
102#
103# * It is pre-qualified and lives under libs/third_party/ -- its
104# justification documents are filed under docs/SOUP/ (see
105# docs/SOUP/README.md).
106# * The largest current offenders (miniz mz_zip_reader_*,
107# tinfl_decompress_mem_to_*, ~10 kB frames) are deflate / zip
108# decoder helpers that are only invoked from the ereader app's
109# dedicated worker thread, which carries a generously-sized stack.
110#
111# The FIRST_PARTY_EXEMPTIONS list is the explicit, documented escape
112# hatch for first-party functions that have a justified large frame.
113# Add a tuple `(tu_substring, function_name, max_bytes, rationale)` per
114# entry; the gate will accept any first-party frame at or below
115# `max_bytes` for that function. Empty list today means *no* first-
116# party function is exempt.
117
118THIRD_PARTY_PATH_FRAGMENTS = (
119 "/libs/third_party/",
120 # Vendor-supplied port shims live under port/ but mostly call into
121 # third_party/ libraries; they remain first-party and gated.
122)
123
124FIRST_PARTY_EXEMPTIONS = (
125 # ("tu_substring", "function_name", max_bytes, "rationale"),
126 #
127 # Example template -- intentionally empty today (every first-party
128 # function is currently under the 2048-byte default limit, see
129 # `python3 scripts/checks/stack_usage_check.py --strict` against
130 # HEAD as of the commit that added this list):
131 #
132 # (
133 # "libs/ra8_epub/src/ra8_epub_open.c",
134 # "priv_parse_archive",
135 # 4344,
136 # "ZIP central-directory parser: 4 kB scratch struct on stack "

143def is_first_party(tu_path: str) -> bool:
144 """Whether a .su translation-unit path is ours rather than vendored SOUP.
145
146 Decided by path fragment against the vendored trees, because the TU path
147 inside a .su is whatever the compiler was given and is not reliably
148 repo-relative -- a prefix test would miss an absolute one.
149 """
150 return not any(frag in tu_path for frag in THIRD_PARTY_PATH_FRAGMENTS)

153def exemption_for(tu_path: str, func: str) -> int:
154 """Return the exempt max-bytes for a first-party (tu, func) pair, or 0."""
155 for tu_frag, fn_name, max_bytes, _why in FIRST_PARTY_EXEMPTIONS:
156 if tu_frag in tu_path and fn_name == func:
157 return max_bytes
158 return 0

167class StackEntry:
168 """One function's stack frame, as reported by a single line of a .su file.
169
170 ``tu`` is the path the COMPILER recorded, not where the .su was found. The
171 two diverge for the shared-library archive, and the gates key off ``tu``
172 precisely so that a critical-path frame is caught wherever it was built.
173 """
174
175 __slots__ = ("app", "bytes_", "func", "qualifier", "su_file", "tu")
176
177 def __init__( # noqa: PLR0913 # data class init, all fields are required
178 self,
179 app: str,
180 tu: str,
181 func: str,
182 bytes_: int,
183 qualifier: str,
184 su_file: Path,
185 ) -> None:
186 """Record one parsed .su line; every field is required and none is derived."""
187 self.app = app
188 self.tu = tu
189 self.func = func
190 self.bytes_ = bytes_
191 self.qualifier = qualifier
192 self.su_file = su_file
193
194 @property
195 def is_dynamic(self) -> bool:
196 """Whether gcc reported this frame as dynamically sized.
197
198 The qualifier is a comma-separated set, so it is split and matched as a
199 whole token rather than by substring -- a compound qualifier must be
200 decomposed, not pattern-matched.
201
202 ``dynamic,bounded`` counts as dynamic here even though gcc could bound
203 it. That is the conservative reading and it is deliberate: a bounded
204 dynamic frame still has no single compile-time size to put in a
205 worst-case stack budget, so it is surfaced for a human to judge.
206 """
207 return "dynamic" in self.qualifier.split(",")
208
209 @property
210 def is_critical(self) -> bool:
211 """Whether this frame belongs to a module held to the tighter 256-byte limit.
212
213 Matches the module name in the TU path OR in the function name, since a
214 critical-path function can be defined in a differently-named TU; either
215 hit is enough to pull the frame under the stricter budget.
216 """
217 return any(m in self.tu for m in CRITICAL_MODULES) or any(
218 m in self.func for m in CRITICAL_MODULES
219 )

222# The CI cross-build fast path (scripts/builders/all_examples.sh) compiles the
223# universal first-party library set (ra8_core / ra8_hal / ra8_net_pal /
224# ra8_usb_pal / board / secure_app) ONCE into a static archive under
225# build/shared_libs/ instead of recompiling it into every app, so those
226# sources' .su files land there rather than under examples/<app>/build/. That
227# archive holds the critical-path modules (ra8_isr / ra8_check / ra8_err /
228# ra8_mpu / ra8_cgc / ra8_pfs), so it MUST be aggregated too -- otherwise the
229# gate would silently lose its whole shared-library surface. See
230# cmake/ra8_shared_libs.cmake.
231SHARED_LIB_BUILD_SUBDIR = "build/shared_libs"
232SHARED_LIB_APP_NAME = "ra8_shared"

235def find_su_files(repo_root: Path) -> list:
236 """Collect every gcc ``.su`` file the cross-build emits.
237
238 Two roots:
239
240 * ``examples/`` -- each app's per-app CMake output lives in ``<app>/
241 build*/``, nested 2-4 levels below ``examples/`` (``examples/<tier>/
242 .../<app>/``). A ``.su`` is only ever emitted inside such a build tree,
243 so we recurse to any depth rather than assume a fixed layout. This also
244 covers the ineligible apps (TrustZone, non-ek board) that the fast path
245 still compiles from source per-app.
246
247 * ``build/shared_libs/`` -- the prebuilt universal-library archive the
248 fast path compiles once (see ``SHARED_LIB_BUILD_SUBDIR``). Absent on a
249 per-app-only build, so this is additive and never regresses the old
250 examples-only sweep.
251 """
252 found: list = []
253 examples = repo_root / "examples"
254 if examples.is_dir():
255 found.extend(p for p in examples.rglob("*.su") if p.is_file())
256 shared = repo_root / SHARED_LIB_BUILD_SUBDIR
257 if shared.is_dir():
258 found.extend(p for p in shared.rglob("*.su") if p.is_file())
259 return sorted(found)

262def app_name_for(su_file: Path, repo_root: Path) -> str:
263 """Attribute a .su file to the app whose build tree produced it.
264
265 Used only for GROUPING the per-app reports. The critical-path and
266 first-party gates deliberately key off the TU path recorded inside the .su
267 instead, so a shared-library frame is still judged correctly despite being
268 filed under a synthetic app name.
269
270 Returns "unknown" rather than raising for a .su outside both known roots:
271 an unattributable file should still appear in the aggregate report.
272 """
273 # Archive .su files live under build/shared_libs/, not examples/. Attribute
274 # them to a synthetic app so the per-app report groups them; the
275 # critical-path and first-party gates key off the real TU path (e.g.
276 # libs/ra8_hal/src/ra8_isr.c), which is preserved inside the .su, so those
277 # gates still fire on shared-library frames.
278 try:
279 su_file.relative_to(repo_root / SHARED_LIB_BUILD_SUBDIR)
280 except ValueError:
281 pass
282 else:
283 return SHARED_LIB_APP_NAME
284 try:
285 rel = su_file.relative_to(repo_root / "examples")
286 except ValueError:
287 return "unknown"
288 parts = rel.parts
289 # The app directory is the path component immediately preceding the
290 # first `build*` segment: examples/<tier>/.../<app>/build*/... .
291 for i, part in enumerate(parts):
292 if part.startswith("build"):
293 return parts[i - 1] if i > 0 else "unknown"
294 return parts[0] if parts else "unknown"

297def parse_su(su_file: Path, app: str) -> list:
298 """Parse one .su file into StackEntry records, skipping anything unrecognised.
299
300 Every failure mode here is a silent skip -- an unreadable file, a line that
301 does not match the .su grammar, a byte count that is not an integer. That
302 tolerance suits a format gcc owns and may extend, but note the consequence:
303 if the .su grammar ever changed wholesale this would return no entries, and
304 ``main`` reports a parse of zero functions as success. The gate would go
305 quiet rather than fail. Nothing currently detects that.
306 """
307 entries = []
308 try:
309 text = su_file.read_text(encoding="utf-8", errors="replace")
310 except OSError:
311 return entries
312 for raw_line in text.splitlines():
313 line = raw_line.strip()
314 if not line:
315 continue
316 match = _SU_LINE.match(line)
317 if not match:
318 continue
319 try:
320 nbytes = int(match.group("bytes"))
321 except ValueError:
322 continue
323 entries.append(
324 StackEntry(
325 app,
326 match.group("path"),
327 match.group("func"),
328 nbytes,
329 match.group("qual"),
330 su_file,
331 )
332 )
333 return entries

336def write_csv(out_path: Path, entries: list) -> None:
337 """Write the flat machine-readable report of every parsed frame.
338
339 Written with ``newline=""`` as the csv module requires, and in ASCII so a
340 non-ASCII symbol name fails here rather than producing a file the rest of
341 the ASCII-only toolchain cannot read.
342 """
343 out_path.parent.mkdir(parents=True, exist_ok=True)
344 with out_path.open("w", encoding="ascii", newline="") as fh:
345 writer = csv.writer(fh)
346 writer.writerow(["app", "tu", "function", "frame_bytes", "qualifier"])
347 for e in entries:
348 writer.writerow([e.app, e.tu, e.func, e.bytes_, e.qualifier])

351def write_per_app(out_dir: Path, entries: list) -> None:
352 """Write one human-readable report per app, biggest frame first.
353
354 Sorted descending because these files are read to answer "what should I
355 shrink?", so the answer belongs at the top rather than after a scroll.
356 """
357 by_app = defaultdict(list)
358 for e in entries:
359 by_app[e.app].append(e)
360 out_dir.mkdir(parents=True, exist_ok=True)
361 for app, rows in sorted(by_app.items()):
362 rows.sort(key=lambda r: r.bytes_, reverse=True)
363 path = out_dir / f"stack_usage_{app}.txt"
364 with path.open("w", encoding="ascii") as fh:
365 fh.write(f"# stack usage report for {app}\n")
366 fh.write(f"# {len(rows)} functions analysed\n")
367 fh.write("# columns: bytes qualifier function tu\n\n")
368 for r in rows:
369 fh.write(f"{r.bytes_:>8} {r.qualifier:<16} {r.func} {r.tu}\n")

372def find_violations(entries: list, frame_limit: int) -> tuple[list, list]:
373 """Partition entries into critical-path and soft breaches.
374
375 The two buckets are mutually exclusive by construction: a critical-path
376 frame is judged ONLY against the 256-byte limit and never also counted as
377 a soft breach, so one oversized function cannot be reported twice.
378
379 Returns ``(critical, soft)``.
380 """
381 critical = []
382 soft = []
383 for e in entries:
384 if e.is_critical:
385 if e.bytes_ > CRITICAL_FRAME_LIMIT or e.is_dynamic:
386 critical.append(e)
387 elif e.bytes_ > frame_limit or e.is_dynamic:
388 soft.append(e)
389 return critical, soft

394

494def _report_strict(soft: list, frame_limit: int) -> int:
495 """Promote a first-party soft violation into a hard failure.
496
497 Third-party SOUP stays exempt; a first-party function with a justified
498 large frame must be enrolled in FIRST_PARTY_EXEMPTIONS with a rationale.
499 """
500 offenders = []
501 for e in soft:
502 if not is_first_party(e.tu):
503 continue
504 allowed = exemption_for(e.tu, e.func)
505 if allowed > 0 and e.bytes_ <= allowed:
506 continue
507 offenders.append(e)
508 if not offenders:
509 return 0
510 print(
511 f"\nSTRICT first-party violations: "
512 f"{len(offenders)} function(s) exceed "
513 f"{frame_limit} bytes and are not exempt:"
514 )
515 for e in sorted(offenders, key=lambda r: r.bytes_, reverse=True):
516 print(f" [{e.app}] {e.func} ({e.tu}): {e.bytes_} bytes [{e.qualifier}]")
517 print(
518 "\nFix: either reduce the frame size (move scratch "
519 "buffers to module-static storage) or enroll the "
520 "function in FIRST_PARTY_EXEMPTIONS at the top of "
521 "scripts/checks/stack_usage_check.py with a written "
522 "rationale."
523 )
524 return 1

538def _report_dynamic(all_entries: list) -> int:
539 """NASA P10 Rule 3: a `dynamic` qualifier is a VLA or alloca.
540
541 Forbidden in this firmware regardless of frame size, in any module, and
542 the gate applies even in --warn-only mode. No deviation procedure.
543 """
544 dyn = [e for e in all_entries if e.is_dynamic]
545 if not dyn:
546 return 0
547 print(
548 f"\nNASA P10 Rule 3 violation: "
549 f"{len(dyn)} function(s) carry a `dynamic` qualifier "
550 f"(VLA or alloca). Forbidden in this firmware -- "
551 f"replace with a fixed-size buffer or an enum-bounded "
552 f"static array."
553 )
554 for e in sorted(dyn, key=lambda r: r.bytes_, reverse=True):
555 print(f" [{e.app}] {e.func} ({e.tu}): {e.bytes_} bytes [{e.qualifier}]")
556 return 1

578def _check_enumeration(
579 su_count: int, func_count: int, min_functions: int, allow_empty: bool
580) -> int:
581 """Decide whether the sweep measured enough to trust its verdict (#386).
582
583 A gate that examined nothing must FAIL, not pass. The three collapse modes
584 are graded distinctly:
585
586 * NO .su files -- an unbuilt or moved build tree. Under ``--allow-empty``
587 (the pre-commit path) this is a fresh clone and is tolerated; otherwise
588 it is a build that never ran and fails.
589 * .su files that decode to ZERO functions -- corruption or a grammar the
590 parser no longer understands. This is never a fresh clone, so it fails
591 even under ``--allow-empty``.
592 * FEWER functions than the floor -- a partially collapsed enumeration. A
593 run reporting far fewer results than a real build must not read as an
594 improvement, so it fails; ``--allow-empty`` skips the floor because a
595 single-app local build legitimately parses fewer.
596
597 Returns ``RC_OK`` when the sweep is trustworthy, ``RC_ENUMERATION_BROKE``
598 otherwise (after printing the reason to stderr).
599 """
600 if su_count == 0:
601 if allow_empty:
602 print(
603 "stack_usage_check: no .su files found; nothing built with "
604 "-fstack-usage yet -- allowed under --allow-empty.",
605 file=sys.stderr,
606 )
607 return RC_OK
608 print(
609 "stack_usage_check: FAIL -- no .su files found under examples/*/*/build*/ "
610 "or build/shared_libs/.\n"
611 " The build never ran or its output moved: there is no stack budget to "
612 "measure, so this is a failure, not a pass.\n"
613 " Run the build-cross gate first (it compiles every app with "
614 "-fstack-usage).",
615 file=sys.stderr,
616 )
617 return RC_ENUMERATION_BROKE
618
619 if func_count == 0:
620 print(
621 f"stack_usage_check: FAIL -- found {su_count} .su file(s) but parsed 0 "
622 "functions.\n"
623 " .su files present that decode to nothing is corruption or a changed "
624 ".su grammar, not a fresh clone; the stack budget went unmeasured.",
625 file=sys.stderr,
626 )
627 return RC_ENUMERATION_BROKE
628
629 if not allow_empty and func_count < min_functions:
630 print(
631 f"stack_usage_check: FAIL -- parsed only {func_count} function(s) from "
632 f"{su_count} .su file(s), below the floor of {min_functions}.\n"
633 " The enumeration collapsed (a partial build, a moved build dir, or a "
634 "compiler swap). A degraded sweep reporting fewer frames must not read as "
635 "a pass.",
636 file=sys.stderr,
637 )
638 return RC_ENUMERATION_BROKE
639
640 return RC_OK

643def main(argv: list) -> int:
644 """Aggregate every .su file, write the reports, and gate on the violations.
645
646 A green run is only evidence that stack budgets were checked when the sweep
647 actually parsed functions. Every no-input path is therefore graded by
648 ``_check_enumeration`` and prints the file/function census: an empty sweep
649 or one collapsed below ``--min-functions`` FAILS with
650 ``RC_ENUMERATION_BROKE`` rather than passing vacuously (#386). The only
651 tolerated empty case is ``--allow-empty`` with no .su files at all -- the
652 pre-commit fresh-clone path, where forcing a full build inside the hook
653 would cost minutes per commit.
654
655 Severity is tiered rather than uniform. Critical-path modules breach at 256
656 bytes, everything else at ``--frame-limit`` (2048), and a ``dynamic``
657 qualifier anywhere is a P10 Rule 3 violation regardless of size --
658 that last check runs on ALL entries and is the one thing ``--warn-only``
659 cannot downgrade.
660
661 Returns ``RC_ENUMERATION_BROKE`` (2) when the sweep measured too little to
662 trust, ``RC_VIOLATION`` (1) on a strict-mode soft breach, an un-waived
663 critical breach, or any dynamic frame, and ``RC_OK`` (0) otherwise.
664 """
665 args = _build_parser().parse_args(argv)
666
667 if args.selftest:
668 return selftest()
669
670 repo_root = args.repo_root.resolve()
671 su_files = find_su_files(repo_root)
672 all_entries = _collect_entries(repo_root, su_files)
673
674 su_count = len(su_files)
675 func_count = len(all_entries)
676 _print_census(su_count, func_count)
677
678 verdict = _check_enumeration(su_count, func_count, args.min_functions, args.allow_empty)
679 if verdict != RC_OK:
680 return verdict
681 if su_count == 0:
682 # --allow-empty with a fresh clone: nothing to measure, nothing to gate.
683 return RC_OK
684
685 out_dir = repo_root / "build"
686 write_csv(out_dir / "stack_usage.csv", all_entries)
687 write_per_app(out_dir, all_entries)
688
689 critical, soft = find_violations(all_entries, args.frame_limit)
690
691 if not args.quiet:
692 print(f" CSV: {out_dir / 'stack_usage.csv'}")
693 print(f" per-app: {out_dir}/stack_usage_<app>.txt")
694 print_top_n(all_entries, args.top)
695
696 if soft:
697 print(
698 f"\nSOFT violations: {len(soft)} function(s) over {args.frame_limit} bytes or dynamic:"
699 )
700 for e in sorted(soft, key=lambda r: r.bytes_, reverse=True):
701 print(f" [{e.app}] {e.func} ({e.tu}): {e.bytes_} bytes [{e.qualifier}]")
702
703 if args.strict and _report_strict(soft, args.frame_limit):
704 return RC_VIOLATION
705
706 if critical:
707 _report_critical(critical)
708 if not args.warn_only:
709 return RC_VIOLATION
710
711 return _report_dynamic(all_entries)

720def selftest() -> int:
721 """Assert the gate fires in BOTH directions, so a collapse cannot pass clean.
722
723 Builds throwaway .su fixtures in a temp tree and drives ``main`` against
724 them. A vacuous selftest would defeat the whole point of #386, so each
725 direction is asserted against a real return code:
726
727 * an empty sweep (no .su files, no --allow-empty) FAILS;
728 * .su files that decode to zero functions FAIL even under --allow-empty;
729 * a sweep below the floor FAILS, but the same fixture PASSES once the floor
730 is lowered -- proving the floor, not something else, is what tripped;
731 * an over-budget first-party frame FAILS under --strict;
732 * a genuine clean run PASSES.
733
734 Returns ``RC_OK`` when every direction behaves, ``RC_VIOLATION`` otherwise.
735 """
736 failures: list = []
737
738 def expect_red(got: int, label: str) -> None:
739 if got == RC_OK:
740 failures.append(f"{label}: expected non-zero, got {got}")
741
742 def expect_green(got: int, label: str) -> None:
743 if got != RC_OK:
744 failures.append(f"{label}: expected 0, got {got}")
745
746 with tempfile.TemporaryDirectory() as tmp:
747 root = Path(tmp)
748 build = root / "examples" / "app" / "build"
749 low = ["--strict", "--min-functions", "1"]
750 clean_line = "libs/ra8_epub/src/ok.c:10:1:small_fn\t128\tstatic"
751
752 # 1. Empty sweep -- no .su anywhere. Must FAIL without --allow-empty ...
753 expect_red(main(["--repo-root", str(root)]), "empty sweep")
754 # ... and be tolerated WITH --allow-empty (fresh clone).
755 expect_green(main(["--repo-root", str(root), "--allow-empty"]), "empty sweep, allow-empty")
756
757 # 2. .su present but decoding to zero functions -- corruption. Must FAIL
758 # even under --allow-empty.
759 _write_su_fixture(build, "junk.su", ["not a stack-usage line at all", ""])
760 expect_red(
761 main(["--repo-root", str(root), "--allow-empty"]),
762 "zero functions, allow-empty",
763 )
764
765 # 3. One clean function. Below the default floor -> FAIL on the floor ...
766 _write_su_fixture(build, "junk.su", [clean_line])
767 expect_red(main(["--repo-root", str(root), "--strict"]), "below floor")
768 # ... but PASS once the floor is lowered to admit it (isolates the floor).
769 expect_green(main(["--repo-root", str(root), *low]), "clean run, floor=1")
770
771 # 4. An over-budget first-party frame must FAIL under --strict even with
772 # the floor satisfied.
773 _write_su_fixture(
774 build,
775 "over.su",
776 [clean_line, "libs/ra8_epub/src/big.c:20:1:priv_scratch\t9000\tstatic"],
777 )
778 expect_red(main(["--repo-root", str(root), *low]), "over-budget frame")
779
780 if failures:
781 print("SELFTEST FAILED:", file=sys.stderr)
782 for problem in failures:
783 print(f" - {problem}", file=sys.stderr)
784 return RC_VIOLATION
785 print("selftest: stack_usage_check empty-sweep + floor + over-budget detection OK")
786 return RC_OK
