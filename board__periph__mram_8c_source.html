196static uint64_t mram_reg_read(uc_engine* uc, uint64_t addr, unsigned size)
197{
198 (void)uc;
199 (void)size;
200 const uint64_t off = addr - (uint64_t)k_mram_reg_base;
201 if (off == (uint64_t)k_mram_off_mentryr) {
202 return (s_mram.pe_active != 0U) ? (uint64_t)k_mram_mentryr_pe_bit : 0U;
203 }
204 if (off == (uint64_t)k_mram_off_mstatr) {
205 return (uint64_t)k_mram_mstatr_mrdy; /* command complete, no error bits */
206 }
207 if ((off / 4U) >= (uint64_t)k_mram_reg_words) {
208 return 0U;
209 }
210 return s_mram.regs[off / 4U];
211}

249static void maci_commit(uc_engine* uc)
250{
251 if (s_mram.payload_len == 0U) {
252 return;
253 }
254 /* HUM Ch 59.7.4.5 Table 59.15 p 3592: the Program command only accepts the
255 * option-setting / OTP addresses in [k_mram_pgm_lo, k_mram_pgm_hi]. Anything
256 * else -- notably the 0x2700_0000 "extra-MRAM data area" that exists in no
257 * RA8D2 memory map -- is an illegal command on silicon. This model used to
258 * write it straight through to mapped RAM, which let ra8_io_mram_demo round
259 * trip in the emulator while the bench returned Error=516 (#170). Reject it
260 * here so the emulator reports the bench result. */
261 if (s_mram.msaddr < (uint32_t)k_mram_pgm_lo) {
262 maci_reject();
263 return;
264 }
265 if (s_mram.msaddr > (uint32_t)k_mram_pgm_hi) {
266 maci_reject();
267 return;
268 }
269 (void)uc_mem_write(uc, (uint64_t)s_mram.msaddr, s_mram.payload, (size_t)s_mram.payload_len);
270 s_mram.programs++;
271}

274static void maci_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
275{
276 (void)addr;
277 if (size == 2U) {
278 if (s_mram.maci_state == (uint8_t)k_maci_collect) {
279 if ((s_mram.payload_len + 2U) <= (uint32_t)k_mram_payload_max) {
280 s_mram.payload[s_mram.payload_len] = (uint8_t)(value & (uint64_t)k_mram_byte_mask);
281 s_mram.payload[s_mram.payload_len + 1U] =
282 (uint8_t)((value >> (uint64_t)k_mram_hi_shift) & (uint64_t)k_mram_byte_mask);
283 s_mram.payload_len += 2U;
284 }
285 }
286 return;
287 }
288 const uint32_t byte = (uint32_t)value & (uint32_t)k_mram_byte_mask;
289 /* Either opener (Program 0xE8 or Config-Set 0x40) starts a command; kept as
290 * two single-condition checks so the model carries no compound decision. */
291 if (byte == (uint32_t)k_maci_cmd_program) {
292 s_mram.maci_state = (uint8_t)k_maci_opener1;
293 return;
294 }
295 if (byte == (uint32_t)k_maci_cmd_config_set) {
296 s_mram.maci_state = (uint8_t)k_maci_opener1;
297 return;
298 }
299 if ((byte == (uint32_t)k_maci_cmd_word_n) && (s_mram.maci_state == (uint8_t)k_maci_opener1)) {
300 s_mram.maci_state = (uint8_t)k_maci_collect;
301 s_mram.payload_len = 0U;
302 return;
303 }
304 if ((byte == (uint32_t)k_maci_cmd_final) && (s_mram.maci_state == (uint8_t)k_maci_collect)) {
305 maci_commit(uc);
306 s_mram.maci_state = (uint8_t)k_maci_idle;
307 return;
308 }
309 s_mram.maci_state = (uint8_t)k_maci_idle; /* forced-stop / status-clear / unknown */
310}

322static uint64_t mrpgm_read(uc_engine* uc, uint64_t addr, unsigned size)
323{
324 (void)uc;
325 (void)size;
326 const uint64_t off = addr - (uint64_t)k_mrpgm_base;
327 if (off == (uint64_t)k_mrpgm_off_mrcps) {
328 return (uint64_t)k_mrcps_ready; /* always idle-and-ready: no MACI sequencing here. */
329 }
330 if ((off / 4U) >= (uint64_t)k_mrpgm_words) {
331 return 0U;
332 }
333 return s_mram.pgm_regs[off / 4U]; /* MRCPC0/1 / MRCFLR read back their written value. */
334}

337static void mrpgm_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
338{
339 (void)uc;
340 (void)size;
341 const uint64_t off = addr - (uint64_t)k_mrpgm_base;
342 /* MRCPC0/1 (program gate), MRCFLR (flush), MRCPS (W1C error clear): the mapped
343 * MRAM window already accepts the direct STR programming, so these only need
344 * to shadow so a gate read-back sees the key that was written. */
345 if ((off / 4U) < (uint64_t)k_mrpgm_words) {
346 s_mram.pgm_regs[off / 4U] = (uint32_t)value;
347 }
348}

357static void mram_report(void)
358{
359 if (s_mram.rejected != 0U) {
360 /* Loud: a Program aimed outside HUM Table 59.15's option-setting window
361 * is an illegal command on silicon (command-locked, MSTATR.ILGCOMERR). */
362 (void)fprintf(stderr,
363 " Extra-MRAM : %u MACI program commands, %u REJECTED "
364 "(MSADDR outside HUM Table 59.15 option-setting window)\n",
365 s_mram.programs,
366 s_mram.rejected);
367 return;
368 }
369 if (s_mram.programs == 0U) {
370 return;
371 }
372 (void)fprintf(stderr, " Extra-MRAM : %u MACI program commands\n", s_mram.programs);
373}
