209RA8_INTERNAL static uint64_t internal_mram_reg_read(uc_engine* uc, uint64_t addr, unsigned size)
210{
211 (void)uc;
212 (void)size;
213 const uint64_t off = addr - (uint64_t)k_mram_reg_base;
214 if (off == (uint64_t)k_mram_off_mentryr) {
215 return (s_mram.pe_active != 0U) ? (uint64_t)k_mram_mentryr_pe_bit : 0U;
216 }
217 if (off == (uint64_t)k_mram_off_mstatr) {
218 return (uint64_t)k_mram_mstatr_mrdy; /* command complete, no error bits */
219 }
220 if ((off / 4U) >= (uint64_t)k_mram_reg_words) {
221 return 0U;
222 }
223 return s_mram.regs[off / 4U];
224}

286RA8_INTERNAL static void internal_maci_commit(uc_engine* uc)
287{
288 if (s_mram.payload_len == 0U) {
289 return;
290 }
291 /* HUM Ch 59.7.4.5 Table 59.15 p 3592: the Program command only accepts the
292 * option-setting / OTP addresses in [k_mram_pgm_lo, k_mram_pgm_hi]. Anything
293 * else -- notably the 0x2700_0000 "extra-MRAM data area" that exists in no
294 * RA8D2 memory map -- is an illegal command on silicon. This model used to
295 * write it straight through to mapped RAM, which let ra8_io_mram_demo round
296 * trip in the emulator while the bench returned Error=516 (#170). Reject it
297 * here so the emulator reports the bench result. */
298 if (s_mram.msaddr < (uint32_t)k_mram_pgm_lo) {
299 internal_maci_reject();
300 return;
301 }
302 if (s_mram.msaddr > (uint32_t)k_mram_pgm_hi) {
303 internal_maci_reject();
304 return;
305 }
306 (void)emu_mem_write(uc, (uint64_t)s_mram.msaddr, s_mram.payload, (size_t)s_mram.payload_len);
307 s_mram.programs++;
308}

322RA8_INTERNAL static void
323internal_maci_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
324{
325 (void)addr;
326 if (size == 2U) {
327 if (s_mram.maci_state == (uint8_t)k_maci_collect) {
328 if ((s_mram.payload_len + 2U) <= (uint32_t)k_mram_payload_max) {
329 s_mram.payload[s_mram.payload_len] = (uint8_t)(value & (uint64_t)k_mram_byte_mask);
330 s_mram.payload[s_mram.payload_len + 1U] =
331 (uint8_t)((value >> (uint64_t)k_mram_hi_shift) & (uint64_t)k_mram_byte_mask);
332 s_mram.payload_len += 2U;
333 }
334 }
335 return;
336 }
337 const uint32_t byte = (uint32_t)value & (uint32_t)k_mram_byte_mask;
338 /* Either opener (Program 0xE8 or Config-Set 0x40) starts a command; kept as
339 * two single-condition checks so the model carries no compound decision. */
340 if (byte == (uint32_t)k_maci_cmd_program) {
341 s_mram.maci_state = (uint8_t)k_maci_opener1;
342 return;
343 }
344 if (byte == (uint32_t)k_maci_cmd_config_set) {
345 s_mram.maci_state = (uint8_t)k_maci_opener1;
346 return;
347 }
348 if ((byte == (uint32_t)k_maci_cmd_word_n) && (s_mram.maci_state == (uint8_t)k_maci_opener1)) {
349 s_mram.maci_state = (uint8_t)k_maci_collect;
350 s_mram.payload_len = 0U;
351 return;
352 }
353 if ((byte == (uint32_t)k_maci_cmd_final) && (s_mram.maci_state == (uint8_t)k_maci_collect)) {
354 internal_maci_commit(uc);
355 s_mram.maci_state = (uint8_t)k_maci_idle;
356 return;
357 }
358 s_mram.maci_state = (uint8_t)k_maci_idle; /* forced-stop / status-clear / unknown */
359}

395RA8_INTERNAL static uint64_t internal_mrpgm_read(uc_engine* uc, uint64_t addr, unsigned size)
396{
397 (void)uc;
398 (void)size;
399 const uint64_t off = addr - (uint64_t)k_mrpgm_base;
400 if (off == (uint64_t)k_mrpgm_off_mrcps) {
401 return (uint64_t)k_mrcps_ready; /* always idle-and-ready: no MACI sequencing here. */
402 }
403 if ((off / 4U) >= (uint64_t)k_mrpgm_words) {
404 return 0U;
405 }
406 return s_mram.pgm_regs[off / 4U]; /* MRCPC0/1 / MRCFLR read back their written value. */

421RA8_INTERNAL static void
422internal_mrpgm_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
423{
424 (void)uc;
425 (void)size;
426 const uint64_t off = addr - (uint64_t)k_mrpgm_base;
427 /* MRCPC0/1 (program gate), MRCFLR (flush), MRCPS (W1C error clear): the mapped
428 * MRAM window already accepts the direct STR programming, so these only need
429 * to shadow so a gate read-back sees the key that was written. */
430 if ((off / 4U) < (uint64_t)k_mrpgm_words) {
431 s_mram.pgm_regs[off / 4U] = (uint32_t)value;

456RA8_INTERNAL static void internal_mram_report(void)
457{
458 if (s_mram.rejected != 0U) {
459 /* Loud: a Program aimed outside HUM Table 59.15's option-setting window
460 * is an illegal command on silicon (command-locked, MSTATR.ILGCOMERR). */
461 (void)priv_emu_io_errf(" Extra-MRAM : %u MACI program commands, %u REJECTED "
462 "(MSADDR outside HUM Table 59.15 option-setting window)\n",
463 s_mram.programs,
464 s_mram.rejected);
465 return;
466 }
467 if (s_mram.programs == 0U) {
468 return;
469 }
470 (void)priv_emu_io_errf(" Extra-MRAM : %u MACI program commands\n", s_mram.programs);
471}
