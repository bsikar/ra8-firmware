270ra8_err_t ra8_i2c_internal_target_drain_rx(volatile r_i2c_regs_t* reg,
271 uint8_t* buf,
272 uint32_t capacity,
273 uint32_t* out_count)
274{
275 uint32_t count = 0U;
276 ra8_err_t err = k_ra8_ok;
277 for (uint32_t i = 0U; i <= capacity; i++) {
278 err = internal_i2c_target_wait(
279 reg,
280 (uint8_t)((uint8_t)k_ra8_i2c_msk_icsr2_rdrf | (uint8_t)k_ra8_i2c_msk_icsr2_stop));
281 if (err != k_ra8_ok) {
282 break;
283 }
284 /* HUM Ch 39.2.10 "ICSR2 : I2C Bus Status Register 2" p 2384 */
285 const uint8_t icsr2 = reg->ICSR2;
286 if (!ra8_i2c_internal_peripheral_rx_continue(icsr2, count, capacity)) {
287 /* STOP detected or buffer full: drain a final pending byte if room. */
288 if (((icsr2 & (uint8_t)k_ra8_i2c_msk_icsr2_rdrf) != 0U) && (count < capacity)) {
289 /* HUM Ch 39.2.18 "ICDRR : I2C Bus Receive Data Register" p 2393 */
290 buf[count] = reg->ICDRR;
291 count++;
292 }
293 break;
294 }
295 if ((icsr2 & (uint8_t)k_ra8_i2c_msk_icsr2_rdrf) != 0U) {
296 /* HUM Ch 39.2.18 "ICDRR : I2C Bus Receive Data Register" p 2393 */
297 buf[count] = reg->ICDRR;
298 count++;
299 }
300 }
301 *out_count = count;
302 return err;
303}

328static void internal_i2c_target_fill_tx(volatile r_i2c_regs_t* reg,
329 const uint8_t* data,
330 uint32_t len,
331 uint32_t* out_sent)
332{
333 uint32_t sent = 0U;
334 for (uint32_t i = 0U; i <= len; i++) {
335 /* Step 3 (peripheral transmit): wait for TDRE before each byte.
336 * HUM Ch 39.3.5 "Peripheral Transmit Operation" p 2405 */
337 if (internal_i2c_target_wait(reg, (uint8_t)k_ra8_i2c_msk_icsr2_tdre) != k_ra8_ok) {
338 break;
339 }
340 /* HUM Ch 39.2.10 "ICSR2 : I2C Bus Status Register 2" p 2384 */
341 const uint8_t icsr2 = reg->ICSR2;
342 if (!ra8_i2c_internal_peripheral_tx_continue(icsr2, sent, len)) {
343 break;
344 }
345 /* HUM Ch 39.2.17 "ICDRT : I2C Bus Transmit Data Register" p 2393 */
346 reg->ICDRT = data[sent];
347 sent++;
348 }
349 *out_sent = sent;
350}

374static bool internal_i2c_target_finish_tx(volatile r_i2c_regs_t* reg)
375{
376 /* Step 4: wait for the controller's TEND or NACK to mark frame end. */
377 (void)internal_i2c_target_wait(
378 reg,
379 (uint8_t)((uint8_t)k_ra8_i2c_msk_icsr2_tend | (uint8_t)k_ra8_i2c_msk_icsr2_nackf));
380 /* HUM Ch 39.2.10 "ICSR2 : I2C Bus Status Register 2" p 2384 */
381 const bool ended = ra8_i2c_internal_peripheral_tx_done(reg->ICSR2);
382
383 /* Step 5: dummy-read ICDRR to release SCL.
384 * HUM Ch 39.2.18 "ICDRR : I2C Bus Receive Data Register" p 2393 */
385 (void)reg->ICDRR;
386
387 /* Step 7: clear NACKF and STOP (W0C) for the next transfer.
388 * HUM Ch 39.2.10 "ICSR2 : I2C Bus Status Register 2 -- W0C" p 2384 */
389 reg->ICSR2 = (uint8_t)(reg->ICSR2 & (uint8_t)~(uint8_t)((uint8_t)k_ra8_i2c_msk_icsr2_nackf |
390 (uint8_t)k_ra8_i2c_msk_icsr2_stop));
391 return ended;
392}

400{
401 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
402 RA8_CHECK_NULL_PTR(reg, s_i2c_tag, "i2c_target_init: channel");

404 if (cfg->own_addr_7b > (uint8_t)k_ra8_i2c_peripheral_addr_7b_max) {
405 return k_ra8_err_invalid_arg;
406 }
407 if ((uint8_t)cfg->slot > (uint8_t)k_ra8_i2c_peripheral_slot_max) {
408 return k_ra8_err_invalid_arg;
409 }
410
411 const uint8_t slot = (uint8_t)cfg->slot;
412 internal_i2c_target_set_addr(reg, slot, cfg->own_addr_7b);

414 /* HUM Ch 39.2.7 "ICSER : I2C Bus Status Enable Register" p 2380 */
415 reg->ICSER = internal_i2c_target_icser_mask(slot, cfg->general_call);

419 reg->ICMR3 = (uint8_t)(reg->ICMR3 | (uint8_t)k_ra8_i2c_msk_icmr3_wait);

479ra8_err_t
480ra8_i2c_peripheral_receive(uint8_t channel, uint8_t* buf, uint32_t capacity, uint32_t* out_received)
481{
482 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
483 RA8_CHECK_NULL_PTR(reg, s_i2c_tag, "i2c_target_receive: channel");
484 RA8_CHECK_NULL_PTR(buf, s_i2c_tag, "i2c_target_receive: buf");
485 RA8_CHECK_NULL_PTR(out_received, s_i2c_tag, "i2c_target_receive: out_received");
486 *out_received = 0U;
487 if (capacity == 0U) {
488 return k_ra8_err_invalid_arg;
489 }
490 if (!s_i2c_state[channel].peripheral_active) {
491 return k_ra8_err_not_initialized;
492 }
493
494 /* Step 3 (peripheral receive): the address-phase RDRF, then dummy-read
495 * ICDRR to discard the matched address byte and clear RDRF.
496 * HUM Ch 39.3.6 "Peripheral Receive Operation" p 2408 */
497 ra8_err_t err = internal_i2c_target_wait(reg, (uint8_t)k_ra8_i2c_msk_icsr2_rdrf);
498 if (err != k_ra8_ok) {
499 return err;
500 }
501 /* HUM Ch 39.2.18 "ICDRR : I2C Bus Receive Data Register" p 2393 */
502 (void)reg->ICDRR;
503
504 /* Steps 4-5: drain data bytes until STOP or the buffer fills. */
505 uint32_t count = 0U;
506 err = ra8_i2c_internal_target_drain_rx(reg, buf, capacity, &count);
507
508 /* Step 6: clear the STOP flag (W0C) for the next transfer.
509 * HUM Ch 39.2.10 "ICSR2 : I2C Bus Status Register 2 -- STOP W0C" p 2384 */
510 reg->ICSR2 = (uint8_t)(reg->ICSR2 & (uint8_t)~(uint8_t)k_ra8_i2c_msk_icsr2_stop);
511 *out_received = count;
512 return (count == 0U) ? err : k_ra8_ok;
513}

515ra8_err_t
516ra8_i2c_peripheral_transmit(uint8_t channel, const uint8_t* data, uint32_t len, uint32_t* out_sent)
517{
518 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
519 RA8_CHECK_NULL_PTR(reg, s_i2c_tag, "i2c_target_transmit: channel");
520 RA8_CHECK_NULL_PTR(data, s_i2c_tag, "i2c_target_transmit: data");
521 RA8_CHECK_NULL_PTR(out_sent, s_i2c_tag, "i2c_target_transmit: out_sent");
522 *out_sent = 0U;
523 if (len == 0U) {
524 return k_ra8_err_invalid_arg;
525 }
526 if (!s_i2c_state[channel].peripheral_active) {
527 return k_ra8_err_not_initialized;
528 }
529
530 /* Step 3: push data bytes to ICDRT until NACK or every byte is queued. */
531 uint32_t sent = 0U;
532 internal_i2c_target_fill_tx(reg, data, len, &sent);
533
534 /* Steps 4-7: confirm the controller ended the read, release SCL, clear
535 * NACKF / STOP. ``ended`` is k_ra8_ok-equivalent: the end-of-frame poll only
536 * times out (ended == false) when no TEND / NACK ever arrives, which is the
537 * single failure mode for both an empty and a non-empty transmit. */
538 const bool ended = internal_i2c_target_finish_tx(reg);
539 *out_sent = sent;
540 return ended ? k_ra8_ok : k_ra8_err_hw_timeout;
541}
