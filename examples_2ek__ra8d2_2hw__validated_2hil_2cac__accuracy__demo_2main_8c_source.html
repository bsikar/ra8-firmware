197[[nodiscard]] static ra8_err_t cac_demo_configure(void)
198{
199 const uint32_t expected = cac_demo_expected();
200 const uint32_t tol = expected >> (uint32_t)k_cac_demo_tol_shift;
201 const uint16_t upper = (uint16_t)(expected + tol);
202 const uint16_t lower = (uint16_t)(expected - tol);
203
204 const ra8_err_t err = ra8_cac_init(upper, lower);
205 if (err != k_ra8_ok) {
206 return err;
207 }
208 /* HUM Ch 10.2.2 "CACR1 : CAC Control Register 1" p 421 -- target =
209 * main osc (FMCS=000), no division, rising edge. Valid while CFME=0. */
210 ra8_cac()->CACR1 = (uint8_t)k_cac_demo_cacr1_main;
211 /* HUM Ch 10.2.3 "CACR2 : CAC Control Register 2" p 422 -- internal
212 * reference = LOCO (RSCS=100) divided by 32 (RCDS=00), no filter. */
213 ra8_cac()->CACR2 = (uint8_t)k_cac_demo_cacr2_loco;
214 return k_ra8_ok;
215}

238[[nodiscard]] static ra8_err_t cac_demo_sample(uint8_t* out_ok)
239{
240 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
241
242 uint16_t count = 0U;
243 const ra8_err_t meas = ra8_cac_measure(&count);
244 const uint8_t meas_ok = (meas == k_ra8_ok) ? 1U : 0U;
245
246 uint8_t status = 0U;
247 const ra8_err_t serr = ra8_cac_get_status(&status);
248 if (serr != k_ra8_ok) {
249 return serr;
250 }
251 g_cac_status = (uint32_t)status;
252 if (meas_ok != 0U) {
253 g_cac_count = (uint32_t)count;
254 }
255 const uint8_t ferrf = ((status & (uint8_t)k_ra8_cac_status_ferrf) != 0U) ? 1U : 0U;
256 const uint8_t ovff = ((status & (uint8_t)k_ra8_cac_status_ovff) != 0U) ? 1U : 0U;
257 *out_ok = (meas_ok != 0U && ferrf == 0U && ovff == 0U) ? 1U : 0U;
258 return k_ra8_ok;
259}

263int32_t main(void)
264{
265 cac_demo_setup_or_halt();
266 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
267 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No NVIC sources
268 * are armed by this demo. */
269 ra8_isr_globals_enable();
270
271 if (cac_demo_configure() != k_ra8_ok) {
272 cac_demo_panic_halt();
273 }
274
275 while (1) {
276 uint8_t ok = 0U;
277 const ra8_err_t err = cac_demo_sample(&ok);
278 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
279 g_cac_ok = (uint32_t)good;
280 if (good != 0U) {
281 (void)ra8_board_uart_console_write(k_cac_demo_ok_msg,
282 (size_t)(sizeof(k_cac_demo_ok_msg) - 1U));
283 (void)ra8_board_led_toggle(k_ra8_board_led1);
284 } else {
285 (void)ra8_board_uart_console_write(k_cac_demo_bad_msg,
286 (size_t)(sizeof(k_cac_demo_bad_msg) - 1U));
287 (void)ra8_board_led_toggle(k_ra8_board_led2);
288 }
289 ++g_cac_heartbeat;
290 ra8_delay_ms(k_cac_demo_period_ms);
291 }
292 cac_demo_panic_halt();
293 return 0;
294}
