59static void on_scb_ctrl_write(uc_engine* uc,
60 uc_mem_type type,
61 uint64_t addr,
62 int size,
63 int64_t value,
64 void* user)
65{
66 (void)type;
67 (void)size;
68 (void)user;
69 const uint32_t word = (uint32_t)addr & ~3U;
70 if (word == (uint32_t)k_scb_ccr) {
71 if (((uint32_t)value & (uint32_t)k_ccr_div_0_trp) != 0U) {
72 div0_patch_sites(uc); /* firmware opted in: overwrite the divides with UDF. */
73 }
74 return;
75 }
76 if (word != (uint32_t)k_scb_aircr) {
77 return; /* an intervening SCB word (VTOR / SCR): not ours. */
78 }
79 if (((uint32_t)value & (1U << (uint32_t)k_aircr_sysresetreq)) == 0U) {
80 return; /* a non-reset AIRCR write (e.g. priority grouping) */
81 }
82 s_reboot_request = true;
83 (void)uc_emu_stop(uc); /* end the chunk; the run loop performs the reboot */
84}

118 (void)type;

121 const bool is_set = (addr >= (uint64_t)k_nvic_iser_base) &&
122 (addr < ((uint64_t)k_nvic_iser_base + (uint64_t)k_nvic_en_span));
123 const uint64_t base = is_set ? (uint64_t)k_nvic_iser_base : (uint64_t)k_nvic_icer_base;
124 const uint32_t word = (uint32_t)((addr - base) / 4U);
125 const uint32_t bits = (uint32_t)value;
126 for (uint32_t b = 0U; b < 32U; b++) {

128 board_periph_nvic_set_enable((word * 32U) + b, is_set);

185void dwt_cyccnt_advance(uc_engine* uc)
186{
187 const uint32_t demcr = rd32(uc, (uint64_t)k_scb_demcr_addr);
188 if ((demcr & (uint32_t)k_scb_demcr_trcena) == 0U) {
189 return; /* Trace subsystem off: CYCCNT does not count. */
190 }
191 const uint32_t ctrl = rd32(uc, (uint64_t)k_dwt_ctrl_addr);
192 if ((ctrl & (uint32_t)k_dwt_ctrl_cyccntena) == 0U) {
193 return; /* Cycle counter disabled: leave CYCCNT untouched. */
194 }
195 const uint32_t next = rd32(uc, (uint64_t)k_dwt_cyccnt_addr) + (uint32_t)k_dwt_cyccnt_per_chunk;
196 wr32(uc, (uint64_t)k_dwt_cyccnt_addr, next);
197}

200void emu_exc_install_scb_nvic(uc_engine* uc)
201{
202 /* Watch the SCB control words AIRCR..CCR with ONE hook (on_scb_ctrl_write): a
203 * SYSRESETREQ store triggers a warm reboot and a CCR.DIV_0_TRP store arms the
204 * div-0 trap (which then patches divides -- no per-access cost). Both are PPB
205 * RAM, so a write-hook is the only way to observe them. */
206 static uc_hook s_h_scb;
207 (void)uc_hook_add(uc,
208 &s_h_scb,
209 UC_HOOK_MEM_WRITE,
210 (void*)on_scb_ctrl_write,
211 nullptr,
212 (uint64_t)k_scb_aircr,
213 (uint64_t)k_scb_ccr + 3U);
214 /* NVIC ISER / ICER are set-enable / clear-enable: fold each written bit into
215 * board_periph's enable shadow so enabling several lines does not clobber the
216 * earlier ones (see on_nvic_en_write). The PPB is RAM, so this hook is the
217 * only place the W1S/W1C semantics can be applied. */
218 static uc_hook s_h_nvic_iser;
219 static uc_hook s_h_nvic_icer;
220 (void)uc_hook_add(uc,
221 &s_h_nvic_iser,
222 UC_HOOK_MEM_WRITE,
223 (void*)on_nvic_en_write,
224 nullptr,
225 (uint64_t)k_nvic_iser_base,
226 (uint64_t)k_nvic_iser_base + (uint64_t)k_nvic_en_span - 1U);
227 (void)uc_hook_add(uc,
228 &s_h_nvic_icer,
229 UC_HOOK_MEM_WRITE,
230 (void*)on_nvic_en_write,
231 nullptr,
232 (uint64_t)k_nvic_icer_base,
233 (uint64_t)k_nvic_icer_base + (uint64_t)k_nvic_en_span - 1U);
234}
