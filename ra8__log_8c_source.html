165RA8_INTERNAL static inline bool internal_itm_ready(void)
166{
167 /* A redirected byte sink needs no ITM/debugger -- it is always ready. */
168 if (s_byte_sink != nullptr) {
169 return true;
170 }
171#ifndef RA8_OFF_TARGET
172 /* Hardening: if DEMCR.TRCENA is clear the ITM block is powered down
173 * and any read of its registers (TCR, TENR, STIM) will bus-fault.
174 * Pre-check TRCENA in DEMCR (always accessible) before touching ITM.
175 *
176 * This was the root cause of escalation-to-LOCKUP observed during
177 * USB bring-up: the original USB fault entered the fault handler,
178 * which called this function, which read ITM_TCR with TRCENA=0 ->
179 * second fault -> LOCKUP at PC=0xEFFFFFFE, hiding the real PC. */
180 const uint32_t demcr = *(volatile uint32_t*)k_ra8_scb_demcr_addr;
181 if ((demcr & (uint32_t)k_ra8_demcr_trcena_mask) == 0U) {
182 return false;
183 }
184 /* Additional belt-and-braces: never poke ITM from a fault context.
185 * IPSR != 0 means we are inside an exception handler. Even if
186 * TRCENA happens to be on, dropping log lines is preferable to a
187 * second fault that masks the original PC. */
188 /* cppcheck-suppress unreadVariable -- the inline asm mrs writes IPSR into the variable; cppcheck cannot see through the asm. */
189 /* cppcheck-suppress knownConditionTrueFalse -- cppcheck assumes the asm-written IPSR value stays 0. */
190 volatile uint32_t ipsr = 0U;
191 __asm__ volatile("mrs %0, ipsr" : "=r"(ipsr));
192 if (ipsr != 0U) {
193 return false;
194 }
195#endif
196 const uint32_t tcr = *internal_itm_tcr();
197 const uint32_t tenr = *internal_itm_tenr();
198 /* TCR bit 0 = ITMENA. TENR bit 0 = stimulus port 0 enabled. */
199 if ((tcr & 1U) == 0U) {
200 return false;
201 }
202 if ((tenr & 1U) == 0U) {
203 return false;
204 }
205 return *internal_itm_stim0() != 0U;
206}

226RA8_INTERNAL static inline void internal_itm_putc(uint8_t byte)
227{
228 if (s_byte_sink != nullptr) {
229 s_byte_sink(s_byte_sink_ctx, byte);
230 return;
231 }
232 /* A real build should pre-check `internal_itm_ready()` once during
233 * init and skip the call entirely when the debugger is absent. For
234 * now we poll with a small bound so a disconnected debugger never
235 * stalls the firmware. */
236 enum : uint32_t {
237 k_ra8_itm_poll_limit = 1000U,
238 };
239 for (uint32_t i = 0U; i < k_ra8_itm_poll_limit; i++) {
240 if (*internal_itm_stim0() != 0U) {
241 *(volatile uint8_t*)k_ra8_itm_stim_base = byte;
242 return;
243 }
244 }
245}

295/* Itm put u32 -- see implementation for details. */
296RA8_INTERNAL static inline void internal_itm_put_u32(uint32_t value)
297{
298 char buf[k_ra8_u32_max_digits + 1U] = {};
299 uint8_t i = 0U;
300 if (value == 0U) {
301 internal_itm_putc('0');
302 return;
303 }
304 /* mcdc-deactivated: digit-buffer bound; uint32_t max is 10 digits (k_ra8_u32_max_digits == 10), so the loop always terminates by `value == 0` first; the bound is a defensive watchdog. */
305 while (value != 0U && i < k_ra8_u32_max_digits) {
306 buf[i++] = (char)('0' + (char)(value % (uint32_t)k_ra8_decimal_base));
307 value /= (uint32_t)k_ra8_decimal_base;
308 }
309 /* Digits were pushed LSB-first; emit MSB-first. */
310 while (i > 0U) {
311 i--;
312 internal_itm_putc((uint8_t)buf[i]);
313 }
314}

334RA8_INTERNAL static inline void internal_itm_put_i32(int32_t value)
335{
336 if (value < 0) {
337 internal_itm_putc((uint8_t)'-');
338 /* Casting INT32_MIN negated is UB; subtract through uint32_t. */
339 internal_itm_put_u32((uint32_t)(-(int64_t)value));
340 return;
341 }
342 internal_itm_put_u32((uint32_t)value);
343}
