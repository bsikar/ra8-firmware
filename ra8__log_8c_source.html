161RA8_INTERNAL static inline bool internal_itm_ready(void)
162{
163 /* A redirected byte sink needs no ITM/debugger -- it is always ready. */
164 if (s_byte_sink != nullptr) {
165 return true;
166 }
167#ifndef RA8_OFF_TARGET
168 /* Hardening: if DEMCR.TRCENA is clear the ITM block is powered down
169 * and any read of its registers (TCR, TENR, STIM) will bus-fault.
170 * Pre-check TRCENA via the shared ra8_scb primitive (DEMCR is always
171 * accessible) before touching ITM.
172 *
173 * This was the root cause of escalation-to-LOCKUP observed during
174 * USB bring-up: the original USB fault entered the fault handler,
175 * which called this function, which read ITM_TCR with TRCENA=0 ->
176 * second fault -> LOCKUP at PC=0xEFFFFFFE, hiding the real PC. */
177 if (!ra8_scb_trace_enabled()) {
178 return false;
179 }
180 /* Additional belt-and-braces: never poke ITM from a fault context.
181 * IPSR != 0 means we are inside an exception handler. Even if
182 * TRCENA happens to be on, dropping log lines is preferable to a
183 * second fault that masks the original PC. */
184 /* cppcheck-suppress unreadVariable -- the inline asm mrs writes IPSR into the variable; cppcheck cannot see through the asm. */
185 /* cppcheck-suppress knownConditionTrueFalse -- cppcheck assumes the asm-written IPSR value stays 0. */
186 volatile uint32_t ipsr = 0U;
187 __asm__ volatile("mrs %0, ipsr" : "=r"(ipsr));
188 if (ipsr != 0U) {
189 return false;
190 }
191#endif
192 const uint32_t tcr = *internal_itm_tcr();
193 const uint32_t tenr = *internal_itm_tenr();
194 /* TCR bit 0 = ITMENA. TENR bit 0 = stimulus port 0 enabled. */
195 if ((tcr & 1U) == 0U) {
196 return false;
197 }
198 if ((tenr & 1U) == 0U) {
199 return false;
200 }
201 return *internal_itm_stim0() != 0U;
202}

222RA8_INTERNAL static inline void internal_itm_putc(uint8_t byte)
223{
224 if (s_byte_sink != nullptr) {
225 s_byte_sink(s_byte_sink_ctx, byte);
226 return;
227 }
228 /* A real build should pre-check `internal_itm_ready()` once during
229 * init and skip the call entirely when the debugger is absent. For
230 * now we poll with a small bound so a disconnected debugger never
231 * stalls the firmware. */
232 enum : uint32_t {
233 k_ra8_itm_poll_limit = 1000U,
234 };
235 for (uint32_t i = 0U; i < k_ra8_itm_poll_limit; i++) {
236 if (*internal_itm_stim0() != 0U) {
237 *(volatile uint8_t*)k_ra8_itm_stim_base = byte;
238 return;
239 }
240 }
241}

291/* Itm put u32 -- see implementation for details. */
292RA8_INTERNAL static inline void internal_itm_put_u32(uint32_t value)
293{
294 char buf[k_ra8_u32_max_digits + 1U] = {};
295 uint8_t i = 0U;
296 if (value == 0U) {
297 internal_itm_putc('0');
298 return;
299 }
300 /* mcdc-deactivated: digit-buffer bound; uint32_t max is 10 digits (k_ra8_u32_max_digits == 10), so the loop always terminates by `value == 0` first; the bound is a defensive watchdog. */
301 while (value != 0U && i < k_ra8_u32_max_digits) {
302 buf[i++] = (char)('0' + (char)(value % (uint32_t)k_ra8_decimal_base));
303 value /= (uint32_t)k_ra8_decimal_base;
304 }
305 /* Digits were pushed LSB-first; emit MSB-first. */
306 while (i > 0U) {
307 i--;
308 internal_itm_putc((uint8_t)buf[i]);
309 }
310}

330RA8_INTERNAL static inline void internal_itm_put_i32(int32_t value)
