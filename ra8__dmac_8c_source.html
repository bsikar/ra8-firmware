360static inline void internal_program_channel(volatile r_dmac_channel_regs_t* reg,
361 const ra8_dmac_config_t* cfg)
362{
363 /* HUM 17.2.14 DMCNT, p 743 -- clear DTE before reprogramming. */
364 reg->DMCNT = 0U;
365 /* HUM 17.2.10 DMTMD, p 738. */
366 reg->DMTMD = internal_dmtmd_value(cfg);
367 /* HUM 17.2.12 DMAMD, p 741. */
368 reg->DMAMD = internal_dmamd_value(cfg->src_inc, cfg->dst_inc);
369 /* HUM 17.2.4 DMSAR p 734, 17.2.6 DMDAR p 735. */
370 reg->DMSAR = cfg->src;
371 reg->DMDAR = cfg->dst;
372 /* HUM 17.2.8 DMCRA. */
373 reg->DMCRA = internal_dmcra_value(cfg);
374 /* HUM 17.2.9 DMCRB p 737 -- "Set the same value for DMCRBH and DMCRBL
375 * in repeat transfer mode, block transfer mode and repeat-block
376 * transfer mode." Mirror block_count into both halves; normal mode
377 * leaves DMCRB at 0 ("In normal transfer mode, DMCRB is not used"). */
378 if (cfg->mode == k_ra8_dmac_mode_normal) {
379 reg->DMCRB = 0U;
380 } else {
381 const uint32_t bc = (uint32_t)cfg->block_count;
382 reg->DMCRB = bc | (bc << k_ra8_dmcra_high_pos);
383 }
384 /* HUM 17.2.13 DMOFR p 743 -- offset-addition not exposed. */
385 reg->DMOFR = 0U;
386 /* HUM 17.2.11 DMINT p 739. */
387 reg->DMINT = internal_dmint_value(cfg);
388}

395ra8_err_t ra8_dmac_start(uint8_t channel, const ra8_dmac_config_t* cfg)
396{
397 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
398 const ra8_err_t verr = internal_validate_cfg(cfg);
399 if (verr != k_ra8_ok) {
400 return verr;
401 }
402
403 volatile r_dmac_channel_regs_t* reg = ra8_dmac(channel);
404 if (reg == nullptr) {
405 return k_ra8_err_out_of_range;
406 }
407
408 /* DMAC0 + DTC0 share MSTPA22; the ref count tracks how many DMAC
409 * channels (or the DTC) currently need the bit cleared.
410 * HUM Ch 11.2.6 "MSTPCRA : Module Stop Control Register A", p 443. */
411 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_dmac0_dtc0);
412 RA8_RETURN_ON_ERROR(mst_err, s_tag, "dmac_start: mstp enable"); /* GCOVR_EXCL_BR_LINE */
413
414 internal_program_channel(reg, cfg);
415
416 /* HUM 17.2.20 DMAST p 749 -- the shared activation gate must be
417 * set before any per-channel DTE write takes effect. FSP does the
418 * same in `R_DMAC_Open` (`r_dmac.c` line 178). Idempotent: leaving
419 * it set across multiple channels is the documented behaviour. */
420 ra8_dma_shared()->DMAST = k_ra8_dmast_dmst_mask;
421
422 /* HUM 17.2.14 DMCNT p 743 -- arm the channel. */
423 reg->DMCNT = k_ra8_dmcnt_dte_mask;
424
425 ra8_log_info_val(s_tag, "dmac_start ch", (uint32_t)channel);
426 return k_ra8_ok;
427}

429ra8_err_t ra8_dmac_stop(uint8_t channel)
430{
431 volatile r_dmac_channel_regs_t* reg = ra8_dmac(channel);
432 if (reg == nullptr) {
433 return k_ra8_err_out_of_range;
434 }
435 /* HUM 17.2.14 DMCNT p 743. */
436 reg->DMCNT = 0U;
437 /* Drop the matching MSTP reference acquired in ra8_dmac_start. */
438 return ra8_mstp_disable(k_ra8_mstp_dmac0_dtc0);
439}

489ra8_err_t ra8_dmac_set_address_mode(uint8_t channel,
490 ra8_dmac_addr_mode_t src_mode,
491 ra8_dmac_addr_mode_t dest_mode)
492{
493 if ((src_mode > k_ra8_dmac_addr_decrement) || (dest_mode > k_ra8_dmac_addr_decrement)) {
494 return k_ra8_err_invalid_arg;
495 }
496 volatile r_dmac_channel_regs_t* reg = ra8_dmac(channel);
497 if (reg == nullptr) {
498 return k_ra8_err_out_of_range;
499 }
500 /* HUM 17.2.12 DMAMD p 741 -- preserve DARA/SARA/DADR/SADR fields and
501 * only rewrite the SM/DM 2-bit slots. */
502 uint16_t v = reg->DMAMD;
503 v &= (uint16_t)~(k_ra8_dmamd_sm_mask | k_ra8_dmamd_dm_mask);
504 v |= (uint16_t)((uint16_t)src_mode << k_ra8_dmamd_sm_pos);
505 v |= (uint16_t)((uint16_t)dest_mode << k_ra8_dmamd_dm_pos);
506 reg->DMAMD = v;
507 return k_ra8_ok;
508}
