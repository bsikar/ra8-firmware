359static void ls_print_hex(uint32_t value)
360{
361 uint8_t buf[k_ls_hex_nibbles];
362 for (uint32_t i = 0U; i < (uint32_t)k_ls_hex_nibbles; i++) {
363 const uint32_t shift = ((uint32_t)k_ls_hex_nibbles - 1U - i) * (uint32_t)k_ls_nibble_bits;
364 const uint32_t nib = (value >> shift) & (uint32_t)k_ls_nibble_mask;
365 buf[i] = (uint8_t)((nib < (uint32_t)k_ls_dec_ten) ? ('0' + nib) : ('A' + (nib - k_ls_dec_ten)));
366 }
367 ls_print(buf, (uint32_t)k_ls_hex_nibbles);
368}

371static void ls_print_uint(uint32_t value)
372{
373 uint8_t buf[k_ls_dec_digits];
374 uint32_t n = 0U;
375 if (value == 0U) {
376 buf[n] = '0';

380 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_ls_dec_ten));

383 }
384 for (uint32_t i = 0U; i < n; i++) {

537static uint32_t ls_build_atlas(void)
538{
539 const uint32_t width = (uint32_t)k_ls_canvas_w;
540 const uint32_t height = (uint32_t)k_ls_canvas_h;
541 const uint32_t band_h = (uint32_t)k_ls_band_h;
542 const uint32_t bands = (uint32_t)k_ls_bands;
543 for (uint32_t i = 0U; i < (uint32_t)k_ls_hdr_bytes; i++) {
544 s_atlas[i] = 0U;
545 }
546 s_atlas[0] = 'J';
547 s_atlas[1] = 'O';
548 s_atlas[2] = 'F';
549 s_atlas[3] = '1';
550 ls_put_u16(&s_atlas[4], (uint16_t)width);
551 ls_put_u16(&s_atlas[6], (uint16_t)height);
552 ls_put_u16(&s_atlas[8], (uint16_t)width); /* tile_w == width (band column) */
553 ls_put_u16(&s_atlas[10], (uint16_t)band_h);
554 s_atlas[12] = (uint8_t)k_ls_bpp;
555 s_atlas[13] = (uint8_t)k_ra8_jof_codec_raw;
556 ls_put_u32(&s_atlas[16], bands);
557
558 const uint32_t index_off = (uint32_t)k_ls_hdr_bytes; /* no pixel region */
559 for (uint32_t n = 0U; n < bands; n++) {
560 const uint32_t y0 = n * band_h;
561 const uint32_t th = ((height - y0) < band_h) ? (height - y0) : band_h;
562 ls_put_u32(&s_atlas[index_off + (n * (uint32_t)k_ls_idx_entry)], index_off);
563 ls_put_u32(&s_atlas[index_off + (n * (uint32_t)k_ls_idx_entry) + 4U],
564 width * th * (uint32_t)k_ls_bpp);
565 }
566 const uint32_t ftr = index_off + (bands * (uint32_t)k_ls_idx_entry);
567 ls_put_u32(&s_atlas[ftr], index_off);
568 ls_put_u32(&s_atlas[ftr + 4U], bands);
569 ls_put_u32(&s_atlas[ftr + 8U], (uint32_t)k_ls_atlas_total);
570 const uint32_t mag = ftr + (uint32_t)k_ls_ftr_magic_off;
571 s_atlas[mag + 0U] = 'J';
572 s_atlas[mag + 1U] = 'O';
573 s_atlas[mag + 2U] = 'F';
574 s_atlas[mag + 3U] = 'E';
575 return (uint32_t)k_ls_atlas_total;
576}

603static void ls_row_color(uint32_t base, uint32_t r, uint32_t th, uint8_t* rgb)
604{
605 if (r < (uint32_t)k_ls_seam_h) {
606 rgb[0] = (uint8_t)k_ls_seam_level;
607 rgb[1] = (uint8_t)k_ls_seam_level;
608 rgb[2] = (uint8_t)k_ls_seam_level;
609 return;
610 }
611 const uint32_t span = (th > 1U) ? (th - 1U) : 1U;
612 const uint32_t pct = (uint32_t)k_ls_grad_lo + (((uint32_t)k_ls_grad_span * r) / span);
613 rgb[0] =
614 ls_scale_u8((uint8_t)((base >> (uint32_t)k_ls_shift_16) & (uint32_t)k_ls_byte_mask), pct);
615 rgb[1] = ls_scale_u8((uint8_t)((base >> (uint32_t)k_ls_shift_8) & (uint32_t)k_ls_byte_mask), pct);
616 rgb[2] = ls_scale_u8((uint8_t)(base & (uint32_t)k_ls_byte_mask), pct);
617}

635static ra8_err_t ls_band_decode(void* ctx,
636 const ra8_tile_key_t* key,
637 uint8_t* cell,
638 uint32_t cell_bytes,
639 uint16_t* out_w,
640 uint16_t* out_h)
641{
642 RA8_CHECK_NULL_PTR(ctx, s_tag, "decode ctx must not be nullptr");
643 RA8_CHECK_NULL_PTR(key, s_tag, "key must not be nullptr");
644 RA8_CHECK_NULL_PTR(cell, s_tag, "cell must not be nullptr");
645 RA8_CHECK_NULL_PTR(out_w, s_tag, "out_w must not be nullptr");
646 RA8_CHECK_NULL_PTR(out_h, s_tag, "out_h must not be nullptr");
647 const ls_paint_ctx_t* pc = (const ls_paint_ctx_t*)ctx;
648 const uint32_t y0 = (uint32_t)key->tile_y * (uint32_t)pc->band_h;
649 uint32_t th = (uint32_t)pc->band_h;
650 if ((y0 + th) > pc->canvas_h) {
651 th = pc->canvas_h - y0;
652 }
653 if ((pc->width * th * (uint32_t)pc->bpp) > cell_bytes) {
654 return k_ra8_err_out_of_range;
655 }
656 const uint32_t base = k_ls_palette[(uint32_t)key->tile_y % (uint32_t)k_ls_palette_size];
657 uint32_t off = 0U;
658 for (uint32_t r = 0U; r < th; r++) { /* bounded by band_h */
659 uint8_t rgb[k_ls_bpp];
660 ls_row_color(base, r, th, rgb);
661 for (uint32_t x = 0U; x < pc->width; x++) { /* bounded by canvas width */
662 cell[off] = rgb[0];
663 cell[off + 1U] = rgb[1];
664 cell[off + 2U] = rgb[2];
665 off += (uint32_t)pc->bpp;
666 }
667 }
668 *out_w = (uint16_t)pc->width;
669 *out_h = (uint16_t)th;
670 return k_ra8_ok;
671}
