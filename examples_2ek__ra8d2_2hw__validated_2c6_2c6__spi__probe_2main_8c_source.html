284static void c6_probe_run(void)
285{
286 uint8_t hs_idx =
287 c6_probe_best(s_c6_stats.hs_vote, (uint8_t)k_c6_sb_count, (uint32_t)k_c6_probe_min_votes);
288 bool link_live = false;
289 ra8_spi_mode_t used = k_c6_mode_order[0];
290
291 for (uint8_t m = 0U; m < (uint8_t)k_c6_probe_mode_count && !link_live; m++) {
292 used = k_c6_mode_order[m];
293 link_live = c6_probe_sweep_mode(used, s_c6_pclka_hz, &s_c6_stats, &hs_idx);
294 }
295
296 /* Here, and not before the sweep: a freshly-booted C6 has its INIT event
297 * queued and holds DATA_READY high until a transaction drains it, so an
298 * early contest would find nothing sunk. By now the queue is empty (or the
299 * link is dead and the contest is the only evidence there is) and the
300 * chip-select has been released, which is DATA_READY's true idle state. */
301 if (c6_probe_pull_contest(&s_c6_stats) != k_ra8_ok) {
302 c6_probe_puts("c6_probe: pull-up contest failed; map falls back to votes\r\n");
303 }
304
305 uint8_t dr_idx = (uint8_t)k_c6_sb_count;
306 c6_probe_resolve_map(&s_c6_stats, &hs_idx, &dr_idx);
307 c6_probe_print_map(&s_c6_stats, hs_idx, dr_idx);
308 c6_probe_print_verdict(&s_c6_stats, used, link_live);
309 if (!link_live) {
310 (void)ra8_board_led_on(k_ra8_board_led2);
311 }
312}

328int32_t main(void)
329{
330 c6_probe_setup_or_halt();
331 ra8_isr_globals_enable();
332
333 c6_probe_puts("\r\nc6_probe: EK-RA8D2 <-> ESP32-C6 esp-hosted SPI probe\r\n");
334 c6_probe_puts("c6_probe: pmod1 sci2 simple-spi controller, cs on P804\r\n");
335 ra8_delay_ms((uint32_t)k_c6_probe_boot_wait_ms);
336
337 c6_sideband_sample_t idle = {};
338 c6_probe_sample_sideband(&idle);
339 c6_probe_print_sideband("c6_probe: idle sideband", &idle);
340
341 c6_probe_wire_test();
342 (void)c6_probe_cs_hunt(&s_c6_stats);
343
344 if (c6_probe_spi_pins_init() != k_ra8_ok) {
345 c6_probe_puts("c6_probe: spi pin routing failed\r\n");
346 c6_probe_panic_halt();
347 }
348 c6_probe_run();
349
350 while (1) {
351 (void)ra8_board_led_toggle(k_ra8_board_led1);
352 ra8_delay_ms((uint32_t)k_c6_probe_idle_ms);
353 }
354
355 c6_probe_panic_halt();
356 return 0;
357}
