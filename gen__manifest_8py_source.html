23def ascii_only(text: str) -> str:
24 """Fold text to 7-bit ASCII that is safe inside a C string literal.
25
26 NFKD decomposition first, so an accented letter splits into a plain letter
27 plus a combining mark and the letter survives the ASCII encode; encoding
28 without it would drop the whole character. Anything still non-ASCII after
29 that -- CJK, emoji, typographic dashes -- is dropped silently, so a title in
30 a non-Latin script can come back empty. That is accepted: the repository
31 encoding policy requires pure 7-bit ASCII in generated sources.
32
33 Double quotes become apostrophes and backslashes are deleted, which is what
34 keeps the result from terminating or escaping the surrounding C literal.
35 Callers must still not wrap the result in anything but a plain "..." .
36
37 Args:
38 text: Arbitrary Unicode, typically a book title or author.
39
40 Returns:
41 ASCII-only text containing neither a double quote nor a backslash. May
42 be shorter than the input, and may be empty.
43 """
44 norm = unicodedata.normalize("NFKD", text)
45 out = norm.encode("ascii", "ignore").decode("ascii")
46 return out.replace('"', "'").replace("\\", "")

49def unwrap_container(data: bytes) -> bytes:
50 """Inflate a chunked RBKC .rabook container back to its flat blob.
51
52 Keep in sync with ra8_book_container_t in libs/ra8_book/inc/ra8_book.h:
53 "RBKC" + <I chunk_bytes + <Q total + <I count + <I reserved(0), a
54 (count + 1)-entry <Q offset table, then count concatenated zlib streams.
55 """
56 if data[:4] != b"RBKC":
57 msg = "not an RBKC container"
58 raise ValueError(msg)
59 chunk_bytes, total, count, reserved = struct.unpack_from("<IQII", data, 4)
60 if reserved != 0 or chunk_bytes == 0 or count != (total + chunk_bytes - 1) // chunk_bytes:
61 msg = "malformed RBKC header"
62 raise ValueError(msg)
63 offsets = struct.unpack_from(f"<{count + 1}Q", data, 24)
64 payload = 24 + 8 * (count + 1)
65 blob = b"".join(
66 zlib.decompress(data[payload + offsets[i] : payload + offsets[i + 1]]) for i in range(count)
67 )
68 if len(blob) != total:
69 msg = "RBKC inflated size mismatch"
70 raise ValueError(msg)
71 return blob

74def read_meta(path: Path) -> dict[str, object]:
75 """Inflate one .rabook and extract the fields the manifest header needs.
76
77 Reads the whole container into memory and inflates all of it just to reach
78 the header and a few pool strings -- wasteful per file, but it is the only
79 way to learn the inflated size, which is the number the firmware sizes its
80 scratch buffer from.
81
82 Note the two different sizes returned and do not confuse them: `file_size`
83 is the compressed bytes on the SD card, `inflated_size` is the RAM the
84 device must have free to open the book.
85
86 Args:
87 path: Path to a .rabook container.
88
89 Returns:
90 Dict with "title", "author", "language" (decoded with replacement, so
91 never raises on bad UTF-8), "file_size", "inflated_size", "chapters"
92 and "images".
93
94 Raises:
95 ValueError: Not a valid RBKC container, prefixed with the file path so
96 the caller's loop reports which book failed.
97 OSError: `path` cannot be read.
98 """
99 with Path(path).open("rb") as fh:
100 container = fh.read()
101 try:
102 flat = unwrap_container(container)
103 except ValueError as exc:
104 msg = f"{path}: {exc}"
105 raise ValueError(msg) from exc
106 inflated_size = len(flat)
107 h = struct.unpack("<8s23I", flat[:100])
108 string_off = h[19]
109
110 def s(off: int) -> str:
111 start = string_off + off
112 end = flat.index(b"\x00", start)
113 return flat[start:end].decode("utf-8", "replace")
114
115 return {
116 "title": s(h[4]),
117 "author": s(h[5]),
118 "language": s(h[6]),
119 "file_size": len(container),
120 "inflated_size": inflated_size,
121 "chapters": h[9],
122 "images": h[17],
123 }

187def _generated_table(entries: list[dict[str, object]]) -> list[str]:
188 """The book table itself, one initialiser row per book."""
189 lines = [
190 "/** @brief The bundled book index. Generated; data table, magic numbers ok. */",
191 "// NOLINTBEGIN(readability-magic-numbers)",
192 "static const ra8_book_library_entry_t g_ra8_book_library[k_ra8_book_library_count] = {",
193 ]
194 lines.extend(
195 f' {{ "{ascii_only(e["title"])}", "{ascii_only(e["author"])}", '
196 f'"{ascii_only(e["filename"])}", {e["file_size"]}U, {e["inflated_size"]}U }},'
197 for e in entries
198 )
199 lines += ["};", "// NOLINTEND(readability-magic-numbers)", ""]
200 return lines

203def _build_header_lines(entries: list[dict[str, object]]) -> tuple[list[str], int]:
204 """Return the C header lines for the given book entries, and the worst-case size."""
205 biggest = max((e["inflated_size"] for e in entries), default=0)
206 lines = [
207 *_HEADER_PREAMBLE,
208 *_generated_constants(entries, biggest),
209 *_generated_table(entries),
210 ]
211 return lines, biggest

214def main() -> int:
215 """Scan a directory of .rabook files and write the generated C header.
216
217 Books are ordered by filename, and that order fixes the indices in
218 `g_ra8_book_library[]`. Anything that persists a book index -- a saved
219 reading position, a baked fixture -- is invalidated by adding or renaming a
220 file in the directory.
221
222 The header is written with `encoding="ascii"`, so a title that survived
223 `ascii_only` but still holds a non-ASCII byte raises here rather than
224 producing a source file the encoding gate would later reject.
225
226 Non-.rabook files in the directory are ignored; an unreadable or corrupt
227 .rabook aborts the whole run rather than being skipped, because a manifest
228 that silently omits a book would under-size the inflate scratch buffer.
229 """
230 ap = argparse.ArgumentParser(description="Generate the ra8_book library manifest header.")
231 ap.add_argument("compiled_dir", help="directory of .rabook files")
232 ap.add_argument("output", help="path to the generated header")
233 args = ap.parse_args()
234
235 compiled_dir = Path(args.compiled_dir)
236 files = sorted(f.name for f in compiled_dir.iterdir() if f.name.endswith(".rabook"))
237 entries = []
238 for name in files:
239 meta = read_meta(compiled_dir / name)
240 meta["filename"] = name
241 entries.append(meta)
242
243 lines, biggest = _build_header_lines(entries)
244
245 with Path(args.output).open("w", encoding="ascii") as fh:
246 fh.write("\n".join(lines))
247 total_disk = sum(e["file_size"] for e in entries)
248 print(
249 f"wrote {args.output}: {len(entries)} books, "
250 f"{total_disk // 1024 // 1024} MB on disk, max inflate {biggest // 1024} KB"
251 )
