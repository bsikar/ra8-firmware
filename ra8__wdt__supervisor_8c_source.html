209{
210 (void)arg;
211 /* NASA Rule 2: outer loop is the canonical "main control loop"
212 * exception. Body is finite. */

228ra8_err_t ra8_wdt_supervisor_init(const ra8_wdt_sup_cfg_t* cfg)
229{
230 const ra8_err_t cfg_err = internal_validate_cfg(cfg);
231 if (cfg_err != k_ra8_ok) {
232 return cfg_err;
233 }
234 if (s_state.initialized) {
235 return k_ra8_err_busy;
236 }
237
238 (void)memset(&s_state.slots[0], 0, sizeof s_state.slots);
239 s_state.cfg = *cfg;
240 s_state.now = internal_default_now;
241 s_state.refresh = internal_default_refresh;
242 s_state.started = false;
243
244 const UINT mx = tx_mutex_create(&s_state.mutex, (CHAR*)(uintptr_t)"ra8_wdt_sup", TX_NO_INHERIT);
245 if (mx != TX_SUCCESS) {
246 return k_ra8_err_rtos_error; /* GCOVR_EXCL_LINE -- shim always returns TX_SUCCESS */
247 }
248
249 s_state.initialized = true;
250 return k_ra8_ok;
251}

305ra8_wdt_supervisor_register_thread(const char* name, uint32_t deadline_ms, uint8_t* out_handle)
306{
307 if (out_handle == nullptr) {
308 return k_ra8_err_null_ptr;
309 }
310 *out_handle = (uint8_t)k_ra8_wdt_sup_handle_invalid;
311
312 if (name == nullptr) {
313 return k_ra8_err_null_ptr;
314 }
315 if (deadline_ms == 0U) {
316 return k_ra8_err_invalid_arg;
317 }
318 if (!s_state.initialized) {
319 return k_ra8_err_not_initialized;
320 }
321
322 const UINT mx = tx_mutex_get(&s_state.mutex, TX_WAIT_FOREVER);
323 if (mx != TX_SUCCESS) {
324 return k_ra8_err_rtos_error; /* GCOVR_EXCL_LINE -- shim always returns TX_SUCCESS */
325 }
326
327 ra8_err_t result = k_ra8_err_no_mem;
328 for (uint8_t i = 0U; i < (uint8_t)k_ra8_wdt_sup_max_threads; ++i) {
329 if (s_state.slots[i].state == (uint8_t)k_ra8_wdt_sup_slot_free) {
330 internal_fill_slot(i, name, deadline_ms);
331 *out_handle = i;
332 result = k_ra8_ok;
333 break;
334 }
335 }
336
337 (void)tx_mutex_put(&s_state.mutex);
338 return result;
339}

392ra8_err_t ra8_wdt_supervisor_tick(bool* out_did_refresh)
393{
394 if (!s_state.initialized) {
395 if (out_did_refresh != nullptr) {
396 *out_did_refresh = false;
397 }
398 return k_ra8_err_not_initialized;
399 }
400
401 const UINT mx = tx_mutex_get(&s_state.mutex, TX_WAIT_FOREVER);
402 if (mx != TX_SUCCESS) {
403 /* GCOVR_EXCL_START -- shim always returns TX_SUCCESS */
404 if (out_did_refresh != nullptr) {
405 *out_did_refresh = false;
406 }
407 return k_ra8_err_rtos_error;
408 /* GCOVR_EXCL_STOP */
409 }
410
411 bool all_alive = true;
412 bool any_present = false;
413 const uint32_t now_ms = s_state.now();
414 for (uint8_t i = 0U; i < (uint8_t)k_ra8_wdt_sup_max_threads; ++i) {
415 if (s_state.slots[i].state != (uint8_t)k_ra8_wdt_sup_slot_used) {
416 continue;
417 }
418 any_present = true;
419 if (internal_is_overdue(now_ms,
420 s_state.slots[i].last_checkin_ms,
421 s_state.slots[i].deadline_ms)) {
422 all_alive = false;
423 break;
424 }
425 }
426
427 (void)tx_mutex_put(&s_state.mutex);
428
429 /* Only refresh if at least one thread is registered AND every
430 * registered thread is alive. With zero workers we deliberately do
431 * NOT kick the dog -- the supervisor would mask a degenerate config. */
432 const bool will_refresh = any_present && all_alive;
433 // mcdc-deactivated: ra8_wdt_supervisor_tick refresh dispatch gate; s_state.refresh is registered at init and remains non-NULL for the lifetime of the supervisor (it is the WDT kick callback, mandatory per the public-API contract). will_refresh varies independently and is fully covered, but the second condition is invariant on every reachable post-init path.
434 if (will_refresh && (s_state.refresh != nullptr)) {
435 s_state.refresh();
436 }
437 if (out_did_refresh != nullptr) {
438 *out_did_refresh = will_refresh;
439 }
440 return k_ra8_ok;
441}
