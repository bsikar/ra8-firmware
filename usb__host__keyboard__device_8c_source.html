302void hid_fill_report_body(uint8_t* out, uint32_t len)
303{
304 /* Boot-keyboard report: byte 1 = reserved, bytes 2.. = up to six keycodes
305 * (the typed "RA8D2"), the remainder 0 (no key). Byte 0 (modifier) is set by
306 * the caller. Deterministic, so the host computes the same body to verify. */
307 for (uint32_t i = (uint32_t)k_hid_body_idx; i < len; i++) {
308 out[i] = 0U;
309 }
310 for (uint32_t i = 0U; i < (uint32_t)k_hid_nkeys; i++) {
311 const uint32_t idx = (uint32_t)k_hid_key0_idx + i;
312 if (idx < len) {
313 out[idx] = s_kbd_keys[i];
314 }
315 }
316}

492static void hid_send_iter(uint32_t* seq)
493{
494 if (_ux_system_slave != UX_NULL) {
495 _ux_system_slave->ux_system_slave_device.ux_slave_device_state =
496 (unsigned long)UX_DEVICE_CONFIGURED;
497 }
498 UX_SLAVE_CLASS_HID_EVENT ev;
499 (void)memset(&ev, 0, sizeof(ev));
500 ev.ux_device_class_hid_event_length = (ULONG)k_hid_report_len;
501 ev.ux_device_class_hid_event_report_id = 0UL;
502 ev.ux_device_class_hid_event_report_type = (ULONG)UX_DEVICE_CLASS_HID_REPORT_TYPE_INPUT;
503 hid_fill_report_body(ev.ux_device_class_hid_event_buffer, (uint32_t)k_hid_report_len);
504 ev.ux_device_class_hid_event_buffer[k_hid_seq_idx] = (UCHAR)(*seq & (uint32_t)k_hid_byte_mask);
505 if (_ux_device_class_hid_event_set(s_hid_class, &ev) == UX_SUCCESS) {
506 (*seq)++;
507 s_dbg_dev_sent++;
508 (void)ra8_board_led_toggle(k_ra8_board_led1);
509 }
510 tx_thread_sleep(1U);
511}

513VOID hid_device_worker(ULONG arg)
514{
515 (void)arg;
516
517 UINT ux = hid_usbx_stack_up();
518 if (ux != UX_SUCCESS) {
519 s_dbg_dev_err = (uint32_t)ux;
520 return;
521 }
522 s_dbg_dev_step = (uint32_t)k_hid_dev_step_stack;
523 ux = hid_class_register();
524 if (ux != UX_SUCCESS) {
525 s_dbg_dev_err = (uint32_t)ux;
526 return;
527 }
528 s_dbg_dev_step = (uint32_t)k_hid_dev_step_class;
529 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
530 if (e != k_ra8_ok) {
531 s_dbg_dev_err = (uint32_t)e;
532 return;
533 }
534 s_dbg_dev_step = (uint32_t)k_hid_dev_step_dcd;
535 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
536 if (e != k_ra8_ok) {
537 s_dbg_dev_err = (uint32_t)e;
538 return;
539 }
540 s_dbg_dev_step = (uint32_t)k_hid_dev_step_attach;
541
542 uint32_t seq = 0U;
543 while (1) {
544 if (s_hid_class == UX_NULL) {
545 (void)tx_semaphore_get(&s_usb_host_keyboard_hid_active_sem, TX_WAIT_FOREVER);
546 continue;
547 }
548 s_dbg_dev_step = (uint32_t)k_hid_dev_step_send;
549 hid_send_iter(&seq);
550 }
551}
