204static ra8_err_t internal_sdramc_route_pins(void)
205{
206 const uint32_t count = (uint32_t)(sizeof(s_sdram_bus_pins) / sizeof(s_sdram_bus_pins[0]));
207 for (uint32_t i = 0U; i < count; ++i) {
208 const ra8_err_t route_err =
209 ra8_pfs_route_peripheral(s_sdram_bus_pins[i], k_ra8_psel_bus, "ra8_sdramc");
210 if (route_err != k_ra8_ok) {
211 return route_err;
212 }
213 const ra8_err_t drive_err =
214 ra8_pfs_set_drive_strength(s_sdram_bus_pins[i], k_ra8_pfs_dscr_high_speed_high);
215 if (drive_err != k_ra8_ok) {
216 return drive_err;
217 }
218 }
219 return k_ra8_ok;
220}

244static ra8_err_t internal_sdramc_wait(uint8_t mask)
245{
246 volatile r_sdramc_regs_t* const reg = ra8_sdramc();
247 for (uint32_t spin = 0U; spin < (uint32_t)k_ra8_sdramc_spin_max; ++spin) {
248 if ((uint8_t)(reg->SDSR & mask) == 0U) {
249 return k_ra8_ok;
250 }
251 }
252 ra8_log_error(s_tag, "sdramc: SDSR status bits never cleared");
253 return k_ra8_err_hw_timeout;

256ra8_err_t ra8_sdramc_init(void)
257{
258 const ra8_err_t pin_err = internal_sdramc_route_pins();
259 if (pin_err != k_ra8_ok) {
260 ra8_log_error(s_tag, "sdramc: bus-pin routing failed");
261 return pin_err;
262 }
263
264 volatile r_sdramc_regs_t* const reg = ra8_sdramc();
265
266 /* HUM Ch 15.6.10: confirm SDSR is clear, then program the
267 * one-shot SDRAM Initialization Register. SDIR is writable only
268 * once after reset. */
269 ra8_err_t err = internal_sdramc_wait((uint8_t)k_ra8_sdsr_all);
270 if (err != k_ra8_ok) {
271 return err;
272 }
273 reg->SDIR = (uint16_t)k_ra8_sdir_init;
274 reg->SDCCR = (uint8_t)k_ra8_sdccr_cfg_32bit;
275
276 /* Enable the SDCLK output to the device (PRCR-gated, CGC group). */
277 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
278 {
279 *ra8_sdramc_sdckocr() = (uint8_t)k_ra8_sdramc_kick;
280 }
281
282 /* Run the hardware init sequencer (precharge-all + auto-refresh). */
283 reg->SDICR = (uint8_t)k_ra8_sdramc_kick;
284 err = internal_sdramc_wait((uint8_t)k_ra8_sdsr_inist);
285 if (err != k_ra8_ok) {
286 return err;
287 }
288
289 /* Access + endian mode, then issue the LMR command (SDMOD).
290 * SDAMOD.BE = 1 keeps multi-beat accesses continuous so the
291 * sub-word address increments correctly. */
292 reg->SDAMOD = (uint8_t)k_ra8_sdamod_be;
293 reg->SDCMOD = 0U;
294 err = internal_sdramc_wait((uint8_t)k_ra8_sdsr_all);
295 if (err != k_ra8_ok) {
296 return err;
297 }
298 reg->SDMOD = (uint16_t)k_ra8_sdmod_lmr;
299 err = internal_sdramc_wait((uint8_t)k_ra8_sdsr_mrsst);
300 if (err != k_ra8_ok) {
301 return err;
302 }
303
304 /* Timing, address-multiplex, refresh -- then enable SDRAM access. */
305 reg->SDTR = (uint32_t)k_ra8_sdtr_init;
306 reg->SDADR = (uint8_t)k_ra8_sdadr_mxc_9col;
307 reg->SDRFCR = (uint16_t)k_ra8_sdrfcr_init;
308 reg->SDRFEN = (uint8_t)k_ra8_sdramc_kick;
309 reg->SDCCR = (uint8_t)k_ra8_sdccr_enable;
310
311 ra8_log_info(s_tag, "sdramc_init (64 MiB @ 0x68000000)");
312 return k_ra8_ok;
313}

359ra8_err_t ra8_sdramc_enter_self_refresh(void)
360{
361 volatile r_sdramc_regs_t* const reg = ra8_sdramc();
362
363 /* Precondition 1: all SDSR status bits must be clear before writing
364 * SDSELF.SFEN (HUM Table 15.8 requirement). */
365 /* HUM Ch 15.3.22 "SDSR : SDRAM Status Register" p 613 */
366 if ((uint8_t)(reg->SDSR & (uint8_t)k_ra8_sdsr_all) != 0U) {
367 ra8_log_error(s_tag, "sdramc: SDSR busy on self-refresh entry");
368 return k_ra8_err_invalid_state;
369 }
370
371 /* Precondition 2: SDSELF.SFEN must be 0 -- already in self-refresh
372 * is a caller error. */
373 /* HUM Ch 15.3.14 "SDSELF : SDRAM Self-Refresh Control Register" p 606 */
374 if ((uint8_t)(reg->SDSELF & (uint8_t)k_ra8_sdself_sfen) != 0U) {
375 ra8_log_error(s_tag, "sdramc: already in self-refresh");
376 return k_ra8_err_invalid_state;
377 }
378
379 /* Disable auto-refresh before asserting self-refresh. */
380 /* HUM Ch 15.3.16 "SDRFEN : SDRAM Auto-Refresh Control Register" p 608 */
381 reg->SDRFEN = 0U;
382
383 /* Assert SFEN: the hardware issues a final auto-refresh then
384 * switches the SDRAM to self-refresh mode. */
385 /* HUM Ch 15.3.14 "SDSELF : SDRAM Self-Refresh Control Register" p 606 */
386 reg->SDSELF = (uint8_t)k_ra8_sdself_sfen;
387
388 /* Wait for the transition to complete (SRFST clears when the
389 * SDRAM has entered self-refresh). */
390 const ra8_err_t err = internal_sdramc_wait((uint8_t)k_ra8_sdsr_srfst);
391 if (err != k_ra8_ok) {
392 return err;
393 }
394
395 ra8_log_info(s_tag, "sdramc: entered self-refresh");
396 return k_ra8_ok;
397}

399ra8_err_t ra8_sdramc_exit_self_refresh(void)
400{
401 volatile r_sdramc_regs_t* const reg = ra8_sdramc();
402
403 /* Precondition 1: SDSELF.SFEN must be 1 -- must be in self-refresh
404 * before exiting. */
405 /* HUM Ch 15.3.14 "SDSELF : SDRAM Self-Refresh Control Register" p 606 */
406 if ((uint8_t)(reg->SDSELF & (uint8_t)k_ra8_sdself_sfen) == 0U) {
407 ra8_log_error(s_tag, "sdramc: not in self-refresh on exit");
408 return k_ra8_err_invalid_state;
409 }
410
411 /* Precondition 2: SDSR.SRFST must be clear -- no transition may be
412 * in progress when writing SDSELF. */
413 /* HUM Ch 15.3.22 "SDSR : SDRAM Status Register" p 613 */
414 if ((uint8_t)(reg->SDSR & (uint8_t)k_ra8_sdsr_srfst) != 0U) {
415 ra8_log_error(s_tag, "sdramc: SDSR SRFST busy on self-refresh exit");
416 return k_ra8_err_invalid_state;
417 }
418
419 /* Clear SFEN: the hardware performs the self-refresh clearing
420 * cycles (count set by SDRFCR.REFW) then returns to normal. */
421 /* HUM Ch 15.3.14 "SDSELF : SDRAM Self-Refresh Control Register" p 606 */
422 reg->SDSELF = 0U;
423
424 /* Wait for the recovery to complete (SRFST clears when auto-
425 * refresh is ready to resume). */
426 const ra8_err_t err = internal_sdramc_wait((uint8_t)k_ra8_sdsr_srfst);
427 if (err != k_ra8_ok) {
428 return err;
429 }
430
431 /* Re-enable the auto-refresh engine. */
432 /* HUM Ch 15.3.16 "SDRFEN : SDRAM Auto-Refresh Control Register" p 608 */
433 reg->SDRFEN = (uint8_t)k_ra8_sdramc_kick;
434
435 ra8_log_info(s_tag, "sdramc: exited self-refresh");
436 return k_ra8_ok;
