134RA8_INTERNAL static bool ra8_c6link_tlv_named(const uint8_t* payload)
135{
136 bool named = true;
137 for (uint16_t i = 0U; i < (uint16_t)k_ra8_c6link_tlv_ep_len; i++) {
138 const uint8_t got = payload[(uint16_t)k_ra8_c6link_tlv_value + i];
139 if ((got != (uint8_t)RPC_EP_NAME_RSP[i]) && (got != (uint8_t)RPC_EP_NAME_EVT[i])) {
140 named = false;
141 }
142 }

150 return nullptr;
151 }
152 *proto_len = 0U;

154 (uint16_t)((uint16_t)k_ra8_c6link_tlv_value + (uint16_t)k_ra8_c6link_tlv_ep_len);
155
156 if ((len < (uint16_t)k_ra8_c6link_tlv_overhead) ||
157 (payload[k_ra8_c6link_tlv_type] != (uint8_t)k_ra8_c6link_tlv_t_epname) ||
158 (payload[data_tag] != (uint8_t)k_ra8_c6link_tlv_t_data)) {
159 return nullptr;
160 }

165 return nullptr;
166 }
167

169 if ((body == 0U) || (((uint32_t)body + (uint32_t)k_ra8_c6link_tlv_overhead) > (uint32_t)len)) {
170 return nullptr;
171 }
172 *proto_len = body;
