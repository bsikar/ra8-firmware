136static uint8_t elc_demo_uint_to_dec(uint8_t* buf, uint32_t v)
137{
138 uint8_t tmp[k_elc_demo_uint_dec_max];
139 uint8_t n = 0U;
140 uint32_t r = v;
141 if (r == 0U) {
142 buf[0] = (uint8_t)k_elc_demo_ascii_zero;
143 return 1U;
144 }
145 while ((r > 0U) && (n < (uint8_t)k_elc_demo_uint_dec_max)) {
146 tmp[n] =
147 (uint8_t)((uint8_t)k_elc_demo_ascii_zero + (uint8_t)(r % (uint32_t)k_elc_demo_dec_base));
148 r = r / (uint32_t)k_elc_demo_dec_base;
149 n++;
150 }
151 for (uint8_t i = 0U; i < n; i++) {
152 buf[i] = tmp[(uint8_t)(n - 1U - i)];
153 }
154 return n;
155}

170static uint32_t elc_demo_format_line(uint8_t* out, bool enabled, uint32_t trig_count)
171{
172 uint32_t off = 0U;
173 const uint8_t prefix[] = "elc: en=";
174 const uint8_t mid[] = " trig=";
175 const uint8_t suffix[] = "\r\n";
176 for (uint32_t i = 0U; i < sizeof(prefix) - 1U; i++) {
177 out[off++] = prefix[i];
178 }
179 out[off++] = enabled ? (uint8_t)k_elc_demo_ascii_one : (uint8_t)k_elc_demo_ascii_zero;
180 for (uint32_t i = 0U; i < sizeof(mid) - 1U; i++) {
181 out[off++] = mid[i];
182 }
183 off += elc_demo_uint_to_dec(&out[off], trig_count);
184 for (uint32_t i = 0U; i < sizeof(suffix) - 1U; i++) {
185 out[off++] = suffix[i];
186 }
187 return off;
188}

205int32_t main(void)
206{
207 elc_demo_setup_or_halt();
208 ra8_isr_globals_enable();
209
210 uint32_t trig_count = 0U;
211 while (1) {
212 if (ra8_elc_software_trigger((uint8_t)k_elc_demo_sw_event_idx) != k_ra8_ok) {
213 (void)ra8_board_led_on(k_ra8_board_led2);
214 /* Emit FAIL banner so the HIL negative regex catches a silently
215 * broken SW-trigger path; without this the t=N counter would
216 * stop advancing but the demo would keep emitting "elc: en=1
217 * t=0" indefinitely and the probe would pass. */
218 const uint8_t trig_fail[] = "elc: FAIL trigger\r\n";
219 (void)ra8_board_uart_console_write(trig_fail, (size_t)(sizeof(trig_fail) - 1U));
220 } else {
221 trig_count++;
222 }
223 bool enabled = false;
224 if (ra8_elc_is_enabled(&enabled) != k_ra8_ok) {
225 (void)ra8_board_led_on(k_ra8_board_led2);
226 const uint8_t enabled_fail[] = "elc: FAIL is_enabled\r\n";
227 (void)ra8_board_uart_console_write(enabled_fail, (size_t)(sizeof(enabled_fail) - 1U));
228 }
229 uint8_t out[k_elc_demo_print_buf] = {};
230 const uint32_t off = elc_demo_format_line(out, enabled, trig_count);
231 if (ra8_board_uart_console_write(out, (size_t)off) != k_ra8_ok) {
232 break;
233 }
234 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
235 break;
236 }
237 ra8_delay_ms(k_elc_demo_period_ms);
238 }
239
240 elc_demo_panic_halt();
241 return 0;
242}
