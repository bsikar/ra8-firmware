182{
183 if (sz_code == (uint32_t)k_dmac_sz_word) {

193static uint32_t dmac_total_units(const dmac_chan_t* c)
194{
195 uint32_t count = c->dmcra & (uint32_t)k_dmac_dmcra_low_mask;
196 if (count == 0U) {
197 count = (uint32_t)k_dmac_dmcra_low_mask + 1U; /* DMCRAL==0 means 1024 units */
198 }
199 uint32_t blocks = c->dmcrb & (uint32_t)k_u16_mask;
200 if (blocks == 0U) {
201 blocks = 1U; /* normal mode: a single block of `count` units */
202 }
203 uint64_t total = (uint64_t)count * (uint64_t)blocks;
204 if (total > (uint64_t)k_dmac_max_units) {
205 total = (uint64_t)k_dmac_max_units;
206 }
207 return (uint32_t)total;
208}

211static void dmac_copy_units(uc_engine* uc, dmac_chan_t* c, uint32_t units, uint32_t unit)
212{
213 uint64_t src = (uint64_t)c->dmsar;
214 uint64_t dst = (uint64_t)c->dmdar;
215 const uint32_t sm = (c->dmamd >> (uint32_t)k_dmac_sm_pos) & (uint32_t)k_dmac_am_mask;
216 const uint32_t dm = (c->dmamd >> (uint32_t)k_dmac_dm_pos) & (uint32_t)k_dmac_am_mask;
217 const bool src_inc = (sm == (uint32_t)k_dmac_am_incr);
218 const bool dst_inc = (dm == (uint32_t)k_dmac_am_incr);
219 uint8_t word[k_dmac_unit_word] = {};
220 for (uint32_t i = 0U; i < units; i++) {
221 if (uc_mem_read(uc, src, word, unit) != UC_ERR_OK) {
222 break;
223 }
224 if (uc_mem_write(uc, dst, word, unit) != UC_ERR_OK) {
225 break;
226 }
227 if (src_inc) {
228 src += unit;
229 }
230 if (dst_inc) {
231 dst += unit;
232 }
233 c->units++;
234 }
235 c->dmsar = (uint32_t)src;
236 c->dmdar = (uint32_t)dst;
237}

240static void dmac_run_transfer(uc_engine* uc, uint32_t ch)
241{
242 dmac_chan_t* c = &s_dmac[ch];
243 if ((c->dmcnt & (uint8_t)k_dmac_dte_mask) == 0U) {
244 return; /* channel disarmed: SWREQ does nothing */
245 }
246 const uint32_t sz = (c->dmtmd >> (uint32_t)k_dmac_sz_pos) & (uint32_t)k_dmac_sz_mask;
247 const uint32_t unit = dmac_unit_bytes(sz);
248 const uint32_t units = dmac_total_units(c);
249 c->units = 0U;
250 dmac_copy_units(uc, c, units, unit);
251 c->dmcra &= ~(uint32_t)k_dmac_dmcra_low_mask; /* count drained to zero */
252 c->dmsts &= (uint8_t)~k_dmac_act_mask; /* synchronous: idle after copy */
253 c->dmsts |= (uint8_t)k_dmac_dtif_mask; /* transfer-end status latched */
254 c->copies++;
255 /* Console DMA tab: one line per completed DMAC transfer (not per unit). */
256 char ln[k_dmac_console_line_cap];
257 (void)snprintf(ln,
258 sizeof(ln),
259 "DMAC%u %u units x %uB",
260 (unsigned)ch,
261 (unsigned)c->units,
262 (unsigned)unit);
263 board_console_push(k_board_console_ch_dma, ln);
264 if (board_periph_trace()) {
265 (void)fprintf(stderr, " DMAC%u : copied %u units x %uB (SWREQ)\n", ch, c->units, unit);
266 }
267 if ((c->dmint & (uint8_t)k_dmac_dtif_mask) != 0U) {
268 board_periph_icu_raise_event(uc, (uint16_t)((uint16_t)k_dmac_event_ch0 + (uint16_t)ch));
269 }
270}

382static uint64_t shadow_read(const uint8_t* shadow, uint64_t span, uint64_t off, unsigned size)
383{
384 uint64_t v = 0U;

386 const uint64_t b = off + (uint64_t)i;

388 v |= (uint64_t)shadow[b] << (8U * i);
389 }
390 }
391 return v;
392}
