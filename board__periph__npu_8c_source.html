234static bool npu_guest_word(uc_engine* uc, uint64_t addr, uint32_t* out)
235{
236 uint8_t b[4] = {};
237 if (uc_mem_read(uc, addr, b, sizeof(b)) != UC_ERR_OK) {
238 return false;
239 }
240 /* Little-endian assembly over the fixed 4-byte buffer -- the loop keeps the
241 * byte shift a named multiple (i * 8U) instead of a bare 24U high-byte
242 * literal, and is statically bounded by sizeof(b) (NASA P10 Rule 2). */
243 *out = 0U;
244 for (uint32_t i = 0U; i < sizeof(b); ++i) {
245 *out |= (uint32_t)b[i] << (i * 8U);
246 }
247 return true;
248}

258static npu_dec_t npu_decode(uc_engine* uc, npu_cmd_t* out)
259{
260 const uint64_t qbase = npu_reg64((uint64_t)k_npu_off_qbase);
261 const uint32_t qsize = s_npu.reg[npu_word((uint64_t)k_npu_off_qsize)];
262 if (qsize < (uint32_t)k_ra8_npu_fake_header_bytes) {
263 return k_npu_dec_parse;
264 }
265 uint32_t w[k_ra8_npu_fake_word_num] = {};
266 for (uint32_t i = 0U; i < (uint32_t)k_ra8_npu_fake_word_num; i++) {
267 if (!npu_guest_word(uc, qbase + ((uint64_t)i * 4UL), &w[i])) {
268 return k_npu_dec_bus;
269 }
270 }
271 if ((w[k_ra8_npu_fake_word_op] & (uint32_t)k_ra8_npu_fake_magic_mask) !=
272 (uint32_t)k_ra8_npu_fake_magic) {
273 return k_npu_dec_parse; /* Real Vela stream / not ours: cannot interpret. */
274 }
275 out->op = w[k_ra8_npu_fake_word_op] & (uint32_t)k_ra8_npu_fake_op_mask;
276 out->src = w[k_ra8_npu_fake_word_src];
277 out->dst = w[k_ra8_npu_fake_word_dst];
278 out->count = w[k_ra8_npu_fake_word_count];
279 out->konst = w[k_ra8_npu_fake_word_const];
280 const bool bad_region =
281 (out->src >= (uint32_t)k_npu_region_count) || (out->dst >= (uint32_t)k_npu_region_count);
282 const bool bad_count = (out->count == 0U) || (out->count > (uint32_t)k_ra8_npu_fake_max_bytes);
283 if (bad_region || bad_count) {
284 return k_npu_dec_parse;
285 }
286 return k_npu_dec_ok;
287}

290static bool
291npu_apply(uc_engine* uc, const npu_cmd_t* c, uint64_t src, uint64_t dst, uint32_t* out_check)
292{
293 uint8_t buf[k_npu_chunk_bytes] = {};
294 uint32_t check = (uint32_t)k_npu_fnv_offset;
295 uint32_t done = 0U;
296 for (uint32_t chunk = 0U; chunk < (uint32_t)k_npu_max_chunks; chunk++) {
297 if (done >= c->count) {
298 break;
299 }
300 uint32_t n = c->count - done;
301 if (n > (uint32_t)k_npu_chunk_bytes) {
302 n = (uint32_t)k_npu_chunk_bytes;
303 }
304 if (uc_mem_read(uc, src + done, buf, n) != UC_ERR_OK) {
305 return false;
306 }
307 if (c->op == (uint32_t)k_ra8_npu_fake_op_addk) {
308 for (uint32_t i = 0U; i < n; i++) {
309 buf[i] = (uint8_t)((buf[i] + c->konst) & (uint32_t)k_ra8_npu_fake_byte_mask);
310 }
311 }
312 if (uc_mem_write(uc, dst + done, buf, n) != UC_ERR_OK) {
313 return false;
314 }
315 for (uint32_t i = 0U; i < n; i++) {
316 check = (check ^ (uint32_t)buf[i]) * (uint32_t)k_npu_fnv_prime;
317 }
318 done += n;
319 }
320 *out_check = check;
321 return true;
322}

352static void npu_execute(uc_engine* uc)
353{
354 npu_cmd_t c = {};
355 const npu_dec_t dec = npu_decode(uc, &c);
356 if (dec == k_npu_dec_bus) {
357 npu_fault((uint32_t)k_npu_status_buserr_bit);
358 return;
359 }
360 if (dec != k_npu_dec_ok) {
361 npu_fault((uint32_t)k_npu_status_parse_bit);
362 return;
363 }
364 const uint64_t src = npu_region_base(c.src);
365 const uint64_t dst = npu_region_base(c.dst);
366 if ((src == 0U) || (dst == 0U)) {
367 npu_fault((uint32_t)k_npu_status_parse_bit);
368 return;
369 }
370 uint32_t check = 0U;
371 if (!npu_apply(uc, &c, src, dst, &check)) {
372 npu_fault((uint32_t)k_npu_status_buserr_bit);
373 return;
374 }
375 s_npu.status = ((uint32_t)1U << (uint32_t)k_npu_status_cmdend_bit) |
376 ((uint32_t)1U << (uint32_t)k_npu_status_irq_bit);
377 s_npu.jobs++;
378 s_npu.last_op = c.op;
379 s_npu.last_bytes = c.count;
380 s_npu.last_check = check;
381 board_periph_icu_raise_event(uc, (uint16_t)k_npu_event_irq);
382 npu_console_job(&c, check);
383}

386static void npu_on_cmd(uc_engine* uc, uint32_t value)
387{
388 if ((value & ((uint32_t)1U << (uint32_t)k_npu_cmd_clear_irq_bit)) != 0U) {
389 s_npu.status &= ~((uint32_t)1U << (uint32_t)k_npu_status_irq_bit);
390 }
391 if ((value & ((uint32_t)1U << (uint32_t)k_npu_cmd_run_bit)) != 0U) {
392 npu_execute(uc);
393 }

397static uint64_t npu_read(uc_engine* uc, uint64_t addr, unsigned size)
398{
399 (void)uc;
400 (void)size;
401 const uint64_t off = addr - (uint64_t)k_npu_base;
402 if (off >= (uint64_t)k_npu_span) {
403 return 0U;
404 }
405 s_npu.reads++;
406 if (off == (uint64_t)k_npu_off_id) {
407 return (uint64_t)s_npu_id;
408 }
409 if (off == (uint64_t)k_npu_off_status) {
410 return (uint64_t)s_npu.status; /* STATUS.reset stays 0: reset is instant. */
411 }
412 return (uint64_t)s_npu.reg[npu_word(off)];
413}

416static void npu_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
417{
418 (void)size;
419 const uint64_t off = addr - (uint64_t)k_npu_base;
420 if (off >= (uint64_t)k_npu_span) {
421 return;
422 }
423 s_npu.writes++;
424 s_npu.reg[npu_word(off)] = (uint32_t)value;
425 if (off == (uint64_t)k_npu_off_cmd) {
426 npu_on_cmd(uc, (uint32_t)value);
427 return;
428 }
429 if (off == (uint64_t)k_npu_off_reset) {
430 s_npu.status = 0U; /* Reset discards job state; STATUS.reset reads 0. */
431 }
432}

441static void npu_report(void)
442{
443 if ((s_npu.jobs == 0U) && (s_npu.faults == 0U)) {
444 if ((s_npu.reads != 0U) || (s_npu.writes != 0U)) {
445 (void)fprintf(stderr,

447 s_npu.reads,
448 s_npu.writes);
449 }
450 return;

452 (void)fprintf(stderr,
453 " NPU(Ethos-U55): jobs=%u faults=%u last=%s bytes=%u check=0x%08X"
454 " (ra8_emulator stand-in, not Vela)\n",
455 s_npu.jobs,
456 s_npu.faults,
457 npu_op_name(s_npu.last_op),
458 s_npu.last_bytes,
459 s_npu.last_check);
