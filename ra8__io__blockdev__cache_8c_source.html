88RA8_INTERNAL
89static uint32_t cache_pick_victim(const ra8_io_blockdev_cache_state_t* st)
90{
91 uint32_t best = 0;
92 for (uint32_t i = 0; i < st->n_slots; ++i) {
93 if (!st->slots[i].valid) {
94 return i;
95 }
96 if (st->slots[i].last_use < st->slots[best].last_use) {
97 best = i;
98 }
99 }
100 return best;
101}

127RA8_INTERNAL
128static ra8_err_t cache_read_block(ra8_io_blockdev_cache_state_t* st, uint32_t lba, uint8_t* dst)
129{
130 st->clock++;
131 const uint32_t hit = cache_find(st, lba);
132 const size_t span = (size_t)k_ra8_io_block_size_bytes;
133 if (hit != st->n_slots) {
134 st->hits++;
135 (void)memcpy(dst, &st->data[(size_t)hit * span], span);
136 st->slots[hit].last_use = st->clock;
137 return k_ra8_ok;
138 }
139 st->misses++;
140 const uint32_t v = cache_pick_victim(st);
141 RA8_RETURN_ON_ERROR(ra8_io_blockdev_read(st->under, lba, 1, &st->data[(size_t)v * span]),
142 s_tag,
143 "backend read");
144 st->slots[v].lba = lba;
145 st->slots[v].valid = true;
146 st->slots[v].last_use = st->clock;
147 (void)memcpy(dst, &st->data[(size_t)v * span], span);
148 return k_ra8_ok;
149}

174RA8_INTERNAL
175static ra8_err_t
176cache_write_block(ra8_io_blockdev_cache_state_t* st, uint32_t lba, const uint8_t* src)
177{
178 RA8_RETURN_ON_ERROR(ra8_io_blockdev_write(st->under, lba, 1, src), s_tag, "backend write");
179 st->clock++;
180 const size_t span = (size_t)k_ra8_io_block_size_bytes;
181 uint32_t idx = cache_find(st, lba);
182 if (idx == st->n_slots) {
183 idx = cache_pick_victim(st);
184 }
185 (void)memcpy(&st->data[(size_t)idx * span], src, span);
186 st->slots[idx].lba = lba;
187 st->slots[idx].valid = true;
188 st->slots[idx].last_use = st->clock;
189 return k_ra8_ok;
190}

292RA8_INTERNAL
293static ra8_err_t cache_erase(void* ctx, uint32_t lba, uint32_t count)
294{
295 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
296 ra8_io_blockdev_cache_state_t* st = (ra8_io_blockdev_cache_state_t*)ctx;
297 RA8_RETURN_ON_ERROR(ra8_io_blockdev_erase(st->under, lba, count), s_tag, "backend erase");
298 const uint32_t end = lba + count;
299 for (uint32_t i = 0; i < st->n_slots; ++i) {
300 if (!st->slots[i].valid) {
301 continue;
302 }
303 if (st->slots[i].lba < lba) {
304 continue;
305 }
306 if (st->slots[i].lba >= end) {
307 continue;
308 }
309 st->slots[i].valid = false;
310 }
311 return k_ra8_ok;
312}

455 ra8_io_blockdev_cache_state_t* state,
456 const ra8_io_blockdev_t* under,

460{
461 RA8_CHECK_NULL_PTR(bd, s_tag, "bd must not be nullptr");
462 RA8_CHECK_NULL_PTR(state, s_tag, "state must not be nullptr");
463 RA8_CHECK_NULL_PTR(under, s_tag, "under must not be nullptr");
464 RA8_CHECK_NULL_PTR(data, s_tag, "data must not be nullptr");
465 RA8_CHECK_NULL_PTR(slots, s_tag, "slots must not be nullptr");
466 if (n_slots == 0U) {
467 return k_ra8_err_invalid_size;
468 }
469 cache_state_init(state, under, data, slots, n_slots);
470 cache_reset_slots(slots, n_slots);
471 bd->iface = &k_cache_iface;
472 bd->ctx = state;
473 return k_ra8_ok;
