199void internal_select_cfifo(volatile r_usb_regs_t* reg, uint16_t pipe_num, bool is_in_dir)
200{
201 /* HUM Ch 36.2.7 / 37.2.8 CFIFOSEL p 1976 / 2071. The USBHS module
202 * (IP1) requires MBW=32 -- FSP wires `USB1_CFIFO_MBW = USB_MBW_32`
203 * unconditionally for this controller. With MBW=16 the SIE refuses
204 * to arm an IN response (BSTS reads 0 and PID stays effectively
205 * NAK on the wire even though DCPCTR.PID=BUF), and the host sees
206 * unending NAKs / "device descriptor read/N, error -110". The
207 * USBFS module (IP0) keeps MBW=16. */
208 uint16_t sel = (uint16_t)(pipe_num & k_ra8_fifosel_curpipe);
209 if (internal_is_hs(reg)) {
210 sel = (uint16_t)(sel | k_ra8_fifosel_mbw_32);
211 } else {
212 sel = (uint16_t)(sel | k_ra8_fifosel_mbw_16);
213 }
214 if (is_in_dir) {
215 sel = (uint16_t)(sel | k_ra8_fifosel_isel);
216 }
217 reg->CFIFOSEL = sel;
218 /* FSP (usb_cstd_chg_curpipe) polls the CURPIPE/ISEL readback until the
219 * window switch takes effect; without this a FIFO access issued right
220 * after the select can land on the previous pipe's buffer (observed on
221 * hardware as a bulk-OUT payload stuck with INBUFM set, never sent). */
222 const uint16_t key = (uint16_t)(sel & (uint16_t)(k_ra8_fifosel_curpipe | k_ra8_fifosel_isel));
223 for (uint32_t i = 0U; i < (uint32_t)k_ra8_usb_fifosel_settle_limit; ++i) {
224 const uint16_t now =
225 (uint16_t)(reg->CFIFOSEL & (uint16_t)(k_ra8_fifosel_curpipe | k_ra8_fifosel_isel));
226 if (now == key) {
227 return;
228 }
229 }
230}

251ra8_err_t internal_wait_frdy(volatile r_usb_regs_t* reg)
252{
253 /* HUM Ch 36.2.8 "CFIFOCTR : CFIFO Port Control Register", p 1979.
254 * Loop bound is large (~10 ms ceiling at 1 GHz) because the DCP
255 * is single-buffered: between consecutive EP0 IN chunks FRDY stays
256 * low until the host actually pulls the previous chunk off the
257 * wire. See ra8_usb_internal_lim32_t for the rationale. */
258 for (uint32_t i = 0U; i < (uint32_t)k_ra8_usb_frdy_poll_limit; ++i) {
259 const bool frdy = ((reg->CFIFOCTR & k_ra8_fifoctr_frdy) != 0U);
260#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
261 /* Host MMIO fault seam, keyed on CFIFOCTR (the polled register). */
262 if (ra8_fake_mmio_wait_eval(&reg->CFIFOCTR, i, frdy)) {
263 return k_ra8_ok;
264 }
265#else
266 if (frdy) {
267 return k_ra8_ok;
268 }
269#endif
270 }
271 return k_ra8_err_hw_timeout;
272}

400uint16_t
401internal_pipecfg_word(uint8_t ep_addr, ra8_usb_ep_dir_t dir, ra8_usb_ep_type_t type, bool dblb_in)
402{
403 uint16_t cfg = (uint16_t)((uint16_t)ep_addr & k_ra8_pipecfg_epnum_mask);
404 if (dir == k_ra8_usb_ep_dir_in) {
405 cfg = (uint16_t)(cfg | k_ra8_pipecfg_dir_in);
406 }
407 if (type == k_ra8_usb_ep_type_bulk) {
408 cfg = (uint16_t)(cfg | k_ra8_pipecfg_type_bulk);
409 cfg = (uint16_t)(cfg | k_ra8_pipecfg_shtnak);
410 /* Device OUT (receive): double-buffer so the host's next packet lands
411 * in bank B while the ISR drains bank A. Single-banked OUT NAKs (and
412 * wedges) the host's next packet during the per-packet re-arm of a
413 * sustained multi-packet WRITE data phase. PIPEBUF already reserves
414 * 2*MPS, and internal_irq_complete_out drains every ready bank per
415 * ISR. Device IN stays single-banked (the free-bank handshake after
416 * an MPS-exact fill is unreliable -- see header). HUM Ch 36.2.24. */
417 if (dir == k_ra8_usb_ep_dir_out) {
418 cfg = (uint16_t)(cfg | k_ra8_pipecfg_dblb);
419 }
420 /* DBLB on bulk IN only when the caller asks (host-mode IN). */
421 if (dblb_in) {
422 if (dir == k_ra8_usb_ep_dir_in) {
423 cfg = (uint16_t)(cfg | k_ra8_pipecfg_dblb);
424 }
425 }
426 } else if (type == k_ra8_usb_ep_type_intr) {
427 cfg = (uint16_t)(cfg | k_ra8_pipecfg_type_intr);
428 } else {
429 cfg = (uint16_t)(cfg | k_ra8_pipecfg_type_iso);
430 }
431 return cfg;
432}

505RA8_INTERNAL
506static void
507internal_fifo_write_hs_tail(volatile r_usb_regs_t* reg, const uint8_t* data, uint16_t len)
508{
509 const uint16_t tail = (uint16_t)(len & 0x3U);
510 if (tail == 0U) {
511 return;
512 }
513 const uint16_t sel_save = reg->CFIFOSEL;
514 const uint16_t sel_base = (uint16_t)(sel_save & (uint16_t)~(uint16_t)k_ra8_fifosel_mbw_msk);
515 uint16_t offset = (uint16_t)(len & (uint16_t)~(uint16_t)0x3U);
516 /* For USBHS in little-endian mode, FSP writes the 16-bit residual
517 * halfword to CFIFOH (CFIFO base + 0x02) and the 8-bit residual
518 * byte to CFIFOHH (CFIFO base + 0x03), NOT to CFIFO itself. See
519 * FSP r_usb_reg_access.h: USB1_CFIFO16=USB_M1->CFIFOH and
520 * USB1_CFIFO8=USB_M1->CFIFOHH (in USB_CFG_LITTLE branch). The
521 * chip uses these address aliases to commit the trailing 0..3
522 * bytes when MBW is narrowed; a write to CFIFO at offset +0x00
523 * in MBW=16/8 mode is silently dropped on USBHS, leaving DTLN
524 * unchanged at 4N (verified empirically: 18-byte push showed
525 * DTLN=16 instead of 18, and Linux saw -EOVERFLOW). */
526 volatile uint16_t* const cfifoh = (volatile uint16_t*)((uintptr_t)&reg->CFIFO + 2U);
527 volatile uint8_t* const cfifohh = (volatile uint8_t*)((uintptr_t)&reg->CFIFO + 3U);
528 if ((tail & 0x2U) != 0U) {
529 reg->CFIFOSEL = (uint16_t)(sel_base | k_ra8_fifosel_mbw_16);
530 const uint16_t lo = (uint16_t)data[offset + 0U];
531 const uint16_t hi = (uint16_t)data[offset + 1U];
532 *cfifoh = (uint16_t)(lo | (uint16_t)(hi << k_ra8_usb_byte_bits));
533 offset = (uint16_t)(offset + 2U);
534 }
535 if ((tail & 0x1U) != 0U) {
536 reg->CFIFOSEL = (uint16_t)(sel_base | k_ra8_fifosel_mbw_8);
537 *cfifohh = data[offset];
538 }
539 reg->CFIFOSEL = sel_save;
540}

591void internal_fifo_write(volatile r_usb_regs_t* reg, const uint8_t* data, uint16_t len)
592{
593 /* HUM Ch 36.2.5 / 37.2.7 CFIFO p 1973 / 2070. CFIFO access width
594 * must match CFIFOSEL.MBW. USBHS = 32-bit, USBFS = 16-bit. */
595 if (internal_is_hs(reg)) {
596 internal_fifo_write_hs_head(reg, data, len);
597 internal_fifo_write_hs_tail(reg, data, len);
598 return;
599 }
600 /* USBFS / MBW=16 path. */
601 const uint16_t even = (uint16_t)(len >> 1U);
602 for (uint16_t i = 0U; i < even; ++i) {
603 const uint16_t lo = (uint16_t)data[(2U * i) + 0U];
604 const uint16_t hi = (uint16_t)data[(2U * i) + 1U];
605 reg->CFIFO = (uint16_t)(lo | (uint16_t)(hi << k_ra8_usb_byte_bits));
606 }
607 if ((len & 1U) != 0U) {
608 /* Trailing odd byte: switch to MBW=8 and write one byte. A 16-bit
609 * write here adds a phantom zero byte to DTLN and the host sees
610 * EOVERFLOW on every odd-length descriptor. Mirrors FSP
611 * hw_usb_write_fifo8 for USBFS. HUM Ch 36.2.6 CFIFOSEL.MBW. */
612 const uint16_t sel_save = reg->CFIFOSEL;
613 const uint16_t sel_8 =
614 (uint16_t)((sel_save & (uint16_t)~(uint16_t)k_ra8_fifosel_mbw_msk) | k_ra8_fifosel_mbw_8);
615 volatile uint8_t* const cfifo8 = (volatile uint8_t*)(uintptr_t)&reg->CFIFO;
616 reg->CFIFOSEL = sel_8;
617 *cfifo8 = data[len - 1U];
618 reg->CFIFOSEL = sel_save;
619 }
620}

666RA8_INTERNAL
667static void internal_fifo_read_hs_tail(volatile r_usb_regs_t* reg, uint8_t* data, uint16_t len)
668{
669 const uint16_t tail = (uint16_t)(len & 0x3U);
670 if (tail == 0U) {
671 return;
672 }
673 /* HUM Ch 37.2.7 CFIFO: at MBW=32 the FIFO read pointer only advances
674 * on 32-bit reads at CFIFO+0. Narrow CFIFOH/CFIFOHH reads at MBW=16/8
675 * do not advance the pointer reliably (a 3-byte bulk-OUT packet drains
676 * as bytes [0],[1],[0] when mixed). For the trailing 1..3 bytes of a
677 * non-quadword packet we keep MBW=32 and consume one final 32-bit
678 * word, extracting only the `tail` low-order bytes (the FIFO presents
679 * unused-byte slots as don't-care). One read, one pointer advance,
680 * no MBW transition. */
681 const uint32_t word = *(volatile ra8_usb_cfifo32_t*)(uintptr_t)&reg->CFIFO;
682 const uint16_t off_base = (uint16_t)(len & (uint16_t)~(uint16_t)0x3U);
683 const uint8_t bytes[4] = {
684 (uint8_t)(word & k_ra8_usb_byte_mask),
685 (uint8_t)((word >> k_ra8_usb_shift_b1) & k_ra8_usb_byte_mask),
686 (uint8_t)((word >> k_ra8_usb_shift_b2) & k_ra8_usb_byte_mask),
687 (uint8_t)((word >> k_ra8_usb_shift_b3) & k_ra8_usb_byte_mask),
688 };
689 for (uint16_t i = 0U; i < tail; ++i) {
690 data[off_base + i] = bytes[i];
691 }
692}

710void internal_fifo_read(volatile r_usb_regs_t* reg, uint8_t* data, uint16_t len)
711{
712 /* HUM Ch 37.2.7 "CFIFO : CFIFO Port Register" p 2070 */
713 if (internal_is_hs(reg)) {
714 internal_fifo_read_hs_head(reg, data, len);
715 internal_fifo_read_hs_tail(reg, data, len);
716 return;
717 }
718 /* USBFS / MBW=16 path. */
719 const uint16_t even = (uint16_t)(len >> 1U);
720 for (uint16_t i = 0U; i < even; ++i) {
721 const uint16_t word = reg->CFIFO;
722 data[(2U * i) + 0U] = (uint8_t)(word & k_ra8_usb_byte_mask);
723 data[(2U * i) + 1U] = (uint8_t)((word >> k_ra8_usb_byte_bits) & k_ra8_usb_byte_mask);
724 }
725 if ((len & 1U) != 0U) {
726 /* Trailing odd byte: switch to MBW=8 and read one byte so DTLN
727 * advances by exactly 1. Mirrors FSP hw_usb_read_fifo8 for USBFS. */
728 const uint16_t sel_save = reg->CFIFOSEL;
729 const uint16_t sel_8 =
730 (uint16_t)((sel_save & (uint16_t)~(uint16_t)k_ra8_fifosel_mbw_msk) | k_ra8_fifosel_mbw_8);
731 volatile uint8_t* const cfifo8 = (volatile uint8_t*)(uintptr_t)&reg->CFIFO;
732 reg->CFIFOSEL = sel_8;
733 data[len - 1U] = *cfifo8;
734 reg->CFIFOSEL = sel_save;
735 }

769{
770 const ra8_err_t ready = internal_wait_frdy(reg);
771 RA8_RETURN_ON_ERROR(ready, s_tag, "dcp_in_data: FRDY timeout (chunk)");

773 /* HUM Ch 36.2.8 "CFIFOCTR : CFIFO Port Control Register", p 1979.
774 *
775 * 2026-05-20 attempt: FSP r_usb_plibusbip.c only sets BVAL on the
776 * short last chunk (n < MPS), letting hardware auto-flag full-MPS
777 * chunks. Tried that here; usb_cdc_echo dropped off the bus and
778 * the chip didn't enumerate. Our synchronous multi-chunk push (vs
779 * FSP's BRDY-IRQ-per-chunk) apparently requires the explicit BVAL
780 * pulse on every chunk to commit each one before the next FRDY
781 * poll. Keep the unconditional write. */

783 return k_ra8_ok;
