270[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_sample(uint8_t* out_ok)
271{
272 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
273 if (s_ecc_buf == nullptr) {
274 return k_ra8_err_not_initialized;
275 }
276
277 for (uint32_t i = 0U; i < (uint32_t)k_ecc_demo_test_words; ++i) {
278 s_ecc_buf[i] = internal_pattern(i);
279 }
280 uint8_t rw = 1U;
281 for (uint32_t i = 0U; i < (uint32_t)k_ecc_demo_test_words; ++i) {
282 if (s_ecc_buf[i] != internal_pattern(i)) {
283 rw = 0U;
284 }
285 }
286 g_ecc_rw_ok = (uint32_t)rw;
287
288 ra8_sram_status_t st = {};
289 const ra8_err_t err = ra8_sram_get_status(&st);
290 if (err != k_ra8_ok) {
291 return err;
292 }
293 g_ecc_esr = (uint32_t)st.raw_esr;
294 g_ecc_1bit = (uint32_t)st.one_bit_mask;
295 g_ecc_2bit = (uint32_t)st.two_bit_mask;
296 /* Verdict gates only on the ECC-protected round-trip, which is
297 * deterministic. The latched 1-bit / 2-bit error masks (above) are the
298 * hardware-error-reporting payload -- meaningful only on silicon, where a
299 * real bit-flip sets them; ra8_emulator does not model ECC, so they stay out
300 * of the headless verdict (else it would flake on the unmodelled ESR). */
301 *out_ok = (rw != 0U) ? 1U : 0U;
302 return k_ra8_ok;
303}

305void main(void)
306{
307 internal_setup_or_halt();
308 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
309 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). The ECC NMI is
310 * non-maskable and unaffected; no maskable NVIC source is armed. */
311 ra8_isr_globals_enable();
312
313 if (internal_configure() != k_ra8_ok) {
314 internal_panic_halt();
315 }
316
317 while (1) {
318 uint8_t ok = 0U;
319 const ra8_err_t err = internal_sample(&ok);
320 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
321 g_ecc_ok = (uint32_t)good;
322 if (good != 0U) {
323 (void)ra8_board_uart_console_write(s_ecc_demo_ok_msg,
324 (size_t)(sizeof(s_ecc_demo_ok_msg) - 1U));
325 (void)ra8_board_led_toggle(k_ra8_board_led1);
326 } else {
327 (void)ra8_board_uart_console_write(s_ecc_demo_bad_msg,
328 (size_t)(sizeof(s_ecc_demo_bad_msg) - 1U));
329 (void)ra8_board_led_toggle(k_ra8_board_led2);
330 }
331 ++g_ecc_heartbeat;
332 ra8_delay_ms(k_ecc_demo_period_ms);
333 }
334 internal_panic_halt();
335}
