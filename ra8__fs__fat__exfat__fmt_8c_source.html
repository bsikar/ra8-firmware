148RA8_INTERNAL
149static ra8_err_t priv_exfat_geometry(uint32_t total_sectors, exfat_geom_t* g)
150{
151 g->total_sectors = total_sectors;
152 g->spc_shift = priv_exfat_spc_shift(total_sectors);
153 g->spc = 1U << g->spc_shift;
154 g->fat_offset = (uint32_t)k_exfat_fmt_boot_secs;
155 if (total_sectors <= g->fat_offset) {
156 return k_ra8_err_invalid_size;
157 }
158 uint32_t clusters = (total_sectors - g->fat_offset) / g->spc;
159 g->fat_length = 0U;
160 g->heap_offset = 0U;
161 for (uint32_t i = 0U; i < (uint32_t)k_exfat_fmt_geom_iters; i++) {
162 const uint64_t fat_bytes = ((uint64_t)clusters + 2U) * 4U;
163 const uint32_t fat_len = (uint32_t)((fat_bytes + (uint32_t)k_ra8_fs_bytes_per_sector - 1U) /
164 (uint32_t)k_ra8_fs_bytes_per_sector);
165 const uint32_t heap = g->fat_offset + fat_len;
166 if (total_sectors <= heap) {
167 return k_ra8_err_invalid_size;
168 }
169 const uint32_t next = (total_sectors - heap) / g->spc;
170 g->fat_length = fat_len;
171 g->heap_offset = heap;
172 if (next == clusters) {
173 break;
174 }
175 clusters = next;
176 }
177 g->cluster_count = (total_sectors - g->heap_offset) / g->spc;
178 const uint32_t cluster_bytes = g->spc * (uint32_t)k_ra8_fs_bytes_per_sector;
179 g->bitmap_bytes =
180 (g->cluster_count + (uint32_t)k_exfat_fmt_byte_bits - 1U) / (uint32_t)k_exfat_fmt_byte_bits;
181 g->bitmap_clusters = (g->bitmap_bytes + cluster_bytes - 1U) / cluster_bytes;
182 g->upcase_cluster = (uint32_t)k_exfat_fmt_first_clus + g->bitmap_clusters;
183 g->root_cluster = g->upcase_cluster + 1U;
184 g->used_clusters = g->bitmap_clusters + 2U; /* bitmap + up-case + root */
185 if (g->cluster_count < g->used_clusters) {
186 return k_ra8_err_invalid_size;
187 }
188 return k_ra8_ok;
189}

337RA8_INTERNAL
338static ra8_err_t priv_exfat_write_boot_tail(const ra8_fs_backend_t* backend, uint32_t cs)
339{
340 /* Sectors 1-8: extended boot (ext-boot signature only). */
341 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
342 s_scratch[i] = 0U;
343 }
344 priv_wr32(&s_scratch[k_exfat_foff_ext_sig], (uint32_t)k_exfat_fmt_ext_sig);
345 for (uint32_t s = (uint32_t)k_exfat_fmt_ext_first;
346 s < (uint32_t)k_exfat_fmt_ext_first + (uint32_t)k_exfat_fmt_ext_count;
347 s++) {
348 cs = priv_exfat_csum32(cs, s_scratch, (uint32_t)k_ra8_fs_bytes_per_sector);
349 const ra8_err_t e = priv_exfat_wr_dual(backend, s, s_scratch);
350 if (e != k_ra8_ok) {
351 return e;
352 }
353 }
354 /* Sectors 9 (OEM) + 10 (reserved): all zero. */
355 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
356 s_scratch[i] = 0U;
357 }
358 cs = priv_exfat_csum32(cs, s_scratch, (uint32_t)k_ra8_fs_bytes_per_sector);
359 ra8_err_t err = priv_exfat_wr_dual(backend, (uint32_t)k_exfat_fmt_oem_lba, s_scratch);
360 if (err != k_ra8_ok) {
361 return err;
362 }
363 cs = priv_exfat_csum32(cs, s_scratch, (uint32_t)k_ra8_fs_bytes_per_sector);
364 err = priv_exfat_wr_dual(backend, (uint32_t)k_exfat_fmt_resv_lba, s_scratch);
365 if (err != k_ra8_ok) {
366 return err;
367 }
368 /* Sector 11: every 32-bit word = the boot checksum. */
369 for (uint32_t i = 0U; i < (uint32_t)k_exfat_fmt_csum_copies; i++) {
370 priv_exfat_put32(s_scratch, i, cs);
371 }
372 return priv_exfat_wr_dual(backend, (uint32_t)k_exfat_fmt_csum_lba, s_scratch);
373}

403RA8_INTERNAL
404static ra8_err_t priv_exfat_write_boot(const ra8_fs_backend_t* backend, const exfat_geom_t* g)
405{
406 priv_exfat_build_vbr(s_scratch, g);
407 /* Boot checksum over sector 0 skips bytes 106,107 (VolumeFlags) and 112
408 * (PercentInUse): fold [0,106), [108,112), [113,512). */
409 uint32_t cs = priv_exfat_csum32(0U, &s_scratch[0], (uint32_t)k_exfat_fmt_csum_skip0);
410 cs = priv_exfat_csum32(cs,
411 &s_scratch[k_exfat_off_bps_shift],
412 (uint32_t)k_exfat_fmt_csum_skip2 - (uint32_t)k_exfat_off_bps_shift);
413 cs = priv_exfat_csum32(cs,
414 &s_scratch[(uint32_t)k_exfat_fmt_csum_skip2 + 1U],
415 (uint32_t)k_ra8_fs_bytes_per_sector -
416 ((uint32_t)k_exfat_fmt_csum_skip2 + 1U));
417 const ra8_err_t err = priv_exfat_wr_dual(backend, 0U, s_scratch);
418 if (err != k_ra8_ok) {
419 return err;
420 }
421 return priv_exfat_write_boot_tail(backend, cs);
422}

451RA8_INTERNAL
452static ra8_err_t priv_exfat_write_fat(const ra8_fs_backend_t* backend, const exfat_geom_t* g)
453{
454 ra8_err_t err = priv_fmt_clear_region(backend, g->fat_offset, g->fat_length);
455 if (err != k_ra8_ok) {
456 return err;
457 }
458 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
459 s_scratch[i] = 0U;
460 }
461 priv_exfat_put32(s_scratch, 0U, (uint32_t)k_exfat_fmt_fat_media);
462 priv_exfat_put32(s_scratch, 1U, (uint32_t)k_exfat_fmt_fat_eoc);
463 const uint32_t bmp_last = (uint32_t)k_exfat_fmt_first_clus + g->bitmap_clusters - 1U;
464 for (uint32_t c = (uint32_t)k_exfat_fmt_first_clus; c <= bmp_last; c++) {
465 const uint32_t next = (c < bmp_last) ? (c + 1U) : (uint32_t)k_exfat_fmt_fat_eoc;
466 priv_exfat_put32(s_scratch, c, next);
467 }
468 priv_exfat_put32(s_scratch, g->upcase_cluster, (uint32_t)k_exfat_fmt_fat_eoc);
469 priv_exfat_put32(s_scratch, g->root_cluster, (uint32_t)k_exfat_fmt_fat_eoc);
470 return backend->write_block(backend->ctx, g->fat_offset, 1U, s_scratch);
471}

499static ra8_err_t priv_exfat_write_bitmap(const ra8_fs_backend_t* backend, const exfat_geom_t* g)
500{
501 const uint32_t bmp_lba = g->heap_offset;
502 const uint32_t bmp_sectors = g->bitmap_clusters * g->spc;
503 ra8_err_t err = priv_fmt_clear_region(backend, bmp_lba, bmp_sectors);
504 if (err != k_ra8_ok) {
505 return err;
506 }
507 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
508 s_scratch[i] = 0U;
509 }
510 const uint32_t full = g->used_clusters / (uint32_t)k_exfat_fmt_byte_bits;
511 for (uint32_t i = 0U; i < full; i++) {
512 s_scratch[i] = (uint8_t)k_exfat_fmt_byte_full;
513 }
514 const uint32_t rem = g->used_clusters % (uint32_t)k_exfat_fmt_byte_bits;
515 if (rem != 0U) {
516 s_scratch[full] = (uint8_t)((1U << rem) - 1U);
517 }
518 return backend->write_block(backend->ctx, bmp_lba, 1U, s_scratch);
519}

553RA8_INTERNAL
554static ra8_err_t
555priv_exfat_write_upcase(const ra8_fs_backend_t* backend, const exfat_geom_t* g, uint32_t* out_csum)
556{
557 const uint32_t lba =
558 g->heap_offset + ((g->upcase_cluster - (uint32_t)k_exfat_fmt_first_clus) * g->spc);
559 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
560 s_scratch[i] = 0U;
561 }
562 uint32_t off = 0U;
563 priv_wr16(&s_scratch[off], (uint16_t)k_exfat_fmt_upc_marker);
564 off += 2U;
565 priv_wr16(&s_scratch[off], (uint16_t)k_exfat_fmt_upc_id0); /* identity 0x0000-0x0060 */
566 off += 2U;
567 for (uint16_t u = (uint16_t)k_exfat_fmt_upc_a; u <= (uint16_t)k_exfat_fmt_upc_z; u++) {
568 priv_wr16(&s_scratch[off],
569 (uint16_t)((uint16_t)k_exfat_fmt_upc_au + (u - (uint16_t)k_exfat_fmt_upc_a)));
570 off += 2U;
571 }
572 priv_wr16(&s_scratch[off], (uint16_t)k_exfat_fmt_upc_marker);
573 off += 2U;
574 priv_wr16(&s_scratch[off], (uint16_t)k_exfat_fmt_upc_id1); /* identity 0x007B-0xFFFF */
575 *out_csum = priv_exfat_csum32(0U, s_scratch, (uint32_t)k_exfat_fmt_upc_bytes);
576 return backend->write_block(backend->ctx, lba, 1U, s_scratch);
577}

648static ra8_err_t priv_exfat_write_root(const ra8_fs_backend_t* backend,
649 const exfat_geom_t* g,
650 uint32_t upcase_csum,
651 const char* label)
652{
653 const uint32_t lba =
654 g->heap_offset + ((g->root_cluster - (uint32_t)k_exfat_fmt_first_clus) * g->spc);
655 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
656 s_scratch[i] = 0U;
657 }
658 /* Allocation Bitmap entry (0x81). */
659 s_scratch[0] = (uint8_t)k_exfat_entry_bitmap;
660 priv_wr32(&s_scratch[k_exfat_de_first_clus], (uint32_t)k_exfat_fmt_first_clus);
661 priv_wr32(&s_scratch[k_exfat_de_data_len], g->bitmap_bytes);
662 /* Up-case Table entry (0x82). */
663 uint8_t* upc = &s_scratch[k_exfat_de_second];
664 upc[0] = (uint8_t)k_exfat_entry_upcase;
665 priv_wr32(&upc[k_exfat_de_upc_csum], upcase_csum);
666 priv_wr32(&upc[k_exfat_de_first_clus], g->upcase_cluster);
667 priv_wr32(&upc[k_exfat_de_data_len], (uint32_t)k_exfat_fmt_upc_bytes);
668 /* Volume Label entry (0x83). */
669 uint8_t* lbl = &s_scratch[k_exfat_de_third];
670 const uint32_t n = priv_exfat_label_utf16(&lbl[k_exfat_de_lbl_name], label);
671 lbl[0] = (uint8_t)k_exfat_entry_label;
672 lbl[k_exfat_de_lbl_cnt] = (uint8_t)n;
673 return backend->write_block(backend->ctx, lba, 1U, s_scratch);
674}

677ra8_err_t
678priv_exfat_format(const ra8_fs_backend_t* backend, uint32_t total_sectors, const char* label)
679{
680 if (total_sectors < (uint32_t)k_exfat_fmt_min_sectors) {
681 return k_ra8_err_not_supported;
682 }
683 exfat_geom_t g = {};
684 ra8_err_t err = priv_exfat_geometry(total_sectors, &g);
685 if (err != k_ra8_ok) {
686 return err;
687 }
688 /* The bitmap/up-case/root chains are seeded in FAT sector 0 only. */
689 if ((g.root_cluster * 4U) >= (uint32_t)k_ra8_fs_bytes_per_sector) {
690 return k_ra8_err_not_supported;
691 }
692 err = priv_exfat_write_boot(backend, &g);
693 if (err != k_ra8_ok) {
694 return err;
695 }
696 err = priv_exfat_write_fat(backend, &g);
697 if (err != k_ra8_ok) {
698 return err;
699 }
700 err = priv_exfat_write_bitmap(backend, &g);
701 if (err != k_ra8_ok) {
702 return err;
703 }
704 uint32_t upcase_csum = 0U;
705 err = priv_exfat_write_upcase(backend, &g, &upcase_csum);
706 if (err != k_ra8_ok) {
707 return err;
708 }
709 return priv_exfat_write_root(backend, &g, upcase_csum, label);
710}
