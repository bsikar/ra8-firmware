296RA8_INTERNAL static ra8_err_t internal_open_and_join(const ra8_net_credentials_t* credentials)
297{
298 if ((credentials == nullptr) || (credentials->ssid[0] == '\0')) {
299 return k_ra8_err_invalid_arg;
300 }
301 ra8_c6link_cfg_t link_config = {};
302 ra8_err_t err = ra8_esp_hosted_c6link_bind(&link_config.transport);
303 if (err != k_ra8_ok) {
304 return err;
305 }
306 link_config.arena = s_c6_arena;
307 link_config.arena_bytes = sizeof(s_c6_arena);
308 link_config.event_cb = internal_on_event;
309 err = ra8_c6link_open(&s_link, &link_config);
310 if (err != k_ra8_ok) {
311 return err;
312 }
313 ra8_c6link_fw_version_t version = {};
314 err = ra8_c6link_await_ready(&s_link, (uint16_t)k_ra8_c6link_announce_transfers, &version);
315 if (err == k_ra8_ok) {
316 err = ra8_c6link_wifi_start(&s_link);
317 }
318 ra8_c6link_sta_cfg_t station = {};
319 if (err == k_ra8_ok) {
320 err = ra8_c6link_sta_cfg_set(&station, credentials->ssid, credentials->psk);
321 }
322 if (err == k_ra8_ok) {
323 err = ra8_c6link_wifi_join(&s_link, &station);
324 }
325 ra8_secure_memzero(&station, sizeof(station));
326 if ((err == k_ra8_ok) && !internal_wait_connected()) {
327 err = k_ra8_err_timeout;
328 }
329 return err;

405RA8_INTERNAL static ra8_err_t internal_inflate(const void* source,
406 size_t source_bytes,
407 void* destination,
408 size_t destination_capacity,
409 size_t* output_bytes)
410{
411 if ((source == nullptr) || (destination == nullptr) || (output_bytes == nullptr)) {
412 return k_ra8_err_null_ptr;

414 const size_t inflated = tinfl_decompress_mem_to_mem(
415 destination,
416 destination_capacity,

421 return k_ra8_err_validation_failed;

558RA8_INTERNAL static ra8_err_t internal_consume(void)
559{
560 ra8_err_t err = mdl_rabook_vfs_open(&s_rabook, s_destination);
561 book_header_t header = {};
562 uint64_t flat_size = 0U;
563 if (err == k_ra8_ok) {
564 err = mdl_rabook_vfs_info(&s_rabook, &header, &flat_size);
565 }
566 uint32_t first_bytes = 0U;
567 if (err == k_ra8_ok) {
568 first_bytes = s_rabook.reader.chunk_bytes;
569 if (flat_size < first_bytes) {
570 first_bytes = (uint32_t)flat_size;
571 }
572 if ((first_bytes == 0U) || (first_bytes > sizeof(s_chunk)) ||
573 ((uint64_t)header.total_size != flat_size)) {
574 err = k_ra8_err_invalid_size;
575 }
576 }
577 if (err == k_ra8_ok) {
578 err = mdl_rabook_vfs_read_chunk(&s_rabook, 0U, s_chunk, first_bytes);
579 }
580 const ra8_err_t close_err = mdl_rabook_vfs_close(&s_rabook);
581 return ((err == k_ra8_ok) ? close_err : err);
582}

631RA8_INTERNAL static bool internal_run(ra8_net_credentials_t* configuration)
632{
633 if (s_port_err != k_ra8_ok) {
634 return internal_fail("c6 port", s_port_err);
635 }
636 if ((configuration == nullptr) || (configuration->url[0] == '\0')) {
637 return internal_fail("runtime configuration", k_ra8_err_invalid_arg);
638 }
639 ra8_err_t err = internal_mount_sd();
640 if (err != k_ra8_ok) {
641 return internal_fail("sd mount", err);
642 }
643 const ra8_rsip_config_t rsip_config = {.run_bist = true};
644 err = ra8_rsip_init(&rsip_config);
645 if (err != k_ra8_ok) {
646 return internal_fail("rsip", err);
647 }
648 ra8_delay_ms((uint32_t)k_media_c6_boot_wait_ms);
649 err = internal_open_and_join(configuration);
650 if (err != k_ra8_ok) {
651 return internal_fail("c6 join", err);
652 }
653 ra8_secure_memzero(configuration->ssid, sizeof(configuration->ssid));
654 ra8_secure_memzero(configuration->psk, sizeof(configuration->psk));
655 configuration->ssid_len = 0U;
656 configuration->psk_len = 0U;
657 err = internal_transfer_and_consume(configuration);
658 (void)ra8_c6link_close(&s_link);
659 if (err != k_ra8_ok) {
660 return internal_fail("transfer/consume", err);
661 }
662 internal_puts("media_download: PASS C6->RABOOK->SD publish and read\r\n");
663 return true;
664}
