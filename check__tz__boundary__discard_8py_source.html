4"""Ban silent error discards at TrustZone boot boundaries.
5
6C23 makes an explicit ``(void)`` cast the sanctioned suppression for a
7``[[nodiscard]]`` diagnostic, so ``-Wall -Wextra -Werror`` can never flag
8
9 (void)ra8_cgc_init(); /* discarded right before BLXNS */
10
11even though the callee is ``[[nodiscard]] ra8_err_t``. That exact pattern
12shipped a Secure boot that would BLXNS into the Non-secure image on a dead
13clock tree (tracker issue #191, T2-05), and it recurred after being fixed
14once because per-app boot files are copied from siblings. The compiler is
15structurally unable to police the void-cast; this gate is the backstop.
16
17Rules (first-party C/C++ only):
18
19 A. The TrustZone world-switch API family is must-handle EVERYWHERE: a
20 ``(void)``-cast discard of any ``ra8_tz_secure_boot_*()`` call is
21 banned in every first-party file. A returned error from that family
22 is a root-of-trust DENIAL or a rejected NS vector table -- ignoring
23 it means treating an unauthenticated NS world as live.
24
25 B. Inside a boot translation unit -- any ``.c`` that defines
26 ``SystemInit`` or ``ra8_trustzone_init`` -- a ``(void)``-cast discard
27 of ANY ``ra8_*()`` call is banned. Every fallible step in those TUs
28 gates the S->NS world switch; a tolerated failure must be an explicit
29 ``if`` (halt, or a documented S-side fallback) or
30 ``RA8_ERROR_CHECK`` / ``RA8_ERROR_CHECK_NO_ABORT``, never a silent
31 cast.
32
33Scope:
34 C / C++ sources (.c .h .cpp .hpp) under libs/, src/, tests/, examples/,
35 port/, tools/. Vendored SOUP (libs/third_party/*) and generated font
36 data (libs/ra8_fonts/*) are exempt, as are build trees.
37
38Exemptions:
39 * Comment lines / prose (a match whose cast sits inside a // or /*
40 comment does not count).
41 * Function-pointer discards like ``(void)ra8_trustzone_init;`` (no call
42 parentheses) -- those silence -Wunused, they do not discard an error.
43 * Any line carrying `TZ-DISCARD-OK: <reason>` (reason text required).
44
45Exit 0 when clean, 1 on any discard, 2 when the whole-tree sweep collapses
46below FILE_FLOOR (see the constant).
47"""

59ROOTS = ("libs", "src", "tests", "examples", "port", "tools")
60EXTS = (".c", ".h", ".cpp", ".hpp")
61EXEMPT_DIRS = ("/third_party/", "/ra8_fonts/")
62
63# Rule A: the world-switch family, discarded anywhere.
64FAMILY_RE = re.compile(r"\‍(\s*void\s*\‍)\s*(ra8_tz_secure_boot_[a-z0-9_]+)\s*\‍(")
65# Rule B: any ra8_ call, discarded inside a boot TU.
66ANY_RA8_RE = re.compile(r"\‍(\s*void\s*\‍)\s*(ra8_[a-z0-9_]+)\s*\‍(")
67# Boot-TU marker: a definition (or same-file declaration) of one of the two
68# canonical boot entry points every app's boot files provide.
69BOOT_TU_RE = re.compile(
70 r"^\s*void\s+(?:SystemInit|ra8_trustzone_init)\s*\‍(\s*void\s*\‍)", re.MULTILINE
71)
72WAIVER_RE = re.compile(r"TZ-DISCARD-OK:\s*\S")
73
74# A tree this size cannot legitimately collapse to a handful of files. If the
75# whole-tree sweep returns less than this, something broke (a bad cwd -- ROOTS
76# are relative, so this walks nothing when run from elsewhere -- or a renamed
77# root) and reporting "clean" would be a lie: no boot TU would be read, so no
78# discard could ever be reported. Measured 2026-07-28: 2125 first-party C/C++
79# files. Same trip-wire as check_ruff.py.
80FILE_FLOOR = 1700

83def _is_comment_pos(line: str, pos: int) -> bool:
84 """True if column `pos` sits inside a // or /* comment on this line."""
85 stripped = line.lstrip()
86 if stripped.startswith(("*", "//", "/*")):
87 return True
88 before = line[:pos]
89 if "//" in before:
90 return True
91 # crude single-line /* ... */ check
92 return "/*" in before and "*/" not in before

95def discover() -> list[str]:
96 """Every boot-boundary source file, for the whole-tree sweep."""
97 out: list[str] = []
98 for root in ROOTS:
99 base = Path(root)
100 if not base.is_dir():
101 continue
102 for ext in EXTS:
103 for f in base.rglob(f"*{ext}"):
104 s = str(f)
105 if is_build_output_path(s) or any(d in s for d in EXEMPT_DIRS):
106 continue
107 out.append(s)
108 return out

111def check_file(path: str) -> list[tuple[int, str, str]]:
112 """Return (line, rule, snippet) findings for one file."""
113 findings: list[tuple[int, str, str]] = []
114 try:
115 text = Path(path).read_text(encoding="utf-8")
116 except (OSError, UnicodeDecodeError):
117 return findings
118 if "(void)" not in text.replace(" ", ""):
119 return findings
120 boot_tu = path.endswith(".c") and bool(BOOT_TU_RE.search(text))
121 rules: list[tuple[re.Pattern[str], str]] = [(FAMILY_RE, "A")]
122 if boot_tu:
123 rules.append((ANY_RA8_RE, "B"))
124 lines = text.splitlines()
125 for i, raw in enumerate(lines, 1):
126 # A discard split across lines: `(void)` alone, call on the next line.
127 line = raw
128 if re.search(r"\‍(\s*void\s*\‍)\s*$", raw) and i < len(lines):
129 line = raw + lines[i]
130 if WAIVER_RE.search(line):
131 continue
132 seen: set[tuple[int, int]] = set()
133 for rule_re, rule in rules:
134 for m in rule_re.finditer(line):
135 span = (m.start(), m.end())
136 if span in seen or _is_comment_pos(line, m.start()):
137 continue
138 seen.add(span)
139 findings.append((i, rule, raw.strip()[:100]))
140 return findings

143def main() -> int:
144 """Fail when a TrustZone boot-boundary call discards its ra8_err_t.
145
146 These call sites are the worst place in the tree to drop a status: a
147 failed SAU or MPU configuration returns an error and then, ignored, hands
148 control to the non-secure world with the boundary not actually in force.
149 The build links, the board boots, and the isolation is simply absent.
150
151 FILE_FLOOR is enforced on the whole-tree sweep only, and exits 2 below it.
152 An explicit argv file list is a deliberately narrowed scope -- not a
153 collapsed one -- so it is exempt; the sweep is where a broken enumeration
154 would silently report the tree clean.
155
156 Returns 1 listing each discard, 0 when clean, 2 when the whole-tree sweep
157 enumerated too few files to trust.
158 """
159 args = [a for a in sys.argv[1:] if not a.startswith("-")]
160 whole_tree = not args
161 files = args or discover()
162 if whole_tree and len(files) < FILE_FLOOR:
163 print(
164 f"check_tz_boundary_discard.py: FATAL -- only {len(files)} first-party source "
165 f"file(s) in scope, floor is {FILE_FLOOR}. A collapsed sweep reports a "
166 "clean tree because it scanned nothing.",
167 file=sys.stderr,
168 )
169 return 2
170 total = 0
171 for path in sorted(set(files)):
172 if (
173 not path.endswith(EXTS)
174 or is_build_output_path(path)
175 or any(d in path for d in EXEMPT_DIRS)
176 ):
177 continue
178 for ln, rule, snippet in check_file(path):
179 what = (
180 "world-switch result discarded" if rule == "A" else "boot-TU ra8_* result discarded"
181 )
182 print(
183 f"{path}:{ln}: [rule {rule}] {what} -- handle the ra8_err_t "
184 f"(halt or a documented fallback; RA8_ERROR_CHECK[_NO_ABORT]); "
185 f"never (void)-cast it at a TrustZone boot boundary; {snippet}"
186 )
187 total += 1
188 if total:
189 print(
190 f"\ncheck_tz_boundary_discard: {total} violation(s). A (void)-cast "
191 f"silences [[nodiscard]] by ISO C23 rule, so -Werror cannot catch "
192 f"these; check the result and fail safe instead (add "
193 f"`TZ-DISCARD-OK: <reason>` only for a justified exception)."
194 )
195 return 1
196 print(
197 "check_tz_boundary_discard: clean -- no silent ra8_err_t discards at "
198 "TrustZone boot boundaries."
199 )
200 return 0
