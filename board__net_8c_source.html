190static void put32(uint8_t* p, uint32_t v)
191{
192 p[0] = (uint8_t)(v >> (uint32_t)k_shift24);
193 p[1] = (uint8_t)((v >> 16) & (uint32_t)k_byte_mask);

207 return ((uint32_t)p[0] << (uint32_t)k_shift24) | ((uint32_t)p[1] << 16) | ((uint32_t)p[2] << 8) |

212static uint16_t net_checksum(const uint8_t* d, uint32_t len, uint32_t seed)
213{
214 uint32_t sum = seed;
215 for (uint32_t i = 0U; (i + 1U) < len; i += 2U) {
216 sum += ((uint32_t)d[i] << 8) | (uint32_t)d[i + 1U];
217 }
218 if ((len & 1U) != 0U) {
219 sum += (uint32_t)d[len - 1U] << 8;
220 }
221 while ((sum >> 16) != 0U) {
222 sum = (sum & (uint32_t)k_u16_mask) + (sum >> 16);
223 }
224 return (uint16_t)(~sum & (uint32_t)k_u16_mask);
225}

253static void net_send_arp_request(void)
254{
255 static const uint8_t bcast[k_mac_len] = {(uint8_t)k_byte_mask,
256 (uint8_t)k_byte_mask,
257 (uint8_t)k_byte_mask,
258 (uint8_t)k_byte_mask,
259 (uint8_t)k_byte_mask,
260 (uint8_t)k_byte_mask};
261 uint8_t f[k_eth_hdr + k_arp_len];
262 (void)memset(f, 0, sizeof(f));
263 net_eth_hdr(f, bcast, (uint16_t)k_eth_arp);
264 uint8_t* a = &f[k_eth_hdr];
265 put16(&a[0], 1U); /* htype = Ethernet. */
266 put16(&a[2], (uint16_t)k_eth_ipv4); /* ptype = IPv4. */
267 a[4] = (uint8_t)k_mac_len;
268 a[k_arp_plen_off] = (uint8_t)k_arp_plen;
269 put16(&a[6], 1U); /* op = request. */
270 (void)memcpy(&a[8], s_peer_mac, k_mac_len);
271 put32(&a[k_arp_spa_off], (uint32_t)k_net_peer_ip);
272 put32(&a[k_arp_tpa_off], (uint32_t)k_net_fw_ip);
273 net_queue(f, sizeof(f));
274}

301static void net_ip_hdr(uint8_t* ip, uint8_t proto, uint16_t payload_len)
302{
303 (void)memset(ip, 0, k_ip_hdr);
304 ip[0] = (uint8_t)k_ipv4_ver_ihl; /* version 4, IHL 5. */
305 put16(&ip[2], (uint16_t)(k_ip_hdr + payload_len));
306 put16(&ip[4], (uint16_t)k_ip_ident); /* identification. */
307 put16(&ip[6], (uint16_t)k_ip_flag_df); /* don't fragment. */
308 ip[8] = (uint8_t)k_ip_ttl; /* TTL. */
309 ip[k_ip_proto_off] = proto;
310 put32(&ip[k_ip_src_off], (uint32_t)k_net_peer_ip);
311 put32(&ip[k_ip_dst_off], (uint32_t)k_net_fw_ip);
312 put16(&ip[k_ip_csum_off], net_checksum(ip, k_ip_hdr, 0U));
313}

316static void net_send_ping(void)
317{
318 if (!s_fw_mac_known) {
319 return;
320 }
321 uint8_t f[k_eth_hdr + k_ip_hdr + k_icmp_hdr + 16U];
322 (void)memset(f, 0, sizeof(f));
323 net_eth_hdr(f, s_fw_mac, (uint16_t)k_eth_ipv4);
324 const uint16_t icmp_len = (uint16_t)(k_icmp_hdr + 16U);
325 net_ip_hdr(&f[k_eth_hdr], (uint8_t)k_ip_proto_icmp, icmp_len);
326 uint8_t* ic = &f[k_eth_hdr + k_ip_hdr];
327 ic[0] = 8U; /* echo request. */
328 s_ping_seq++;
329 put16(&ic[4], (uint16_t)k_icmp_ident); /* identifier. */
330 put16(&ic[6], s_ping_seq);
331 for (uint32_t i = 0U; i < 16U; i++) {
332 ic[k_icmp_hdr + i] = (uint8_t)((uint32_t)k_icmp_pat_base + i); /* payload pattern. */
333 }
334 put16(&ic[2], net_checksum(ic, icmp_len, 0U));
335 net_queue(f, sizeof(f));
336}

344static void net_send_tcp(uint8_t flags, const uint8_t* payload, uint16_t payload_len)
345{
346 if (!s_fw_mac_known) {
347 return;
348 }
349 uint8_t f[k_eth_hdr + k_ip_hdr + k_tcp_hdr + k_tcp_payload_max];
350 const uint16_t tcp_len = (uint16_t)((uint16_t)k_tcp_hdr + payload_len);
351 if (((uint32_t)k_eth_hdr + (uint32_t)k_ip_hdr + (uint32_t)tcp_len) > sizeof(f)) {
352 return;
353 }
354 (void)memset(f, 0, sizeof(f));
355 net_eth_hdr(f, s_fw_mac, (uint16_t)k_eth_ipv4);
356 net_ip_hdr(&f[k_eth_hdr], (uint8_t)k_ip_proto_tcp, tcp_len);
357 uint8_t* t = &f[k_eth_hdr + k_ip_hdr];
358 put16(&t[0], (uint16_t)k_net_peer_port);
359 put16(&t[2], (uint16_t)k_net_echo_port);
360 put32(&t[4], s_tcp_our_seq);
361 put32(&t[8], s_tcp_their_seq);
362 t[k_tcp_off_dataoff] = (uint8_t)k_tcp_data_off; /* data offset = 5 32-bit words. */
363 t[k_tcp_off_flags] = flags;
364 put16(&t[k_tcp_off_window], (uint16_t)k_tcp_window); /* window. */
365 if (payload_len > 0U) {
366 (void)memcpy(&t[k_tcp_off_payload], payload, payload_len);
367 }
368 /* TCP checksum covers the IPv4 pseudo-header + the segment. */
369 const uint32_t pseudo =
370 ((uint32_t)k_net_peer_ip >> 16) + ((uint32_t)k_net_peer_ip & (uint32_t)k_u16_mask) +
371 ((uint32_t)k_net_fw_ip >> 16) + ((uint32_t)k_net_fw_ip & (uint32_t)k_u16_mask) +
372 (uint32_t)k_ip_proto_tcp + (uint32_t)tcp_len;
373 put16(&t[16], net_checksum(t, tcp_len, pseudo));
374 net_queue(f, (uint32_t)k_eth_hdr + (uint32_t)k_ip_hdr + (uint32_t)tcp_len);
375}

409static void net_rx_tcp(const uint8_t* t, uint32_t len)
410{
411 if (len < (uint32_t)k_tcp_hdr) {
412 return;
413 }
414 if ((get16(&t[0]) != (uint16_t)k_net_echo_port) || (get16(&t[2]) != (uint16_t)k_net_peer_port)) {
415 return;
416 }
417 const uint32_t seq = get32(&t[4]);
418 const uint8_t flags = t[13];
419 const uint32_t doff =
420 (uint32_t)((t[k_tcp_off_dataoff] >> (uint32_t)k_tcp_doff_shift) & (uint32_t)k_ip_ihl_mask) *
421 (uint32_t)k_ihl_word;
422 if ((doff < (uint32_t)k_tcp_hdr) || (doff > len)) {
423 return;
424 }
425 const uint32_t plen = len - doff;
426 const uint8_t* pdata = &t[doff];
427 if (s_trace) {
428 (void)fprintf(stderr,
429 " [net] RX TCP flags=0x%02X seq=%u ack=%u plen=%u (state %u)\n",
430 (unsigned)flags,
431 seq,
432 get32(&t[8]),
433 plen,
434 (unsigned)s_state);
435 }
436
437 if ((s_state == (uint8_t)k_net_syn) && ((flags & (uint8_t)k_tcp_syn) != 0U) &&
438 ((flags & (uint8_t)k_tcp_ack) != 0U)) {
439 s_tcp_their_seq = seq + 1U; /* their SYN consumes one sequence number. */
440 s_tcp_our_seq += 1U; /* our SYN consumed one. */
441 net_send_tcp((uint8_t)k_tcp_ack, nullptr, 0U);
442 /* Defer the payload a few ticks so the firmware's accept() binds the socket
443 * and the echo thread is waiting in receive() before the data arrives. */
444 s_tcp_need_data = true;
445 s_tcp_estab_wait = 0U;
446 s_state = (uint8_t)k_net_estab;
447 return;
448 }
449 if ((s_state == (uint8_t)k_net_estab) && (plen > 0U)) {
450 net_tcp_on_estab(seq, pdata, plen);
451 return;
452 }
453 if ((s_state == (uint8_t)k_net_fin) && ((flags & (uint8_t)k_tcp_fin) != 0U)) {
454 s_tcp_their_seq = seq + plen + 1U; /* their FIN consumes one. */
455 net_send_tcp((uint8_t)k_tcp_ack, nullptr, 0U);
456 s_state = (uint8_t)k_net_done;
457 return;
458 }
459}

467static void net_rx_arp(const uint8_t* a, uint32_t len)
468{
469 if (len < (uint32_t)k_arp_len) {
470 return;
471 }
472 const uint16_t op = get16(&a[6]);
473 const uint32_t spa = get32(&a[14]);
474 const uint32_t tpa = get32(&a[24]);
475 if (spa == (uint32_t)k_net_fw_ip) {
476 (void)memcpy(s_fw_mac, &a[8], k_mac_len); /* sender HW = firmware MAC. */
477 s_fw_mac_known = true;
478 if (op == 2U) {
479 s_arp_replies++;
480 }
481 }
482 if ((op == 1U) && (tpa == (uint32_t)k_net_peer_ip)) {
483 net_send_arp_reply(&a[8], spa); /* firmware asked who-has us. */
484 }
485 if (s_fw_mac_known && (s_state == (uint8_t)k_net_arp)) {
486 net_send_ping();
487 s_state = (uint8_t)k_net_ping;
488 s_wait = 0U;
489 }

510static void net_rx_ipv4(const uint8_t* ip, uint32_t len)
511{
512 if (len < (uint32_t)k_ip_hdr) {
513 return;
514 }
515 const uint32_t ihl = (uint32_t)(ip[0] & (uint32_t)k_ip_ihl_mask) * (uint32_t)k_ihl_word;
516 if ((ihl < (uint32_t)k_ip_hdr) || (ihl > len)) {
517 return;
518 }
519 /* Use the IP total-length field, not the frame length: a short frame is padded
520 * to the 60-byte Ethernet minimum, and that padding must not be mistaken for
521 * upper-layer payload (e.g. a bare TCP ACK would otherwise look like 6 data
522 * bytes). */
523 uint32_t actual = len;
524 const uint16_t ip_total = get16(&ip[2]);
525 if (((uint32_t)ip_total >= ihl) && ((uint32_t)ip_total <= len)) {
526 actual = (uint32_t)ip_total;
527 }
528 const uint8_t proto = ip[9];
529 if (proto == (uint8_t)k_ip_proto_icmp) {
530 net_rx_icmp(&ip[ihl], actual - ihl);
531 } else if (proto == (uint8_t)k_ip_proto_tcp) {
532 net_rx_tcp(&ip[ihl], actual - ihl);
533 }
534}

560uint32_t board_net_poll_rx(uint8_t* buf, uint32_t max)
561{
562 s_polls++;
563 if (s_rxq_head == s_rxq_tail) {
564 return 0U; /* ring empty. */
565 }
566 const uint32_t n = s_rxq_len[s_rxq_head];
567 if (n > max) {
568 return 0U;
569 }
570 (void)memcpy(buf, s_rxq[s_rxq_head], n);
571 s_rxq_head = (s_rxq_head + 1U) % (uint32_t)k_net_qdepth;
572 s_delivered++;
573 /* Console NET tab: one line per frame the peer delivers to the firmware. */
574 char ln[k_net_console_line_cap];
575 (void)snprintf(ln, sizeof(ln), "NET rx %uB (#%u)", (unsigned)n, (unsigned)s_delivered);
576 board_console_push(k_board_console_ch_net, ln);
577 return n;
578}

580void board_net_tick(void)
581{
582 s_wait++;
583 if (s_state == (uint8_t)k_net_init) {
584 net_send_arp_request();
585 s_state = (uint8_t)k_net_arp;
586 s_wait = 0U;
587 return;
588 }
589 if ((s_state == (uint8_t)k_net_estab) && s_tcp_need_data) {
590 enum : uint32_t { k_net_data_delay = 800U };
591 s_tcp_estab_wait++;
592 if (s_tcp_estab_wait > k_net_data_delay) {
593 net_send_data(); /* connection settled; send the echo payload. */
594 s_tcp_need_data = false;
595 }
596 return;
597 }
598 /* Retransmit the pending step if the firmware has not answered for a while
599 * (the stack may still be bringing the interface up on the first attempts). */
600 enum : uint32_t { k_net_retry = 2000U };
601 if (s_wait > k_net_retry) {
602 s_wait = 0U;
603 if (s_state == (uint8_t)k_net_arp) {
604 net_send_arp_request();
605 } else if (s_state == (uint8_t)k_net_ping) {
606 net_send_ping();
607 } else if (s_state == (uint8_t)k_net_syn) {
608 net_send_syn();
609 }
610 }
611}
