189static void put32(uint8_t* p, uint32_t v)
190{
191 p[0] = (uint8_t)(v >> (uint32_t)k_shift24);
192 p[1] = (uint8_t)((v >> 16) & (uint32_t)k_byte_mask);

205{
206 return ((uint32_t)p[0] << (uint32_t)k_shift24) | ((uint32_t)p[1] << 16) | ((uint32_t)p[2] << 8) |

211static uint16_t net_checksum(const uint8_t* d, uint32_t len, uint32_t seed)
212{
213 uint32_t sum = seed;
214 for (uint32_t i = 0U; (i + 1U) < len; i += 2U) {
215 sum += ((uint32_t)d[i] << 8) | (uint32_t)d[i + 1U];
216 }
217 if ((len & 1U) != 0U) {
218 sum += (uint32_t)d[len - 1U] << 8;
219 }
220 while ((sum >> 16) != 0U) {
221 sum = (sum & (uint32_t)k_u16_mask) + (sum >> 16);
222 }
223 return (uint16_t)(~sum & (uint32_t)k_u16_mask);
224}

252static void net_send_arp_request(void)
253{
254 static const uint8_t bcast[k_mac_len] = {(uint8_t)k_byte_mask,
255 (uint8_t)k_byte_mask,
256 (uint8_t)k_byte_mask,
257 (uint8_t)k_byte_mask,
258 (uint8_t)k_byte_mask,
259 (uint8_t)k_byte_mask};
260 uint8_t f[k_eth_hdr + k_arp_len];
261 (void)memset(f, 0, sizeof(f));
262 net_eth_hdr(f, bcast, (uint16_t)k_eth_arp);
263 uint8_t* a = &f[k_eth_hdr];
264 put16(&a[0], 1U); /* htype = Ethernet. */
265 put16(&a[2], (uint16_t)k_eth_ipv4); /* ptype = IPv4. */
266 a[4] = (uint8_t)k_mac_len;
267 a[k_arp_plen_off] = (uint8_t)k_arp_plen;
268 put16(&a[6], 1U); /* op = request. */
269 (void)memcpy(&a[8], s_peer_mac, k_mac_len);
270 put32(&a[k_arp_spa_off], (uint32_t)k_net_peer_ip);
271 put32(&a[k_arp_tpa_off], (uint32_t)k_net_fw_ip);
272 net_queue(f, sizeof(f));
273}

300static void net_ip_hdr(uint8_t* ip, uint8_t proto, uint16_t payload_len)
301{
302 (void)memset(ip, 0, k_ip_hdr);
303 ip[0] = (uint8_t)k_ipv4_ver_ihl; /* version 4, IHL 5. */
304 put16(&ip[2], (uint16_t)(k_ip_hdr + payload_len));
305 put16(&ip[4], (uint16_t)k_ip_ident); /* identification. */
306 put16(&ip[6], (uint16_t)k_ip_flag_df); /* don't fragment. */
307 ip[8] = (uint8_t)k_ip_ttl; /* TTL. */
308 ip[k_ip_proto_off] = proto;
309 put32(&ip[k_ip_src_off], (uint32_t)k_net_peer_ip);
310 put32(&ip[k_ip_dst_off], (uint32_t)k_net_fw_ip);
311 put16(&ip[k_ip_csum_off], net_checksum(ip, k_ip_hdr, 0U));
312}

315static void net_send_ping(void)
316{
317 if (!s_fw_mac_known) {
318 return;
319 }
320 uint8_t f[k_eth_hdr + k_ip_hdr + k_icmp_hdr + 16U];
321 (void)memset(f, 0, sizeof(f));
322 net_eth_hdr(f, s_fw_mac, (uint16_t)k_eth_ipv4);
323 const uint16_t icmp_len = (uint16_t)(k_icmp_hdr + 16U);
324 net_ip_hdr(&f[k_eth_hdr], (uint8_t)k_ip_proto_icmp, icmp_len);
325 uint8_t* ic = &f[k_eth_hdr + k_ip_hdr];
326 ic[0] = 8U; /* echo request. */
327 s_ping_seq++;
328 put16(&ic[4], (uint16_t)k_icmp_ident); /* identifier. */
329 put16(&ic[6], s_ping_seq);
330 for (uint32_t i = 0U; i < 16U; i++) {
331 ic[k_icmp_hdr + i] = (uint8_t)((uint32_t)k_icmp_pat_base + i); /* payload pattern. */
332 }
333 put16(&ic[2], net_checksum(ic, icmp_len, 0U));
334 net_queue(f, sizeof(f));
335}

343static void net_send_tcp(uint8_t flags, const uint8_t* payload, uint16_t payload_len)
344{
345 if (!s_fw_mac_known) {
346 return;
347 }
348 uint8_t f[k_eth_hdr + k_ip_hdr + k_tcp_hdr + k_tcp_payload_max];
349 const uint16_t tcp_len = (uint16_t)((uint16_t)k_tcp_hdr + payload_len);
350 if (((uint32_t)k_eth_hdr + (uint32_t)k_ip_hdr + (uint32_t)tcp_len) > sizeof(f)) {
351 return;
352 }
353 (void)memset(f, 0, sizeof(f));
354 net_eth_hdr(f, s_fw_mac, (uint16_t)k_eth_ipv4);
355 net_ip_hdr(&f[k_eth_hdr], (uint8_t)k_ip_proto_tcp, tcp_len);
356 uint8_t* t = &f[k_eth_hdr + k_ip_hdr];
357 put16(&t[0], (uint16_t)k_net_peer_port);
358 put16(&t[2], (uint16_t)k_net_echo_port);
359 put32(&t[4], s_tcp_our_seq);
360 put32(&t[8], s_tcp_their_seq);
361 t[k_tcp_off_dataoff] = (uint8_t)k_tcp_data_off; /* data offset = 5 32-bit words. */
362 t[k_tcp_off_flags] = flags;
363 put16(&t[k_tcp_off_window], (uint16_t)k_tcp_window); /* window. */
364 if (payload_len > 0U) {
365 (void)memcpy(&t[k_tcp_off_payload], payload, payload_len);
366 }
367 /* TCP checksum covers the IPv4 pseudo-header + the segment. */
368 const uint32_t pseudo =
369 ((uint32_t)k_net_peer_ip >> 16) + ((uint32_t)k_net_peer_ip & (uint32_t)k_u16_mask) +
370 ((uint32_t)k_net_fw_ip >> 16) + ((uint32_t)k_net_fw_ip & (uint32_t)k_u16_mask) +
371 (uint32_t)k_ip_proto_tcp + (uint32_t)tcp_len;
372 put16(&t[16], net_checksum(t, tcp_len, pseudo));
373 net_queue(f, (uint32_t)k_eth_hdr + (uint32_t)k_ip_hdr + (uint32_t)tcp_len);
374}

408static void net_rx_tcp(const uint8_t* t, uint32_t len)
409{
410 if (len < (uint32_t)k_tcp_hdr) {
411 return;
412 }
413 if ((get16(&t[0]) != (uint16_t)k_net_echo_port) || (get16(&t[2]) != (uint16_t)k_net_peer_port)) {
414 return;
415 }
416 const uint32_t seq = get32(&t[4]);
417 const uint8_t flags = t[13];
418 const uint32_t doff =
419 (uint32_t)((t[k_tcp_off_dataoff] >> (uint32_t)k_tcp_doff_shift) & (uint32_t)k_ip_ihl_mask) *
420 (uint32_t)k_ihl_word;
421 if ((doff < (uint32_t)k_tcp_hdr) || (doff > len)) {
422 return;
423 }
424 const uint32_t plen = len - doff;
425 const uint8_t* pdata = &t[doff];
426 if (s_trace) {
427 (void)fprintf(stderr,
428 " [net] RX TCP flags=0x%02X seq=%u ack=%u plen=%u (state %u)\n",
429 (unsigned)flags,
430 seq,
431 get32(&t[8]),
432 plen,
433 (unsigned)s_state);
434 }
435
436 if ((s_state == (uint8_t)k_net_syn) && ((flags & (uint8_t)k_tcp_syn) != 0U) &&
437 ((flags & (uint8_t)k_tcp_ack) != 0U)) {
438 s_tcp_their_seq = seq + 1U; /* their SYN consumes one sequence number. */
439 s_tcp_our_seq += 1U; /* our SYN consumed one. */
440 net_send_tcp((uint8_t)k_tcp_ack, nullptr, 0U);
441 /* Defer the payload a few ticks so the firmware's accept() binds the socket
442 * and the echo thread is waiting in receive() before the data arrives. */
443 s_tcp_need_data = true;
444 s_tcp_estab_wait = 0U;
445 s_state = (uint8_t)k_net_estab;
446 return;
447 }
448 if ((s_state == (uint8_t)k_net_estab) && (plen > 0U)) {
449 net_tcp_on_estab(seq, pdata, plen);
450 return;
451 }
452 if ((s_state == (uint8_t)k_net_fin) && ((flags & (uint8_t)k_tcp_fin) != 0U)) {
453 s_tcp_their_seq = seq + plen + 1U; /* their FIN consumes one. */
454 net_send_tcp((uint8_t)k_tcp_ack, nullptr, 0U);
455 s_state = (uint8_t)k_net_done;
456 return;
457 }
458}

466static void net_rx_arp(const uint8_t* a, uint32_t len)
467{
468 if (len < (uint32_t)k_arp_len) {
469 return;
470 }
471 const uint16_t op = get16(&a[6]);
472 const uint32_t spa = get32(&a[14]);
473 const uint32_t tpa = get32(&a[24]);
474 if (spa == (uint32_t)k_net_fw_ip) {
475 (void)memcpy(s_fw_mac, &a[8], k_mac_len); /* sender HW = firmware MAC. */
476 s_fw_mac_known = true;
477 if (op == 2U) {
478 s_arp_replies++;
479 }
480 }
481 if ((op == 1U) && (tpa == (uint32_t)k_net_peer_ip)) {
482 net_send_arp_reply(&a[8], spa); /* firmware asked who-has us. */
483 }
484 if (s_fw_mac_known && (s_state == (uint8_t)k_net_arp)) {
485 net_send_ping();
486 s_state = (uint8_t)k_net_ping;
487 s_wait = 0U;
488 }

509static void net_rx_ipv4(const uint8_t* ip, uint32_t len)
510{
511 if (len < (uint32_t)k_ip_hdr) {
512 return;
513 }
514 const uint32_t ihl = (uint32_t)(ip[0] & (uint32_t)k_ip_ihl_mask) * (uint32_t)k_ihl_word;
515 if ((ihl < (uint32_t)k_ip_hdr) || (ihl > len)) {
516 return;
517 }
518 /* Use the IP total-length field, not the frame length: a short frame is padded
519 * to the 60-byte Ethernet minimum, and that padding must not be mistaken for
520 * upper-layer payload (e.g. a bare TCP ACK would otherwise look like 6 data
521 * bytes). */
522 uint32_t actual = len;
523 const uint16_t ip_total = get16(&ip[2]);
524 if (((uint32_t)ip_total >= ihl) && ((uint32_t)ip_total <= len)) {
525 actual = (uint32_t)ip_total;
526 }
527 const uint8_t proto = ip[9];
528 if (proto == (uint8_t)k_ip_proto_icmp) {
529 net_rx_icmp(&ip[ihl], actual - ihl);
530 } else if (proto == (uint8_t)k_ip_proto_tcp) {
531 net_rx_tcp(&ip[ihl], actual - ihl);
532 }
533}

559uint32_t board_net_poll_rx(uint8_t* buf, uint32_t max)
560{
561 s_polls++;
562 if (s_rxq_head == s_rxq_tail) {
563 return 0U; /* ring empty. */
564 }
565 const uint32_t n = s_rxq_len[s_rxq_head];
566 if (n > max) {
567 return 0U;
568 }

570 s_rxq_head = (s_rxq_head + 1U) % (uint32_t)k_net_qdepth;
571 s_delivered++;
572 /* Console NET tab: one line per frame the peer delivers to the firmware. */
573 char ln[k_net_console_line_cap];

575 board_console_push(k_board_console_ch_net, ln);
576 return n;
577}

579void board_net_tick(void)
580{
581 s_wait++;
582 if (s_state == (uint8_t)k_net_init) {
583 net_send_arp_request();
584 s_state = (uint8_t)k_net_arp;
585 s_wait = 0U;
586 return;
587 }
588 if ((s_state == (uint8_t)k_net_estab) && s_tcp_need_data) {
589 enum : uint32_t { k_net_data_delay = 800U };
590 s_tcp_estab_wait++;
591 if (s_tcp_estab_wait > k_net_data_delay) {
592 net_send_data(); /* connection settled; send the echo payload. */
593 s_tcp_need_data = false;
594 }
595 return;
596 }
597 /* Retransmit the pending step if the firmware has not answered for a while
598 * (the stack may still be bringing the interface up on the first attempts). */
599 enum : uint32_t { k_net_retry = 2000U };
600 if (s_wait > k_net_retry) {
601 s_wait = 0U;
602 if (s_state == (uint8_t)k_net_arp) {
603 net_send_arp_request();
604 } else if (s_state == (uint8_t)k_net_ping) {
605 net_send_ping();
606 } else if (s_state == (uint8_t)k_net_syn) {
607 net_send_syn();
608 }
609 }
610}
