88static uint32_t ep_crc32(const uint8_t* data, size_t len)
89{
90 uint32_t crc = (uint32_t)k_ep_crc_init;
91 for (size_t i = 0U; i < len; i++) {
92 crc ^= (uint32_t)data[i];
93 for (uint32_t bit = 0U; bit < (uint32_t)k_ep_crc_bits; bit++) {
94 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
95 crc = (crc >> 1U) ^ ((uint32_t)k_ep_crc_poly & mask);
96 }
97 }
98 return ~crc;
99}

102static void ep_print_hex(uint32_t value)
103{
104 uint8_t buf[k_ep_hex_nibbles];
105 for (uint32_t i = 0U; i < (uint32_t)k_ep_hex_nibbles; i++) {
106 const uint32_t shift = ((uint32_t)k_ep_hex_nibbles - 1U - i) * (uint32_t)k_ep_nibble_bits;
107 const uint32_t nib = (value >> shift) & (uint32_t)k_ep_nibble_mask;
108 buf[i] = (uint8_t)((nib < (uint32_t)k_ep_dec_ten) ? ('0' + nib) : ('A' + (nib - k_ep_dec_ten)));
109 }
110 ep_print(buf, (uint32_t)k_ep_hex_nibbles);
111}

114static void ep_print_uint(uint32_t value)
115{
116 uint8_t buf[k_ep_dec_ten];
117 uint32_t n = 0U;
118 if (value == 0U) {
119 buf[n] = '0';
120 n++;
121 }
122 while ((value > 0U) && (n < (uint32_t)k_ep_dec_ten)) {
123 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_ep_dec_ten));
124 n++;
125 value /= (uint32_t)k_ep_dec_ten;
126 }
127 for (uint32_t i = 0U; i < n; i++) {
128 ep_print(&buf[n - 1U - i], 1U);
129 }
130}

161static uint16_t ep_parse_or_halt(uint32_t* out_crc)
162{
163 const ra8_epub_mem_media_t media = {.data = k_epub_fixture, .size = (size_t)k_epub_fixture_len};
164 if (ra8_epub_open(&media, "book.epub", &s_book) != k_ra8_ok) {
165 ep_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
166 }
167 uint16_t chapters = 0U;
168 if ((ra8_epub_get_chapter_count(&s_book, &chapters) != k_ra8_ok) ||
169 (chapters != (uint16_t)k_ep_expect_chap)) {
170 ep_panic_halt(k_msg_chap, (uint32_t)sizeof(k_msg_chap) - 1U);
171 }
172 size_t got = 0U;
173 if ((ra8_epub_load_chapter(&s_book, 0U, s_chapter, (size_t)k_ep_chap_cap, &got) != k_ra8_ok) ||
174 (got == 0U)) {
175 ep_panic_halt(k_msg_load, (uint32_t)sizeof(k_msg_load) - 1U);
176 }
177 ra8_epub_metadata_t meta = {};
178 if (ra8_epub_get_metadata(&s_book, &meta) != k_ra8_ok) {
179 ep_panic_halt(k_msg_meta, (uint32_t)sizeof(k_msg_meta) - 1U);
180 }
181 *out_crc = ep_crc32(s_chapter, got);
182 return chapters;
183}
