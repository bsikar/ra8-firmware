4"""Pre-flash IMAGE guard: refuse a firmware image that would brick recovery.
5
6Owner policy (2026-07-23): a bad image must never be able to permanently
7disable device recovery on the RA8D2 -- not even one that came from outside
8this tree. This scanner inspects the ACTUAL bytes of the ``.hex`` / ``.elf``
9about to be programmed and REFUSES it if it writes a lockdown value into the
10anti-recovery option-setting / security / OTP region.
11
12What it refuses
13---------------
14A *programmed* (non-erased) value landing in the permanent security / OTP
15sub-region of the option-setting memory. On the RA8D2 the option-setting
16memory (HUM Ch 7 "Option-Setting Memory") is erased to all-ones; a real
17application leaves every one of these words at ``0xFFFFFFFF``. A byte that is
18NOT ``0xFF`` in one of these windows means the image is trying to *set* it:
19
20 * First-Stage Boot Loader control (FSBLCTRL) -- boot lock.
21 * MRAM/secure attribution + access control (SAMR / SACC).
22 * PERMANENT block protect (PBPS / PBPS_SEC) -- irreversible.
23 * HUK zeroize (ZHUK) -- destroys the device-unique key.
24 * The extra-MRAM OTP window (FSBL / code cert / PBPS / POFSPS / REVOKE /
25 HUK-zeroize / anti-rollback, HUM Ch 59.7.4.5 Table 59.15) and the
26 anti-rollback counters (ARCCS / ARC_SEC / ARC_NSEC).
27
28None of the project's firmware programs any of these -- verified: every app's
29``.option_setting_otp_*`` word is ``0xFFFFFFFF`` and no example overrides the
30default. So any non-erased value there is, by construction, not something this
31project produces, and the safe answer is to refuse (this is the conservative
32policy the owner asked for: the firmware never touches the lockdown region, so
33ANY programmed data there is rejected).
34
35What it ALLOWS (must not be flagged)
36------------------------------------
37 * Normal code / rodata / vectors / data anywhere in MRAM, SRAM, or external
38 memory.
39 * The benign option bytes every app legitimately sets, AT ANY VALUE: OFS0 /
40 OFS1 / OFS2 / OFS3 (watchdog, LVD, HOCO, extended clocks), SAS + the
41 OFS*_SEC / OFS*_SEL TrustZone attribution selectors, and BPS / BPS_SEC
42 block-protect for normal use. (e.g. ``bkup_survival_demo`` legitimately
43 programs ``OFS1 = 0xFFFFFFF0`` to enable LVD0 -- that is allowed.)
44 * Any word in the anti-recovery region that is still erased (``0xFFFFFFFF``)
45 -- a real image carries those sections erased.
46
47The addresses use the flash-programming view of the option-setting memory
48(``0x0300A000`` block) -- the addresses that actually appear in a linked
49``.hex`` / ``.elf`` (confirmed against a built ``uart_hello.elf``) -- plus the
50extra-MRAM / config-set / anti-rollback windows a lockdown could otherwise
51target.
52
53Run::
54
55 check_image_no_antirecovery.py IMAGE.hex # or IMAGE.elf
56 check_image_no_antirecovery.py --selftest # prove both directions
57
58Exit 0 when the image is safe to flash, 1 when it must be refused (or on a
59bad/unreadable image, or a failing selftest).
60"""

72# The erased state of option-setting / OTP flash: all-ones. A byte that is not
73# this is a programmed bit.
74ERASED = 0xFF

76# Intel HEX record layout + record types (Intel HEX specification).
77IHEX_MIN_RECORD_LEN = 5 # length + addr(2) + type + checksum
78IHEX_DATA = 0x00
79IHEX_EOF = 0x01
80IHEX_EXT_SEG_ADDR = 0x02
81IHEX_START_SEG_ADDR = 0x03
82IHEX_EXT_LIN_ADDR = 0x04
83IHEX_START_LIN_ADDR = 0x05

85# Anti-recovery windows: (name, start, end_inclusive, hum_cite). A non-erased
86# byte anywhere in one of these is a lockdown attempt and the image is refused.
87#
88# These are the permanent / security / OTP structures. The benign option words
89# (OFS0-3 at 0x0300A100-0x0300A113, SAS, the OFS*_SEC/SEL selectors at
90# 0x0300A200-0x0300A213, and BPS/BPS_SEC at 0x0300A300/0x0300A400) are
91# deliberately NOT here -- apps set those legitimately.
92ANTI_RECOVERY_WINDOWS: tuple[tuple[str, int, int, str], ...] = (
93 (
94 "OTP security/permanent block (FSBLCTRL/SAMR/SACC/PBPS/ZHUK)",
95 0x0300A500,
96 0x0300A9FF,
97 "HUM Ch 7 Option-Setting Memory p 278-299",
98 ),
99 (
100 "extra-MRAM OTP window (FSBL/cert/PBPS/POFSPS/REVOKE/HUK-zeroize)",

122def _objcopy() -> str:
123 """Return the arm-none-eabi-objcopy path, or fail loudly if it is absent.
124
125 An image guard that silently skipped the ELF because a tool was missing
126 would be a guard that passes a brick image -- so this fails rather than
127 degrading to a no-op (CLAUDE.md: gates fail loudly on a missing tool).
128 """
129 for cand in (
130 os.environ.get("RA8_OBJCOPY", ""),
131 "arm-none-eabi-objcopy",
132 "objcopy",
133 ):
134 if cand and shutil.which(cand):
135 return shutil.which(cand) # type: ignore[return-value]
136 sys.exit(
137 "check_image_no_antirecovery.py: FATAL -- no objcopy found to read an ELF.\n"
138 " Install arm-none-eabi-objcopy or set RA8_OBJCOPY=/path/to/objcopy.\n"
139 " Refusing to skip the ELF: a skipped image check is a brick waiting to happen."
140 )

143def parse_ihex(text: str) -> dict[int, int]:
144 """Parse Intel HEX text into an ``{address: byte}`` map.
145
146 Handles record types 00 (data), 01 (EOF), 02 (extended segment), 04
147 (extended linear address) and 05 (start linear). The extended-address
148 records are essential: a linked RA8D2 image sits at 0x0200_0000 and
149 0x0300_A000, so without type-04 handling the option bytes would be read at
150 the wrong address and silently missed.
151 """
152 mem: dict[int, int] = {}
153 base = 0
154 for raw_line in text.splitlines():
155 line = raw_line.strip()
156 if not line or not line.startswith(":"):
157 continue
158 try:
159 rec = bytes.fromhex(line[1:])
160 except ValueError:
161 sys.exit(f"check_image_no_antirecovery.py: FATAL -- malformed HEX record: {line!r}")
162 if len(rec) < IHEX_MIN_RECORD_LEN or (sum(rec) & 0xFF) != 0:
163 sys.exit(f"check_image_no_antirecovery.py: FATAL -- bad HEX record: {line!r}")
164 length, addr_hi, addr_lo, rectype = rec[0], rec[1], rec[2], rec[3]
165 data = rec[4 : 4 + length]
166 if rectype == IHEX_DATA:
167 start = base + ((addr_hi << 8) | addr_lo)
168 for i, byte in enumerate(data):
169 mem[start + i] = byte
170 elif rectype == IHEX_EXT_LIN_ADDR:
171 base = ((data[0] << 8) | data[1]) << 16
172 elif rectype == IHEX_EXT_SEG_ADDR:
173 base = ((data[0] << 8) | data[1]) << 4
174 elif rectype in (IHEX_EOF, IHEX_START_SEG_ADDR, IHEX_START_LIN_ADDR):
175 continue
176 return mem

179def load_image(path: Path) -> dict[int, int]:
180 """Load a ``.hex`` or ``.elf`` firmware image into an ``{address: byte}`` map.
181
182 ``.hex`` is parsed directly. ``.elf`` is converted to Intel HEX with
183 objcopy first, so load (LMA) addresses -- what actually gets programmed --
184 are used, not the VMA.
185 """
186 suffix = path.suffix.lower()
187 if suffix in (".hex", ".ihex", ".mot", ""):
188 return parse_ihex(path.read_text(encoding="latin-1"))
189 if suffix in (".elf", ".axf", ".out"):
190 objcopy = _objcopy()
191 with tempfile.TemporaryDirectory() as td:
192 out = Path(td) / "image.hex"
193 proc = subprocess.run( # noqa: S603 # fixed argv, resolved tool
194 [objcopy, "-O", "ihex", str(path), str(out)],
195 capture_output=True,
196 text=True,
197 check=False,
198 )
199 if proc.returncode != 0:
200 sys.exit(
201 f"check_image_no_antirecovery.py: FATAL -- objcopy failed on {path}:\n"
202 f"{proc.stderr}"
203 )
204 return parse_ihex(out.read_text(encoding="latin-1"))
205 # Unknown extension: try Intel HEX, which is the common flashing format.
206 return parse_ihex(path.read_text(encoding="latin-1"))

209def scan_memory(mem: dict[int, int]) -> list[dict]:
210 """Return every non-erased byte that lands in an anti-recovery window."""
211 findings: list[dict] = []
212 for addr, byte in sorted(mem.items()):
213 if byte == ERASED:
214 continue
215 for name, start, end, cite in ANTI_RECOVERY_WINDOWS:
216 if start <= addr <= end:
217 findings.append(
218 {"addr": addr, "byte": byte, "window": name, "start": start, "cite": cite}
219 )
220 break
221 return findings

224def _report(image: str, findings: list[dict]) -> None:
225 """Explain which lockdown writes were found and why the flash is refused."""
226 print(
227 f"check_image_no_antirecovery.py: REFUSING to flash {image}\n"
228 f" {len(findings)} byte(s) program a lockdown value into the anti-recovery\n"
229 f" option-setting / security / OTP region:\n",
230 file=sys.stderr,
231 )
232 for f in findings[:MAX_FINDINGS_SHOWN]:
233 print(
234 f" 0x{f['addr']:08X} = 0x{f['byte']:02X} [{f['window']}] ({f['cite']})",
235 file=sys.stderr,
236 )
237 if len(findings) > MAX_FINDINGS_SHOWN:
238 print(f" ... {len(findings) - MAX_FINDINGS_SHOWN} more (truncated)", file=sys.stderr)
239 print(
240 "\nOwner policy (2026-07-23): this project must NEVER permanently disable\n"
241 "device recovery. No first-party image programs this region -- every\n"
242 "option-setting OTP word ships erased (0xFFFFFFFF). A non-erased value\n"
243 "here is a boot-lock / permanent-block-protect / HUK-zeroize / anti-\n"
244 "rollback lockdown, which the recovery scripts cannot undo.\n"
245 "\nBenign option bytes (OFS0/OFS1/OFS2/OFS3, SAS, OFS*_SEC/SEL, BPS) are\n"
246 "allowed at any value; only the security/OTP lockdown region is refused.\n"
247 "\nIf you are DELIBERATELY provisioning a board and accept the brick risk,\n"
248 "re-run with RA8_ALLOW_ANTIRECOVERY_FLASH=1 (loud, explicit override).",
249 file=sys.stderr,
250 )

256def _ihex_record(rectype: int, addr16: int, data: list[int]) -> str:
257 """Build one Intel HEX record line (with checksum)."""
258 body = [len(data), (addr16 >> 8) & 0xFF, addr16 & 0xFF, rectype, *data]
259 checksum = (-sum(body)) & 0xFF
260 return ":" + "".join(f"{b:02X}" for b in [*body, checksum])

263def _ela(upper16: int) -> str:
264 """Extended-linear-address record setting the upper 16 bits of the address."""
265 return _ihex_record(0x04, 0, [(upper16 >> 8) & 0xFF, upper16 & 0xFF])

268def _word_le(value: int) -> list[int]:
269 """Little-endian 4-byte list for a 32-bit word."""
270 return [value & 0xFF, (value >> 8) & 0xFF, (value >> 16) & 0xFF, (value >> 24) & 0xFF]

273def selftest() -> int:
274 """Assert the scanner refuses lockdown images and passes benign ones.
275
276 Both directions: a synthetic image that programs the disable-initialize /
277 permanent-lock OTP region is REFUSED, and one that carries only code plus
278 benign (even non-erased) OFS bytes plus erased OTP words is ALLOWED.
279 """
280 eof = _ihex_record(0x01, 0, [])
281
282 # BAD: programs PBPS (permanent block protect, 0x0300A800) and ZHUK
283 # (HUK zeroize, 0x0300A900) with non-erased values -> must REFUSE.
284 bad = "\n".join(
285 [
286 _ela(0x0200),
287 _ihex_record(0x00, 0x0000, _word_le(0x20001000)), # a reset vector
288 _ela(0x0300),
289 _ihex_record(0x00, 0xA800, _word_le(0x00000000)), # PBPS lockdown
290 _ihex_record(0x00, 0xA900, _word_le(0x00000000)), # ZHUK lockdown
291 eof,
292 ]
293 )
294 # BAD-2: a lockdown in the extra-MRAM OTP window (0x02E07600).
295 bad2 = "\n".join(
296 [
297 _ela(0x02E0),
298 _ihex_record(0x00, 0x7600, _word_le(0x00000000)),
299 eof,
300 ]
301 )
302 # GOOD: code at 0x02000000, OFS1 programmed to 0xFFFFFFF0 (benign LVD
303 # enable), and the OTP words left erased (0xFFFFFFFF) -- must ALLOW.
304 good = "\n".join(
305 [
306 _ela(0x0200),
307 _ihex_record(0x00, 0x0000, _word_le(0x20001000)),
308 _ihex_record(0x00, 0x0200, [0x00, 0xBF, 0x00, 0xBF]), # nop; nop
309 _ela(0x0300),
310 _ihex_record(0x00, 0xA104, _word_le(0xFFFFFFF0)), # OFS1 (benign)
311 _ihex_record(0x00, 0xA800, _word_le(0xFFFFFFFF)), # PBPS erased
312 _ihex_record(0x00, 0xA900, _word_le(0xFFFFFFFF)), # ZHUK erased
313 eof,
314 ]
315 )
316
317 cases: list[tuple[str, str, bool]] = [
318 ("lockdown: PBPS + ZHUK programmed", bad, True),
319 ("lockdown: extra-MRAM OTP programmed", bad2, True),
320 ("benign: code + OFS1=0xFFFFFFF0 + erased OTP", good, False),
321 ]
322 failures: list[str] = []
323 for label, text, should_refuse in cases:
324 refused = bool(scan_memory(parse_ihex(text)))
325 if refused != should_refuse:
326 verb = "did not refuse" if should_refuse else "refused"
327 failures.append(f" {label}: scanner {verb} (unexpected)")
328
329 if failures:
330 print("check_image_no_antirecovery.py --selftest: FAILED\n", file=sys.stderr)
331 print("\n".join(failures), file=sys.stderr)
332 return 1
333 refused_n = sum(1 for c in cases if c[2])
334 print(
335 f"check_image_no_antirecovery.py --selftest: PASS "
336 f"({len(cases)} cases: {refused_n} must refuse, {len(cases) - refused_n} must allow)"
337 )
338 return 0

341def main(argv: list[str]) -> int:
342 """Scan a firmware image for anti-recovery lockdown writes, or run the selftest.
343
344 Returns 0 when the image is safe to flash, 1 when it must be refused (a
345 lockdown value in the security/OTP region), on an unreadable image, or on a
346 failing selftest.
347 """
348 ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
349 ap.add_argument("image", nargs="?", help="path to a .hex or .elf image")
350 ap.add_argument("--selftest", action="store_true", help="prove both directions")
351 args = ap.parse_args(argv[1:])
352
353 if args.selftest:
354 return selftest()
355
356 if not args.image:
357 ap.error("an image path is required (or use --selftest)")
358 path = Path(args.image)
359 if not path.is_file():
360 print(f"check_image_no_antirecovery.py: FATAL -- image not found: {path}", file=sys.stderr)
361 return 1
362
363 mem = load_image(path)
364 findings = scan_memory(mem)
365 if not findings:
366 print(
367 f"check_image_no_antirecovery.py: OK ({path} -- "
368 f"{len(mem)} bytes, no anti-recovery lockdown writes)"
369 )
370 return 0
371
372 _report(str(path), findings)
373 return 1
