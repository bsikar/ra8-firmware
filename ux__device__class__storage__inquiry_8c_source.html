100{
101 buf[UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_PERIPHERAL_TYPE] =
102 (UCHAR)storage->ux_slave_class_storage_lun[lun].ux_slave_class_storage_media_type;
103 buf[UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_REMOVABLE_MEDIA] =
104 (UCHAR)storage->ux_slave_class_storage_lun[lun].ux_slave_class_storage_media_removable_flag;
105 if (storage->ux_slave_class_storage_lun[lun].ux_slave_class_storage_media_type ==
106 UX_SLAVE_CLASS_STORAGE_MEDIA_CDROM) {
107 buf[k_inq_off_data_format] = (UCHAR)k_inq_format_cdrom;
108 buf[k_inq_off_addl_len] = UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_LENGTH_CD_ROM;
109 } else {
110 buf[k_inq_off_version] = (UCHAR)k_inq_version_spc2;
111 buf[k_inq_off_data_format] = (UCHAR)k_inq_format_spc;
112 buf[k_inq_off_addl_len] = (UCHAR)k_inq_addl_len_std;
113 }
114 _ux_utility_memory_copy(buf + UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_VENDOR_INFORMATION,
115 storage->ux_slave_class_storage_vendor_id,
116 (ULONG)k_inq_vendor_id_len); /* Use case of memcpy is verified. */
117 _ux_utility_memory_copy(buf + UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_PRODUCT_ID,
118 storage->ux_slave_class_storage_product_id,
119 (ULONG)k_inq_product_id_len); /* Use case of memcpy is verified. */
120 _ux_utility_memory_copy(buf + UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_PRODUCT_REVISION,
121 storage->ux_slave_class_storage_product_rev,
122 (ULONG)k_inq_product_rev_len); /* Use case of memcpy is verified. */
123}

237RA8_INTERNAL static void internal_inquiry_send(UX_SLAVE_CLASS_STORAGE* storage,
238 UX_SLAVE_ENDPOINT* endpoint_in,
239 UX_SLAVE_TRANSFER* transfer_request,
240 ULONG inquiry_length)
241{
242#ifdef UX_DEVICE_STANDALONE
243 UX_PARAMETER_NOT_USED(endpoint_in);
244 storage->ux_device_class_storage_state = UX_DEVICE_CLASS_STORAGE_STATE_TRANS_START;
245 storage->ux_device_class_storage_cmd_state = UX_DEVICE_CLASS_STORAGE_CMD_READ;
246 storage->ux_device_class_storage_transfer = transfer_request;
247 storage->ux_device_class_storage_device_length = inquiry_length;
248 storage->ux_device_class_storage_data_length = inquiry_length;
249 storage->ux_device_class_storage_data_count = 0;
250#else
251 if (inquiry_length != 0U) {
252 _ux_device_stack_transfer_request(transfer_request, inquiry_length, inquiry_length);
253 }
254 if (storage->ux_slave_class_storage_host_length != inquiry_length) {
255 storage->ux_slave_class_storage_csw_residue =
256 storage->ux_slave_class_storage_host_length - inquiry_length;
257 _ux_device_stack_endpoint_stall(endpoint_in);
258 }
259#endif
260}

287UINT _ux_device_class_storage_inquiry(UX_SLAVE_CLASS_STORAGE* storage,
288 ULONG lun,
289 UX_SLAVE_ENDPOINT* endpoint_in,
290 UX_SLAVE_ENDPOINT* endpoint_out,
291 UCHAR* cbwcb)
292{
293 UX_PARAMETER_NOT_USED(endpoint_out);
294 UX_ASSERT(UX_SLAVE_REQUEST_DATA_MAX_LENGTH >= (ULONG)k_inq_serial_resp_len);
295 UX_TRACE_IN_LINE_INSERT(UX_TRACE_DEVICE_CLASS_STORAGE_INQUIRY,
296 storage,
297 lun,
298 0,
299 0,
300 UX_TRACE_DEVICE_CLASS_EVENTS,
301 0,
302 0)
303
304#ifndef UX_DEVICE_STANDALONE
305 /* A data phase the host marked OUT cannot satisfy INQUIRY: phase error. */
306 if (storage->ux_slave_class_storage_host_length != 0U) {
307 if ((storage->ux_slave_class_storage_cbw_flags & (UCHAR)k_inq_dir_in_bit) == 0U) {
308 _ux_device_stack_endpoint_stall(endpoint_out);
309 storage->ux_slave_class_storage_csw_status = UX_SLAVE_CLASS_STORAGE_CSW_PHASE_ERROR;
310 return (UX_ERROR);
311 }
312 }
313#endif
314
315 UX_SLAVE_TRANSFER* transfer_request = &endpoint_in->ux_slave_endpoint_transfer_request;
316 UCHAR* buf = transfer_request->ux_slave_transfer_request_data_pointer;
317 const UCHAR evpd = (UCHAR)(cbwcb[k_inq_cdb_off_evpd] & (UCHAR)k_inq_cdb_evpd_bit);
318 const UCHAR page = cbwcb[UX_SLAVE_CLASS_STORAGE_INQUIRY_PAGE_CODE];
319 ULONG inquiry_length = storage->ux_slave_class_storage_host_length;
320
321 _ux_utility_memory_set(buf,
322 0,
323 UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_LENGTH); /* Use case of memset is
324 verified. */
325 storage->ux_slave_class_storage_csw_status = UX_SLAVE_CLASS_STORAGE_CSW_PASSED;
326
327 ULONG response_length = 0U;
328 if (evpd == 0U) {
329 /* Standard INQUIRY: SPC requires PAGE CODE = 0 when EVPD is clear. */
330 if (page != (UCHAR)UX_SLAVE_CLASS_STORAGE_INQUIRY_PAGE_CODE_STANDARD) {
331 return internal_inquiry_reject(storage, lun, endpoint_in);
332 }
333 internal_inquiry_fill_standard(storage, lun, buf);
334 response_length = UX_SLAVE_CLASS_STORAGE_INQUIRY_RESPONSE_LENGTH;
335 } else if (page == (UCHAR)UX_SLAVE_CLASS_STORAGE_INQUIRY_PAGE_CODE_STANDARD) {
336 response_length = internal_inquiry_fill_vpd_pages(buf);
337 } else if (page == (UCHAR)UX_SLAVE_CLASS_STORAGE_INQUIRY_PAGE_CODE_SERIAL) {
338 response_length = internal_inquiry_fill_serial(storage, buf);
339 } else {
340 return internal_inquiry_reject(storage, lun, endpoint_in);
341 }
342
343 if (inquiry_length > response_length) {
344 inquiry_length = response_length;
345 }
346
347 internal_inquiry_send(storage, endpoint_in, transfer_request, inquiry_length);
348 return (UX_SUCCESS);
349}
