66/* Strong SysTick override: route the tick into BOTH the ra8_time millisecond
67 * counter (for ra8_delay_ms) AND ThreadX's timer (for tx_thread_sleep and
68 * semaphore timeouts). The default weak ra8_time SysTick handler only advances
69 * the ms counter; without _tx_timer_interrupt ThreadX time never advances and
70 * tx_thread_sleep / USBX class-thread scheduling stall. The project's
71 * tx_initialize_low_level.S configures SysTick but relies on the application
72 * to publish the handler. */
73
74extern void _tx_timer_interrupt(void);
75void SysTick_Handler(void);
76void SysTick_Handler(void)
77{
78 ra8_time_on_tick();
79 _tx_timer_interrupt();
80 /* Re-enable the USB IRQ at the NVIC level: the bridge's storm guard
81 * masks it to break the USBFS event-less interrupt storm, and this
82 * 1 ms pulse is its recovery clock -- a masked line is re-enabled
83 * within one period so real USB events are never lost. */
84 ux_dcd_ra8_usb_irq_reenable();
85}
86#endif

360static UINT demo_msc_read(VOID* storage,
361 ULONG lun,

377 number_blocks * (ULONG)k_demo_block_size);
378 *media_status = 0UL;
379 (void)ra8_board_led_toggle(k_ra8_board_led1);
380 return UX_SUCCESS;
381}

541static VOID demo_worker(ULONG arg)
542{
543 (void)arg;
544
545 if (demo_usbx_stack_up() != UX_SUCCESS) {
546 return;
547 }
548 if (demo_msc_class_register() != UX_SUCCESS) {
549 return;
550 }
551 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs) != k_ra8_ok) {
552 return;
553 }
554 if (ra8_usb_device_attach(k_ra8_usb_speed_fs, true) != k_ra8_ok) {
555 return;
556 }
557
558 /* Idle. USBX runs the SCSI/BBB state machine on its own threads. */
559 while (1) {
560 tx_thread_sleep(k_demo_idle_ticks);
561 }
562}

662int32_t main(void)
663{
664 uint32_t cpuclk0_hz = 0U;
665
666 if (ra8_cgc_init() != k_ra8_ok) {
667 demo_panic_halt();
668 }
669 /* Bring up PLL2 -> USBCKCR / USBCKDIVCR so USBFS sees a spec-compliant
670 * 48 MHz reference (PLL2P 240 MHz / 5). Must run BEFORE any caller
671 * releases MSTPB11 (USBFS) -- the SREQ -> SRDY handshake silently
672 * hangs otherwise (HUM Ch 9 "Clock selection switching procedure"
673 * step 1). Without this the SIE never sees a 48 MHz clock and the
674 * host never enumerates the device. */
675 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
676 demo_panic_halt();
677 }
678 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
679 demo_panic_halt();
680 }
681 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
682 demo_panic_halt();
683 }
684 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
685 demo_panic_halt();
686 }
687 if (demo_pins_init() != k_ra8_ok) {
688 demo_panic_halt();
689 }
690
691 ra8_isr_globals_enable();
692
693#ifndef RA8_OFF_TARGET
694 /* tx_kernel_enter is __noreturn -- it never comes back. */
695 tx_kernel_enter();
696#endif
697
698 demo_panic_halt();
699 return 0;
700}
