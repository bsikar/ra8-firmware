169RA8_INTERNAL static void internal_rcs_print_u32(uint32_t value)
170{
171 char buf[k_rcs_u32_buf_size];
172 uint32_t i = (uint32_t)sizeof(buf);
173 buf[--i] = '\0';
174 if (value == 0U) {
175 buf[--i] = '0';
176 } else {
177 while ((value != 0U) && (i > 0U)) {
178 buf[--i] = (char)('0' + (value % (uint32_t)k_rcs_radix_dec));
179 value /= (uint32_t)k_rcs_radix_dec;
180 }
181 }
182 internal_rcs_print(&buf[i]);
183}

256void main(void)
257{
258 internal_rcs_setup_or_halt();
259 ra8_isr_globals_enable();
260 internal_rcs_print("[rcs] cache_store demo: mounting RAM-backed LevelX...\r\n");
261
262 cache_store_demo_cfg_t cfg = internal_rcs_build_cfg();
263 cache_store_demo_result_t res = {};
264 const ra8_err_t rc = cache_store_demo_run(&cfg, &res);
265 internal_rcs_report(rc, &res);
266
267 /* Idle, re-emitting the verdict so the STOP_ON / scrape always sees the
268 * steady-state banner and the run reaches its budget cleanly. */
269 while (1) {
270 ra8_delay_ms((uint32_t)k_rcs_reemit_ms);
271 internal_rcs_report(rc, &res);
272 }
