128/* `priv_exfat_next_entry()`: see header for the documented contract. */
129ra8_err_t priv_exfat_next_entry(const ra8_fs_mount_t* m, exfat_cursor_t* cur, uint8_t* out)
130{
131 const uint32_t per_cluster =
132 (m->sectors_per_cluster * k_ra8_fs_bytes_per_sector) / (uint32_t)k_exfat_entry_bytes;
133 if (cur->entry_in_cluster >= per_cluster) {
134 uint32_t next = 0U;
135 ra8_err_t e = priv_fat_get(m, cur->cluster, &next);
136 if (e != k_ra8_ok) {
137 return e;
138 }
139 if (priv_is_eoc(m, next) != 0U) {
140 return k_ra8_err_not_found;
141 }
142 cur->cluster = next;
143 cur->entry_in_cluster = 0U;
144 }
145 const uint32_t byte_off = cur->entry_in_cluster * (uint32_t)k_exfat_entry_bytes;
146 const uint32_t lba =
147 priv_cluster_to_lba(m, cur->cluster) + (byte_off / k_ra8_fs_bytes_per_sector);
148 uint8_t sec[k_ra8_fs_bytes_per_sector] = {};
149 ra8_err_t e = priv_read_sector(m, lba, sec);
150 if (e != k_ra8_ok) {
151 return e;
152 }
153 priv_byte_copy(out, &sec[byte_off % k_ra8_fs_bytes_per_sector], (uint32_t)k_exfat_entry_bytes);
154 cur->entry_in_cluster++;
155 cur->scanned++;
156 return k_ra8_ok;
157}

160uint8_t
161priv_exfat_name_chunk_eq(const uint8_t* entry, const char* path, uint32_t pos, uint32_t nlen)
162{
163 for (uint32_t i = 0U; i < (uint32_t)k_exfat_name_per_entry; i++) {
164 if ((pos + i) >= nlen) {
165 return 1U;
166 }
167 const uint32_t b = (uint32_t)k_exfat_name_off + (i * 2U);
168 if (entry[b + 1U] != 0U) {
169 return 0U;
170 }
171 if (priv_ascii_upper((char)entry[b]) != priv_ascii_upper(path[pos + i])) {
172 return 0U;
173 }
174 }
175 return 1U;
176}

202RA8_INTERNAL
203static ra8_err_t priv_exfat_match_set(const ra8_fs_mount_t* m,
204 exfat_cursor_t* cur,
205 const char* path,
206 uint32_t* out_first,
207 uint32_t* out_size,
208 uint8_t* out_nofat)
209{
210 uint8_t strm[k_exfat_entry_bytes] = {};
211 ra8_err_t e = priv_exfat_next_entry(m, cur, strm);
212 if (e != k_ra8_ok) {
213 return e;
214 }
215 if (strm[0] != (uint8_t)k_exfat_entry_stream) {
216 return k_ra8_err_not_found;
217 }
218 const uint32_t nlen = (uint32_t)strm[k_exfat_strm_off_nlen];
219 if (nlen != priv_strlen(path)) {
220 return k_ra8_err_not_found;
221 }
222 for (uint32_t pos = 0U; pos < nlen; pos += (uint32_t)k_exfat_name_per_entry) {
223 uint8_t nm[k_exfat_entry_bytes] = {};
224 e = priv_exfat_next_entry(m, cur, nm);
225 if (e != k_ra8_ok) {
226 return e;
227 }
228 if (nm[0] != (uint8_t)k_exfat_entry_name) {
229 return k_ra8_err_not_found;
230 }
231 if (priv_exfat_name_chunk_eq(nm, path, pos, nlen) == 0U) {
232 return k_ra8_err_not_found;
233 }
234 }
235 *out_first = priv_rd32(&strm[k_exfat_strm_off_clus]);
236 *out_size = priv_rd32(&strm[k_exfat_strm_off_dlen]);
237 *out_nofat = ((strm[k_exfat_strm_off_flags] & (uint8_t)k_exfat_secflag_no_fat) != 0U) ? 1U : 0U;
238 return k_ra8_ok;
239}

241/* `priv_exfat_find()`: see header for the documented contract. */
242ra8_err_t priv_exfat_find(const ra8_fs_mount_t* m,
243 const char* path,
244 uint32_t* out_first,
245 uint32_t* out_size,
246 uint8_t* out_nofat)
247{
248 /* Leading slashes are not part of the name; match FAT's priv_path_to_83
249 * behavior so ra8_fs_open("/name") resolves on exFAT too (#93). */
250 while (*path == '/') {
251 path++;
252 }
253 exfat_cursor_t cur = {.cluster = m->root_cluster, .entry_in_cluster = 0U, .scanned = 0U};
254 while (cur.scanned < (uint32_t)k_exfat_scan_limit) {
255 uint8_t entry[k_exfat_entry_bytes] = {};
256 ra8_err_t e = priv_exfat_next_entry(m, &cur, entry);
257 if (e != k_ra8_ok) {
258 return e;
259 }
260 if (entry[0] == (uint8_t)k_exfat_entry_eod) {
261 return k_ra8_err_not_found;
262 }
263 if (entry[0] != (uint8_t)k_exfat_entry_file) {
264 continue;
265 }
266 e = priv_exfat_match_set(m, &cur, path, out_first, out_size, out_nofat);
267 if (e == k_ra8_ok) {
268 return k_ra8_ok;
269 }
270 if (e != k_ra8_err_not_found) {
271 return e;
272 }
273 }
274 return k_ra8_err_not_found; /* GCOVR_EXCL_LINE -- 65536 entries required */
275}

278ra8_err_t priv_exfat_open(ra8_fs_mount_t* handle,
279 const char* path,
280 ra8_fs_mode_t mode,
281 ra8_fs_file_t** out_file)
282{
283 if (mode != k_ra8_fs_mode_read) {
284 return k_ra8_err_not_supported;
285 }
286 uint32_t first = 0U;
287 uint32_t size = 0U;
288 uint8_t nofat = 0U;
289 ra8_err_t e = priv_exfat_find(handle, path, &first, &size, &nofat);
290 if (e != k_ra8_ok) {
291 return e;
292 }
293 ra8_fs_file_t* f = priv_alloc_file_slot();
294 if (f == nullptr) {
295 return k_ra8_err_no_mem;
296 }
297 f->mount = handle;
298 f->first_cluster = first;
299 f->cur_cluster = first;
300 f->walk_cache_idx = 0U; /* read accelerator seeded at the chain head */
301 f->walk_cache_cluster = first;
302 f->size_bytes = size;
303 f->offset = 0U;
304 f->dir_entry_lba = 0U;
305 f->dir_entry_idx = 0U;
306 f->mode = mode;
307 f->no_fat_chain = nofat;
308 f->in_use = 1U;
309 *out_file = f;
310 return k_ra8_ok;
311}
