91RA8_PRIV int priv_mdl_compose_dispatch(const mdl_args_t* a,
92 mdl_cli_mode_t mode,
93 mdl_format_t format,
94 const mdl_run_opts_t* opts,
95 const mdl_nums_t* nums,
96 const mdl_series_run_t* run)
97{
98 switch (mode) {
99 case k_mdl_cli_mode_series:
100 return mdl_app_run_series(run);
101 case k_mdl_cli_mode_search:
102 case k_mdl_cli_mode_browse: {
103 const mdl_discover_run_t request = {.cfg_path = a->cfg,
104 .term = a->search,
105 .pick = nums->pick,
106 .timeout = nums->timeout,
107 .seed = nums->seed,
108 .browse = a->browse};
109 return mdl_app_run_discover(&request, opts, run);
110 }
111 case k_mdl_cli_mode_list:
112 return mdl_app_run_list(a->out);
113 case k_mdl_cli_mode_update_all:
114 return mdl_app_run_update_all(run);
115 case k_mdl_cli_mode_remove:
116 return mdl_app_run_remove(a->out, a->remove_series);
117 case k_mdl_cli_mode_verify:
118 return mdl_app_run_verify((a->verify_dir != nullptr) ? a->verify_dir : a->out);
119 case k_mdl_cli_mode_init_site:
120 return mdl_app_run_init_site(a->init_site_url, a->out);
121 case k_mdl_cli_mode_pack:
122 return mdl_app_run_pack(a->pack, format);
123 case k_mdl_cli_mode_artifact:
124 return mdl_app_run_artifact(a->page_url, a->out, nums->timeout, opts);
125 case k_mdl_cli_mode_page:
126 return mdl_app_run_page(a->page_url,
127 a->out,
128 a->attr,
129 nums->max_imgs,
130 nums->seed,
131 nums->timeout,
132 opts);
133 default:
134 return 2;
135 }
136}
