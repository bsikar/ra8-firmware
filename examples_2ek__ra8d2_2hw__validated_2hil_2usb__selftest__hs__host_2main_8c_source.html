417static VOID selftest_device_worker(ULONG arg)
418{
419 (void)arg;
420
421 if (selftest_usbx_stack_up() != UX_SUCCESS) {
422 return;
423 }
424 if (selftest_msc_class_register() != UX_SUCCESS) {
425 return;
426 }
427 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs) != k_ra8_ok) {
428 return;
429 }
430 if (ra8_usb_device_attach(k_ra8_usb_speed_fs, true) != k_ra8_ok) {
431 return;
432 }
433
434 /* Idle. USBX runs the SCSI/BBB state machine on its own threads. */
435 while (1) {
436 tx_thread_sleep(k_selftest_idle_ticks);
437 }
438}

458static VOID selftest_host_worker(ULONG arg)
459{
460 (void)arg;
461
462 tx_thread_sleep(k_selftest_boot_wait_ticks);
463 for (;;) {
464 const ra8_err_t err = selftest_host_pass();
465 if (err == k_ra8_ok) {
466 break;
467 }
468 tx_thread_sleep(k_selftest_retry_ticks);
469 }
470 while (1) {
471 tx_thread_sleep(k_selftest_idle_ticks);
472 }
473}

557static void selftest_route_usb_or_halt(void)
558{
559 /* FS port: device role. */

561 k_ra8_ok) {
562 selftest_panic_halt();
563 }
564 if (ra8_gpio_output_init(k_selftest_pin_fs_vbusen, k_ra8_level_low) != k_ra8_ok) {
565 selftest_panic_halt();
566 }
567 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_dp, k_ra8_psel_usb_fs, "selftest.fs_dp") !=
568 k_ra8_ok) {
569 selftest_panic_halt();
570 }
571 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_dm, k_ra8_psel_usb_fs, "selftest.fs_dm") !=
572 k_ra8_ok) {
573 selftest_panic_halt();
574 }
575 /* HS port: host role. */
576 if (ra8_board_io_expander_set_usbhs_host_mode() != k_ra8_ok) {
577 selftest_panic_halt();
578 }
579 if (ra8_gpio_output_init(k_selftest_pin_hs_pwr, k_ra8_level_high) != k_ra8_ok) {
580 selftest_panic_halt();
581 }

583 k_ra8_ok) {
584 selftest_panic_halt();

603static void selftest_setup_or_halt(void)
604{
605 uint32_t cpuclk0_hz = 0U;
606 if (ra8_cgc_init() != k_ra8_ok) {
607 selftest_panic_halt();
608 }
609 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
610 selftest_panic_halt();
611 }
612 if (ra8_cgc_usbhs_pll_enable() != k_ra8_ok) {
613 selftest_panic_halt();
614 }
615 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
616 selftest_panic_halt();
617 }
618 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
619 selftest_panic_halt();
620 }
621 if (ra8_board_uart_console_init((uint32_t)k_selftest_baud) != k_ra8_ok) {
622 selftest_panic_halt();
623 }
624 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
625 selftest_panic_halt();
626 }
627 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
628 selftest_panic_halt();
629 }
630 selftest_route_usb_or_halt();
631}
