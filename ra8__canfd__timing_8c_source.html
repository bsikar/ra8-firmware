101RA8_INTERNAL
102static ra8_err_t internal_solve_timing(uint32_t clock_hz,
103 uint32_t bitrate_bps,
104 uint32_t prescaler_max,
105 ra8_canfd_timing_t* out)
106{
107 /* mcdc-deactivated: both args are validated by ra8_canfd_init upstream; defensive duplicate. */
108 if ((bitrate_bps == 0U) || (clock_hz == 0U)) {
109 return k_ra8_err_invalid_arg;
110 }
111 for (uint32_t tq = k_ra8_canfd_tq_search_hi; tq >= k_ra8_canfd_tq_search_lo; tq--) {
112 const uint32_t denom = bitrate_bps * tq;
113 if ((clock_hz % denom) != 0U) {
114 continue;
115 }
116 const uint32_t prescaler = clock_hz / denom;
117 // mcdc-deactivated: ra8_canfd_deinit (bit-timing solver) prescaler-range guard; the search-loop tq bounds (k_ra8_canfd_tq_search_lo..hi) and clock_hz/bitrate_bps caller validation upstream make either the lower or upper bound condition the dominant branch for any valid input -- the opposing condition cannot independently flip without violating the documented clock/bitrate range.
118 if ((prescaler < k_ra8_canfd_prescaler_min) || (prescaler > prescaler_max)) {
119 continue;
120 }
121 /* 75% sample point: TSEG1 = 3*(tq-1)/4, TSEG2 = tq - 1 - TSEG1. */
122 const uint32_t tseg1 = ((tq - 1U) * 3U) / 4U;
123 const uint32_t tseg2 = (tq - 1U) - tseg1;
124 const uint32_t sjw = (tseg2 < k_ra8_canfd_sjw_max) ? tseg2 : k_ra8_canfd_sjw_max;
125 out->prescaler = prescaler;
126 out->tseg1 = tseg1;
127 out->tseg2 = tseg2;
128 out->sjw = sjw;
129 return k_ra8_ok;
130 }
131 return k_ra8_err_invalid_arg;
132}

152static uint32_t internal_pack_ncfg(const ra8_canfd_timing_t* t)
153{
154 const uint32_t brp_field = ((t->prescaler - 1U) & k_ra8_cncfg_mask_nbrp)
155 << (uint32_t)k_ra8_cncfg_shift_nbrp;
156 const uint32_t tseg1_field = (t->tseg1 & k_ra8_cncfg_mask_ntseg1)
157 << (uint32_t)k_ra8_cncfg_shift_ntseg1;
158 const uint32_t tseg2_field = (t->tseg2 & k_ra8_cncfg_mask_ntseg2)
159 << (uint32_t)k_ra8_cncfg_shift_ntseg2;
160 const uint32_t sjw_field = ((t->sjw - 1U) & k_ra8_cncfg_mask_nsjw)
161 << (uint32_t)k_ra8_cncfg_shift_nsjw;
162 return brp_field | tseg1_field | tseg2_field | sjw_field;
163}

197ra8_err_t ra8_canfd_set_bitrate(uint8_t channel, uint32_t bitrate_bps, uint32_t data_bitrate_bps)
198{
199 volatile r_canfd_t* reg = ra8_canfd(channel);
200 RA8_CHECK_NULL_PTR(reg, s_tag, "channel out of range");
201
202 uint32_t pclka_hz = 0U;
203 const ra8_err_t clk_err = ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz);
204 if (clk_err != k_ra8_ok) {
205 return clk_err;
206 }
207
208 /* Nominal phase: 10-bit prescaler ceiling = 1024. */
209 ra8_canfd_timing_t nominal = {};
210 const ra8_err_t n_err =
211 internal_solve_timing(pclka_hz, bitrate_bps, k_ra8_canfd_prescaler_max, &nominal);
212 if (n_err != k_ra8_ok) {
213 return n_err;
214 }
215
216 /* NCFG / DCFG are only writable in CH_RESET or CH_HALT. Use
217 * CH_RESET: halt is a graceful transition that waits for any
218 * in-flight TX to finish, and on internal-loopback bring-up the
219 * channel may be stuck trying to TX onto a bus with no
220 * acknowledger -- halt never converges. CH_RESET is the
221 * immediate abort path, which is what FSP r_canfd does too.
222 * HUM Ch 41 "CFDCnNCFG.NTSEG2" p 2706 */
223 const ra8_err_t halt_err = ra8_canfd_internal_set_channel_mode(reg, k_ra8_chmdc_reset);
224 if (halt_err != k_ra8_ok) {
225 return halt_err;
226 }
227
228 /* HUM Ch 41 "CFDCnNCFG" p 2705 */
229 reg->CFDC[0].NCFG = internal_pack_ncfg(&nominal);
230
231 if ((data_bitrate_bps != 0U) && (data_bitrate_bps > bitrate_bps)) {
232 /* Data phase: 8-bit prescaler ceiling = 256. */
233 ra8_canfd_timing_t data = {};
234 const ra8_err_t d_err =
235 internal_solve_timing(pclka_hz, data_bitrate_bps, k_ra8_canfd_data_prescaler_max, &data);
236 if (d_err != k_ra8_ok) {
237 /* Best-effort: return the channel to CH_OPERATION so the
238 * caller does not observe a half-applied edit. */
239 (void)ra8_canfd_internal_set_channel_mode(reg, k_ra8_chmdc_operation);
240 return d_err;
241 }
242 /* HUM Ch 41 "CFDCnDCFG" p 2785 */
243 reg->CFDC2[0].DCFG = internal_pack_dcfg(&data);
244 }
245
246 const ra8_err_t op_err = ra8_canfd_internal_set_channel_mode(reg, k_ra8_chmdc_operation);
247 if (op_err != k_ra8_ok) {
248 return op_err;
249 }
250
251 ra8_log_info_val(s_tag, "set_bitrate bps", bitrate_bps);
252 return k_ra8_ok;
253}

276static ra8_err_t internal_program_data_phase(volatile r_canfd_t* reg, uint32_t data_bitrate)

284 return clk_err;
285 }

289 if (err != k_ra8_ok) {
290 return err;
291 }
292 /* HUM Ch 41 "CFDCnDCFG" p 2785 */ /* "CFDCnDCFG" */
