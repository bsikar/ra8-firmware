248[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_mecc_inject(bool two_bit, uint8_t* out_caught)
249{
250 RA8_CHECK_NULL_PTR(out_caught, s_tag, "out_caught must not be nullptr");
251
252 bool caught = false;
253 const ra8_err_t serr =
254 ra8_sram_self_test((uint8_t)k_mecc_bank, (uint32_t)k_mecc_probe_offset, two_bit, &caught);
255 if (serr != k_ra8_ok) {
256 return serr;
257 }
258
259 ra8_sram_status_t st = {};
260 const ra8_err_t gerr = ra8_sram_get_status(&st);
261 if (gerr != k_ra8_ok) {
262 return gerr;
263 }
264 if (two_bit) {
265 g_mecc_2bit_mask = (uint32_t)st.two_bit_mask;
266 } else {
267 g_mecc_1bit_mask = (uint32_t)st.one_bit_mask;
268 }
269
270 const ra8_err_t cerr = ra8_sram_clear_status(st.raw_esr);
271 if (cerr != k_ra8_ok) {
272 return cerr;
273 }
274
275 *out_caught = caught ? 1U : 0U;
276 return k_ra8_ok;
277}

297[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_mecc_run_pass(uint8_t* out_ok)
298{
299 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
300 *out_ok = 0U;
301
302 uint8_t caught1 = 0U;
303 const ra8_err_t e1 = internal_mecc_inject(false, &caught1);
304 if (e1 != k_ra8_ok) {
305 return e1;
306 }
307 g_mecc_1bit_caught = (uint32_t)caught1;
308
309 uint8_t caught2 = 0U;
310 const ra8_err_t e2 = internal_mecc_inject(true, &caught2);
311 if (e2 != k_ra8_ok) {
312 return e2;
313 }
314 g_mecc_2bit_caught = (uint32_t)caught2;
315
316 if (caught1 == 0U) {
317 return k_ra8_ok;
318 }
319 if (caught2 == 0U) {
320 return k_ra8_ok;
321 }
322 *out_ok = 1U;
323 return k_ra8_ok;
324}

326void main(void)
327{
328 internal_mecc_setup_or_halt();
329 /* Clear PRIMASK so SysTick drives ra8_delay_ms(); the ECC NMI is
330 * non-maskable and unaffected (no maskable NVIC source is armed). */
331 ra8_isr_globals_enable();
332
333 if (internal_mecc_configure() != k_ra8_ok) {
334 internal_mecc_panic_halt();
335 }
336
337 while (1) {
338 uint8_t ok = 0U;
339 const ra8_err_t err = internal_mecc_run_pass(&ok);
340 uint8_t good = 0U;
341 if (err == k_ra8_ok) {
342 good = ok;
343 }
344 if (good != 0U) {
345 (void)ra8_board_uart_console_write(s_mecc_ok_msg, (size_t)(sizeof(s_mecc_ok_msg) - 1U));
346 (void)ra8_board_led_toggle(k_ra8_board_led1);
347 } else {
348 (void)ra8_board_uart_console_write(s_mecc_bad_msg, (size_t)(sizeof(s_mecc_bad_msg) - 1U));
349 (void)ra8_board_led_toggle(k_ra8_board_led2);
350 }
351 ++g_mecc_heartbeat;
352 ra8_delay_ms(k_mecc_period_ms);
353 }
354 internal_mecc_panic_halt();
355}
