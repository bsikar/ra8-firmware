108static ra8_err_t cam_fill_ceu_config(ra8_ceu_config_t* cfg)
109{
110 RA8_CHECK_NULL_PTR(cfg, "cam", "ceu_cfg");
111 const ra8_ceu_config_t seed = {
112 /* Data-synchronous 8-bit fetch counts raw bus cycles, not pixels:
113 CMCYR.HCYL must equal the byte-cycles per active line (pixels x 2
114 for YUV422) or the CEU flags an HD line-count mismatch (IGHS) and
115 writes zero bytes. So width == line bytes, matching x_capture_px. */
116 .width_px = (uint16_t)k_cam_line_bytes,
117 .height_px = (uint16_t)k_cam_height_px,
118 .x_start_px = 0U,
119 .y_start_px = 0U,
120 .x_capture_px = (uint16_t)k_cam_line_bytes,
121 .y_capture_lines = (uint16_t)k_cam_height_px,
122 .dst_stride = (uint16_t)k_cam_line_bytes,
123 .frame_drop = 0U,
124 .bytes_per_pixel = (uint8_t)k_cam_bytes_per_px,
125 .interrupts = 0U,
126 .capture_format = k_ra8_ceu_fmt_data_synchronous,
127 .capture_mode = k_ra8_ceu_capture_single,
128 .data_bus = k_ra8_ceu_bus_8_bit,
129 .hsync_polarity = k_ra8_ceu_pol_high_active,
130 .vsync_polarity = k_ra8_ceu_pol_high_active,
131 .field_polarity = k_ra8_ceu_pol_high_active,
132 .input_order = k_ra8_ceu_input_y0_cb0_y1_cr0,
133 .output_format = k_ra8_ceu_output_ycbcr_422,
134 .burst_mode = k_ra8_ceu_burst_32,
135 .first_field = k_ra8_ceu_field_immediate,
136 .edge = {k_ra8_ceu_edge_rising,
137 k_ra8_ceu_edge_rising,
138 k_ra8_ceu_edge_rising,
139 k_ra8_ceu_edge_rising},
140 .byte_swap = {false, false, false},
141 .scale = {0U, 0U, 0U, 0U, 0U, 0U},
142 .interlace = false,
143 .one_field_only = false,
144 .bundle_write = false,
145 .low_pass_filter = false,
146 .image_area_size = (uint32_t)k_cam_frame_bytes,
147 };
148 *cfg = seed;
149 return k_ra8_ok;
150}

157ra8_err_t cam_route_ceu_pins(void)
158{
159 const uint32_t count = (uint32_t)(sizeof(s_ceu_pins) / sizeof(s_ceu_pins[0]));
160 RA8_CHECK_RANGE(count, 1U, 32U, k_ra8_err_invalid_arg);
161 for (uint32_t i = 0U; i < count; i += 1U) {
162 const ra8_port_pin_t pin = RA8_PIN(s_ceu_pins[i].port, s_ceu_pins[i].pin);
163 const ra8_err_t err = ra8_pfs_route_peripheral(pin, k_ra8_psel_ceu, "cam.ceu");
164 if (err != k_ra8_ok) {
165 return err;
166 }
167 }
168 return k_ra8_ok;
169}

181ra8_err_t cam_capture_one(void)
182{
183 ra8_err_t err = ra8_ceu_capture_arm(s_frame);
184 if (err != k_ra8_ok) {
185 return err;
186 }
187 for (uint32_t i = 0U; i < (uint32_t)k_cam_capture_tries; i += 1U) {
188 uint32_t evt = 0U;
189 err = ra8_ceu_get_status(&evt);
190 if (err != k_ra8_ok) {
191 return err;
192 }
193 if ((evt & (uint32_t)k_ra8_ceu_evt_cpe) != 0U) {
194 (void)ra8_ceu_clear_status((uint32_t)k_ra8_ceu_evt_cpe);
195 return k_ra8_ok;
196 }
197 ra8_delay_ms((uint32_t)k_cam_capture_poll_ms);
198 }
199 return k_ra8_err_hw_timeout;
200}
