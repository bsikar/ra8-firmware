236 /* HUM Ch 45.3.2 "CONVAREAD" p 3049 */
237 /* end address is inclusive. */
238 const ra8_dotf_region_t r = {
239 .start_addr = start,
240 .end_addr = start + len - 1U,
241 .key_index = 0U,
242 .region_id = 0U,
243 };
244 return ra8_dotf_set_region(channel, &r);
