38/*
39 * =============================================================================
40 * Divide-by-zero UsageFault (CCR.DIV_0_TRP) -- CPU-model seam.
41 *
42 * Unicorn's Cortex-M core executes UDIV/SDIV with the Arm default divide-by-zero
43 * result (quotient 0) and never raises the UsageFault that real Armv7-M/Armv8-M
44 * silicon takes when CCR.DIV_0_TRP is set. ra8_emulator closes that gap by scanning
45 * the image for every UDIV/SDIV site and -- only after the firmware sets
46 * CCR.DIV_0_TRP (watched via on_scb_ctrl_write) -- overwriting those sites with an
47 * undefined instruction (UDF). Each divide then traps through the already-armed
48 * ::on_invalid_insn hook into ::emulate_div0_patched, which either takes a decoded
49 * UsageFault (divisor zero) or emulates the divide in software (::div0_quotient)
50 * and continues. Patching -- rather than a per-site UC_HOOK_CODE -- is the point:
51 * a code hook disables Unicorn's engine-wide block chaining and would roughly
52 * quarter throughput for ANY busy firmware that arms the trap, whereas the invalid
53 * hook has no such cost. A firmware that never opts in keeps its native divides
54 * and pays nothing. Faithful in both directions: no faked trap, no masked one.
55 * =============================================================================
56 */

131static bool udiv_sdiv_decode(uint16_t hw1,
132 uint16_t hw2,
133 uint32_t* out_rn,
134 uint32_t* out_rd,
135 uint32_t* out_rm,
136 bool* out_signed)
137{
138 if ((out_rn == nullptr) || (out_rd == nullptr) || (out_rm == nullptr) ||
139 (out_signed == nullptr)) {
140 return false;
141 }
142 const bool is_udiv = ((uint32_t)hw1 & (uint32_t)k_div0_hw1_mask) == (uint32_t)k_div0_hw1_udiv;
143 const bool is_sdiv = ((uint32_t)hw1 & (uint32_t)k_div0_hw1_mask) == (uint32_t)k_div0_hw1_sdiv;
144 if (!is_udiv && !is_sdiv) {
145 return false;
146 }
147 if (((uint32_t)hw2 & (uint32_t)k_div0_hw2_mask) != (uint32_t)k_div0_hw2_fixed) {
148 return false;
149 }
150 const uint32_t rn = (uint32_t)hw1 & (uint32_t)k_div0_reg_mask;
151 const uint32_t rd = ((uint32_t)hw2 >> (uint32_t)k_div0_rd_shift) & (uint32_t)k_div0_reg_mask;
152 const uint32_t rm = (uint32_t)hw2 & (uint32_t)k_div0_reg_mask;
153 /* d/n/m == SP or PC is UNPREDICTABLE for UDIV/SDIV: never a real divide, so
154 * this window is a false positive straddling two instructions. Reject it so
155 * the arming pass does not overwrite executed code with UDF. */
156 if ((rn == (uint32_t)k_div0_reg_sp) || (rn == (uint32_t)k_div0_reg_pc) ||
157 (rd == (uint32_t)k_div0_reg_sp) || (rd == (uint32_t)k_div0_reg_pc) ||
158 (rm == (uint32_t)k_div0_reg_sp) || (rm == (uint32_t)k_div0_reg_pc)) {
159 return false;
160 }
161 *out_rn = rn;
162 *out_rd = rd;
163 *out_rm = rm;
164 *out_signed = is_sdiv;
165 return true;
166}

185static uint32_t div0_quotient(uint32_t vn, uint32_t vm, bool is_signed)
186{
187 if (vm == 0U) {
188 return 0U; /* Arm default when DIV_0_TRP is clear. */
189 }
190 if (!is_signed) {
191 return vn / vm;
192 }
193 if ((vn == (uint32_t)k_div0_int32_min) && ((int32_t)vm == -1)) {
194 return (uint32_t)k_div0_int32_min; /* Arm SDIV overflow: INT32_MIN / -1 = INT32_MIN. */
195 }
196 return (uint32_t)((int32_t)vn / (int32_t)vm);
197}

230bool emulate_div0_patched(uc_engine* uc, uint32_t pc, const uint8_t code[4])
231{
232 (void)code;
233 const div0_site_t* site = nullptr;
234 for (uint32_t i = 0U; i < s_div0_site_n; i++) {
235 if (s_div0_site[i].va == pc) {
236 site = &s_div0_site[i];
237 break;
238 }
239 }
240 if (site == nullptr) {
241 return false; /* not an armed divide -- let the other invalid-insn handlers try. */
242 }
243 uint32_t rn = 0U;
244 uint32_t rd = 0U;
245 uint32_t rm = 0U;
246 bool sign = false;
247 if (!udiv_sdiv_decode(site->hw1, site->hw2, &rn, &rd, &rm, &sign)) {
248 return false; /* a tracked non-divide (scan false-positive): not ours to service. */
249 }
250 uint32_t vn = 0U;
251 uint32_t vm = 0U;
252 (void)uc_reg_read(uc, k_arm_reg_id[rn], &vn);
253 (void)uc_reg_read(uc, k_arm_reg_id[rm], &vm);
254 const uint32_t ccr = rd32(uc, (uint64_t)k_scb_ccr);
255 if ((vm == 0U) && ((ccr & (uint32_t)k_ccr_div_0_trp) != 0U)) {
256 s_div0_fault = true;
257 s_div0_fault_pc = pc; /* stacked PC == the divide, as real hardware does. */
258 return true; /* PC left at the site; the run loop synthesises UsageFault. */
259 }
260 const uint32_t q = div0_quotient(vn, vm, sign);
261 (void)uc_reg_write(uc, k_arm_reg_id[rd], &q);
262 uint32_t next = pc + (uint32_t)k_div0_insn_len;
263 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
264 /* Emulating one divide consumes no modelled time: mark the stop a zero-time
265 * seam relaunch so the inner run loop re-enters at `next` WITHOUT advancing
266 * SysTick or charging an outer chunk (like on_sdmmc_read_block). Without this,
267 * a divide in a hot loop (e.g. ra8_keycache_put's modulo hashing) charges one
268 * of the 40000 chunks per execution, exhausting the budget mid-render. The
269 * zero-divisor trap path above returns earlier and is handled by the
270 * s_div0_fault branch, which is already zero-time. */
271 emu_seam_request_relaunch();
272 return true;
273}

299void div0_patch_sites(uc_engine* uc)
300{
301 if (s_div0_armed) {
302 return;
303 }
304 for (uint32_t i = 0U; i < s_div0_site_n; i++) {
305 (void)uc_mem_write(uc, (uint64_t)s_div0_site[i].va, k_div0_udf, sizeof(k_div0_udf));
306 }
307 s_div0_armed = true;
308 (void)fprintf(stderr,
309 " div-0 seam: CCR.DIV_0_TRP set -- patched %u UDIV/SDIV site(s)\n",
310 (unsigned)s_div0_site_n);
311}

335static bool div0_scan_segment(const elf_exec_segment_t* seg, void* ctx)
336{
337 (void)ctx;
338 for (uint32_t off = 0U; (off + (uint32_t)k_div0_insn_len) <= seg->filesz; off += 2U) {
339 const uint8_t* p = seg->bytes + off;
340 const uint16_t hw1 = (uint16_t)(p[0] | ((uint16_t)p[1] << 8));
341 const uint16_t hw2 = (uint16_t)(p[2] | ((uint16_t)p[3] << 8));
342 uint32_t rn = 0U;
343 uint32_t rd = 0U;
344 uint32_t rm = 0U;
345 bool sign = false;
346 if (!udiv_sdiv_decode(hw1, hw2, &rn, &rd, &rm, &sign)) {
347 continue;
348 }
349 if (s_div0_site_n >= (uint32_t)k_div0_sites_max) {
350 (void)fprintf(stderr, " div-0 seam: site cap %u reached\n", (unsigned)k_div0_sites_max);
351 return false;
352 }
353 s_div0_site[s_div0_site_n] =
354 (div0_site_t){.va = (uint32_t)seg->vaddr + off, .hw1 = hw1, .hw2 = hw2};
355 s_div0_site_n++;
356 }
357 return true;
358}

424void div0_synth_usagefault(uc_engine* uc, uint32_t vtor_base)
425{
426 const uint32_t cfsr = rd32(uc, (uint64_t)k_scb_cfsr) | (uint32_t)k_div0_cfsr_divzero;
427 wr32(uc, (uint64_t)k_scb_cfsr, cfsr);
428 const uint32_t fault_pc = emu_div0_fault_pc();
429 (void)uc_reg_write(uc, UC_ARM_REG_PC, &fault_pc);
430 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_usagefault);
431 if (handler != 0U) {
432 emu_div0_count_trap();
433 exc_enter(uc, (uint32_t)k_exc_usagefault, handler);
434 }
435}
