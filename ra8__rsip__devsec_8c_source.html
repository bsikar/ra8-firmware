82/*
83 * The device-security registers modelled below (lifecycle state, debug level,
84 * tamper control / status, DPA arm) are NOT documented RA8D2 registers: HUM
85 * Ch 51 "Security Features" (p 3263-3301) is a prose feature index and HUM
86 * Ch 52 "Renesas Secure IP (RSIP-E50D)" (p 3302-3307) is a conceptual overview
87 * with no register map -- the real state lives in the DLM / option-setting
88 * memory / SAU, not an RSIP MMIO word. The command-path bodies here only
89 * round-trip the host register fake; they compute no real security state.
90 * They compile only under the insecure-stub / off-target guard so a production
91 * image gets the fail-closed #else and can never mistake these bytes for a
92 * genuine lifecycle / debug / tamper reading. The register pokes below
93 * therefore carry NO HUM citation -- there is no real register map to cite; the
94 * former "HUM Ch 51.1 / 51.5 / 51.6" citations were fabricated and are removed.
95 */

105ra8_err_t ra8_rsip_life_advance(ra8_rsip_life_state_t state)
106{
107 if ((uint32_t)state > k_ra8_rsip_life_rma) {
108 return k_ra8_err_invalid_arg;
109 }
110 const uint32_t cur = *ra8_rsip_reg32(k_ra8_rsip_off_life_state);
111 if ((uint32_t)state < cur) {
112 return k_ra8_err_invalid_state;
113 }
114 *ra8_rsip_reg32(k_ra8_rsip_off_life_state) = (uint32_t)state;
115 return k_ra8_ok;
116}

162ra8_err_t ra8_rsip_dpa_arm(bool enable)
163{
164 volatile uint32_t* ctrl = ra8_rsip_reg32(k_ra8_rsip_off_ctrl);
165 if (enable) {
166 *ctrl |= k_ra8_rsip_mask_ctrl_dpa_arm;
167 } else {
168 *ctrl &= ~k_ra8_rsip_mask_ctrl_dpa_arm;
169 }
170 *ra8_rsip_reg32(k_ra8_rsip_off_dpa_ctrl) = (uint32_t)enable;
171 return k_ra8_ok;
172}
