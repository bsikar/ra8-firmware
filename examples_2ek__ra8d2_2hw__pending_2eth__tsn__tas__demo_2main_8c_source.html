185static uint32_t tsn_u32_to_dec(uint8_t* buf, uint32_t val)
186{
187 if (val == 0U) {
188 buf[0] = (uint8_t)'0';
189 return 1U;
190 }
191 uint8_t tmp[k_tsn_dec_u32_max];
192 uint32_t n = 0U;
193 uint32_t v = val;
194 while (v != 0U) {
195 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint32_t)k_tsn_radix));
196 v = v / (uint32_t)k_tsn_radix;
197 n++;
198 }
199 for (uint32_t i = 0U; i < n; i++) {
200 buf[i] = tmp[n - 1U - i];
201 }
202 return n;
203}

314static bool tsn_check_time_base(void)
315{
316 uint64_t sec0 = 0U;
317 uint32_t nsec0 = 0U;
318 uint64_t sec1 = 0U;
319 uint32_t nsec1 = 0U;
320 bool ok = true;
321
322 const uint32_t ms0 = ra8_time_ms();
323 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec0, &nsec0) != k_ra8_ok) {
324 ok = false;
325 }
326 ra8_delay_ms((uint32_t)k_tsn_gptp_window_ms);
327 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec1, &nsec1) != k_ra8_ok) {
328 ok = false;
329 }
330 const uint32_t elapsed_ms = ra8_time_ms() - ms0;
331 const uint64_t ns0 = tsn_flatten_ns(sec0, nsec0);
332 const uint64_t ns1 = tsn_flatten_ns(sec1, nsec1);
333 const uint64_t advance_ns = (ns1 > ns0) ? (ns1 - ns0) : 0ULL;
334 const uint64_t expected_ns = (uint64_t)elapsed_ms * (uint64_t)k_tsn_ns_per_ms;
335 const uint64_t band_ns = (expected_ns * (uint64_t)k_tsn_tolerance_pct) / (uint64_t)k_tsn_pct_full;
336
337 tsn_write(k_tsn_base_prefix, (uint32_t)(sizeof(k_tsn_base_prefix) - 1U));
338 tsn_write_u64(advance_ns);
339 tsn_write(k_tsn_base_sep, (uint32_t)(sizeof(k_tsn_base_sep) - 1U));
340 tsn_write_u32(elapsed_ms);
341 tsn_write(k_tsn_crlf, (uint32_t)(sizeof(k_tsn_crlf) - 1U));
342
343 if (elapsed_ms == 0U) {
344 return false;
345 }
346 if (advance_ns > (expected_ns + band_ns)) {
347 ok = false;
348 }
349 if (advance_ns < (expected_ns - band_ns)) {
350 ok = false;
351 }
352 return ok;
353}

411static bool tsn_program_tas(void)
412{
413 bool ok = true;
414 static const ra8_etha_tas_entry_t entries[k_tsn_gate_entries] = {
415 {.gate_time_ns = (uint32_t)k_tsn_win_units, .gate_open = true},
416 {.gate_time_ns = (uint32_t)k_tsn_win_units, .gate_open = false},
417 };
418 ra8_etha_tas_queue_t queues[k_ra8_etha_tc_count] = {};
419 queues[k_ra8_etha_tc_7].entries = entries;
420 queues[k_ra8_etha_tc_7].count = (uint16_t)k_tsn_gate_entries;
421
422 if (ra8_etha_tas_ram_reset(k_ra8_etha_port_0) != k_ra8_ok) {
423 ok = false;
424 }
425 const ra8_err_t sched_err = ra8_etha_set_tas_schedule(k_ra8_etha_port_0,
426 queues,
427 (uint8_t)k_tsn_gate_ptp_only,
428 (uint32_t)k_tsn_cycle_units,
429 0U);
430 if (sched_err != k_ra8_ok) {
431 ok = false;
432 }
433 /* Read every entry back: this is what makes the verdict mean something. */
434 for (uint8_t index = 0U; index < (uint8_t)k_tsn_gate_entries; ++index) {
435 if (!tsn_tas_entry_matches(index, &entries[index])) {
436 ok = false;
437 }
438 }
439 if (ra8_etha_enable_tas(k_ra8_etha_port_0, 1U) != k_ra8_ok) {
440 ok = false;
441 }
442 tsn_write(k_tsn_tas_prefix, (uint32_t)(sizeof(k_tsn_tas_prefix) - 1U));
443 tsn_write_u32((uint32_t)k_tsn_gate_entries);
444 tsn_write(k_tsn_crlf, (uint32_t)(sizeof(k_tsn_crlf) - 1U));
445 return ok;
446}

649#pragma GCC diagnostic push
650#pragma GCC diagnostic ignored "-Wmain"
651int32_t main(void)
652{
653 tsn_setup_or_halt();
654 ra8_isr_globals_enable();
655
656 if (tsn_arm() != k_ra8_ok) {
657 tsn_panic_halt();
658 }
659
660 while (1) {
661 const bool healthy = tsn_run_cycle();
662 if (healthy) {
663 tsn_write(k_tsn_verdict_pass, (uint32_t)(sizeof(k_tsn_verdict_pass) - 1U));
664 } else {
665 tsn_write(k_tsn_verdict_fail, (uint32_t)(sizeof(k_tsn_verdict_fail) - 1U));
666 }
667 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
668 break;
669 }
670 ra8_delay_ms((uint32_t)k_tsn_period_ms);
671 }
672 tsn_panic_halt();
673 return 0;
674}
