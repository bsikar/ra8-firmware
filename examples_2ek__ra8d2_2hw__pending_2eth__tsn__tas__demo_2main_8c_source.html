179static uint32_t tsn_u32_to_dec(uint8_t* buf, uint32_t val)
180{
181 if (val == 0U) {
182 buf[0] = (uint8_t)'0';
183 return 1U;
184 }
185 uint8_t tmp[k_tsn_dec_u32_max];
186 uint32_t n = 0U;
187 uint32_t v = val;
188 while (v != 0U) {
189 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint32_t)k_tsn_radix));
190 v = v / (uint32_t)k_tsn_radix;
191 n++;
192 }
193 for (uint32_t i = 0U; i < n; i++) {
194 buf[i] = tmp[n - 1U - i];
195 }
196 return n;
197}

308static bool tsn_check_time_base(void)
309{
310 uint64_t sec0 = 0U;
311 uint32_t nsec0 = 0U;
312 uint64_t sec1 = 0U;
313 uint32_t nsec1 = 0U;
314 bool ok = true;
315
316 const uint32_t ms0 = ra8_time_ms();
317 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec0, &nsec0) != k_ra8_ok) {
318 ok = false;
319 }
320 ra8_delay_ms((uint32_t)k_tsn_gptp_window_ms);
321 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec1, &nsec1) != k_ra8_ok) {
322 ok = false;
323 }
324 const uint32_t elapsed_ms = ra8_time_ms() - ms0;
325 const uint64_t ns0 = tsn_flatten_ns(sec0, nsec0);
326 const uint64_t ns1 = tsn_flatten_ns(sec1, nsec1);
327 const uint64_t advance_ns = (ns1 > ns0) ? (ns1 - ns0) : 0ULL;
328 const uint64_t expected_ns = (uint64_t)elapsed_ms * (uint64_t)k_tsn_ns_per_ms;
329 const uint64_t band_ns = (expected_ns * (uint64_t)k_tsn_tolerance_pct) / (uint64_t)k_tsn_pct_full;
330
331 tsn_write(k_tsn_base_prefix, (uint32_t)(sizeof(k_tsn_base_prefix) - 1U));
332 tsn_write_u64(advance_ns);
333 tsn_write(k_tsn_base_sep, (uint32_t)(sizeof(k_tsn_base_sep) - 1U));
334 tsn_write_u32(elapsed_ms);
335 tsn_write(k_tsn_crlf, (uint32_t)(sizeof(k_tsn_crlf) - 1U));
336
337 if (elapsed_ms == 0U) {
338 return false;
339 }
340 if (advance_ns > (expected_ns + band_ns)) {
341 ok = false;
342 }
343 if (advance_ns < (expected_ns - band_ns)) {
344 ok = false;
345 }
346 return ok;
347}

405static bool tsn_program_tas(void)
406{
407 bool ok = true;
408 static const ra8_etha_tas_entry_t entries[k_tsn_gate_entries] = {
409 {.gate_time_ns = (uint32_t)k_tsn_win_units, .gate_open = true},
410 {.gate_time_ns = (uint32_t)k_tsn_win_units, .gate_open = false},
411 };
412 ra8_etha_tas_queue_t queues[k_ra8_etha_tc_count] = {};
413 queues[k_ra8_etha_tc_7].entries = entries;
414 queues[k_ra8_etha_tc_7].count = (uint16_t)k_tsn_gate_entries;
415
416 if (ra8_etha_tas_ram_reset(k_ra8_etha_port_0) != k_ra8_ok) {
417 ok = false;
418 }
419 const ra8_err_t sched_err = ra8_etha_set_tas_schedule(k_ra8_etha_port_0,
420 queues,
421 (uint8_t)k_tsn_gate_ptp_only,
422 (uint32_t)k_tsn_cycle_units,
423 0U);
424 if (sched_err != k_ra8_ok) {
425 ok = false;
426 }
427 /* Read every entry back: this is what makes the verdict mean something. */
428 for (uint8_t index = 0U; index < (uint8_t)k_tsn_gate_entries; ++index) {
429 if (!tsn_tas_entry_matches(index, &entries[index])) {
430 ok = false;
431 }
432 }
433 if (ra8_etha_enable_tas(k_ra8_etha_port_0, 1U) != k_ra8_ok) {
434 ok = false;
435 }
436 tsn_write(k_tsn_tas_prefix, (uint32_t)(sizeof(k_tsn_tas_prefix) - 1U));
437 tsn_write_u32((uint32_t)k_tsn_gate_entries);
438 tsn_write(k_tsn_crlf, (uint32_t)(sizeof(k_tsn_crlf) - 1U));
439 return ok;
440}

643#pragma GCC diagnostic push
644#pragma GCC diagnostic ignored "-Wmain"
645int32_t main(void)
646{
647 tsn_setup_or_halt();
648 ra8_isr_globals_enable();
649
650 if (tsn_arm() != k_ra8_ok) {
651 tsn_panic_halt();
652 }
653
654 while (1) {
655 const bool healthy = tsn_run_cycle();
656 if (healthy) {
657 tsn_write(k_tsn_verdict_pass, (uint32_t)(sizeof(k_tsn_verdict_pass) - 1U));
658 } else {
659 tsn_write(k_tsn_verdict_fail, (uint32_t)(sizeof(k_tsn_verdict_fail) - 1U));
660 }
661 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
662 break;
663 }
664 ra8_delay_ms((uint32_t)k_tsn_period_ms);
665 }
666 tsn_panic_halt();
667 return 0;
668}
