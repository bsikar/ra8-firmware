148static bool wdt_window_demo_iter(void)
149{
150 uint16_t counter = 0U;
151 if (ra8_wdt_get_counter(&counter) != k_ra8_ok) {
152 wdt_window_demo_banner(k_wdt_window_demo_msg_counter_failed,
153 (uint32_t)(sizeof(k_wdt_window_demo_msg_counter_failed) - 1U));
154 return false;
155 }
156 if (!wdt_window_demo_in_window(counter)) {
157 return true;
158 }
159 if (ra8_wdt_refresh_for(k_ra8_wdt0) != k_ra8_ok) {
160 wdt_window_demo_banner(k_wdt_window_demo_msg_refresh_failed,
161 (uint32_t)(sizeof(k_wdt_window_demo_msg_refresh_failed) - 1U));
162 return false;
163 }
164 if (ra8_wdt_clear_status() != k_ra8_ok) {
165 wdt_window_demo_banner(k_wdt_window_demo_msg_clear_failed,
166 (uint32_t)(sizeof(k_wdt_window_demo_msg_clear_failed) - 1U));
167 return false;
168 }
169 (void)ra8_board_led_toggle(k_ra8_board_led1);
170 wdt_window_demo_banner(k_wdt_window_demo_msg_refresh,
171 (uint32_t)(sizeof(k_wdt_window_demo_msg_refresh) - 1U));
172 return true;
173}

177int32_t main(void)
178{
179 wdt_window_demo_setup_or_halt();
180 ra8_isr_globals_enable();

182 /* Boot banner -- emit before ra8_wdt_init so the HIL host can see
183 * the firmware booted even if the WDT bring-up fails. */
184 (void)ra8_board_uart_console_write(k_wdt_window_demo_msg_boot,
185 (size_t)(sizeof(k_wdt_window_demo_msg_boot) - 1U));
186
187 /* on_expiry=k_ra8_wdt_on_expiry_nmi (RSTIRQS=0) routes a window
188 * violation through NMI/IRQ instead of a silent reset, so a bug
189 * in the demo's window math shows up as a banner-loss rather than
190 * a reset-loop that confuses HIL bench parsing. */
191 const ra8_wdt_cfg_t cfg = {
192 .timeout = k_ra8_wdt_timeout_1024,
193 .clock_div = k_ra8_wdt_clkdiv_4,
194 .window_start = k_ra8_wdt_window_start_75,
195 .window_end = k_ra8_wdt_window_end_25,
196 .on_expiry = k_ra8_wdt_on_expiry_nmi,
197 .stop_in_sleep = k_ra8_wdt_sleep_stop_count,
198 };
199 if (ra8_wdt_init(&cfg) != k_ra8_ok) {
200 wdt_window_demo_banner(k_wdt_window_demo_msg_init_failed,
201 (uint32_t)(sizeof(k_wdt_window_demo_msg_init_failed) - 1U));
202 wdt_window_demo_panic_halt();
203 }
204
205 while (wdt_window_demo_iter()) {
206 ra8_delay_ms((uint32_t)k_wdt_window_demo_poll_ms);
207 }
208 wdt_window_demo_panic_halt();
209 return 0;
210}
