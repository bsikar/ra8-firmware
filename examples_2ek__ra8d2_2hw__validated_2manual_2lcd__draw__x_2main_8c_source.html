104static void lcd_draw_x(uint16_t color)
105{
106 for (uint16_t x = 0U; x < (uint16_t)k_fb_w; x++) {
107 const int32_t y_down = ((int32_t)x * (int32_t)k_fb_h) / (int32_t)k_fb_w;
108 const int32_t y_up = ((int32_t)k_fb_h - 1) - y_down;
109 for (int32_t dy = -(int32_t)k_lcd_x_thickness; dy <= (int32_t)k_lcd_x_thickness; dy++) {
110 const int32_t yy_down = y_down + dy;
111 const int32_t yy_up = y_up + dy;
112 if ((yy_down >= 0) && (yy_down < (int32_t)k_fb_h)) {
113 s_framebuffer[((uint32_t)yy_down * (uint32_t)k_fb_w) + (uint32_t)x] = color;
114 }
115 if ((yy_up >= 0) && (yy_up < (int32_t)k_fb_h)) {
116 s_framebuffer[((uint32_t)yy_up * (uint32_t)k_fb_w) + (uint32_t)x] = color;
117 }
118 }
119 }
120 __asm__ volatile("dsb" ::: "memory");
121}

139static void lcd_bringup_clocks(void)
140{

142 if (ra8_cgc_init() != k_ra8_ok) {
143 lcd_panic_halt();
144 }

148 if (ra8_mstp_init() != k_ra8_ok) {
149 lcd_panic_halt();

152 lcd_panic_halt();
153 }
154 if (ra8_board_led_init(k_ra8_board_led_blue) != k_ra8_ok) {
155 lcd_panic_halt();
156 }
157 if (ra8_board_led_init(k_ra8_board_led_red) != k_ra8_ok) {
158 lcd_panic_halt();
159 }
160 ra8_isr_globals_enable();
161}
