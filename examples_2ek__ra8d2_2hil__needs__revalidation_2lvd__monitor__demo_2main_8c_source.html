66/* PRCR write values to (un)lock the PVD protection group. Not a Doxygen block:
67 * it documents constants that live in ra8_lvd's headers, not the declaration
68 * that follows it.
69 *
70 * The PVD control registers (PVD1CMPCR / PVD1CR0 / PVD1CR1 / PVD1SR /
71 * PVD1FCR) are write-protected by PRCR.PRC3. Each PVD register page carries
72 * the note "Set the PRCR.PRC3 bit to 1 (write enabled) before rewriting this
73 * register" (HUM Ch 8.2.2 p 303, Ch 8.2.4 p 305, Ch 8.2.6 p 307, Ch 8.2.7
74 * p 307). PRCR is R_SYSTEM + 0x3FA; bits 15:8 are the mandatory 0xA5 key and
75 * k_ra8_prcr_grp3_pvd (0x0008) is the PRC3 group bit, so
76 * k_ra8_prcr_unlock_pvd (0xA508) enables writes and k_ra8_prcr_lock_all (the
77 * key alone) relocks. */

179[[nodiscard]] static ra8_err_t lvd_demo_configure(void)
180{
181 const ra8_lvd_cfg_t cfg = {
182 .threshold = k_ra8_lvd_pvdlvl_2_80v,
183 .edge = k_ra8_lvd_edge_both,
184 .irq_type = k_ra8_lvd_irq_maskable,
185 .response = k_ra8_lvd_response_none, /* SAFE: flags only, no reset/NMI. */
186 .negate = k_ra8_lvd_negate_after_voltage,
187 .hysteresis = k_ra8_lvd_hysteresis_lvd,
188 .filter_div = k_ra8_lvd_loco_div_8,
189 .filter_en = false,
190 .irq_enable = false,
191 .clear_status = true,
192 };
193
194 /* HUM Ch 8.2.4 "PVDmCR0" p 305: "Set the PRCR.PRC3 bit to 1 (write
195 * enabled) before rewriting this register." PRCR = R_SYSTEM + 0x3FA,
196 * key 0xA5 in [15:8], PRC3 = bit 3. */
197 *ra8_sys_prcr() = (uint16_t)k_ra8_prcr_unlock_pvd;
198 const ra8_err_t err = ra8_lvd_channel_init(k_ra8_lvd_ch1, &cfg);
199 ra8_sys_prcr_lock_all();
200
201 g_lvd_cfg_err = (err == k_ra8_ok) ? 0U : (uint32_t)err;
202 return err;
203}

226[[nodiscard]] static ra8_err_t lvd_demo_sample(uint8_t* out_ok)
227{
228 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
229
230 ra8_lvd_status_t st = {.crossed = false, .above = false};
231 const ra8_err_t err = ra8_lvd_get_status(k_ra8_lvd_ch1, &st);
232 if (err != k_ra8_ok) {
233 return err;
234 }
235 g_lvd_mon_above = st.above ? 1U : 0U;
236 g_lvd_det = st.crossed ? 1U : 0U;
237 *out_ok = (st.above && !st.crossed) ? 1U : 0U;
238 return k_ra8_ok;
239}

243int32_t main(void)
244{
245 lvd_demo_setup_or_halt();
246 /* Clear PRIMASK so SysTick can dispatch: ra8_delay_ms() uses the
247 * SysTick tick path only when IRQs are globally enabled, otherwise it
248 * busy-waits on DWT_CYCCNT. No NVIC sources are armed (response=none). */
249 ra8_isr_globals_enable();
250 if (lvd_demo_configure() != k_ra8_ok) {
251 lvd_demo_panic_halt();
252 }
253 ra8_delay_ms((uint32_t)k_lvd_demo_stab_ms);
254
255 while (1) {
256 uint8_t ok = 0U;
257 const ra8_err_t err = lvd_demo_sample(&ok);
258 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
259 g_lvd_ok = (uint32_t)good;
260 if (good != 0U) {
261 (void)ra8_board_uart_console_write(k_lvd_demo_ok_msg,
262 (size_t)(sizeof(k_lvd_demo_ok_msg) - 1U));
263 (void)ra8_board_led_toggle(k_ra8_board_led1);
264 } else {
265 (void)ra8_board_uart_console_write(k_lvd_demo_bad_msg,
266 (size_t)(sizeof(k_lvd_demo_bad_msg) - 1U));
267 (void)ra8_board_led_toggle(k_ra8_board_led2);
268 }
269 ++g_lvd_heartbeat;
270 ra8_delay_ms(k_lvd_demo_period_ms);
271 }
272 lvd_demo_panic_halt();
273 return 0;
274}
