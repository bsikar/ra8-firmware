76static uint64_t dac_read(uc_engine* uc, uint64_t addr, unsigned size)
77{
78 (void)uc;
79 (void)size;
80 const uint64_t off = addr - (uint64_t)k_dac_base;
81 const uint32_t inst = (uint32_t)(off / (uint64_t)k_dac_stride);
82 if (inst >= (uint32_t)k_dac_count) {
83 return 0U;
84 }
85 const uint64_t inst_off = off % (uint64_t)k_dac_stride;
86 if (inst_off >= (uint64_t)k_dac_stride) {
87 return 0U;
88 }
89 return s_dac[inst].reg[dac_word(inst_off)];

93static void dac_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
94{
95 (void)uc;
96 (void)size;
97 const uint64_t off = addr - (uint64_t)k_dac_base;
98 const uint32_t inst = (uint32_t)(off / (uint64_t)k_dac_stride);
99 if (inst >= (uint32_t)k_dac_count) {
100 return;
101 }
102 const uint64_t inst_off = off % (uint64_t)k_dac_stride;
103 if (inst_off >= (uint64_t)k_dac_stride) {
104 return;
105 }
106 s_dac[inst].reg[dac_word(inst_off)] = (uint32_t)value;
107 if (inst_off == (uint64_t)k_dac_off_dadr) {
108 const uint16_t code = (uint16_t)((uint32_t)value & (uint32_t)k_dac_dadr_mask);
109 s_dac[inst].last = code;
110 if (code > s_dac[inst].peak) {
111 s_dac[inst].peak = code;
112 }
113 s_dac[inst].writes++;
114 /* Console DAC tab: anti-flood -- the first update and then every Nth update
115 * (a continuous waveform writes per-sample, so one line per N keeps the ring
116 * readable while still showing the channel is active). */
117 bool should_report = (s_dac[inst].writes == 1U);
118 if ((s_dac[inst].writes % (uint32_t)k_dac_console_every) == 0U) {
119 should_report = true;
120 }
121 if (should_report) {
122 char ln[k_dac_console_line_cap];
123 (void)snprintf(ln,
124 sizeof(ln),
125 "DAC%u code=%u (n=%u)",
126 (unsigned)inst,
127 (unsigned)code,
128 (unsigned)s_dac[inst].writes);
129 board_console_push(k_board_console_ch_dac, ln);
130 }
131 if (board_periph_trace()) {
132 (void)fprintf(stderr, " [trace] DAC_B%u <- code=%u\n", inst, code);
133 }
134 }
135}

146static void dac_report(void)
147{
148 for (uint32_t i = 0U; i < (uint32_t)k_dac_count; i++) {
149 if (s_dac[i].writes == 0U) {
150 continue;
151 }
152 (void)fprintf(stderr,
153 " DAC_B%u : writes=%u last=%u peak=%u enabled=%s\n",
154 i,
155 s_dac[i].writes,
156 s_dac[i].last,
157 s_dac[i].peak,
158 (s_dac[i].reg[dac_word((uint64_t)k_dac_off_dacr0)] & (uint32_t)k_dac_dacen_mask)
159 ? "yes"
160 : "no");
161 }
