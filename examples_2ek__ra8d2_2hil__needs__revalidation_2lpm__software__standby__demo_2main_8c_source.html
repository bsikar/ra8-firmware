165static void lpm_swstd_add_offset(const ra8_rtc_datetime_t* now, ra8_rtc_datetime_t* out)
166{
167 *out = *now;
168 const uint16_t s = (uint16_t)now->second + (uint16_t)k_lpm_swstd_alarm_offset_s;
169 out->second = (uint8_t)(s % (uint16_t)k_lpm_swstd_secs_per_min);
170 const uint16_t carm = (uint16_t)(s / (uint16_t)k_lpm_swstd_secs_per_min);
171 const uint16_t m = (uint16_t)now->minute + carm;
172 out->minute = (uint8_t)(m % (uint16_t)k_lpm_swstd_mins_per_hour);
173 const uint16_t carh = (uint16_t)(m / (uint16_t)k_lpm_swstd_mins_per_hour);
174 const uint16_t h = (uint16_t)now->hour + carh;
175 out->hour = (uint8_t)(h % (uint16_t)k_lpm_swstd_hours_per_day);
176}

197[[nodiscard]] static ra8_err_t lpm_swstd_arm_wake(const ra8_rtc_datetime_t* now)
198{
199 ra8_rtc_datetime_t a = {};
200 lpm_swstd_add_offset(now, &a);
201 ra8_err_t err = ra8_rtc_set_alarm(&a);
202 if (err != k_ra8_ok) {
203 return err;
204 }
205 err = ra8_rtc_set_irq_enable((uint8_t)k_ra8_rtc_irq_alarm);
206 if (err != k_ra8_ok) {
207 return err;
208 }
209 return ra8_lpm_arm_wupen0_bits((uint32_t)k_ra8_lpm_wupen0_rtcalm);
210}

214int32_t main(void)
215{
216 lpm_swstd_setup_or_halt();
217 ra8_isr_globals_enable();
218
219 /* Emit the boot banner *before* the standby entry so the HIL gate
220 * confirms the firmware booted even when the sub-clock crystal is
221 * silent (see the SOSC caveat in the file header). */
222 (void)ra8_board_uart_console_write(k_lpm_swstd_boot_msg,
223 (size_t)(sizeof(k_lpm_swstd_boot_msg) - 1U));
224
225 while (1) {
226 ra8_rtc_datetime_t now = {};
227 if (ra8_rtc_get(&now) != k_ra8_ok) {
228 break;
229 }
230 if (lpm_swstd_arm_wake(&now) != k_ra8_ok) {
231 break;
232 }
233 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_software_std) != k_ra8_ok) {
234 break;
235 }
236 g_lpm_swstd_wake_count++;
237 /* Clear the alarm flag so the next cycle starts from a known
238 * state. Errors here are non-fatal -- the loop body will simply
239 * re-arm the alarm. */
240 (void)ra8_rtc_clear_status((uint8_t)k_ra8_rtc_irq_alarm);
241 }
242 lpm_swstd_panic_halt();
243 return 0;
244}
