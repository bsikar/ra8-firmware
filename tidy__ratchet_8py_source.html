1#!/usr/bin/env python3
2"""tidy_ratchet.py -- clang-tidy ratchet gate (compare vs committed baseline).
3
4WHY THIS EXISTS
5---------------
6#369 and #370 brought 435 firmware C files and 12 first-party C++/Objective-C
7files into clang-tidy's scope for the first time. They had never been analysed,
8so they arrive carrying pre-existing findings. Two bad options were available
9and both were rejected:
10
11* leave the new surface OUT of the gate -- which is the status quo the two
12 issues exist to end, and the reason nobody knew what was in there;
13* switch the offending checks off -- which converts a measured debt into a
14 permanent blind spot, and is exactly the "gate that silently does nothing"
15 pattern this tree keeps finding.
16
17So the new surface is IN the gate, every finding is counted, and this ratchet
18holds the line: it is the same shape as `misra_ratchet.py`, which this tree
19already uses for precisely this situation.
20
21* NEW findings (any per-file-per-check count above the baseline, or any
22 file/check pair absent from the baseline) FAIL.
23* Shrinkage PASSES with a notice to re-baseline, which locks the progress in
24 so the debt can never quietly grow back.
25
26Closing the debt means the baseline reaching zero rows and being deleted --
27not being regenerated larger. `--update` refuses to grow a bucket for exactly
28that reason; a genuine increase has to be justified by a human editing the
29file, which leaves a reviewable diff.
30
31BASELINE NORMALISATION -- per-file-per-check COUNTS, not raw finding lines.
32Raw clang-tidy findings carry line numbers, which churn on every unrelated edit
33above them, and message text, which embeds identifiers. A `(file, check) ->
34count` map is invariant under both, still trips the moment a file gains another
35violation of a check, and stays small and diffable.
36
37USAGE
38 python3 scripts/checks/tidy_ratchet.py --selftest # assert it fires
39 python3 scripts/checks/tidy_ratchet.py --check LOGFILE # the gate
40 python3 scripts/checks/tidy_ratchet.py --update LOGFILE # re-baseline
41
42Copyright (c) 2026 Brighton Sikarskie
43SPDX-License-Identifier: MIT

57MAX_DETAIL_LINES = 10
58"""Cap on offending buckets echoed before the report truncates."""
59
60BASELINE_COLUMNS = 3
61"""Column count of one baseline row: file, check, count."""
62
63DIAGNOSTIC_ERROR_CHECK = "clang-diagnostic-error"
64"""The clang "check" that is really a PARSE failure, not a lint finding."""
65
66MAX_DIAGNOSTIC_ERROR_RATE = 0.20
67"""Refuse to ratchet when more than this fraction of findings are parse errors.
68
69A clang-diagnostic-error means clang-tidy could not PARSE a translation unit
70(a missing header, an unusable compile command) rather than that it found a
71real defect. When those dominate a run, nothing was actually analysed -- the
72#387 case, where an unusable arm-none-eabi-gcc stripped the firmware pass of
73every system include and turned ~100 TUs into "'string.h' file not found". The
74healthy baseline runs at 37 of 415 findings (9%) with the toolchain correct, so
750.20 clears real runs with margin while a mass parse failure (which pushes the
76rate past 25%) trips it.
77"""
78
79DIAGNOSTIC_ERROR_FLOOR = 60
80"""...and only once at least this many parse errors are present, so a tiny run
81whose handful of findings happen to be parse errors does not trip. Above the
82healthy baseline's 37, below the ~137 a broken firmware pass produces."""
83
84# One clang-tidy diagnostic line:
85# /abs/path/file.c:12:34: warning: text [check-name,-warnings-as-errors]
86FINDING_RE = re.compile(
87 r"^(?P<path>[^:\s][^:]*):(?P<line>\d+):(?P<col>\d+):\s+"
88 r"(?:warning|error):\s+.*\‍[(?P<checks>[A-Za-z0-9_.,-]+)\‍]\s*$"
89)
90
91# clang-tidy appends this to every check name when WarningsAsErrors is on; it
92# is not a check and must not become a baseline key.
93NOT_A_CHECK = "-warnings-as-errors"

96def parse_log(text: str) -> Counter:
97 """Return a {(repo-relative file, check): count} Counter for `text`.
98
99 A finding may list several aliases for one diagnostic
100 (`bugprone-reserved-identifier,cert-dcl37-c,cert-dcl51-cpp`). The FIRST is
101 used as the canonical key so an alias-set reordering between clang-tidy
102 versions cannot silently rewrite every baseline row.
103 """
104 counts: Counter = Counter()
105 seen: set[tuple[str, int, int, str]] = set()
106 for raw in text.splitlines():
107 m = FINDING_RE.match(raw.strip())
108 if not m:
109 continue
110 names = [c for c in m.group("checks").split(",") if c and c != NOT_A_CHECK]
111 if not names:
112 continue
113 check = names[0]
114 path = m.group("path")
115 try:
116 rel = str(Path(path).resolve().relative_to(REPO_ROOT))
117 except ValueError:
118 rel = path
119 # clang-tidy repeats a header diagnostic once per TU that includes it.
120 key = (rel, int(m.group("line")), int(m.group("col")), check)
121 if key in seen:
122 continue
123 seen.add(key)
124 counts[(rel, check)] += 1
125 return counts

128def load_baseline() -> Counter:
129 """Read the committed baseline into a Counter. Missing file means empty."""
130 counts: Counter = Counter()
131 if not BASELINE_FILE.is_file():
132 return counts
133 for raw in BASELINE_FILE.read_text(encoding="ascii").splitlines():
134 line = raw.strip()
135 if not line or line.startswith("#"):
136 continue
137 parts = line.split("\t")
138 if len(parts) != BASELINE_COLUMNS:
139 continue
140 counts[(parts[0], parts[1])] = int(parts[2])
141 return counts

144def write_baseline(counts: Counter) -> None:
145 """Write `counts` out in the committed, sorted, diffable form."""
146 lines = [
147 "# clang-tidy ratchet baseline -- per-file-per-check finding counts.",
148 "# Consumed by scripts/checks/tidy_ratchet.py --check (CI gate: tidy).",
149 "#",
150 "# These are PRE-EXISTING findings on the surface that #369 and #370",
151 "# brought into clang-tidy's scope for the first time. The gate fails on",
152 "# any INCREASE, so the debt is frozen and can only be burned down.",
153 "#",
154 "# Regenerate after burning findings down:",
155 "# bash scripts/checks/clang_tidy.sh --check > /tmp/tidy.log 2>&1",
156 "# python3 scripts/checks/tidy_ratchet.py --update /tmp/tidy.log",
157 "#",
158 "# Closing this out means this file reaching zero rows and being",
159 "# DELETED -- never regenerated larger.",
160 "#",
161 "# file<TAB>check<TAB>count",
162 ]
163 for (path, check), n in sorted(counts.items()):
164 if n:
165 lines.append(f"{path}\t{check}\t{n}")
166 BASELINE_FILE.write_text("\n".join(lines) + "\n", encoding="ascii")

169def diagnostic_error_reason(current: Counter) -> str | None:
170 """Return a refusal reason if the run is dominated by parse errors, else None.
171
172 Guards both directions of the ratchet (#387): a run whose findings are
173 mostly clang-diagnostic-error has not analysed its code, it has failed to
174 compile it. Writing that as the baseline freezes ~100 files of parse errors
175 as the accepted state, and comparing against it hides every real finding in
176 those files. So refuse to ``--update`` or ``--check`` such a run and name
177 the likely cause instead.
178
179 Trips only when BOTH the absolute count clears ``DIAGNOSTIC_ERROR_FLOOR``
180 and the fraction clears ``MAX_DIAGNOSTIC_ERROR_RATE`` -- a healthy run
181 carrying a few baselined parse errors must not be mistaken for a broken one.
182 """
183 total = sum(current.values())
184 if total == 0:
185 return None
186 diag = sum(n for (_path, check), n in current.items() if check == DIAGNOSTIC_ERROR_CHECK)
187 rate = diag / total
188 if diag < DIAGNOSTIC_ERROR_FLOOR or rate <= MAX_DIAGNOSTIC_ERROR_RATE:
189 return None
190 return (
191 f"{diag} of {total} finding(s) ({rate:.0%}) are {DIAGNOSTIC_ERROR_CHECK} "
192 f"-- over the {MAX_DIAGNOSTIC_ERROR_RATE:.0%} ceiling.\n"
193 " clang-tidy could not PARSE most of its input. The usual cause is an\n"
194 " unusable cross-compiler stripping the firmware pass of its system\n"
195 " includes (#387): run the tidy gate with the pinned 13.3\n"
196 " arm-none-eabi-gcc on PATH. A baseline built from a broken parse is\n"
197 " worse than no baseline, so this run is refused for ratcheting."
198 )

201def report(current: Counter, baseline: Counter) -> int:
202 """Compare and print. Returns the process exit code."""
203 grown = []
204 for key, n in sorted(current.items()):
205 was = baseline.get(key, 0)
206 if n > was:
207 grown.append((key, was, n))
208
209 total_now = sum(current.values())
210 total_was = sum(baseline.values())
211
212 if grown:
213 print(file=sys.stderr)
214 print("clang-tidy ratchet: NEW findings above the committed baseline", file=sys.stderr)
215 print(file=sys.stderr)
216 for (path, check), was, now in grown[:MAX_DETAIL_LINES]:
217 print(f" {path}\n {check}: {was} -> {now}", file=sys.stderr)
218 if len(grown) > MAX_DETAIL_LINES:
219 print(f" ... and {len(grown) - MAX_DETAIL_LINES} more bucket(s)", file=sys.stderr)
220 print(file=sys.stderr)
221 print(
222 "Fix them. The baseline is a burn-down of debt that predates the\n"
223 "scope widening, not a place to record new debt.",
224 file=sys.stderr,
225 )
226 return 1
227
228 print(f"clang-tidy ratchet: {total_now} finding(s), baseline {total_was} -- no growth.")
229 if total_now < total_was:
230 burned = total_was - total_now
231 print(f" {burned} finding(s) burned down. Re-baseline to lock it in:")
232 print(" python3 scripts/checks/tidy_ratchet.py --update <log>")
233 return 0

236def _selftest_parse() -> list[str]:
237 """Assertions about log parsing."""
238 failures: list[str] = []
239
240 sample = (
241 "/repo/examples/a/main.c:12:3: warning: x is bad "
242 "[readability-magic-numbers,-warnings-as-errors]\n"
243 "/repo/examples/a/main.c:14:5: error: y is worse "
244 "[bugprone-reserved-identifier,cert-dcl37-c]\n"
245 "some unrelated build noise\n"
246 " 12 | static uint32_t s_pass = 0U;\n"
247 )
248 counts = parse_log(sample)
249 if sum(counts.values()) != 2: # noqa: PLR2004 -- the fixture plants exactly 2
250 failures.append(f"parse_log() found {sum(counts.values())} finding(s), expected 2")
251 checks = {check for _, check in counts}
252 if "bugprone-reserved-identifier" not in checks:
253 failures.append("parse_log() did not canonicalise an alias set to its first name")
254 if NOT_A_CHECK in checks:
255 failures.append("parse_log() treated -warnings-as-errors as a check name")
256
257 # A repeated header diagnostic (same file/line/col/check) counts ONCE.
258 dupe = parse_log(
259 "/repo/libs/x.h:9:1: warning: z [misc-x,-warnings-as-errors]\n"
260 "/repo/libs/x.h:9:1: warning: z [misc-x,-warnings-as-errors]\n"
261 )
262 if sum(dupe.values()) != 1:
263 failures.append(f"parse_log() counted a duplicated header finding {sum(dupe.values())}x")
264
265 return failures

268def _selftest_ratchet() -> list[str]:
269 """Assertions about the growth verdict and baseline round-tripping."""
270 failures: list[str] = []
271
272 # The gate must FAIL on growth ...
273 base = Counter({("f.c", "misc-x"): 1})
274 if report(Counter({("f.c", "misc-x"): 2}), base) == 0:
275 failures.append("report() passed a bucket that GREW above the baseline")
276 # ... and on a bucket the baseline has never seen ...
277 if report(Counter({("new.c", "misc-x"): 1}), base) == 0:
278 failures.append("report() passed a finding in a file absent from the baseline")
279 # ... and PASS when equal or shrinking.
280 if report(Counter({("f.c", "misc-x"): 1}), base) != 0:
281 failures.append("report() failed an unchanged bucket")
282 if report(Counter(), base) != 0:
283 failures.append("report() failed a fully burned-down baseline")
284
285 # Round-trip: what is written must read back identically, or a re-baseline
286 # would silently reshape the debt it claims to be freezing.
287 original = BASELINE_FILE.read_text(encoding="ascii") if BASELINE_FILE.is_file() else None
288 try:
289 fixture = Counter({("a/b.c", "misc-x"): 3, ("a/c.c", "readability-y"): 1})
290 write_baseline(fixture)
291 if load_baseline() != fixture:
292 failures.append("write_baseline()/load_baseline() did not round-trip")
293 finally:
294 if original is None:
295 BASELINE_FILE.unlink(missing_ok=True)
296 else:
297 BASELINE_FILE.write_text(original, encoding="ascii")
298
299 return failures

302def _selftest_diag_guard() -> list[str]:
303 """Assertions about the parse-error guard (#387), in both directions."""
304 failures: list[str] = []
305
306 # A run dominated by parse errors must be refused for ratcheting ...
307 broken = Counter({(f"examples/tu{i}.c", DIAGNOSTIC_ERROR_CHECK): 1 for i in range(100)})
308 if diagnostic_error_reason(broken) is None:
309 failures.append("diagnostic_error_reason() accepted a run that is 100% parse errors")
310
311 # ... while a healthy mix carrying a few baselined parse errors is accepted.
312 healthy: Counter = Counter(
313 {(f"libs/f{i}.c", "readability-magic-numbers"): 1 for i in range(378)}
314 )
315 healthy.update({(f"examples/m{i}.c", DIAGNOSTIC_ERROR_CHECK): 1 for i in range(37)})
316 if diagnostic_error_reason(healthy) is not None:
317 failures.append("diagnostic_error_reason() rejected the healthy 37/415 baseline mix")
318
319 # A tiny run whose few findings are all parse errors must not trip (below
320 # the absolute floor): the signature is a MASS failure, not a stray error.
321 tiny = Counter({(f"x{i}.c", DIAGNOSTIC_ERROR_CHECK): 1 for i in range(5)})
322 if diagnostic_error_reason(tiny) is not None:
323 failures.append("diagnostic_error_reason() tripped on a handful of parse errors")
324
325 return failures

328def selftest() -> int:
329 """Assert the parser and the ratchet fire, in BOTH directions."""
330 failures = _selftest_parse() + _selftest_ratchet() + _selftest_diag_guard()
331
332 if failures:
333 print("SELFTEST FAILED:", file=sys.stderr)
334 for problem in failures:
335 print(f" - {problem}", file=sys.stderr)
336 return 1
337 print("selftest: clang-tidy ratchet parse + growth detection OK")
338 return 0

341def main() -> int:
342 """Ratchet clang-tidy findings against the committed baseline.
343
344 A ratchet, not a floor: ``--check`` fails when the count RISES, and
345 ``--update`` lowers the baseline once findings are fixed. The baseline can
346 therefore only move downward, which is what stops a large legacy count
347 from being permanently accepted.
348
349 ``--selftest`` asserts the comparison still fires, so a ratchet that
350 stopped detecting growth cannot pass as clean.
351
352 Returns 0 when the count is at or below the baseline, 1 when it grew.
353 """
354 parser = argparse.ArgumentParser(description=__doc__)
355 parser.add_argument("log", nargs="?", help="clang-tidy output to compare")
356 parser.add_argument("--check", action="store_true", help="gate against the baseline")
357 parser.add_argument("--update", action="store_true", help="rewrite the baseline")
358 parser.add_argument("--selftest", action="store_true", help="assert this gate still fires")
359 args = parser.parse_args()
360
361 if args.selftest:
362 return selftest()
363 if not args.log:
364 parser.error("a clang-tidy log path is required unless --selftest is given")
365
366 text = Path(args.log).read_text(encoding="utf-8", errors="replace")
367 current = parse_log(text)
368
369 # Refuse to ratchet a run that failed to parse its input, in EITHER
370 # direction, before comparing or writing anything (#387). A baseline built
371 # from a broken toolchain would freeze ~100 files of parse errors forever.
372 broken = diagnostic_error_reason(current)
373 if broken:
374 verb = "--update" if args.update else "--check"
375 print(f"refusing to {verb}: {broken}", file=sys.stderr)
376 return 1
377
378 if args.update:
379 baseline = load_baseline()
380 # Seeding the very first baseline necessarily "grows" every bucket from
381 # nothing, so the no-growth rule applies only once a baseline exists.
382 seeding = not BASELINE_FILE.is_file()
383 grew = [] if seeding else [k for k, n in current.items() if n > baseline.get(k, 0)]
384 if grew:
385 print(
386 f"refusing to --update: {len(grew)} bucket(s) would GROW. "
387 "The baseline is a burn-down; fix the new findings instead.",
388 file=sys.stderr,
389 )
390 for key in grew[:MAX_DETAIL_LINES]:
391 print(f" {key[0]} {key[1]}", file=sys.stderr)
392 return 1
393 write_baseline(current)
394 print(f"baseline updated: {sum(current.values())} finding(s) recorded")
395 return 0
396
397 return report(current, load_baseline())
