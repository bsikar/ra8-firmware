200[[gnu::section(".ns_text")]] static bool ns_exercise_veneers(void)
201{
202 /* ---- Phase C veneer milestone: call the 3 NSC CGC veneers from NS. ----
203 * Plain calls; the CMSE import library binds these names to the Secure-
204 * Gateway stubs so the call transitions into Secure (see the import note). */
205 g_tz_nsc_cgc_usb_init_step = (uint32_t)k_ns_step_pll2;
206 if (ra8_nsc_cgc_pll2_enable((uint8_t)k_ns_pll2_mul_int,
207 (uint8_t)k_ns_pll2_mul_quarters,
208 k_ra8_plodiv_div4) != k_ra8_ok) {
209 g_tz_nsc_cgc_usb_mismatch += 1U;
210 return false;
211 }
212
213 g_tz_nsc_cgc_usb_init_step = (uint32_t)k_ns_step_usbfs;
214 if (ra8_nsc_cgc_usbfs_clock_enable() != k_ra8_ok) {
215 g_tz_nsc_cgc_usb_mismatch += 1U;
216 return false;
217 }
218
219 /* Capture an NS stack address so a J-Link read confirms where MSP_NS sits
220 * relative to the SRAM NS alias (MSP_NS top = 0x32180000). A genuine NS
221 * stack reads back in 0x321xxxxx; the old broken build showed a Secure
222 * 0x220xxxxx stack here. */
223 uint32_t sp_local = 0U;
224 g_tz_nsc_cgc_usb_sp_probe = (uint32_t)(uintptr_t)&sp_local;
225 g_tz_nsc_cgc_usb_init_step = (uint32_t)k_ns_step_query;
226 /* hz_out points at an NS-resident global (in .ns_bss, inside the SAU NS
227 * region) rather than a stack local near MSP_NS, so the veneer's
228 * cmse_check_address_range guard passes. */
229 if (ra8_nsc_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0,
230 (uint32_t*)(uintptr_t)&g_tz_nsc_cgc_usb_clock_hz) != k_ra8_ok) {
231 g_tz_nsc_cgc_usb_mismatch += 1U;
232 return false;
233 }
234
235 /* All three veneers SG-trapped into Secure, ran ra8_cgc_*, and returned OK
236 * with an NS-resident pointer arg. The NSC wall works from real NS memory. */
237 g_tz_nsc_cgc_usb_init_step = (uint32_t)k_ns_step_veneers_ok;
238 return true;
239}

260[[gnu::section(".ns_text"), noreturn]] static void ns_reset_handler(void)
261{
262 /* Zero the NS BSS via uintptr_t arithmetic (cppcheck flags pointer
263 * comparison between two distinct externs as ISO C UB even though the
264 * linker fixes them to a contiguous range). */
265 const uintptr_t bss_start = (uintptr_t)&g_ra8_ls_ns_bss_start;
266 const uintptr_t bss_end = (uintptr_t)&g_ra8_ls_ns_bss_end;
267 for (uintptr_t addr = bss_start; addr < bss_end; addr += sizeof(uint32_t)) {
268 *(volatile uint32_t*)addr = 0U;
269 }
270
271 /* Phase C: exercise the three NSC CGC veneers from genuine NS memory. */
272 if (!ns_exercise_veneers()) {
273 ns_park();
274 }
275
276 /* Point VTOR_NS at the NS vector table so ThreadX's PendSV / SysTick
277 * exceptions vector to the NS handlers. ThreadX's tx_initialize_low_level
278 * deliberately leaves VTOR alone (it expects a SystemInit, which the NS
279 * image has no equivalent of), so set it here in NS state -- 0xE000ED08 is
280 * the current-domain (NS) VTOR alias. */
281 *(volatile uint32_t*)k_ns_scb_vtor_addr = (uint32_t)(uintptr_t)&g_ra8_ls_ns_run_start;
282
283 /* Phase C (#96) milestone 1: hand off to ThreadX, running entirely inside
284 * the NS image. tx_kernel_enter() initialises the kernel, calls
285 * tx_application_define() (which spawns the worker below), and starts the
286 * scheduler -- it never returns. The worker advances g_tz_nsc_cgc_usb_match,
287 * so a continuing advance proves the NS-resident RTOS is scheduling. */
288 tx_kernel_enter();
289 ns_park(); /* Unreachable -- tx_kernel_enter() does not return. */
290}
