26/* `priv_entry_first_cluster()`: see header for the documented contract. */
27uint32_t priv_entry_first_cluster(const uint8_t* entry)
28{
29 const uint32_t hi = priv_rd16(&entry[k_dir_off_fst_clus_hi]);
30 const uint32_t lo = priv_rd16(&entry[k_dir_off_fst_clus_lo]);
31 return (hi << k_shift_two_bytes) | lo;
32}

34/* `priv_entry_set_cluster_size()`: see header for the documented contract. */
35void priv_entry_set_cluster_size(uint8_t* entry, uint32_t cluster, uint32_t size)
36{
37 priv_wr16(&entry[k_dir_off_fst_clus_hi],
38 (uint16_t)((cluster >> k_shift_two_bytes) & k_word_mask));
39 priv_wr16(&entry[k_dir_off_fst_clus_lo], (uint16_t)(cluster & k_word_mask));
40 priv_wr32(&entry[k_dir_off_file_size], size);
41}

74RA8_INTERNAL
75static ra8_err_t
76priv_truncate_existing(ra8_fs_mount_t* handle, ra8_fs_file_t* f, uint64_t lba, uint32_t off)
77{
78 if (f->first_cluster >= k_cluster_first_data) {
79 ra8_err_t err = priv_free_chain(handle, f->first_cluster);
80 if (err != k_ra8_ok) {
81 return err;
82 }
83 }
84 f->first_cluster = 0;
85 f->cur_cluster = 0;
86 f->walk_cache_idx = 0;
87 f->walk_cache_cluster = 0; /* < 2: no read cache for a fresh file */
88 f->size_bytes = 0;
89 f->offset = 0;
90 uint8_t* const buf = priv_sec_walk();
91 ra8_err_t err = priv_read_sector(handle, lba, buf);
92 if (err != k_ra8_ok) {
93 return err;
94 }
95 priv_entry_set_cluster_size(&buf[off], 0, 0);
96 /* Truncation IS a content change -- the file went from N bytes to zero -- so
97 * the modification time has to move even if the caller never writes a byte
98 * afterwards. Without this, `open(write)` + `close()` left a PC-authored
99 * mtime describing contents that no longer exist (#601). */
100 priv_fat_entry_stamp_write(&buf[off]);
101 /* A content change also sets the archive attribute, the FAT/exFAT convention a
102 * backup tool clears and the OS re-sets on every modification (#681). */
103 priv_fat_entry_apply_attr(&buf[off], (uint8_t)k_ra8_fs_attr_archive, 0U);
104 return priv_write_sector(handle, lba, buf);
105}

137RA8_INTERNAL
138static ra8_err_t priv_open_existing(ra8_fs_mount_t* handle,
139 const uint8_t* entry,
140 uint64_t lba,
141 uint32_t off,
142 ra8_fs_mode_t mode,
143 ra8_fs_file_t** out_file)
144{
145 /* A directory is not a file (#604). Without this the write path ran
146 * priv_truncate_existing() on it -- freeing the chain that held every child
147 * and stamping cluster 0 / size 0 into an entry still flagged ATTR_DIRECTORY
148 * -- and the read path handed back a zero-byte handle, because a directory's
149 * DIR_FileSize is 0 by specification. Rejected for every mode, before the
150 * file slot is allocated, so a refused open consumes nothing. */
151 if ((entry[k_dir_off_attr] & (uint8_t)k_ra8_fs_attr_directory) != 0U) {
152 return k_ra8_err_invalid_arg;
153 }
154 /* Honor the read-only attribute (#681). A writing open truncates or appends,
155 * so a file a host marked read-only must be refused BEFORE the slot is taken
156 * and BEFORE priv_truncate_existing() frees its chain -- the DOS/Windows
157 * semantics say `open(read-only, write)` is denied, not silently obeyed. A
158 * read open is untouched: reading a read-only file is exactly what the bit
159 * permits. */
160 if ((mode != k_ra8_fs_mode_read) &&
161 ((entry[k_dir_off_attr] & (uint8_t)k_ra8_fs_attr_read_only) != 0U)) {
162 return k_ra8_err_access_denied;
163 }
164 ra8_fs_file_t* f = priv_alloc_file_slot();
165 if (f == nullptr) {
166 return k_ra8_err_no_mem;
167 }
168 f->mount = handle;
169 f->first_cluster = priv_entry_first_cluster(entry);
170 f->cur_cluster = f->first_cluster;
171 f->walk_cache_idx = 0U; /* read accelerator seeded at the chain head */
172 f->walk_cache_cluster = f->first_cluster;
173 f->size_bytes = priv_rd32(&entry[k_dir_off_file_size]);
174 f->dir_entry_lba = lba;
175 f->dir_entry_idx = off;
176 f->mode = mode;
177 f->no_fat_chain = 0U;
178 f->dirty = 0U;
179 f->in_use = 1;
180 if (mode == k_ra8_fs_mode_write) {
181 ra8_err_t err = priv_truncate_existing(handle, f, lba, off);
182 if (err != k_ra8_ok) {
183 f->in_use = 0;
184 return err;
185 }
186 } else if (mode == k_ra8_fs_mode_append) {
187 f->offset = f->size_bytes;
188 } else {
189 f->offset = 0;
190 }
191 *out_file = f;
192 return k_ra8_ok;
193}

276static ra8_err_t priv_enter_subdir(const ra8_fs_mount_t* m,
277 const dir_loc_t* cur,
278 const char* comp,
279 uint32_t len,
280 dir_loc_t* out)
281{
282 /* A BYTE bound, because the component is UTF-8 here: the unit-count limit is
283 * ::k_lfn_write_max and lands in priv_name_classify(). Bounding the bytes by
284 * the unit cap would refuse a perfectly storable name three characters into
285 * a Cyrillic directory (#606). */
286 if (len >= (uint32_t)k_lfn_utf8_cap) {
287 return k_ra8_err_invalid_arg;
288 }
289 /* Sized for a long name, not an 8.3 one: once `mkdir` can create
290 * `/Reading List`, every path THROUGH it has to resolve as well, and a
291 * 13-byte buffer would have refused the component before the lookup. */
292 char namebuf[k_lfn_utf8_cap] = {};
293 for (uint32_t i = 0; i < len; i++) {
294 namebuf[i] = comp[i];
295 }
296 namebuf[len] = '\0';
297 uint64_t lba = 0;
298 uint32_t off = 0;
299 uint8_t entry[k_ra8_fs_dir_entry_bytes] = {};
300 ra8_err_t err = priv_dir_lookup_any(m, cur, namebuf, &lba, &off, entry);
301 if (err != k_ra8_ok) {
302 return err;
303 }
304 if ((entry[k_dir_off_attr] & k_ra8_fs_attr_directory) == 0U) {
305 return k_ra8_err_invalid_arg;
306 }
307 const uint32_t cl = priv_entry_first_cluster(entry);
308 if (cl < (uint32_t)k_cluster_first_data) {
309 return k_ra8_err_protocol_error;
310 }
311 out->is_root = 0U;
312 out->cluster = cl;
313 return k_ra8_ok;
314}

316/* `priv_resolve_parent()`: see header for the documented contract. */
317ra8_err_t priv_resolve_parent(const ra8_fs_mount_t* m,
318 const char* path,
319 dir_loc_t* out_parent,
320 const char** out_leaf)
321{
322 const char* p = path;
323 while (*p == '/') {
324 p++;
325 }
326 dir_loc_t cur = {.is_root = 1U, .cluster = 0U};
327 for (uint32_t depth = 0; depth < (uint32_t)k_path_max_depth; depth++) {
328 const char* end = p;
329 while (*end != '\0') {
330 if (*end == '/') {
331 break;
332 }
333 end++;
334 }
335 if (*end == '\0') {
336 *out_parent = cur;
337 *out_leaf = p;
338 return k_ra8_ok;
339 }
340 dir_loc_t next = {};
341 const ra8_err_t err = priv_enter_subdir(m, &cur, p, (uint32_t)(end - p), &next);
342 if (err != k_ra8_ok) {
343 return err;
344 }
345 cur = next;
346 p = end;
347 while (*p == '/') {
348 p++;
349 }
350 }
351 return k_ra8_err_invalid_arg; /* deeper than k_path_max_depth */
352}

354/* `priv_resolve_dir()`: see header for the documented contract. */
355ra8_err_t priv_resolve_dir(const ra8_fs_mount_t* m, const char* path, dir_loc_t* out)
356{
357 const char* p = path;
358 while (*p == '/') {
359 p++;
360 }
361 if (*p == '\0') {
362 out->is_root = 1U;
363 out->cluster = 0U;
364 return k_ra8_ok;
365 }
366 dir_loc_t parent = {};
367 const char* leaf = nullptr;
368 const ra8_err_t err = priv_resolve_parent(m, path, &parent, &leaf);
369 if (err != k_ra8_ok) {
370 return err;
371 }
372 uint32_t len = 0;
373 while (leaf[len] != '\0') {
374 len++;
375 }
376 if (len == 0U) {
377 *out = parent; /* trailing slash: the parent is the directory */
378 return k_ra8_ok;
379 }
380 return priv_enter_subdir(m, &parent, leaf, len, out);
381}

416RA8_INTERNAL
417static ra8_err_t priv_create_new(ra8_fs_mount_t* handle,
418 const dir_loc_t* parent,
419 const char* leaf,
420 ra8_fs_mode_t mode,
421 ra8_fs_file_t** out_file)
422{
423 dir_insert_t plan = {};
424 ra8_err_t err = priv_dir_reserve(handle, parent, leaf, &plan);
425 if (err != k_ra8_ok) {
426 return err;
427 }
428 ra8_fs_file_t* f = priv_alloc_file_slot();
429 if (f == nullptr) {
430 return k_ra8_err_no_mem;
431 }
432 uint8_t tmpl[k_ra8_fs_dir_entry_bytes] = {};
433 tmpl[k_dir_off_attr] = (uint8_t)k_ra8_fs_attr_archive;
434 priv_entry_set_cluster_size(tmpl, 0U, 0U);
435 /* The template is zero-filled above, and a zero FAT date is not a date: month
436 * and day are both 1-based, so 0x0000 claims month 0 of day 0 and every host
437 * decodes it differently (#601). Stamped here rather than inside
438 * `priv_dir_commit()` because that same commit re-files an EXISTING entry for
439 * `rename`, which must keep the creation date it already has. */
440 priv_fat_entry_stamp_create(tmpl);
441 uint64_t lba = 0;
442 uint32_t off = 0;
443 err = priv_dir_commit(handle, &plan, tmpl, &lba, &off);
444 if (err != k_ra8_ok) {
445 return err;
446 }
447 priv_init_new_file(f, handle, mode, lba, off);
448 *out_file = f;
449 return k_ra8_ok;
450}

452/* `priv_open_locked()`: see header for the documented contract. */
453ra8_err_t priv_open_locked(ra8_fs_mount_t* handle,
454 const char* path,
455 ra8_fs_mode_t mode,
456 ra8_fs_file_t** out_file)
457{
458 if (handle == nullptr || path == nullptr || out_file == nullptr) {
459 return k_ra8_err_null_ptr;
460 }
461 if (handle->in_use == 0U) {
462 return k_ra8_err_invalid_state;
463 }
464 if (handle->type == k_ra8_fs_type_exfat) {
465 return priv_exfat_open(handle, path, mode, out_file);
466 }
467 dir_loc_t parent = {};
468 const char* leaf = nullptr;
469 const ra8_err_t rerr = priv_resolve_parent(handle, path, &parent, &leaf);
470 if (rerr != k_ra8_ok) {
471 return rerr;
472 }
473 uint64_t lba = 0;
474 uint32_t off = 0;
475 uint8_t entry[k_ra8_fs_dir_entry_bytes] = {};
476 const ra8_err_t err = priv_dir_lookup_any(handle, &parent, leaf, &lba, &off, entry);
477 if (err == k_ra8_ok) {
478 return priv_open_existing(handle, entry, lba, off, mode, out_file);
479 }
480 if (err != k_ra8_err_not_found) {
481 return err;
482 }
483 if (mode == k_ra8_fs_mode_read) {
484 return k_ra8_err_not_found;
485 }
486 /* Creation no longer needs an 8.3-representable name: `priv_dir_reserve()`
487 * generates the alias and reserves the chain's slots (#600). */
488 return priv_create_new(handle, &parent, leaf, mode, out_file);
489}

517RA8_INTERNAL
518static ra8_err_t priv_close_stamp(ra8_fs_file_t* file)
519{
520 ra8_fs_mount_t* m = file->mount;
521 /* exFAT keeps its metadata in a checksummed entry set, not one 32-byte
522 * directory entry, so the close flush is the same commit every write already
523 * performs -- stamp, patch, checksum, write (#602). */
524 if (m->type == k_ra8_fs_type_exfat) {
525 return priv_exfat_flush_set(file);
526 }
527 uint8_t* const sec = priv_sec_walk();
528 ra8_err_t err = priv_read_sector(m, file->dir_entry_lba, sec);
529 if (err != k_ra8_ok) {
530 return err;
531 }
532 priv_fat_entry_stamp_write(&sec[file->dir_entry_idx]);
533 err = priv_write_sector(m, file->dir_entry_lba, sec);
534 if (err != k_ra8_ok) {
535 return err;
536 }
537 return priv_fsinfo_flush(m);
538}

540/* `priv_close_locked()`: see header for the documented contract. */
541ra8_err_t priv_close_locked(ra8_fs_file_t* file)
542{
543 if (file == nullptr) {
544 return k_ra8_err_null_ptr;
545 }
546 ra8_err_t err = k_ra8_ok;
547 /* Never touch the volume through a handle that is already closed or whose
548 * mount has gone: the dir-entry LBA would be read against a mount slot that
549 * now describes some other card. */
550 if ((file->dirty != 0U) && (file->in_use != 0U) && (file->mount != nullptr) &&
551 (file->mount->in_use != 0U)) {
552 err = priv_close_stamp(file);
553 }
554 file->dirty = 0;
555 file->in_use = 0;
556 file->mount = nullptr;
557 return err;
558}
