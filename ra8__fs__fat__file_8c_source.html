27/* `priv_entry_first_cluster()`: see header for the documented contract. */
28uint32_t priv_entry_first_cluster(const uint8_t* entry)
29{
30 const uint32_t hi = priv_rd16(&entry[k_dir_off_fst_clus_hi]);
31 const uint32_t lo = priv_rd16(&entry[k_dir_off_fst_clus_lo]);
32 return (hi << k_shift_two_bytes) | lo;
33}

35/* `priv_entry_set_cluster_size()`: see header for the documented contract. */
36void priv_entry_set_cluster_size(uint8_t* entry, uint32_t cluster, uint32_t size)
37{
38 priv_wr16(&entry[k_dir_off_fst_clus_hi],
39 (uint16_t)((cluster >> k_shift_two_bytes) & k_word_mask));
40 priv_wr16(&entry[k_dir_off_fst_clus_lo], (uint16_t)(cluster & k_word_mask));
41 priv_wr32(&entry[k_dir_off_file_size], size);
42}

121RA8_INTERNAL
122static ra8_err_t priv_open_existing(ra8_fs_mount_t* handle,
123 const uint8_t* entry,
124 uint32_t lba,
125 uint32_t off,
126 ra8_fs_mode_t mode,
127 ra8_fs_file_t** out_file)
128{
129 ra8_fs_file_t* f = priv_alloc_file_slot();
130 if (f == nullptr) {
131 return k_ra8_err_no_mem;
132 }
133 f->mount = handle;
134 f->first_cluster = priv_entry_first_cluster(entry);
135 f->cur_cluster = f->first_cluster;
136 f->walk_cache_idx = 0U; /* read accelerator seeded at the chain head */
137 f->walk_cache_cluster = f->first_cluster;
138 f->size_bytes = priv_rd32(&entry[k_dir_off_file_size]);
139 f->dir_entry_lba = lba;
140 f->dir_entry_idx = off;
141 f->mode = mode;
142 f->no_fat_chain = 0U;
143 f->in_use = 1;
144 if (mode == k_ra8_fs_mode_write) {
145 ra8_err_t err = priv_truncate_existing(handle, f, lba, off);
146 if (err != k_ra8_ok) {
147 f->in_use = 0;
148 return err;
149 }
150 } else if (mode == k_ra8_fs_mode_append) {
151 f->offset = f->size_bytes;
152 } else {
153 f->offset = 0;
154 }
155 *out_file = f;
156 return k_ra8_ok;
157}

262static ra8_err_t priv_enter_subdir(const ra8_fs_mount_t* m,
263 const dir_loc_t* cur,
264 const char* comp,
265 uint32_t len,
266 dir_loc_t* out)
267{
268 if (len > (uint32_t)k_ra8_fs_short_name_len) {
269 return k_ra8_err_invalid_arg;
270 }
271 char namebuf[(uint32_t)k_ra8_fs_short_name_len + 1U] = {};
272 for (uint32_t i = 0; i < len; i++) {
273 namebuf[i] = comp[i];
274 }
275 namebuf[len] = '\0';
276 uint8_t name83[k_max_8_3_name] = {};
277 if (priv_path_to_83(namebuf, name83) == 0U) {
278 return k_ra8_err_invalid_arg;
279 }
280 uint32_t lba = 0;
281 uint32_t off = 0;
282 uint8_t entry[k_ra8_fs_dir_entry_bytes] = {};
283 ra8_err_t err = priv_dir_find(m, cur, name83, &lba, &off, entry);
284 if (err != k_ra8_ok) {
285 return err;
286 }
287 if ((entry[k_dir_off_attr] & k_ra8_fs_attr_directory) == 0U) {
288 return k_ra8_err_invalid_arg;
289 }
290 const uint32_t cl = priv_entry_first_cluster(entry);
291 if (cl < (uint32_t)k_cluster_first_data) {
292 return k_ra8_err_protocol_error;
293 }
294 out->is_root = 0U;
295 out->cluster = cl;
296 return k_ra8_ok;
297}

299/* `priv_resolve_parent()`: see header for the documented contract. */
300ra8_err_t priv_resolve_parent(const ra8_fs_mount_t* m,
301 const char* path,
302 dir_loc_t* out_parent,
303 const char** out_leaf)
304{
305 const char* p = path;
306 while (*p == '/') {
307 p++;
308 }
309 dir_loc_t cur = {.is_root = 1U, .cluster = 0U};
310 for (uint32_t depth = 0; depth < (uint32_t)k_path_max_depth; depth++) {
311 const char* end = p;
312 while (*end != '\0') {
313 if (*end == '/') {
314 break;
315 }
316 end++;
317 }
318 if (*end == '\0') {
319 *out_parent = cur;
320 *out_leaf = p;
321 return k_ra8_ok;
322 }
323 dir_loc_t next = {};
324 const ra8_err_t err = priv_enter_subdir(m, &cur, p, (uint32_t)(end - p), &next);
325 if (err != k_ra8_ok) {
326 return err;
327 }
328 cur = next;
329 p = end;
330 while (*p == '/') {
331 p++;
332 }
333 }
334 return k_ra8_err_invalid_arg; /* deeper than k_path_max_depth */
335}

337/* `priv_resolve_dir()`: see header for the documented contract. */
338ra8_err_t priv_resolve_dir(const ra8_fs_mount_t* m, const char* path, dir_loc_t* out)
339{
340 const char* p = path;
341 while (*p == '/') {
342 p++;
343 }
344 if (*p == '\0') {
345 out->is_root = 1U;
346 out->cluster = 0U;
347 return k_ra8_ok;
348 }
349 dir_loc_t parent = {};
350 const char* leaf = nullptr;
351 const ra8_err_t err = priv_resolve_parent(m, path, &parent, &leaf);
352 if (err != k_ra8_ok) {
353 return err;
354 }
355 uint32_t len = 0;
356 while (leaf[len] != '\0') {
357 len++;
358 }
359 if (len == 0U) {
360 *out = parent; /* trailing slash: the parent is the directory */
361 return k_ra8_ok;
362 }
363 return priv_enter_subdir(m, &parent, leaf, len, out);
364}

452ra8_err_t
453ra8_fs_open(ra8_fs_mount_t* handle, const char* path, ra8_fs_mode_t mode, ra8_fs_file_t** out_file)
454{
455 if (handle == nullptr || path == nullptr || out_file == nullptr) {
456 return k_ra8_err_null_ptr;
457 }
458 if (handle->in_use == 0U) {
459 return k_ra8_err_invalid_state;
460 }
461 if (handle->type == k_ra8_fs_type_exfat) {
462 return priv_exfat_open(handle, path, mode, out_file);
463 }
464 dir_loc_t parent = {};
465 const char* leaf = nullptr;
466 const ra8_err_t rerr = priv_resolve_parent(handle, path, &parent, &leaf);
467 if (rerr != k_ra8_ok) {
468 return rerr;
469 }
470 uint8_t name83[k_max_8_3_name] = {};
471 const uint8_t have83 = priv_path_to_83(leaf, name83);
472
473 uint32_t lba = 0;
474 uint32_t off = 0;
475 uint8_t entry[k_ra8_fs_dir_entry_bytes] = {};
476 ra8_err_t err = k_ra8_err_not_found;
477 if (have83 != 0U) {
478 err = priv_dir_find(handle, &parent, name83, &lba, &off, entry);
479 }
480 if (err == k_ra8_err_not_found) {
481 /* VFAT long-name fallback: a name that is not 8.3-representable (e.g. a
482 * 4-char ".epub" extension), or an 8.3 lookup that missed, can still match a
483 * file's long name. */
484 err = priv_dir_find_long(handle, &parent, leaf, &lba, &off, entry);
485 }
486 if (err == k_ra8_ok) {
487 return priv_open_existing(handle, entry, lba, off, mode, out_file);
488 }
489 if (err != k_ra8_err_not_found) {
490 return err;
491 }
492 if (mode == k_ra8_fs_mode_read) {
493 return k_ra8_err_not_found;
494 }
495 /* Creation needs an 8.3-representable name (LFN write is out of scope, #101). */
496 if (have83 == 0U) {
497 return k_ra8_err_invalid_arg;
498 }
499 return priv_create_new(handle, &parent, name83, mode, out_file);
500}
