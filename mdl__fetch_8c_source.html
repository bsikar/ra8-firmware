208RA8_INTERNAL static bool copy_file(const char* src, const char* dst)
209{
210 if ((src == nullptr) || (dst == nullptr)) {
211 return false;
212 }
213 FILE* in = fopen(src, "rb");
214 if (in == nullptr) {
215 return false;
216 }
217 /* Same rule as every other writer here: a partial copy must not be able to
218 * destroy a good `dst` that already exists (see mdl_atomic.h). */
219 char tmp[PATH_MAX];
220 if (!mdl_atomic_tmp_path(dst, tmp, sizeof(tmp))) {
221 (void)fclose(in);
222 return false;
223 }
224 FILE* out = fopen(tmp, "wb");
225 if (out == nullptr) {
226 (void)fclose(in);
227 return false;
228 }
229 bool ok = true;
230 uint8_t buf[k_fetch_copy_chunk];
231 for (uint32_t chunk = 0U; chunk < (uint32_t)k_fetch_copy_max_chunks; ++chunk) {
232 const size_t n = fread(buf, 1U, sizeof(buf), in);
233 if ((n > 0U) && (fwrite(buf, 1U, n, out) != n)) {
234 ok = false;
235 break;
236 }
237 if (n < sizeof(buf)) {
238 ok = (ferror(in) == 0);
239 break;
240 }
241 }
242 ok = ((fclose(out) == 0) && ok);
243 (void)fclose(in);
244 if (!ok) {
245 mdl_atomic_abort(tmp);
246 return false;
247 }
248 return mdl_atomic_commit(tmp, dst);
249}

269{

271 if (held == nullptr) {
272 return false;
273 }

277 return false;

385RA8_INTERNAL static ra8_err_t fetch_one_page(mdl_fetch_ctx_t* ctx,
386 const char* chapter_url,
387 const char* dest_abs,
388 const char* dest_rel,
389 size_t page_no,
390 size_t idx,
391 mdl_chapter_rec_t* rec,
392 mdl_fetch_stats_t* stats,
393 mdl_page_outcome_t* out)
394{
395 const char* url = ctx->images->urls[idx];
396 char leaf[k_fetch_leaf_bytes];
397 if (!page_leaf(page_no, url, leaf, sizeof(leaf))) {
398 return k_ra8_fail;
399 }
400 char target_abs[PATH_MAX];
401 char target_rel[k_mdl_relpath_max];
402 const int an = snprintf(target_abs, sizeof(target_abs), "%s/%s", dest_abs, leaf);
403 const int rn = snprintf(target_rel, sizeof(target_rel), "%s/%s", dest_rel, leaf);
404 if ((an < 0) || ((size_t)an >= sizeof(target_abs)) || (rn < 0) ||
405 ((size_t)rn >= sizeof(target_rel))) {
406 return k_ra8_fail;
407 }
408 if (try_reuse(ctx, mdl_hash_str(url), target_abs, target_rel)) {
409 stats->pages_reused += 1U;
410 *out = (mdl_page_outcome_t){.bytes = 0U, .elapsed_ms = 0U, .reused = true};
411 } else {
412 const int64_t t0 = mono_ms(ctx);
413 size_t got = 0U;
414 const ra8_err_t rc = do_fetch_page(ctx, chapter_url, url, target_abs, target_rel, &got);
415 if (rc != k_ra8_ok) {
416 stats->pages_failed += 1U;
417 return rc;
418 }
419 stats->pages_fetched += 1U;
420 *out = (mdl_page_outcome_t){.bytes = (uint64_t)got,
421 .elapsed_ms = (uint32_t)(mono_ms(ctx) - t0),
422 .reused = false};
423 }
424 rec->pages_done = (uint16_t)(idx + 1U);
425 checkpoint(ctx);
426 return k_ra8_ok;
427}
