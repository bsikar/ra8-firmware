118RA8_INTERNAL static void internal_c6_cam_on_event(void* ctx, const ra8_c6link_event_t* event)
119{
120 (void)ctx;
121 if (event == nullptr) {
122 return;
123 }
124 s_events++;
125 if (event->kind == k_ra8_c6link_event_sta_connected) {
126 s_connected = 1U;
127 } else if (event->kind == k_ra8_c6link_event_sta_disconnected) {
128 s_disconnected = 1U;
129 s_disconnect_reason = event->reason;
130 } else {
131 /* Boot and bare Wi-Fi events are counted but need no state latch. */
132 }

197RA8_INTERNAL static bool internal_c6_cam_wait_connected(void)
198{
199 for (uint32_t i = 0U; i < (uint32_t)k_c6_cam_assoc_polls; i++) {
200 (void)ra8_c6link_poll(&s_link, (uint16_t)k_ra8_c6link_announce_transfers, nullptr);
201 if (s_connected != 0U) {
202 return true;

204 if (s_disconnected != 0U) {

266RA8_INTERNAL static bool internal_c6_cam_associate(ra8_c6link_mac_t* mac,
267 const ra8_net_credentials_t* credentials)
268{
269 ra8_err_t err = ra8_c6link_wifi_start(&s_link);
270 if (err != k_ra8_ok) {
271 internal_c6_cam_report_fault("wifi_start", err);
272 return false;
273 }
274 err = ra8_c6link_wifi_mac(&s_link, mac);
275 if (err != k_ra8_ok) {
276 internal_c6_cam_report_fault("wifi_mac", err);
277 return false;
278 }
279 ra8_c6link_sta_cfg_t station = {};
280 err = ra8_c6link_sta_cfg_set(&station, credentials->ssid, credentials->psk);
281 if (err != k_ra8_ok) {
282 ra8_secure_memzero(&station, sizeof(station));
283 internal_c6_cam_report_fault("sta_cfg_set", err);
284 return false;
285 }
286 s_connected = 0U;
287 s_disconnected = 0U;
288 s_disconnect_reason = 0U;
289 err = ra8_c6link_wifi_join(&s_link, &station);
290 ra8_secure_memzero(&station, sizeof(station));
291 if (err != k_ra8_ok) {
292 internal_c6_cam_report_fault("wifi_join", err);
293 return false;
294 }
295 if (!internal_c6_cam_wait_connected()) {
296 c6_cam_puts("c6_cam: FAIL association reason=");
297 c6_cam_put_u32((uint32_t)s_disconnect_reason);
298 c6_cam_puts(" events=");
299 c6_cam_put_u32((uint32_t)s_events);
300 c6_cam_puts("\r\n");
301 return false;
302 }
303 c6_cam_puts("c6_cam: associated events=");
304 c6_cam_put_u32((uint32_t)s_events);
305 c6_cam_puts("\r\n");
306 return true;
307}

432RA8_INTERNAL static void internal_c6_cam_worker_entry(ULONG input)
433{
434 (void)input;
435 if (s_port_err != k_ra8_ok) {
436 c6_cam_puts("c6_cam: FAIL esp-hosted port init err=");
437 c6_cam_puts(ra8_err_to_str(s_port_err));
438 c6_cam_puts("\r\n");
439 internal_c6_cam_halt();
440 }
441 ra8_delay_ms((uint32_t)k_c6_cam_boot_wait_ms);
442 ra8_net_credentials_t credentials = {};
443 const ra8_err_t provisioned = ra8_net_provision_receive(&s_provision_uart,
444 (uint32_t)k_ra8_net_provision_timeout_ms,
445 &credentials);
446 if (provisioned != k_ra8_ok) {
447 internal_c6_cam_report_fault("runtime provisioning", provisioned);
448 ra8_net_provision_clear(&credentials);
449 internal_c6_cam_halt();
450 }
451 if (!internal_c6_cam_prepare_media()) {
452 ra8_net_provision_clear(&credentials);
453 internal_c6_cam_halt();
454 }
455 if (!internal_c6_cam_prepare_link()) {
456 ra8_net_provision_clear(&credentials);
457 internal_c6_cam_halt();
458 }
459 c6_cam_lease_t lease = {};
460 if (!internal_c6_cam_join(&lease, &credentials)) {
461 ra8_net_provision_clear(&credentials);
462 internal_c6_cam_halt();
463 }
464 ra8_net_provision_clear(&credentials);
465 c6_cam_puts("c6_cam: PASS Wi-Fi and DHCP ip=");
466 c6_cam_put_ip(lease.ip);
467 c6_cam_puts("\r\n");
468 c6_cam_puts("\r\nc6_cam: open http://");
469 c6_cam_put_ip(lease.ip);
470 c6_cam_puts("/\r\n");
471 c6_cam_http_serve();
472}
