183 uint64_t any_lba = 0U;

185 const uint32_t sector = i / eps;
186 const uint32_t offset = (i % eps) * (uint32_t)k_gpt_entry_bytes;
187 if (offset == 0U) {

189 if (err != k_ra8_ok) {
190 return err;
191 }
192 }
193 priv_gpt_note_entry(&s_scratch[offset], &basic_lba, &any_lba);
194 }
195 if (basic_lba != 0U) {

197 return k_ra8_ok;
198 }
199 if (any_lba != 0U) {

231static ra8_err_t priv_gpt_read_geom(ra8_fs_mount_t* m, uint64_t* out_entry_lba, uint32_t* out_count)
232{
233 const ra8_err_t err = priv_read_sector(m, (uint64_t)k_gpt_header_lba, s_scratch);
234 if (err != k_ra8_ok) {
235 return err;
236 }
237 for (uint32_t i = 0U; i < (uint32_t)k_gpt_sig_len; i++) {
238 if (s_scratch[i] != k_gpt_signature[i]) {
239 return k_ra8_err_validation_failed;
240 }
241 }
242 const uint64_t entry_lba = priv_rd64(&s_scratch[k_gpt_off_entry_lba]);
243 const uint32_t entry_size = priv_rd32(&s_scratch[k_gpt_off_entry_size]);
244 uint32_t count = priv_rd32(&s_scratch[k_gpt_off_entry_count]);
245 if (entry_lba == 0U) {
246 return k_ra8_err_validation_failed;
247 }
248 if (entry_size != (uint32_t)k_gpt_entry_bytes) {
249 return k_ra8_err_not_supported;
250 }
251 if (count > (uint32_t)k_gpt_entry_scan_max) {
252 count = (uint32_t)k_gpt_entry_scan_max;
253 }
254 *out_entry_lba = entry_lba;
255 *out_count = count;
256 return k_ra8_ok;
