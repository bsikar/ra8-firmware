113void board_periph_register_block(const board_periph_block_t* block)
114{
115 if (block == nullptr) {
116 return;
117 }
118 if (s_block_count >= (uint32_t)k_block_max) {
119 /* NEVER drop a block silently: an un-registered block reads as an
120 * unmodelled peripheral and fails its apps with no hint of the cause. */
121 (void)fprintf(stderr,
122 "board_periph: block registry FULL (k_block_max=%u) -- "
123 "DROPPING '%s'; raise k_block_max\n",
124 (unsigned)k_block_max,
125 (block->name != nullptr) ? block->name : "?");
126 return;
127 }
128 s_blocks[s_block_count] = block;
129 s_block_count++;
130 s_order_built = false; /* a new block invalidates the cached tick/report order */
131}

145static void board_periph_build_order(void)
146{
147 for (uint32_t i = 0U; i < s_block_count; i++) {
148 uint32_t j = i;
149 while ((j > 0U) && (s_blocks[s_block_order[j - 1U]]->order > s_blocks[i]->order)) {
150 s_block_order[j] = s_block_order[j - 1U];
151 j--;
152 }
153 s_block_order[j] = (uint8_t)i;
154 }
155 s_order_built = true;
156}

219static const board_periph_block_t* block_for_addr(uint64_t addr)
220{
221 if ((s_last_block != nullptr) && in_range(addr, s_last_block->base, s_last_block->span)) {
222 return s_last_block;
223 }
224 for (uint32_t i = 0U; i < s_block_count; i++) {
225 if (in_range(addr, s_blocks[i]->base, s_blocks[i]->span)) {
226 if (!block_on_active_device(s_blocks[i])) {
227 continue; /* RA8P1-only block on a non-RA8P1 run: fall through to sparse. */
228 }
229 s_last_block = s_blocks[i];
230 return s_blocks[i];
231 }
232 }
233 return nullptr;
234}

241/* ICU IELSR event-link table. The ICU registers live inside the callback-MMIO
242 * peripheral window (0x40006xxx), so the model owns this state itself rather
243 * than reading it back through the MMIO callback. The firmware's IELSR writes
244 * (ra8_isr_register / ra8_icu_route) land here; icu_raise_event scans it. */
245static uint32_t s_ielsr[k_icu_ielsr_cnt];
246
247/* Pending-IRQ ring: the ICU pushes a line here when a linked, NVIC-enabled
248 * event fires; the run loop pops it and the engine vectors it in. */
249static uint32_t s_irq_ring[k_irq_queue_len];
250static uint32_t s_irq_head;
251static uint32_t s_irq_tail;
252static uint32_t s_irq_taken[k_irq_track_max];
253static uint32_t s_irq_total;

255/* NVIC set-enable shadow. The Cortex-M NVIC ISER/ICER registers are
256 * set-enable / clear-enable (a written 1 sets / clears that line, a written 0
257 * is ignored), but ra8_emulator maps the PPB as plain RAM, so a raw store of
258 * "1 << bit" to ISER would clobber every other already-enabled line. main.c
259 * decodes ISER/ICER writes and routes the set/clear bits here, where this
260 * accumulating mask is the authoritative enable state the ICU model checks --
261 * so a firmware that enables several NVIC lines (e.g. SCI RXI + TXI + TEI, or
262 * the USB controller lines later) has them all stay enabled. */
263typedef enum : uint32_t {
264 k_nvic_enable_words = 8U,
265} nvic_shadow_t;
266static uint32_t s_nvic_iser_shadow[k_nvic_enable_words];

282static bool nvic_enabled(uc_engine* uc, uint32_t irq)
283{
284 (void)uc; /* enable state is the accumulating shadow, not raw PPB RAM */
285 const uint32_t word = irq / 32U;
286 if (word >= (uint32_t)k_nvic_enable_words) {
287 return false;
288 }
289 return ((s_nvic_iser_shadow[word] >> (irq % 32U)) & 1U) != 0U;
290}

293static void nvic_set_pending(uc_engine* uc, uint32_t irq)
294{
295 const uint64_t word = (uint64_t)k_nvic_ispr_base + ((uint64_t)(irq / 32U) * 4U);
296 uint32_t ispr = periph_rd32(uc, word);
297 ispr |= (1U << (irq % 32U));
298 (void)uc_mem_write(uc, word, &ispr, sizeof(ispr));
299}

301void board_periph_nvic_set_enable(uint32_t irq, bool enable)
302{
303 const uint32_t word = irq / 32U;
304 if (word >= (uint32_t)k_nvic_enable_words) {
305 return;
306 }
307 const uint32_t bit = 1U << (irq % 32U);
308 if (enable) {
309 s_nvic_iser_shadow[word] |= bit;
310 } else {
311 s_nvic_iser_shadow[word] &= ~bit;
312 }

321static void irq_ring_push(uint32_t irq)
322{
323 const uint32_t next = (s_irq_tail + 1U) % (uint32_t)k_irq_queue_len;
324 if (next == s_irq_head) {
325 return; /* ring full: the run loop drains it every boundary, so rare */
326 }
327 s_irq_ring[s_irq_tail] = irq;

331void board_periph_icu_raise_event(uc_engine* uc, uint16_t event)
332{
333 for (uint32_t slot = 0U; slot < (uint32_t)k_icu_ielsr_cnt; slot++) {
334 if ((s_ielsr[slot] & (uint32_t)k_ielsr_iels_mask) != (uint32_t)event) {
335 continue;
336 }
337 s_ielsr[slot] |= (uint32_t)k_ielsr_ir_mask; /* latch IR (W0C by handler) */
338 if (nvic_enabled(uc, slot)) {
339 nvic_set_pending(uc, slot);
340 irq_ring_push(slot);
341 if (s_trace) {
342 (void)fprintf(stderr, " [trace] ICU event 0x%03X -> NVIC IRQ %u pended\n", event, slot);
343 }
344 }
345 return; /* one IELSR slot owns a given event */
346 }

362static uint32_t icu_ielsr_slot(uint64_t addr)
363{
364 const uint64_t lo = (uint64_t)k_icu_base + (uint64_t)k_icu_off_ielsr;
365 const uint64_t hi = lo + ((uint64_t)k_icu_ielsr_cnt * 4U);
366 if ((addr < lo) || (addr >= hi)) {
367 return (uint32_t)k_icu_ielsr_cnt;
368 }
369 return (uint32_t)((addr - lo) / 4U);

380static void icu_write(uint64_t addr, uint32_t value)
381{
382 const uint32_t slot = icu_ielsr_slot(addr);
383 if (slot >= (uint32_t)k_icu_ielsr_cnt) {
384 return;
385 }
386 /* IR (bit 16) is WRITE-ZERO-to-clear, not write-one-to-clear: HUM Ch 14.2.17
387 * p 547, "IR flag (Interrupt Status Flag)" -> [Clearing condition] -- "the IR
388 * flag is cleared to 0 by writing 0". A written 0 clears the latched flag; a
389 * written 1 leaves it set. Every other bit takes the written value.
390 *
391 * This model had the polarity inverted (RW1C), which silently satisfied a
392 * driver that ORed a 1 into IR to "clear" it -- a write real silicon ignores.
393 * The emulator therefore ran every ISR-driven app cleanly while the bench
394 * live-locked in an interrupt storm (#170: lpm_periodic_idle emitted no UART
395 * at all, stuck in IRQ0_Handler with IELSR0 = 0x00010080). Modelling the real
396 * polarity makes the emulator reproduce that storm, so the bug cannot hide here
397 * again. */
398 const uint32_t ir_now = s_ielsr[slot] & (uint32_t)k_ielsr_ir_mask;
399 const uint32_t ir_keep = ((value & (uint32_t)k_ielsr_ir_mask) != 0U) ? ir_now : 0U;
400 s_ielsr[slot] = (value & ~(uint32_t)k_ielsr_ir_mask) | ir_keep;
401}

429void board_periph_init(bool trace)
430{
431 s_trace = trace;
432 s_last_block = nullptr; /* drop the dispatch cache across a warm reboot */
433 if (!s_order_built) {
434 board_periph_build_order();
435 }
436 /* Reset every registered block to its power-on state. */
437 for (uint32_t i = 0U; i < s_block_count; i++) {
438 const board_periph_block_t* b = s_blocks[s_block_order[i]];
439 if (b->reset != nullptr) {
440 b->reset();
441 }
442 }
443 /* Core ICU / NVIC bookkeeping. */
444 for (uint32_t i = 0U; i < (uint32_t)k_icu_ielsr_cnt; i++) {
445 s_ielsr[i] = 0U;
446 }
447 for (uint32_t i = 0U; i < (uint32_t)k_irq_track_max; i++) {
448 s_irq_taken[i] = 0U;
449 }
450 for (uint32_t i = 0U; i < (uint32_t)k_nvic_enable_words; i++) {
451 s_nvic_iser_shadow[i] = 0U;
452 }
453 s_irq_head = 0U;
454 s_irq_tail = 0U;
455 s_irq_total = 0U;
456 /* Bring the USBFS controller model + virtual host up and wire its interrupt
457 * pend through this module's ICU/NVIC path (Phase 3, issue #67). */
458 board_usb_init(trace);
459 board_usb_set_irq_raiser(usb_irq_raiser);
460 /* USBHS HOST-mode model (self-loop peer of the USBFS device above). Dormant
461 * until main.c grants the loop AND the firmware selects host mode. */
462 board_usb_host_init(trace);
463}

470uint64_t board_periph_read(uc_engine* uc, uint64_t addr, unsigned size, bool* handled)
471{
472 *handled = true;
473 const board_periph_block_t* block = block_for_addr(addr);
474 if ((block != nullptr) && !block->observe) {
475 if (board_mstp_addr_stopped(addr)) {
476 /* Module-stopped peripheral: unclocked on silicon, so it reads 0 and the
477 * modelled block must NOT answer (#405). This is the same fidelity shape
478 * as the #247 graphics power domain -- a block whose enable was never
479 * cancelled does nothing on hardware, and now nothing in the emulator. */
480 board_mstp_note_gated_access(addr, false);
481 return 0U;
482 }
483 return block->read(uc, addr, size);
484 }
485 /* An observe-only block (e.g. GLCDC) does not own its window: fall through so
486 * the caller's sparse model answers the read, exactly as if it were
487 * unmodelled. */
488 {
489 bool usb_hit = false;
490 const uint64_t usb_val = board_usb_read(uc, addr, size, &usb_hit);
491 if (usb_hit) {
492 return usb_val;
493 }
494 }
495 {
496 bool usbh_hit = false;
497 const uint64_t usbh_val = board_usb_host_read(uc, addr, size, &usbh_hit);
498 if (usbh_hit) {
499 return usbh_val;
500 }
501 }
502 if (icu_ielsr_slot(addr) < (uint32_t)k_icu_ielsr_cnt) {
503 return icu_read(addr);
504 }
505 *handled = false;
506 return 0U;
507}

509void board_periph_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value, bool* handled)
510{
511 *handled = true;
512 const board_periph_block_t* block = block_for_addr(addr);
513 if (block != nullptr) {
514 if (!block->observe && board_mstp_addr_stopped(addr)) {
515 /* Module-stopped peripheral: unclocked on silicon, so the write is
516 * dropped (#405). Reported handled -- the access is consumed and
517 * discarded exactly as the hardware discards it (no fault, no flag). */
518 board_mstp_note_gated_access(addr, true);
519 return;
520 }
521 block->write(uc, addr, size, value);
522 /* An observe-only block snoops the write but does not consume it: report
523 * NOT handled so the caller's sparse model also records it (the panel
524 * compositor reads the GLCDC registers back from that shadow). */
525 *handled = !block->observe;
526 return;
527 }
528 {
529 bool usb_hit = false;
530 board_usb_write(uc, addr, size, value, &usb_hit);
531 if (usb_hit) {
532 return;
533 }
534 }
535 {
536 bool usbh_hit = false;
537 board_usb_host_write(uc, addr, size, value, &usbh_hit);
538 if (usbh_hit) {
539 return;
540 }
541 }
542 if (icu_ielsr_slot(addr) < (uint32_t)k_icu_ielsr_cnt) {
543 icu_write(addr, (uint32_t)value);
544 return;
545 }
546 *handled = false;
547}

620void board_periph_report(uc_engine* uc)
621{
622 (void)uc;
623 bool irq_done = false;
624 for (uint32_t i = 0U; i < s_block_count; i++) {
625 const board_periph_block_t* b = s_blocks[s_block_order[i]];
626 /* The core's NVIC-IRQ section sits between SCI and the touch line. */
627 if (!irq_done && (b->order >= (uint32_t)k_core_irq_report_order)) {
628 report_irqs();

630 }
631 if (b->report != nullptr) {
