112void board_periph_register_block(const board_periph_block_t* block)
113{
114 if (block == nullptr) {
115 return;
116 }
117 if (s_block_count >= (uint32_t)k_block_max) {
118 /* NEVER drop a block silently: an un-registered block reads as an
119 * unmodelled peripheral and fails its apps with no hint of the cause. */
120 (void)fprintf(stderr,
121 "board_periph: block registry FULL (k_block_max=%u) -- "
122 "DROPPING '%s'; raise k_block_max\n",
123 (unsigned)k_block_max,
124 (block->name != nullptr) ? block->name : "?");
125 return;
126 }
127 s_blocks[s_block_count] = block;
128 s_block_count++;
129 s_order_built = false; /* a new block invalidates the cached tick/report order */
130}

144static void board_periph_build_order(void)
145{
146 for (uint32_t i = 0U; i < s_block_count; i++) {
147 uint32_t j = i;
148 while ((j > 0U) && (s_blocks[s_block_order[j - 1U]]->order > s_blocks[i]->order)) {
149 s_block_order[j] = s_block_order[j - 1U];
150 j--;
151 }
152 s_block_order[j] = (uint8_t)i;
153 }
154 s_order_built = true;
155}

218static const board_periph_block_t* block_for_addr(uint64_t addr)
219{
220 if ((s_last_block != nullptr) && in_range(addr, s_last_block->base, s_last_block->span)) {
221 return s_last_block;
222 }
223 for (uint32_t i = 0U; i < s_block_count; i++) {
224 if (in_range(addr, s_blocks[i]->base, s_blocks[i]->span)) {
225 if (!block_on_active_device(s_blocks[i])) {
226 continue; /* RA8P1-only block on a non-RA8P1 run: fall through to sparse. */
227 }
228 s_last_block = s_blocks[i];
229 return s_blocks[i];
230 }
231 }
232 return nullptr;
233}

240/* ICU IELSR event-link table. The ICU registers live inside the callback-MMIO
241 * peripheral window (0x40006xxx), so the model owns this state itself rather
242 * than reading it back through the MMIO callback. The firmware's IELSR writes
243 * (ra8_isr_register / ra8_icu_route) land here; icu_raise_event scans it. */
244static uint32_t s_ielsr[k_icu_ielsr_cnt];
245
246/* Pending-IRQ ring: the ICU pushes a line here when a linked, NVIC-enabled
247 * event fires; the run loop pops it and the engine vectors it in. */
248static uint32_t s_irq_ring[k_irq_queue_len];
249static uint32_t s_irq_head;
250static uint32_t s_irq_tail;
251static uint32_t s_irq_taken[k_irq_track_max];
252static uint32_t s_irq_total;

254/* NVIC set-enable shadow. The Cortex-M NVIC ISER/ICER registers are
255 * set-enable / clear-enable (a written 1 sets / clears that line, a written 0
256 * is ignored), but ra8_emulator maps the PPB as plain RAM, so a raw store of
257 * "1 << bit" to ISER would clobber every other already-enabled line. main.c
258 * decodes ISER/ICER writes and routes the set/clear bits here, where this
259 * accumulating mask is the authoritative enable state the ICU model checks --
260 * so a firmware that enables several NVIC lines (e.g. SCI RXI + TXI + TEI, or
261 * the USB controller lines later) has them all stay enabled. */
262typedef enum : uint32_t {
263 k_nvic_enable_words = 8U,
264} nvic_shadow_t;
265static uint32_t s_nvic_iser_shadow[k_nvic_enable_words];

281static bool nvic_enabled(uc_engine* uc, uint32_t irq)
282{
283 (void)uc; /* enable state is the accumulating shadow, not raw PPB RAM */
284 const uint32_t word = irq / 32U;
285 if (word >= (uint32_t)k_nvic_enable_words) {
286 return false;
287 }
288 return ((s_nvic_iser_shadow[word] >> (irq % 32U)) & 1U) != 0U;
289}

292static void nvic_set_pending(uc_engine* uc, uint32_t irq)
293{
294 const uint64_t word = (uint64_t)k_nvic_ispr_base + ((uint64_t)(irq / 32U) * 4U);
295 uint32_t ispr = periph_rd32(uc, word);
296 ispr |= (1U << (irq % 32U));
297 (void)uc_mem_write(uc, word, &ispr, sizeof(ispr));
298}

300void board_periph_nvic_set_enable(uint32_t irq, bool enable)
301{
302 const uint32_t word = irq / 32U;
303 if (word >= (uint32_t)k_nvic_enable_words) {
304 return;
305 }
306 const uint32_t bit = 1U << (irq % 32U);
307 if (enable) {
308 s_nvic_iser_shadow[word] |= bit;
309 } else {
310 s_nvic_iser_shadow[word] &= ~bit;
311 }

321{

323 if (next == s_irq_head) {
324 return; /* ring full: the run loop drains it every boundary, so rare */

330void board_periph_icu_raise_event(uc_engine* uc, uint16_t event)
331{
332 for (uint32_t slot = 0U; slot < (uint32_t)k_icu_ielsr_cnt; slot++) {
333 if ((s_ielsr[slot] & (uint32_t)k_ielsr_iels_mask) != (uint32_t)event) {
334 continue;
335 }
336 s_ielsr[slot] |= (uint32_t)k_ielsr_ir_mask; /* latch IR (W0C by handler) */
337 if (nvic_enabled(uc, slot)) {
338 nvic_set_pending(uc, slot);
339 irq_ring_push(slot);
340 if (s_trace) {
341 (void)fprintf(stderr, " [trace] ICU event 0x%03X -> NVIC IRQ %u pended\n", event, slot);
342 }
343 }
344 return; /* one IELSR slot owns a given event */
345 }

361static uint32_t icu_ielsr_slot(uint64_t addr)
362{
363 const uint64_t lo = (uint64_t)k_icu_base + (uint64_t)k_icu_off_ielsr;
364 const uint64_t hi = lo + ((uint64_t)k_icu_ielsr_cnt * 4U);
365 if ((addr < lo) || (addr >= hi)) {
366 return (uint32_t)k_icu_ielsr_cnt;
367 }
368 return (uint32_t)((addr - lo) / 4U);

379static void icu_write(uint64_t addr, uint32_t value)
380{
381 const uint32_t slot = icu_ielsr_slot(addr);
382 if (slot >= (uint32_t)k_icu_ielsr_cnt) {
383 return;
384 }
385 /* IR (bit 16) is WRITE-ZERO-to-clear, not write-one-to-clear: HUM Ch 14.2.17
386 * p 547, "IR flag (Interrupt Status Flag)" -> [Clearing condition] -- "the IR
387 * flag is cleared to 0 by writing 0". A written 0 clears the latched flag; a
388 * written 1 leaves it set. Every other bit takes the written value.
389 *
390 * This model had the polarity inverted (RW1C), which silently satisfied a
391 * driver that ORed a 1 into IR to "clear" it -- a write real silicon ignores.
392 * The emulator therefore ran every ISR-driven app cleanly while the bench
393 * live-locked in an interrupt storm (#170: lpm_periodic_idle emitted no UART
394 * at all, stuck in IRQ0_Handler with IELSR0 = 0x00010080). Modelling the real
395 * polarity makes the emulator reproduce that storm, so the bug cannot hide here
396 * again. */
397 const uint32_t ir_now = s_ielsr[slot] & (uint32_t)k_ielsr_ir_mask;
398 const uint32_t ir_keep = ((value & (uint32_t)k_ielsr_ir_mask) != 0U) ? ir_now : 0U;
399 s_ielsr[slot] = (value & ~(uint32_t)k_ielsr_ir_mask) | ir_keep;
400}

428void board_periph_init(bool trace)
429{
430 s_trace = trace;
431 s_last_block = nullptr; /* drop the dispatch cache across a warm reboot */
432 if (!s_order_built) {
433 board_periph_build_order();
434 }
435 /* Reset every registered block to its power-on state. */
436 for (uint32_t i = 0U; i < s_block_count; i++) {
437 const board_periph_block_t* b = s_blocks[s_block_order[i]];
438 if (b->reset != nullptr) {
439 b->reset();
440 }
441 }
442 /* Core ICU / NVIC bookkeeping. */
443 for (uint32_t i = 0U; i < (uint32_t)k_icu_ielsr_cnt; i++) {
444 s_ielsr[i] = 0U;
445 }
446 for (uint32_t i = 0U; i < (uint32_t)k_irq_track_max; i++) {
447 s_irq_taken[i] = 0U;
448 }
449 for (uint32_t i = 0U; i < (uint32_t)k_nvic_enable_words; i++) {
450 s_nvic_iser_shadow[i] = 0U;
451 }
452 s_irq_head = 0U;
453 s_irq_tail = 0U;
454 s_irq_total = 0U;
455 /* Bring the USBFS controller model + virtual host up and wire its interrupt
456 * pend through this module's ICU/NVIC path (Phase 3, issue #67). */
457 board_usb_init(trace);
458 board_usb_set_irq_raiser(usb_irq_raiser);
459 /* USBHS HOST-mode model (self-loop peer of the USBFS device above). Dormant
460 * until main.c grants the loop AND the firmware selects host mode. */
461 board_usb_host_init(trace);
462}

469uint64_t board_periph_read(uc_engine* uc, uint64_t addr, unsigned size, bool* handled)
470{
471 *handled = true;
472 const board_periph_block_t* block = block_for_addr(addr);
473 if ((block != nullptr) && !block->observe) {
474 if (board_mstp_addr_stopped(addr)) {
475 /* Module-stopped peripheral: unclocked on silicon, so it reads 0 and the
476 * modelled block must NOT answer (#405). This is the same fidelity shape
477 * as the #247 graphics power domain -- a block whose enable was never
478 * cancelled does nothing on hardware, and now nothing in the emulator. */
479 board_mstp_note_gated_access(addr, false);
480 return 0U;
481 }
482 return block->read(uc, addr, size);
483 }
484 /* An observe-only block (e.g. GLCDC) does not own its window: fall through so
485 * the caller's sparse model answers the read, exactly as if it were
486 * unmodelled. */
487 {
488 bool usb_hit = false;
489 const uint64_t usb_val = board_usb_read(uc, addr, size, &usb_hit);
490 if (usb_hit) {
491 return usb_val;
492 }
493 }
494 {
495 bool usbh_hit = false;
496 const uint64_t usbh_val = board_usb_host_read(uc, addr, size, &usbh_hit);
497 if (usbh_hit) {
498 return usbh_val;
499 }
500 }
501 if (icu_ielsr_slot(addr) < (uint32_t)k_icu_ielsr_cnt) {
502 return icu_read(addr);
503 }
504 *handled = false;
505 return 0U;
506}

508void board_periph_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value, bool* handled)
509{
510 *handled = true;
511 const board_periph_block_t* block = block_for_addr(addr);
512 if (block != nullptr) {
513 if (!block->observe && board_mstp_addr_stopped(addr)) {
514 /* Module-stopped peripheral: unclocked on silicon, so the write is
515 * dropped (#405). Reported handled -- the access is consumed and
516 * discarded exactly as the hardware discards it (no fault, no flag). */
517 board_mstp_note_gated_access(addr, true);
518 return;
519 }
520 block->write(uc, addr, size, value);
521 /* An observe-only block snoops the write but does not consume it: report
522 * NOT handled so the caller's sparse model also records it (the panel
523 * compositor reads the GLCDC registers back from that shadow). */
524 *handled = !block->observe;
525 return;
526 }
527 {
528 bool usb_hit = false;
529 board_usb_write(uc, addr, size, value, &usb_hit);
530 if (usb_hit) {
531 return;
532 }
533 }
534 {
535 bool usbh_hit = false;
536 board_usb_host_write(uc, addr, size, value, &usbh_hit);
537 if (usbh_hit) {
538 return;
539 }
540 }
541 if (icu_ielsr_slot(addr) < (uint32_t)k_icu_ielsr_cnt) {
542 icu_write(addr, (uint32_t)value);
543 return;
544 }
545 *handled = false;
546}

619void board_periph_report(uc_engine* uc)
620{
621 (void)uc;
622 bool irq_done = false;
623 for (uint32_t i = 0U; i < s_block_count; i++) {
624 const board_periph_block_t* b = s_blocks[s_block_order[i]];
625 /* The core's NVIC-IRQ section sits between SCI and the touch line. */
626 if (!irq_done && (b->order >= (uint32_t)k_core_irq_report_order)) {
627 report_irqs();
628 irq_done = true;
629 }
630 if (b->report != nullptr) {
631 b->report();
632 }

635 report_irqs();
