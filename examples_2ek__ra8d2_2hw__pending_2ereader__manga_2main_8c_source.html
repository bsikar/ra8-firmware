113/*
114 * ::k_mg_cells is DERIVED from the panel + tile geometry, never hand-picked
115 * (#338). The reader pages the atlas one tile at a time (get / blit / put), so
116 * the cache does not need every on-screen tile pinned at once -- but it must
117 * still be able to *hold* the tiles a frame touches, or a one-tile pan evicts
118 * tiles that are still on screen and re-inflates them next frame. That is the
119 * thrash: with a sub-frame budget every pan re-decodes ~a full screen of
120 * 64 KiB DEFLATE tiles.
121 *
122 * At 1:1 a 1024x600 panel with a 48 px status bar shows a 1024x552 content
123 * window over 256x256 tiles, straddling at most ::k_mg_view_cols x
124 * ::k_mg_view_rows = 5 x 4 = 20 tiles. Sizing the cache to that frame plus one
125 * tile of margin on each axis (::k_mg_cells = 6 x 5 = 30, ~1.9 MiB of the
126 * 64 MB SDRAM) means the union of any single-step pan's before/after frames
127 * stays resident, so a pan re-decodes only the newly exposed row/column, never
128 * a tile still on screen. `tests/test_app_ereader_manga.c`
129 * (test_manga_pan_no_thrash) measures the eviction/decode count for a
130 * representative pan against the old 4-cell budget and asserts it collapses.
131 *
132 * Two compile-time tripwires keep the budget honest:
133 * - the first static_assert fails the build if ::k_mg_cells is ever set below
134 * the frame demand (a manual override, or a geometry change that outgrows
135 * it), the condition that reintroduces the thrash;
136 * - the second bounds the resident-tile arena by ::k_mg_cell_budget_bytes so
137 * the derivation cannot silently balloon the SDRAM working set.
138 * The 4-cell budget lives on only as the explicit eviction-stress case in the
139 * host test, which is what it was always right for.
140 */
141static_assert((uint64_t)k_mg_cells >= (uint64_t)k_mg_view_cols * (uint64_t)k_mg_view_rows,
142 "tile cache is smaller than the 1:1 viewport tile demand: re-derive k_mg_cells from "
143 "the panel + tile geometry so a single-step pan cannot thrash (#338)");
144static_assert((uint64_t)k_mg_cells * (uint64_t)k_mg_cell_bytes <= (uint64_t)k_mg_cell_budget_bytes,
145 "resident tile arena exceeds its SDRAM budget: the tile-cache derivation grew past "
146 "k_mg_cell_budget_bytes");

148/*
149 * ::k_mg_work_bytes is DERIVED, never hand-picked: it is exactly
150 * `ra8_jof_work_bytes(k_mg_cap_edge, k_mg_cap_edge, k_mg_tile_edge,
151 * k_mg_tile_edge)` -- the producer's own worst-case arena requirement for the
152 * cap this app advertises, taken at the format's maximum 4 bpp. A round number
153 * here is how the advertised cap and the real capability drift apart: a 2 MiB
154 * arena backs a cap of only 1016 px, so the app used to advertise 2048 and
155 * fail-closed (`k_ra8_err_invalid_size`) on any in-budget source wider than
156 * that. Three layers keep the two in agreement:
157 * - the static_assert below is a compile-time tripwire on the band term
158 * (`cap * tile_h * bpp_max`), a strict lower bound on the full
159 * requirement, so raising ::k_mg_cap_edge without re-deriving the arena
160 * fails the build;
161 * - `tests/test_app_ereader_manga.c` asserts the arena against the real
162 * `ra8_jof_work_bytes()` return, which is the exact and
163 * authoritative check, and runs in CI;
164 * - `mg_build_atlas()` re-checks the same call at boot, so a mismatch that
165 * somehow reaches silicon reports `FAIL atlas` instead of silently
166 * under-delivering the advertised cap.
167 * The band term alone is a lower bound, not the whole requirement, so the
168 * static_assert can only catch an under-sized arena, never certify a
169 * sufficient one -- that is the test's and the boot check's job.
170 */
171static_assert((uint64_t)k_mg_work_bytes >=
172 ((uint64_t)k_mg_cap_edge * (uint64_t)k_mg_tile_edge * (uint64_t)k_ra8_jof_bpp_max),
173 "work arena is below the producer band term for the advertised cap: re-derive "
174 "k_mg_work_bytes from ra8_jof_work_bytes() for k_mg_cap_edge");

291static void mg_print_hex(uint32_t value)
292{
293 uint8_t buf[k_mg_hex_nibbles];
294 for (uint32_t i = 0U; i < (uint32_t)k_mg_hex_nibbles; ++i) {
295 const uint32_t shift = ((uint32_t)k_mg_hex_nibbles - 1U - i) * (uint32_t)k_mg_nibble_bits;
296 const uint32_t nib = (value >> shift) & (uint32_t)k_mg_nibble_mask;
297 buf[i] = (uint8_t)((nib < (uint32_t)k_mg_dec_ten) ? ('0' + nib) : ('A' + (nib - k_mg_dec_ten)));
298 }
299 mg_print(buf, (uint32_t)k_mg_hex_nibbles);
300}

303static void mg_print_uint(uint32_t value)
304{
305 uint8_t buf[k_mg_dec_ten];
306 uint32_t n = 0U;
307 if (value == 0U) {
308 buf[n] = '0';
309 n++;
310 }
311 while ((value > 0U) && (n < (uint32_t)k_mg_dec_ten)) {
312 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_mg_dec_ten));
313 n++;
314 value /= (uint32_t)k_mg_dec_ten;
315 }
316 for (uint32_t i = 0U; i < n; ++i) {
317 mg_print(&buf[n - 1U - i], 1U);
318 }

395static ra8_err_t mg_png_pull(void* ctx, uint8_t* buf, size_t cap, size_t* got)
396{
397 mg_png_cursor_t* c = (mg_png_cursor_t*)ctx;
398 const uint32_t avail = c->len - c->pos;
399 size_t take = ((uint32_t)cap < avail) ? cap : (size_t)avail;
400 (void)memcpy(buf, &c->data[c->pos], take);
401 c->pos += (uint32_t)take;
402 *got = take;
403 return k_ra8_ok;
404}

407static bool mg_build_atlas(void)
408{
409 /* The advertised cap must be one this arena can actually serve. The producer
410 * carves from the *decoded* geometry, so an under-sized arena would not show
411 * up until some in-budget source happened to be wide enough or deep enough in
412 * bpp to exhaust it; asking the producer for its own requirement up front
413 * turns that latent capability lie into a deterministic boot failure. */
414 const uint32_t need = ra8_jof_work_bytes((uint16_t)k_mg_cap_edge,
415 (uint16_t)k_mg_cap_edge,
416 (uint16_t)k_mg_tile_edge,
417 (uint16_t)k_mg_tile_edge);
418 if ((need == 0U) || (need > (uint32_t)sizeof(s_work))) {
419 return false;
420 }
421 s_png_cursor = (mg_png_cursor_t){.data = k_mg_png, .len = k_mg_png_len, .pos = 0U};
422 s_store = (ra8_jof_memstore_t){.buf = s_store_buf, .cap = sizeof(s_store_buf), .len = 0U};
423 const ra8_jof_produce_cfg_t cfg = {
424 .pull = mg_png_pull,
425 .pull_ctx = &s_png_cursor,
426 .sink = ra8_jof_memstore_sink,
427 .sink_ctx = &s_store,
428 .tile_w = (uint16_t)k_mg_tile_edge,
429 .tile_h = (uint16_t)k_mg_tile_edge,
430 /* Deflate tiles (codec 1): each tile is an intra-coded raw-DEFLATE stream,
431 * so the 3 MiB page normalizes to a ~20 KiB atlas and every cache miss does
432 * real decode-on-demand work (one bounded inflate into the pinned cell) --
433 * the streaming-larger-than-RAM property the tile cache exists for. On the
434 * 1 GHz M85 a 64 KiB-tile inflate is microseconds; the cost only shows under
435 * the ra8_emulator CPU emulator, where boot still completes well inside budget. */
436 .codec = (uint8_t)k_ra8_jof_codec_deflate,
437 .max_width = (uint16_t)k_mg_cap_edge,
438 .max_height = (uint16_t)k_mg_cap_edge,
439 .work = s_work,
440 .work_cap = sizeof(s_work),
441 .webp_work = nullptr,
442 };
443 if (ra8_jof_produce(&cfg, &s_info) != k_ra8_ok) {
444 return false;
445 }
446 return ra8_jof_parse(ra8_jof_memstore_pread, &s_store, s_store.len, &s_info) == k_ra8_ok;
447}

493static void mg_poll_touch(bool* was_touching)
494{
495 ra8_touch_point_t pt = {};
496 uint8_t got = 0U;
497 if (ra8_touch_read(&pt, 1U, &got) != k_ra8_ok) {
498 return;
499 }
500 const bool touching = (got > 0U);
501 if (touching && !(*was_touching)) {
502 if (mg_reader_tap(&s_reader, (int32_t)pt.x, (int32_t)pt.y)) {
503 if (mg_reader_render(&s_reader) == k_ra8_ok) {
504 mg_present();
505 /* Idle-window read-ahead: warm the next pan step's tiles so a follow-on
506 * pan finds them resident instead of stalling on a cold decode (#341). */
507 (void)mg_reader_prefetch(&s_reader);
508 }
509 }
510 }
511 *was_touching = touching;
512}
