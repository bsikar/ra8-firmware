175void main(void)
176{
177 internal_gpt_3p_demo_setup_or_halt();
178 ra8_isr_globals_enable();
179
180 if (internal_gpt_3p_demo_arm() != k_ra8_ok) {
181 internal_gpt_3p_demo_panic_halt();
182 }
183
184 /* Cycle the three-phase duties via ra8_gpt_three_phase_set_duty
185 * once per iteration. The call exercises GTCCR writes on all
186 * three channels and proves the driver's open state is intact.
187 * On a scope the U/V/W outputs would sweep their duty cycles in
188 * lockstep; on the HIL bench we just count successful set_duty
189 * calls. */
190 uint32_t duty_offset = 0U;
191 while (1) {
192 const uint32_t period = (uint32_t)k_gpt_3p_demo_period;
193 const uint32_t u_duty = ((uint32_t)k_gpt_3p_demo_duty_u + duty_offset) & period;
194 const uint32_t v_duty = ((uint32_t)k_gpt_3p_demo_duty_v + duty_offset) & period;
195 const uint32_t w_duty = ((uint32_t)k_gpt_3p_demo_duty_w + duty_offset) & period;
196 if (ra8_gpt_three_phase_set_duty(u_duty, v_duty, w_duty) == k_ra8_ok) {
197 g_gpt_three_phase_match += 1U;
198 (void)ra8_board_led_toggle(k_ra8_board_led1);
199 } else {
200 g_gpt_three_phase_mismatch += 1U;
201 }
202 duty_offset = (duty_offset + k_gpt_duty_advance) & period;
203 ra8_delay_ms((uint32_t)k_gpt_3p_demo_step_ms);
204 }
205 internal_gpt_3p_demo_panic_halt();
206}
