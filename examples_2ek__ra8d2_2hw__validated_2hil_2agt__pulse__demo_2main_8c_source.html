175static void agt_pulse_format_tick(uint32_t tick, uint8_t* digits)
176{
177 for (uint32_t i = 0U; i < (uint32_t)k_agt_pulse_hex_digits; ++i) {
178 const uint32_t shift =
179 ((uint32_t)k_agt_pulse_hex_digits - 1U - i) * (uint32_t)k_agt_pulse_hex_shift;
180 const uint32_t nib = (tick >> shift) & (uint32_t)k_agt_pulse_hex_mask;
181 digits[i] = (uint8_t)k_agt_pulse_hex_chars[nib];
182 }
183}

205[[nodiscard]] static ra8_err_t agt_pulse_emit_tick(uint32_t tick)
206{
207 uint8_t digits[k_agt_pulse_hex_digits] = {};
208 agt_pulse_format_tick(tick, digits);
209 ra8_err_t err =
210 ra8_board_uart_console_write(k_agt_pulse_msg_head, (size_t)(sizeof(k_agt_pulse_msg_head) - 1U));
211 if (err != k_ra8_ok) {
212 return err;
213 }
214 err = ra8_board_uart_console_write(digits, (size_t)sizeof(digits));
215 if (err != k_ra8_ok) {
216 return err;
217 }
218 return ra8_board_uart_console_write(k_agt_pulse_msg_tail,
219 (size_t)(sizeof(k_agt_pulse_msg_tail) - 1U));
220}

223#pragma GCC diagnostic ignored "-Wmain"
224int32_t main(void)
225{
226 agt_pulse_setup_or_halt();
227 ra8_isr_globals_enable();
228
229 if (agt_pulse_arm() != k_ra8_ok) {
230 agt_pulse_panic_halt();
231 }
232
233 uint32_t tick = 0U;
234 while (1) {
235 uint8_t status = 0U;
236 if (ra8_agt_get_status((uint8_t)k_agt_pulse_channel, &status) != k_ra8_ok) {
237 break;
238 }
239 if ((status & (uint8_t)k_ra8_agt_agtcr_tundf_msk) != 0U) {
240 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
241 break;
242 }
243 ++tick;
244 if (agt_pulse_emit_tick(tick) != k_ra8_ok) {
245 break;
246 }
247 /* Stop + re-arm to clear AGTCR.TUNDF on real silicon. */
248 if (ra8_agt_stop((uint8_t)k_agt_pulse_channel) != k_ra8_ok) {
249 break;
250 }
251 if (agt_pulse_arm() != k_ra8_ok) {
252 break;
253 }
254 }
255 ra8_delay_ms((uint32_t)k_agt_pulse_poll_ms);
256 }
257 agt_pulse_panic_halt();
258 return 0;
259}
