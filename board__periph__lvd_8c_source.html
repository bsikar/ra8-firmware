77static uint64_t lvd_read(uc_engine* uc, uint64_t addr, unsigned size)
78{
79 (void)uc;
80 (void)size;
81 const uint64_t off = addr - (uint64_t)k_lvd_base;
82 if (off == (uint64_t)k_lvd_off_pvd1cr1) {
83 return s_lvd.pvd1cr1;
84 }
85 if (off == (uint64_t)k_lvd_off_pvd2cr1) {
86 return s_lvd.pvd2cr1;
87 }
88 if ((off == (uint64_t)k_lvd_off_pvd1sr) || (off == (uint64_t)k_lvd_off_pvd2sr)) {
89 s_lvd.reads++;
90 return (uint64_t)k_lvd_sr_healthy; /* MON = above, DET = 0. */
91 }
92 return 0U;
93}

96static void lvd_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
97{
98 (void)uc;
99 (void)size;
100 const uint64_t off = addr - (uint64_t)k_lvd_base;
101 if (off == (uint64_t)k_lvd_off_pvd1cr1) {
102 s_lvd.pvd1cr1 = (uint8_t)value;
103 } else if (off == (uint64_t)k_lvd_off_pvd2cr1) {
104 s_lvd.pvd2cr1 = (uint8_t)value;
105 }
106 /* PVDmSR writes (DET write-0-to-clear) are absorbed: MON stays "above". */
107}
