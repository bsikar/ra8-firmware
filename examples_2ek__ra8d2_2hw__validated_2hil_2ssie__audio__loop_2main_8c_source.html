116[[nodiscard]] static ra8_err_t ssie_loop_run(void)
117{
118 const ra8_ssie_cfg_t cfg = {
119 .role = k_ra8_ssie_role_controller,
120 .format = k_ra8_ssie_format_i2s,
121 .data_word = k_ra8_ssie_dwl_16,
122 .system_word = k_ra8_ssie_swl_32,
123 .bclk_div = k_ra8_ssie_bclk_div_32,
124 .use_gpt_clk = false,
125 .long_frame = true,
126 .bckp_rising = false,
127 .lrckp_low = false,
128 .spdp_high = false,
129 .byte_swap = false,
130 .lr_continue = false,
131 .bck_idle_stop = false,
132 .enable_aucke = true,
133 .tx_threshold = (uint8_t)k_ssie_loop_tx_threshold,
134 .rx_threshold = (uint8_t)k_ssie_loop_rx_threshold,
135 };
136 ra8_err_t err = ra8_ssie_init((uint8_t)k_ssie_loop_channel, &cfg);
137 if (err != k_ra8_ok) {
138 return err;
139 }
140 err = ra8_ssie_start((uint8_t)k_ssie_loop_channel, k_ra8_ssie_dir_tx_rx);
141 if (err != k_ra8_ok) {
142 return err;
143 }
144 for (uint8_t i = 0U; i < (uint8_t)k_ssie_loop_sample_count; i++) {
145 err = ra8_ssie_write_sample((uint8_t)k_ssie_loop_channel, k_ssie_loop_pattern[i]);
146 if (err != k_ra8_ok) {
147 return err;
148 }
149 }
150 return k_ra8_ok;
151}

170 while (1) {
171 __asm__ volatile("wfi");
172 }
173 return 0;
174}
