108static size_t cm_ct_read(void* ctx, uint64_t off, void* buf, size_t len)
109{
110 const cm_ct_blob_t* s = (const cm_ct_blob_t*)ctx;
111 if ((s == nullptr) || (buf == nullptr) || (off >= (uint64_t)s->n)) {
112 return 0U;
113 }
114 const uint64_t avail = (uint64_t)s->n - off;
115 const size_t k = (len > (size_t)avail) ? (size_t)avail : len;
116 (void)memcpy(buf, &s->d[off], k);
117 return k;
118}

121static uint32_t cm_ct_fnv(uint32_t h, const uint8_t* p, size_t n)
122{
123 for (size_t i = 0U; i < n; ++i) {
124 h = (h ^ (uint32_t)p[i]) * (uint32_t)k_ct_fnv_prime;
125 }
126 return h;
127}

178static bool cm_ct_hash_tiles(const ra8_jof_info_t* info, uint32_t* out_crc)
179{
180 uint32_t crc = (uint32_t)k_ct_fnv_offset;
181 for (uint16_t ty = 0U; ty < info->tile_rows; ++ty) {
182 for (uint16_t tx = 0U; tx < info->tile_cols; ++tx) {
183 ra8_tile_t tile = {};
184 if (ra8_comic_tiles_tile(&s_ct_reader, tx, ty, &tile) != k_ra8_ok) {
185 return false;
186 }
187 const size_t n = (size_t)tile.width * (size_t)tile.height * (size_t)info->bpp;
188 crc = cm_ct_fnv(crc, tile.pixels, n);
189 if (ra8_comic_tiles_release(&s_ct_reader, tile.pixels) != k_ra8_ok) {
190 return false;
191 }
192 }
193 }
194 *out_crc = crc;
195 return true;
196}

215cm_tiled_result_t cm_comic_tiled_selfcheck(void)
216{
217 cm_tiled_result_t r = {};
218 size_t got = 0U;
219 if (!cm_ct_open(&got)) {
220 return r;
221 }
222 uint16_t w = 0U;
223 uint16_t h = 0U;
224 uint64_t dec = 0U;
225 if (ra8_comic_tiles_footprint(s_ct_pagebuf, got, &w, &h, &dec) != k_ra8_ok) {
226 (void)ra8_comic_close(&s_ct_comic);
227 return r;
228 }
229 if (!ra8_comic_tiles_over_budget(dec, (uint64_t)k_ct_budget)) {
230 (void)ra8_comic_close(&s_ct_comic);
231 return r; /* fixture must be oversized; a fit page is not the case under test */
232 }
233 ra8_tile_cache_cfg_t storage = cm_ct_cache_cfg();
234 if (ra8_comic_tiles_init(&s_ct_reader, &storage, s_ct_scratch, (uint32_t)k_ct_scratch) !=
235 k_ra8_ok) {
236 (void)ra8_comic_close(&s_ct_comic);
237 return r;
238 }
239 const ra8_comic_tiles_import_cfg_t cfg = {
240 .tile_w = (uint16_t)k_ct_tile,
241 .tile_h = (uint16_t)k_ct_tile,
242 .codec = (uint8_t)k_ra8_jof_codec_deflate,
243 .max_width = (uint16_t)k_comic_large_w,
244 .max_height = (uint16_t)k_comic_large_h,
245 .work = s_ct_work,
246 .work_cap = sizeof s_ct_work,
247 .webp_work = nullptr,
248 .webp_work_cap = 0U,
249 .atlas = s_ct_atlas,
250 .atlas_cap = sizeof s_ct_atlas,
251 };
252 if (ra8_comic_tiles_import(&s_ct_reader, s_ct_pagebuf, got, &cfg) != k_ra8_ok) {
253 (void)ra8_comic_close(&s_ct_comic);
254 return r;
255 }
256 ra8_jof_info_t info = {};
257 if (ra8_comic_tiles_info(&s_ct_reader, &info) != k_ra8_ok) {
258 (void)ra8_comic_close(&s_ct_comic);
259 return r;
260 }
261 uint32_t crc = 0U;
262 r.ok = cm_ct_hash_tiles(&info, &crc);
263 r.w = (uint32_t)info.width;
264 r.h = (uint32_t)info.height;
265 r.tiles = info.tile_count;
266 r.crc = r.ok ? crc : 0U;
267 (void)ra8_comic_close(&s_ct_comic);
268 return r;
