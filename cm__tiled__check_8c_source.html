107static size_t cm_ct_read(void* ctx, uint64_t off, void* buf, size_t len)
108{
109 const cm_ct_blob_t* s = (const cm_ct_blob_t*)ctx;
110 if ((s == nullptr) || (buf == nullptr) || (off >= (uint64_t)s->n)) {
111 return 0U;
112 }
113 const uint64_t avail = (uint64_t)s->n - off;
114 const size_t k = (len > (size_t)avail) ? (size_t)avail : len;
115 (void)memcpy(buf, &s->d[off], k);
116 return k;
117}

120static uint32_t cm_ct_fnv(uint32_t h, const uint8_t* p, size_t n)
121{
122 for (size_t i = 0U; i < n; ++i) {
123 h = (h ^ (uint32_t)p[i]) * (uint32_t)k_ct_fnv_prime;
124 }
125 return h;
126}

177static bool cm_ct_hash_tiles(const ra8_jof_info_t* info, uint32_t* out_crc)
178{
179 uint32_t crc = (uint32_t)k_ct_fnv_offset;
180 for (uint16_t ty = 0U; ty < info->tile_rows; ++ty) {
181 for (uint16_t tx = 0U; tx < info->tile_cols; ++tx) {
182 ra8_tile_t tile = {};
183 if (ra8_comic_tiles_tile(&s_ct_reader, tx, ty, &tile) != k_ra8_ok) {
184 return false;
185 }
186 const size_t n = (size_t)tile.width * (size_t)tile.height * (size_t)info->bpp;
187 crc = cm_ct_fnv(crc, tile.pixels, n);
188 if (ra8_comic_tiles_release(&s_ct_reader, tile.pixels) != k_ra8_ok) {
189 return false;
190 }
191 }
192 }
193 *out_crc = crc;
194 return true;
195}

214cm_tiled_result_t cm_comic_tiled_selfcheck(void)
215{
216 cm_tiled_result_t r = {};
217 size_t got = 0U;
218 if (!cm_ct_open(&got)) {
219 return r;
220 }
221 uint16_t w = 0U;
222 uint16_t h = 0U;
223 uint64_t dec = 0U;
224 if (ra8_comic_tiles_footprint(s_ct_pagebuf, got, &w, &h, &dec) != k_ra8_ok) {
225 (void)ra8_comic_close(&s_ct_comic);
226 return r;
227 }
228 if (!ra8_comic_tiles_over_budget(dec, (uint64_t)k_ct_budget)) {
229 (void)ra8_comic_close(&s_ct_comic);
230 return r; /* fixture must be oversized; a fit page is not the case under test */
231 }
232 ra8_tile_cache_cfg_t storage = cm_ct_cache_cfg();
233 if (ra8_comic_tiles_init(&s_ct_reader, &storage, s_ct_scratch, (uint32_t)k_ct_scratch) !=
234 k_ra8_ok) {
235 (void)ra8_comic_close(&s_ct_comic);
236 return r;
237 }
238 const ra8_comic_tiles_import_cfg_t cfg = {
239 .tile_w = (uint16_t)k_ct_tile,
240 .tile_h = (uint16_t)k_ct_tile,
241 .codec = (uint8_t)k_ra8_jof_codec_deflate,
242 .max_width = (uint16_t)k_comic_large_w,
243 .max_height = (uint16_t)k_comic_large_h,
244 .work = s_ct_work,
245 .work_cap = sizeof s_ct_work,
246 .webp_work = nullptr,
247 .webp_work_cap = 0U,
248 .atlas = s_ct_atlas,
249 .atlas_cap = sizeof s_ct_atlas,
250 };
251 if (ra8_comic_tiles_import(&s_ct_reader, s_ct_pagebuf, got, &cfg) != k_ra8_ok) {
252 (void)ra8_comic_close(&s_ct_comic);
253 return r;
254 }
255 ra8_jof_info_t info = {};
256 if (ra8_comic_tiles_info(&s_ct_reader, &info) != k_ra8_ok) {
257 (void)ra8_comic_close(&s_ct_comic);
258 return r;
259 }
260 uint32_t crc = 0U;
261 r.ok = cm_ct_hash_tiles(&info, &crc);
262 r.w = (uint32_t)info.width;
263 r.h = (uint32_t)info.height;
264 r.tiles = info.tile_count;
265 r.crc = r.ok ? crc : 0U;
266 (void)ra8_comic_close(&s_ct_comic);
267 return r;
