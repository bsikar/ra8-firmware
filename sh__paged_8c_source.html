110static ra8_err_t sh_paged_mram_read(void* ctx, uint64_t offset, uint8_t* buf, uint32_t len)
111{
112 RA8_CHECK_NULL_PTR(ctx, s_shp_tag, "mram: null ctx");
113 RA8_CHECK_NULL_PTR(buf, s_shp_tag, "mram: null buf");
114 const sh_paged_mram_t* m = (const sh_paged_mram_t*)ctx;
115 if ((offset + (uint64_t)len) > m->len) {
116 return k_ra8_err_out_of_range;
117 }
118 (void)memcpy(buf, &m->data[offset], (size_t)len);
119 return k_ra8_ok;
120}

145static bool sh_paged_hdr_ok(const ra8_book_header_t* hdr, uint64_t inflated_total)
146{
147 const char k_magic[8] = {'R', 'A', 'B', 'O', 'O', 'K', '1', '\0'};
148 if (memcmp(hdr->magic, k_magic, sizeof k_magic) != 0) {
149 return false;
150 }
151 if (hdr->format_version != (uint32_t)k_ra8_book_format_version) {
152 return false;
153 }
154 if ((uint64_t)hdr->total_size != inflated_total) {
155 return false;
156 }
157 if (hdr->total_size < (uint32_t)sizeof(ra8_book_header_t)) {
158 return false;
159 }
160 const struct {
161 uint32_t off;
162 uint32_t count;
163 uint32_t elem;
164 } k_ext[] = {
165 {hdr->chapter_off, hdr->chapter_count, (uint32_t)sizeof(ra8_book_chapter_t)},
166 {hdr->node_off, hdr->node_count, (uint32_t)sizeof(ra8_book_node_t)},
167 {hdr->attr_off, hdr->attr_count, (uint32_t)sizeof(ra8_book_attr_t)},
168 {hdr->stylesheet_off, hdr->stylesheet_count, (uint32_t)sizeof(ra8_book_stylesheet_t)},
169 {hdr->image_off, hdr->image_count, (uint32_t)sizeof(ra8_book_image_t)},
170 {hdr->string_off, hdr->string_size, 1U},
171 {hdr->image_pool_off, hdr->image_pool_size, 1U},
172 };
173 for (size_t i = 0U; i < (sizeof(k_ext) / sizeof(k_ext[0])); ++i) {
174 if (!sh_paged_extent_ok(k_ext[i].off, k_ext[i].count, k_ext[i].elem, hdr->total_size)) {
175 return false;
176 }
177 }
178 return true;
179}

183{

209 return false;
210 }

222bool sh_paged_open(ra8_vsource_read_fn file_read, void* file_ctx, uint64_t file_len)
223{
224 /* Unbind any previous source, but do NOT sh_paged_close() here: the caller
225 * has already torn the old book down (sh_book_open), and a full close would
226 * wipe the MRAM context / close the SD file that @p file_read is about to
227 * read from. The failure paths below do close, releasing the SD backing. */
228 g_sh.book_src = (ra8_book_src_t){};
229 s_shp_rd = (ra8_book_chunked_t){};
230 if ((file_read == nullptr) || (file_len == 0U)) {
231 sh_paged_close();
232 return false;
233 }
234 if (ra8_book_chunked_open(&s_shp_rd,
235 file_read,
236 file_ctx,
237 file_len,
238 sh_inflate,
239 s_shp_table,
240 (uint32_t)k_shp_table_entries,
241 s_shp_staging,
242 (uint32_t)k_shp_staging_bytes) != k_ra8_ok) {
243 sh_paged_close();
244 return false;
245 }
246 /* One chunk == one cache frame: the read callback's contract. A container
247 * compiled with a different --chunk-bytes cannot be paged through this
248 * cache, so reject it here instead of failing obscurely per read. */
249 if (s_shp_rd.chunk_bytes != (uint32_t)k_shp_frame_bytes) {
250 sh_paged_close();
251 return false;
252 }
253 /* ra8_book_src_t carries a uint32 size; the table budget already caps books
254 * at 64 MiB, so this guard is belt-and-braces against a lying header. */
255 if (s_shp_rd.inflated_total > (uint64_t)UINT32_MAX) {
256 sh_paged_close();
257 return false;
258 }
259 if (!sh_paged_bind()) {
260 sh_paged_close();
261 return false;
262 }
263 return true;
264}
