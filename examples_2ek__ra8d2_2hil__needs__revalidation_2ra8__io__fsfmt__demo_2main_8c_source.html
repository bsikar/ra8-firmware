88static bool stub_probe(const ra8_fs_backend_t* be)
89{
90 if (be == nullptr) {
91 return false;
92 }
93 if (be->read_block == nullptr) {
94 return false;

98 return false;
99 }
100 return blk[0] == (uint8_t)k_demo_stub_magic;
101}

143static ra8_err_t demo_probe_fat(const ra8_io_fsfmt_t** out_fmt)
144{
145 RA8_CHECK_NULL_PTR(out_fmt, s_tag, "out_fmt");
146 RA8_RETURN_ON_ERROR(
147 ra8_io_blockdev_ram_init(&s_bd, &s_bstate, s_disk, (uint32_t)k_demo_disk_blocks, false),
148 s_tag,
149 "blockdev init");
150 RA8_RETURN_ON_ERROR(ra8_io_blockdev_as_fs_backend(&s_bd, &s_be), s_tag, "fs bridge");
151 ra8_fs_format_opts_t opts = {};
152 opts.type = k_ra8_fs_type_fat12;
153 opts.label = "RAIO";
154 RA8_RETURN_ON_ERROR(ra8_fs_format(&s_be, &opts), s_tag, "format");
155
156 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_init(), s_tag, "fsfmt init");
157 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_probe(&s_be, out_fmt), s_tag, "probe fat");
158 if (strcmp((*out_fmt)->name, "fat") != 0) {
159 return k_ra8_err_not_found;
160 }
161 if ((*out_fmt)->caps.read_only || !(*out_fmt)->caps.supports_streaming_write) {
162 return k_ra8_err_not_supported;
163 }
164 if ((*out_fmt)->caps.max_name_len != (uint16_t)k_demo_fat_maxname) {
165 return k_ra8_err_invalid_size;
166 }
167 return k_ra8_ok;
168}

171static ra8_err_t demo_probe_foreign(void)
172{
173 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_register(&k_demo_stub), s_tag, "register stub");
174
175 ra8_io_blockdev_t bd = {};
176 ra8_io_blockdev_ram_state_t state = {};
177 RA8_RETURN_ON_ERROR(
178 ra8_io_blockdev_ram_init(&bd, &state, s_stub_disk, (uint32_t)k_demo_stub_blocks, false),
179 s_tag,
180 "stub blockdev");
181 uint8_t blk[(size_t)k_ra8_io_block_size_bytes] = {};
182 blk[0] = (uint8_t)k_demo_stub_magic;
183 RA8_RETURN_ON_ERROR(ra8_io_blockdev_write(&bd, 0, 1, blk), s_tag, "stub mark");
184
185 ra8_fs_backend_t be = {};
186 RA8_RETURN_ON_ERROR(ra8_io_blockdev_as_fs_backend(&bd, &be), s_tag, "stub bridge");
187 const ra8_io_fsfmt_t* fmt = nullptr;
188 RA8_RETURN_ON_ERROR(ra8_io_fsfmt_probe(&be, &fmt), s_tag, "probe stub");
189 if (strcmp(fmt->name, "stub") != 0) {
190 return k_ra8_err_not_found;
191 }
192 if (!fmt->caps.read_only || !fmt->caps.case_sensitive) {
193 return k_ra8_err_not_supported;
194 }
195 return k_ra8_ok;
196}
