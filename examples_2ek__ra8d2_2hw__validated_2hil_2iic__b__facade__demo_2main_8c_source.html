183static void iic_print_dec(uint32_t value)
184{
185 uint8_t buf[k_iic_dec_digits];
186 uint32_t n = 0U;
187 if (value == 0U) {
188 buf[n] = (uint8_t)'0';
189 n++;
190 }
191 while ((value > 0U) && (n < (uint32_t)k_iic_dec_digits)) {
192 buf[n] = (uint8_t)((uint32_t)'0' + (value % (uint32_t)k_iic_dec_ten));
193 n++;
194 value /= (uint32_t)k_iic_dec_ten;
195 }
196 for (uint32_t i = 0U; i < n; i++) {
197 iic_print(&buf[n - 1U - i], 1U);
198 }
199}

282[[nodiscard]] static ra8_err_t iic_facade_service(uint8_t* last_byte, iic_facade_svc_t* out_svc)
283{
284 uint8_t mask = 0U;
285 if (ra8_i3c_peripheral_status((uint8_t)k_iic_channel, &mask) != k_ra8_ok) {
286 return k_ra8_err_hw_error;
287 }
288 *out_svc = k_iic_svc_none;
289 if (!iic_facade_periph_acted(mask)) {
290 return k_ra8_ok; /* Idle poll: no RX-full / TX-empty event to service. */
291 }
292 if ((mask & (uint8_t)k_ra8_i3c_peripheral_status_rx_full) != 0U) {
293 uint8_t b = 0U;
294 if (ra8_i3c_peripheral_receive((uint8_t)k_iic_channel, &b, 1U) != k_ra8_ok) {
295 return k_ra8_err_hw_error;
296 }
297 *last_byte = b;
298 *out_svc = k_iic_svc_rx;
299 }
300 if ((mask & (uint8_t)k_ra8_i3c_peripheral_status_tx_empty) != 0U) {
301 if (ra8_i3c_peripheral_send((uint8_t)k_iic_channel, last_byte, 1U) != k_ra8_ok) {
302 return k_ra8_err_hw_error;
303 }
304 *out_svc = k_iic_svc_tx;
305 }
306 return k_ra8_ok;
307}

327[[nodiscard]] static iic_facade_ctrl_result_t iic_facade_controller_phase(uint8_t* out_id0)
328{
329 const uint8_t ptr[k_iic_ptr_len] = {(uint8_t)k_iic_prod_ptr_hi, (uint8_t)k_iic_prod_ptr_lo};
330 uint8_t id[k_iic_prod_len] = {};
331 const ra8_err_t err = ra8_io_i2c_bus_transfer(&s_iic_bus,
332 (uint8_t)k_iic_ctrl_addr_7b,
333 ptr,
334 (uint32_t)k_iic_ptr_len,
335 id,
336 (uint32_t)k_iic_prod_len);
337 const iic_facade_ctrl_result_t result = iic_facade_classify_ctrl(err);
338 *out_id0 = (result == k_iic_ctrl_ok) ? id[0] : 0U;
339 return result;
340}

359[[nodiscard]] static uint32_t iic_facade_peripheral_phase(void)
360{
361 const ra8_i3c_peripheral_cfg_t cfg = {
362 .peripheral_addr_7b = (uint8_t)k_iic_periph_addr_7b,
363 .general_call = 0U,
364 };
365 if (ra8_i3c_peripheral_open((uint8_t)k_iic_channel, &cfg) != k_ra8_ok) {
366 return 0U;
367 }
368 uint8_t last = 0U;
369 bool have_rx = false;
370 uint32_t rounds = 0U;
371 for (uint32_t i = 0U;
372 (i < (uint32_t)k_iic_periph_poll_max) && (rounds < (uint32_t)k_iic_periph_rounds);
373 i++) {
374 iic_facade_svc_t svc = k_iic_svc_none;
375 if (iic_facade_service(&last, &svc) != k_ra8_ok) {
376 break;
377 }
378 if (svc == k_iic_svc_rx) {
379 have_rx = true;
380 } else if ((svc == k_iic_svc_tx) && have_rx) {
381 rounds++;
382 have_rx = false;
383 } else {
384 /* Idle poll or an echo without a preceding receive: not a round-trip. */
385 }
386 }
387 return rounds;
388}

449int32_t main(void)
450{
451 uint32_t pclka_hz = 0U;
452 iic_setup_or_halt(&pclka_hz);
453 ra8_isr_globals_enable();
454 iic_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
455
456 /* App-owned bus bring-up: IIC_B in I2C-compat controller mode, bound through
457 * the ra8_io facade. A future board revision that routes this bus to a RIIC
458 * channel only swaps the bind call for ra8_io_i2c_bus_bind_riic. */
459 const ra8_i3c_cfg_t iic_cfg = {
460 .mode = k_ra8_i3c_mode_i2c,
461 .bus_hz = (uint32_t)k_iic_bus_hz,
462 .pclka_hz = pclka_hz,
463 };
464 if (ra8_i3c_init((uint8_t)k_iic_channel, &iic_cfg) != k_ra8_ok) {
465 iic_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
466 }
467 if (ra8_io_i2c_bus_bind_i3c_compat(&s_iic_bus, (uint8_t)k_iic_channel) != k_ra8_ok) {
468 iic_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
469 }
470
471 /* (1) Controller: facade write-pointer / repeated-START / read of the GT911
472 * PRODUCT_ID. (2) Peripheral: reopen the channel as target 0x42 and echo
473 * controller round-trips. Both complete once, before the banner loop. */
474 uint8_t id0 = 0U;
475 const iic_facade_ctrl_result_t ctrl = iic_facade_controller_phase(&id0);
476 const uint32_t periph = iic_facade_peripheral_phase();
477
478 while (1) {
479 iic_print_banner(ctrl, id0, periph);
480 (void)ra8_board_led_toggle(k_ra8_board_led1);
481 ra8_delay_ms((uint32_t)k_iic_period_ms);
482 }
483 return 0;
484}
