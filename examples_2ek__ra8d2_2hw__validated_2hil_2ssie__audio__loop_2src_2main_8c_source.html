165[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_ssie_loop_run(void)
166{
167 const ra8_ssie_cfg_t cfg = {
168 .role = k_ra8_ssie_role_controller,
169 .format = k_ra8_ssie_format_i2s,
170 .data_word = k_ra8_ssie_dwl_16,
171 .system_word = k_ra8_ssie_swl_32,
172 .bclk_div = k_ra8_ssie_bclk_div_32,
173 .use_gpt_clk = false,
174 .long_frame = true,
175 .bckp_rising = false,
176 .lrckp_low = false,
177 .spdp_high = false,
178 .byte_swap = false,
179 .lr_continue = false,
180 .bck_idle_stop = false,
181 .enable_aucke = true,
182 .tx_threshold = (uint8_t)k_ssie_loop_tx_threshold,
183 .rx_threshold = (uint8_t)k_ssie_loop_rx_threshold,
184 };
185 ra8_err_t err = ra8_ssie_init((uint8_t)k_ssie_loop_channel, &cfg);
186 if (err != k_ra8_ok) {
187 return err;
188 }
189 err = ra8_ssie_start((uint8_t)k_ssie_loop_channel, k_ra8_ssie_dir_tx_rx);
190 if (err != k_ra8_ok) {
191 return err;
192 }
193 for (uint8_t i = 0U; i < (uint8_t)k_ssie_loop_sample_count; i++) {
194 err = ra8_ssie_write_sample((uint8_t)k_ssie_loop_channel, s_ssie_loop_pattern[i]);
195 if (err != k_ra8_ok) {
196 return err;
197 }
198 }
199 return k_ra8_ok;
200}
