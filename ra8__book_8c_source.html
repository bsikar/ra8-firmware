138static uint32_t ra8_book_crc32(const uint8_t* data, size_t len)
139{
140 uint32_t crc = k_ra8_book_crc_init;
141 for (size_t i = 0U; i < len; ++i) {
142 const uint8_t idx = (uint8_t)((crc ^ (uint32_t)data[i]) & (uint32_t)k_ra8_book_crc_byte_mask);
143 crc = s_ra8_book_crc_table[idx] ^ (crc >> (uint32_t)k_ra8_book_crc_byte_bits);
144 }
145 return crc ^ k_ra8_book_crc_init;
146}

259ra8_err_t ra8_book_validate(const void* base, size_t size)
260{
261 RA8_CHECK_NULL_PTR(base, s_tag_book, "validate: null base");
262
263 if (size < sizeof(ra8_book_header_t)) {
264 return k_ra8_err_invalid_size;
265 }
266
267 const ra8_book_header_t* hdr = (const ra8_book_header_t*)base;
268
269 if (!ra8_book_magic_ok(hdr)) {
270 return k_ra8_err_invalid_arg;
271 }
272 if (hdr->format_version != k_ra8_book_format_version) {
273 return k_ra8_err_invalid_arg;
274 }
275 /* Unknown feature bits mean the blob relies on a presentation semantic this
276 * firmware does not implement (e.g. a reading-order mode newer than
277 * ra8_book_flag_t) -- refuse it rather than silently mis-render it. */
278 if ((hdr->flags & ~(uint32_t)k_ra8_book_flag_mask_known) != 0U) {
279 return k_ra8_err_invalid_arg;
280 }
281
282 uint32_t total = hdr->total_size;
283 if ((total < sizeof(ra8_book_header_t)) || ((size_t)total > size)) {
284 return k_ra8_err_invalid_size;
285 }
286
287 const bool tables_ok =
288 ra8_book_table_fits(hdr->chapter_off, hdr->chapter_count, sizeof(ra8_book_chapter_t), total) &&
289 ra8_book_table_fits(hdr->node_off, hdr->node_count, sizeof(ra8_book_node_t), total) &&
290 ra8_book_table_fits(hdr->attr_off, hdr->attr_count, sizeof(ra8_book_attr_t), total) &&
291 ra8_book_table_fits(hdr->stylesheet_off,
292 hdr->stylesheet_count,
293 sizeof(ra8_book_stylesheet_t),
294 total) &&
295 ra8_book_table_fits(hdr->image_off, hdr->image_count, sizeof(ra8_book_image_t), total) &&
296 ra8_book_table_fits(hdr->string_off, hdr->string_size, 1U, total) &&
297 ra8_book_table_fits(hdr->image_pool_off, hdr->image_pool_size, 1U, total);
298 if (!tables_ok) {
299 return k_ra8_err_invalid_size;
300 }
301
302 /* Every image descriptor must name a pixel depth this build can unpack; an
303 * unknown depth is refused (fail-closed) rather than fed to the wrong blit. */
304 if (!ra8_book_image_pixfmts_known(base, hdr)) {
305 return k_ra8_err_invalid_arg;
306 }
307
308 const uint8_t* body = (const uint8_t*)base + sizeof(ra8_book_header_t);
309 uint32_t body_len = total - (uint32_t)sizeof(ra8_book_header_t);
310 if (ra8_book_crc32(body, body_len) != hdr->crc32) {
311 return k_ra8_err_range_check_failed;
312 }
313
314 return k_ra8_ok;
315}

318ra8_err_t ra8_book_container_header_fields(const uint8_t* hdr,
319 uint32_t* out_chunk_bytes,
320 uint64_t* out_total,
321 uint32_t* out_count)
322{
323 RA8_CHECK_NULL_PTR(hdr, s_tag_book, "hdr fields: null hdr");
324 RA8_CHECK_NULL_PTR(out_chunk_bytes, s_tag_book, "hdr fields: null out");
325 const char cmagic[k_ra8_book_container_magic_len] = {'R', 'B', 'K', 'C'};
326 for (uint8_t i = 0U; i < k_ra8_book_container_magic_len; ++i) {
327 if ((char)hdr[i] != cmagic[i]) {
328 return k_ra8_err_invalid_arg;
329 }
330 }
331 uint32_t chunk_bytes = 0U;
332 uint64_t total = 0U;
333 uint32_t count = 0U;
334 uint32_t reserved = 0U;
335 memcpy(&chunk_bytes, &hdr[k_ra8_book_cont_off_chunk_bytes], sizeof(chunk_bytes));
336 memcpy(&total, &hdr[k_ra8_book_cont_off_total], sizeof(total));
337 memcpy(&count, &hdr[k_ra8_book_cont_off_count], sizeof(count));
338 memcpy(&reserved, &hdr[k_ra8_book_cont_off_reserved], sizeof(reserved));
339 if ((chunk_bytes == 0U) || (total == 0U) || (reserved != 0U)) {
340 return k_ra8_err_invalid_arg;
341 }
342 const uint64_t want_count = (total + (uint64_t)chunk_bytes - 1U) / (uint64_t)chunk_bytes;
343 if ((uint64_t)count != want_count) {
344 return k_ra8_err_invalid_arg;
345 }
346 *out_chunk_bytes = chunk_bytes;
347 *out_total = total;
348 *out_count = count;
349 return k_ra8_ok;
350}

411RA8_INTERNAL
412static ra8_err_t s_container_view(const uint8_t* bytes,
413 size_t file_len,
414 size_t scratch_cap,
415 ra8_book_container_view_t* out_view)
416{
417 if (file_len < k_ra8_book_container_header_len) {
418 return k_ra8_err_invalid_size;
419 }
420 ra8_err_t err = ra8_book_container_header_fields(bytes,
421 &out_view->chunk_bytes,
422 &out_view->total,
423 &out_view->chunk_count);
424 if (err != k_ra8_ok) {
425 return err;
426 }
427 const uint64_t entries = (uint64_t)out_view->chunk_count + 1U;
428 const uint64_t table_bytes = entries * k_ra8_book_container_entry_len;
429 if ((uint64_t)file_len < ((uint64_t)k_ra8_book_container_header_len + table_bytes)) {
430 return k_ra8_err_invalid_size;
431 }
432 if ((uint64_t)scratch_cap < out_view->total) {
433 return k_ra8_err_invalid_size;
434 }
435 out_view->table = &bytes[k_ra8_book_container_header_len];
436 out_view->payload = &out_view->table[table_bytes];
437 out_view->payload_len = (uint64_t)file_len - k_ra8_book_container_header_len - table_bytes;
438 uint64_t prev = ra8_book_container_table_entry(out_view->table, 0U);
439 if (prev != 0U) {
440 return k_ra8_err_invalid_arg;
441 }
442 for (uint32_t i = 1U; i <= out_view->chunk_count; ++i) { /* bound: validated chunk_count */
443 const uint64_t cur = ra8_book_container_table_entry(out_view->table, i);
444 if (cur <= prev) {
445 return k_ra8_err_invalid_arg;
446 }
447 prev = cur;
448 }
449 if (prev != out_view->payload_len) {
450 return k_ra8_err_invalid_arg;
451 }
452 return k_ra8_ok;
453}

484RA8_INTERNAL
485static ra8_err_t s_inflate_chunks(const ra8_book_container_view_t* view,
486 ra8_book_inflate_fn inflate,
487 uint8_t* scratch)
488{
489 for (uint32_t i = 0U; i < view->chunk_count; ++i) { /* bound: validated chunk_count */
490 const uint64_t off = ra8_book_container_table_entry(view->table, i);
491 const uint64_t next = ra8_book_container_table_entry(view->table, i + 1U);
492 const uint64_t dst_off = (uint64_t)i * view->chunk_bytes;
493 uint64_t expected = view->total - dst_off;
494 if (expected > (uint64_t)view->chunk_bytes) {
495 expected = view->chunk_bytes;
496 }
497 size_t produced = 0U;
498 const ra8_err_t err = inflate(&view->payload[(size_t)off],
499 (size_t)(next - off),
500 &scratch[(size_t)dst_off],
501 (size_t)expected,
502 &produced);
503 if (err != k_ra8_ok) {
504 return err;
505 }
506 if (produced != (size_t)expected) {
507 return k_ra8_err_invalid_size;
508 }
509 }
510 return k_ra8_ok;
511}
