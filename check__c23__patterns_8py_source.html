4"""check_c23_patterns.py -- enforce four C23 source patterns on first-party code.
5
6These four rules previously lived ONLY as inline ``grep`` loops inside the
7local ``scripts/git/pre-commit`` hook and were never run by the CI gate
8``pre-commit-checks`` (``gate_pre_commit_checks`` in
9``scripts/ci/gates/checks.sh``). The workflow claimed the gate mirrored the
10hook, but these four checks were absent from it -- a "local green, CI red"
11drift in the other direction, where a violation the hook rejects sails through
12CI on a machine whose hook is not installed. This checker is the single
13first-party implementation both the hook and the gate now call, so there is
14exactly one definition of each rule.
15
16The rules (CLAUDE.md "C23 Syntax" and "Constants and Macros"):
17
18 1. ``_Static_assert(`` -- C11 spelling; C23 provides ``static_assert``.
19 2. ``= {0}`` -- C11 zero-initializer; C23 uses ``= {}``.
20 3. ``#include <stdbool.h>`` -- unnecessary; ``bool`` is a C23 keyword.
21 4. Object-like ``#define NAME <bare-numeric-literal>`` -- the value must be
22 paren-wrapped (``#define NAME (1000)``) so it stays a single token in
23 every expression context. Function-like macros, bare feature flags, and
24 already-parenthesised values are ignored; hex / binary / float / suffix
25 literal forms are all recognised.
26
27Scope:
28 C / C++ sources (.c .h .cpp .hpp) under libs/, src/, examples/, port/,
29 tools/, tests/. Vendored SOUP (libs/third_party/) and generated font data
30 (libs/ra8_fonts/) are exempt, as are build trees.
31
32Matches inside comments and string / character literals are ignored: the
33comment/string blanking is reused verbatim from ``check_magic_numbers.py`` so
34the two gates agree byte-for-byte on what "code" is.
35
36Usage:
37 python3 scripts/checks/check_c23_patterns.py FILE [FILE ...]
38 python3 scripts/checks/check_c23_patterns.py # staged files
39 python3 scripts/checks/check_c23_patterns.py --all # every tracked file
40 python3 scripts/checks/check_c23_patterns.py --selftest # prove the rules fire
41
42Returns 0 on clean, 1 on findings, 2 on usage / selftest failure.
43"""

59REPO_ROOT = Path(__file__).resolve().parents[2]

61# First-party roots that carry hand-authored C. Mirrors check_no_gnu_attribute.
62ROOTS = ("libs", "src", "examples", "port", "tools", "tests")
63EXTS = (".c", ".h", ".cpp", ".hpp")
64# Path fragments that exclude a file: vendored SOUP and generated font tables.
65EXEMPT_DIRS = ("/third_party/", "/ra8_fonts/")
66
67# Rule 1: C11 _Static_assert at the start of a line (after leading whitespace).
68# C23 spells it `static_assert`.
69_STATIC_ASSERT_RE = re.compile(r"^\s*_Static_assert\s*\‍(")
70
71# Rule 2: the C11 `= {0}` zero-initializer. C23 uses `= {}`.
72_ZERO_INIT_RE = re.compile(r"= \{0\}")
73
74# Rule 3: `#include <stdbool.h>` -- unnecessary, `bool` is a C23 keyword.
75_STDBOOL_RE = re.compile(r"^\s*#\s*include\s+<stdbool\.h>")
76
77# Rule 4: object-like `#define NAME <bare-numeric-literal>` whose value is not
78# paren-wrapped. Faithful translation of the ERE in scripts/git/pre-commit:
79# a bare integer or float literal (with optional U/L/F suffix, and hex / binary
80# / exponent forms), ignoring function-like macros, bare feature flags, and
81# already-parenthesised values. The trailing comment group is retained for
82# fidelity; the blanking below turns any real comment to spaces, which the
83# leading `\s*` absorbs.
84_BARE_DEFINE_RE = re.compile(
85 r"^\s*#\s*define\s+[A-Za-z_][A-Za-z0-9_]*\s+"
86 r"[+-]?(?:"
87 r"0[xX][0-9a-fA-F]+[uUlL]*"
88 r"|0[bB][01]+[uUlL]*"
89 r"|[0-9]+\.[0-9]*(?:[eE][+-]?[0-9]+)?[fFlL]?"
90 r"|\.[0-9]+(?:[eE][+-]?[0-9]+)?[fFlL]?"
91 r"|[0-9]+(?:[eE][+-]?[0-9]+)?[fFlLuU]*"
92 r")\s*(?:/[/*].*)?$"
93)
94
95# Each rule: (id, compiled regex, human-facing message). The id doubles as the
96# selftest key so a rule cannot be added without a fixture proving both
97# directions.
98_RULES: tuple[tuple[str, re.Pattern[str], str], ...] = (
99 ("static_assert", _STATIC_ASSERT_RE, "C11 _Static_assert -- use C23 static_assert"),
100 ("zero_init", _ZERO_INIT_RE, "C11 = {0} initializer -- use C23 = {}"),
101 ("stdbool", _STDBOOL_RE, "unnecessary #include <stdbool.h> -- bool is a C23 keyword"),
102 ("bare_define", _BARE_DEFINE_RE, "bare numeric #define value -- wrap with parens, e.g. (1000)"),
103)

106def find_violations(path: Path) -> list[tuple[int, str, str]]:
107 """Report every C23-pattern violation in one file.
108
109 The scan runs over a comment/string-blanked view of the source so a match
110 inside a comment or a string literal is never reported; line numbers stay
111 valid because the blanking preserves newlines and column positions.
112
113 Args:
114 path: File to scan.
115
116 Returns:
117 A list of ``(line_no, rule_id, snippet)`` tuples, one per finding;
118 an unreadable file yields an empty list rather than raising.
119 """
120 try:
121 text = path.read_text(encoding="utf-8", errors="replace")
122 except OSError:
123 return []
124 code_lines = _strip_comments_and_strings(text).splitlines()
125 orig_lines = text.splitlines()
126 violations: list[tuple[int, str, str]] = []
127 for idx, code in enumerate(code_lines):
128 for rule_id, pattern, _msg in _RULES:
129 if pattern.search(code):
130 snippet = (orig_lines[idx] if idx < len(orig_lines) else code).strip()
131 violations.append((idx + 1, rule_id, snippet))
132 return violations

135def needs_check(path: Path) -> bool:
136 """Whether a path is first-party C/C++ subject to the C23 pattern rules.
137
138 Args:
139 path: Candidate file path.
140
141 Returns:
142 True when the suffix is a C/C++ one and the path is neither a build
143 artifact nor under a vendored / generated tree.
144 """
145 if path.suffix.lower() not in EXTS:
146 return False
147 posix = path.as_posix()
148 if is_build_output_path(posix):
149 return False
150 return not any(frag in f"/{posix}/" for frag in EXEMPT_DIRS)

153def _git_lines(*pathspec: str) -> list[str]:
154 """Return tracked repo-relative paths matching `pathspec`.
155
156 Args:
157 pathspec: git pathspec arguments (e.g. ``"*.c"``).
158
159 Returns:
160 Repo-relative path strings; exits 2 on a git failure.
161 """
162 proc = subprocess.run( # noqa: S603 -- fixed argv, trusted tool path
163 ["git", "ls-files", "-z", "--", *pathspec], # noqa: S607 -- trusted: fixed git argv
164 cwd=REPO_ROOT,
165 capture_output=True,
166 text=True,
167 check=False,
168 )
169 if proc.returncode != 0:
170 sys.stderr.write(proc.stderr)
171 sys.stderr.write(f"git ls-files failed (exit {proc.returncode})\n")
172 sys.exit(2)
173 return [p for p in proc.stdout.split("\0") if p]

176def iter_all_files() -> list[Path]:
177 """Every tracked first-party C/C++ file, for the ``--all`` sweep.
178
179 Returns:
180 Absolute paths under the first-party roots that pass ``needs_check``.
181 """
182 out: list[Path] = []
183 for root in ROOTS:
184 for rel in _git_lines(*(f"{root}/**/*{ext}" for ext in EXTS)):
185 p = REPO_ROOT / rel
186 if needs_check(p):
187 out.append(p)
188 return sorted(set(out))

191def iter_staged_files() -> list[Path]:
192 """Every staged first-party C/C++ file, for the default (hook) mode.
193
194 Returns:
195 Absolute paths of added / copied / modified / renamed staged files
196 that pass ``needs_check`` and still exist on disk.
197 """
198 proc = subprocess.run(
199 ["git", "diff", "--cached", "--name-only", "--diff-filter=ACMR", "-z"], # noqa: S607 -- trusted git
200 cwd=REPO_ROOT,
201 capture_output=True,
202 text=True,
203 check=False,
204 )
205 if proc.returncode != 0:
206 sys.stderr.write(proc.stderr)
207 sys.stderr.write(f"git diff --cached failed (exit {proc.returncode})\n")
208 sys.exit(2)
209 out: list[Path] = []
210 for rel in proc.stdout.split("\0"):
211 if not rel:
212 continue
213 p = REPO_ROOT / rel
214 if needs_check(p) and p.is_file():
215 out.append(p)
216 return out

219# ---------------------------------------------------------------------------
220# Selftest
221#
222# Every rule is asserted in BOTH directions: a bad fixture must fire exactly
223# that rule, and the matching correct-C23 form must stay silent. A comment and
224# a string-literal fixture prove the blanking suppresses matches that are not
225# really code. Fixtures are scanned in-memory via a temp file, so nothing a
226# checker's own tree scan could later trip over is written into the repo.
227# ---------------------------------------------------------------------------
228
229# (rule_id, source, should_fire). ``should_fire`` names the rule the source is
230# expected to trip; the negative cases assert a clean scan of the whole file.
231_SELFTEST_CASES: tuple[tuple[str, str, bool], ...] = (
232 # Rule 1: _Static_assert.
233 ("static_assert", '_Static_assert(sizeof(int) == 4, "width");\n', True),
234 ("static_assert", ' _Static_assert(1, "x");\n', True),
235 ("static_assert", 'static_assert(sizeof(int) == 4, "width");\n', False),
236 # Rule 2: = {0}.
237 ("zero_init", "int table[4] = {0};\n", True),
238 ("zero_init", "int table[4] = {};\n", False),
239 # Rule 3: #include <stdbool.h>.
240 ("stdbool", "#include <stdbool.h>\n", True),
241 ("stdbool", "# include <stdbool.h>\n", True),
242 ("stdbool", "#include <stdint.h>\n", False),
243 # Rule 4: bare numeric #define.
244 ("bare_define", "#define K_TIMEOUT_MS 1000\n", True),
245 ("bare_define", "#define K_MASK 0xFF\n", True),
246 ("bare_define", "#define K_FLAGS 0b1010u\n", True),
247 ("bare_define", "#define K_SCALE 1.5f\n", True),
248 ("bare_define", "#define K_TIMEOUT_MS (1000)\n", False),
249 ("bare_define", "#define K_TIMEOUT_MS 1000 // trailing comment ok\n", True),
250 ("bare_define", "#define RA8_HAS_MVE\n", False),
251 ("bare_define", "#define MAX(a, b) ((a) > (b) ? (a) : (b))\n", False),
252 # Comment / string suppression: no rule may fire on non-code text.
253 ("_clean", "/* _Static_assert(x); int a[1] = {0}; #define K 5 */\n", False),
254 ("_clean", 'const char* s = "= {0}";\n', False),
255 ("_clean", "// #include <stdbool.h>\n", False),
256)

259def selftest(tmp: Path) -> int:
260 """Prove each rule fires on a bad fixture and stays silent on the C23 form.
261
262 Args:
263 tmp: Writable scratch directory for the fixture files.
264
265 Returns:
266 0 when every case matches its expectation, 1 otherwise.
267 """
268 failures: list[str] = []
269 for i, (expect_id, source, should_fire) in enumerate(_SELFTEST_CASES):
270 fixture = tmp / f"case_{i}.c"
271 fixture.write_text(source, encoding="utf-8")
272 fired = {rule_id for _line, rule_id, _snip in find_violations(fixture)}
273 if should_fire:
274 if expect_id not in fired:
275 failures.append(f" case {i}: rule '{expect_id}' did not fire on: {source!r}")
276 elif fired:
277 failures.append(
278 f" case {i}: rules {sorted(fired)} fired but none expected: {source!r}"
279 )
280 if failures:
281 print("check_c23_patterns.py --selftest: FAILED\n", file=sys.stderr)
282 print("\n".join(failures), file=sys.stderr)
283 return 1
284 fires = sum(1 for c in _SELFTEST_CASES if c[2])
285 total = len(_SELFTEST_CASES)
286 print(
287 f"check_c23_patterns.py --selftest: PASS "
288 f"({total} cases: {fires} must fire, {total - fires} must stay silent)"
289 )
290 return 0

293def _resolve_targets(args: argparse.Namespace) -> list[Path]:
294 """Decide which files to scan from the parsed arguments.
295
296 Args:
297 args: Parsed command-line arguments.
298
299 Returns:
300 The list of files to scan, filtered through ``needs_check``.
301 """
302 if args.all:
303 return iter_all_files()
304 if args.files:
305 return [Path(p) for p in args.files if needs_check(Path(p)) and Path(p).is_file()]
306 return iter_staged_files()

309def main(argv: list[str]) -> int:
310 """Scan first-party C/C++ for the four C23 patterns, or run the selftest.
311
312 With no ``--all`` and no explicit files the staged set is scanned, which is
313 how the pre-commit hook stays fast; ``--all`` sweeps every tracked file,
314 which is how the CI gate covers the whole tree.
315
316 Args:
317 argv: Full argument vector (``sys.argv``).
318
319 Returns:
320 0 when clean, 1 on findings or a failing selftest, 2 on a usage error.
321 """
322 parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
323 parser.add_argument("--all", action="store_true", help="scan all tracked C/C++ files")
324 parser.add_argument(
325 "--selftest", action="store_true", help="prove each rule fires and stays silent correctly"
326 )
327 parser.add_argument("files", nargs="*", help="explicit file list (e.g. staged files)")
328 args = parser.parse_args(argv[1:])
329
330 if args.selftest:
331 with tempfile.TemporaryDirectory() as td:
332 return selftest(Path(td))
333
334 targets = _resolve_targets(args)
335
336 messages = {rule_id: msg for rule_id, _pat, msg in _RULES}
337 total = 0
338 for path in targets:
339 rel = path.relative_to(REPO_ROOT) if path.is_relative_to(REPO_ROOT) else path
340 for line_no, rule_id, snippet in find_violations(path):
341 print(f"{rel}:{line_no}: {messages[rule_id]}: {snippet}", file=sys.stderr)
342 total += 1
343
344 if total:
345 print(
346 f"\ncheck_c23_patterns.py: {total} C23-pattern violation(s). "
347 "Use static_assert, `= {}`, drop <stdbool.h>, and wrap bare "
348 "numeric #define values in parens.",
349 file=sys.stderr,
350 )
351 return 1
352 print(f"check_c23_patterns.py: {len(targets)} file(s) scanned, 0 findings.")
353 return 0
