167static uint8_t internal_spbr(uint32_t baud_hz, uint32_t pclka_hz)
168{
169 if ((baud_hz == 0U) || (pclka_hz == 0U)) {
170 return 0U;
171 }
172 const uint32_t divisor = 2U * baud_hz;
173 const uint32_t n = pclka_hz / divisor;
174 if (n == 0U) {
175 return 0U;
176 }
177 const uint32_t result = n - 1U;
178 if (result > (uint32_t)k_ra8_spbr_max) {
179 return k_ra8_spbr_max;
180 }
181 return (uint8_t)result;
182}

206static uint32_t internal_spcmd(const ra8_spi_cfg_t* cfg)
207{
208 uint32_t v = 0U;
209 /* CPHA / CPOL match SPI mode 0..3. */
210 if ((cfg->mode == k_ra8_spi_mode_1) || (cfg->mode == k_ra8_spi_mode_3)) {
211 v |= k_ra8_spcmd_mask_cpha;
212 }
213 if ((cfg->mode == k_ra8_spi_mode_2) || (cfg->mode == k_ra8_spi_mode_3)) {
214 v |= k_ra8_spcmd_mask_cpol;
215 }
216 if (cfg->lsb_first) {
217 v |= k_ra8_spcmd_mask_lsbf;
218 }
219 /* 8-bit frame. */
220 v |= ((uint32_t)k_ra8_spcmd_spb_8bit << k_ra8_spcmd_bit_spb_lo) & k_ra8_spcmd_mask_spb;
221 return v;
222}

279static ra8_err_t internal_wait_spsr(volatile r_spi_regs_t* reg, uint32_t flag_mask)
280{
281 /* Bounded busy-poll of SPSR. On the host test build the ra8_hw_err MMIO fault
282 * seam (ra8_fake_mmio_*) drives this real loop to succeed-after-N or to time out,
283 * so both the success and timeout legs are exercised on host (T1-01) rather
284 * than compiled out behind an RA8_OFF_TARGET short-circuit. On target it is
285 * a plain register spin with a fixed iteration bound. */
286 return ra8_hw_wait_flag_set32(&reg->SPSR, flag_mask, (uint32_t)k_ra8_spi_b_poll_limit);
287}

313static void internal_spi_program_regs(volatile r_spi_regs_t* reg, const ra8_spi_cfg_t* cfg)
314{
315 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
316 reg->SPSRC = k_ra8_spsrc_mask_all;
317
318 /* HUM Ch 43.2.6 "SPCR3 : SPI Control Register 3" p 2891 */
319 const uint8_t spbr = internal_spbr(cfg->baud_hz, cfg->pclka_hz);
320 reg->SPCR3 = ((uint32_t)spbr << k_ra8_spcr3_bit_spbr) & k_ra8_spcr3_mask_spbr;
321
322 /* HUM Ch 43.2.3 "SPDECR : SPI Delay Control Register" p 2883 */
323 reg->SPDECR = 0U;
324
325 /* SPCR2 only honors writes while SPE=0; SPLP2 (bit 17) is the */
326 /* non-inverting loopback (rx = tx). */
327 /* HUM Ch 43.2.4 "SPCR2 : SPI Control Register 2" p 2889 */
328 reg->SPCR2 = (cfg->loopback ? (uint32_t)k_ra8_spcr2_mask_splp2 : 0U);
329
330 /* HUM Ch 43.2.7 "SPCMDm : SPI Command Register" p 2893 */
331 reg->SPCMD[0] = internal_spcmd(cfg);
332
333 /* HUM Ch 43.2.10 "SPDCR : SPI Data Control Register" p 2896 */
334 reg->SPDCR = 0U;
335 reg->SPDCR2 = 0U;
336
337 /* HUM Ch 43.2.14 "SPFCR : SPI FIFO Clear Register" p 2906 */
338 reg->SPFCR = k_ra8_spfcr_mask_spfrst;
339
340 /* SPFRST drains residual FIFO contents through the shifter and */
341 /* can leave SPRF set with a stale 0x00, so the first xfer8 */
342 /* would race; re-clear SPSR right before the SPE assert. */
343 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
344 reg->SPSRC = k_ra8_spsrc_mask_all;
345}

347ra8_err_t ra8_spi_init(uint8_t channel, const ra8_spi_cfg_t* cfg)
348{
349 RA8_CHECK_NULL_PTR(cfg, s_tag, "spi_init: cfg");
350 if (channel >= k_ra8_spi_b_channel_count) {
351 return k_ra8_err_invalid_arg;
352 }
353 volatile r_spi_regs_t* reg = ra8_spi(channel);
354 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
355 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
356 }
357
358 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B" p 444 */
359 const ra8_err_t mst_err = ra8_mstp_enable(s_spi_mstp_table[channel]);
360 RA8_RETURN_ON_ERROR(mst_err, s_tag, "spi_init: mstp"); /* GCOVR_EXCL_BR_LINE */
361
362 /* Disable (SPE=0) before reprogramming. */
363 /* HUM Ch 43.2.4 "SPCR : SPI Control Register" p 2884 */
364 reg->SPCR = 0U;
365
366 internal_spi_program_regs(reg, cfg);
367
368 /* Re-enable with SPE+MSTR set. */
369 /* HUM Ch 43.2.4 "SPCR : SPI Control Register" p 2884 */
370 reg->SPCR = internal_spcr_controller();
371
372 s_spi_state[channel].cb = nullptr;
373 s_spi_state[channel].ctx = nullptr;
374 s_spi_state[channel].initialized = true;
375 ra8_log_info_val(s_tag, "spi_init channel", (uint32_t)channel);
376 return k_ra8_ok;
377}

418ra8_err_t ra8_spi_xfer8(uint8_t channel, uint8_t tx, uint8_t* rx)
419{
420 volatile r_spi_regs_t* reg = ra8_spi(channel);
421 RA8_CHECK_NULL_PTR(reg, s_tag, "channel out of range");
422
423 /* Wait for TX buffer empty. */
424 /* HUM Ch 43.2.9 "SPSR : SPI Status Register" p 2898 */
425 ra8_err_t err = internal_wait_spsr(reg, k_ra8_spsr_mask_sptef);
426 if (err != k_ra8_ok) {
427 return err;
428 }
429
430 /* Push TX byte. */
431 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
432 reg->SPDR = (uint32_t)tx;
433
434 /* Clear TX-empty flag (write-1). */
435 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
436 reg->SPSRC = k_ra8_spsrc_mask_sptefc;
437
438 /* Wait for RX buffer full. */
439 /* HUM Ch 43.2.9 "SPSR : SPI Status Register" p 2898 */
440 err = internal_wait_spsr(reg, k_ra8_spsr_mask_sprf);
441 if (err != k_ra8_ok) {
442 return err;
443 }
444
445 /* Drain RX byte. */
446 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
447 const uint8_t received = (uint8_t)reg->SPDR;
448
449 /* Clear RX-full flag. */
450 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
451 reg->SPSRC = k_ra8_spsrc_mask_sprfc;
452
453 if (rx != nullptr) {
454 *rx = received;
455 }
456 return k_ra8_ok;
457}

549static void internal_push_unit(volatile r_spi_regs_t* reg,
550 const void* tx,
551 uint32_t idx,
552 ra8_spi_bit_width_t bit_width)
553{
554 uint32_t value = 0U;
555 if (tx == nullptr) {
556 if (bit_width == k_ra8_spi_width_32) {
557 value = k_ra8_spi_b_dummy_tx_32;
558 } else if (bit_width == k_ra8_spi_width_16) {
559 value = k_ra8_spi_b_dummy_tx_16;
560 } else {
561 value = k_ra8_spi_b_dummy_tx_8;
562 }
563 } else if (bit_width == k_ra8_spi_width_32) {
564 value = ((const uint32_t*)tx)[idx];
565 } else if (bit_width == k_ra8_spi_width_16) {
566 value = (uint32_t)((const uint16_t*)tx)[idx];
567 } else {
568 value = (uint32_t)((const uint8_t*)tx)[idx];
569 }
570 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
571 reg->SPDR = value;
572}

644static ra8_err_t internal_xfer_common(uint8_t channel,
645 const void* tx,
646 void* rx,
647 uint32_t len,
648 ra8_spi_bit_width_t bit_width)
649{
650 if (channel >= k_ra8_spi_b_channel_count) {
651 return k_ra8_err_invalid_arg;
652 }
653 uint8_t bytes_per_unit = 0U;
654 const ra8_err_t bw_err = internal_unit_bytes(bit_width, &bytes_per_unit);
655 if (bw_err != k_ra8_ok) {
656 return bw_err;
657 }
658 if (len == 0U) {
659 return k_ra8_ok;
660 }
661 // mcdc-deactivated: TU-local helper internal_apply_bit_width null-pair guard; the public-API ra8_spi_b_transfer entry validates that at least one of (tx, rx) is non-NULL before calling this helper, so the AND's two conditions cannot both be true on any reachable path -- defensive depth guard only.
662 if ((tx == nullptr) && (rx == nullptr)) {
663 return k_ra8_err_null_ptr;
664 }
665 volatile r_spi_regs_t* reg = ra8_spi(channel);
666 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
667 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
668 }
669
670 /* Lock SPCMD0.SPB to the requested width before pushing data. */
671 internal_apply_bit_width(reg, bit_width);
672 /* Suppress unused warning when callers never pass a 32-bit frame. */
673 (void)bytes_per_unit;
674
675 for (uint32_t i = 0U; i < len; i++) {
676 /* HUM Ch 43.2.9 "SPSR : SPI Status Register" p 2898 */
677 ra8_err_t err = internal_wait_spsr(reg, k_ra8_spsr_mask_sptef);
678 if (err != k_ra8_ok) {
679 return err;
680 }
681 internal_push_unit(reg, tx, i, bit_width);
682 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
683 reg->SPSRC = k_ra8_spsrc_mask_sptefc;
684
685 err = internal_wait_spsr(reg, k_ra8_spsr_mask_sprf);
686 if (err != k_ra8_ok) {
687 return err;
688 }
689 internal_pop_unit(reg, rx, i, bit_width);
690 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
691 reg->SPSRC = k_ra8_spsrc_mask_sprfc;
692 }
693 return k_ra8_ok;
694}
