84ra8_err_t ra8_scb_read_fault_status(ra8_scb_fault_status_t* out)
85{
87 static const char* const s_tag = "ra8_scb";
88
89 RA8_CHECK_NULL_PTR(out, s_tag, "read_fault_status: out");
90 /* Arm v8-M ARM B3.2 "System Control Block": configurable / hard / debug
91 * fault status, then the MemManage / BusFault / Auxiliary registers. Plain
92 * reads -- no side effect, nothing is cleared. */
93 out->cfsr = *ra8_scb_reg(k_ra8_scb_cfsr);
94 out->hfsr = *ra8_scb_reg(k_ra8_scb_hfsr);
95 out->dfsr = *ra8_scb_reg(k_ra8_scb_dfsr);
96 out->mmfar = *ra8_scb_reg(k_ra8_scb_mmfar);
97 out->bfar = *ra8_scb_reg(k_ra8_scb_bfar);
98 out->afsr = *ra8_scb_reg(k_ra8_scb_afsr);
99 /* Arm v8-M ARM (Security Extension): SFSR / SFAR are banked to the Secure
100 * state. Read from Secure they carry the real cause / address; read from
101 * Non-secure they are RAZ -- never a fault -- so this is safe in either
102 * world without a guard. */
103 out->sfsr = *ra8_scb_reg(k_ra8_scb_sfsr);
104 out->sfar = *ra8_scb_reg(k_ra8_scb_sfar);
105 return k_ra8_ok;
106}

121bool ra8_scb_trace_enabled(void)
122{
123 /* Arm v8-M ARM (Debug) DEMCR.TRCENA (bit 24): trace subsystem power. */
124 const uint32_t demcr = *ra8_scb_reg(k_ra8_scb_demcr);
125 return (demcr & (uint32_t)k_ra8_scb_demcr_trcena) != 0U;
126}

128void ra8_scb_trace_enable(void)
129{
130 /* Arm v8-M ARM (Debug) DEMCR.TRCENA (bit 24): set it, preserving every
131 * other bit -- a read-modify-write, not a blind store. */
132 const uint32_t demcr = *ra8_scb_reg(k_ra8_scb_demcr);
133 *ra8_scb_reg(k_ra8_scb_demcr) = demcr | (uint32_t)k_ra8_scb_demcr_trcena;
134}
