104bool board_periph_reset_take_request(bool* out_watchdog, bool* out_iwdt)
105{
106 if ((out_watchdog == nullptr) || (out_iwdt == nullptr)) {
107 return false;
108 }
109 if (s_reboot_req_cause == (uint32_t)k_reboot_cause_none) {
110 return false;
111 }
112 *out_watchdog = (s_reboot_req_cause == (uint32_t)k_reboot_cause_wdt);
113 *out_iwdt = (s_reboot_req_cause == (uint32_t)k_reboot_cause_iwdt);
114 s_reboot_req_cause = (uint32_t)k_reboot_cause_none;
115 return true;
116}

118void board_periph_reset_set_cause(bool power_on, bool software, bool watchdog, bool iwdt)
119{
120 if (power_on) {
121 /* A power-on / cold reboot: PORF is the cause; leave it set (the reset model
122 * retains it across the warm reboot) and assert nothing else. */
123 s_rstsr0 |= (uint8_t)k_rstsr0_porf;
124 return;
125 }
126 /* Any non-power-on reset clears PORF and latches the specific cause flag(s). */
127 s_rstsr0 &= (uint8_t)~(uint8_t)k_rstsr0_porf;
128 if (software) {
129 s_rstsr1 |= (uint32_t)k_rstsr1_swrf;
130 }
131 if (watchdog) {
132 s_rstsr1 |= (uint32_t)k_rstsr1_wdtrf;
133 }
134 if (iwdt) {
135 s_rstsr1 |= (uint32_t)k_rstsr1_iwdtrf;
136 }
137}

168{
169 (void)uc;

185static void reset0_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
186{
187 (void)uc;
188 (void)size;
189 const uint64_t off = addr - (uint64_t)k_rstsr0_base;
190 if (off == (uint64_t)k_rstsr0_off_r0) {
191 s_rstsr0 &= (uint8_t)value; /* write-0-to-clear */
192 } else if (off == (uint64_t)k_rstsr0_off_r2) {
193 s_rstsr2 = (uint8_t)value; /* RSTSR2.CWSF is write-1 set; shadow it */
194 } else if (off == (uint64_t)k_rstsr0_off_r3) {
195 s_rstsr3 = (uint8_t)value;
196 }
197}

200static void reset_report(void)
201{
202 if ((s_rstsr1 == 0U) && ((s_rstsr0 & (uint8_t)k_rstsr0_porf) == 0U)) {
203 return; /* nothing notable to report */
204 }
205 (void)fprintf(stderr,
206 " RESET-CAUSE : RSTSR0=0x%02X RSTSR1=0x%08X (%s%s%s%s)\n",
207 (unsigned)s_rstsr0,
208 s_rstsr1,
209 ((s_rstsr0 & (uint8_t)k_rstsr0_porf) != 0U) ? "POR " : "",
210 ((s_rstsr1 & (uint32_t)k_rstsr1_swrf) != 0U) ? "SW " : "",
211 ((s_rstsr1 & (uint32_t)k_rstsr1_wdtrf) != 0U) ? "WDT " : "",
212 ((s_rstsr1 & (uint32_t)k_rstsr1_iwdtrf) != 0U) ? "IWDT " : "");
213}
