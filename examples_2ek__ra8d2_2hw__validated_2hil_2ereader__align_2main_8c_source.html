92static void al_print_hex(uint32_t value)
93{
94 uint8_t buf[k_al_hex_nibbles];
95 for (uint32_t i = 0U; i < (uint32_t)k_al_hex_nibbles; i++) {
96 const uint32_t shift = ((uint32_t)k_al_hex_nibbles - 1U - i) * (uint32_t)k_al_nibble_bits;
97 const uint32_t nib = (value >> shift) & (uint32_t)k_al_nibble_mask;
98 buf[i] = (uint8_t)((nib < (uint32_t)k_al_dec_ten) ? ('0' + nib) : ('A' + (nib - k_al_dec_ten)));
99 }
100 al_print(buf, (uint32_t)k_al_hex_nibbles);
101}

104static void al_print_uint(uint32_t value)
105{
106 uint8_t buf[k_al_dec_ten];
107 uint32_t n = 0U;
108 if (value == 0U) {
109 buf[n] = '0';
110 n++;
111 }
112 while ((value > 0U) && (n < (uint32_t)k_al_dec_ten)) {
113 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_al_dec_ten));
114 n++;
115 value /= (uint32_t)k_al_dec_ten;
116 }
117 for (uint32_t i = 0U; i < n; i++) {
118 al_print(&buf[n - 1U - i], 1U);
119 }
120}

123static uint32_t al_geom_hash(void)
124{
125 uint32_t hsh = (uint32_t)k_al_fnv_offset;
126 const size_t nbytes = sizeof(int32_t) * 2U; /* x + y per glyph */
127 for (uint32_t g = 0U; g < s_engine.glyph_count; g++) {
128 const int32_t coords[2] = {s_engine.glyphs[g].x, s_engine.glyphs[g].y};
129 const uint8_t* p = (const uint8_t*)coords;
130 for (size_t i = 0U; i < nbytes; i++) {
131 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_al_fnv_prime;
132 }
133 }
134 return hsh;
135}

167int32_t main(void)
168{
169 al_setup_or_halt();
170 ra8_isr_globals_enable();
171 al_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
172
173 if (ra8_reflow_init((uint16_t)k_al_view_w,
174 (uint16_t)k_al_view_h,
175 k_ahem_ttf,
176 (size_t)k_ahem_ttf_len,
177 (uint16_t)k_al_font_px,
178 (uint32_t)k_al_ink,
179 (uint32_t)k_al_link_col,
180 &s_engine) != k_ra8_ok) {
181 al_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
182 }
183 uint32_t pages = 0U;
184 if (ra8_reflow_layout_chapter(&s_engine,
185 (const uint8_t*)k_al_chapter,
186 (uint32_t)(sizeof(k_al_chapter) - 1U),
187 &pages) != k_ra8_ok) {
188 al_panic_halt(k_msg_lerr, (uint32_t)sizeof(k_msg_lerr) - 1U);
189 }
190
191 al_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
192 al_print_uint(s_engine.glyph_count);
193 al_print(k_msg_geom, (uint32_t)sizeof(k_msg_geom) - 1U);
194 al_print_hex(al_geom_hash());
195 al_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
196
197 while (1) {
198 __asm__ volatile("wfi");
199 }
200}
