153static void bm_print_uint(uint32_t value)
154{
155 uint8_t buf[k_bm_dec_digits];
156 uint32_t n = 0U;
157 if (value == 0U) {
158 buf[n] = '0';
159 n++;
160 }
161 while ((value > 0U) && (n < (uint32_t)k_bm_dec_digits)) {
162 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_bm_dec_ten));
163 n++;
164 value /= (uint32_t)k_bm_dec_ten;
165 }
166 for (uint32_t i = 0U; i < n; i++) {
167 bm_print(&buf[n - 1U - i], 1U);
168 }
169}

172static void bm_print_nag(ra8_batt_nag_t nag, uint8_t soc)
173{
174 if (nag == k_ra8_batt_nag_low) {
175 bm_print(k_msg_nag_low, (uint32_t)sizeof(k_msg_nag_low) - 1U);
176 } else if (nag == k_ra8_batt_nag_critical) {
177 bm_print(k_msg_nag_crit, (uint32_t)sizeof(k_msg_nag_crit) - 1U);

195static void bm_setup_or_halt(uint32_t* out_pclka_hz)
196{
197 uint32_t cpuclk0_hz = 0U;
198 if ((ra8_cgc_init() != k_ra8_ok) || (ra8_mstp_init() != k_ra8_ok)) {
199 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
200 }

202 (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, out_pclka_hz) != k_ra8_ok)) {
203 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
204 }
205 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
206 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
207 }
208 if (bm_pins_init() != k_ra8_ok) {
209 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
210 }
211 if (ra8_board_uart_console_init((uint32_t)k_bm_uart_baud) != k_ra8_ok) {
212 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);

230static void bm_read_or_halt(uint8_t* out_soc, uint8_t* out_chg)
231{
232 uint8_t soc = 0U;
233 if (ra8_smbus_read_byte_data((uint8_t)k_bm_fg_addr, (uint8_t)k_bm_reg_soc, &soc) != k_ra8_ok) {
234 bm_panic_halt(k_msg_nak, (uint32_t)sizeof(k_msg_nak) - 1U);
235 }
236 uint8_t crate_hi = 0U;
237 if (ra8_smbus_read_byte_data((uint8_t)k_bm_fg_addr, (uint8_t)k_bm_reg_crate, &crate_hi) !=
238 k_ra8_ok) {
239 bm_panic_halt(k_msg_nak, (uint32_t)sizeof(k_msg_nak) - 1U);
240 }
241 *out_soc = (soc > (uint8_t)k_bm_soc_max) ? (uint8_t)k_bm_soc_max : soc;
242 *out_chg = ((crate_hi & (uint8_t)k_bm_sign_mask) == 0U) ? 1U : 0U;
243}

247int32_t main(void)
248{
249 uint32_t pclka_hz = 0U;
250 bm_setup_or_halt(&pclka_hz);
251 ra8_isr_globals_enable();
252 bm_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
253
254 /* App-owned bus bring-up: IIC_B in I2C-compat mode, bound through the
255 * ra8_io facade into the SMBus layer's injected seam. A future board
256 * revision that moves the bus onto a RIIC channel only swaps the bind. */
257 const ra8_i3c_cfg_t iic_cfg = {.mode = k_ra8_i3c_mode_i2c,
258 .bus_hz = (uint32_t)k_bm_bus_hz,
259 .pclka_hz = pclka_hz};
260 ra8_i2c_bus_ops_t bus_ops = {};
261 if (ra8_i3c_init((uint8_t)k_bm_iic_chan, &iic_cfg) != k_ra8_ok) {
262 bm_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
263 }
264 if (ra8_io_i2c_bus_bind_i3c_compat(&s_bm_bus, (uint8_t)k_bm_iic_chan) != k_ra8_ok) {
265 bm_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
266 }
267 if (ra8_io_i2c_bus_as_ops(&s_bm_bus, &bus_ops) != k_ra8_ok) {
268 bm_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
269 }
270
271 const ra8_smbus_cfg_t cfg = {.bus = bus_ops, .pec_enabled = false};
272 if (ra8_smbus_init(&cfg) != k_ra8_ok) {
273 bm_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
274 }
275
276 ra8_batt_monitor_t nag_mon;
277 if (ra8_batt_monitor_init(&nag_mon) != k_ra8_ok) {
278 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
279 }
280
281 while (1) {
282 uint8_t soc = 0U;
283 uint8_t chg = 0U;
284 bm_read_or_halt(&soc, &chg);
285 g_bat_soc = (uint32_t)soc;
286 g_bat_chg = (uint32_t)chg;
287
288 bm_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
289 bm_print_uint((uint32_t)soc);
290 bm_print(k_msg_pct, (uint32_t)sizeof(k_msg_pct) - 1U);
291 bm_print((chg != 0U) ? (const uint8_t*)"Y" : (const uint8_t*)"N", 1U);
292 bm_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
293
294 /* Fold the reading into the nag policy: a one-shot LOW / CRITICAL line
295 * prints on the descent into each band (edge-triggered, see ra8_batt). */
296 ra8_batt_nag_t nag = k_ra8_batt_nag_none;
297 if (ra8_batt_update(&nag_mon, soc, (chg != 0U), &nag) == k_ra8_ok) {
298 g_bat_nag = (uint32_t)nag;
299 bm_print_nag(nag, soc);
300 }
301
302 ++g_bat_heartbeat;
303 ra8_delay_ms((uint32_t)k_bm_period_ms);
304 }
305 bm_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
306 return 0;
307}
