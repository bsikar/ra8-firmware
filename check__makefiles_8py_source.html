4r"""Structural and formatting checker for this tree's Makefiles.
5
6WHY NOT checkmake
7=================
8checkmake is the only off-the-shelf Make linter, and it was evaluated first.
9It ships three rules, and none of them survives contact with this tree:
10
11 minphony requires an `all`, `test` and `clean` phony target in EVERY
12 Makefile. That encodes "every Makefile is a project root",
13 which is false for the 224 per-app wrappers (no `test` to
14 run) and for the 7 mk/*.mk include fragments (not
15 independently invocable at all). 248 findings, all noise.
16
17 maxbodylength caps a recipe body at 5 lines. An arbitrary number with no
18 defect behind it; 220 findings.
19
20 phonydeclared the one rule with real intent -- and it is BROKEN for the
21 declaration style this tree uses. checkmake 0.2.2 reads only
22 the FIRST physical line of a continued `.PHONY:`, so every
23 name after a backslash is invisible to it. Reduced case, two
24 files identical but for the line break:
25
26 .PHONY: first \\ .PHONY: first second
27 second
28 first: dep first: dep
29 second: dep second: dep
30
31 the left file reports `Target "second" should be declared
32 PHONY.`; the right reports nothing. mk/quality.mk declares 36
33 phony targets across 6 continued lines, of which checkmake
34 sees 7 -- which is how it reported the already-phony `vela`.
35
36So adopting checkmake would mean either living with a false positive or
37disabling its only applicable rule, at which point the gate enforces nothing
38while looking like it does. The rules below are implemented here instead, with
39a continuation-aware parser, and --selftest asserts both directions.
40
41RULES
42=====
43 MK001 licence header -- SPDX-License-Identifier and a Copyright line.
44 MK002 phony declared -- a target with no `/` or `.` in its name (i.e. one
45 that names no file on disk) must appear in a
46 .PHONY declaration. Continuation-aware.
47 MK003 portable ROOT -- a per-app Makefile derives ROOT from
48 `git rev-parse --show-toplevel`, never from a
49 fixed count of `..` segments. `$(abspath
50 $(CURDIR)/../../..)` is correct only at one
51 directory depth, and this repo promotes apps
52 between tiers with `git mv`, which silently
53 changes that depth and leaves ROOT pointing above
54 the repo.
55 MK004 formatting -- 7-bit ASCII, LF endings, final newline, no
56 trailing whitespace, and recipe lines indented
57 with a tab.
58"""

77def repo_files(root: pathlib.Path) -> list[pathlib.Path]:
78 """Every tracked Makefile and .mk fragment, as absolute paths.
79
80 Derived from ``git ls-files`` rather than a directory walk, so an
81 untracked or ignored Makefile in a build tree is never scanned and a
82 newly-tracked one is picked up with no list to update.
83
84 Matches on the exact basename ``Makefile`` or a ``.mk`` suffix -- notably
85 NOT ``Makefile.am`` or similar, which this tree does not use.
86 """
87 out = subprocess.run( # noqa: S603 # fixed argv, no shell
88 ["git", "-C", str(root), "ls-files"], # noqa: S607 # git from PATH is intended
89 capture_output=True,
90 text=True,
91 check=True,
92 ).stdout.split()
93 keep = [
94 p
95 for p in out
96 if not p.startswith(EXCLUDED_PREFIXES)
97 and (pathlib.Path(p).name == "Makefile" or p.endswith(".mk"))
98 ]
99 return [root / p for p in keep]

102def logical_lines(text: str) -> list[tuple[int, str, bool]]:
103 r"""Join backslash continuations into (lineno, text, is_recipe) triples.
104
105 Continuation joining is the piece checkmake gets wrong, so the selftest
106 pins it hardest. `is_recipe` records whether the FIRST physical line began
107 with a tab -- it has to be captured before the join strips indentation, or
108 a recipe line like `\t@echo "make targets:"` parses as a rule whose target
109 list is `@echo "make` (an earlier revision of this file did exactly that
110 and produced 786 phantom findings).
111 """
112 out: list[tuple[int, str, bool]] = []
113 buf: list[str] = []
114 start = 0
115 recipe = False
116 for i, raw in enumerate(text.splitlines(), 1):
117 if not buf:
118 start = i
119 recipe = raw.startswith("\t")
120 if raw.endswith("\\"):
121 buf.append(raw[:-1])
122 continue
123 buf.append(raw)
124 out.append((start, " ".join(s.strip() for s in buf).strip(), recipe))
125 buf = []
126 if buf:
127 out.append((start, " ".join(s.strip() for s in buf).strip(), recipe))
128 return out

131def phony_names(text: str) -> set[str]:
132 """Every target named by any ``.PHONY`` line, unioned across the file.
133
134 Recipe lines are skipped, so a ``.PHONY`` echoed inside a recipe body is
135 not mistaken for a declaration.
136
137 Unioned rather than last-wins because make itself accumulates ``.PHONY``
138 across every occurrence; treating the last line as authoritative would
139 report already-declared targets as missing.
140 """
141 names: set[str] = set()
142 for _, line, is_recipe in logical_lines(text):
143 if is_recipe:
144 continue
145 m = re.match(r"^\.PHONY\s*:\s*(.*)$", line)
146 if m:
147 names.update(m.group(1).split())
148 return names

151def declared_targets(text: str) -> list[tuple[int, str, str]]:
152 """Every explicit target name, as (lineno, name, recipe_text).
153
154 The recipe comes back with the target because it is the only reliable
155 signal for "this target names a file it builds". A binary has no suffix on
156 Linux, so `cache_bench:` looks exactly like a phony `clean:` by name alone
157 -- but its recipe ends `-o $@`, which says plainly that the rule writes the
158 file the target names.
159 """
160 found: list[tuple[int, str, str]] = []
161 lines = logical_lines(text)
162 for idx, (lineno, line, is_recipe) in enumerate(lines):
163 if is_recipe or not line or line.lstrip().startswith("#"):
164 continue
165 # `target: deps` but not `VAR := value`, `VAR ?= value`, `VAR = value`
166 m = re.match(r"^([^\t:=#]+?)\s*::?(?!=)\s*(.*)$", line)
167 if not m:
168 continue
169 lhs = m.group(1).strip()
170 if not lhs or "=" in lhs:
171 continue
172 recipe_parts: list[str] = []
173 for _, nxt, nxt_recipe in lines[idx + 1 :]:
174 if not nxt_recipe:
175 break
176 recipe_parts.append(nxt)
177 recipe = "\n".join(recipe_parts)
178 for name in lhs.split():
179 if _SPECIAL.match(name) or "%" in name or "$" in name:
180 continue
181 found.append((lineno, name, recipe))
182 return found

185class Finding:
186 """One rule violation, carrying the code the selftest asserts on.
187
188 ``code`` (MK001, MK002, ...) is the stable identity: the selftest asserts
189 on codes rather than on message text, so the wording can be improved
190 without silently disarming the test.

193 def __init__(self, path: pathlib.Path, line: int, code: str, msg: str) -> None:
194 """Record one finding; all four fields are required and none is derived."""
195 self.path, self.line, self.code, self.msg = path, line, code, msg

197 def __str__(self) -> str:
198 """Render as ``path:line: [CODE] message`` -- editor-jumpable."""
199 return f"{self.path}:{self.line}: [{self.code}] {self.msg}"

202def _check_formatting(path: pathlib.Path, raw: bytes) -> tuple[list[Finding], str]:
203 """MK004 -- encoding, line endings, trailing whitespace, final newline."""
204 findings: list[Finding] = []
205 try:
206 text = raw.decode("ascii")
207 except UnicodeDecodeError as exc:
208 findings.append(
209 Finding(
210 path,
211 raw[: exc.start].count(b"\n") + 1,
212 "MK004",
213 f"non-ASCII byte 0x{raw[exc.start]:02x}",
214 )
215 )
216 text = raw.decode("ascii", errors="replace")
217
218 if b"\r\n" in raw:
219 findings.append(Finding(path, 1, "MK004", "CRLF line ending"))
220 if raw and not raw.endswith(b"\n"):
221 findings.append(Finding(path, text.count("\n") + 1, "MK004", "no final newline"))
222 findings.extend(
223 Finding(path, i, "MK004", "trailing whitespace")
224 for i, ln in enumerate(text.splitlines(), 1)
225 if ln != ln.rstrip()
226 )
227 return findings, text

230def _check_licence(path: pathlib.Path, text: str) -> list[Finding]:
231 """MK001 -- SPDX identifier and copyright line in the file head."""
232 head = "\n".join(text.splitlines()[:40])
233 findings: list[Finding] = []
234 if "SPDX-License-Identifier:" not in head:
235 findings.append(
236 Finding(path, 1, "MK001", "no SPDX-License-Identifier in the first 40 lines")
237 )
238 if not re.search(r"Copyright \‍(c\‍) \d{4}", head):
239 findings.append(
240 Finding(path, 1, "MK001", "no 'Copyright (c) <year>' line in the first 40 lines")
241 )
242 return findings

245def _check_phony(path: pathlib.Path, text: str) -> list[Finding]:
246 """MK002 -- a target that names no file must be declared .PHONY."""
247 phony = phony_names(text)
248 findings: list[Finding] = []
249 seen: set[str] = set()
250 for lineno, name, recipe in declared_targets(text):
251 if name in seen:
252 continue
253 seen.add(name)
254 if "/" in name or "." in name:
255 continue # names a real file
256 if "$@" in recipe:
257 continue # the recipe writes the file this target names
258 if name not in phony:
259 findings.append(
260 Finding(path, lineno, "MK002", f"target '{name}' names no file but is not .PHONY")
261 )
262 return findings

265def _check_root_derivation(path: pathlib.Path, text: str) -> list[Finding]:
266 """MK003 -- an app Makefile derives ROOT from git, not a fixed '..' depth."""
267 findings: list[Finding] = []
268 for lineno, line, is_recipe in logical_lines(text):
269 if is_recipe:
270 continue
271 m = re.match(r"^ROOT\s*:?=\s*(.+)$", line)
272 if not m:
273 continue
274 value = m.group(1).strip()
275 if GIT_ROOT_FORM not in value:
276 findings.append(
277 Finding(
278 path,
279 lineno,
280 "MK003",
281 "ROOT is derived from a fixed '..' depth "
282 f'({value}); use $(shell git -C "$(CURDIR)" '
283 "rev-parse --show-toplevel) so moving the app between "
284 "tiers cannot silently break it",
285 )
286 )
287 return findings

290def check_file(path: pathlib.Path, raw: bytes, is_app_makefile: bool) -> list[Finding]:
291 """Every Makefile rule, one function per finding code.
292
293 The rule list is the call sequence below: MK004 formatting, MK001 licence,
294 MK002 .PHONY, and MK003 ROOT derivation (app Makefiles only).
295 """
296 findings, text = _check_formatting(path, raw)
297 findings += _check_licence(path, text)
298 findings += _check_phony(path, text)
299 if is_app_makefile:
300 findings += _check_root_derivation(path, text)
301 return findings

355def selftest() -> int:
356 """Prove the checker still fires on a malformed Makefile and spares a good one.
357
358 Asserted in BOTH directions on purpose: a detector whose patterns stopped
359 matching would report a clean tree forever, and only the must-fire half
360 catches that, while only the must-not-fire half catches a rule that
361 became indiscriminate.
362
363 Returns 0 when both directions hold, 1 otherwise.
364 """
365 rc = 0
366 with tempfile.TemporaryDirectory() as td:
367 bad = pathlib.Path(td) / "Makefile"
368 bad.write_bytes(MALFORMED.encode())
369 got = check_file(bad, bad.read_bytes(), is_app_makefile=True)
370 codes = {f.code for f in got}
371 expected = {"MK001", "MK002", "MK003"}
372 missing = expected - codes
373 if missing:
374 print(f"SELFTEST FAIL: malformed Makefile did not report {sorted(missing)}")
375 for f in got:
376 print(f" got: {f}")
377 rc = 1
378 else:
379 print(f"selftest: malformed Makefile -> {len(got)} findings {sorted(codes)} OK")
380 flagged = {f.msg.split("'")[1] for f in got if f.code == "MK002"}
381 if flagged != {"release", "build"}:
382 print(f"SELFTEST FAIL: MK002 flagged {sorted(flagged)}, want release+build")
383 rc = 1
384 else:
385 print("selftest: MK002 names both undeclared phony targets OK")
386
387 good = pathlib.Path(td) / "Tricky.mk"
388 good.write_bytes(TRICKY.encode())
389 got = check_file(good, good.read_bytes(), is_app_makefile=True)
390 if got:
391 print("SELFTEST FAIL: tricky Makefile should be clean but reported:")
392 for f in got:
393 print(f" {f}")
394 rc = 1
395 else:
396 print("selftest: tricky Makefile -> 0 findings OK")
397
398 # The specific defect that disqualified checkmake: names after a
399 # backslash continuation must be visible to the .PHONY parser.
400 cont = ".PHONY: first \\\n second \\\n third\n"
401 got_names = phony_names(cont)
402 if got_names != {"first", "second", "third"}:
403 print(f"SELFTEST FAIL: continued .PHONY parsed as {sorted(got_names)}")
404 rc = 1
405 else:
406 print("selftest: continued .PHONY yields all 3 names OK")
407
408 if phony_names(".PHONY: only\n") != {"only"}:
409 print("SELFTEST FAIL: single-line .PHONY misparsed")
410 rc = 1
411 else:
412 print("selftest: single-line .PHONY OK")
413
414 return rc

417def main() -> int:
418 """Check every tracked Makefile, or run one of the two introspection modes.
419
420 With no positional paths the scan covers every tracked Makefile; naming
421 paths narrows it, which is how the pre-commit hook passes only what is
422 staged.
423
424 ``--list-files`` prints the scope and exits 0 without checking anything.
425 It exists so check_lint_coverage.py can ASK this gate what it covers
426 instead of restating the answer, which is what keeps the two from
427 disagreeing.
428
429 Returns 0 when every scanned file is clean, 1 on any finding.
430 """
431 ap = argparse.ArgumentParser(description=__doc__)
432 ap.add_argument("--selftest", action="store_true")
433 # Scope introspection for check_lint_coverage.py: print what this gate
434 # would scan, so the coverage gate can ask rather than restate the scope.
435 ap.add_argument("--list-files", action="store_true", help="print the scanned file list")
436 ap.add_argument("paths", nargs="*")
437 args = ap.parse_args()
438
439 if args.selftest:
440 return selftest()
441
442 root = pathlib.Path(
443 subprocess.run(
444 ["git", "rev-parse", "--show-toplevel"], # noqa: S607 # git from PATH is intended
445 capture_output=True,
446 text=True,
447 check=True,
448 ).stdout.strip()
449 )
450 paths = [pathlib.Path(p) for p in args.paths] or repo_files(root)
451
452 if args.list_files:
453 print("\n".join(sorted(str(p.relative_to(root)) for p in paths)))
454 return 0
455 if not paths:
456 print("ERROR: no Makefiles found; refusing to report success.", file=sys.stderr)
457 return 1
458
459 findings: list[Finding] = []
460 for p in paths:
461 rel = p.relative_to(root) if p.is_absolute() else p
462 is_app = str(rel).startswith("examples/") and rel.name == "Makefile"
463 findings.extend(check_file(p, p.read_bytes(), is_app))
464
465 for f in findings:
466 print(f)
467 if findings:
468 print(f"\n{len(findings)} Makefile finding(s) in {len(paths)} file(s)")
469 return 1
470 print(f"Makefiles clean ({len(paths)} files)")
471 return 0
