77static uint8_t internal_i3c_i2c_decode_errors(uint32_t bst)
78{
79 uint8_t mask = k_ra8_i3c_i2c_err_none;
80 if ((bst & k_ra8_i3c_i2c_msk_bst_alf) != 0U) {
81 mask |= k_ra8_i3c_i2c_err_arb_lost;
82 }
83 if ((bst & k_ra8_i3c_i2c_msk_bst_nackdf) != 0U) {
84 mask |= k_ra8_i3c_i2c_err_nack;

87 mask |= k_ra8_i3c_i2c_err_timeout;
88 }

97ra8_err_t internal_i3c_i2c_abort(uint8_t channel)
98{
99 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
100 if (reg == nullptr) {
101 return k_ra8_err_invalid_arg;
102 }
103 /* Mask interrupts before tearing down (mirrors FSP
104 * controller abort-sequence helper).
105 * HUM Ch 40.2.48 "BIE", p 2495 / Ch 40.2.52 "NTIE" p 2504. */
106 reg->BIE = 0U;
107 reg->NTIE = 0U;
108
109 internal_i3c_i2c_stop(reg);
110 internal_i3c_i2c_clear_bst(reg);
111 s_iic_b_state[channel].bus_held = false;
112 return k_ra8_ok;
113}

116 * Bus probe -- single-byte address-only transaction.
117 * =============================================================================
118 */
119
120ra8_err_t internal_i3c_i2c_scan(uint8_t channel, uint8_t target_7b, bool* out_acked)
121{
122 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
123 RA8_CHECK_NULL_PTR(reg, s_tag, "iic_b_scan: channel");
124 RA8_CHECK_NULL_PTR(out_acked, s_tag, "iic_b_scan: out_acked");
125
126 *out_acked = false;
127 internal_i3c_i2c_clear_bst(reg);
128 internal_i3c_i2c_start(reg);
129
130 const uint8_t address_byte = (uint8_t)(((uint32_t)target_7b << k_ra8_i3c_i2c_ctrl_addr_shift) |
131 k_ra8_i3c_i2c_ctrl_addr_rw_write);
132 ra8_err_t err = internal_i3c_i2c_send_address(reg, address_byte);
133 if (err != k_ra8_ok) {
134 internal_i3c_i2c_stop(reg);
135 return err;
136 }
137
138 /* Wait for either TENDF (peripheral ACKed the address) or NACKDF. */
139 err = k_ra8_err_hw_timeout;
140 for (uint32_t i = 0U; i < k_ra8_i3c_i2c_ctrl_poll_limit; i++) { /* GCOVR_EXCL_BR_LINE */
141 const uint32_t bst = reg->BST;
142 if ((bst & (k_ra8_i3c_i2c_msk_bst_tendf | k_ra8_i3c_i2c_msk_bst_nackdf)) !=
143 0U) { /* GCOVR_EXCL_BR_LINE */
144 *out_acked = (bst & k_ra8_i3c_i2c_msk_bst_nackdf) == 0U;
145 err = k_ra8_ok;
146 break;
147 }
148 }
149
150 /* Stop is best-effort -- record the outcome of the probe regardless. */
151 internal_i3c_i2c_stop(reg);
152 internal_i3c_i2c_clear_bst(reg);
153 return err;
154}

172ra8_err_t internal_i3c_i2c_clear_errors(uint8_t channel)
173{
174 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
175 if (reg == nullptr) {
176 return k_ra8_err_invalid_arg;
177 }
178 /* HUM Ch 40.2.46 "BST : Bus Status Register" p 2490 */
179 enum : uint32_t {
180 k_ra8_i3c_i2c_err_clear_mask = k_ra8_i3c_i2c_msk_bst_alf | k_ra8_i3c_i2c_msk_bst_nackdf |
181 k_ra8_i3c_i2c_msk_bst_todf,
182 };
183 reg->BST = reg->BST & ~k_ra8_i3c_i2c_err_clear_mask;

192ra8_err_t internal_i3c_i2c_attach_handler(uint8_t channel, ra8_i3c_i2c_complete_fn_t fn, void* ctx)
193{
194 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
195 if (reg == nullptr) {
196 return k_ra8_err_invalid_arg;
197 }
198 s_iic_b_state[channel].cb = fn;
199 s_iic_b_state[channel].ctx = ctx;
200
201 /* Toggle the interrupt sources used by the polling driver as a
202 * single group when (de)attaching a handler. Per-bit tuning lands
203 * when the first interrupt-mode consumer arrives. */
204 if (fn != nullptr) {
205 /* HUM Ch 40.2.48 "BIE : Bus Interrupt Enable Register" p 2495 */
206 reg->BIE = k_ra8_i3c_i2c_msk_bie_nackdie | k_ra8_i3c_i2c_msk_bie_alie |
207 k_ra8_i3c_i2c_msk_bie_todie | k_ra8_i3c_i2c_msk_bie_tendie;
208 /* HUM Ch 40.2.52 "NTIE : Normal Transfer Interrupt Enable" p 2504 */
209 reg->NTIE = k_ra8_i3c_i2c_msk_ntie_tdbeie0 | k_ra8_i3c_i2c_msk_ntie_rdbfie0;
210 } else {
211 /* HUM Ch 40.2.48 "BIE : Bus Interrupt Enable Register" p 2495 */
212 reg->BIE = 0U;
213 /* HUM Ch 40.2.52 "NTIE : Normal Transfer Interrupt Enable" p 2504 */
214 reg->NTIE = 0U;
215 }
216 return k_ra8_ok;
217}
