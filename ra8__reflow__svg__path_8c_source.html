119priv_path_step(char u, bool rel, const int32_t* args, int32_t na, int32_t* cx, int32_t* cy)
120{
121 if (u == 'h') {
122 *cx = rel ? (*cx + args[0]) : args[0];
123 return;
124 }
125 if (u == 'v') {
126 *cy = rel ? (*cy + args[0]) : args[0];
127 return;

130 const int32_t ey = args[na - 1];

132 *cy = rel ? (*cy + ey) : ey;

165static float priv_bezier1(float tt, float c0, float c1, float c2, float c3)
166{
167 const float mt = 1.0F - tt;
168 return ((mt * mt * mt) * c0) + (s_svg_bez3 * (mt * mt) * tt * c1) +
169 (s_svg_bez3 * mt * (tt * tt) * c2) + ((tt * tt * tt) * c3);
170}

554RA8_INTERNAL
555static int32_t priv_path_curve(const svg_xform_t* t,
556 char u,
557 bool rel,
558 const int32_t* args,
559 path_state_t* st,
560 int32_t* xs,
561 int32_t* ys,
562 int32_t n)
563{
564 const svg_pt_t p0 = {.x = st->cx, .y = st->cy};
565 if ((u == 'q') || (u == 't')) {
566 const bool sm = (u == 't');
567 const svg_pt_t p1 =
568 sm ? priv_smooth_ctrl(p0, st, 'q') : priv_arg_pt(rel, st->cx, st->cy, args[0], args[1]);
569 const int32_t e = sm ? 0 : 2;
570 const svg_pt_t p2 = priv_arg_pt(rel, st->cx, st->cy, args[e], args[e + 1]);
571 return priv_emit_quad(t, p0, p1, p2, st, xs, ys, n);
572 }
573 if ((u == 'c') || (u == 's')) {
574 const bool sm = (u == 's');
575 const svg_pt_t p1 =
576 sm ? priv_smooth_ctrl(p0, st, 'c') : priv_arg_pt(rel, st->cx, st->cy, args[0], args[1]);
577 const int32_t b = sm ? 0 : 2;
578 const svg_pt_t p2 = priv_arg_pt(rel, st->cx, st->cy, args[b], args[b + 1]);
579 const svg_pt_t p3 = priv_arg_pt(rel, st->cx, st->cy, args[b + 2], args[b + 3]);
580 return priv_emit_cubic(t, p0, p1, p2, p3, st, xs, ys, n);
581 }
582 if (u == 'a') {
583 const svg_pt_t p_end = priv_arg_pt(rel, st->cx, st->cy, args[k_svg_arc_ex], args[k_svg_arc_ey]);
584 int32_t m = n;
585 ra8_svgp_flatten_arc(t, p0, args, p_end, xs, ys, &m);
586 st->kind = 0; /* an arc breaks the smooth-reflection chain */
587 st->cx = p_end.x;
588 st->cy = p_end.y;
589 return m;
590 }
591 return -1;
592}

624static char priv_next_cmd(const uint8_t* d, size_t dlen, size_t* i, char* last)
625{
626 while ((*i < dlen) && (ra8_svgp_ws((char)d[*i]) || (d[*i] == ','))) {
627 ++(*i);
628 }
629 if (*i >= dlen) {
630 return 0;
631 }
632 const char c = (char)d[*i];
633 if (((c >= 'A') && (c <= 'Z')) || ((c >= 'a') && (c <= 'z'))) {
634 *last = c;
635 ++(*i);
636 return c;
637 }
638 return *last;
639}

672RA8_INTERNAL
673static int32_t
674priv_parse_path(const uint8_t* d, size_t dlen, const svg_xform_t* t, int32_t* xs, int32_t* ys)
675{
676 int32_t n = 0;
677 path_state_t st = {.cx = 0, .cy = 0, .ctrl = {.x = 0, .y = 0}, .kind = 0};
678 char last = 0;
679 size_t i = 0U;
680 /* Bounded: <= k_svg_poly_max commands. Every non-'z' command emits at least
681 * one vertex (n advances) or hits the vertex cap, and the 'z' arm advances i
682 * on a malformed implicit repeat, so the loop always makes progress and
683 * cannot spin on untrusted input (NASA P10 Rule 2). */
684 while ((i < dlen) && (n < (int32_t)k_svg_poly_max)) {
685 const size_t i_before = i;
686 const char c = priv_next_cmd(d, dlen, &i, &last);
687 const char u = ra8_svgp_lc(c);
688 /*
689 * priv_next_cmd() only ever yields c in {0} U ['A'..'Z'] U ['a'..'z'] (an
690 * ASCII path-command letter or the 0 end-marker). Whenever c >= 'a' holds,
691 * c is a lowercase letter (0x61..0x7A) and therefore always <= 'z'; no
692 * yielded value can make (c >= 'a') true while (c <= 'z') is false. The
693 * (c <= 'z') condition thus cannot be flipped independently.
694 */
695 /* mcdc-deactivated: c is a command letter or 0; (c<='z') is always true once (c>='a') holds. */
696 const bool rel = (c >= 'a') && (c <= 'z');
697 const int32_t na = (c != 0) ? priv_cmd_argc(u) : -1;
698 if (na < 0) {
699 break; /* unknown / no current command */
700 }
701 if (u == 'z') {
702 /* Close: the fill closes implicitly. 'z' takes no args, so an implicit
703 * repeat after a non-command byte (e.g. the digit in d="Z2") leaves the
704 * cursor unmoved and would spin forever -- skip the byte to guarantee
705 * forward progress. */
706 if (i == i_before) {
707 ++i;
708 }
709 continue;
710 }
711 int32_t args[k_svg_path_args] = {};
712 for (int32_t a = 0; a < na; ++a) {
713 args[a] = ra8_svgp_num(d, dlen, &i);
714 }
715 const int32_t cn = priv_path_curve(t, u, rel, args, &st, xs, ys, n);
716 if (cn >= 0) {
717 n = cn; /* C/S/Q/T flattened into segments */
718 } else {
719 priv_path_step(u, rel, args, na, &st.cx, &st.cy);
720 ra8_svgp_map_point(t, st.cx, st.cy, &xs[n], &ys[n]);
721 ++n;
722 st.kind = 0; /* a non-curve breaks the smooth-reflection chain */
723 }
724 if (u == 'm') {
725 last = rel ? 'l' : 'L'; /* implicit coords after M are line-tos */
726 }
727 }
728 return n;
729}
