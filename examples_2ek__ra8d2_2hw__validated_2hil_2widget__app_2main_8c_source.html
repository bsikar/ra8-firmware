126static uint32_t wa_framebuffer_hash(void)
127{
128 const uint8_t* p = (const uint8_t*)s_framebuffer;
129 const size_t n = sizeof(s_framebuffer);
130 uint32_t hsh = (uint32_t)k_wa_fnv_offset;
131 for (size_t i = 0U; i < n; i++) {
132 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_wa_fnv_prime;
133 }
134 return hsh;
135}

138static void wa_print_hex(uint32_t value)
139{
140 uint8_t buf[k_wa_hex_nibbles];
141 for (uint32_t i = 0U; i < (uint32_t)k_wa_hex_nibbles; i++) {
142 const uint32_t shift = ((uint32_t)k_wa_hex_nibbles - 1U - i) * (uint32_t)k_wa_nibble_bits;
143 const uint32_t nib = (value >> shift) & (uint32_t)k_wa_nibble_mask;
144 buf[i] = (uint8_t)((nib < (uint32_t)k_wa_dec_ten) ? ('0' + nib) : ('A' + (nib - k_wa_dec_ten)));
145 }
146 wa_print(buf, (uint32_t)k_wa_hex_nibbles);
147}

253static void wa_verify_or_halt(uint32_t lib_crc, uint32_t rdr_crc)
254{
255 /* The two app compositions must differ (different content widget). */
256 if (lib_crc == rdr_crc) {
257 wa_panic_halt(k_msg_fcrc, (uint32_t)sizeof(k_msg_fcrc) - 1U);
258 }
259 /* The focus lifecycle must have fired exactly once each. */
260 if ((s_library.leaves != 1U) || (s_reader.enters != 1U) || (s_library.enters != 1U)) {
261 wa_panic_halt(k_msg_flife, (uint32_t)sizeof(k_msg_flife) - 1U);
262 }
263 /* Partial-flush: invalidate ONLY the status bar -> damage is just its rect. */
264 (void)ra8_widget_invalidate(&s_reader.widgets[0], k_ra8_widget_refresh_fast);
265 ra8_ui_rect_t dmg = {};
266 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
267 uint16_t ndirty = 0U;
268 (void)ra8_widget_damage(s_reader.widgets, 2U, &dmg, &hint, &ndirty);
269 if ((ndirty != 1U) || (dmg.h != (int32_t)k_wa_statusbar_h) || (dmg.y != 0) ||
270 (dmg.w != (int32_t)k_wa_fb_w) || (hint != k_ra8_widget_refresh_fast)) {
271 wa_panic_halt(k_msg_fdmg, (uint32_t)sizeof(k_msg_fdmg) - 1U);
272 }
273}

287int32_t main(void)
288{
289 wa_setup_or_halt();
290 ra8_isr_globals_enable();
291 wa_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
292
293 if (ra8_gfx_init(s_framebuffer,
294 (uint16_t)k_wa_fb_w,
295 (uint16_t)k_wa_fb_h,
296 k_ra8_gfx_format_rgb565) != k_ra8_ok) {
297 wa_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
298 }
299
300 /* Two apps, each a widget tree. */
301 wa_state_init(&s_library, (uint32_t)k_wa_col_library);
302 wa_state_init(&s_reader, (uint32_t)k_wa_col_reader);
303 ra8_app_t library = {.vt = &k_wa_app_vt,
304 .ctx = &s_library,
305 .id = (uint16_t)k_wa_app_library,
306 .name = "Library",
307 .removable = false};
308 ra8_app_t reader = {.vt = &k_wa_app_vt,
309 .ctx = &s_reader,
310 .id = (uint16_t)k_wa_app_reader,
311 .name = "Reader",
312 .removable = false};
313
314 ra8_app_t* slots[2];
315 ra8_app_registry_t reg = {};
316 if ((ra8_app_registry_init(&reg, slots, 2U) != k_ra8_ok) ||
317 (ra8_app_register(&reg, &library) != k_ra8_ok) ||
318 (ra8_app_register(&reg, &reader) != k_ra8_ok)) {
319 wa_panic_halt(k_msg_freg, (uint32_t)sizeof(k_msg_freg) - 1U);
320 }
321
322 uint32_t lib_crc = 0U;
323 uint32_t rdr_crc = 0U;
324 wa_run_two_apps(&reg, &lib_crc, &rdr_crc);
325 wa_verify_or_halt(lib_crc, rdr_crc);
326
327 wa_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
328 wa_print_hex(lib_crc);
329 wa_print(k_msg_rdr, (uint32_t)sizeof(k_msg_rdr) - 1U);
330 wa_print_hex(rdr_crc);
331 wa_print(k_msg_flush, (uint32_t)sizeof(k_msg_flush) - 1U);
332 wa_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
333
334 while (1) {
335 __asm__ volatile("wfi");
336 }
337}
