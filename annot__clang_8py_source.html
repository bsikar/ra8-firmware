3"""Getting libclang to actually parse this tree, and proving that it did.
4
5Two responsibilities that belong together because one is the evidence for the
6other: build the compile flags that let a translation unit parse completely,
7then measure whether it did.
8
9The measurement is not optional bookkeeping. Every call-graph rule in this
10checker is a claim about a graph libclang built, so an include that stops
11resolving does not produce a loud error -- it removes edges, the rules stop
12firing, and the gate prints *fewer* violations, which reads as an improvement.
13:data:`MIN_CALL_RESOLUTION` and :func:`check_parse_integrity` exist to make
14that failure mode fatal, and they live in the same module as the flags whose
15degradation they detect.
16"""
17

48# None so a missing kind degrades to "no section attribute detected" instead of

54#: libclang diagnostic severity for Error (3) and Fatal (4); anything at or
55#: above this level means the parse did not see the code it was given.
56DIAG_ERROR = 3
57
58#: Minimum fraction of ``CallExpr`` cursors whose callee declaration
59#: libclang must resolve for the call-graph rules to mean anything. The
60#: tree parses at >99.8%; the floor sits far enough below that to absorb
61#: genuinely unresolvable constructs (host-only intrinsics, test hooks
62#: behind build macros) while still catching an include path that has
63#: come apart. When resolution collapses the call-graph rules stop
64#: policing anything and report a clean tree, so this is fatal.
65MIN_CALL_RESOLUTION = 0.98

67#: Headers that only exist after a build step generated them. A clean
68#: checkout legitimately does not have these, so an unresolved include
69#: naming one is not an include-path defect.
70GENERATED_HEADERS = frozenset(
71 {
72 "literata_latin1.h", # libs/ra8_fonts generator output
73 "ra8_npu_model_addk_fake.h", # Vela model-compiler output

77#: Headers a clean checkout legitimately lacks on THIS platform -- distinct
78#: from GENERATED_HEADERS (never produced without a build step, on any host):
79#: these exist only on the OS that ships them. ``sys/personality.h`` is
80#: glibc/Linux-only (the ``personality()`` syscall this checker's own probe

83#: CLAUDE.md already documents that the host test suite does not run natively
84#: on macOS at all (mmap of peripheral RAM below 4 GiB is refused there), so

88NON_LINUX_MISSING_HEADERS = frozenset({"personality.h"}) if sys.platform != "linux" else frozenset()

109 "port/*/inc",

113 # headers the vendored driver includes). Derive those roots rather than
114 # naming them: a port that grows one and is not listed here parses with
115 # an unresolved include, and every declaration behind it drops out of
116 # the call graph.
117 "port/*/inc/*",
118 "port/*/src",
119 # tools/<tool>/{inc,src}: the same public/private split the libraries
120 # use. cache_bench keeps its headers flat beside its sources, and the

123 "tools/*",
124 "tools/*/inc",
125 "tools/*/src",
126 # a tool whose src/ is grouped into domain subdirs (ra8_emulator:
127 # engine/periph/usb/display/io) keeps its *_internal.h beside the .c;
128 # a cross-boundary include from tests/ needs the subdir on the path,
129 # so derive those roots too (the same courtesy libs/*/* already gets).
130 "tools/*/src/*",
131 "tools/*/generated",

135def _vendored_include_roots() -> list[pathlib.Path]:
136 """Return the header roots inside ``libs/third_party/``.
137
138 Vendored trees are excluded from *analysis* but must stay on the
139 *include* path: ``port/`` implements interfaces those headers declare --
140 ``ble_npl_*`` for NimBLE, ``tx_application_define`` for ThreadX,
141 ``_ux_device_class_storage_*`` for USBX. Without their headers those are
142 undeclared external symbols; with them they are what they actually are,
143 implementations of a published contract.
144 """
145 third_party = repo_root() / "libs" / "third_party"
146 if not third_party.is_dir():
147 return []
148 vendored = list(third_party.iterdir())
149 vendored += list(third_party.rglob("inc"))
150 vendored += list(third_party.rglob("include"))
151 # Not every vendored tree uses an inc/ or include/ convention. esp-hosted
152 # puts its public headers directly in host/ and common/**, and its driver
153 # includes them by bare name, so the two conventions above miss them
154 # entirely -- `esp_hosted_os_abstraction.h` went unresolved and the whole
155 # OS-abstraction seam disappeared from the call graph.
156 #
157 # The tempting fix, deriving roots from wherever a `.h` actually sits, was
158 # tried and rejected: it puts ~600 directories on the path, several of them
159 # C++ trees, and a namesake header there then shadows the one a C
160 # translation unit meant -- forty examples/ TUs started failing to resolve
161 # `cstdint`. Include shadowing is exactly what a wider path buys, so the
162 # extra roots are named per tree instead. Add a row here when a vendored
163 # tree lands whose public headers are not under inc/ or include/.
164 for extra in ("esp-hosted/host", "esp-hosted/host/api/priv", "esp-hosted/common"):
165 vendored.append(third_party / extra)
166 vendored += sorted((third_party / "esp-hosted" / "common").glob("*"))
167 vendored += sorted((third_party / "esp-hosted" / "host" / "drivers").rglob("*"))
168 seen: set[pathlib.Path] = set()
169 ordered: list[pathlib.Path] = []
170 for d in vendored:
171 if d in seen or not d.is_dir() or is_build_output(d):
172 continue
173 seen.add(d)
174 ordered.append(d)
175 return ordered

178def _include_args() -> list[str]:
179 """Return -I flags covering every header root the tree can include.
180
181 The include path must be complete, and it must be derived only from
182 the repo layout. It used to list seven hand-picked directories on the
183 theory that unresolved headers still yield a good-enough AST. They do
184 not: a call whose declaration was never seen does not resolve, so
185 ``cursor.referenced`` is None and the call site is dropped. That
186 silently starved the call-graph rules (``ra8_priv``,
187 ``ra8_test_helper``) of exactly the cross-module call sites they
188 exist to police -- the light path resolved 43k of 126k call sites,
189 and which ones resolved varied with ambient state, so the same tree
190 could pass standalone and fail under the pre-commit hook.
191
192 Widening it again is not cosmetic either. The linkage rule reads a
193 function's published interface off the file its prototype lives in,
194 so a public header the parse cannot reach turns every symbol it
195 declares into a phantom violation: ``src/secure_app/inc`` was absent,
196 and the whole key-vault and OTA-commit API looked undeclared.
197 """
198 root = repo_root()
199 roots: list[pathlib.Path] = []
200 for pattern in _INCLUDE_ROOT_PATTERNS:
201 roots.extend(sorted(root.glob(pattern)))
202 # Applications keep their headers beside their sources with no inc/
203 # convention, so derive those roots from where the headers actually
204 # are. A quote-include always searches the including file's own
205 # directory first, so a namesake in a sibling app cannot displace an
206 # app's own header.
207 roots.extend(sorted({h.parent for h in (root / "examples").rglob("*.h")}))
208 roots = [d for d in roots if d.is_dir() and not is_excluded(d)]
209 roots.extend(_vendored_include_roots())
210
211 out: list[str] = []
212 seen: set[str] = set()
213 for d in roots:
214 if str(d) in seen:
215 continue
216 seen.add(str(d))
217 out.append(f"-I{d}")
218 return out

232def _probe_is_clean(extra: list[str]) -> bool:
233 """True when ``#include <stddef.h>`` parses without a fatal error."""
234 # Alongside this checker, wherever it lives -- a hardcoded directory
235 # here becomes a FileNotFoundError the moment the file moves.
236 probe = pathlib.Path(__file__).resolve().parent / ".ra8_annotations_probe.c"
237 try:
238 probe.write_text("#include <stddef.h>\n#include <stdint.h>\nsize_t ra8_probe(void);\n")
239 tu = cindex.Index.create().parse(
240 str(probe), args=["-std=c23", "-x", "c", "-DRA8_HOST_BUILD=1", *extra]
241 )
242 return not [d for d in (tu.diagnostics if tu else []) if d.severity >= DIAG_ERROR]
243 except cindex.TranslationUnitLoadError:
244 return False
245 finally:
246 with contextlib.suppress(OSError):
247 probe.unlink()

250def _homebrew_keg_only_clang_candidates() -> list[str]:
251 """Return full paths to keg-only Homebrew ``clang-<N>`` binaries, if any.
252
253 Homebrew's ``llvm@<N>`` formulas are keg-only (a system default `clang`
254 must stay whatever Apple or the unversioned `llvm` formula provides), so
255 their ``bin/`` is never linked onto PATH -- ``command -v clang-18`` fails
256 even when ``brew install llvm@18`` succeeded and the binary is sitting
257 right there. ``brew --prefix llvm@<N>`` finds it directly, tried at the
258 same versions ``_resource_dir_from_driver`` already tries by bare name
259 (which is enough on Linux, where the devcontainer's clang-18 package
260 installs onto PATH normally). A missing ``brew`` (Linux, or a Mac without
261 it) degrades to an empty list.
262 """
263 out: list[str] = []
264 for major in ("22", "21", "20", "19", "18"):
265 try:
266 res = subprocess.run( # noqa: S603 -- fixed argv, no shell
267 ["brew", "--prefix", f"llvm@{major}"], # noqa: S607 -- brew from PATH is intended
268 capture_output=True,
269 text=True,
270 timeout=20,
271 check=False,
272 )
273 except (OSError, subprocess.SubprocessError):
274 continue
275 if res.returncode != 0 or not res.stdout.strip():
276 continue
277 candidate = pathlib.Path(res.stdout.strip()) / "bin" / f"clang-{major}"
278 if candidate.is_file():
279 out.append(str(candidate))
280 return out

283def _resource_dir_from_driver() -> list[str]:
284 """Ask a real clang binary where its builtin headers live."""
285 candidates = [
286 os.environ.get("RA8_CLANG"),
287 "clang",
288 "clang-22",
289 "clang-21",
290 "clang-20",
291 "clang-19",
292 "clang-18",
293 *_homebrew_keg_only_clang_candidates(),
294 ]
295 for exe in candidates:
296 if not exe:
297 continue
298 try:
299 res = subprocess.run( # noqa: S603 # fixed argv, no shell
300 [exe, "-print-resource-dir"],
301 capture_output=True,
302 text=True,
303 timeout=20,
304 check=False,
305 )
306 except (OSError, subprocess.SubprocessError):
307 continue
308 if res.returncode != 0:
309 continue
310 inc = pathlib.Path(res.stdout.strip()) / "include"
311 if not (inc / "stddef.h").is_file():
312 continue
313 # Only adopt this resource dir if it is actually compatible with the
314 # loaded libclang. A mismatched pair (e.g. clang 22 headers against
315 # older bindings) fails inside stdint.h with "__INT32_C is not
316 # defined", which is worse than not setting it at all.
317 if _probe_is_clean([f"-isystem{inc}"]):
318 return [f"-isystem{inc}"]
319 return []

322def _darwin_sysroot_args() -> list[str]:
323 """Return -isysroot and Homebrew include flags, on macOS only.
324
325 libclang loaded through the Python bindings has no notion of the
326 platform SDK a real ``clang`` driver invocation picks up by default, so
327 on macOS libc headers behind the SDK sysroot (``<stdio.h>``, most of
328 ``<stdint.h>``'s transitive includes, and everything else under
329 ``usr/include``) do not resolve -- unlike the compiler's OWN builtin
330 headers (``stddef.h``, ``stdarg.h``, ...), which ``_builtin_include_args``
331 already supplies from the resource directory and which resolve with or
332 without a sysroot. ``xcrun --show-sdk-path`` reports the active SDK
333 regardless of whether it is a full Xcode install or just the Command
334 Line Tools, so ask it rather than hardcode a path that drifts with
335 every OS/Xcode update. Measured effect on this tree: 86.6% call-site
336 resolution without a sysroot, 99.9%+ with one (#488).
337
338 ``tools/ra8_emulator`` includes ``<unicorn/unicorn.h>``, which on macOS
339 is a Homebrew package living outside any path clang searches by
340 default (unlike Linux, where the devcontainer installs it under
341 ``/usr/local/include``). Add the active Homebrew prefix's ``include/``
342 so that header, and any other Homebrew-installed header, resolves the
343 same way it does for a real local ``clang`` invocation.
344
345 A missing ``xcrun`` / ``brew`` (this checker also runs on Linux, where
346 neither exists) degrades to an empty list rather than raising -- the
347 Linux parse already reaches the floor without either flag.
348 """
349 if sys.platform != "darwin":
350 return []
351 out: list[str] = []
352 try:
353 sdk = subprocess.run(
354 ["xcrun", "--show-sdk-path"], # noqa: S607 -- xcrun from PATH is intended
355 capture_output=True,
356 text=True,
357 timeout=20,
358 check=False,
359 )
360 except (OSError, subprocess.SubprocessError):
361 sdk = None
362 if sdk is not None and sdk.returncode == 0 and sdk.stdout.strip():
363 out.append(f"-isysroot{sdk.stdout.strip()}")
364 try:
365 brew = subprocess.run(
366 ["brew", "--prefix"], # noqa: S607 -- brew from PATH is intended
367 capture_output=True,
368 text=True,
369 timeout=20,
370 check=False,
371 )
372 except (OSError, subprocess.SubprocessError):
373 brew = None
374 if brew is not None and brew.returncode == 0 and brew.stdout.strip():
375 inc = pathlib.Path(brew.stdout.strip()) / "include"
376 if inc.is_dir():
377 out.append(f"-I{inc}")
378 return out

381def _builtin_include_args() -> list[str]:
382 """Return -isystem flags for clang's own builtin headers.
383
384 libclang loaded through the Python bindings does not know where its
385 resource directory lives, so ``stddef.h`` and friends may not resolve.
386 That is not cosmetic: a failed system include aborts the rest of the
387 include chain, later declarations are never seen, and the calls that
388 depend on them silently vanish from the call graph. Ask a real clang
389 binary where its resource dir is and put that on the path.
390 """
391 from_driver = _resource_dir_from_driver()
392 if from_driver:
393 return from_driver
394 # No clang driver on PATH: the pip wheel bundles libclang.so but no
395 # builtin headers, so fall back to whatever a distro LLVM package left
396 # on disk. Newest first, so a box with several LLVMs picks the one
397 # closest to the bindings.
398 for inc in sorted(_resource_dirs_on_disk(), reverse=True):
399 if (inc / "stddef.h").is_file() and _probe_is_clean([f"-isystem{inc}"]):
400 return [f"-isystem{inc}"]
401 return []

404def _crypto_config_args() -> list[str]:
405 """Return the Mbed TLS / TF-PSA config selectors ``cmake/mbedtls.cmake`` sets.
406
407 The vendored crypto headers are a maze of ``#if`` on the project
408 config, and without pointing them at the same config file the build
409 uses they expose a different API surface. ``psa/crypto_extra.h``
410 declares ``mbedtls_psa_external_get_random`` only under
411 ``MBEDTLS_PSA_CRYPTO_EXTERNAL_RNG``, which lives in that config -- so
412 without this the RNG callback the secure-boot app supplies looks like
413 an undeclared external symbol rather than the PSA hook it is.
414 """
415 port_inc = repo_root() / "port" / "mbedtls" / "inc"
416 out: list[str] = []
417 for macro, name in (

419 ("TF_PSA_CRYPTO_CONFIG_FILE", "tf_psa_crypto_config.h"),
420 ):
421 header = port_inc / name
422 if header.is_file():
423 out.append(f'-D{macro}="{header}"')

427def tu_args(path: pathlib.Path) -> list[str]:
428 """Return the compile flags for ``path``.
429
430 ``.cpp`` sources are parsed as C++. Forcing ``-x c`` over them, as
431 this used to, makes every ``<algorithm>`` / ``<cassert>`` include fail
432 and truncates the parse of the XML-shim and reflow-v2 translation
433 units.
434
435 Host tests carry the two macros ``tests/CMakeLists.txt`` compiles them
436 with. Without them the mock TUs parse under a configuration nothing
437 ever builds: the ``ra8_fake_mmio_*`` fault-seam prototypes in
438 ``ra8_hw_err.h`` sit behind ``RA8_OFF_TARGET && UNIT_TEST``, so
439 the definitions in ``tests/mocks/`` looked like undeclared external
440 symbols and every call through the seam went unresolved.
441 """
442 global _INCLUDE_ARGS # noqa: PLW0603 # one-shot memo of a pure repo-layout scan
443 if not _INCLUDE_ARGS:
444 _INCLUDE_ARGS = _builtin_include_args() + _darwin_sysroot_args() + _include_args()
445 # -fsized-deallocation matches GCC, which builds every .cpp here and
446 # turns it on from C++14. Clang leaves it off, so <new> compiles out
447 # the two sized `operator delete` declarations and the replacements in
448 # ra8_epub_cpp_alloc.cpp look like symbols no header declares.
449 lang = (
450 ["-std=c++20", "-x", "c++", "-fsized-deallocation"]
451 if path.suffix == ".cpp"
452 else ["-std=c23", "-x", "c"]
453 )
454 config = ["-DRA8_HOST_BUILD=1", *_crypto_config_args()]
455 if is_test_path(str(path)):
456 config += ["-DRA8_OFF_TARGET", "-DUNIT_TEST"]
457 return [*lang, *config, *_INCLUDE_ARGS]

460def parse_tu(path: pathlib.Path, stats: ParseStats) -> cindex.TranslationUnit | None:
461 """Parse a single TU and record what the parse could not resolve."""
462 index = cindex.Index.create()
463 try:
464 tu = index.parse(
465 str(path),
466 args=tu_args(path),
467 options=cindex.TranslationUnit.PARSE_DETAILED_PROCESSING_RECORD,
468 )
469 except cindex.TranslationUnitLoadError:
470 stats.unparsed.append(str(path))
471 return None
472 if tu is None:
473 stats.unparsed.append(str(path))
474 return None
475 for diag in tu.diagnostics:
476 if diag.severity < DIAG_ERROR:
477 continue
478 match = _MISSING_INCLUDE_RE.search(diag.spelling)
479 if not match:
480 continue
481 header_name = pathlib.PurePath(match.group(1)).name
482 if header_name not in GENERATED_HEADERS and header_name not in NON_LINUX_MISSING_HEADERS:
483 stats.missing_includes.add((str(path), match.group(1)))
484 return tu

487def check_parse_integrity(stats: ParseStats, tu_count: int) -> list[Violation]:
488 """Fail when the parse did not see the code it was handed.
489
490 Every call-graph rule here is a claim about a graph libclang built.
491 When headers stop resolving the graph loses edges, the rules stop
492 firing, and the gate prints a smaller number of violations -- which
493 reads as an improvement. It is the opposite, so a parse that came
494 apart fails the gate on its own terms, before any rule runs.
495 """
496 out: list[Violation] = []
497 out.extend(
498 Violation(
499 "ra8_parse_integrity",
500 src,
501 0,
502 f"include '{header}' does not resolve; every declaration behind it "
503 f"is invisible to the call graph",
504 )
505 for src, header in sorted(stats.missing_includes)
506 )
507 out.extend(
508 Violation("ra8_parse_integrity", src, 0, "translation unit failed to parse")
509 for src in sorted(stats.unparsed)
510 )
511 if not stats.calls_seen:
512 out.append(
513 Violation(
514 "ra8_parse_integrity",
515 str(repo_root()),
516 0,
517 f"no call sites found across {tu_count} translation units",
518 )
519 )
520 return out
521 rate = stats.calls_resolved / stats.calls_seen
522 if rate < MIN_CALL_RESOLUTION:
523 out.append(
524 Violation(
525 "ra8_parse_integrity",
526 str(repo_root()),
527 0,
528 f"only {stats.calls_resolved} of {stats.calls_seen} call sites "
529 f"resolved ({rate:.1%}, floor {MIN_CALL_RESOLUTION:.0%}) -- the "
530 f"call-graph rules are not seeing the tree",
531 )
532 )
533 return out
