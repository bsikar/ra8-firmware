168[[nodiscard]] ra8_err_t ra8_dac_b_write(uint8_t channel, uint16_t value)
169{
170 volatile r_dac_b_regs_t* reg = ra8_dac_b(channel);
171 if (reg == nullptr) {
172 return k_ra8_err_invalid_arg;
173 }
174 const uint16_t clamped = internal_ra8_dac_b_clamp(value);
175 /* HUM Ch 54 "12-Bit D/A Converter (DAC12)" p 3490 -- FSP R_DAC_B_Write
176 * is a single 16-bit store to DADR; DACEN is asserted via
177 * ra8_dac_b_set_output_enable() / ra8_dac_b_init_configured(), NOT here. */
178 reg->DADR = clamped;

209static void internal_apply_cfg(uint8_t channel, const ra8_dac_b_cfg_t* cfg)
210{
211 volatile r_dac_b_regs_t* reg = ra8_dac_b(channel);
212 if (reg == nullptr) {
213 return;
214 }
215 /* HUM Ch 54 "12-Bit D/A Converter (DAC12)" p 3490..3496 -- mirror
216 * FSP R_DAC_B_Open: clear DACR0, set DACR1.DPSEL, DACR2.OFSSEL,
217 * zero DADR, then drive the DAOUTDIS bit per the descriptor. */
218 reg->DACR0 = 0U;
219 reg->DACR1 = ((uint32_t)cfg->data_format) << (uint32_t)k_ra8_dacr1_bit_dpsel;
220 reg->DACR2 = ((uint32_t)cfg->vref) << (uint32_t)k_ra8_dacr2_bit_ofssel;
221 reg->DADR = 0U;
222 /* internal_output_enabled == true clears DAOUTDIS (route to pin);
223 * false sets DAOUTDIS (Hi-Z). FSP encodes the same way: it writes
224 * the boolean directly into the 1-bit DAOUTDIS field. */
225 if (!cfg->internal_output_enabled) {
226 reg->DACR0 = reg->DACR0 | k_ra8_dacr0_mask_daoutdis;
227 }
228}

252ra8_err_t ra8_dac_b_deinit(void)
253{
254 /* HUM Ch 54 "12-Bit D/A Converter (DAC12)" p 3490 -- FSP R_DAC_B_Close
255 * writes DACR0 = DAOUTDIS_Msk to fully disable output, operation,
256 * and coupled operation. We then release MSTP. */
257 volatile r_dac_b_regs_t* reg0 = ra8_dac_b(k_ra8_dac_b_channel_0);
258 volatile r_dac_b_regs_t* reg1 = ra8_dac_b(k_ra8_dac_b_channel_1);
259 if (reg0 != nullptr) {
260 reg0->DACR0 = k_ra8_dacr0_mask_daoutdis;
261 reg0->DADR = 0U;
262 }
263 if (reg1 != nullptr) {
264 reg1->DACR0 = k_ra8_dacr0_mask_daoutdis;
265 reg1->DADR = 0U;
266 }
267 s_dac_b_state.fn = nullptr;
268 s_dac_b_state.ctx = nullptr;
269 (void)ra8_mstp_disable(k_ra8_mstp_dac12_1);
270 return ra8_mstp_disable(k_ra8_mstp_dac12_0);
271}

273ra8_err_t ra8_dac_b_set_vref(ra8_dac_b_vref_t vref)
274{
275 /* HUM Ch 54 "12-Bit D/A Converter (DAC12)" p 3490 -- DACR2.OFSSEL is
276 * a single bit at position 8; preserve all other bits. */
277 volatile r_dac_b_regs_t* reg0 = ra8_dac_b(k_ra8_dac_b_channel_0);
278 volatile r_dac_b_regs_t* reg1 = ra8_dac_b(k_ra8_dac_b_channel_1);
279 const uint32_t shifted = ((uint32_t)vref) << (uint32_t)k_ra8_dacr2_bit_ofssel;
280 reg0->DACR2 = (reg0->DACR2 & ~k_ra8_dacr2_mask_ofssel) | (shifted & k_ra8_dacr2_mask_ofssel);
281 reg1->DACR2 = (reg1->DACR2 & ~k_ra8_dacr2_mask_ofssel) | (shifted & k_ra8_dacr2_mask_ofssel);
282 return k_ra8_ok;
283}

285ra8_err_t ra8_dac_b_set_output_enable(uint8_t channel, bool enable)
286{
287 volatile r_dac_b_regs_t* reg = ra8_dac_b(channel);
288 if (reg == nullptr) {
289 return k_ra8_err_invalid_arg;
290 }
291 /* HUM Ch 54 "12-Bit D/A Converter (DAC12)" p 3490 -- FSP toggles
292 * DACEN (bit 0) only; DAE (batch mode) and DAOUTDIS are managed
293 * separately. */
294 if (enable) {
295 reg->DACR0 = reg->DACR0 | k_ra8_dacr0_mask_dacen;
296 } else {
297 reg->DACR0 = reg->DACR0 & ~k_ra8_dacr0_mask_dacen;
298 }
299 return k_ra8_ok;
300}

302ra8_err_t ra8_dac_b_get_status(uint8_t* out_mask)
303{
304 RA8_CHECK_NULL_PTR(out_mask, s_tag, "out_mask must not be nullptr");
305 /* Build a single-byte composite status flag:
306 * bit 0 -- channel 0 DACEN
307 * bit 1 -- channel 1 DACEN
308 */
309 uint8_t flags = 0U;
310 volatile r_dac_b_regs_t* reg0 = ra8_dac_b(k_ra8_dac_b_channel_0);
311 volatile r_dac_b_regs_t* reg1 = ra8_dac_b(k_ra8_dac_b_channel_1);
312 if ((reg0->DACR0 & k_ra8_dacr0_mask_dacen) != 0U) {
313 flags |= 0x1U;
314 }
315 if ((reg1->DACR0 & k_ra8_dacr0_mask_dacen) != 0U) {
316 flags |= 0x2U;
317 }
318 *out_mask = flags;
319 return k_ra8_ok;
