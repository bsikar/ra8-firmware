182static uint32_t eoh_crc32(const uint8_t* data, size_t len)
183{
184 uint32_t crc = (uint32_t)k_eoh_crc_init;
185 for (size_t i = 0U; i < len; i++) {
186 crc ^= (uint32_t)data[i];
187 for (uint32_t bit = 0U; bit < (uint32_t)k_eoh_crc_bits; bit++) {
188 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
189 crc = (crc >> 1U) ^ ((uint32_t)k_eoh_crc_poly & mask);
190 }
191 }
192 return ~crc;
193}

196static void eoh_print_hex(uint32_t value)
197{
198 uint8_t buf[k_eoh_hex_nibbles];
199 for (uint32_t i = 0U; i < (uint32_t)k_eoh_hex_nibbles; i++) {
200 const uint32_t shift = ((uint32_t)k_eoh_hex_nibbles - 1U - i) * (uint32_t)k_eoh_nibble_bits;
201 const uint32_t nib = (value >> shift) & (uint32_t)k_eoh_nibble_mask;
202 buf[i] =
203 (uint8_t)((nib < (uint32_t)k_eoh_dec_ten) ? ('0' + nib) : ('A' + (nib - k_eoh_dec_ten)));
204 }
205 eoh_print(buf, (uint32_t)k_eoh_hex_nibbles);
206}

209static void eoh_print_uint(uint32_t value)
210{
211 uint8_t buf[k_eoh_dec_ten];
212 uint32_t n = 0U;
213 if (value == 0U) {
214 buf[n] = '0';
215 n++;
216 }
217 while ((value > 0U) && (n < (uint32_t)k_eoh_dec_ten)) {
218 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_eoh_dec_ten));
219 n++;
220 value /= (uint32_t)k_eoh_dec_ten;
221 }
222 for (uint32_t i = 0U; i < n; i++) {
223 eoh_print(&buf[n - 1U - i], 1U);
224 }
225}

354static uint16_t eoh_parse_or_halt(ra8_fs_mount_t* mount, uint32_t* out_crc)
355{
356 if (ra8_epub_open_streamed_fs(mount, k_eoh_book_path, &s_epub_io, &s_book) != k_ra8_ok) {
357 eoh_fail((uint32_t)k_eoh_err_open, k_msg_fopen, (uint32_t)sizeof(k_msg_fopen) - 1U);
358 }
359 uint16_t chapters = 0U;
360 if ((ra8_epub_get_chapter_count(&s_book, &chapters) != k_ra8_ok) ||
361 (chapters != (uint16_t)k_eoh_expect_chap)) {
362 eoh_fail((uint32_t)k_eoh_err_chap, k_msg_fchap, (uint32_t)sizeof(k_msg_fchap) - 1U);
363 }
364 size_t got = 0U;
365 if ((ra8_epub_load_chapter(&s_book, 0U, s_chapter, (size_t)k_eoh_chap_cap, &got) != k_ra8_ok) ||
366 (got == 0U)) {
367 eoh_fail((uint32_t)k_eoh_err_load, k_msg_fload, (uint32_t)sizeof(k_msg_fload) - 1U);
368 }
369 const uint32_t crc = eoh_crc32(s_chapter, got);
370 if (crc != (uint32_t)k_eoh_expect_crc) {
371 eoh_fail((uint32_t)k_eoh_err_crc, k_msg_fcrc, (uint32_t)sizeof(k_msg_fcrc) - 1U);
372 }
373 ra8_epub_metadata_t meta = {};
374 if ((ra8_epub_get_metadata(&s_book, &meta) != k_ra8_ok) || (meta.title == nullptr) ||
375 (meta.title[0] == '\0')) {
376 eoh_fail((uint32_t)k_eoh_err_meta, k_msg_fmeta, (uint32_t)sizeof(k_msg_fmeta) - 1U);
377 }
378 *out_crc = crc;
379 return chapters;
380}

396int32_t main(void)
397{
398 uint32_t pclka_hz = 0U;
399 eoh_setup_or_halt(&pclka_hz);
400 ra8_isr_globals_enable();
401 ra8_log_init();
402 eoh_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
403
404 eoh_init_card_or_halt(&pclka_hz);
405 ra8_fs_mount_t* const mount = eoh_mount_or_halt();
406 eoh_provision_or_halt(mount);
407
408 uint32_t ch0_crc = 0U;
409 const uint16_t chapters = eoh_parse_or_halt(mount, &ch0_crc);
410
411 /* Latch the results for the J-Link / `--dump-sym` memprobe gate. */
412 g_eoh_chapters = (uint32_t)chapters;
413 g_eoh_crc = ch0_crc;
414
415 eoh_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
416 eoh_print_uint((uint32_t)chapters);
417 eoh_print(k_msg_crc, (uint32_t)sizeof(k_msg_crc) - 1U);
418 eoh_print_hex(ch0_crc);
419 eoh_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
420
421 /* Idle with a heartbeat that advances ONLY here -- the failure path parks in
422 * WFI without bumping it -- so a memprobe gate proves the full SD -> ra8_fs ->
423 * ra8_epub open/parse pipeline ran and the chapter-0 CRC matched. */
424 while (1) {
425 ra8_delay_ms(k_eoh_frame_ms);
426 g_eoh_heartbeat++;
427 }
428}
