62RA8_INTERNAL static bool is_reserved_base(const char* name)
63{
64 char base[k_reserved_base_max];
65 base_of(name, base, sizeof(base));
66 for (size_t i = 0U; i < (sizeof(s_reserved_exact) / sizeof(s_reserved_exact[0])); ++i) {
67 if (strcmp(base, s_reserved_exact[i]) == 0) {
68 return true;
69 }
70 }
71 if ((strlen(base) == (size_t)k_reserved_len) && (base[k_reserved_digit_at] >= '1') &&
72 (base[k_reserved_digit_at] <= '9')) {
73 for (size_t i = 0U; i < (sizeof(s_reserved_numbered) / sizeof(s_reserved_numbered[0])); ++i) {
74 if (strncmp(base, s_reserved_numbered[i], strlen(s_reserved_numbered[i])) == 0) {
75 return true;
76 }
77 }
78 }
79 return false;
80}

83RA8_INTERNAL static bool copy_sanitised(const char* raw, char* out, size_t cap, size_t* out_len)
84{
85 bool clean = true;
86 size_t n = 0U;
87 size_t i = 0U;
88 if (raw != nullptr) {
89 while ((raw[i] != '\0') && ((n + 1U) < cap)) {
90 if (is_allowed_char(raw[i])) {
91 out[n] = raw[i];
92 } else {
93 out[n] = '_';
94 clean = false;
95 }
96 ++n;
97 ++i;
98 }
99 if (raw[i] != '\0') {
100 clean = false; /* input did not fit: truncated */
101 }
102 }
103 out[n] = '\0';
104 *out_len = n;
105 return clean;
106}

109RA8_INTERNAL static void prepend_underscore(char* out, size_t cap, size_t len)
110{
111 size_t keep = len;
112 if ((keep + 2U) > cap) {
113 keep = cap - 2U; /* drop the tail that no longer fits after the prefix */
114 }
115 memmove(out + 1, out, keep);
116 out[0] = '_';
117 out[keep + 1] = '\0';
118}

120bool mdl_sanitize_segment(const char* raw, char* out, size_t cap)
121{
122 if ((out == nullptr) || (cap < 2U)) {
123 return false;
124 }
125 size_t len = 0U;
126 bool clean = copy_sanitised(raw, out, cap, &len);
127 if (is_dot_segment(out)) {
128 (void)snprintf(out, cap, "%s", s_fallback_name);
129 return false;
130 }
131 if (is_reserved_base(out)) {
132 prepend_underscore(out, cap, len);
133 return false;
134 }
135 return clean;
136}

138bool mdl_path_contained(const char* parent, const char* candidate)
139{
140 if ((parent == nullptr) || (candidate == nullptr)) {
141 return false;
142 }
143 size_t plen = strlen(parent);
144 while ((plen > 0U) && (parent[plen - 1U] == '/')) {
145 --plen;
146 }
147 if (plen == 0U) {
148 return false;
149 }
150 if (strncmp(candidate, parent, plen) != 0) {
151 return false;
152 }
153 const char sep = candidate[plen];
154 return (sep == '/') || (sep == '\0');
155}

163bool mdl_path_join(const char* parent, const char* seg, char* out, size_t cap)
164{
165 if ((out == nullptr) || (cap == 0U)) {
166 return false;
167 }
168 out[0] = '\0';
169 if ((parent == nullptr) || (seg == nullptr)) {
170 return false;
171 }
172 if (is_dot_segment(seg) || has_separator(seg)) {
173 return false; /* empty, `.`, `..`, or a `/`-bearing/absolute segment */
174 }
175 const size_t plen = strlen(parent);
176 const size_t slen = strlen(seg);
177 const size_t need = plen + 1U + slen + 1U; /* parent + '/' + seg + NUL */
178 if (need > cap) {
179 return false; /* refuse a truncated (thus different) path */
180 }
181 memcpy(out, parent, plen);
182 out[plen] = '/';
183 memcpy(out + plen + 1U, seg, slen);
184 out[plen + 1U + slen] = '\0';
185 return true;
186}

207bool mdl_xml_escape(const char* src, char* out, size_t cap)
208{
209 if ((out == nullptr) || (cap == 0U)) {
210 return false;
211 }
212 out[0] = '\0';
213 if (src == nullptr) {
214 return false;
215 }
216 size_t n = 0U;
217 for (size_t i = 0U; src[i] != '\0'; ++i) {
218 const char* ent = xml_entity(src[i]);
219 const size_t need = (ent != nullptr) ? strlen(ent) : 1U;
220 if ((n + need + 1U) > cap) {
221 out[0] = '\0';
222 return false;
223 }
224 if (ent != nullptr) {
225 memcpy(out + n, ent, need);
226 } else {
227 out[n] = src[i];
228 }
229 n += need;
230 }
231 out[n] = '\0';
232 return true;
233}
