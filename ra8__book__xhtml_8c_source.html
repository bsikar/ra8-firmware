159static bool ra8_book_emit_escaped(char* out, size_t cap, size_t* pos, const char* str, bool in_attr)
160{
161 for (const char* p = str; *p != '\0'; ++p) {
162 bool ok = true;
163 switch (*p) {
164 case '&':
165 ok = ra8_book_emit_cstr(out, cap, pos, "&amp;");
166 break;
167 case '<':
168 ok = ra8_book_emit_cstr(out, cap, pos, "&lt;");
169 break;
170 case '>':
171 ok = ra8_book_emit_cstr(out, cap, pos, "&gt;");
172 break;
173 case '"':
174 ok = in_attr ? ra8_book_emit_cstr(out, cap, pos, "&quot;")
175 : ra8_book_emit(out, cap, pos, p, 1U);
176 break;
177 default:
178 ok = ra8_book_emit(out, cap, pos, p, 1U);
179 break;
180 }
181 if (!ok) {
182 return false;
183 }
184 }
185 return true;
186}

338static bool ra8_book_open_element(const void* base,
339 const ra8_book_node_t* node,
340 char* out,
341 size_t cap,
342 size_t* pos,
343 ra8_book_walk_entry_t* stack,
344 uint32_t* sp)
345{
346 const char* name = ra8_book_string(base, node->name_off);
347 if (!ra8_book_emit(out, cap, pos, "<", 1U) || !ra8_book_emit_cstr(out, cap, pos, name) ||
348 !ra8_book_emit_attrs(base, node, out, cap, pos)) {

352 return ra8_book_emit(out, cap, pos, "/>", 2U);

355 return false;
356 }
357 stack[(*sp)++] = (ra8_book_walk_entry_t){true, node->name_off}; /* close after children */
358 stack[(*sp)++] = (ra8_book_walk_entry_t){false, node->first_child}; /* children first */
359 return true;

398RA8_INTERNAL
399static bool ra8_book_walk_to_xhtml(const void* base,
400 uint32_t root,
401 uint32_t node_count,
402 char* out,
403 size_t cap,
404 size_t* pos)
405{
406 const ra8_book_node_t* nodes = ra8_book_nodes(base);
407 /* Explicit DFS stack (~4 KiB) kept in module-static storage so this frame
408 * stays within the stack-usage budget; the walk is iterative (no recursion)
409 * and single-threaded, so the shared buffer never overlaps. */
410 static ra8_book_walk_entry_t s_xhtml_stack[k_ra8_book_xhtml_stack];
411 ra8_book_walk_entry_t* stack = s_xhtml_stack;
412 uint32_t sp = 0U;
413 bool ok = true;
414 stack[sp++] = (ra8_book_walk_entry_t){false, root};
415
416 const uint32_t max_iter = (node_count * k_ra8_book_xhtml_iter_x) + k_ra8_book_xhtml_stack;
417 uint32_t guard = 0U;
418 while (sp > 0U && ok && guard < max_iter) {
419 ++guard;
420 ra8_book_walk_entry_t e = stack[--sp];
421 if (e.is_close) {
422 ok = ra8_book_emit(out, cap, pos, "</", 2U) &&
423 ra8_book_emit_cstr(out, cap, pos, ra8_book_string(base, e.value)) &&
424 ra8_book_emit(out, cap, pos, ">", 1U);
425 continue;
426 }
427 if (e.value == k_ra8_book_nil) {
428 continue;
429 }
430 const ra8_book_node_t* node = &nodes[e.value];
431 if (sp >= k_ra8_book_xhtml_stack) {
432 return false;
433 }
434 /* Continue this level's sibling chain after the whole subtree + close. */
435 stack[sp++] = (ra8_book_walk_entry_t){false, node->next_sibling};
436 if (node->kind == (uint8_t)k_ra8_book_node_text) {
437 ok = ra8_book_emit_escaped(out, cap, pos, ra8_book_string(base, node->text_off), false);
438 } else {
439 ok = ra8_book_open_element(base, node, out, cap, pos, stack, &sp);
440 }
441 }
442 return ok && (guard < max_iter);
443}

528bool ra8_book_emit_text(char* out, size_t cap, size_t* pos, const char* str, bool* at_break)
529{
530 for (const char* p = str; *p != '\0'; ++p) {
531 const char c = *p;
532 const bool ws = (c == ' ') || (c == '\t') || (c == '\n') || (c == '\r');
533 if (!ws) {
534 if (!ra8_book_emit(out, cap, pos, &c, 1U)) {
535 return false;
536 }
537 *at_break = false;
538 continue;
539 }
540 if (*at_break) {
541 continue;
542 }
543 if (!ra8_book_emit(out, cap, pos, " ", 1U)) {
544 return false;
545 }
546 *at_break = true;
547 }
548 return true;
549}

585bool ra8_book_emit_break(char* out, size_t cap, size_t* pos, bool* at_break)
586{
587 while ((*pos > 0U) && (out[*pos - 1U] == ' ')) {
588 (*pos)--;
589 }
590 *at_break = true;
591 if ((*pos > 0U) && (out[*pos - 1U] == '\n')) {
592 return true;
593 }
594 return ra8_book_emit(out, cap, pos, "\n", 1U);
595}

635RA8_INTERNAL
636static bool ra8_book_walk_text(const void* base,
637 uint32_t root,
638 uint32_t node_count,
639 char* out,
640 size_t cap,
641 size_t* pos)
642{
643 const ra8_book_node_t* nodes = ra8_book_nodes(base);
644 /* Explicit DFS stack (2 KiB) kept in module-static storage so this frame
645 * stays within the stack-usage budget; iterative (no recursion) and
646 * single-threaded, so the shared buffer never overlaps. */
647 static uint32_t s_text_stack[k_ra8_book_xhtml_stack];
648 uint32_t* stack = s_text_stack;
649 uint32_t sp = 0U;
650 bool ok = true;
651 bool at_break = true;
652 stack[sp++] = root;
653
654 const uint32_t max_iter = (node_count * k_ra8_book_xhtml_iter_x) + k_ra8_book_xhtml_stack;
655 uint32_t guard = 0U;
656 while (sp > 0U && ok && guard < max_iter) {
657 ++guard;
658 const uint32_t n = stack[--sp];
659 if (n == k_ra8_book_nil) {
660 continue;
661 }
662 const ra8_book_node_t* node = &nodes[n];
663 if (sp >= k_ra8_book_xhtml_stack) {
664 return false;
665 }
666 stack[sp++] = node->next_sibling; /* sibling chain after this subtree */
667 if (node->kind == (uint8_t)k_ra8_book_node_text) {
668 ok = ra8_book_emit_text(out, cap, pos, ra8_book_string(base, node->text_off), &at_break);
669 continue;
670 }
671 if (ra8_book_is_block(ra8_book_string(base, node->name_off))) {
672 ok = ra8_book_emit_break(out, cap, pos, &at_break);
673 }
674 if (ok && (sp < k_ra8_book_xhtml_stack)) {
675 stack[sp++] = node->first_child; /* descend, pre-order */
676 }
677 }
678 return ok && (guard < max_iter);
679}
