170static bool priv_backing_read(ra8_epub_book_t* book, uint64_t archive_ofs, uint8_t* buf, size_t n)
171{
172 if (book->zip_bytes != nullptr) {
173 if ((archive_ofs + (uint64_t)n) > (uint64_t)book->zip_size) {
174 return false; /* GCOVR_EXCL_LINE -- offsets come from a validated central-directory entry */
175 }
176 (void)memcpy(buf, &book->zip_bytes[(size_t)archive_ofs], n);
177 return true;
178 }
179 if (book->stream_media.read == nullptr) {
180 return false; /* GCOVR_EXCL_LINE -- a streamed book always carries a validated read callback */
181 }
182 const size_t got = book->stream_media.read(book->stream_media.ctx, archive_ofs, buf, n);
183 return got == n;
184}

203RA8_INTERNAL
204static bool priv_data_offset(const uint8_t* hdr, uint64_t local_header_ofs, uint64_t* out_data_ofs)
205{
206 const uint8_t sig[4] = {(uint8_t)k_ra8_epub_ldh_sig_0,
207 (uint8_t)k_ra8_epub_ldh_sig_1,
208 (uint8_t)k_ra8_epub_ldh_sig_2,
209 (uint8_t)k_ra8_epub_ldh_sig_3};
210 if (memcmp(hdr, sig, sizeof(sig)) != 0) {
211 return false; /* GCOVR_EXCL_LINE -- a located, stat'd entry always has a valid local header */
212 }
213 const uint32_t fname = (uint32_t)hdr[(uint8_t)k_ra8_epub_ldh_fname_len_ofs + k_ra8_epub_le_lo] |
214 ((uint32_t)hdr[(uint8_t)k_ra8_epub_ldh_fname_len_ofs + k_ra8_epub_le_hi]
215 << k_ra8_epub_le_shift);
216 const uint32_t extra = (uint32_t)hdr[(uint8_t)k_ra8_epub_ldh_extra_len_ofs + k_ra8_epub_le_lo] |
217 ((uint32_t)hdr[(uint8_t)k_ra8_epub_ldh_extra_len_ofs + k_ra8_epub_le_hi]
218 << k_ra8_epub_le_shift);
219 *out_data_ofs =
220 local_header_ofs + (uint64_t)k_ra8_epub_ldh_size + (uint64_t)fname + (uint64_t)extra;
221 return true;
222}

243RA8_INTERNAL
244static ra8_err_t priv_open_iter(ra8_epub_book_t* book,
245 const char* path,
246 mz_zip_reader_extract_iter_state** out_iter,
247 uint64_t* out_uncomp)
248{
249 mz_zip_archive* zip = priv_zip(book);
250 int32_t idx = 0;
251 const ra8_err_t err = priv_locate(zip, book, path, &idx);
252 if (err != k_ra8_ok) {
253 return err;
254 }
255 mz_zip_archive_file_stat st;
256 if (mz_zip_reader_file_stat(zip, (mz_uint)idx, &st) == MZ_FALSE) {
257 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- stat cannot fail on a located entry */
258 }
259 const ra8_err_t gerr = ra8_epub_zip_guard_entry(&st);
260 if (gerr != k_ra8_ok) {
261 return gerr; /* lying header / declared bomb: reject before inflation */
262 }
263 mz_zip_reader_extract_iter_state* iter = mz_zip_reader_extract_iter_new(zip, (mz_uint)idx, 0U);
264 if (iter == nullptr) {
265 return k_ra8_err_validation_failed;
266 }
267 *out_iter = iter;
268 *out_uncomp = (uint64_t)st.m_uncomp_size;
269 return k_ra8_ok;
270}

292RA8_INTERNAL
293static ra8_err_t priv_stored_data_offset(ra8_epub_book_t* book,
294 const char* path,
295 uint64_t* out_data_ofs,
296 uint64_t* out_uncomp)
297{
298 mz_zip_archive* zip = priv_zip(book);
299 int32_t idx = 0;
300 const ra8_err_t err = priv_locate(zip, book, path, &idx);
301 if (err != k_ra8_ok) {
302 return err;
303 }
304 mz_zip_archive_file_stat st;
305 if (mz_zip_reader_file_stat(zip, (mz_uint)idx, &st) == MZ_FALSE) {
306 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- stat cannot fail on a located entry */
307 }
308 const ra8_err_t gerr = ra8_epub_zip_guard_entry(&st);
309 if (gerr != k_ra8_ok) {
310 return gerr; /* lying header / declared bomb: reject before the copy */
311 }
312 if (st.m_method != 0U) {
313 return k_ra8_err_not_supported; /* DEFLATE -> use the forward cursor */
314 }
315 uint8_t hdr[k_ra8_epub_ldh_size] = {};
316 if (!priv_backing_read(book, (uint64_t)st.m_local_header_ofs, hdr, sizeof(hdr))) {
317 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- header bytes are within the archive */
318 }
319 if (!priv_data_offset(hdr, (uint64_t)st.m_local_header_ofs, out_data_ofs)) {
320 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- signature valid for a located entry */
321 }
322 *out_uncomp = (uint64_t)st.m_uncomp_size;
323 return k_ra8_ok;
324}

331ra8_err_t ra8_epub_entry_open(ra8_epub_book_t* book,
332 const char* path,
333 ra8_epub_entry_reader_t* out_reader,
334 uint64_t* out_size)
335{
336 RA8_CHECK_NULL_PTR(book, s_tag, "book must not be nullptr");
337 RA8_CHECK_NULL_PTR(path, s_tag, "path must not be nullptr");
338 RA8_CHECK_NULL_PTR(out_reader, s_tag, "out_reader must not be nullptr");
339 (void)memset(out_reader, 0, sizeof(*out_reader));
340 if (ra8_epub_internal_book_not_ready(book->in_use, book->zip_archive_active)) {
341 return k_ra8_err_not_initialized;
342 }
343
344 mz_zip_reader_extract_iter_state* iter = nullptr;
345 uint64_t uncomp = 0U;
346 const ra8_err_t err = priv_open_iter(book, path, &iter, &uncomp);
347 if (err != k_ra8_ok) {
348 return err;
349 }
350 out_reader->iter = iter;
351 out_reader->book = book;
352 out_reader->total = uncomp;
353 out_reader->done = (uncomp == 0U) ? 1U : 0U;
354 if (out_size != nullptr) {
355 *out_size = uncomp;
356 }
357 return k_ra8_ok;
358}

360ra8_err_t
361ra8_epub_entry_read(ra8_epub_entry_reader_t* reader, uint8_t* buf, size_t cap, size_t* got)
362{
363 RA8_CHECK_NULL_PTR(reader, s_tag, "reader must not be nullptr");
364 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
365 RA8_CHECK_NULL_PTR(got, s_tag, "got must not be nullptr");
366 *got = 0U;
367 if (reader->iter == nullptr) {
368 return k_ra8_err_not_initialized;
369 }
370 if (cap == 0U) {
371 return k_ra8_err_invalid_size;
372 }
373 if (reader->done != 0U) {
374 return k_ra8_ok; /* clean EOF already reported; *got stays 0 */
375 }
376
377 mz_zip_reader_extract_iter_state* iter = (mz_zip_reader_extract_iter_state*)reader->iter;
378 const size_t n = mz_zip_reader_extract_iter_read(iter, buf, cap);
379 reader->consumed += (uint64_t)n;
380 *got = n;
381 if (n < cap) {
382 /* A short read is end-of-entry only if every byte arrived; otherwise the
383 * compressed stream ended early / failed to inflate. */
384 if (reader->consumed >= reader->total) {
385 reader->done = 1U;
386 return k_ra8_ok;
387 }
388 return k_ra8_err_validation_failed;
389 }
390 return k_ra8_ok;
391}

393ra8_err_t ra8_epub_entry_close(ra8_epub_entry_reader_t* reader)
394{
395 RA8_CHECK_NULL_PTR(reader, s_tag, "reader must not be nullptr");
396 if (reader->iter == nullptr) {
397 return k_ra8_ok; /* idempotent: already closed / never opened */
398 }
399 mz_zip_reader_extract_iter_state* iter = (mz_zip_reader_extract_iter_state*)reader->iter;
400 bool full = false;
401 if (reader->total > 0U) {
402 if (reader->consumed >= reader->total) {
403 full = true;
404 }
405 }
406 const mz_bool ok = mz_zip_reader_extract_iter_free(iter);
407 reader->iter = nullptr;
408 if (full) {
409 if (ok == MZ_FALSE) {
410 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- good archives pass CRC/size */
411 }
412 }
413 return k_ra8_ok;
414}

447ra8_err_t ra8_epub_entry_pread(ra8_epub_book_t* book,
448 const char* path,
449 uint64_t offset,
450 uint8_t* buf,
451 size_t len,
452 size_t* got)
453{
454 const ra8_err_t nz = priv_pread_null_ok(book, path, buf, got);
455 if (nz != k_ra8_ok) {
456 return nz;
457 }
458 *got = 0U;
459 if (ra8_epub_internal_book_not_ready(book->in_use, book->zip_archive_active)) {
460 return k_ra8_err_not_initialized;
461 }
462
463 uint64_t data_ofs = 0U;
464 uint64_t uncomp = 0U;
465 const ra8_err_t err = priv_stored_data_offset(book, path, &data_ofs, &uncomp);
466 if (err != k_ra8_ok) {
467 return err;
468 }
469 if (offset >= uncomp) {
470 return k_ra8_ok; /* window starts at/after EOF; *got stays 0 */
471 }
472 const uint64_t avail = uncomp - offset;
473 const size_t n = ((uint64_t)len > avail) ? (size_t)avail : len;
474 if (!priv_backing_read(book, data_ofs + offset, buf, n)) {
475 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- data window is within the archive */
476 }
477 *got = n;
478 return k_ra8_ok;
