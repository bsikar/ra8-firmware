73def default_signal_test_command(*command: str) -> tuple[str, ...]:
74 """Wrap one test child so an asynchronous parent cannot mask its signals."""
75 if not command:
76 _fail("default-signal fixture command is empty")
77 program = """\
78import os
79import signal
80import sys
81for item in (signal.SIGHUP, signal.SIGINT, signal.SIGQUIT, signal.SIGTERM):
82 signal.signal(item, signal.SIG_DFL)
83os.execv(sys.argv[1], sys.argv[1:])
84"""
85 return (sys.executable, "-I", "-c", program, *command)

341def _installer_transaction_case(base: Path) -> None:
342 root = base / "installer-transaction"
343 root.mkdir()
344 _seed_launcher_repo(root)
345 result = _run_installer(root)
346 if result.returncode:
347 _fail(f"initial transactional install failed: {result.stderr}")
348 launcher = root / "scripts/git/hook-launcher"
349 with launcher.open("ab") as stream:
350 stream.write(b"\n# padding keeps the transaction observable\n")
351 stream.write(os.urandom(8 * 1024 * 1024))
352 _git(root, "add", "scripts/git/hook-launcher")
353 _git(root, "commit", "--quiet", "-m", "large launcher transaction fixture")
354 environment = sanitized_git_environment()
355 common = _managed_dir(root).parent
356 process = subprocess.Popen( # noqa: S603 -- audited private-repo installer
357 default_signal_test_command("/bin/bash", "-p", str(INSTALLER)),
358 cwd=root,
359 env=environment,
360 stdout=subprocess.PIPE,
361 stderr=subprocess.PIPE,
362 text=True,
363 start_new_session=True,
364 )
365 try:
366 _wait_for_staging(common, process)
367 os.kill(process.pid, signal.SIGSTOP)
368 contender = _run_installer(root)
369 if contender.returncode == 0 or "another hook installer" not in contender.stderr:
370 _fail("concurrent installer did not fail on the common-dir lock")
371 os.kill(process.pid, signal.SIGCONT)
372 os.killpg(process.pid, signal.SIGTERM)
373 process.communicate(timeout=15)
374 finally:
375 if process.poll() is None:
376 with suppress(ProcessLookupError):
377 os.killpg(process.pid, signal.SIGKILL)
378 process.wait(timeout=5)
379 residue = tuple(common.glob("ra8-hooks.*"))
380 hooks = tuple(sorted(path.name for path in _managed_dir(root).iterdir()))
381 if residue or hooks != tuple(sorted(HOOK_NAMES)):
382 _fail("interrupted installer left residue or a partial hook generation")

420def _configure_fixture(root: Path, state: str) -> tuple[ConfigState, ManagedState]:
421 """Prepare one exact pre-transaction config/directory state."""
422 if state == "managed":
423 result = _run_installer(root)

425 _fail(f"managed fixture install failed: {result.stderr}")

435def _signal_injected_installer(
436 base: Path,
437 label: str,
438 insertions: tuple[tuple[str, str], ...],
439) -> Path:
440 """Materialize the real installer with deterministic signal injections."""
441 text = INSTALLER.read_text(encoding="ascii")

465class _BoundaryCase:
466 """One exact signal/configuration transaction interruption."""
467
468 label: str
469 boundary: str
470 committed: bool
471 config_state: str = "managed"
472 signal_name: str = "TERM"
473 second_cleanup_signal: bool = False

476def _installer_boundary_signal_case(base: Path, case: _BoundaryCase) -> None:
477 """Signal one exact transaction boundary and verify the resulting generation."""
478 root = base / f"installer-boundary-{case.label}"
479 root.mkdir()
480 _seed_launcher_repo(root)
481 original_config, original_state = _configure_fixture(root, case.config_state)
482 launcher = root / "scripts/git/hook-launcher"
483 with launcher.open("a", encoding="ascii") as stream:
484 stream.write("\n# boundary signal generation\n")
485 _git(root, "add", "scripts/git/hook-launcher")
486 _git(root, "commit", "--quiet", "-m", "new launcher generation")
487
488 committed_state = _committed_state(root)
489 insertions = [(case.boundary, f' kill -{case.signal_name} "$$"')]
490 if case.second_cleanup_signal:
491 insertions.append((" trap '' HUP INT QUIT TERM", ' kill -TERM "$$"'))
492 installer = _signal_injected_installer(base, case.label, tuple(insertions))
493 result = _run_installer(root, installer=installer)
494 if result.returncode != ABORTED:
495 _fail(f"{case.label}: injected termination returned {result.returncode}, not 3")
496 if case.committed:
497 expected_config = (True, str(_managed_dir(root)))
498 _assert_transaction_state(root, case.label, expected_config, committed_state)
499 else:
500 _assert_transaction_state(root, case.label, original_config, original_state)
501 residue = tuple(_managed_dir(root).parent.glob("ra8-hooks.*"))
502 if residue:
503 _fail(f"{case.label}: interrupted transaction left residue: {residue!r}")
