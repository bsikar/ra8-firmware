160void board_periph_sci_feed_rx(uint8_t channel, const uint8_t* data, uint32_t len)
161{
162 if ((channel >= (uint32_t)k_sci_count) || (data == nullptr)) {
163 return;
164 }
165 sci_state_t* s = &s_sci[channel];
166 for (uint32_t i = 0U; i < len; i++) {
167 const uint32_t next = (s->rx_tail + 1U) % (uint32_t)k_sci_rx_queue_len;
168 if (next == s->rx_head) {
169 s->rx_dropped += (len - i); /* ring full: drop the remaining bytes */
170 if (board_periph_trace()) {
171 (void)
172 fprintf(stderr, " [trace] SCI%u RX queue full, dropped %u bytes\n", channel, len - i);
173 }
174 return;
175 }
176 s->rx[s->rx_tail] = data[i];
177 s->rx_tail = next;
178 }
179}

202static uint32_t sci_csr_value(const sci_state_t* s)
203{
204 /* TDRE + TEND are held set so ra8_sci's "wait for transmit empty / end" polls
205 * (ra8_sci_putc_polling on TDRE, internal_wait_tx_end on TEND) fall through.
206 * RXDMON reads high because an idle UART line idles high. RDRF tracks the
207 * host RX queue so ra8_sci_getc_polling completes only when a byte is ready. */
208 uint32_t csr = (uint32_t)k_sci_csr_tdre | (uint32_t)k_sci_csr_tend | (uint32_t)k_sci_csr_rxdmon;
209 if (sci_rx_available(s)) {
210 csr |= (uint32_t)k_sci_csr_rdrf;
211 }
212 return csr;
213}

216static uint64_t sci_reg_read(uint32_t ch, uint64_t off)
217{
218 sci_state_t* s = &s_sci[ch];
219 if (off == (uint64_t)k_sci_off_rdr) {
220 if (!sci_rx_available(s)) {
221 return 0U; /* drained: nothing queued */
222 }
223 const uint8_t b = s->rx[s->rx_head];
224 s->rx_head = (s->rx_head + 1U) % (uint32_t)k_sci_rx_queue_len;
225 s->received++;
226 return (uint64_t)b;
227 }
228 if (off == (uint64_t)k_sci_off_csr) {
229 return sci_csr_value(s);
230 }
231 if (off == (uint64_t)k_sci_off_ccr0) {
232 return s->ccr0;
233 }
234 if (off == (uint64_t)k_sci_off_frsr) {
235 return sci_rx_available(s) ? ((uint32_t)k_sci_frsr_dr | (uint32_t)k_sci_frsr_rdf) : 0U;
236 }
237 if (off == (uint64_t)k_sci_off_ftsr) {
238 return (uint32_t)k_sci_ftsr_tdfe; /* TX FIFO is always empty in the model */
239 }
240 return 0U; /* CFCLR / FFCLR read as 0; other regs unmodelled -> 0 */
241}

256static void sci_capture_tx_line(uint8_t byte)
257{
258 if (byte == (uint8_t)'\n') {
259 for (uint32_t i = 0U; i < s_uart_pend_len; i++) {
260 s_uart_last[i] = s_uart_pend[i];

264 * the tabbed board-view console + its deeper scrollback). */
265 board_console_push(k_board_console_ch_uart, s_uart_last);
266 s_uart_pend_len = 0U;
267 return;
268 }
269 if ((byte != (uint8_t)'\r') && (s_uart_pend_len < (uint32_t)(k_uart_line_cap - 1U))) {
270 s_uart_pend[s_uart_pend_len++] = (char)byte;

275static void sci_reg_write(uint32_t ch, uint64_t off, uint32_t value)
276{
277 sci_state_t* s = &s_sci[ch];
278 if (off == (uint64_t)k_sci_off_tdr) {
279 /* Both the polled (ra8_sci_putc_polling) and interrupt (ra8_sci_dispatch_txi)
280 * paths launch a frame by writing TDAT[7:0]; FIFO mode also writes TDR. */
281 const uint8_t byte = (uint8_t)(value & (uint32_t)k_sci_data_mask);
282 s->transmitted++;
283 /* Only the console channel carries UART text. Other channels (e.g. SCI0 in
284 * Simple-SPI mode for the microSD) move binary frames, so capturing + sinking
285 * their TX would spew the SPI traffic as "[uart] SCIn:" garbage -- skip it. */
286 if (ch == (uint32_t)k_sci_console_ch) {
287 sci_capture_tx_line(byte);
288 if (s_sci_tx_sink != nullptr) {
289 s_sci_tx_sink((uint8_t)ch, byte);
290 }
291 }
292 /* Pmod2 microSD is SCI0 in Simple-SPI mode: every TDR write clocks a
293 * full-duplex frame, so feed the modelled card's response back into this
294 * channel's RX queue (RDR/RDRF) -- the real ra8_sci_spi + ra8_sdmmc_spi path
295 * then runs against the FAT image exactly as on hardware. */
296 if (ch == (uint32_t)k_sci_sd_ch) {
297 /* With a card attached, return its modelled response; with none, return
298 * 0xFF (CIPO idles high on an empty bus). Always feeding a byte is what
299 * keeps RDRF advancing -- otherwise the firmware's full-duplex read polls
300 * RDRF forever and every ra8_sdmmc probe burns its whole timeout budget
301 * (seconds, per byte), which is the dominant cold-boot stall for any app
302 * that probes the SD without a card present. With 0xFF the probe simply
303 * sees no valid R1 and fails "no card" promptly, as on real hardware. */
304 const uint8_t resp =
305 board_sd_attached() ? board_sd_exchange(byte) : (uint8_t)k_sci_spi_cipo_idle;
306 board_periph_sci_feed_rx((uint8_t)ch, &resp, 1U);
307 }
308 /* MikroBUS UART (SCI7) with a modelled AT modem attached: forward each
309 * transmitted byte to the modem line model. Most bytes buffer silently; the
310 * terminating CR closes an AT command and the model returns the modem's
311 * reply, which we feed back into this channel's RX queue so the firmware's
312 * genuine ra8_modem_at -> ra8_sci_getc_polling path drains it (EIL == HIL). */
313 if ((ch == (uint32_t)board_modem_channel()) && board_modem_attached()) {
314 uint8_t mresp[k_sci_modem_resp_cap] = {};
315 const uint32_t mn = board_modem_feed_tx(byte, mresp, (uint32_t)k_sci_modem_resp_cap);
316 if (mn > 0U) {
317 board_periph_sci_feed_rx((uint8_t)ch, mresp, mn);
318 }
319 }
320 } else if (off == (uint64_t)k_sci_off_ccr0) {
321 s->ccr0 = value;
322 }
323 /* CFCLR / FFCLR are write-1-to-clear: in this model TDRE/TEND stay asserted
324 * and RDRF is derived from the live RX queue, so clearing them is a no-op
325 * (the firmware re-reads the queue-backed state on its next poll). */
326}

329static void sci_tick_channel(uc_engine* uc, uint32_t ch)
330{
331 if (ch != (uint32_t)k_sci_console_ch) {
332 return; /* only the console channel has modelled ELC event numbers */
333 }
334 const sci_state_t* s = &s_sci[ch];
335 /* TX is always empty/ended in the model: while the firmware keeps TIE/TEIE
336 * armed (an interrupt-driven ra8_sci_write in flight), re-pend TXI/TEI so the
337 * dispatcher pushes the next byte exactly as a real TDRE/TEND would. */
338 if ((s->ccr0 & (uint32_t)k_sci_ccr0_te) != 0U) {
339 if ((s->ccr0 & (uint32_t)k_sci_ccr0_tie) != 0U) {
340 board_periph_icu_raise_event(uc, (uint16_t)k_event_sci8_txi);
341 }
342 if ((s->ccr0 & (uint32_t)k_sci_ccr0_teie) != 0U) {
343 board_periph_icu_raise_event(uc, (uint16_t)k_event_sci8_tei);
344 }
345 }
346 /* RX: while RIE is armed and a host byte is queued, pend RXI so the firmware
347 * reads it from RDR in handler context (interrupt-driven receive). */
348 if (((s->ccr0 & (uint32_t)k_sci_ccr0_rie) != 0U) && sci_rx_available(s)) {
349 board_periph_icu_raise_event(uc, (uint16_t)k_event_sci8_rxi);
350 }
351}

395static void sci_report(void)
396{
397 for (uint32_t ch = 0U; ch < (uint32_t)k_sci_count; ch++) {
398 if ((s_sci[ch].transmitted > 0U) || (s_sci[ch].received > 0U)) {
399 (void)fprintf(stderr,
400 " SCI%u UART : TX %u bytes RX %u bytes (RX dropped %u)\n",
401 ch,
402 s_sci[ch].transmitted,
403 s_sci[ch].received,
404 s_sci[ch].rx_dropped);
405 }
406 }
407 /* An attached AT modem answers on the modem channel; summarise its traffic
408 * alongside the raw SCI byte counts (self-contained -- no main.c edit). */
409 board_modem_report();
410}
