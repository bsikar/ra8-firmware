145[[nodiscard]] static ra8_err_t rng_demo_emit_one_line(void)
146{
147 uint8_t rng[k_rng_demo_bytes_per_line] = {};
148 uint8_t hex[(uint16_t)k_rng_demo_bytes_per_line * k_rng_demo_hex_per_byte] = {};
149 const uint32_t hex_len =
150 (uint32_t)((uint16_t)k_rng_demo_bytes_per_line * (uint16_t)k_rng_demo_hex_per_byte);
151
152 if (ra8_psa_crypto_random(rng, sizeof(rng)) != k_ra8_ok) {
153 return k_ra8_err_hw_error;
154 }
155 /* Stuck-bit detector: if every byte in the 32-byte sample is the
156 * same value, the TRNG entropy path is almost certainly broken
157 * (probability of 32 random bytes all matching is 1 in 2^248).
158 * Without this check the demo would happily print "trng: 00...00"
159 * or "trng: ff...ff" forever and the probe would pass. */
160 bool all_same = true;
161 for (uint8_t i = 1U; i < (uint8_t)k_rng_demo_bytes_per_line; ++i) {
162 if (rng[i] != rng[0]) {
163 all_same = false;
164 break;
165 }
166 }
167 if (all_same) {
168 const uint8_t fail_banner[] = "rng_demo: FAIL stuck\r\n";
169 (void)ra8_board_uart_console_write(fail_banner, (size_t)(sizeof(fail_banner) - 1U));
170 return k_ra8_err_hw_error;
171 }
172 for (uint8_t i = 0U; i < (uint8_t)k_rng_demo_bytes_per_line; ++i) {
173 const size_t hi = (size_t)i * (size_t)k_rng_demo_hex_per_byte;
174 hex[hi] = rng_demo_nibble_to_hex((uint8_t)(rng[i] >> (uint8_t)k_rng_demo_nibble_shift));
175 hex[hi + 1U] = rng_demo_nibble_to_hex(rng[i]);
176 }
177
178 if (ra8_board_uart_console_write(k_rng_demo_prefix, (size_t)(sizeof(k_rng_demo_prefix) - 1U)) !=
179 k_ra8_ok) {
180 return k_ra8_err_hw_error;
181 }
182 if (ra8_board_uart_console_write(hex, (size_t)hex_len) != k_ra8_ok) {
183 return k_ra8_err_hw_error;
184 }
185 if (ra8_board_uart_console_write(k_rng_demo_eol, (size_t)(sizeof(k_rng_demo_eol) - 1U)) !=
186 k_ra8_ok) {
187 return k_ra8_err_hw_error;
188 }
189 /* Success-only verdict for the HIL scrape; fire-and-forget so it adds
190 * no decision to the documented MC/DC vector set above. */
191 (void)ra8_board_uart_console_write(k_rng_demo_pass_msg,
192 (size_t)(sizeof(k_rng_demo_pass_msg) - 1U));
193 return k_ra8_ok;
194}
