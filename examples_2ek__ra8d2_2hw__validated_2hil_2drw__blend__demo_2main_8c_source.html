301[[nodiscard]] static ra8_err_t drw_blend_arm(void)
302{
303 const ra8_drw_gradient_t grad = {
304 .color1_argb8888 = (uint32_t)k_drw_blend_fg_argb,
305 .color2_argb8888 = (uint32_t)k_drw_blend_bg_argb,
306 };
307 const ra8_err_t gerr = ra8_drw_set_gradient(&grad);
308 if (gerr != k_ra8_ok) {
309 return gerr;
310 }
311
312 /* True source-over: out = src*a + dst*(1-a). Source factor = alpha
313 * (BSF=1, BSI=0); destination factor = 1-alpha (BDF=1, BDI=1). The
314 * previous descriptor left dst_invert=false (dst factor = alpha, an
315 * additive mix), which is not source-over. */
316 const ra8_drw_blend_t blend = {
317 .use_alpha_channel = true,
318 .src_factor = true,
319 .dst_factor = true,
320 .src_invert = false,
321 .dst_invert = true,
322 .src_factor_alpha = true,
323 .dst_factor_alpha = true,
324 .src_invert_alpha = false,
325 .dst_invert_alpha = true,
326 .use_color2_dst = false,
327 .global_alpha = (uint8_t)k_drw_blend_alpha,
328 };
329 return ra8_drw_set_blend(&blend);
330}

333static uint32_t drw_blend_framebuffer_hash(void)
334{
335 const uint32_t words = (uint32_t)k_drw_blend_fb_dim * (uint32_t)k_drw_blend_fb_dim;
336 uint32_t hsh = (uint32_t)k_drw_blend_fnv_offset;
337 for (uint32_t w = 0U; w < words; ++w) {
338 const uint32_t px = g_drw_blend_fb[w];
339 for (uint32_t b = 0U; b < (uint32_t)sizeof(uint32_t); ++b) {
340 const uint8_t byte =
341 (uint8_t)((px >> (b * (uint32_t)k_drw_blend_byte_bits)) & (uint32_t)k_drw_blend_byte_mask);
342 hsh = (hsh ^ (uint32_t)byte) * (uint32_t)k_drw_blend_fnv_prime;
343 }
344 }
345 return hsh;
346}

369[[nodiscard]] static ra8_err_t drw_blend_render(uint32_t* out_crc)
370{
371 RA8_CHECK_NULL_PTR(out_crc, s_tag, "out_crc must not be nullptr");
372
373 const ra8_err_t bg = drw_blend_fill((int16_t)k_drw_blend_full_org,
374 (int16_t)k_drw_blend_full_org,
375 (uint16_t)k_drw_blend_fb_dim,
376 (uint16_t)k_drw_blend_fb_dim,
377 (uint32_t)k_drw_blend_bg_argb);
378 if (bg != k_ra8_ok) {
379 g_drw_blend_err = (uint32_t)bg;
380 return bg;
381 }
382
383 const ra8_err_t sp = drw_blend_fill((int16_t)k_drw_blend_sp_xy,
384 (int16_t)k_drw_blend_sp_xy,
385 (uint16_t)k_drw_blend_sp_wh,
386 (uint16_t)k_drw_blend_sp_wh,
387 (uint32_t)k_drw_blend_sp_argb);
388 if (sp != k_ra8_ok) {
389 g_drw_blend_err = (uint32_t)sp;
390 return sp;
391 }
392
393 const ra8_err_t armed = drw_blend_arm();
394 if (armed != k_ra8_ok) {
395 g_drw_blend_err = (uint32_t)armed;
396 return armed;
397 }
398
399 /* The blended fill carries the global alpha IN its colour: fill_rect
400 * writes the full COLOR1 register, so passing the opaque source colour
401 * here would clobber the armed alpha back to 0xFF and blend nothing. */
402 const ra8_err_t fg = drw_blend_fill((int16_t)k_drw_blend_fg_xy,
403 (int16_t)k_drw_blend_fg_xy,
404 (uint16_t)k_drw_blend_fg_wh,
405 (uint16_t)k_drw_blend_fg_wh,
406 (uint32_t)k_drw_blend_fg_a80);
407 if (fg != k_ra8_ok) {
408 g_drw_blend_err = (uint32_t)fg;
409 return fg;
410 }
411
412 g_drw_blend_err = 0U;
413 *out_crc = drw_blend_framebuffer_hash();
414 return k_ra8_ok;
415}

418static void drw_blend_print_hex(uint32_t value)
419{
420 static const uint8_t k_hex[] = "0123456789ABCDEF";
421 uint8_t out[(uint32_t)k_drw_blend_hex_digit];
422 for (uint32_t i = 0U; i < (uint32_t)k_drw_blend_hex_digit; ++i) {
423 const uint32_t shift = ((uint32_t)k_drw_blend_hi_nibble - i) * (uint32_t)k_drw_blend_nib_bits;
424 out[i] = k_hex[(value >> shift) & (uint32_t)k_drw_blend_nib_mask];
425 }
426 (void)ra8_board_uart_console_write(out, (size_t)sizeof(out));
427}

439#pragma GCC diagnostic push
440#pragma GCC diagnostic ignored "-Wmain"
441int32_t main(void)
442{
443 drw_blend_setup_or_halt();
444 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
445 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No NVIC sources
446 * are armed by this demo. */
447 ra8_isr_globals_enable();
448
449 if (drw_blend_configure() != k_ra8_ok) {
450 drw_blend_panic_halt();
451 }
452
453 while (1) {
454 uint32_t crc = 0U;
455 const ra8_err_t err = drw_blend_render(&crc);
456 if (err == k_ra8_ok) {
457 g_drw_blend_crc = crc;
458 drw_blend_print_pass(crc);
459 (void)ra8_board_led_toggle(k_ra8_board_led1);
460 } else {
461 (void)ra8_board_uart_console_write(k_drw_blend_fail_msg,
462 (size_t)(sizeof(k_drw_blend_fail_msg) - 1U));
463 (void)ra8_board_led_toggle(k_ra8_board_led2);
464 }
465 ++g_drw_blend_heartbeat;
466 ra8_delay_ms(k_drw_blend_period_ms);
467 }
468 drw_blend_panic_halt();
469 return 0;
470}
