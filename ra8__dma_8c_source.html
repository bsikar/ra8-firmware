214ra8_err_t ra8_dma_init(void)
215{
216 ra8_log_info(s_tag, "ra8_dma_init");
217
218 /* HUM Ch 11.2.6 "MSTPCRA : Module Stop Control Register A", p 443
219 * -- DMAC0 + DTC0 share MSTPA22. */
220 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_dmac0_dtc0);
221 if (mst_err != k_ra8_ok) {
222 ra8_log_error_val(s_tag, "mstp enable failed", (uint32_t)mst_err);
223 return k_ra8_err_hw_init_failed;
224 }
225
226 for (uint8_t ch = 0U; ch < k_ra8_dma_channel_count; ++ch) {
227 s_channels[ch].on_complete = nullptr;
228 s_channels[ch].ctx = nullptr;
229 s_channels[ch].in_use = false;
230 }
231 s_initialized = true;
232 return k_ra8_ok;
233}

256ra8_err_t ra8_dma_deinit(void)
257{
258 /* Tear down every in-use channel before releasing the MSTP
259 * reference so no pending transfer can wedge the DMAC block. */
260 for (uint8_t ch = 0U; ch < k_ra8_dma_channel_count; ++ch) {
261 if (s_channels[ch].in_use) {
262 /* Ignore dmac_stop errors -- the goal is to get every
263 * channel into a safe state before releasing MSTP. */
264 (void)ra8_dmac_stop(ch);
265 s_channels[ch].on_complete = nullptr;
266 s_channels[ch].ctx = nullptr;
267 s_channels[ch].in_use = false;
268 }
269 }
270
271 const ra8_err_t mst_err = ra8_mstp_disable(k_ra8_mstp_dmac0_dtc0);
272 if (mst_err != k_ra8_ok) {
273 ra8_log_error_val(s_tag, "mstp disable failed", (uint32_t)mst_err);
274 return k_ra8_err_hw_error;
275 }
276 s_initialized = false;
277 return k_ra8_ok;
278}

307ra8_err_t ra8_dma_request(const ra8_dma_request_t* req, uint8_t* out_channel)
308{
309 RA8_CHECK_NULL_PTR(req, s_tag, "request must not be NULL");
310 RA8_CHECK_NULL_PTR(out_channel, s_tag, "out_channel must not be NULL");
311 if (!s_initialized) {
312 return k_ra8_err_not_initialized;
313 }
314
315 const ra8_err_t verr = internal_validate_request(req);
316 if (verr != k_ra8_ok) {
317 return verr;
318 }
319
320 const uint8_t ch = internal_find_free();
321 if (ch == k_ra8_dma_channel_none) {
322 ra8_log_error(s_tag, "no free channel");
323 return k_ra8_err_no_mem;
324 }
325
326 ra8_dmac_config_t cfg = {};
327 internal_pack_dmac_cfg(req, &cfg);
328
329 const ra8_err_t derr = ra8_dmac_start(ch, &cfg);
330 if (derr != k_ra8_ok) {
331 ra8_log_error_val(s_tag, "dmac_start failed", (uint32_t)derr);
332 return k_ra8_err_hw_error;
333 }
334
335 s_channels[ch].on_complete = req->on_complete;
336 s_channels[ch].ctx = req->ctx;
337 s_channels[ch].in_use = true;
338#ifdef RA8_OFF_TARGET
339 s_fake_requests[ch] = *req;
340#endif
341 *out_channel = ch;
342 return k_ra8_ok;
343}

369ra8_err_t ra8_dma_release(uint8_t channel)
370{
371 if (channel >= k_ra8_dma_channel_count) {
372 return k_ra8_err_invalid_arg;
373 }
374 if (!s_channels[channel].in_use) {
375 return k_ra8_err_invalid_state;
376 }
377
378 /* ra8_dmac_stop itself drops one MSTP reference -- the bit
379 * stays cleared because ra8_dma_init holds the root reference. */
380 const ra8_err_t stop_err = ra8_dmac_stop(channel);
381 if (stop_err != k_ra8_ok) {
382 ra8_log_error_val(s_tag, "dmac_stop failed", (uint32_t)stop_err);
383 return k_ra8_err_hw_error;
384 }
385
386 s_channels[channel].on_complete = nullptr;
387 s_channels[channel].ctx = nullptr;
388 s_channels[channel].in_use = false;
389 return k_ra8_ok;
390}
