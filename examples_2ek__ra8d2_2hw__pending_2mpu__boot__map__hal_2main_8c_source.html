205static bool mpu_boot_test_map(void)
206{
207 uint8_t count = 0U;
208 const ra8_mpu_region_t* map = ra8_mpu_boot_map(&count);
209 if (map == nullptr) {
210 return false;
211 }
212 if (count != (uint8_t)k_ra8_mpu_boot_region_count) {
213 return false;
214 }
215 const ra8_mpu_region_t* shram = &map[k_mpu_boot_idx_shram];
216 if (shram->base != (uintptr_t)k_mpu_boot_shram_base) {
217 return false;
218 }
219 if (shram->size != (uint32_t)k_mpu_boot_shram_size) {
220 return false;
221 }
222 if (shram->attr_idx != k_ra8_mpu_attr_idx_1) {
223 return false;
224 }
225 return true;
226}

375int main(void)
376{
377 ra8_log_init();
378 ra8_log_info("MPU_BOOT", "==== mpu_boot_map_hal: MPU boot map via ra8_mpu HAL ====");
379
380 if (!mpu_boot_setup()) {
381 ra8_log_info("MPU_BOOT", "setup FAILED (CGC / console) -- halting");
382 mpu_boot_emit(k_mpu_boot_fail_setup, sizeof(k_mpu_boot_fail_setup) - 1U);
383 mpu_boot_wfi_forever();
384 }
385
386 const mpu_boot_result_t result = mpu_boot_run_selftest();
387
388 switch (result) {
389 case k_mpu_boot_result_pass:
390 ra8_log_info("MPU_BOOT", "mpu-via-hal PASS (enabled + canonical map + Device MMIO)");
391 mpu_boot_emit(k_mpu_boot_pass_banner, sizeof(k_mpu_boot_pass_banner) - 1U);
392 break;
393 case k_mpu_boot_result_fail_mpu:
394 ra8_log_info("MPU_BOOT", "MPU-enabled step FAILED (apply_boot_map did not enable)");
395 mpu_boot_emit(k_mpu_boot_fail_mpu, sizeof(k_mpu_boot_fail_mpu) - 1U);
396 break;
397 case k_mpu_boot_result_fail_map:
398 ra8_log_info("MPU_BOOT", "boot-map step FAILED (region table mismatch)");
399 mpu_boot_emit(k_mpu_boot_fail_map, sizeof(k_mpu_boot_fail_map) - 1U);
400 break;
401 case k_mpu_boot_result_fail_mmio:
402 default:
403 ra8_log_info("MPU_BOOT", "device-MMIO step FAILED");
404 mpu_boot_emit(k_mpu_boot_fail_mmio, sizeof(k_mpu_boot_fail_mmio) - 1U);
405 break;
406 }
407
408 mpu_boot_wfi_forever();
409}
