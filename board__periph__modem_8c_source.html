106static uint32_t modem_emit(const char* resp, uint8_t* out, uint32_t out_cap)
107{
108 uint32_t n = 0U;
109 while ((n < out_cap) && (resp[n] != '\0')) {
110 out[n] = (uint8_t)resp[n];
111 n++;
112 }
113 return n;

123static uint32_t modem_answer_line(uint8_t* out, uint32_t out_cap)
124{
125 s_modem.cmd[s_modem.cmd_len] = '\0';
126 const uint32_t count = (uint32_t)(sizeof(k_modem_script) / sizeof(k_modem_script[0]));
127 for (uint32_t i = 0U; i < count; i++) {
128 if (strcmp(s_modem.cmd, k_modem_script[i].cmd) == 0) {
129 s_modem.answered++;
130 return modem_emit(k_modem_script[i].resp, out, out_cap);
131 }
132 }
133 s_modem.errors++;
134 return modem_emit(k_modem_cme_error, out, out_cap);
135}

154uint32_t board_modem_feed_tx(uint8_t tx, uint8_t* out, uint32_t out_cap)
155{
156 if ((out == nullptr) || (out_cap == 0U)) {
157 return 0U;
158 }
159 if (tx == (uint8_t)k_modem_lf) {
160 return 0U; /* modems terminate on CR; LF (if any) is not a delimiter */
161 }
162 if (tx == (uint8_t)k_modem_cr) {
163 const uint32_t n = modem_answer_line(out, out_cap);
164 s_modem.cmd_len = 0U;
165 return n;
166 }
167 if (s_modem.cmd_len < (uint32_t)(k_modem_cmd_cap - 1U)) {
168 s_modem.cmd[s_modem.cmd_len] = (char)tx;
169 s_modem.cmd_len++;
170 }
171 return 0U;
172}
