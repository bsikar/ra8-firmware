124RA8_INTERNAL static uint32_t ra8_crashlog_crc32(const volatile uint8_t* data, uint32_t len)
125{
126 if (data == nullptr) {
127 return 0U;
128 }
129 uint32_t crc = (uint32_t)k_ra8_crashlog_crc_seed;
130 for (uint32_t i = 0U; i < len; i++) {
131 crc ^= (uint32_t)data[i];
132 for (uint8_t bit = 0U; bit < (uint8_t)k_ra8_crashlog_crc_bits; bit++) {
133 const uint32_t mask = 0U - (crc & 1U);
134 crc = (crc >> 1U) ^ ((uint32_t)k_ra8_crashlog_crc_poly & mask);
135 }
136 }
137 return crc ^ (uint32_t)k_ra8_crashlog_crc_seed;
138}

159RA8_INTERNAL static uint32_t ra8_crashlog_payload_crc(void)

202void ra8_crashlog_record_fault(const volatile ra8_exception_last_t* decoded)
203{
204 if (decoded == nullptr) {
205 return;
206 }
207 /* Carry the prior crash count forward only if the existing record still
208 * validates; a cold-boot / corrupted record restarts the count at 1. */
209 uint32_t next = 1U;
210 if (ra8_crashlog_is_valid()) {
211 const uint32_t prior = s_ra8_crashlog_record.boot_loops;
212 next = (prior < (uint32_t)k_ra8_crashlog_loops_max) ? (prior + 1U)
213 : (uint32_t)k_ra8_crashlog_loops_max;
214 }
215 s_ra8_crashlog_record.magic = 0U; /* invalidate for the write window */
216 s_ra8_crashlog_record.boot_loops = next;
217 s_ra8_crashlog_record.fault = *decoded;
218 s_ra8_crashlog_record.crc = ra8_crashlog_payload_crc();
219 s_ra8_crashlog_record.magic = (uint32_t)k_ra8_crashlog_magic_valid; /* validate LAST */
220}
