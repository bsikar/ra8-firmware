111static void sd_print_hex8(uint8_t value)
112{
113 uint8_t buf[2];
114 const uint32_t hi = ((uint32_t)value >> (uint32_t)k_sd_hex_shift) & (uint32_t)k_sd_hex_mask;
115 const uint32_t lo = (uint32_t)value & (uint32_t)k_sd_hex_mask;
116 buf[0] = (uint8_t)((hi < (uint32_t)k_sd_dec_ten) ? ('0' + hi) : ('A' + (hi - k_sd_dec_ten)));
117 buf[1] = (uint8_t)((lo < (uint32_t)k_sd_dec_ten) ? ('0' + lo) : ('A' + (lo - k_sd_dec_ten)));
118 sd_print(buf, 2U);
119}

131static void sd_setup_or_halt(uint32_t* out_pclka_hz)
132{
133 uint32_t cpuclk0_hz = 0U;
134 if ((ra8_cgc_init() != k_ra8_ok) || (ra8_mstp_init() != k_ra8_ok)) {
135 sd_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
136 }
137 if ((ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) ||
138 (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, out_pclka_hz) != k_ra8_ok)) {
139 sd_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
140 }
141 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
142 sd_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
143 }
144 if (sd_pins_init() != k_ra8_ok) {
145 sd_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
146 }
147 if (ra8_board_uart_console_init((uint32_t)k_sd_uart_baud) != k_ra8_ok) {
148 sd_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);

164static void sd_read_whoami_or_halt(uint8_t* out_a, uint8_t* out_b)
165{
166 uint8_t a = 0U;
167 if (ra8_smbus_read_byte_data((uint8_t)k_sd_lsm_addr, (uint8_t)k_sd_reg_whoami, &a) != k_ra8_ok) {
168 sd_panic_halt(k_msg_nak, (uint32_t)sizeof(k_msg_nak) - 1U);
169 }
170 uint8_t b = 0U;
171 if (ra8_smbus_send_byte((uint8_t)k_sd_lsm_addr, (uint8_t)k_sd_reg_whoami) != k_ra8_ok) {
172 sd_panic_halt(k_msg_nak, (uint32_t)sizeof(k_msg_nak) - 1U);
173 }
174 if (ra8_smbus_receive_byte((uint8_t)k_sd_lsm_addr, &b) != k_ra8_ok) {
175 sd_panic_halt(k_msg_nak, (uint32_t)sizeof(k_msg_nak) - 1U);
176 }
177 if ((a != (uint8_t)k_sd_whoami_val) || (b != (uint8_t)k_sd_whoami_val)) {
178 sd_panic_halt(k_msg_bad, (uint32_t)sizeof(k_msg_bad) - 1U);
179 }
180 *out_a = a;
181 *out_b = b;
182}

196int32_t main(void)
197{
198 uint32_t pclka_hz = 0U;
199 sd_setup_or_halt(&pclka_hz);
200 ra8_isr_globals_enable();
201 sd_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
202
203 /* App-owned bus bring-up: IIC_B in I2C-compat mode, bound through the
204 * ra8_io facade into the SMBus layer's injected seam. A future board
205 * revision that moves the bus onto a RIIC channel only swaps the bind. */
206 const ra8_i3c_cfg_t iic_cfg = {.mode = k_ra8_i3c_mode_i2c,
207 .bus_hz = (uint32_t)k_sd_bus_hz,
208 .pclka_hz = pclka_hz};
209 ra8_i2c_bus_ops_t bus_ops = {};
210 if (ra8_i3c_init((uint8_t)k_sd_iic_chan, &iic_cfg) != k_ra8_ok) {
211 sd_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
212 }
213 if (ra8_io_i2c_bus_bind_i3c_compat(&s_sd_bus, (uint8_t)k_sd_iic_chan) != k_ra8_ok) {
214 sd_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
215 }
216 if (ra8_io_i2c_bus_as_ops(&s_sd_bus, &bus_ops) != k_ra8_ok) {
217 sd_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
218 }
219
220 const ra8_smbus_cfg_t cfg = {.bus = bus_ops, .pec_enabled = false};
221 if (ra8_smbus_init(&cfg) != k_ra8_ok) {
222 sd_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
223 }
224
225 uint8_t a = 0U;
226 uint8_t b = 0U;
227 sd_read_whoami_or_halt(&a, &b);
228
229 sd_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
230 sd_print_hex8(a);
231 sd_print(k_msg_mid, (uint32_t)sizeof(k_msg_mid) - 1U);
232 sd_print_hex8(b);
233 sd_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
234
235 while (1) {
236 __asm__ volatile("wfi");
237 }
238}
