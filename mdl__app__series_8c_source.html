62RA8_INTERNAL static bool internal_list_range(const mdl_url_list_t* l, double* lo, double* hi)
63{
64 *lo = 0.0;
65 *hi = 0.0;
66 if (l->count == 0U) {
67 return false;
68 }
69 if (!mdl_urlname_chapter_parse(l->urls[0], lo)) {
70 return false;
71 }
72 *hi = *lo;
73 for (size_t i = 1U; i < l->count; ++i) {
74 double num = 0.0;
75 if (!mdl_urlname_chapter_parse(l->urls[i], &num)) {
76 return false;
77 }
78 *lo = (num < *lo) ? num : *lo;
79 *hi = (num > *hi) ? num : *hi;
80 }
81 return true;
82}

101internal_select_window(bool from_present, double from_num, size_t count, mdl_url_list_t* out)
102{

112 }
113 }
114 out->count = 0U;
115 for (size_t i = start; (i < priv_mdl_app_context()->chapters.count) && (out->count < count);

117 memcpy(out->urls[out->count], priv_mdl_app_context()->chapters.urls[i], k_mdl_url_max);
118 out->count += 1U;
119 }

594RA8_INTERNAL static int internal_run_prepared(const mdl_series_run_t* r,
595 const mdl_site_t* site,
596 const char* abs_dir,
597 const char* state_path,
598 mdl_cache_t* cache)
599{
600 const mdl_url_list_t* sel = nullptr;
601 if (internal_select_run_window(r, &sel) != k_ra8_ok) {
602 return 1;
603 }
604 char display_slug[k_slug_bytes];
605 (void)mdl_sanitize_segment(priv_mdl_app_context()->state.series_title,
606 display_slug,
607 sizeof(display_slug));
608 char combined_rel[k_leaf_name_bytes];
609 mdl_fetch_layout_t layout =
610 internal_choose_layout(r, sel, combined_rel, sizeof(combined_rel), display_slug);
611
612 mdl_governor_t gov;
613 const mdl_gov_cfg_t cfg = mdl_config_gov_cfg(site);
614 mdl_governor_init(&gov, &cfg, r->seed);
615 memset(&s_faillog, 0, sizeof(s_faillog));
616 mdl_fetch_ctx_t ctx = internal_make_ctx(r, site, abs_dir, state_path, &gov, cache);
617 const ra8_err_t crc = priv_mdl_app_ensure_series_cover(&ctx, abs_dir);
618 if (crc != k_ra8_ok) {
619 internal_report_cover_failure(crc);
620 return 1;
621 }
622 if (internal_save_series_state(state_path, "mdl: failed to checkpoint series metadata '") !=
623 k_ra8_ok) {
624 return 1;
625 }
626 const int64_t run_start = (int64_t)time(nullptr);
627 mdl_fetch_stats_t stats;
628 const ra8_err_t frc = mdl_fetch_run(&ctx,
629 sel,
630 layout,
631 (layout == k_mdl_layout_combined) ? combined_rel : nullptr,
632 &stats);
633 if (ctx.progress_error != k_ra8_ok) {
634 return 1;
635 }
636 const ra8_err_t src =
637 internal_save_series_state(state_path, "mdl: failed to save library state '");
638 if (internal_report_stats(abs_dir, &stats) != k_ra8_ok) {
639 return 1;
640 }
641 if (mdl_report_failures(priv_mdl_app_context()->diagnostic, &s_faillog) != k_ra8_ok) {
642 return 1;
643 }
644 const size_t efail =
645 internal_export_after(r, abs_dir, layout, combined_rel, sel, &stats, run_start);
646 return ((frc == k_ra8_ok) && (src == k_ra8_ok) && (efail == 0U)) ? 0 : 1;
647}
