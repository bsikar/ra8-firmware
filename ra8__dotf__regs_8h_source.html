236static inline volatile ra8_dotf_regs_t* ra8_dotf_regs(uint8_t channel)
237{
238 if ((uint16_t)channel >= (uint16_t)k_ra8_dotf_channel_count) {
239 return nullptr;
240 }

242 ((uintptr_t)channel * k_ra8_dotf_stride));
243}
