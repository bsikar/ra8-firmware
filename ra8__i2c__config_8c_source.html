106static uint8_t internal_i2c_pick_cks(uint32_t* total)
107{
108 uint32_t cks = 0U;
109 for (uint32_t i = 0U; i <= (uint32_t)k_ra8_i2c_cks_max; i++) { /* GCOVR_EXCL_BR_LINE */
110 const uint32_t half = *total / (uint32_t)k_ra8_i2c_period_split;
111 if (half <= ((uint32_t)k_ra8_i2c_br_field_max + 1U)) {
112 break;
113 }
114 cks = i + 1U;
115 *total = *total >> 1U;
116 }
117 if (cks > (uint32_t)k_ra8_i2c_cks_max) {
118 cks = (uint32_t)k_ra8_i2c_cks_max;
119 }
120 return (uint8_t)cks;
121}

142static uint8_t internal_i2c_clamp_half(uint32_t total)
143{
144 uint32_t half = total / (uint32_t)k_ra8_i2c_period_split;
145 if (half == 0U) {
146 half = 1U;
147 }
148 uint32_t field = half - 1U;
149 if (field > (uint32_t)k_ra8_i2c_br_field_max) {
150 field = (uint32_t)k_ra8_i2c_br_field_max;
151 }
152 return (uint8_t)field;
153}

182static ra8_err_t internal_i2c_bitrate(uint32_t bus_hz,
183 uint32_t pclkb_hz,
184 uint8_t* out_cks,
185 uint8_t* out_brh,
186 uint8_t* out_brl)
187{
188 RA8_CHECK_NULL_PTR(out_cks, s_i2c_tag, "bitrate: out_cks");
189 RA8_CHECK_NULL_PTR(out_brh, s_i2c_tag, "bitrate: out_brh");
190 if (ra8_i2c_internal_clk_invalid(bus_hz, pclkb_hz)) {
191 return k_ra8_err_invalid_arg;
192 }
193
194 /* Each transferred bit needs (BRH+1)+(BRL+1) IICphi cycles. Pick the
195 * smallest CKS that fits both half-periods in the 5-bit field, then
196 * derive the clamped half-period count for ICBRL / ICBRH. */
197 uint32_t total = pclkb_hz / bus_hz;
198 const uint8_t cks = internal_i2c_pick_cks(&total);
199 const uint8_t field = internal_i2c_clamp_half(total);
200
201 *out_cks = cks;
202 *out_brh = (uint8_t)((uint32_t)k_ra8_i2c_br_reserved_hi | (uint32_t)field);
203 *out_brl = (uint8_t)((uint32_t)k_ra8_i2c_br_reserved_hi | (uint32_t)field);
204 return k_ra8_ok;
205}

226static uint8_t internal_i2c_decode_errors(uint8_t icsr2)
227{
228 uint8_t mask = (uint8_t)k_ra8_i2c_err_none;
229 if ((icsr2 & (uint8_t)k_ra8_i2c_msk_icsr2_al) != 0U) {
230 mask |= (uint8_t)k_ra8_i2c_err_arb_lost;
231 }
232 if ((icsr2 & (uint8_t)k_ra8_i2c_msk_icsr2_nackf) != 0U) {
233 mask |= (uint8_t)k_ra8_i2c_err_nack;
234 }
235 /* TMOF lives in ICSR2 bit 0; reuse the timeout mask via the position. */
236 if ((icsr2 & (uint8_t)(1U << (uint8_t)k_ra8_i2c_icsr2_tmof_pos)) != 0U) {
237 mask |= (uint8_t)k_ra8_i2c_err_timeout;
238 }
239 return mask;
240}

269static void internal_i2c_apply_init_regs(volatile r_i2c_regs_t* reg,
270 uint8_t cks,
271 uint8_t brh,
272 uint8_t brl,
273 bool fast_plus)
274{
275 /* Initial-settings step 1 (Figure 39.5): ICE = 0, pins inactive.
276 * HUM Ch 39.2.1 "ICCR1 : I2C Bus Control Register 1" p 2369 */
277 reg->ICCR1 = 0U;
278 /* Initial-settings step 2: IICRST = 1 (IIC reset, ICE still 0).
279 * HUM Ch 39.2.1 "ICCR1 : I2C Bus Control Register 1" p 2369 */
280 reg->ICCR1 = (uint8_t)k_ra8_i2c_msk_iccr1_iicrst;
281 /* Initial-settings step 3: ICE = 1 (internal reset, pins active).
282 * HUM Ch 39.2.1 "ICCR1 : I2C Bus Control Register 1" p 2369 */
283 reg->ICCR1 = (uint8_t)((uint8_t)k_ra8_i2c_msk_iccr1_iicrst | (uint8_t)k_ra8_i2c_msk_iccr1_ice);
284
285 /* HUM Ch 39.2.3 "ICMR1 : I2C Bus Mode Register 1 -- CKS[6:4]" p 2374 */
286 reg->ICMR1 = (uint8_t)((uint32_t)cks << (uint32_t)k_ra8_i2c_icmr1_cks_pos);
287 /* HUM Ch 39.2.15 "ICBRL : I2C Bus Bit Rate Low-Level Register" p 2391 */
288 reg->ICBRL = brl;
289 /* HUM Ch 39.2.16 "ICBRH : I2C Bus Bit Rate High-Level Register" p 2392 */
290 reg->ICBRH = brh;
291
292 /* Enable arbitration-lost detection, NACK transfer suspension, the SCL
293 * synchronous circuit, and (for Fm+) the slope-control circuit.
294 * HUM Ch 39.2.6 "ICFER : I2C Bus Function Enable Register" p 2378 */
295 uint8_t icfer = (uint8_t)((uint8_t)k_ra8_i2c_msk_icfer_male | (uint8_t)k_ra8_i2c_msk_icfer_nacke |
296 (uint8_t)k_ra8_i2c_msk_icfer_scle);
297 if (fast_plus) {
298 icfer |= (uint8_t)k_ra8_i2c_msk_icfer_fmpe;
299 }
300 reg->ICFER = icfer;
301
302 /* Initial-settings step 5: release the internal reset (IICRST = 0).
303 * HUM Ch 39.2.1 "ICCR1 : I2C Bus Control Register 1" p 2369 */
304 reg->ICCR1 = (uint8_t)k_ra8_i2c_msk_iccr1_ice;
305}

307ra8_err_t ra8_i2c_init(uint8_t channel, const ra8_i2c_cfg_t* cfg)
308{
309 RA8_CHECK_NULL_PTR(cfg, s_i2c_tag, "i2c_init: cfg");
310 if (ra8_i2c_internal_clk_invalid(cfg->bus_hz, cfg->pclkb_hz)) {
311 return k_ra8_err_invalid_arg;
312 }
313 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
314 if (reg == nullptr) {
315 return k_ra8_err_invalid_arg;
316 }
317
318 const ra8_err_t mst_err = ra8_mstp_enable(internal_i2c_mstp_id(channel));
319 RA8_RETURN_ON_ERROR(mst_err, s_i2c_tag, "i2c_init: mstp"); /* GCOVR_EXCL_BR_LINE */
320
321 uint8_t cks = 0U;
322 uint8_t brh = 0U;
323 uint8_t brl = 0U;
324 const ra8_err_t br_err = internal_i2c_bitrate(cfg->bus_hz, cfg->pclkb_hz, &cks, &brh, &brl);
325 RA8_RETURN_ON_ERROR(br_err, s_i2c_tag, "i2c_init: bitrate"); /* GCOVR_EXCL_BR_LINE */
326
327 internal_i2c_apply_init_regs(reg,
328 cks,
329 brh,
330 brl,
331 cfg->bus_hz >= (uint32_t)k_ra8_i2c_speed_fast_plus);
332
333 s_i2c_state[channel].initialized = true;
334 s_i2c_state[channel].bus_held = false;
335
336 ra8_log_info_val(s_i2c_tag, "i2c_init channel", (uint32_t)channel);
337 return k_ra8_ok;
338}

340ra8_err_t ra8_i2c_deinit(uint8_t channel)
341{
342 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
343 if (reg == nullptr) {
344 return k_ra8_err_invalid_arg;
345 }
346 /* ICE = 0: place the SCL/SDA pins back in the inactive state.
347 * HUM Ch 39.2.1 "ICCR1 : I2C Bus Control Register 1" p 2369 */
348 reg->ICCR1 = 0U;
349 s_i2c_state[channel].initialized = false;
350 s_i2c_state[channel].bus_held = false;
351 return ra8_mstp_disable(internal_i2c_mstp_id(channel));

354ra8_err_t ra8_i2c_set_clock(uint8_t channel, uint32_t bus_hz, uint32_t pclkb_hz)
355{
356 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
357 if (reg == nullptr) {
358 return k_ra8_err_invalid_arg;
359 }
360 uint8_t cks = 0U;
361 uint8_t brh = 0U;
362 uint8_t brl = 0U;
363 const ra8_err_t br_err = internal_i2c_bitrate(bus_hz, pclkb_hz, &cks, &brh, &brl);

365 return br_err;
366 }
367 /* HUM Ch 39.2.3 "ICMR1 : I2C Bus Mode Register 1 -- CKS[6:4]" p 2374 */
368 reg->ICMR1 = (uint8_t)(((uint32_t)reg->ICMR1 & ~((uint32_t)(uint8_t)k_ra8_i2c_cks_max
369 << (uint32_t)k_ra8_i2c_icmr1_cks_pos)) |
370 ((uint32_t)cks << (uint32_t)k_ra8_i2c_icmr1_cks_pos));
371 /* HUM Ch 39.2.15 "ICBRL : I2C Bus Bit Rate Low-Level Register" p 2391 */
372 reg->ICBRL = brl;
373 /* HUM Ch 39.2.16 "ICBRH : I2C Bus Bit Rate High-Level Register" p 2392 */
374 reg->ICBRH = brh;
375 return k_ra8_ok;
