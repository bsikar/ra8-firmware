188{
189 ra8_sram_config_t cfg = {};
190 cfg.banks[k_ecc_demo_bank].ecc_mode = k_ra8_sram_ecc_with_chk;
191 cfg.banks[k_ecc_demo_bank].on_error = k_ra8_sram_on_error_interrupt;
192 cfg.banks[k_ecc_demo_bank].enable_1bit_latch = true;
193 cfg.banks[k_ecc_demo_bank].eccrgn = k_ra8_sram_region_128kb;
194 cfg.banks[k_ecc_demo_bank].zero_init = true;
195
196 const ra8_err_t ierr = ra8_sram_init(&cfg);
197 if (ierr != k_ra8_ok) {
198 return ierr;

203 return berr;
204 }
205 s_ecc_buf = (volatile uint32_t*)info.data_base;
206 return k_ra8_ok;

228[[nodiscard]] static ra8_err_t ecc_demo_sample(uint8_t* out_ok)
229{
230 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
231 if (s_ecc_buf == nullptr) {
232 return k_ra8_err_not_initialized;
233 }
234
235 for (uint32_t i = 0U; i < (uint32_t)k_ecc_demo_test_words; ++i) {
236 s_ecc_buf[i] = ecc_demo_pattern(i);
237 }
238 uint8_t rw = 1U;
239 for (uint32_t i = 0U; i < (uint32_t)k_ecc_demo_test_words; ++i) {
240 if (s_ecc_buf[i] != ecc_demo_pattern(i)) {
241 rw = 0U;
242 }
243 }
244 g_ecc_rw_ok = (uint32_t)rw;
245
246 ra8_sram_status_t st = {};
247 const ra8_err_t err = ra8_sram_get_status(&st);
248 if (err != k_ra8_ok) {
249 return err;
250 }
251 g_ecc_esr = (uint32_t)st.raw_esr;
252 g_ecc_1bit = (uint32_t)st.one_bit_mask;
253 g_ecc_2bit = (uint32_t)st.two_bit_mask;
254 /* Verdict gates only on the ECC-protected round-trip, which is
255 * deterministic. The latched 1-bit / 2-bit error masks (above) are the
256 * hardware-error-reporting payload -- meaningful only on silicon, where a
257 * real bit-flip sets them; ra8_emulator does not model ECC, so they stay out
258 * of the headless verdict (else it would flake on the unmodelled ESR). */
259 *out_ok = (rw != 0U) ? 1U : 0U;
260 return k_ra8_ok;
261}

265int32_t main(void)
266{
267 ecc_demo_setup_or_halt();
268 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
269 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). The ECC NMI is
270 * non-maskable and unaffected; no maskable NVIC source is armed. */
271 ra8_isr_globals_enable();
272
273 if (ecc_demo_configure() != k_ra8_ok) {
274 ecc_demo_panic_halt();
275 }
276
277 while (1) {
278 uint8_t ok = 0U;
279 const ra8_err_t err = ecc_demo_sample(&ok);
280 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
281 g_ecc_ok = (uint32_t)good;
282 if (good != 0U) {
283 (void)ra8_board_uart_console_write(k_ecc_demo_ok_msg,
284 (size_t)(sizeof(k_ecc_demo_ok_msg) - 1U));
285 (void)ra8_board_led_toggle(k_ra8_board_led1);
286 } else {
287 (void)ra8_board_uart_console_write(k_ecc_demo_bad_msg,
288 (size_t)(sizeof(k_ecc_demo_bad_msg) - 1U));
289 (void)ra8_board_led_toggle(k_ra8_board_led2);
290 }
291 ++g_ecc_heartbeat;
292 ra8_delay_ms(k_ecc_demo_period_ms);
293 }
294 ecc_demo_panic_halt();
295 return 0;
296}
