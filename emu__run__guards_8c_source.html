105static void guard_read_wall(double* wall_s, bool* wall_guard_on)
106{
107 const char* const e_wall = getenv("RA8_EMU_WALL_S");
108 if (e_wall == nullptr) {
109 return;
110 }
111 const long v = strtol(e_wall, nullptr, (int)k_env_strtol_base);
112 if (v > 0L) {
113 *wall_s = (double)v;
114 } else if (v == 0L) {
115 *wall_guard_on = false; /* explicit opt-out: bound the run by chunks only */
116 }
117}

251run_guards_t run_read_guards(const emu_run_cfg_t* cfg, const board_view_t* view)
252{
253 /* RA8_EMU_CLICK_SETTLE=N: widen the post-click drain for a tap that kicks off
254 * a long operation (e.g. opening a big book from SD). Unset keeps the default. */
255 const uint32_t click_settle_chunks =
256 guard_env_u32("RA8_EMU_CLICK_SETTLE", (uint32_t)k_click_settle_chunks);
257
258 /* Chunked run: one SysTick (one ThreadX tick) per outer chunk. Headless runs
259 * stop on a chunk budget + wall-clock guard; --view runs until the window is
260 * closed. RA8_EMU_WALL_S / MAX_CHUNKS override the guards without a recompile
261 * and have no effect in --view mode. */
262 uint32_t max_chunks =
263 (view != nullptr) ? (uint32_t)k_view_max_chunks : (uint32_t)k_run_max_chunks;
264 double wall_s = (double)k_run_wall_s;
265 bool wall_guard_on = true;
266 guard_read_wall(&wall_s, &wall_guard_on);
267 max_chunks = guard_env_u32_headless("RA8_EMU_MAX_CHUNKS", max_chunks, view);
268 guard_setup_record(cfg, view, &max_chunks);
269
270 /* Headless-only early-stop knobs: idle steady-state (IDLE_STOP), USB device
271 * CONFIGURED (USB_STOP) and USB host complete (USBH_STOP). Off (0) by default. */
272 const uint32_t idle_stop_chunks = guard_env_u32_headless("RA8_EMU_IDLE_STOP", 0U, view);
273 const uint32_t usb_stop_settle = guard_env_u32_headless("RA8_EMU_USB_STOP", 0U, view);
274 const uint32_t usbh_stop_settle = guard_env_u32_headless("RA8_EMU_USBH_STOP", 0U, view);
275 const char* const stop_on = guard_read_stop_on(view);
276 guard_apply_stop_pc();
277 uint32_t prof_idle_insns = 0U;
278 uint32_t prof_idle_need = 0U;
279 uint32_t prof_idle_arm = 0U;
280 guard_read_prof_idle(&prof_idle_insns, &prof_idle_need, &prof_idle_arm);
281
282 return (run_guards_t){
283 .click_settle_chunks = click_settle_chunks,
284 .max_chunks = max_chunks,
285 .wall_s = wall_s,
286 .wall_guard_on = wall_guard_on,
287 .idle_stop_chunks = idle_stop_chunks,
288 .usb_stop_settle = usb_stop_settle,
289 .usbh_stop_settle = usbh_stop_settle,
290 .stop_on = stop_on,
291 .prof_idle_insns = prof_idle_insns,
292 .prof_idle_need = prof_idle_need,
293 .prof_idle_arm = prof_idle_arm,
294 };
