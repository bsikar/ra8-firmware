4"""Prove that EVERY code file in this repository is linted and formatted.
5
6WHY THIS EXISTS
7---------------
8"Is everything linted?" was, until this gate, answerable only by opening each
9checker and reading its scan list by hand. That audit was performed five times
10and was wrong five times -- #296, #332, #358, #359, #360. Every one of those
11was the same defect in a different checker: a hardcoded root list that stopped
12matching the tree, so the checker reported a clean run over a subset while
13files outside it sat unchecked for months. A gate that scans nothing reports
14success, and success is indistinguishable from having done the work.
15
16This gate answers the question mechanically:
17
18 1. Enumerate every file from ``git ls-files`` -- never a directory list.
19 A hardcoded directory list is the exact defect being killed here, so this
20 gate must not contain one.
21 2. Classify each file by exact name, then extension, then shebang.
22 3. Ask each checker, in its own "list what you would scan" mode, which files
23 it claims. The gate does NOT restate any checker's scope: a second copy of
24 the coverage map is a new instance of the original bug.
25 4. Assert every CODE file is claimed by at least one linter and at least one
26 formatter.
27 5. FAIL on any file whose type has no classification rule at all.
28
29Point 5 is the one that earns the gate its keep. The day someone commits a
30``.rs``, a ``.ts`` or a ``.proto``, this goes red and somebody has to decide
31how that language is checked -- rather than it entering the tree silently and
32being discovered by the sixth hand audit.
33
34LAYOUT AGNOSTICISM IS A DESIGN REQUIREMENT
35------------------------------------------
36Checkers are located by BASENAME through ``git ls-files``, not by a hardcoded
37path. ``scripts/`` was reorganised into subdirectories in #359 and every
38checker this gate resolves changed directory; the basename lookup carried that
39move without a single edit. A gate that hardcoded full paths would have broken
40on it and -- far worse -- could then have been "fixed" by dropping the
41provider, silently shrinking coverage. By name, a MOVE is invisible and a
42DELETION is loud.
43
44USAGE
45-----
46 check_lint_coverage.py --selftest # assert the gate itself still fires
47 check_lint_coverage.py # the real check
48 check_lint_coverage.py --matrix # print the coverage matrix and exit 0
49"""

152def git_files() -> list[str]:
153 """Every tracked or untracked-but-not-ignored path, repo-relative."""
154 proc = subprocess.run(
155 [ # noqa: S607 -- trusted: fixed git argv
156 "git",
157 "ls-files",
158 "-z",
159 "--cached",
160 "--others",
161 "--exclude-standard",
162 ],
163 cwd=REPO_ROOT,
164 capture_output=True,
165 text=True,
166 check=False,
167 )
168 if proc.returncode != 0:
169 sys.stderr.write(proc.stderr)
170 sys.stderr.write("check_lint_coverage.py: FATAL -- `git ls-files` failed\n")
171 sys.exit(2)
172 return sorted(p for p in proc.stdout.split("\0") if p)

175def read_shebang(rel: str) -> str:
176 """First line of `rel` if it is a shebang, else the empty string."""
177 try:
178 with (REPO_ROOT / rel).open("rb") as handle:
179 first = handle.readline(200)
180 except OSError:
181 return ""
182 if not first.startswith(b"#!"):
183 return ""
184 return first.decode("utf-8", errors="replace").strip()

187def classify(rel: str) -> str | None:
188 """Return the class name for `rel`, or None when nothing claims it.
189
190 Order is exact name, then extension, then shebang. Name beats extension so
191 ``CMakeLists.txt`` is cmake rather than text; shebang comes last so it only
192 rescues files the tables genuinely miss -- which is how an extensionless
193 ``scripts/git/pre-commit`` is recognised as shell.
194 """
195 name = rel.rsplit("/", 1)[-1]
196 if name in NAME_CLASS:
197 return NAME_CLASS[name]
198 suffix = ""
199 if "." in name[1:]:
200 suffix = "." + name.rsplit(".", 1)[-1]
201 if suffix.lower() in EXT_CLASS:
202 return EXT_CLASS[suffix.lower()]
203 line = read_shebang(rel)
204 if line:
205 for token, cls in SHEBANG_CLASS:
206 if token in line:
207 return cls
208 return None

214def resolve_script(basename: str, tracked: list[str]) -> str | None:
215 """Locate a checker by basename anywhere in the tree. None if absent."""
216 hits = [p for p in tracked if p.rsplit("/", 1)[-1] == basename]
217 if len(hits) != EXACTLY_ONE:
218 return None
219 return hits[0]

222def provider_files(prov: Provider, tracked: list[str]) -> tuple[set[str], str | None]:
223 """Run `prov` in list mode. Returns (files, error). Never swallows failure."""
224 path = resolve_script(prov.script, tracked)
225 if path is None:
226 return set(), f"cannot locate {prov.script} (moved, deleted or ambiguous)"
227 runner = shutil.which(prov.runner) or prov.runner
228 proc = subprocess.run( # noqa: S603 -- argv built from the fixed table above
229 [runner, path, *prov.args],
230 cwd=REPO_ROOT,
231 capture_output=True,
232 text=True,
233 check=False,
234 )
235 if proc.returncode != 0:
236 detail = proc.stderr.strip().splitlines()
237 tail = detail[-1] if detail else f"exit {proc.returncode}"
238 return set(), f"{prov.script} --list-files failed: {tail}"
239 return {ln.strip() for ln in proc.stdout.splitlines() if ln.strip()}, None

242# ---------------------------------------------------------------------------
243# The pure evaluation core -- shared by the real run and by --selftest.
244# ---------------------------------------------------------------------------
245class Report:
246 """Outcome of one evaluation."""
247
248 def __init__(self) -> None:
249 """Start an empty report with every failure bucket distinct.
250
251 The buckets are kept separate rather than merged into one findings
252 list because they fail for different reasons and carry different
253 remedies -- an unclassified file type needs a rule, an uncovered file
254 needs a checker, and gap growth needs the gap closed.
255 """
256 self.unclassified: list[str] = []
257 self.uncovered: list[tuple[str, str, str]] = []
258 self.gap_growth: list[str] = []
259 self.gap_sizes: dict[str, int] = {}
260 self.counts: dict[str, int] = {}
261 self.exempt: int = 0
262
263 @property
264 def ok(self) -> bool:
265 """Whether the report is clean across every failing bucket.
266
267 Note ``gap_sizes`` and ``exempt`` are deliberately NOT consulted: a
268 recorded gap of unchanged size is the accepted state, and only its
269 GROWTH is a failure.
270 """
271 return not (self.unclassified or self.uncovered or self.gap_growth)

274def _read_text(rel: str) -> str:
275 """File contents for a gap predicate, empty when unreadable or binary."""
276 try:
277 return (REPO_ROOT / rel).read_text(errors="replace")
278 except OSError:
279 return ""

282def _bucket_gaps(raw: list[tuple[str, str, str]], report: Report) -> list[tuple[str, str, str]]:
283 """Split uncovered pairs into recorded gaps and genuine violations.
284
285 Also runs the ratchet: a gap that has grown past its recorded count is a
286 failure, because "carried deliberately while it is closed" and "quietly
287 becoming permanent" must not look the same.
288 """
289 hits: dict[str, set[str]] = {gap.name: set() for gap in KNOWN_GAPS}
290 violations: list[tuple[str, str, str]] = []
291 text_cache: dict[str, str] = {}
292 for rel, cls, role in raw:
293 if rel not in text_cache:
294 text_cache[rel] = _read_text(rel)
295 ctx = GapCtx(rel=rel, cls=cls, text=text_cache[rel])
296 for gap in KNOWN_GAPS:
297 if gap.match(ctx):
298 hits[gap.name].add(rel)
299 break
300 else:
301 violations.append((rel, cls, role))
302
303 for gap in KNOWN_GAPS:
304 got = len(hits[gap.name])
305 report.gap_sizes[gap.name] = got
306 if got > gap.count:
307 report.gap_growth.append(
308 f"known gap {gap.name!r} ({gap.issue}) grew from {gap.count} to {got} "
309 "file(s). Close it -- raising the recorded count needs a stated reason."
310 )
311 return violations

314def evaluate(files: list[str], claimed: dict[str, set[str]]) -> Report:
315 """Decide coverage for `files` given each provider's claimed set.
316
317 `claimed` maps provider name -> the set of paths that provider scans. The
318 real run fills it from the checkers themselves; --selftest fills it by
319 hand, which is what makes both directions assertable without touching the
320 working tree.
321 """
322 report = Report()
323 lint_by: dict[str, set[str]] = {}
324 fmt_by: dict[str, set[str]] = {}
325 for prov in PROVIDERS:
326 got = claimed.get(prov.name, set())
327 if LINT in prov.roles:
328 lint_by[prov.name] = got
329 if FORMAT in prov.roles:
330 fmt_by[prov.name] = got
331
332 all_lint = set().union(*lint_by.values()) if lint_by else set()
333 all_fmt = set().union(*fmt_by.values()) if fmt_by else set()
334
335 raw: list[tuple[str, str, str]] = []
336 for rel in files:
337 if exemption_reason(rel) is not None:
338 report.exempt += 1
339 continue
340 cls = classify(rel)
341 if cls is None:
342 report.unclassified.append(rel)
343 continue
344 report.counts[cls] = report.counts.get(cls, 0) + 1
345 if CLASSES[cls].kind != "code":
346 continue
347 for role, pool in ((LINT, all_lint), (FORMAT, all_fmt)):
348 if rel not in pool:
349 raw.append((rel, cls, role))
350
351 report.uncovered = _bucket_gaps(raw, report)
352 return report

358def print_matrix(report: Report, claimed: dict[str, set[str]]) -> None:
359 """Print the class-by-provider coverage matrix.
360
361 The human-readable answer to "which checker claims this file type?", which
362 is the question that goes unasked until a whole language turns out to have
363 had no checker at all.
364 """
365 by_class: dict[str, list[str]] = {}
366 for prov in PROVIDERS:
367 for cls in prov.classes:
368 by_class.setdefault(cls, []).append(prov.name)
369 print(f"{'CLASS':<18}{'COUNT':>7} {'KIND':<6} PROVIDERS")
370 print("-" * 78)
371 for cls in sorted(report.counts):
372 spec = CLASSES[cls]
373 provs = ", ".join(by_class.get(cls, [])) or "-- none --"
374 n = report.counts[cls]
375 print(f"{cls:<18}{n:>7} {spec.kind:<6} {provs}")
376 print("-" * 78)
377 total = sum(report.counts.values())
378 print(f"{'total classified':<18}{total:>7}")
379 print(f"{'exempt':<18}{report.exempt:>7}")
380 for prov in PROVIDERS:
381 print(f" scanned by {prov.name:<22} {len(claimed.get(prov.name, set())):>6} file(s)")
382 if report.gap_sizes:
383 print("\nRECORDED GAPS -- code with no checker, held flat by the ratchet:")
384 for gap in KNOWN_GAPS:
385 got = report.gap_sizes.get(gap.name, 0)
386 print(f" {gap.name:<28}{got:>4}/{gap.count:<4} {gap.issue} {gap.reason[:60]}")

389def print_failures(report: Report) -> None:
390 """Print each failing bucket to stderr, capped per bucket.
391
392 Capped because a newly-added file type can produce hundreds of identical
393 findings, and the first few plus a count communicate the same thing
394 without burying the other buckets.
395 """
396 if report.unclassified:
397 print("\nUNCLASSIFIED FILE TYPES -- no rule says how these are checked:", file=sys.stderr)
398 for rel in report.unclassified[:MAX_SHOWN]:
399 print(f" {rel}", file=sys.stderr)
400 extra = len(report.unclassified) - MAX_SHOWN
401 if extra > 0:
402 print(f" ... and {extra} more", file=sys.stderr)
403 print(
404 " Add the type to EXT_CLASS/NAME_CLASS in lint_coverage_rules.py and,\n"
405 " if it is code, wire a linter and a formatter for it.",
406 file=sys.stderr,
407 )
408 if report.uncovered:
409 print(
410 f"\nUNCOVERED CODE FILES -- {len(report.uncovered)} file/role pair(s) "
411 "that no checker claims:",
412 file=sys.stderr,
413 )
414 noun = {LINT: "linter", FORMAT: "formatter"}
415 for rel, cls, role in report.uncovered[:MAX_SHOWN]:
416 print(f" {rel} [{cls}] has no {noun[role]}", file=sys.stderr)
417 extra = len(report.uncovered) - MAX_SHOWN
418 if extra > 0:
419 print(f" ... and {extra} more", file=sys.stderr)
420 for msg in report.gap_growth:
421 print(f"\nGAP RATCHET: {msg}", file=sys.stderr)

461def _assert_quiet(files: list[str], claimed: dict[str, set[str]], failures: list[str]) -> None:
462 """Assert the model stays silent on trees that are genuinely covered.
463
464 Split out along the QUIET / MUST-FIRE boundary this suite already drew in
465 comments: the two directions share only the fixture, and a reader checking
466 "does a covered tree pass?" should not have to step over the fires cases.

469 expect(base.ok, "a fully-covered tree passes", failures)

474 )
475
476 plus = [*files, "libs/ra8_core/src/ra8_new.c"]
477 claimed2 = {k: set(v) for k, v in claimed.items()}
478 claimed2["clang-tidy"].add("libs/ra8_core/src/ra8_new.c")
479 claimed2["clang-format"].add("libs/ra8_core/src/ra8_new.c")
480 expect(
481 evaluate(plus, claimed2).ok,
482 "a new file of a covered type in a covered dir stays quiet",
483 failures,
484 )

562def selftest() -> int:
563 """Prove the coverage model both fires and stays quiet, against fixtures.
564
565 Validates the classification tables first: a rule keyed on a class no
566 provider claims, or a provider claiming a class that does not exist,
567 makes every later answer meaningless.
568
569 Returns 0 when both directions hold, 1 otherwise.
570 """
571 print("check_lint_coverage.py --selftest")
572 failures: list[str] = []
573
574 problems = validate_tables()
575 expect(not problems, f"classification tables self-consistent ({problems})", failures)
576
577 files, claimed = _fixture()
578 _assert_quiet(files, claimed, failures)
579 _assert_fires(files, claimed, failures)
580 _assert_ratchet(files, claimed, failures)
581
582 return report(failures)

585def run_check(show_matrix: bool) -> int:
586 """Verify every tracked code file is claimed by at least one checker.
587
588 Enforces a FILE FLOOR before anything else and exits 2 below it. That is
589 the load-bearing part: if the enumeration collapses, every file is
590 trivially covered and the gate reports perfect coverage precisely because
591 it saw nothing -- the exact failure mode it exists to detect in others.
592
593 Returns 0 when every file is covered, 1 on a coverage failure, 2 when the
594 enumeration is too small to trust.
595 """
596 tracked = git_files()
597 if len(tracked) < FILE_FLOOR:
598 sys.stderr.write(
599 f"check_lint_coverage.py: FATAL -- only {len(tracked)} file(s) enumerated, "
600 f"floor is {FILE_FLOOR}.\n"
601 " A collapsed enumeration reports full coverage because it saw nothing.\n"
602 )
603 return 2
604
605 claimed: dict[str, set[str]] = {}
606 errors: list[str] = []
607 for prov in PROVIDERS:
608 got, err = provider_files(prov, tracked)
609 if err:
610 errors.append(f"{prov.name}: {err}")
611 claimed[prov.name] = got
612 if errors:
613 sys.stderr.write("check_lint_coverage.py: FATAL -- provider enumeration failed.\n")
614 for err in errors:
615 sys.stderr.write(f" {err}\n")
616 sys.stderr.write(
617 " A provider that cannot report its scope leaves coverage unknown;\n"
618 " unknown is a failure, never a pass.\n"
619 )
620 return 2
621
622 report = evaluate(tracked, claimed)
623 if show_matrix or report.ok:
624 print_matrix(report, claimed)
625 if report.ok:
626 held = sum(report.gap_sizes.values())
627 if held:
