68RA8_PRIV void* ra8_c6link_priv_arena_alloc(void* ctx, size_t size)
69{
70 ra8_c6link_t* link = (ra8_c6link_t*)ctx;
71 if ((link == nullptr) || (link->arena == nullptr)) {
72 return nullptr;
73 }
74
75 /* `arena_used` never exceeds `arena_bytes`, so this cannot underflow, and
76 every later comparison stays inside the remaining space -- which is what
77 lets the rounding below be checked without an overflow test that no
78 reachable input could ever make true. */
79 const uint32_t avail = link->arena_bytes - link->arena_used;
80 if (size > (size_t)avail) {
81 return nullptr;
82 }
83 const uint32_t want = (uint32_t)size;
84 const uint32_t pad = (uint32_t)((0U - want) & (uint32_t)k_ra8_c6link_arena_mask);
85 if (pad > (avail - want)) {
86 return nullptr;
87 }
88
89 const uint32_t at = link->arena_used;
90 link->arena_used = at + want + pad;
91 link->arena_last = at + 1U;
92 return &link->arena[at];
93}

95RA8_PRIV void ra8_c6link_priv_arena_free(void* ctx, void* pointer)
96{
97 ra8_c6link_t* link = (ra8_c6link_t*)ctx;
98 if ((link == nullptr) || (pointer == nullptr) || (link->arena_last == 0U)) {
99 return;
100 }
101 const uint32_t at = link->arena_last - 1U;
102 if (pointer == &link->arena[at]) {
103 link->arena_used = at;
104 link->arena_last = 0U;
105 }
106}
