67static ra8_err_t internal_claim(ra8_port_pin_t pin, ra8_port_t* out_port, ra8_pin_t* out_pin)
68{
69 const ra8_port_t p = RA8_PIN_PORT(pin);
70 const ra8_pin_t b = RA8_PIN_PIN(pin);
71 if ((uint8_t)p > k_ra8_port_max) {
72 return k_ra8_err_gpio_invalid_port;
73 }
74 if ((uint8_t)b > k_ra8_pin_max) {
75 return k_ra8_err_gpio_invalid_pin;
76 }
77 const ra8_err_t err = ra8_pin_validator_claim(pin, s_tag);
78 if (err != k_ra8_ok) {
79 return err;
80 }
81 *out_port = p;
82 *out_pin = b;
83 return k_ra8_ok;
84}

86ra8_err_t ra8_gpio_output_init(ra8_port_pin_t pin, ra8_level_t init_level)
87{
88 ra8_port_t port = k_ra8_port_0;
89 ra8_pin_t bit = k_ra8_pin_0;
90
91 ra8_err_t err = internal_claim(pin, &port, &bit);
92 if (err != k_ra8_ok) {
93 ra8_log_error_val(s_tag, "claim failed", (uint32_t)err);
94 return err;
95 }
96
97 /* Programme PFS: PMR=0 (GPIO), PDR=1 (output), PODR=init level. */
98 volatile uint32_t* pfs = ra8_pfs_pmn(port, bit);
99 if (pfs == nullptr) {
100 /* The claim is already taken by internal_claim above, so it has to be
101 * handed back before bailing out -- otherwise the pin stays owned by a
102 * call that configured nothing, and every later claim of it fails. This
103 * mirrors ra8_pfs_route_peripheral, which releases on the same branch. */
104 (void)ra8_pin_validator_release(pin);
105 return k_ra8_err_hw_unmapped;
106 }
107
108 const uint32_t new_val =
109 (uint32_t)((init_level == k_ra8_level_high) ? k_ra8_pfs_mask_podr : 0U) | k_ra8_pfs_mask_pdr;
110
111 ra8_pfs_pwpr_unlock();
112 *pfs = new_val;
113 ra8_pfs_pwpr_lock();
114
115 ra8_log_info_val(s_tag, "output init pin", (uint32_t)pin);
116 return k_ra8_ok;
117}

119ra8_err_t ra8_gpio_input_init(ra8_port_pin_t pin, ra8_pin_pull_t pull)
120{
121 ra8_port_t port = k_ra8_port_0;
122 ra8_pin_t bit = k_ra8_pin_0;
123
124 ra8_err_t err = internal_claim(pin, &port, &bit);
125 if (err != k_ra8_ok) {
126 ra8_log_error_val(s_tag, "claim failed", (uint32_t)err);
127 return err;
128 }
129
130 volatile uint32_t* pfs = ra8_pfs_pmn(port, bit);
131 if (pfs == nullptr) {
132 /* Same ownership hand-back as ra8_gpio_output_init: internal_claim has
133 * already taken the pin, so returning without releasing would strand it. */
134 (void)ra8_pin_validator_release(pin);
135 return k_ra8_err_hw_unmapped;
136 }
137
138 /* PMR=0 (GPIO), PDR=0 (input), PCR according to pull. */
139 uint32_t new_val = 0U;
140 if (pull == k_ra8_pull_up) {
141 new_val |= k_ra8_pfs_mask_pcr;
142 }
143
144 ra8_pfs_pwpr_unlock();
145 *pfs = new_val;
146 ra8_pfs_pwpr_lock();
147
148 ra8_log_info_val(s_tag, "input init pin", (uint32_t)pin);
149 return k_ra8_ok;
150}

152ra8_err_t ra8_gpio_write(ra8_port_pin_t pin, ra8_level_t level)
153{
154 const ra8_port_t port = RA8_PIN_PORT(pin);
155 const ra8_pin_t bit = RA8_PIN_PIN(pin);
156 if ((uint8_t)port > k_ra8_port_max) {
157 return k_ra8_err_gpio_invalid_port;
158 }
159 if ((uint8_t)bit > k_ra8_pin_max) {
160 return k_ra8_err_gpio_invalid_pin;
161 }
162
163 volatile r_port_regs_t* port_regs = ra8_port(port);
164 if (port_regs == nullptr) {
165 return k_ra8_err_hw_unmapped;
166 }
167
168 /* PCNTR3 high half = PORR (clear), low half = POSR (set). Writing
169 * a 1 to the relevant half drives the pin; writing 0 leaves other
170 * pins alone -- race-free. */
171 const uint32_t bit_mask = (uint32_t)(1UL << (uint32_t)bit);
172 if (level == k_ra8_level_high) {
173 port_regs->PCNTR3 = bit_mask; /* POSR in low half. */
174 } else {
175 port_regs->PCNTR3 = bit_mask << (uint32_t)k_ra8_pcntr_high_half_shift;
176 }
177 return k_ra8_ok;
178}

180ra8_err_t ra8_gpio_toggle(ra8_port_pin_t pin)
181{
182 const ra8_port_t port = RA8_PIN_PORT(pin);
183 const ra8_pin_t bit = RA8_PIN_PIN(pin);
184 if ((uint8_t)port > k_ra8_port_max) {
185 return k_ra8_err_gpio_invalid_port;
186 }
187 if ((uint8_t)bit > k_ra8_pin_max) {
188 return k_ra8_err_gpio_invalid_pin;
189 }
190
191 volatile r_port_regs_t* port_regs = ra8_port(port);
192 if (port_regs == nullptr) {
193 return k_ra8_err_hw_unmapped;
194 }
195
196 /* Read PODR (high half of PCNTR1) and invert the bit. */
197 const uint32_t pcntr1_val = port_regs->PCNTR1;
198 const uint32_t podr = (pcntr1_val >> (uint32_t)k_ra8_pcntr_high_half_shift);
199 const uint32_t bit_mask = (uint32_t)(1UL << (uint32_t)bit);
200 if ((podr & bit_mask) != 0U) {
201 port_regs->PCNTR3 = bit_mask << (uint32_t)k_ra8_pcntr_high_half_shift; /* PORR clear */
202 } else {
203 port_regs->PCNTR3 = bit_mask; /* POSR set */
204 }
205 return k_ra8_ok;
206}

236ra8_err_t ra8_pfs_route_peripheral(ra8_port_pin_t pin, ra8_psel_t psel, const char* owner)
237{
238 RA8_CHECK_NULL_PTR(owner, s_tag, "owner must not be nullptr");
239
240 const ra8_port_t port = RA8_PIN_PORT(pin);
241 const ra8_pin_t bit = RA8_PIN_PIN(pin);
242 if ((uint8_t)port > k_ra8_port_max) {
243 return k_ra8_err_gpio_invalid_port;
244 }
245 if ((uint8_t)bit > k_ra8_pin_max) {
246 return k_ra8_err_gpio_invalid_pin;
247 }
248
249 const ra8_err_t claim = ra8_pin_validator_claim(pin, owner);
250 if (claim != k_ra8_ok) {
251 ra8_log_error_val(s_tag, "peripheral claim failed", (uint32_t)claim);
252 return claim;
253 }
254
255 volatile uint32_t* pfs = ra8_pfs_pmn(port, bit);
256 if (pfs == nullptr) {
257 (void)ra8_pin_validator_release(pin);
258 return k_ra8_err_hw_unmapped;
259 }
260
261 /* HUM Ch 20.2.4 "Notes on the PmnPFS Register Setting" p 859 -- PMR
262 * MUST be cleared before specifying a new PSEL, otherwise the pin can
263 * glitch through whatever the previous peripheral function happened
264 * to be while the new PSEL is decoded. The correct sequence is:
265 * 1. PMR := 0 (force GPIO mode)
266 * 2. PFS := PSEL only, PMR still 0 (programme the new function)
267 * 3. PFS := PSEL | PMR=1 (hand the pin to the peripheral)
268 * @note PMSAR defaults to all-Secure on RA8D2 reset, so writing PWPR
269 * (NS) and PWPRS (S) unconditionally is harmless: the chip
270 * silently drops whichever side does not own the port. */
271 const uint32_t psel_only = (((uint32_t)psel) << (uint32_t)k_ra8_pfs_bit_psel0);
272 const uint32_t new_val = k_ra8_pfs_mask_pmr | psel_only;
273
274 ra8_pfs_pwpr_unlock();
275 /* HUM Ch 20.2.4 "Notes on the PmnPFS Register Setting" p 859 */
276 *pfs = 0U; /* step 1: PMR = 0. */
277 *pfs = psel_only; /* step 2: PSEL with PMR=0. */
278 *pfs = new_val; /* step 3: PSEL | PMR=1. */
279 ra8_pfs_pwpr_lock();
280
281 ra8_log_info_val(s_tag, "peripheral route pin", (uint32_t)pin);
282 return k_ra8_ok;
283}

285ra8_err_t ra8_pfs_set_drive_strength(ra8_port_pin_t pin, ra8_pfs_dscr_t dscr)
286{
287 const ra8_port_t port = RA8_PIN_PORT(pin);
288 const ra8_pin_t bit = RA8_PIN_PIN(pin);
289 if ((uint8_t)port > k_ra8_port_max) {
290 return k_ra8_err_gpio_invalid_port;
291 }
292 if ((uint8_t)bit > k_ra8_pin_max) {
293 return k_ra8_err_gpio_invalid_pin;
294 }
295
296 volatile uint32_t* pfs = ra8_pfs_pmn(port, bit);
297 if (pfs == nullptr) {
298 return k_ra8_err_hw_unmapped;
299 }
300
301 /* Read-modify-write only the DSCR[1:0] field; PSEL / PMR / direction
302 * are preserved so an already-routed pin keeps its function.
303 * HUM Ch 20.2.4 "Notes on the PmnPFS Register Setting" p 859 -- the
304 * PMR-clear dance is required only when PSEL changes; a DSCR-only
305 * update of an in-function pin does not retouch PSEL. */
306 const uint32_t dscr_field =
307 ((uint32_t)dscr << (uint32_t)k_ra8_pfs_bit_dscr0) & (uint32_t)k_ra8_pfs_mask_dscr;
308 ra8_pfs_pwpr_unlock();
309 *pfs = (*pfs & ~(uint32_t)k_ra8_pfs_mask_dscr) | dscr_field;
310 ra8_pfs_pwpr_lock();
311
312 ra8_log_info_val(s_tag, "pin drive strength", (uint32_t)pin);
313 return k_ra8_ok;
314}

339ra8_err_t ra8_gpio_attach_irq(ra8_port_pin_t pin,
340 uint8_t irq_num,
341 const ra8_gpio_irq_cfg_t* cfg,
342 ra8_isr_handler_t handler,
343 void* ctx)
344{
345 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
346 RA8_CHECK_NULL_PTR(handler, s_tag, "handler must not be nullptr");
347 if (irq_num > k_ra8_gpio_irq_num_max) {
348 return k_ra8_err_invalid_arg;
349 }
350
351 ra8_err_t err = ra8_gpio_input_init(pin, cfg->pull);
352 if (err != k_ra8_ok) {
353 return err;
354 }
355
356 const ra8_icu_irq_cfg_t icu_cfg = {
357 .sense = cfg->sense,
358 .filter_div = cfg->filter_div,
359 .filter_en = cfg->filter_en,
360 };
361 err = ra8_icu_configure_irq_pin(irq_num, &icu_cfg);
362 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE -- irq_num bounded */
363 /* GCOVR_EXCL_START */
364 (void)ra8_pin_validator_release(pin);
365 return err;
366 /* GCOVR_EXCL_STOP */
367 }
368
369 const ra8_elc_event_t event = internal_event_for_irq(irq_num);
370 err = ra8_isr_register(event, handler, ctx, cfg->priority, nullptr);
371 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE -- fresh reset_irq_state */
372 /* GCOVR_EXCL_START */
373 const ra8_icu_irq_cfg_t zero_cfg = {
374 .sense = (ra8_icu_irqmd_t)0U,
375 .filter_div = (ra8_icu_fclksel_t)0U,
376 .filter_en = false,
377 };
378 (void)ra8_icu_configure_irq_pin(irq_num, &zero_cfg);
379 (void)ra8_pin_validator_release(pin);
380 return err;
381 /* GCOVR_EXCL_STOP */
382 }
383
384 ra8_log_info_val(s_tag, "attach irq pin", (uint32_t)pin);
385 return k_ra8_ok;
386}
