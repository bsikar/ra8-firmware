401static UINT ospirw_msc_read(VOID* storage,
402 ULONG lun,
403 UCHAR* data_pointer,
404 ULONG number_blocks,
405 ULONG lba,
406 ULONG* media_status)
407{
408 (void)storage;
409 (void)lun;
410 s_dbg_read_calls++;
411 if ((lba + number_blocks) > (ULONG)k_ospirw_sectors) {
412 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
413 k_scsi_asc_lba_out_of_range,
414 k_scsi_ascq_none);
415 return UX_ERROR;
416 }
417 const uint32_t addr = (uint32_t)k_ospirw_offset + ((uint32_t)lba * (uint32_t)k_ospirw_block_size);
418 const ra8_err_t rerr =
419 ra8_xspi_flash_read((uint8_t)k_ospirw_instance,
420 addr,
421 data_pointer,
422 (uint32_t)number_blocks * (uint32_t)k_ospirw_block_size);
423 if (rerr != k_ra8_ok) {
424 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
425 k_scsi_asc_lba_out_of_range,
426 k_scsi_ascq_none);
427 return UX_ERROR;
428 }
429 *media_status = 0UL;
430 (void)ra8_board_led_toggle(k_ra8_board_led1);
431 return UX_SUCCESS;
432}

640/* The OSPI bring-up + write-window pre-erase step (::ospirw_ospi_provision)
641 * lives in usb_selftest_ospi_rw_device.c, alongside the s_dbg_ospi_prov /
642 * s_dbg_ospi_id probes that record its outcome. */
643
662static VOID ospirw_device_worker(ULONG arg)
663{
664 (void)arg;
665
666 if (ospirw_ospi_provision() != k_ra8_ok) {
667 return; /* s_dbg_ospi_prov records the failure; host READ/WRITE will FAIL */
668 }
669 UINT ux = ospirw_usbx_stack_up();
670 if (ux != UX_SUCCESS) {
671 s_dbg_dev_err = (uint32_t)ux;
672 return;
673 }
674 s_dbg_dev_step = (uint32_t)k_ospirw_dev_step_stack;
675 ux = ospirw_class_register();
676 if (ux != UX_SUCCESS) {
677 s_dbg_dev_err = (uint32_t)ux;
678 return;
679 }
680 s_dbg_dev_step = (uint32_t)k_ospirw_dev_step_class;
681 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
682 if (e != k_ra8_ok) {
683 s_dbg_dev_err = (uint32_t)e;
684 return;
685 }
686 s_dbg_dev_step = (uint32_t)k_ospirw_dev_step_dcd;
687 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
688 if (e != k_ra8_ok) {
689 s_dbg_dev_err = (uint32_t)e;
690 return;
691 }
692 s_dbg_dev_step = (uint32_t)k_ospirw_dev_step_attach;
693
694 while (1) {
695 s_dbg_dev_step = (uint32_t)k_ospirw_dev_step_parked;
696 tx_thread_sleep(k_ospirw_idle_ticks);
697 }
698}
