179static uint32_t pc_crc32(const uint8_t* data, size_t len)
180{
181 uint32_t crc = (uint32_t)k_pc_crc_init;
182 for (size_t i = 0U; i < len; i++) {
183 crc ^= (uint32_t)data[i];
184 for (uint32_t bit = 0U; bit < (uint32_t)k_pc_crc_bits; bit++) {
185 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
186 crc = (crc >> 1U) ^ ((uint32_t)k_pc_crc_poly & mask);
187 }
188 }
189 return ~crc;
190}

412static void pc_reload_check_or_halt(uint32_t font_len, size_t n)
413{
414 if (!pc_engine_init((uint16_t)k_pc_font_px, font_len)) {
415 pc_fail((uint32_t)k_pc_err_layout);
416 }
417 if (ra8_reflow_cache_load(&s_engine,
418 (const uint8_t*)k_pc_body,

420 s_read_blob,

422 pc_fail((uint32_t)k_pc_err_load);
423 }
424 size_t n2 = 0U;
425 if (ra8_reflow_cache_serialize(&s_engine,
426 (const uint8_t*)k_pc_body,
427 (size_t)(sizeof(k_pc_body) - 1U),
428 s_reser_blob,

469int32_t main(void)
470{
471 uint32_t pclka_hz = 0U;
472 pc_setup_or_halt(&pclka_hz);
473 ra8_isr_globals_enable();
474 ra8_log_init();
475 pc_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
476
477 pc_init_card_or_halt(&pclka_hz);
478 ra8_fs_mount_t* const mount = pc_mount_or_halt();
479
480 /* Reset-survival: a prior boot's persisted cache file is the hit signal. */
481 g_pc_hit = (pc_read_cache(mount, s_read_blob, (uint32_t)k_pc_blob_cap) != 0U) ? 1U : 0U;
482
483 uint32_t font_len = 0U;
484 pc_font_or_halt(mount, &font_len);
485
486 size_t n = 0U;
487 pc_live_layout_or_halt(font_len, &n);
488 pc_persist_and_verify_or_halt(mount, n);
489 pc_reload_check_or_halt(font_len, n);
490 pc_invalidate_check_or_halt(font_len, n);
491
492 pc_print(k_msg_pass, (uint32_t)sizeof(k_msg_pass) - 1U);
493 pc_print((const uint8_t*)(g_pc_hit != 0U ? "1" : "0"), 1U);
494 pc_print(k_msg_ok, (uint32_t)sizeof(k_msg_ok) - 1U);
495
496 /* Idle with a heartbeat that advances ONLY here -- every failure path parks
497 * in WFI without bumping it -- so a memprobe gate proves the full SD -> ra8_fs
498 * -> ra8_reflow_cache round-trip + invalidation ran clean. */
499 while (1) {
500 ra8_delay_ms(k_pc_frame_ms);
501 g_pc_heartbeat++;
502 }
503}
