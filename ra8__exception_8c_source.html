123void ra8_exception_capture_diagnostics(ra8_exception_diagnostics_t* out)
124{
125 if (out == nullptr) {
126 return;
127 }
128 out->cfsr = internal_read32(k_ra8_scb_cfsr_addr);
129 out->hfsr = internal_read32(k_ra8_scb_hfsr_addr);
130 out->dfsr = internal_read32(k_ra8_scb_dfsr_addr);
131 out->mmfar = internal_read32(k_ra8_scb_mmfar_addr);
132 out->bfar = internal_read32(k_ra8_scb_bfar_addr);
133 out->afsr = internal_read32(k_ra8_scb_afsr_addr);
134 /* TrustZone SecureFault pair: banked to the Secure state. Reads from
135 * the Secure world (every boot here) return the real cause/address;
136 * reads from the Non-secure world are architecturally RAZ -- never a
137 * fault -- so no world guard is needed. */
138 out->sfsr = internal_read32(k_ra8_scb_sfsr_addr);
139 out->sfar = internal_read32(k_ra8_scb_sfar_addr);
140}

298void ra8_exception_report(const ra8_exception_frame_t* frame, uint32_t exc_number)
299{
300 /* Step 1: capture EVERYTHING into fixed SRAM FIRST -- before any
301 * function call that might itself fault. This guarantees that even
302 * if the log backend, ITM, or anything else takes a secondary fault,
303 * a debugger can still recover the original fault context from
304 * g_ra8_exception_last. */
305 g_ra8_exception_last.magic = 0U;
306 g_ra8_exception_last.exc_number = exc_number;
307 g_ra8_exception_last.frame_ptr = (uintptr_t)frame;
308 if (frame != nullptr) {
309 g_ra8_exception_last.frame.r0 = frame->r0;
310 g_ra8_exception_last.frame.r1 = frame->r1;
311 g_ra8_exception_last.frame.r2 = frame->r2;
312 g_ra8_exception_last.frame.r3 = frame->r3;
313 g_ra8_exception_last.frame.r12 = frame->r12;
314 g_ra8_exception_last.frame.lr = frame->lr;
315 g_ra8_exception_last.frame.pc = frame->pc;
316 g_ra8_exception_last.frame.xpsr = frame->xpsr;
317 }
318 ra8_exception_diagnostics_t diag = {};
319 ra8_exception_capture_diagnostics(&diag);
320 g_ra8_exception_last.diag = diag;
321 /* Fold in the ICU cause staged by ra8_exception_report_nmi(), then
322 * clear the stage so a later non-NMI record cannot inherit it. */
323 g_ra8_exception_last.nmisr = s_ra8_exception_nmi_stage;
324 s_ra8_exception_nmi_stage = 0U;
325 g_ra8_exception_last.magic = (uint32_t)k_ra8_exc_magic_valid;
326
327 /* Step 1b: persist the completed snapshot across the coming reset (if a
328 * sink is armed). Runs before any code that might take a secondary fault
329 * so a field unit's post-mortem + reset-loop count survive to the next
330 * boot. A plain SRAM copy -- fault-context safe. See ra8_crashlog.c. */
331 if (s_ra8_exception_persist != nullptr) {
332 s_ra8_exception_persist(&g_ra8_exception_last);
333 }
334
335 /* Step 2: best-effort logging, strictly after the snapshot. */
336 internal_log_fault_dump(frame, exc_number, &diag, g_ra8_exception_last.nmisr);
337
338 /* Step 3: halt at a named symbol. On target we do NOT call
339 * internal_ra8_fatal_error from a fault context -- that path issues
340 * `bkpt #0`, which on a board without an attached debugger re-enters
341 * HardFault and can escalate to LOCKUP at PC=0xEFFFFFFE. The
342 * dedicated halt loop parks PC at a symbol the debugger can name.
343 * On host (fake) builds we still go through the overridable
344 * fatal hook so unit tests can longjmp out. */
345 RA8_EXCEPTION_HALT("EXC", "fault", exc_number);
346}

349void ra8_exception_report_nmi(const ra8_exception_frame_t* frame, uint32_t nmisr)
350{
351 /* Plain SRAM store first (cannot fault), so even a secondary fault
352 * inside the common path leaves the cause recoverable: the stage is
353 * copied into g_ra8_exception_last.nmisr before magic is set. */
354 s_ra8_exception_nmi_stage = nmisr;
355 ra8_exception_report(frame, (uint32_t)k_ra8_exc_num_nmi);
356}
