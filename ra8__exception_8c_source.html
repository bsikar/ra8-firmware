268void ra8_exception_report(const ra8_exception_frame_t* frame, uint32_t exc_number)
269{
270 /* Step 1: capture EVERYTHING into fixed SRAM FIRST -- before any
271 * function call that might itself fault. This guarantees that even
272 * if the log backend, ITM, or anything else takes a secondary fault,
273 * a debugger can still recover the original fault context from
274 * g_ra8_exception_last. */
275 g_ra8_exception_last.magic = 0U;
276 g_ra8_exception_last.exc_number = exc_number;
277 g_ra8_exception_last.frame_ptr = (uintptr_t)frame;
278 if (frame != nullptr) {
279 g_ra8_exception_last.frame.r0 = frame->r0;
280 g_ra8_exception_last.frame.r1 = frame->r1;
281 g_ra8_exception_last.frame.r2 = frame->r2;
282 g_ra8_exception_last.frame.r3 = frame->r3;
283 g_ra8_exception_last.frame.r12 = frame->r12;
284 g_ra8_exception_last.frame.lr = frame->lr;
285 g_ra8_exception_last.frame.pc = frame->pc;
286 g_ra8_exception_last.frame.xpsr = frame->xpsr;
287 }
288 ra8_exception_diagnostics_t diag = {};
289 ra8_exception_capture_diagnostics(&diag);
290 g_ra8_exception_last.diag = diag;
291 /* Fold in the ICU cause staged by ra8_exception_report_nmi(), then
292 * clear the stage so a later non-NMI record cannot inherit it. */
293 g_ra8_exception_last.nmisr = s_ra8_exception_nmi_stage;
294 s_ra8_exception_nmi_stage = 0U;
295 g_ra8_exception_last.magic = (uint32_t)k_ra8_exc_magic_valid;
296
297 /* Step 1b: persist the completed snapshot across the coming reset (if a
298 * sink is armed). Runs before any code that might take a secondary fault
299 * so a field unit's post-mortem + reset-loop count survive to the next
300 * boot. A plain SRAM copy -- fault-context safe. See ra8_crashlog.c. */
301 if (s_ra8_exception_persist != nullptr) {
302 s_ra8_exception_persist(&g_ra8_exception_last);
303 }
304
305 /* Step 2: best-effort logging, strictly after the snapshot. */
306 internal_log_fault_dump(frame, exc_number, &diag, g_ra8_exception_last.nmisr);
307
308 /* Step 3: halt at a named symbol. On target we do NOT call
309 * internal_ra8_fatal_error from a fault context -- that path issues
310 * `bkpt #0`, which on a board without an attached debugger re-enters
311 * HardFault and can escalate to LOCKUP at PC=0xEFFFFFFE. The
312 * dedicated halt loop parks PC at a symbol the debugger can name.
313 * On host (fake) builds we still go through the overridable
314 * fatal hook so unit tests can longjmp out. */
315 RA8_EXCEPTION_HALT("EXC", "fault", exc_number);
316}

319void ra8_exception_report_nmi(const ra8_exception_frame_t* frame, uint32_t nmisr)
320{
321 /* Plain SRAM store first (cannot fault), so even a secondary fault
322 * inside the common path leaves the cause recoverable: the stage is
323 * copied into g_ra8_exception_last.nmisr before magic is set. */
324 s_ra8_exception_nmi_stage = nmisr;
325 ra8_exception_report(frame, (uint32_t)k_ra8_exc_num_nmi);
326}
