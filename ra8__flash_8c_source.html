386ra8_err_t ra8_flash_init(const ra8_flash_cfg_t* cfg)
387{
388 RA8_CHECK_NULL_PTR(cfg, s_flash_tag, "cfg must not be nullptr");
389 if (cfg->mrcfreq_mhz > (uint16_t)k_ra8_flash_max_mrcfreq_mhz) {
390 return k_ra8_err_invalid_arg;
391 }
392 if (cfg->mrefreq_mhz > (uint8_t)k_ra8_flash_max_mrefreq_mhz) {
393 return k_ra8_err_invalid_arg;
394 }
395
396 /* Frequency-down procedure -- HUM Ch 59.4.3 Figure 59.6 p 3550. */
397 ra8_flash_internal_set_prefetch(false);
398
399 /* HUM Ch 59.5.2 "MRCFREQ : Code MRAM Frequency Notifications Register" p 3551 */
400 const uint32_t mrcfreq_word =
401 (k_ra8_flash_mrcfreq_key << k_ra8_flash_freq_key_shift) | (uint32_t)cfg->mrcfreq_mhz;
402 *ra8_mram_reg32(k_ra8_mram_off_mrcfreq) = mrcfreq_word;
403
404 /* HUM Ch 59.5.3 "MREFREQ : Extra MRAM Frequency Notifications Register" p 3552 */
405 const uint32_t mrefreq_word =
406 (k_ra8_flash_mrefreq_key << k_ra8_flash_freq_key_shift) | (uint32_t)cfg->mrefreq_mhz;
407 *ra8_mram_reg32(k_ra8_mram_off_mrefreq) = mrefreq_word;
408
409 /* HUM Ch 59 "MRCEECC : Code MRAM ECC Encoder Control" p 3624 */
410 uint16_t mrceecc_bits = 0U;
411 if (cfg->ecc_encoder_enable) {
412 mrceecc_bits = k_ra8_mrceecc_mask_eccen;
413 }
414 *ra8_mram_reg16(k_ra8_mram_off_mrceecc) = (uint16_t)(k_ra8_mrceecc_key_shift | mrceecc_bits);
415
416 /* HUM Ch 59 "MRCDECC : Code MRAM ECC Decoder Control" p 3554 */
417 uint16_t mrcdecc_bits = 0U;
418 if (cfg->ecc_decoder_enable) {
419 mrcdecc_bits = k_ra8_mrcdecc_mask_dececen;
420 }
421 *ra8_mram_reg16(k_ra8_mram_off_mrcdecc) = (uint16_t)(k_ra8_mrcdecc_key_shift | mrcdecc_bits);
422
423 ra8_flash_internal_set_prefetch(cfg->prefetch_en);
424
425 /* Lock both program gates so no stray store can program MRAM. */
426 internal_set_program_gate(k_ra8_flash_world_ns, false);
427 internal_set_program_gate(k_ra8_flash_world_s, false);
428 internal_set_hsp_mode(false);
429
430 /* Clear sticky errors so the new run starts from a clean state. */
431 /* HUM Ch 59 "MRCPS : Code MRAM Program Status Register" p 3601 */
432 *ra8_mram_reg8(k_ra8_mram_off_mrcps) = k_ra8_mrcps_mask_errors;
433 /* HUM Ch 59 "MRCRAES : Code MRAM Read Access Error Status" p 3554 */
434 *ra8_mram_reg8(k_ra8_mram_off_mrcraes) = 0U;
435 /* HUM Ch 59 "MRERAES : Extra MRAM Read Access Error Status" p 3557 */
436 *ra8_mram_reg8(k_ra8_mram_off_mreraes) = 0U;
437
438 s_flash_rt.initialized = true;
439 s_flash_rt.cb = nullptr;
440 s_flash_rt.user_ctx = nullptr;
441
442 ra8_log_info_val(s_flash_tag, "flash_init mrcfreq", (uint32_t)cfg->mrcfreq_mhz);
443 return k_ra8_ok;
444}

446ra8_err_t ra8_flash_deinit(void)
447{
448 /* Lock everything, clear sticky errors, re-enable prefetch. */
449 internal_set_program_gate(k_ra8_flash_world_ns, false);
450 internal_set_program_gate(k_ra8_flash_world_s, false);
451 internal_set_hsp_mode(false);
452 /* If we accidentally left P/E mode on, drop back to read mode. */
453 /* HUM Ch 59 "MENTRYR : Extra MRAM Program-Mode Entry" p 3582 */
454 *ra8_mram_reg16(k_ra8_mram_off_mentryr) = k_ra8_mentryr_read_mode;
455 /* HUM Ch 59 "MRCPS : Code MRAM Program Status Register" p 3601 */
456 *ra8_mram_reg8(k_ra8_mram_off_mrcps) = k_ra8_mrcps_mask_errors;
457 ra8_flash_internal_set_prefetch(true);
458
459 s_flash_rt.initialized = false;
460 s_flash_rt.cb = nullptr;
461 s_flash_rt.user_ctx = nullptr;
462 return k_ra8_ok;
463}

622static ra8_err_t internal_validate_write_block(uint32_t mram_addr, uint32_t len)
623{
624 if (len == 0U || len > k_ra8_mram_write_size_bytes) {
625 return k_ra8_err_invalid_arg;
626 }
627 const uint32_t end_excl =
628 (uint32_t)((uint64_t)mram_addr + (uint64_t)len); /* avoids 32-bit overflow */
629 if (mram_addr < (uint32_t)k_ra8_flash_code_start) {
630 return k_ra8_err_invalid_arg;
631 }
632 if (end_excl > (uint32_t)k_ra8_flash_code_start + (uint32_t)k_ra8_flash_code_size) {
633 return k_ra8_err_invalid_arg;
634 }
635 const uint32_t page_mask = k_ra8_mram_write_size_bytes - 1U;
636 if ((mram_addr & ~page_mask) != ((end_excl - 1U) & ~page_mask)) {
637 return k_ra8_err_invalid_arg;
638 }
639 if (!ra8_flash_internal_window_allows((uintptr_t)mram_addr, len)) {
640 return k_ra8_err_out_of_range;
641 }
642 return k_ra8_ok;
643}

677static ra8_err_t internal_flash_program_window(uint32_t mram_addr,
678 const uint8_t* src,
679 uint32_t len,
680 ra8_flash_world_t world)
681{
682 ra8_flash_internal_set_prefetch(false);
683 internal_set_hsp_mode(true);
684 internal_set_program_gate(world, true);
685
686 volatile uint8_t* dst = (volatile uint8_t*)(uintptr_t)mram_addr;
687 for (uint32_t i = 0U; i < len; ++i) {
688 dst[i] = src[i];
689 }
690
691 __asm__ volatile("" ::: "memory");
692 /* HUM Ch 59 "MRCFLR : Code MRAM Flush Register" p 3601 */
693 *ra8_mram_reg16(k_ra8_mram_off_mrcflr) = k_ra8_mrcflr_key_flush;
694
695 const ra8_err_t err = internal_wait_commit_done(k_ra8_flash_busy_spin_limit);
696
697 /* Tear down regardless of err so the gate cannot stay open. */
698 internal_set_program_gate(world, false);
699 internal_set_hsp_mode(false);
700 ra8_flash_internal_set_prefetch(true);
701
702 return err;
703}

705ra8_err_t
706ra8_flash_write_block(uint32_t mram_addr, const uint8_t* src, uint32_t len, ra8_flash_world_t world)
707{
708 RA8_CHECK_NULL_PTR(src, s_flash_tag, "src must not be nullptr");
709 const ra8_err_t v_err = internal_validate_write_block(mram_addr, len);
710 RA8_RETURN_ON_ERROR(v_err, s_flash_tag, "flash_write: validate"); /* GCOVR_EXCL_BR_LINE */
711
712 /* Step 1: wait for the controller to be idle. */
713 ra8_err_t err = internal_wait_buffer_ready(k_ra8_flash_busy_spin_limit);
714 RA8_RETURN_ON_ERROR(err, s_flash_tag, "flash_write: busy wait"); /* GCOVR_EXCL_BR_LINE */
715
716 /* Steps 2-6: gate, write, flush, commit, teardown. */
717 err = internal_flash_program_window(mram_addr, src, len, world);
718 RA8_RETURN_ON_ERROR(err, s_flash_tag, "flash_write: commit wait"); /* GCOVR_EXCL_BR_LINE */
719
720 /* Step 7: error check.
721 * HUM Ch 59 "MRCPS : Code MRAM Program Status Register" p 3601 */
722 const uint8_t status = *ra8_mram_reg8(k_ra8_mram_off_mrcps);
723 if ((status & k_ra8_mrcps_mask_errors) != 0U) {
724 *ra8_mram_reg8(k_ra8_mram_off_mrcps) = k_ra8_mrcps_mask_errors;
725 ra8_log_error_val(s_flash_tag, "flash_write: hw error", (uint32_t)status);
726 return k_ra8_err_hw_error;
727 }
728 return k_ra8_ok;
729}

750ra8_err_t ra8_flash_block_protect_set(ra8_flash_world_t world, bool lock, bool permanent)
751{
752 if (permanent && !lock) {
753 /* Permanent unlock makes no sense -- the fuse is one-shot. */
754 return k_ra8_err_invalid_arg;
755 }
756 uint16_t value;
757 if (world == k_ra8_flash_world_s) {
758 if (lock) {
759 value = k_ra8_mrcbprot1_key_lock;
760 } else {
761 value = k_ra8_mrcbprot1_key_unlock;
762 }
763 /* HUM Ch 59 "MRCBPROT1 : Code MRAM Block Protection (S)" p 3605 */
764 *ra8_mram_reg16(k_ra8_mram_off_mrcbprot1) = value;
765 } else {
766 if (lock) {
767 value = k_ra8_mrcbprot0_key_lock;
768 } else {
769 value = k_ra8_mrcbprot0_key_unlock;
770 }
771 /* HUM Ch 59 "MRCBPROT0 : Code MRAM Block Protection (NS)" p 3604 */
772 *ra8_mram_reg16(k_ra8_mram_off_mrcbprot0) = value;
773 }
774 /* The "permanent" flag is conveyed via a configuration-set update of
775 * the OFS region; the per-register write above only takes effect for
776 * this boot. We expose the request for transparency, but defer the
777 * persistent fuse to ``ra8_flash_config_set_write`` for callers who
778 * truly want it. The HUM Ch 7 p 278 OFS layout is the trust anchor. */
779 if (permanent) {
780 ra8_log_warn(s_flash_tag, "permanent block protect requires config-set");
781 }
782 return k_ra8_ok;
783}

822ra8_err_t ra8_flash_suspend(void)
823{
824 /* HUM Ch 59 "MENTRYR : Extra MRAM Program-Mode Entry" pp 3582+ --
825 * driving MENTRYR with the keyed pause pattern (KEY=0xAA, MENTRY=1,
826 * PCKA=1) halts an in-flight MACI command at the next page
827 * boundary. Poll for PCKA=1 to confirm the pause. */
828 *ra8_mram_reg16(k_ra8_mram_off_mentryr) = k_ra8_mentryr_pe_pause;
829
830 for (uint32_t i = 0U; i < k_ra8_flash_pe_spin_limit; ++i) { /* GCOVR_EXCL_BR_LINE */
831 /* HUM Ch 59 "MENTRYR" p 3582 */ /* + */
832 const uint16_t v = *ra8_mram_reg16(k_ra8_mram_off_mentryr);
833 if ((v & k_ra8_mentryr_mask_pcka) != 0U) { /* GCOVR_EXCL_BR_LINE */
834 return k_ra8_ok;
835 }
836 }

840ra8_err_t ra8_flash_resume(void)
841{
842 /* HUM Ch 59 "MENTRYR : Extra MRAM Program-Mode Entry" pp 3582+ --
843 * resume key clears PCKA, leaving MENTRY=1 so programming continues.
844 * Poll for PCKA=0 before returning. */
845 *ra8_mram_reg16(k_ra8_mram_off_mentryr) = k_ra8_mentryr_pe_resume;
846
847 for (uint32_t i = 0U; i < k_ra8_flash_pe_spin_limit; ++i) { /* GCOVR_EXCL_BR_LINE */
848 /* HUM Ch 59 "MENTRYR" p 3582 */ /* + */
849 const uint16_t v = *ra8_mram_reg16(k_ra8_mram_off_mentryr);
850 if ((v & k_ra8_mentryr_mask_pcka) == 0U) { /* GCOVR_EXCL_BR_LINE */
851 return k_ra8_ok;
852 }
853 }
854 return k_ra8_err_hw_timeout;
855}

857ra8_err_t ra8_flash_lock_set(uintptr_t addr, uint16_t lock_bits)
858{
859 /* Address must lie inside the 1 MiB code-MRAM window. */
860 const uintptr_t code_lo = k_ra8_flash_code_start;
861 const uintptr_t code_hi = code_lo + k_ra8_flash_code_size;
862 if ((addr < code_lo) || (addr >= code_hi)) {
863 return k_ra8_err_invalid_arg;
864 }
865 /* Validate the keyed value: bits [15:8] must be one of the two
866 * documented keys (0x88 for MRCBPROT0, 0x44 for MRCBPROT1). */
867 enum : uint16_t {
868 k_ra8_mrcbprot_key_byte_mask = 0xFF00U,
869 k_ra8_mrcbprot_key_ns = 0x8800U,
870 k_ra8_mrcbprot_key_s = 0x4400U,
871 };
872 enum : uint32_t {
873 k_ra8_mrcbprot_secure_bit = 0x00080000UL,
874 };
875 const uint16_t key = (uint16_t)(lock_bits & k_ra8_mrcbprot_key_byte_mask);
876 const bool key_ns = (key == k_ra8_mrcbprot_key_ns);
877 const bool key_s = (key == k_ra8_mrcbprot_key_s);
878 if (!key_ns && !key_s) {
879 return k_ra8_err_invalid_arg;
880 }
881
882 const bool is_secure = ((addr & (uintptr_t)k_ra8_mrcbprot_secure_bit) != 0U);
883 if (is_secure) {
884 /* HUM Ch 59 "MRCBPROT1 : Code MRAM Block Protection (S)" p 3605 */
885 *ra8_mram_reg16(k_ra8_mram_off_mrcbprot1) = lock_bits;
886 } else {
887 /* HUM Ch 59 "MRCBPROT0 : Code MRAM Block Protection (NS)" p 3604 */
888 *ra8_mram_reg16(k_ra8_mram_off_mrcbprot0) = lock_bits;
889 }
890 return k_ra8_ok;
891}

898ra8_err_t ra8_flash_force_stop(void)
899{
900 /* HUM Ch 59 "MACI Command-Issuing Area" p 3550 */
901 ra8_flash_internal_maci_cmd8(k_ra8_maci_cmd_forced_stop);
902 ra8_err_t err = ra8_flash_internal_wait_mrdy(k_ra8_flash_maci_spin_limit);
903 if (err != k_ra8_ok) {
904 return err;
905 }
906
907 /* HUM Ch 59 "MASTAT : Extra MRAM Access Status Register" p 3577 */
908 const uint8_t mastat = *ra8_mram_reg8(k_ra8_mram_off_mastat);
909 if ((mastat & k_ra8_mastat_mask_cmdlk) != 0U) {
910 return k_ra8_err_hw_error;
911 }
912 return k_ra8_ok;
913}

915ra8_err_t ra8_flash_reset(void)
916{
917 RA8_VALIDATE_INIT(s_flash_rt.initialized, s_flash_tag, "flash_reset before init");
918 ra8_err_t err = ra8_flash_enter_pe_mode();
919 if (err != k_ra8_ok) {
920 return err;
921 }
922 ra8_err_t stop_err = ra8_flash_force_stop();
923 /* Clear status whether or not the stop succeeded. */
924 /* HUM Ch 59 "MACI Command-Issuing Area" p 3550 */
925 ra8_flash_internal_maci_cmd8(k_ra8_maci_cmd_status_clear);
926 (void)ra8_flash_internal_wait_mrdy(k_ra8_flash_maci_spin_limit);
927 ra8_err_t exit_err = ra8_flash_exit_pe_mode();
928 if (stop_err != k_ra8_ok) {
929 return stop_err;
930 }
931 return exit_err;
