165[[nodiscard]] static ra8_err_t demo_pins_init(void)
166{
167 /* The HS PHY data lines (USBHSDP / USBHSDM / USBHSRREF) are
168 * dedicated package balls on the BGA and bypass the PFS PSEL path.
169 * VBUS / VBUSEN / OVRCUR are sourced by the on-board USB-PD
170 * controller. So only P4_08 needs PFS routing for the controller. */
171 ra8_err_t err =
172 ra8_pfs_route_peripheral(k_demo_pin_hs_vbus, k_ra8_psel_usb_hs, "usb_cdc_hs.vbus");
173 if (err != k_ra8_ok) {
174 return err;
175 }
176 /* PD07 (J7 USB-HS role select, UM 6.2 p 34): drive LOW for Device
177 * mode. Mirrors the FS demo's P5_00 / VBUSEN GPIO drive in shape:
178 * the role line is owned by the application early so the analog
179 * block sees a stable strap before the controller is brought up.
180 * Previously this happened inside ra8_board_usbhs_device_init from
181 * the worker thread, but that caller also invoked ra8_usb_device_init
182 * which is then re-invoked by ux_dcd_ra8_usb_initialize -- the
183 * duplicate PHY bring-up is what kept INTENB0 from sticking and is
184 * the most plausible reason the host enumerates FS but never sends
185 * SETUP on HS. Owning PD07 here lets the worker drop the redundant
186 * board init call entirely. */
187 return ra8_gpio_output_init(k_demo_pin_pd07_role, k_ra8_level_low);
188}

207 uint32_t cpuclk0_hz = 0U;

209 if (ra8_cgc_init() != k_ra8_ok) {

213 /* Bring up the USBHS PHY clock (USB60CKCR / USB60CLK = PLL2P / 4 =
214 * 60 MHz) BEFORE any caller releases MSTPB12 (USBHS) -- mirrors the
215 * FS demo's ra8_cgc_usbfs_clock_enable() pattern. The bridge's
216 * ra8_usb_device_init invocation later releases MSTPB12 and immediately
217 * starts the PHY-PLL CLKSEL bisect; without this clock arm step the
218 * PHY block has no reference and PLLLOCK never asserts.
219 *
220 * Previously this lived inside ra8_board_usbhs_device_init() called
221 * from the worker, but that wrapper also called ra8_usb_device_init
222 * which is then re-invoked by ux_dcd_ra8_usb_initialize -- the
223 * duplicate PHY bring-up clobbered INTENB0 and is the most plausible
224 * explanation for the "ISR fires but host never sends SETUP" symptom
225 * on HS. With the clock armed here the worker can drop the wrapper
226 * call entirely, making the HS demo's worker-side init sequence a
227 * structural mirror of the working FS demo. */
228 if (ra8_cgc_usbhs_pll_enable() != k_ra8_ok) {
229 demo_panic_halt();
230 }
