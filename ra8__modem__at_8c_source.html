440ra8_modem_line_kind_t ra8_modem_at_internal_classify(const char* line,
441 const char* cmd_echo,
442 const char* expected_response)
443{
444 if (line[0] == '\0') {
445 return k_ra8_modem_line_kind_empty;
446 }
447 if ((cmd_echo != nullptr) && (ra8_modem_at_internal_str_eq(line, cmd_echo) != 0U)) {
448 return k_ra8_modem_line_kind_echo;
449 }
450 uint8_t is_err = 0U;
451 if (internal_classify_final(line, &is_err) != 0U) {
452 return (is_err != 0U) ? k_ra8_modem_line_kind_final_err : k_ra8_modem_line_kind_final_ok;
453 }
454 /* If the caller didn't ask for a specific prefix, allow URC dispatch. */
455 if ((expected_response == nullptr) || (expected_response[0] == '\0') ||
456 (ra8_modem_at_internal_starts_with(line, expected_response) == 0U)) {
457 if (internal_dispatch_urc(line) != 0U) {
458 return k_ra8_modem_line_kind_urc;
459 }
460 }
461 return k_ra8_modem_line_kind_payload;

480static void internal_accumulate(uint8_t byte, const char** line_out)
481{
482 *line_out = nullptr;
483 if ((byte == (uint8_t)'\r') || (byte == (uint8_t)'\n')) {
484 if (s_mod.line_len > 0U) {
485 s_mod.cfg.line_buf[s_mod.line_len] = (uint8_t)'\0';
486 *line_out = (const char*)s_mod.cfg.line_buf;
487 s_mod.line_len = 0U;
488 }
489 return;
490 }
491 if ((s_mod.line_len + 1U) >= s_mod.cfg.line_buf_len) {
492 /* Overflow: terminate, emit what we have, drop the new byte. */
493 s_mod.cfg.line_buf[s_mod.line_len] = (uint8_t)'\0';
494 *line_out = (const char*)s_mod.cfg.line_buf;
495 s_mod.line_len = 0U;
496 return;
497 }
498 s_mod.cfg.line_buf[s_mod.line_len] = byte;
499 ++s_mod.line_len;
500}

621static ra8_modem_line_action_t internal_handle_line(const char* line,
622 ra8_modem_line_kind_t kind,
623 const char* expected_response,
624 uint8_t* seen_exp,
625 char* capture,
626 size_t capture_len,
627 size_t* used)
628{
629 switch (kind) {
630 case k_ra8_modem_line_kind_empty:
631 case k_ra8_modem_line_kind_urc:
632 return k_ra8_modem_line_action_continue;
633 case k_ra8_modem_line_kind_echo:
634 s_mod.state = k_ra8_modem_at_state_await_resp;
635 return k_ra8_modem_line_action_continue;
636 case k_ra8_modem_line_kind_final_ok:
637 s_mod.state = k_ra8_modem_at_state_done;
638 if (*seen_exp == 0U) {

640 return k_ra8_modem_line_action_done_err;
641 }
642 return k_ra8_modem_line_action_done_ok;
643 case k_ra8_modem_line_kind_final_err:
644 s_mod.state = k_ra8_modem_at_state_done;
645 return k_ra8_modem_line_action_done_err;
646 case k_ra8_modem_line_kind_payload:
647 default:
648 if (ra8_modem_at_internal_handle_line_payload_prefix_match(line, expected_response) != 0U) {
649 *seen_exp = 1U;
650 }
651 ra8_modem_at_internal_capture_line(line, capture, capture_len, used);
652 if (s_mod.state == k_ra8_modem_at_state_await_echo) {
653 s_mod.state = k_ra8_modem_at_state_await_resp;
654 }
655 return k_ra8_modem_line_action_continue;
656 }

687static ra8_modem_line_action_t internal_pump_one(const ra8_modem_wait_ctx_t* wc)
688{
689 uint8_t byte = 0U;
690 const ra8_err_t rx_err = s_mod.cfg.io.rx_byte(s_mod.cfg.io.ctx, &byte);
691 if (rx_err != k_ra8_ok) {
692 return k_ra8_modem_line_action_continue;
693 }
694 const char* line = nullptr;
695 internal_accumulate(byte, &line);
696 if (line == nullptr) {
697 return k_ra8_modem_line_action_continue;
698 }
699 const ra8_modem_line_kind_t kind =
700 ra8_modem_at_internal_classify(line, wc->cmd, wc->expected_response);
701 return internal_handle_line(line,
702 kind,
703 wc->expected_response,
704 wc->seen_exp,
705 wc->capture,
706 wc->capture_len,
707 wc->used);

729RA8_INTERNAL
730static ra8_err_t internal_wait_response(const char* cmd,
731 const char* expected_response,
732 uint16_t timeout_ms,
733 char* capture,
734 size_t capture_len)
735{
736 const uint32_t start = s_mod.cfg.io.now_ms(s_mod.cfg.io.ctx);
737 size_t used = 0U;
738 uint8_t seen_exp = (uint8_t)((expected_response == nullptr) || (expected_response[0] == '\0'));
739
740 if (ra8_modem_at_internal_wait_response_should_clear_capture(capture, capture_len) != 0U) {
741 capture[0] = '\0';
742 }
743
744 s_mod.state = k_ra8_modem_at_state_await_echo;
745
746 const ra8_modem_wait_ctx_t wc = {
747 .cmd = cmd,
748 .expected_response = expected_response,
749 .capture = capture,
750 .capture_len = capture_len,
751 .used = &used,
752 .seen_exp = &seen_exp,
753 };
754
755 while (1) {
756 const ra8_modem_line_action_t action = internal_pump_one(&wc);
757 if (action == k_ra8_modem_line_action_done_ok) {
758 internal_reset_line();
759 return k_ra8_ok;
760 }
761 if (action == k_ra8_modem_line_action_done_err) {
762 internal_reset_line();
763 return k_ra8_err_hw_error;
764 }
765 const uint32_t elapsed = s_mod.cfg.io.now_ms(s_mod.cfg.io.ctx) - start;
766 if (elapsed >= (uint32_t)timeout_ms) {
767 s_mod.state = k_ra8_modem_at_state_idle;
768 internal_reset_line();
769 return k_ra8_err_hw_timeout;
770 }
771 }
772}
