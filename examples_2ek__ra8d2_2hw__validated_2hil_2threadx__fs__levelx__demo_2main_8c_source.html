297static void demo_read_test_file(void)
298{
299 ra8_fs_file_t* file = nullptr;
300 if (ra8_fs_open(s_fs_mount, s_demo_file_path, k_ra8_fs_mode_read, &file) != k_ra8_ok) {
301 demo_print("[fslx] ra8_fs_open(read) failed\r\n");
302 return;
303 }
304 demo_print("[fslx] readback: ");
305 uint8_t buf[k_demo_file_chunk];
306 uint32_t got = 0U;
307 do {
308 got = 0U;
309 if (ra8_fs_read(file, buf, (uint32_t)k_demo_file_chunk, &got) != k_ra8_ok) {
310 break;
311 }
312 if (got > 0U) {
313 (void)ra8_board_uart_console_write(buf, (size_t)got);
314 }
315 } while (got > 0U);
316 demo_print("\r\n");
317 (void)ra8_fs_close(file);
318}

330static void demo_thread_entry(ULONG thread_input)
331{
332 (void)thread_input;
333
334 demo_print("[fslx] booting xSPI flash\r\n");
335 /* The LevelX NOR driver owns OCTA bus bring-up: lx_nor_flash_format ->
336 * lx_nor_driver_ra8_xspi_initialize -> priv_bus_init_once routes the
337 * pins, runs the 8D/1S software-reset recovery, calls ra8_xspi_init,
338 * and probes RDID exactly once. Doing it here as well double-routes
339 * the PFS pins (the validator rejects the second route with
340 * k_ra8_err_gpio_conflict), so the driver's initialize bails before
341 * wiring the sector buffer and format returns LX_NO_MEMORY. Let the
342 * driver be the single owner. */
343 demo_print("[fslx] formatting + opening LevelX partition\r\n");
344 demo_lx_open_or_panic();
345
346 demo_print("[fslx] formatting + mounting FAT volume on LevelX\r\n");
347 demo_fs_format_or_panic();
348
349 demo_write_test_file();
350 demo_read_test_file();
351
352 (void)ra8_fs_unmount(s_fs_mount);
353 (void)lx_nor_flash_close(&s_nor_flash);
354 demo_print("[fslx] done\r\n");
355}

373void tx_application_define(void* first_unused_memory)
374{
375 (void)first_unused_memory;
376
377 lx_nor_flash_initialize();
378
379 (void)tx_mutex_create(&s_fs_mutex, "ra8_fs", TX_NO_INHERIT);
380 const ra8_fs_lock_t lock = {
381 .acquire = demo_fs_lock_acquire,
382 .release = demo_fs_lock_release,
383 .ctx = &s_fs_mutex,
384 };
385 if (ra8_fs_set_lock(&lock) != k_ra8_ok) {
386 demo_print("[fslx] ra8_fs_set_lock failed\r\n");
387 }
388
389 (void)tx_thread_create(&s_demo_thread,
390 s_demo_thread_name,
391 demo_thread_entry,
392 0U,
393 s_demo_stack,
394 (ULONG)sizeof(s_demo_stack),
395 8U, /* priority */
396 8U, /* preempt threshold */
397 TX_NO_TIME_SLICE,
398 TX_AUTO_START);
399}
