59RA8_INTERNAL
60static uint64_t priv_fmt_clusters_for(const ra8_fs_fmt_geom_t* g, uint32_t spc, uint32_t* out_fatsz)
61{
62 /* FAT entries per DEVICE sector: 4-byte FAT32 entries or 2-byte FAT16 ones
63 * (FAT12 sizes its FAT with the FAT16 figure, a deliberate overestimate). */
64 const uint32_t entry_bytes = (g->type == k_ra8_fs_type_fat32) ? (uint32_t)k_fmt_fat32_entry_bytes
65 : (uint32_t)k_fmt_fat16_entry_bytes;
66 const uint32_t entry_cap = g->bytes_per_sector / entry_bytes;
67 const uint64_t overhead = (uint64_t)g->reserved_sectors + g->root_sectors;
68 if (g->total_sectors <= overhead) {
69 *out_fatsz = 0U;
70 return 0U;
71 }
72 const uint64_t avail = g->total_sectors - overhead;
73 const uint64_t denom = ((uint64_t)entry_cap * spc) + (uint32_t)k_fmt_num_fats;
74 const uint64_t fatsz = (avail + denom - 1U) / denom;
75 *out_fatsz = (uint32_t)fatsz;
76 const uint64_t fat_total = (uint64_t)k_fmt_num_fats * fatsz;
77 if (avail <= fat_total) {
78 return 0U;
79 }
80 return (avail - fat_total) / spc;
81}

142static uint32_t priv_fmt_log2(uint32_t v)
143{
144 uint32_t r = 0U;
145 uint32_t w = v;
146 while (w > 1U) {
147 w >>= 1U;
148 r++;
149 }

183RA8_INTERNAL
184static uint32_t priv_fmt_fat32_default_spc(uint64_t total_sectors, uint32_t bps)
185{
186 const uint32_t bps_shift = priv_fmt_log2(bps);
187 const uint64_t total_512 = total_sectors
188 << (bps_shift - priv_fmt_log2((uint32_t)k_ra8_fs_sector_min));
189 uint32_t clus_shift = (uint32_t)k_fmt_f32_clus_32k;
190 if (total_512 <= (uint64_t)k_fmt_f32_thr_260m) {
191 clus_shift = (uint32_t)k_fmt_f32_clus_512b;
192 } else if (total_512 <= (uint64_t)k_fmt_f32_thr_8g) {
193 clus_shift = (uint32_t)k_fmt_f32_clus_4k;
194 } else if (total_512 <= (uint64_t)k_fmt_f32_thr_16g) {
195 clus_shift = (uint32_t)k_fmt_f32_clus_8k;
196 } else if (total_512 <= (uint64_t)k_fmt_f32_thr_32g) {
197 clus_shift = (uint32_t)k_fmt_f32_clus_16k;
198 } else {
199 clus_shift = (uint32_t)k_fmt_f32_clus_32k;
200 }
201 if (clus_shift <= bps_shift) {
202 return 1U;
203 }
204 return 1U << (clus_shift - bps_shift);
205}

207/* `priv_fmt_choose_geometry()`: see header for the documented contract. */
208ra8_err_t priv_fmt_choose_geometry(ra8_fs_fmt_geom_t* g, uint32_t spc_hint)
209{
210 /* FAT's BPB records the volume size in a 32-bit field: a device whose
211 * sector count does not fit cannot be described by the format at all --
212 * exFAT is the right choice there, so this fails cleanly instead of
213 * truncating the size (#683). */
214 if (g->total_sectors > (uint64_t)UINT32_MAX) {
215 return k_ra8_err_invalid_size;
216 }
217 bool auto_mode = (spc_hint == 0U);
218 uint32_t spc = spc_hint;
219 if (auto_mode) {
220 /* FAT32 starts the sweep at the size-appropriate cluster (MS table) so a
221 * big card gets a small FAT; FAT12/16 start at the minimum and grow. */
222 spc = (g->type == k_ra8_fs_type_fat32)
223 ? priv_fmt_fat32_default_spc(g->total_sectors, g->bytes_per_sector)
224 : 1U;
225 }
226 for (uint32_t guard = 0U; guard <= (uint32_t)k_fmt_spc_max; guard++) {
227 uint32_t fatsz = 0U;
228 const uint64_t count = priv_fmt_clusters_for(g, spc, &fatsz);
229 if (priv_fmt_count_in_band(g->type, count)) {
230 g->sectors_per_cluster = spc;
231 g->fat_size_sectors = fatsz;
232 g->count_of_clusters = (uint32_t)count;
233 return k_ra8_ok;
234 }
235 if (!auto_mode || spc >= (uint32_t)k_fmt_spc_max) {
236 return k_ra8_err_invalid_size;
237 }
238 spc *= 2U;
239 }
240 return k_ra8_err_invalid_size;
241}

270/* `priv_fmt_label_field()`: see header for the documented contract. */
271void priv_fmt_label_field(uint8_t* dst, const char* label)
272{
273 /* An unlabelled FAT volume stores the spec sentinel "NO NAME ", never
274 * zeros and never a bare run of spaces: fsck.fat reads a blank BS_VolLab as a
275 * corrupt label and strips it on sight, which trains people to ignore fsck
276 * output on every card this firmware writes (#634). A NULL or empty label
277 * therefore resolves to that sentinel before padding. */
278 static const char k_no_name[] = "NO NAME";
279 const char* eff = ((label != nullptr) && (label[0] != '\0')) ? label : k_no_name;
280 bool past_end = false;
281 for (uint32_t i = 0U; i < (uint32_t)k_fmt_label_len; i++) {
282 if (!past_end && (eff[i] == '\0')) {
283 past_end = true;
284 }
285 dst[i] = past_end ? (uint8_t)' ' : (uint8_t)eff[i];
286 }
287}

450RA8_INTERNAL
451static ra8_err_t
452priv_fmt_zero_run(const ra8_fs_backend_t* backend, uint64_t lba, uint64_t count, uint32_t bps)
453{
454 const uint32_t chunk_secs = (uint32_t)k_fmt_zero_chunk_bytes / bps;
455 uint64_t done = 0U;
456 while (done < count) {
457 uint32_t chunk = chunk_secs;
458 if ((uint64_t)chunk > (count - done)) {
459 chunk = (uint32_t)(count - done);
460 }
461 const ra8_err_t err = backend->write_block(backend->ctx, lba + done, chunk, k_fmt_zero_chunk);
462 if (err != k_ra8_ok) {
463 return err;
464 }
465 done += chunk;
466 }
467 return k_ra8_ok;
468}

507RA8_INTERNAL
508static ra8_err_t priv_fmt_seed_fats(const ra8_fs_backend_t* backend, const ra8_fs_fmt_geom_t* g)
509{
510 for (uint32_t i = 0U; i < g->bytes_per_sector; i++) {
511 s_scratch[i] = 0U;
512 }
513 if (g->type == k_ra8_fs_type_fat32) {
514 priv_wr32(&s_scratch[0], (uint32_t)k_cluster_eoc_min_fat32 | (uint32_t)k_fmt_media_fixed);
515 priv_wr32(&s_scratch[4], (uint32_t)k_cluster_eoc_write_fat32);
516 priv_wr32(&s_scratch[8], (uint32_t)k_cluster_eoc_write_fat32);
517 } else if (g->type == k_ra8_fs_type_fat16) {
518 priv_wr16(&s_scratch[0],
519 (uint16_t)((uint32_t)k_cluster_eoc_min_fat16 | (uint32_t)k_fmt_media_fixed));
520 priv_wr16(&s_scratch[2], (uint16_t)k_cluster_eoc_write_fat16);
521 } else {
522 /* FAT12: 1.5-byte entries. FAT[0]=0xF<media>, FAT[1]=0xFFF, packed LE. */
523 s_scratch[0] = (uint8_t)k_fmt_media_fixed;
524 s_scratch[1] = (uint8_t)k_byte_mask;
525 s_scratch[2] = (uint8_t)k_byte_mask;
526 }
527 for (uint32_t f = 0U; f < (uint32_t)k_fmt_num_fats; f++) {
528 const uint64_t fat_lba = (uint64_t)g->reserved_sectors + ((uint64_t)f * g->fat_size_sectors);
529 const ra8_err_t err = backend->write_block(backend->ctx, fat_lba, 1U, s_scratch);
530 if (err != k_ra8_ok) {
531 return err;
532 }
533 }
534 return k_ra8_ok;
535}

598/* `priv_fmt_emit_volume()`: see header for the documented contract. */
599ra8_err_t
600priv_fmt_emit_volume(const ra8_fs_backend_t* backend, const ra8_fs_fmt_geom_t* g, const char* label)
601{
602 uint8_t* const boot = priv_sec_walk();
603 priv_byte_fill(boot, 0U, g->bytes_per_sector);
604 if (g->type == k_ra8_fs_type_fat32) {
605 priv_fmt_build_bpb_f32(boot, g, label);
606 } else {
607 priv_fmt_build_bpb_f16(boot, g, label);
608 }
609 /* Clear the FAT region plus the (contiguous) root region first, so no stale
610 * non-zero FAT word survives as an orphan cluster. Erase the whole span in
611 * one go when the card guarantees zero-after-erase, else stream zeros. Doing
612 * this BEFORE the metadata writes means an erase that rounds into the reserved
613 * region cannot clobber the boot sector / FSInfo written below. */
614 const uint64_t fat_total = (uint64_t)k_fmt_num_fats * g->fat_size_sectors;
615 const uint64_t root_span =
616 (g->type == k_ra8_fs_type_fat32) ? g->sectors_per_cluster : g->root_sectors;
617 ra8_err_t err =
618 priv_fmt_clear_region(backend, g->reserved_sectors, fat_total + root_span, g->bytes_per_sector);
619 if (err != k_ra8_ok) {
620 return err;
621 }
622 /* Now lay the metadata: boot sector, FAT seeds (rewrite FAT sector 0 of each
623 * copy with the media + EOC reserved entries), then FAT32 FSInfo + backup. */
624 err = backend->write_block(backend->ctx, 0U, 1U, boot);
625 if (err != k_ra8_ok) {
626 return err;
627 }
628 err = priv_fmt_seed_fats(backend, g);
629 if (err != k_ra8_ok) {
630 return err;
631 }
632 return priv_fmt_write_fsinfo(backend, g, boot);
633}
