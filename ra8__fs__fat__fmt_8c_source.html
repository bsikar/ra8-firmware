59RA8_INTERNAL
60static uint32_t priv_fmt_clusters_for(const ra8_fs_fmt_geom_t* g, uint32_t spc, uint32_t* out_fatsz)
61{
62 const uint32_t entry_cap = (g->type == k_ra8_fs_type_fat32) ? (uint32_t)k_fmt_fat32_entry_cap
63 : (uint32_t)k_fmt_fat16_entry_cap;
64 const uint32_t overhead = g->reserved_sectors + g->root_sectors;
65 if (g->total_sectors <= overhead) {
66 *out_fatsz = 0U;
67 return 0U;
68 }
69 const uint32_t avail = g->total_sectors - overhead;
70 const uint32_t denom = (entry_cap * spc) + (uint32_t)k_fmt_num_fats;
71 const uint32_t fatsz = (avail + denom - 1U) / denom;
72 *out_fatsz = fatsz;
73 const uint32_t fat_total = (uint32_t)k_fmt_num_fats * fatsz;
74 if (avail <= fat_total) {
75 return 0U;
76 }
77 return (avail - fat_total) / spc;
78}

160/* `priv_fmt_choose_geometry()`: see header for the documented contract. */
161ra8_err_t priv_fmt_choose_geometry(ra8_fs_fmt_geom_t* g, uint32_t spc_hint)
162{
163 bool auto_mode = (spc_hint == 0U);
164 uint32_t spc = spc_hint;
165 if (auto_mode) {
166 /* FAT32 starts the sweep at the size-appropriate cluster (MS table) so a
167 * big card gets a small FAT; FAT12/16 start at the minimum and grow. */
168 spc = (g->type == k_ra8_fs_type_fat32) ? priv_fmt_fat32_default_spc(g->total_sectors) : 1U;
169 }
170 for (uint32_t guard = 0U; guard <= (uint32_t)k_fmt_spc_max; guard++) {
171 uint32_t fatsz = 0U;
172 const uint32_t count = priv_fmt_clusters_for(g, spc, &fatsz);
173 if (priv_fmt_count_in_band(g->type, count)) {
174 g->sectors_per_cluster = spc;
175 g->fat_size_sectors = fatsz;
176 g->count_of_clusters = count;
177 return k_ra8_ok;
178 }
179 if (!auto_mode || spc >= (uint32_t)k_fmt_spc_max) {
180 return k_ra8_err_invalid_size;
181 }
182 spc *= 2U;
183 }
184 return k_ra8_err_invalid_size;
185}

405RA8_INTERNAL
406static ra8_err_t priv_fmt_zero_run(const ra8_fs_backend_t* backend, uint32_t lba, uint32_t count)
407{
408 uint32_t done = 0U;
409 while (done < count) {
410 uint32_t chunk = count - done;
411 if (chunk > (uint32_t)k_fmt_zero_chunk_secs) {
412 chunk = (uint32_t)k_fmt_zero_chunk_secs;
413 }
414 const ra8_err_t err = backend->write_block(backend->ctx, lba + done, chunk, k_fmt_zero_chunk);
415 if (err != k_ra8_ok) {
416 return err;
417 }
418 done += chunk;
419 }
420 return k_ra8_ok;
421}

459RA8_INTERNAL
460static ra8_err_t priv_fmt_seed_fats(const ra8_fs_backend_t* backend, const ra8_fs_fmt_geom_t* g)
461{
462 for (uint32_t i = 0U; i < (uint32_t)k_ra8_fs_bytes_per_sector; i++) {
463 s_scratch[i] = 0U;
464 }
465 if (g->type == k_ra8_fs_type_fat32) {
466 priv_wr32(&s_scratch[0], (uint32_t)k_cluster_eoc_min_fat32 | (uint32_t)k_fmt_media_fixed);
467 priv_wr32(&s_scratch[4], (uint32_t)k_cluster_eoc_write_fat32);
468 priv_wr32(&s_scratch[8], (uint32_t)k_cluster_eoc_write_fat32);
469 } else if (g->type == k_ra8_fs_type_fat16) {
470 priv_wr16(&s_scratch[0],
471 (uint16_t)((uint32_t)k_cluster_eoc_min_fat16 | (uint32_t)k_fmt_media_fixed));
472 priv_wr16(&s_scratch[2], (uint16_t)k_cluster_eoc_write_fat16);
473 } else {
474 /* FAT12: 1.5-byte entries. FAT[0]=0xF<media>, FAT[1]=0xFFF, packed LE. */
475 s_scratch[0] = (uint8_t)k_fmt_media_fixed;
476 s_scratch[1] = (uint8_t)k_byte_mask;
477 s_scratch[2] = (uint8_t)k_byte_mask;
478 }
479 for (uint32_t f = 0U; f < (uint32_t)k_fmt_num_fats; f++) {
480 const uint32_t fat_lba = g->reserved_sectors + (f * g->fat_size_sectors);
481 const ra8_err_t err = backend->write_block(backend->ctx, fat_lba, 1U, s_scratch);
482 if (err != k_ra8_ok) {
483 return err;
484 }
485 }
486 return k_ra8_ok;
487}

550/* `priv_fmt_emit_volume()`: see header for the documented contract. */
551ra8_err_t
552priv_fmt_emit_volume(const ra8_fs_backend_t* backend, const ra8_fs_fmt_geom_t* g, const char* label)
553{
554 uint8_t boot[k_ra8_fs_bytes_per_sector] = {};
555 if (g->type == k_ra8_fs_type_fat32) {
556 priv_fmt_build_bpb_f32(boot, g, label);
557 } else {
558 priv_fmt_build_bpb_f16(boot, g, label);
559 }
560 /* Clear the FAT region plus the (contiguous) root region first, so no stale
561 * non-zero FAT word survives as an orphan cluster. Erase the whole span in
562 * one go when the card guarantees zero-after-erase, else stream zeros. Doing
563 * this BEFORE the metadata writes means an erase that rounds into the reserved
564 * region cannot clobber the boot sector / FSInfo written below. */
565 const uint32_t fat_total = (uint32_t)k_fmt_num_fats * g->fat_size_sectors;
566 const uint32_t root_span =
567 (g->type == k_ra8_fs_type_fat32) ? g->sectors_per_cluster : g->root_sectors;
568 ra8_err_t err = priv_fmt_clear_region(backend, g->reserved_sectors, fat_total + root_span);
569 if (err != k_ra8_ok) {
570 return err;
571 }
572 /* Now lay the metadata: boot sector, FAT seeds (rewrite FAT sector 0 of each
573 * copy with the media + EOC reserved entries), then FAT32 FSInfo + backup. */
574 err = backend->write_block(backend->ctx, 0U, 1U, boot);
575 if (err != k_ra8_ok) {
576 return err;
577 }
578 err = priv_fmt_seed_fats(backend, g);
579 if (err != k_ra8_ok) {
580 return err;
581 }
582 return priv_fmt_write_fsinfo(backend, g, boot);
583}
