152RA8_INTERNAL
153static ra8_err_t internal_wait_idle(void)
154{
155 for (uint32_t i = 0U; i < (uint32_t)k_ra8_ceu_reset_spin; i++) {
156 /* HUM Ch 60.2.23 "CSTSR : Capture Status Register" p 3672 */
157 const uint32_t cstsr = *ra8_ceu_reg32(k_ra8_ceu_off_cstsr);
158 /* HUM Ch 60.2.1 "CAPSR : Capture Start Register" p 3630 */
159 const uint32_t capsr = *ra8_ceu_reg32(k_ra8_ceu_off_capsr);
160 const bool idle = ((cstsr & (uint32_t)k_ra8_ceu_cstsr_mask_cpton) == 0U) &&
161 ((capsr & (uint32_t)k_ra8_ceu_capsr_mask_cpkil) == 0U);
162#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
163 /* Host MMIO fault seam, keyed on CSTSR (the primary status reg). */
164 if (ra8_fake_mmio_wait_eval(ra8_ceu_reg32(k_ra8_ceu_off_cstsr), i, idle)) {
165 return k_ra8_ok;
166 }
167#else
168 if (idle) {
169 return k_ra8_ok;
170 }
171#endif
172 }
173 return k_ra8_err_hw_timeout;
174}

424RA8_ISR_SAFE
425void ra8_ceu_dispatch(void)
426{
427 /* HUM Ch 60.2.22 "CETCR : Capture Event Flag Clear Register" p 3664
428 * CETCR bits clear when 0 is written; preserve bits not observed by
429 * re-reading and writing back `current & ~pending`. Writing a 1 to a
430 * CETCR bit leaves that flag asserted, so the prior observed bits
431 * become 0 and any newly-asserted-during-dispatch bits stay set. */
432 volatile uint32_t* reg = ra8_ceu_reg32(k_ra8_ceu_off_cetcr);
433 const uint32_t pending = *reg;
434 const uint32_t current = *reg;
435 *reg = current & ~pending;
436
437 const ra8_ceu_event_fn_t fn = s_ceu_fn;
438 void* const ctx = s_ceu_ctx;
439 if (fn != nullptr) {
440 fn(ctx, pending & s_ceu_int_enable);
441 }
442}

475RA8_INTERNAL
476static ra8_err_t internal_arm_capture(const ra8_ceu_buffers_t* bufs)
477{
478 /* HUM Ch 60.2.23 "CSTSR : Capture Status Register" p 3672 -- a
479 * non-zero CPTON means a capture is already running. */
480 const uint32_t cstsr = *ra8_ceu_reg32(k_ra8_ceu_off_cstsr);
481 if ((cstsr & k_ra8_ceu_cstsr_mask_cpton) != 0U) {
482 return k_ra8_err_busy;
483 }
484 /* HUM Ch 60.2.1 "CAPSR : Capture Start Register" p 3630 -- if a
485 * software reset is in flight the engine is not ready. */
486 const uint32_t capsr_now = *ra8_ceu_reg32(k_ra8_ceu_off_capsr);
487 if ((capsr_now & k_ra8_ceu_capsr_mask_cpkil) != 0U) {
488 return k_ra8_err_busy;
489 }
490
491 internal_program_addresses(bufs);
492
493 /* HUM Ch 60.2.18 "CFWCR : Firewall Operation Control Register" p 3661
494 * In data-enable-fetch mode arm the firewall to the cached image
495 * area; in image / data-sync modes leave the firewall disabled. */
496 if (s_ceu_capture_format == k_ra8_ceu_fmt_data_enable) {
497 // mcdc-deactivated: TU-local helper internal_arm_capture; bufs->y_top non-NULL is enforced by upstream public-API ra8_ceu_capture_start contract, and s_ceu_image_area is non-zero whenever data-enable-fetch mode is configured -- the AND's two conditions cannot be flipped independently from any reachable call site.
498 if ((s_ceu_image_area != 0U) && (bufs->y_top != nullptr)) {
499 const uint32_t upper = (uint32_t)(uintptr_t)bufs->y_top + s_ceu_image_area - 1U;
500 *ra8_ceu_reg32(k_ra8_ceu_off_cfwcr) =
501 k_ra8_ceu_cfwcr_mask_fwe | (upper & k_ra8_ceu_cfwcr_mask_fwv);
502 }
503 } else {
504 *ra8_ceu_reg32(k_ra8_ceu_off_cfwcr) = 0U;
505 }
506
507 /* HUM Ch 60.2.22 "CETCR : Capture Event Flag Clear Register" p 3664 */
508 *ra8_ceu_reg32(k_ra8_ceu_off_cetcr) = 0U;
509
510 /* HUM Ch 60.2.1 "CAPSR : Capture Start Register" p 3630 -- arm the
511 * CE bit; capture begins on the next VD edge. */
512 *ra8_ceu_reg32(k_ra8_ceu_off_capsr) = k_ra8_ceu_capsr_mask_ce;
513 return k_ra8_ok;
514}

778ra8_err_t ra8_ceu_dma_pump(uint8_t channel, const uint8_t* src, uint8_t* dst, uint32_t bytes)
779{
780 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
781 RA8_CHECK_NULL_PTR(dst, s_tag, "dst must not be nullptr");
782 if (bytes == 0U) {
783 return k_ra8_err_invalid_arg;
784 }
785 if ((bytes & k_ra8_ceu_dma_byte_align_mask) != 0U) {
786 return k_ra8_err_invalid_arg;
787 }
788
789 /* The CEU writes pixels with its own bus initiator; the DMAC is only
790 * used by the application-level pump that ferries finished
791 * frames. Configure for word-wide block transfer with both
792 * pointers incrementing. */
793 const ra8_dmac_config_t cfg = {
794 .src = (uint32_t)(uintptr_t)src,
795 .dst = (uint32_t)(uintptr_t)dst,
796 .count = (uint16_t)(bytes >> k_ra8_ceu_dma_count_shift),
797 .width = k_ra8_dmac_width_word,
798 .src_inc = true,
799 .dst_inc = true,
800 };
801 return ra8_dmac_start(channel, &cfg);
802}

824ra8_err_t ra8_ceu_capture_start(uint32_t num_frames)
825{
826 if (s_ceu_dma_buf == nullptr) {
827 return k_ra8_err_invalid_state;
828 }
829 /* HUM Ch 60.2.2 "CAPCR" p 3634-3635 -- CTNCP[16] = 1 selects
830 * continuous capture, 0 selects single-shot. */
831 volatile uint32_t* capcr_reg = ra8_ceu_reg32(k_ra8_ceu_off_capcr);
832 uint32_t capcr = *capcr_reg;
833 if (num_frames == 0U) {
834 capcr |= ((uint32_t)1U << k_ra8_ceu_capcr_shift_ctncp);
835 } else {
836 capcr &= ~((uint32_t)1U << k_ra8_ceu_capcr_shift_ctncp);
837 }
838 *capcr_reg = capcr;
839
840 return ra8_ceu_capture_arm(s_ceu_dma_buf);
841}
