180[[nodiscard]] static ra8_err_t dma_hal_run_copy(void)
181{
182 const ra8_dmac_config_t cfg = {
183 .src = (uint32_t)(uintptr_t)s_src,
184 .dst = (uint32_t)(uintptr_t)s_dst,
185 .count = (uint16_t)k_dma_hal_buf_words,
186 .block_count = 1U,
187 .width = k_ra8_dmac_width_word,
188 .src_inc = true,
189 .dst_inc = true,
190 };
191 ra8_err_t err = ra8_dmac_start_block((uint8_t)k_dma_hal_channel, &cfg);
192 if (err != k_ra8_ok) {
193 return err;
194 }
195
196 err = ra8_dmac_software_trigger((uint8_t)k_dma_hal_channel);
197 if (err != k_ra8_ok) {
198 (void)ra8_dmac_stop((uint8_t)k_dma_hal_channel);
199 return err;
200 }
201
202 err = ra8_dmac_wait_idle((uint8_t)k_dma_hal_channel, (uint32_t)k_dma_hal_poll_limit);
203 const ra8_err_t stop_err = ra8_dmac_stop((uint8_t)k_dma_hal_channel);
204 if (err != k_ra8_ok) {
205 return err;
206 }
207 return stop_err;
208}

212int32_t main(void)
213{
214 dma_hal_setup_or_halt();
215 ra8_isr_globals_enable();
216
217 while (1) {
218 dma_hal_fill_buffers();
219 const ra8_err_t err = dma_hal_run_copy();
220 uint8_t ok = 0U;
221 if (err == k_ra8_ok) {
222 if (dma_hal_verify() != 0U) {
223 ok = 1U;
224 }
225 }
226 if (ok != 0U) {
227 (void)ra8_board_uart_console_write(k_dma_hal_ok_msg, (size_t)(sizeof(k_dma_hal_ok_msg) - 1U));
228 (void)ra8_board_led_toggle(k_ra8_board_led1);
229 } else {
230 (void)ra8_board_uart_console_write(k_dma_hal_bad_msg,
231 (size_t)(sizeof(k_dma_hal_bad_msg) - 1U));
232 (void)ra8_board_led_toggle(k_ra8_board_led2);
233 }
234 ra8_delay_ms(k_dma_hal_period_ms);
235 }
236 dma_hal_panic_halt();
237 return 0;
238}
