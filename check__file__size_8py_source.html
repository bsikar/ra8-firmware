4"""Gate: source files shall not exceed a maintainability line cap (1000).
5
6The per-function NASA Power-of-10 Rule 4 cap (``check_function_size.py``,
760 lines/function) keeps individual routines small but says nothing about the
8size of the *file* they live in: a translation unit can stay Rule-4-clean while
9growing into a multi-thousand-line god-file, or the same body can be
10copy-pasted across siblings until each is enormous.
11
12Scope is derived, not listed (#359)
13-----------------------------------
14This gate used to carry a hand-written ``SCAN_ROOTS`` tuple that omitted
15``scripts/`` and a ``SOURCE_SUFFIXES`` tuple covering only C and C++. The
16documented 1000-line cap had therefore never applied to a single Python or
17shell file in the repository's history -- and it did not *fail* to apply, it
18reported success over a shrinking slice of the tree. A stale hardcoded list
19looks exactly like a clean one.
20
21The scope now comes from :mod:`lint_targets`, which derives it from
22``git ls-files`` plus per-file language detection (suffix, well-known basename,
23or ``#!`` shebang). A new top-level directory is covered the day it is added,
24and an extensionless executable -- the 670-line ``scripts/git/pre-commit``, for
25instance -- cannot escape by having no suffix.
26
27Every language ``lint_targets`` calls code is in scope: C/C++, Python, shell,
28CMake, YAML, Make and linker scripts. Markdown is deliberately NOT: prose is
29not code, a long reference document is not a maintainability defect, and the
30three docs over the cap (ROADMAP, HARDWARE_BRINGUP, JOF) are reference material
31that would be worse split up.
32
33Waivers
34-------
35A file may waive the cap with an in-file marker in its head:
36 @generated -- machine-emitted; the generator owns its length, not us.
37 FILE-SIZE-OK -- explicit, justified waiver for a hand-authored file.
38
39Run::
40
41 check_file_size.py # scan the whole tree
42 check_file_size.py path/to/file.c ... # scan listed files
43 check_file_size.py --selftest # prove it fires and stays quiet
44
45Exit 0 if every file is at or below the cap, exit 1 (with a table) otherwise.
46"""
47

57# Maintainability cap: a single source file should stay reviewable.
58THRESHOLD_LINES = 1000
59
60# In-file waiver markers; only the head is scanned so a stray occurrence deep
61# in data cannot silently waive a real offender.
62GENERATED_MARKERS = ("@generated", "FILE-SIZE-OK")
63HEAD_SCAN_LINES = 40
64

66def _is_waived(path: Path) -> bool:
67 """Return True if `path`'s head carries a generated / size-OK marker."""
68 try:
69 with path.open("r", encoding="utf-8", errors="replace") as handle:
70 for _ in range(HEAD_SCAN_LINES):
71 line = handle.readline()
72 if not line:
73 break
74 if any(marker in line for marker in GENERATED_MARKERS):
75 return True
76 except OSError:
77 return False
78 return False

81def _count_lines(path: Path) -> int:
82 """Return the physical line count of `path`, or -1 if unreadable."""
83 try:
84 text = path.read_text(encoding="utf-8", errors="replace")
85 except OSError:
86 return -1
87 return len(text.splitlines())

90def _targets_from_args(args: list[str]) -> list[Path]:
91 """Resolve explicitly named files / directories to a scan list."""
92 out: list[Path] = []
93 for raw in args:
94 path = Path(raw)
95 if not path.is_absolute():
96 path = REPO_ROOT / path
97 if path.is_dir():
98 out.extend(
99 p for p in path.rglob("*") if p.is_file() and language_of(_rel(p)) is not None
100 )
101 elif language_of(_rel(path)) is not None:
102 out.append(path)
103 return out

106def _rel(path: Path) -> str:
107 if path.is_absolute() and path.is_relative_to(REPO_ROOT):
108 return str(path.relative_to(REPO_ROOT))
109 return str(path)

112def _all_targets() -> list[Path]:
113 """Every first-party code file, derived from git + language detection."""
114 grouped = files_for()
115 return [REPO_ROOT / rel for paths in grouped.values() for rel in paths]

118def over_cap(targets: list[Path]) -> list[tuple[int, str]]:
119 """The (line count, path) of every unwaived file above the cap."""
120 over = []
121 for path in targets:
122 # Language is re-checked here rather than trusted from the caller, so
123 # the rule is the same however the file arrived -- derived scan,
124 # explicit argument, or selftest fixture.
125 if language_of(_rel(path)) is None or _is_waived(path):
126 continue
127 count = _count_lines(path)
128 if count > THRESHOLD_LINES:
129 over.append((count, _rel(path)))
130 over.sort(reverse=True)
131 return over
132

134# ---------------------------------------------------------------------------
135# Selftest -- asserts BOTH directions before the real scan is trusted.
136# ---------------------------------------------------------------------------

138
139# (filename, body, expected finding count, what the case proves)
140_OVER = THRESHOLD_LINES + 1
141
142_SELFTEST_CASES = (
143 ("over.c", "x\n" * _OVER, 1, "a C file over the cap fires"),
144 ("under.c", "x\n" * THRESHOLD_LINES, 0, "a C file exactly at the cap is clean"),
145 ("over.py", "x\n" * _OVER, 1, "a Python file over the cap fires"),
146 ("over.sh", "x\n" * _OVER, 1, "a shell file over the cap fires"),
147 ("over.cmake", "x\n" * _OVER, 1, "a CMake file over the cap fires"),
148 ("over.md", "x\n" * _OVER, 0, "Markdown is prose and out of scope"),
149 ("over.json", "x\n" * _OVER, 0, "a data file is out of scope"),
150 (
151 "gen.py",
152 "# @generated by a tool\n" + "x\n" * _OVER,
153 0,
154 "the @generated marker waives",
155 ),
156 (
157 "waived.py",
158 "# FILE-SIZE-OK: justified\n" + "x\n" * _OVER,
159 0,
160 "the FILE-SIZE-OK marker waives",
161 ),
162 (
163 "deep.py",
164 "# marker far below the head\n" * HEAD_SCAN_LINES + "@generated\n" + "x\n" * _OVER,
165 1,
166 "a marker below the head does not waive",
167 ),
168)

171def _selftest_cap(cases: tuple) -> list[str]:
172 """Run the threshold / waiver / scope fixtures and return any failures."""
173 failures: list[str] = []
174 with tempfile.TemporaryDirectory() as tmp:
175 root = Path(tmp)
176 for name, body, expected, description in cases:
177 path = root / name
178 path.write_text(body)
179 got = len(over_cap([path]))
180 path.unlink()
181 if got != expected:
182 failures.append(f" FAIL {description}: expected {expected}, got {got}")
183
184 # A shebang script with no suffix is code. This is the case a
185 # suffix-only scope silently drops -- and scripts/git/pre-commit is
186 # 677 lines of exactly it.
187 hook = root / "pre-commit-like"
188 hook.write_text("#!/usr/bin/env bash\n" + "x\n" * (THRESHOLD_LINES + 1))
189 if len(over_cap([hook])) != 1:
190 failures.append(" FAIL an extensionless shebang script is in scope")
191 return failures

194def _selftest_scope() -> tuple[list[str], dict[str, list[str]]]:
195 """Assert the live scope still resolves files for every known language.
196
197 Zero files for a language is always a broken enumeration, never a clean
198 tree -- the failure mode this checker was rewritten to end.
199 """
200 grouped = files_for()
201 failures = [
202 f" FAIL live scope: language {lang!r} resolved to zero files"
203 for lang, paths in grouped.items()
204 if not paths
205 ]
206 return failures, grouped
207
208
209def _selftest() -> int:
210 """Prove the cap fires, the waivers work, and the scope reaches every language."""
211 failures = _selftest_cap(_SELFTEST_CASES)
212 scope_failures, grouped = _selftest_scope()
213 failures.extend(scope_failures)
214
215 if failures:
216 print("check_file_size.py: --selftest FAILED", file=sys.stderr)
217 print("\n".join(failures), file=sys.stderr)
218 return 1
219 print(
220 f"check_file_size.py: --selftest OK ({len(_SELFTEST_CASES) + 1} cases both "
221 f"directions; live scope covers {len(grouped)} language(s), "
222 f"{sum(len(p) for p in grouped.values())} file(s))"
223 )
224 return 0

227def main(argv: list[str]) -> int:
228 """Fail any file over the line cap, or with ``--selftest`` prove the gate still fires.
229
230 Note the THREE distinct exit codes. An empty target set exits 2, not 0:
231 this gate derives its scope from ``git ls-files`` (#359) and so an empty
232 scan means the enumeration broke, which must never be reported as a clean
233 tree -- that is precisely how this checker spent its early life passing
234 over a scope that omitted every Python and shell file.
235
236 Returns 0 when every scanned file is within the cap, 1 when one or more
237 exceed it, and 2 when there was nothing to scan at all.
238 """
239 args = argv[1:]
240 if "--selftest" in args:
241 return _selftest()
242
243 targets = _targets_from_args(args) if args else _all_targets()
244 if not targets:
245 print("check_file_size.py: FATAL -- no files to scan", file=sys.stderr)
246 return 2
247
248 over = over_cap(targets)
249 if not over:
250 print(
251 f"check_file_size.py: {len(targets)} file(s) scanned, "
252 f"none over {THRESHOLD_LINES} lines."
253 )
254 return 0
255
256 print(
257 f"check_file_size.py: {len(over)} file(s) exceed the {THRESHOLD_LINES}-line cap:\n",
258 file=sys.stderr,
259 )
260 print(" lines file", file=sys.stderr)
261 for count, path in over:
262 print(f" {count:5d} {path}", file=sys.stderr)
263 print(
264 "\nSplit the file along its responsibilities, or extract a shared "
265 "helper if the bulk is duplicated. Generated files may carry a "
266 "@generated marker; a justified hand-authored file may use FILE-SIZE-OK.",
267 file=sys.stderr,
268 )
269 return 1
