56void ra8_dfu_launch(uintptr_t src, uint32_t img_len, uint32_t entry)
57{
58 if (src == 0U) {
59 return; /* nothing to copy */
60 }
61 if (!ra8_dfu_run_target_valid(entry, img_len)) {
62 return; /* entry not the run base, or bad length -> caller drops to fallback */
63 }
64
65#ifdef RA8_ENABLE_ROOT_OF_TRUST
66 /* Root-of-trust gate (opt-in, see ra8_rot.h): authenticate the signed image
67 * (SHA-256 + ECDSA-P256) BEFORE copying it to the run base. Default-deny --
68 * any failure (missing trailer, malformed trailer, tampered body, or an
69 * invalid signature) returns to the caller's fallback path without copying or
70 * branching. The signed image is `[ body (img_len) ] [ ra8_rot_trailer_t ]`,
71 * so the trailer sits immediately after the body. With the flag OFF this gate
72 * is absent and the launch proceeds on the CRC32 integrity check alone. */
73 const ra8_rot_trailer_t* trailer = ra8_rot_trailer_after((const void*)src, img_len);
74 if (ra8_rot_verify_image((const uint8_t*)src, img_len, trailer) != k_ra8_ok) {
75 return; /* DEFAULT-DENY: unauthenticated image -> do not launch */
76 }
77
78 /* Anti-rollback gate (downgrade protection): the image is now authentic, so
79 * its trailer is readable. Refuse a version older than the highest ever
80 * accepted (see ra8_dfu_antirollback.h). DEFAULT-DENY on a downgrade OR a store
81 * READ FAULT -- either returns to the caller's fallback path without copying or
82 * branching. A fresh/unprovisioned device reads its erased monotonic counter as
83 * 0, so it ACCEPTS its first image (which becomes the new floor) and is NOT
84 * bricked; only a genuine downgrade or an unreadable counter denies launch. */
85 if (ra8_rot_antirollback_verify(ra8_rot_antirollback_default_store(), trailer->img_version) !=
86 k_ra8_ok) {
87 return; /* DEFAULT-DENY: downgrade / unprovisioned counter -> do not launch */
88 }
89#endif /* RA8_ENABLE_ROOT_OF_TRUST */
90
91#ifndef RA8_OFF_TARGET
92 const volatile uint32_t* s = (const volatile uint32_t*)src;
93 volatile uint32_t* d = (volatile uint32_t*)(uintptr_t)k_ra8_dfu_run_base;
94 const uint32_t words = img_len / (uint32_t)sizeof(uint32_t);
95
96 __asm__ volatile("cpsid i" ::: "memory");
97 /* Bounded by img_len (<= k_ra8_dfu_img_max via ra8_dfu_run_target_valid): a
98 * statically bounded copy -- NASA Rule 2 compliant. */
99 for (uint32_t i = 0U; i < words; ++i) {
100 d[i] = s[i];
101 }
102 __asm__ volatile("dsb 0xF" ::: "memory"); /* image stores reach SRAM before fetch */
103
104 const uint32_t initial_sp = d[0];
105 const uint32_t reset_entry = d[1];
106
107 *(volatile uint32_t*)(uintptr_t)k_ra8_dfu_launch_vtor_addr = (uint32_t)k_ra8_dfu_run_base;
108 __asm__ volatile("dsb 0xF\n isb 0xF\n" ::: "memory");
109 __asm__ volatile("msr msp, %0\n"
110 "bx %1\n"
111 :
112 : "r"(initial_sp), "r"(reset_entry)
113 : "memory");
114 __builtin_unreachable();
115#endif /* !RA8_OFF_TARGET */
116}
