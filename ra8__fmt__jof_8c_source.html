95static bool ra8_fmt_jof_is_webp(const ra8_fmt_blob_t* src)
96{
97 if ((src == nullptr) || (src->bytes == nullptr) || (src->len < (size_t)k_fmt_webp_head_len)) {
98 return false;
99 }
100 return (memcmp(&src->bytes[k_fmt_webp_riff_ofs], s_fmt_webp_riff, sizeof(s_fmt_webp_riff)) ==
101 0) &&
102 (memcmp(&src->bytes[k_fmt_webp_fourcc_ofs], s_fmt_webp_webp, sizeof(s_fmt_webp_webp)) ==
103 0);
104}

138RA8_INTERNAL
139static ra8_err_t fmt_jof_pull(void* ctx, uint8_t* buf, size_t cap, size_t* got)
140{
141 fmt_pull_ctx_t* s = (fmt_pull_ctx_t*)ctx;
142 size_t n = s->len - s->pos;
143 if (n > cap) {
144 n = cap;
145 }
146 (void)memcpy(buf, &s->data[s->pos], n);
147 s->pos += n;
148 *got = n;
149 return k_ra8_ok;
150}

178RA8_INTERNAL
179static ra8_err_t fmt_jof_carve_webp(const ra8_fmt_blob_t* src,
180 uint16_t max_w,
181 uint16_t max_h,
182 uint8_t** out_work,
183 size_t* out_cap)
184{
185 *out_work = nullptr;
186 *out_cap = 0U;
187 if (!ra8_fmt_jof_is_webp(src)) {
188 return k_ra8_ok;
189 }
190 const uint32_t need = ra8_jof_webp_work_bytes(max_w, max_h, (uint32_t)src->len);
191 if (need == 0U) {
192 return k_ra8_err_invalid_size;
193 }
194 uint8_t* mem = (uint8_t*)malloc((size_t)need);
195 if (mem == nullptr) {
196 return k_ra8_err_no_mem;
197 }
198 *out_work = mem;
199 *out_cap = (size_t)need;
200 return k_ra8_ok;
201}

248RA8_INTERNAL
249static ra8_err_t fmt_jof_run_produce(const ra8_fmt_blob_t* src,
250 uint16_t max_w,
251 uint16_t max_h,
252 uint16_t tile_w,
253 uint16_t tile_h,
254 uint8_t codec,
255 uint32_t work_cap,
256 ra8_jof_memstore_t* store,
257 ra8_jof_info_t* out_info)
258{
259 uint8_t* work = (uint8_t*)malloc((size_t)work_cap);
260 if (work == nullptr) {
261 return k_ra8_err_no_mem;
262 }
263 uint8_t* webp_work = nullptr;
264 size_t webp_cap = 0U;
265 const ra8_err_t carve_rc = fmt_jof_carve_webp(src, max_w, max_h, &webp_work, &webp_cap);
266 if (carve_rc != k_ra8_ok) {
267 free(work);
268 return carve_rc;
269 }
270 fmt_pull_ctx_t pull = {.data = src->bytes, .len = src->len, .pos = 0U};
271 const ra8_jof_produce_cfg_t cfg = {.pull = fmt_jof_pull,
272 .pull_ctx = &pull,
273 .sink = ra8_jof_memstore_sink,
274 .sink_ctx = store,
275 .tile_w = tile_w,
276 .tile_h = tile_h,
277 .codec = codec,
278 .max_width = 0U,
279 .max_height = 0U,
280 .work = work,
281 .work_cap = work_cap,
282 .webp_work = webp_work,
283 .webp_work_cap = webp_cap};
284 const ra8_err_t rc = ra8_jof_produce(&cfg, out_info);
285 free(work);
286 free(webp_work);
287 return rc;
288}

290ra8_err_t ra8_fmt_jof_produce(const ra8_fmt_blob_t* src,
291 uint16_t max_w,
292 uint16_t max_h,
293 uint16_t tile_w,
294 uint16_t tile_h,
295 uint8_t codec,
296 ra8_fmt_blob_t* out_atlas,
297 ra8_jof_info_t* out_info)
298{
299 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
300 RA8_CHECK_NULL_PTR(out_atlas, s_tag, "out_atlas must not be nullptr");
301 RA8_CHECK_NULL_PTR(out_info, s_tag, "out_info must not be nullptr");
302 *out_atlas = (ra8_fmt_blob_t){};
303 const uint32_t work_cap = ra8_jof_work_bytes(max_w, max_h, tile_w, tile_h);
304 if (work_cap == 0U) {
305 return k_ra8_err_invalid_size;
306 }
307 /* The atlas can expand slightly over the source for incompressible input;
308 * size the sink from the decoded worst case plus the index and trailer. */
309 const size_t sink_cap = ra8_fmt_jof_sink_cap(src->len);
310 uint8_t* sink = (uint8_t*)malloc(sink_cap);
311 if (sink == nullptr) {
312 return k_ra8_err_no_mem;
313 }
314 ra8_jof_memstore_t store = {.buf = sink, .cap = sink_cap, .len = 0U};
315 const ra8_err_t rc =
316 fmt_jof_run_produce(src, max_w, max_h, tile_w, tile_h, codec, work_cap, &store, out_info);
317 if (rc != k_ra8_ok) {
318 free(sink);
319 return rc;
320 }
321 out_atlas->bytes = sink;
322 out_atlas->len = store.len;
323 return k_ra8_ok;
324}

326ra8_err_t ra8_fmt_jof_convert(const ra8_fmt_blob_t* src, const ra8_fmt_opts_t* opts)
327{
328 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
329 RA8_CHECK_NULL_PTR(opts, s_tag, "opts must not be nullptr");
330 if (opts->out_path == nullptr) {
331 (void)fprintf(opts->report, "ra8_fmt: convert needs --out\n");
332 return k_ra8_err_invalid_arg;
333 }
334 uint16_t w = 0U;
335 uint16_t h = 0U;
336 ra8_err_t rc = ra8_fmt_jof_probe(src, &w, &h);
337 if (rc != k_ra8_ok) {
338 (void)fprintf(opts->report, "ra8_fmt: cannot decode source image (rc=%d)\n", (int)rc);
339 return rc;
340 }
341 const uint16_t band = (h < (uint16_t)k_fmt_jof_band_h) ? h : (uint16_t)k_fmt_jof_band_h;
342 ra8_fmt_blob_t atlas = {};
343 ra8_jof_info_t info = {};
344 rc = ra8_fmt_jof_produce(src, w, h, w, band, (uint8_t)k_ra8_jof_codec_deflate, &atlas, &info);
345 if (rc != k_ra8_ok) {
346 (void)fprintf(opts->report, "ra8_fmt: transcode failed (rc=%d)\n", (int)rc);
347 return rc;
348 }
349 rc = ra8_fmt_write_file(opts->out_path, atlas.bytes, atlas.len);
350 (void)fprintf(opts->report,
351 "convert: %ux%u bpp=%u band=%u tiles=%u -> %s (%zu bytes)\n",
352 (unsigned)info.width,
353 (unsigned)info.height,
354 (unsigned)info.bpp,
355 (unsigned)info.tile_h,
356 (unsigned)info.tile_count,
357 opts->out_path,
358 atlas.len);
359 ra8_fmt_blob_free(&atlas);
360 return rc;
361}
