85static void br_fill(ra8_jpeg_bitreader_t* br)
86{
87 while (br->nbits <= k_jpeg_reservoir_lo && !br->had_eoi) {
88 if (br->pos >= br->len) {
89 br->had_eoi = 1U;
90 return;
91 }
92 uint8_t b = br->buf[br->pos];
93 br->pos++;
94 if (b == (uint8_t)k_ra8_jpeg_marker_byte) {
95 if (br->pos >= br->len) {
96 br->had_eoi = 1U;
97 return;
98 }
99 uint8_t s = br->buf[br->pos];
100 br->pos++;
101 if (s != 0U) {
102 /* A real marker; rewind so caller can inspect. */
103 br->pos -= 2U;
104 br->had_eoi = 1U;
105 return;
106 }
107 /* Stuffed 0xFF byte. */
108 }
109 br->acc = (br->acc << k_ra8_jpeg_byte_shift) | (uint32_t)b;
110 br->nbits = (uint8_t)(br->nbits + k_ra8_jpeg_byte_shift);
111 }
112}

136int32_t ra8_jpeg_sw_br_get_bits(ra8_jpeg_bitreader_t* br, uint8_t n)
137{
138 if (n == 0U) {
139 return 0;
140 }
141 if (br->nbits < n) {
142 br_fill(br);
143 if (br->nbits < n) {
144 return -1;
145 }
146 }
147 uint32_t v = (br->acc >> (br->nbits - n)) & ((1U << n) - 1U);
148 br->nbits = (uint8_t)(br->nbits - n);
149 br->acc &= (br->nbits == 0U) ? 0U : ((1U << br->nbits) - 1U);
150 return (int32_t)v;
151}

178static void htab_build_lookup(ra8_jpeg_htab_t* h)
179{
180 uint16_t j = 0U;
181 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_huff_lengths; i++) {
182 if (h->bits[i] == 0U) {
183 h->maxcode[i] = -1;
184 } else {
185 h->valptr[i] = j;
186 h->mincode[i] = (int32_t)h->huffcode[j];
187 j = (uint16_t)((j + h->bits[i]) - 1U);
188 h->maxcode[i] = (int32_t)h->huffcode[j];
189 j++;
190 }
191 }

194void ra8_jpeg_sw_htab_build(ra8_jpeg_htab_t* h)
195{
196 /* T.81 Annex C, "Generation of size table" + "Generation of code
197 * table". */
198 uint16_t k = 0U;
199 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_huff_lengths; i++) {
200 for (uint8_t j = 0U; j < h->bits[i]; j++) {
201 h->huffsize[k] = (uint8_t)(i + 1U);
202 k++;
203 }
204 }
205 h->huffsize[k] = 0U;
206 h->total = k;
207
208 uint16_t code = 0U;
209 uint8_t si = (k > 0U) ? h->huffsize[0] : 0U;
210 uint16_t kk = 0U;
211 while (h->huffsize[kk] != 0U) {
212 while (h->huffsize[kk] == si) {
213 h->huffcode[kk] = code;
214 code++;
215 kk++;
216 }
217 if (h->huffsize[kk] == 0U) {
218 break;
219 }
220 do {
221 code = (uint16_t)(code << 1U);
222 si++;
223 } while (h->huffsize[kk] != si);
224 }
225
226 /* T.81 Annex F.2.2.3 mincode/maxcode/valptr build. */
227 htab_build_lookup(h);
228}

252int32_t ra8_jpeg_sw_htab_decode(ra8_jpeg_bitreader_t* br, const ra8_jpeg_htab_t* h)
253{
254 br_fill(br);
255 if (br->nbits == 0U) {
256 return -1;
257 }
258 /* Single-bit greedy lookup per F.2.2.3 "Decoder code-length
259 * algorithm". */
260 int32_t code = ra8_jpeg_sw_br_get_bits(br, 1U);
261 /* After br->nbits == 0 guard above, br->nbits >= 1; br_get_bits(br, 1)
262 * with nbits >= 1 always returns 0 or 1 without invoking br_fill, so
263 * code < 0 is unreachable on any host input. */
264 if (code < 0) {
265 return -1; /* GCOVR_EXCL_LINE */
266 }
267 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_huff_lengths; i++) {
268 if (code <= h->maxcode[i]) {
269 uint16_t j = (uint16_t)(h->valptr[i] + (uint16_t)(code - h->mincode[i]));
270 if (j >= h->total) {
271 return -1;
272 }
273 return (int32_t)h->vals[j];
274 }
275 int32_t bit = ra8_jpeg_sw_br_get_bits(br, 1U);
276 if (bit < 0) {
277 return -1;
278 }
279 code = (code << 1) | bit;
280 }
281 return -1;
282}

340static void inv_dct_1d_norm(const int32_t* in, int32_t* out)
341{
342 for (uint8_t n = 0U; n < (uint8_t)k_ra8_jpeg_block_dim; n++) {
343 int64_t s = 0;
344 for (uint8_t k = 0U; k < (uint8_t)k_ra8_jpeg_block_dim; k++) {
345 int64_t v = (int64_t)in[k] * (int64_t)s_dct_w_q14[k];
346 s += (v * (int64_t)s_dct_cos_q14[k][n]) >> k_jpeg_q14_shift;
347 }
348 int64_t r = (s + (1LL << k_jpeg_idct_p2_bias_sh)) >> k_jpeg_q14_shift;
349 out[n] = (int32_t)r;
350 }

508static ra8_err_t
509dims_next_marker(const uint8_t* jpeg_buf, uint32_t jpeg_len, uint32_t* i, uint16_t* out_mk)
510{
511 if (jpeg_buf[*i] != (uint8_t)k_ra8_jpeg_marker_byte) {
512 return k_ra8_err_protocol_error;
513 }
514 /* Skip pad bytes. */
515 // mcdc-deactivated: dims_next_marker JPEG marker-pad skip; the jpeg_len bound is checked by the caller's enclosing while and the marker-byte equality follows the JFIF spec (0xFF padding bytes always within the segment); both conditions are co-dependent in any well-formed stream.
516 while (*i < jpeg_len && jpeg_buf[*i] == (uint8_t)k_ra8_jpeg_marker_byte) {
517 (*i)++;
518 }
519 if (*i >= jpeg_len) {
520 return k_ra8_err_protocol_error;
521 }
522 uint8_t m = jpeg_buf[*i];
523 *out_mk = (uint16_t)(((uint16_t)k_jpeg_byte_mask << k_ra8_jpeg_byte_shift) | m);
524 (*i)++;
525 return k_ra8_ok;
526}

607static ra8_err_t dims_step(const uint8_t* jpeg_buf,
608 uint32_t jpeg_len,
609 uint32_t* i,
610 bool* done,
611 uint16_t* out_w,
612 uint16_t* out_h)
613{
614 uint16_t mk = 0U;
615 ra8_err_t e = dims_next_marker(jpeg_buf, jpeg_len, i, &mk);
616 if (e != k_ra8_ok) {
617 return e;
618 }
619 if (mk == (uint16_t)k_ra8_jpeg_marker_soi || mk == (uint16_t)k_ra8_jpeg_marker_eoi) {
620 return k_ra8_ok;
621 }
622 if (*i + 2U > jpeg_len) {
623 return k_ra8_err_protocol_error;
624 }
625 uint16_t seglen = read_be16(&jpeg_buf[*i]);
626 if (seglen < 2U || (uint32_t)seglen > jpeg_len - *i) {
627 return k_ra8_err_protocol_error;
628 }
629 if (mk == (uint16_t)k_ra8_jpeg_marker_sof0) {
630 e = dims_parse_sof0(jpeg_buf, *i, seglen, out_w, out_h);
631 if (e == k_ra8_ok) {
632 *done = true;
633 }
634 return e;
635 }
636 // mcdc-deactivated: dims_step unsupported-SOFn detector; the 4-condition AND identifies SOF1..SOF15 except DHT/SOF8, but markers >= 0xFFC0 are by definition <= 0xFFCF in the JPEG marker space (range is 16 values), and SOF0 is handled above -- the upper-bound condition cannot independently flip on any reachable SOFn marker.
637 if (mk >= k_jpeg_marker_sof_lo && mk <= k_jpeg_marker_sof_hi &&
638 mk != (uint16_t)k_ra8_jpeg_marker_dht && mk != k_jpeg_marker_jpg) {
639 return k_ra8_err_not_supported;
640 }
641 *i += seglen;
642 return k_ra8_ok;
643}

645ra8_err_t ra8_jpeg_sw_get_dimensions(const uint8_t* jpeg_buf,
646 uint32_t jpeg_len,
647 uint16_t* out_w,
648 uint16_t* out_h)
649{
650 RA8_CHECK_NULL_PTR(jpeg_buf, s_tag, "jpeg_buf is NULL");
651 RA8_CHECK_NULL_PTR(out_w, s_tag, "out_w is NULL");
652 RA8_CHECK_NULL_PTR(out_h, s_tag, "out_h is NULL");

657 return k_ra8_err_protocol_error;

664 return e;
665 }
666 if (done) {
667 return k_ra8_ok;
668 }
669 }
670 return k_ra8_err_protocol_error;
671}
