393static void wd_print_hex(uint32_t value)
394{
395 uint8_t buf[k_wd_hex_nibbles];
396 for (uint32_t i = 0U; i < (uint32_t)k_wd_hex_nibbles; i++) {
397 const uint32_t shift = ((uint32_t)k_wd_hex_nibbles - 1U - i) * (uint32_t)k_wd_nibble_bits;
398 const uint32_t nib = (value >> shift) & (uint32_t)k_wd_nibble_mask;
399 buf[i] = (uint8_t)((nib < (uint32_t)k_wd_dec_ten) ? ('0' + nib) : ('A' + (nib - k_wd_dec_ten)));
400 }
401 wd_print(buf, (uint32_t)k_wd_hex_nibbles);
402}

405static void wd_print_uint(uint32_t value)
406{
407 uint8_t buf[k_wd_dec_ten];
408 uint32_t n = 0U;
409 if (value == 0U) {
410 buf[n] = '0';
411 n++;
412 }
413 while ((value > 0U) && (n < (uint32_t)k_wd_dec_ten)) {
414 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_wd_dec_ten));
415 n++;
416 value /= (uint32_t)k_wd_dec_ten;
417 }
418 for (uint32_t i = 0U; i < n; i++) {
419 wd_print(&buf[n - 1U - i], 1U);
420 }
421}

424static uint32_t wd_framebuffer_hash(void)
425{
426 const uint8_t* p = (const uint8_t*)s_framebuffer;
427 const size_t n = sizeof(s_framebuffer);
428 uint32_t hsh = (uint32_t)k_wd_fnv_offset;
429 for (size_t i = 0U; i < n; i++) {
430 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_wd_fnv_prime;
431 }
432 return hsh;
433}

441static void wd_clock_fmt(void)
442{
443 uint32_t v = s_clock_val;
444 char tmp[k_wd_clock_len];
445 uint32_t n = 0U;
446 if (v == 0U) {
447 tmp[n] = '0';
448 n++;
449 }
450 while ((v > 0U) && (n < (uint32_t)(k_wd_clock_len - 1U))) {
451 tmp[n] = (char)('0' + (v % (uint32_t)k_wd_dec_ten));
452 n++;
453 v /= (uint32_t)k_wd_dec_ten;
454 }
455 for (uint32_t i = 0U; i < n; i++) {
456 s_clock[i] = tmp[n - 1U - i];
457 }
458 s_clock[n] = '\0';
459}

587static bool wd_check_full(uint32_t* out_crc, uint16_t* out_dirty)
588{
589 wd_invalidate_all(k_ra8_widget_refresh_quality);
590 ra8_ui_rect_t dmg = {};
591 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
592 uint16_t nd = 0U;
593 if (wd_compose(&dmg, &hint, &nd) != k_ra8_ok) {
594 return false;
595 }
596 if (nd != (uint16_t)k_wd_root_n) {
597 return false;
598 }
599 if ((dmg.w != (int32_t)k_wd_fb_w) || (dmg.h != (int32_t)k_wd_fb_h)) {
600 return false;
601 }
602 if (hint != k_ra8_widget_refresh_quality) {
603 return false;
604 }
605 *out_crc = wd_framebuffer_hash();
606 *out_dirty = nd;
607 return true;
608}

611static bool wd_check_partial(uint32_t* out_crc, uint16_t* out_dirty)
612{
613 s_clock_val++;
614 wd_clock_fmt();
615 (void)ra8_widget_invalidate(&s_root_kids[k_wd_w_status], k_ra8_widget_refresh_fast);
616 ra8_ui_rect_t dmg = {};
617 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
618 uint16_t nd = 0U;
619 if (wd_compose(&dmg, &hint, &nd) != k_ra8_ok) {
620 return false;
621 }
622 if (nd != 1U) {
623 return false;
624 }
625 if ((dmg.x != 0) || (dmg.y != 0) || (dmg.w != (int32_t)k_wd_fb_w)) {
626 return false;
627 }
628 if (dmg.h != (int32_t)k_wd_status_h) {
629 return false;
630 }
631 if (hint != k_ra8_widget_refresh_fast) {
632 return false;
633 }
634 *out_crc = wd_framebuffer_hash();
635 *out_dirty = nd;
636 return true;
637}

640static bool wd_selfcheck(uint32_t* crc0, uint16_t* d0, uint32_t* crc1, uint16_t* d1)
641{
642 if (!wd_check_full(crc0, d0)) {
643 return false;
644 }
645 if (!wd_check_partial(crc1, d1)) {
646 return false;
647 }
648 return (*crc0 != *crc1); /* the advanced clock changes the status band */
649}

736 }

747 }

750 wd_tick_live();
751 ra8_delay_ms((uint32_t)k_wd_frame_ms);
752 }
