373static ra8_err_t internal_program_and_start_pll1(void)
374{
375 /* PLLMUL is encoded as (integer*4 + quarters) shifted by 6, which
376 * simultaneously fills PLLMULNF[7:6] and PLLMUL[16:8]. */
377 const uint32_t mul_quarters =
378 ((uint32_t)k_ra8_cgc_pllmul_int_default * k_ra8_cgc_quarters_per_unit) +
379 k_ra8_cgc_pllmul_quarters;
380
381 const uint32_t pllccr =
382 (((mul_quarters & k_ra8_pllccr_mask_quarters) << k_ra8_pllccr_shift_quarters) |
383 ((uint32_t)k_ra8_plsrcsel_main << k_ra8_pllccr_shift_plsrcsel) |
384 ((uint32_t)k_ra8_plidiv_div3 & k_ra8_pllccr_mask_plidiv));
385
386 const uint16_t pllccr2 = (uint16_t)(((uint16_t)k_ra8_plodiv_div5 << k_ra8_pllccr2_shift_plodivr) |
387 ((uint16_t)k_ra8_plodiv_div6 << k_ra8_pllccr2_shift_plodivq) |
388 ((uint16_t)k_ra8_plodiv_div2 << k_ra8_pllccr2_shift_plodivp));
389
390 /* HUM Ch 9.2.6 "PLLCCR : PLL Clock Control Register" p 331 */
391 /* Cross-reference: FSP bsp_clocks.c. */
392 *ra8_sys_pllccr() = pllccr;
393 /* HUM Ch 9.2.7 "PLLCCR2 : PLL Clock Control Register 2" p 332 */
394 /* Cross-reference: FSP bsp_clocks.c. */
395 *ra8_sys_pllccr2() = pllccr2;
396
397 /* HUM Ch 9.2.8 "PLLCR : PLL Control Register" p 333 */
398 /* Cross-reference: FSP bsp_clocks.c. */
399 *ra8_sys_pllcr() = k_ra8_pllcr_run;
400 /* HUM Ch 9.2.21 "OSCSF : Oscillation Stabilization Flag Register" p 344 */
401 /* Cross-reference: FSP bsp_clocks.c. */
402 return ra8_cgc_wait_oscsf_set(k_ra8_oscsf_bit_pll1sf);
403}

435static ra8_err_t internal_wait_mrm_freq(volatile uint32_t* reg, uint32_t key, uint32_t freq_mhz)
436{
437 for (uint32_t i = 0U; i < k_ra8_cgc_mrm_spin_limit; i++) {
438#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
439 /* Host MMIO fault seam: the loop-exit decision comes from
440 * ra8_fake_mmio_wait_eval (satisfied on the first poll unless a test
441 * arms a fault) because plain host RAM cannot strip the key byte
442 * on readback the way the silicon latch does. */
443 if (ra8_fake_mmio_wait_eval(reg, i, (*reg == freq_mhz))) {
444 return k_ra8_ok;
445 }
446#else
447 if (*reg == freq_mhz) {
448 return k_ra8_ok;
449 }
450#endif
451 *reg = key | freq_mhz;
452 }
453 return k_ra8_err_hw_timeout;
454}

482static ra8_err_t internal_set_mrm_wait_states(uint32_t mriclk_hz, uint32_t mrpclk_hz)
483{

492 /* Cross-reference: FSP bsp_clocks.c. */
493 ra8_err_t err = internal_wait_mrm_freq(ra8_mrms_mrcfreq(), k_ra8_mrcfreq_key, mri_mhz);

495 return err;

604static ra8_err_t internal_route_sciclk(void)
605{
606 volatile uint8_t* const ckcr = ra8_sys_scickcr();
607 volatile uint8_t* const divcr = ra8_sys_scickdivcr();
608
609 /* HUM Ch 9.2.54 "SCICKCR : SCI Clock Control Register" p 368 */
610 /* step 1 */
611 *ckcr = (uint8_t)(*ckcr | k_ra8_scickcr_cksreq);
612 const ra8_err_t req_err =
613 ra8_hw_wait_flag_set8(ckcr, (uint8_t)k_ra8_scickcr_cksrdy, (uint32_t)k_ra8_cgc_scik_spin_limit);
614 if (req_err != k_ra8_ok) {
615 return req_err;
616 }
617 /* HUM Ch 9.2.49 "SCICKDIVCR : SCI Clock Division Control Register" p 365 */
618 /* step 2 */
619 *divcr = k_ra8_scickdivcr_div4;
620 *ckcr = (uint8_t)(k_ra8_scickcr_sel_pll1r | k_ra8_scickcr_cksreq);
621 /* Step 3: clear CKSREQ to start the new clock. */
622 *ckcr = k_ra8_scickcr_sel_pll1r;
623 return ra8_hw_wait_flag_clear8(ckcr,
624 (uint8_t)k_ra8_scickcr_cksrdy,
625 (uint32_t)k_ra8_cgc_scik_spin_limit);
626}

686static ra8_err_t internal_cgc_init_protected(void)
687{
688 ra8_err_t err = internal_start_main_osc();
689 if (err == k_ra8_ok) {
690 err = internal_set_vscr_not_high_v();
691 }
692 if (err == k_ra8_ok) {
693 err = internal_stop_pll1();
694 }
695 if (err == k_ra8_ok) {
696 err = internal_program_and_start_pll1();
697 }
698 if (err == k_ra8_ok) {
699 /* Step 6: MRMS wait-state latches must come BEFORE the SCKSCR
700 * switch -- the new MRICLK and MRPCLK rates kick in the moment
701 * SCKDIVCR2 is committed. */
702 err = internal_set_mrm_wait_states(k_ra8_mriclk_hz, k_ra8_fclk_hz);
703 }
704 if (err == k_ra8_ok) {
705 /* Step 7: the SRAM wait state, for exactly the same reason and in
706 * exactly the same window as step 6. HUM Ch 58.3.7 "Wait State"
707 * p 3540 is unambiguous -- above half the maximum ICLK, "when the
708 * wait is not inserted, the operation is not guaranteed" -- and
709 * SRAMWTSC.WTEN resets to 0 while this sequence takes ICLK to
710 * k_ra8_iclk_hz, its maximum. Programming it after the SCKSCR
711 * switch would leave a window in which .data, .bss and the stack
712 * are read at full speed with no wait; the ONE fault it is known
713 * to produce is a single dropped bit in a value read back out of
714 * SRAM, so that window has to be empty, not short (tracker #524). */
715 err = ra8_sram_set_wait_state_for_clock(k_ra8_iclk_hz, (uint32_t)k_ra8_iclk_max_hz);
716 }
717 if (err == k_ra8_ok) {
718 internal_program_dividers();
719 /* HUM Ch 9.2.5 "SCKSCR : System Clock Source Control Register" p 330 */
720 *ra8_sys_sckscr() = k_ra8_cksel_pll1;
721 }
722 return err;
723}

725ra8_err_t ra8_cgc_init(void)
726{
727 ra8_log_info(s_tag, "cgc_init -> XTAL -> PLL1 (1 GHz)");
728
729 /* Step 1 happens outside the PRCR window: MRMS isn't a protected block. */
730 internal_clear_pfb();
731
732 ra8_err_t err = k_ra8_ok;
733 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
734 {
735 err = internal_cgc_init_protected();
736 }
737
738 if (err != k_ra8_ok) {
739 ra8_log_error_val(s_tag, "cgc_init failed", (uint32_t)err);
740 return err;
741 }
742
743 /* Step 10 happens after SCKSCR is on PLL1 -- MRMS isn't protected. */
744 internal_set_pfb();
745
746 /* Step 11 needs PRCR unlocked again. */
747 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
748 {
749 err = internal_route_sciclk();
750 }
751 if (err != k_ra8_ok) {
752 ra8_log_error_val(s_tag, "sciclk route failed", (uint32_t)err);
753 return err;
754 }
755
756 internal_publish_clocks();
757 ra8_log_info(s_tag, "system clock = PLL1, CPUCLK0 = 1 GHz");
758 return k_ra8_ok;
759}

812ra8_err_t ra8_cgc_switch_pll1_target(uint32_t new_cpuclk_hz)
813{
814 if (new_cpuclk_hz == 0U) {
815 return k_ra8_err_invalid_arg;
816 }
817 ra8_log_info_val(s_tag, "switch pll1 target", new_cpuclk_hz);
818
819 /* Step 1: temporarily fall back to MOCO so the CPU is not clocked
820 * from PLL1 while we reprogramme it. */
821 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
822 {
823 *ra8_sys_sckscr() = k_ra8_cksel_moco;
824 }
825
826 /* Step 2: stop + restart PLL1 with the existing field encodings.
827 * For now we only support the FSP-quickstart values; a future
828 * change can compute PLLCCR / PLLCCR2 from `new_cpuclk_hz`. */
829 ra8_err_t pll_err = k_ra8_ok;
830 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
831 {
832 pll_err = internal_stop_pll1();
833 if (pll_err == k_ra8_ok) {
834 pll_err = internal_program_and_start_pll1();
835 }
836 if (pll_err != k_ra8_ok) {
837 ra8_log_error_val(s_tag, "pll1 retune failed", (uint32_t)pll_err);
838 /* fall through: leave SCKSCR on MOCO so the CPU keeps running. */
839 }
840 }
841
842 /* Step 3: switch SCKSCR back to PLL1 if the lock succeeded, then
843 * republish the clock tree. */
844 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
845 {
846 if (pll_err == k_ra8_ok) {
847 *ra8_sys_sckscr() = k_ra8_cksel_pll1;
848 }
849 }
850 internal_publish_clocks();
851 s_clock_hz[k_ra8_clock_id_cpuclk0] = new_cpuclk_hz;
852 return k_ra8_ok;
853}
