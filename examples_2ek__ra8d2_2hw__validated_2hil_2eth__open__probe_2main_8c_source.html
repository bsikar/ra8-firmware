220static void eop_print_hex(uint32_t value)
221{
222 eop_print(k_eop_msg_hex_lead, (uint32_t)sizeof(k_eop_msg_hex_lead) - 1U);
223 for (uint32_t i = 0U; i < (uint32_t)k_eop_hex_chars; i++) {
224 const uint32_t shift = ((uint32_t)k_eop_hex_chars - 1U - i) * (uint32_t)k_eop_hex_shift;
225 const uint32_t digit = (value >> shift) & (uint32_t)k_eop_hex_mask;
226 eop_print(&k_eop_hex_digits[digit], 1U);
227 }
228}

246static void eop_print_uint(uint32_t value)
247{
248 uint8_t buf[k_eop_dec_max];
249 uint32_t n = 0U;
250 if (value == 0U) {
251 buf[n] = (uint8_t)'0';
252 n++;
253 }
254 while ((value > 0U) && (n < (uint32_t)k_eop_dec_max)) {
255 buf[n] = (uint8_t)((uint32_t)'0' + (value % (uint32_t)k_eop_dec_base));
256 n++;
257 value /= (uint32_t)k_eop_dec_base;
258 }
259 for (uint32_t i = 0U; i < n; i++) {
260 eop_print(&buf[n - 1U - i], 1U);
261 }
262}

337static void eop_setup_or_halt(uint32_t* out_cpuclk_hz)
338{
339 uint32_t cpuclk0_hz = 0U;
340 if (ra8_cgc_init() != k_ra8_ok) {
341 eop_panic_halt();
342 }
343 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
344 eop_panic_halt();
345 }
346 if (ra8_mstp_init() != k_ra8_ok) {
347 eop_panic_halt();
348 }
349 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
350 eop_panic_halt();
351 }
352 if (ra8_board_uart_console_init((uint32_t)k_eop_uart_baud) != k_ra8_ok) {
353 eop_panic_halt();
354 }
355 *out_cpuclk_hz = cpuclk0_hz;
356}

422int32_t main(void)
423{
424 uint32_t cpuclk0_hz = 0U;
425 eop_setup_or_halt(&cpuclk0_hz);
426 ra8_isr_globals_enable();
427
428 /* Route the shared fault decoder's ra8_log dump onto the bench
429 * console before anything that can fault runs. */
430 ra8_log_set_byte_sink(eop_log_sink, nullptr);
431
432 eop_print(k_eop_msg_boot, (uint32_t)sizeof(k_eop_msg_boot) - 1U);
433 eop_print(k_eop_msg_clock, (uint32_t)sizeof(k_eop_msg_clock) - 1U);
434 eop_print_hex(cpuclk0_hz);
435 eop_print(k_eop_msg_pad, (uint32_t)sizeof(k_eop_msg_pad) - 1U);
436 eop_print_hex(eop_bss_pad_anchor());
437 eop_print(k_eop_msg_padlen, (uint32_t)sizeof(k_eop_msg_padlen) - 1U);
438 eop_print_hex((uint32_t)k_eop_pad_bytes);
439 eop_print(k_eop_msg_newline, (uint32_t)sizeof(k_eop_msg_newline) - 1U);
440
441 if (eop_run_once() != k_ra8_ok) {
442 eop_print(k_eop_msg_fail, (uint32_t)sizeof(k_eop_msg_fail) - 1U);
443 } else {
444 eop_print(k_eop_msg_pass, (uint32_t)sizeof(k_eop_msg_pass) - 1U);
445 }
446
447 while (1) {
448 __asm__ volatile("wfi");
449 }
450 return 0;
451}
