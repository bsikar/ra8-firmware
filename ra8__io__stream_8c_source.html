131ra8_err_t ra8_io_stream_put_u32(ra8_io_stream_t* s, uint32_t value)
132{
133 const ra8_err_t v = internal_validate(s);
134 if (v != k_ra8_ok) {
135 return v;
136 }
137 uint8_t tmp[k_ra8_io_u32_max_digits];
138 uint32_t i = 0;
139 uint32_t x = value;
140 do {
141 tmp[i] = (uint8_t)('0' + (x % (uint32_t)k_ra8_io_dec_base));
142 x /= (uint32_t)k_ra8_io_dec_base;
143 ++i;
144 } while (x != 0U);
145 uint8_t out[k_ra8_io_u32_max_digits];
146 for (uint32_t j = 0; j < i; ++j) {
147 out[j] = tmp[i - 1U - j];
148 }
149 return ra8_io_stream_write(s, out, i, nullptr);

152ra8_err_t ra8_io_stream_put_hex(ra8_io_stream_t* s, uint32_t value, uint8_t min_digits)
153{
154 const ra8_err_t v = internal_validate(s);
155 if (v != k_ra8_ok) {
156 return v;
157 }
158 if (min_digits == 0U) {
159 return k_ra8_err_invalid_arg;
160 }
161 if (min_digits > (uint8_t)k_ra8_io_hex_max_digits) {
162 return k_ra8_err_invalid_arg;
163 }
164 static const char k_hex[] = "0123456789abcdef";
165 uint8_t tmp[k_ra8_io_hex_max_digits];
166 uint32_t i = 0;
167 uint32_t x = value;
168 do {
169 tmp[i] = (uint8_t)k_hex[x % (uint32_t)k_ra8_io_hex_base];
170 x /= (uint32_t)k_ra8_io_hex_base;
171 ++i;
172 } while (x != 0U);
173 while (i < (uint32_t)min_digits) {
174 tmp[i] = (uint8_t)'0';
175 ++i;
176 }
177 uint8_t out[k_ra8_io_hex_max_digits];
178 for (uint32_t j = 0; j < i; ++j) {
179 out[j] = tmp[i - 1U - j];
180 }
