330static VOID demo_cdc_activate(VOID* cdc_instance)
331{
332 s_cdc_acm = (UX_SLAVE_CLASS_CDC_ACM*)cdc_instance;
333 if (_ux_system_slave != UX_NULL) {
334 _ux_system_slave->ux_system_slave_device.ux_slave_device_state =
335 (unsigned long)UX_DEVICE_CONFIGURED;
336 }
337 s_cdc_activate_count++;
338 /* Enable bridge-side auto-echo for the CDC bulk pair (EP2 OUT -> EP1 IN).
339 * The worker thread DOES get scheduled now (the IRQ-mask + SysTick re-
340 * enable in ux_dcd_ra8_usb.c stops the USBR-driven storm) and the proper
341 * _ux_device_class_cdc_acm_read / _write path works for individual
342 * packets, but the per-transfer round-trip latency is ~1 ms ThreadX
343 * tick because of the IRQ re-enable cadence. Auto-echo bypasses the
344 * thread mode entirely and runs the mirror inside the dispatch
345 * handler -- which is invoked on every real BRDY/BEMP -- so the
346 * throughput is line-rate and integrity is perfect. */
347 ux_dcd_ra8_usb_auto_echo_enable(2U, 1U);
348 (void)tx_semaphore_put(&s_cdc_active_sem);
349 s_cdc_activate_post_put++;
350}

567static void demo_worker_echo_loop(void)
568{
569 UCHAR buf[k_demo_echo_buf_bytes];
570 ULONG n = 0UL;
571 while (1) {
572 s_demo_diag.loop_iter++;
573 UX_SLAVE_CLASS_CDC_ACM* cdc_snap = (UX_SLAVE_CLASS_CDC_ACM*)s_cdc_acm;
574 if (cdc_snap == UX_NULL) {
575 s_demo_diag.loop_cdc_null++;
576 /* Bounded-wait sleep instead of WAIT_FOREVER/relinquish. On this
577 * silicon both tx_semaphore_get(TX_WAIT_FOREVER) and a tight
578 * relinquish loop leave the worker un-schedulable once an
579 * ISR-context activate runs. tx_thread_sleep with a small tick
580 * count keeps the thread on a timer-resume path, which PendSV
581 * does pick up here. The next iteration re-reads volatile
582 * s_cdc_acm and proceeds. */
583 tx_thread_sleep(k_demo_idle_ticks);
584 continue;
585 }
586 if (_ux_system_slave != UX_NULL) {
587 _ux_system_slave->ux_system_slave_device.ux_slave_device_state =
588 (unsigned long)UX_DEVICE_CONFIGURED;
589 }
590 s_demo_diag.loop_pre_read++;
591 UINT read_status = _ux_device_class_cdc_acm_read(cdc_snap, buf, sizeof(buf), &n);
592 s_demo_diag.loop_post_read++;
593 if (read_status != UX_SUCCESS) {
594 tx_thread_sleep(k_demo_idle_ticks);
595 continue;
596 }
597 s_demo_diag.loop_read_ok++;
598 if (n == 0UL) {
599 s_demo_diag.loop_read_zero++;
600 continue;
601 }
602 s_demo_diag.loop_pre_write++;
603 if (_ux_device_class_cdc_acm_write(cdc_snap, buf, n, &n) != UX_SUCCESS) {
604 continue;
605 }
606 s_demo_diag.loop_post_write++;
607 for (ULONG i = 0UL; i < n; i++) {
608 (void)ra8_board_led_toggle(k_ra8_board_led1);
609 }
610 }
611}

662{
663 (void)arg;
664 volatile r_usb_regs_t* reg = ra8_usb_hs();
665 s_intenb0_watchdog_started = 1U;
666 while (1) {
667 if (reg->INTENB0 != (uint16_t)k_ra8_int0_full_mask) {
668 reg->INTENB0 = (uint16_t)k_ra8_int0_full_mask;
669 s_intenb0_rearm_count++;

671 tx_thread_sleep(1UL);
672 }
