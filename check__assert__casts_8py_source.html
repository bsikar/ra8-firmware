4"""Gate: reject explicit integer casts inside TEST_ASSERT_EQ arguments.
5
6TEST_ASSERT_EQ widens both arguments to int64_t internally. An outer
7cast like (int) or (int32_t) is therefore redundant -- and a (int) cast
8applied to a uint32_t enum silently truncates the value to 32-bit signed
9before the widening, which can produce a false-passing comparison for
10values >= 0x80000000.
11
12Run: check_assert_casts.py <file> [...]
13Exit 0 if clean, 1 if any violation found.
14"""
15
16import re
17import sys
18from pathlib import Path

20_TYPES = r"u?int(?:8|16|32|64)?_t|int|size_t|ssize_t"
21_CAST_RE = re.compile(r"\‍((?:" + _TYPES + r")\‍)")
22_MACRO = "TEST_ASSERT_EQ("

25def _find_close_paren(text: str, start: int) -> int:
26 depth = 1
27 i = start
28 while i < len(text) and depth:
29 c = text[i]
30 if c == "(":
31 depth += 1
32 elif c == ")":
33 depth -= 1
34 i += 1
35 return i - 1

38def _has_leading_cast(text: str) -> bool:
39 return bool(re.match(r"^\s*\‍((?:" + _TYPES + r")\‍)", text))

42def check(path: Path) -> list[str]:
43 """Report TEST_ASSERT_EQ arguments that OPEN with a redundant integer cast.
44
45 Only a cast in leading position counts, on either argument. A cast deeper
46 in the expression is left alone deliberately -- there it is usually load
47 bearing (narrowing a wider intermediate), whereas an outer one is applied
48 to a value the macro is about to widen to int64_t anyway.
49
50 The macro's two arguments are split at the first comma seen at paren depth
51 zero, so a comma inside a nested call or a compound literal does not shift
52 the second argument. An invocation with no such comma is skipped rather
53 than reported: that is a syntax error, and the compiler names it better.
54
55 Returns one preformatted ``path:line: message`` per violation, in file
56 order; an empty list means clean.
57 """
58 content = path.read_text(encoding="ascii", errors="replace")
59 violations = []
60 macro_len = len(_MACRO)
61 pos = 0
62 while True:
63 idx = content.find(_MACRO, pos)
64 if idx == -1:
65 break
66 inner_start = idx + macro_len
67 close = _find_close_paren(content, inner_start)
68 inner = content[inner_start:close]
69
70 depth = 0
71 split = None
72 for i, ch in enumerate(inner):
73 if ch in "([{":
74 depth += 1
75 elif ch in ")]}":
76 depth -= 1
77 elif ch == "," and depth == 0:
78 split = i
79 break
80
81 line_no = content[:idx].count("\n") + 1
82 if split is None:
83 pos = close + 1
84 continue
85
86 arg1 = inner[:split]
87 arg2 = inner[split + 1 :]
88 if _has_leading_cast(arg1):
89 violations.append(
90 f"{path}:{line_no}: cast in first arg of TEST_ASSERT_EQ: "
91 f"{_MACRO}{arg1.strip()[:60]}..."
92 )
93 if _has_leading_cast(arg2):
94 violations.append(
95 f"{path}:{line_no}: cast in second arg of TEST_ASSERT_EQ: ...{arg2.strip()[:60]}"
96 )
97 pos = close + 1
98 return violations

101def main() -> int:
102 """Scan the files named on argv and print every finding to stdout.
103
104 This gate takes its targets from the caller (the pre-commit hook passes
105 the staged C files) and has no discovery mode: an empty argv is a usage
106 error and exits 1, NOT a clean tree. A checker that treats "no files" as
107 success is the failure mode this repo keeps finding in its own tooling.
108
109 Returns 0 only when every named file is clean; 1 on any violation and on
110 the empty-argv usage error.
111 """
112 paths = [Path(p) for p in sys.argv[1:]]
