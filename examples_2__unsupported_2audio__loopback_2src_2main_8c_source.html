229RA8_INTERNAL static uint32_t internal_audio_loopback_u32_to_dec(uint32_t value, uint8_t* buf)
230{
231 enum : uint8_t {
232 k_ascii_zero = 0x30U,
233 k_radix = 10U,
234 k_max_digits = 10U,
235 };
236 uint8_t tmp[k_max_digits];
237 uint32_t n = 0U;
238 if (value == 0U) {
239 buf[0] = k_ascii_zero;
240 return 1U;
241 }
242 while (value > 0U) {
243 tmp[n] = (uint8_t)(k_ascii_zero + (uint8_t)(value % k_radix));
244 n++;
245 value /= k_radix;
246 }
247 for (uint32_t i = 0U; i < n; i++) {
248 buf[i] = tmp[n - 1U - i];
249 }
250 return n;
251}

302 internal_audio_loopback_init_codec();
303
304 ra8_isr_globals_enable();

306 uint32_t block_count = 0U;

311 break;
312 }
313 block_count++;

315 internal_audio_loopback_print_count(block_count);
316 (void)ra8_board_led_toggle(k_ra8_board_led1);
317 }
318 }
319
320 internal_audio_loopback_panic_halt();
321}
