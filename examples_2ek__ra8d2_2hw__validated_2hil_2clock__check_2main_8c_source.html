160int32_t main(void)
161{
162 if (ra8_cgc_init() != k_ra8_ok) {
163 clock_check_panic_halt();
164 }
165
166 uint32_t cpuclk0_hz = 0U;
167 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
168 clock_check_panic_halt();
169 }
170

172 clock_check_panic_halt();
173 }
174
175 if (clock_check_pins_init() != k_ra8_ok) {
176 clock_check_panic_halt();
177 }
178
179 ra8_isr_globals_enable();
180
181 while (1) {
182 if (clock_check_pins_toggle_all() != k_ra8_ok) {
183 g_clock_check_mismatch += 1U;
184 break;
185 }
186 if (clock_check_verify_all()) {
187 g_clock_check_match += 1U;
188 } else {
189 g_clock_check_mismatch += 1U;
190 }
191 ra8_delay_ms(k_clock_check_half_period_ms);
192 }
193
194 clock_check_panic_halt();
195 return 0;
196}

287 {k_ra8_clock_id_pclke, (uint32_t)k_ra8_pclke_hz},
288 {k_ra8_clock_id_fclk, (uint32_t)k_ra8_fclk_hz},
289 {k_ra8_clock_id_mriclk, (uint32_t)k_ra8_mriclk_hz},
290 };
291 const uint32_t kExpectedCount = (uint32_t)(sizeof(kExpected) / sizeof(kExpected[0]));
292 for (uint32_t i = 0U; i < kExpectedCount; ++i) {
293 uint32_t hz = 0U;
294 if (ra8_cgc_get_clock_hz(kExpected[i].id, &hz) != k_ra8_ok) {
295 return false;
296 }
297 if (hz != kExpected[i].expected_hz) {
298 return false;
299 }
300 }
301 return true;
302}
