57/*
58 * The capacities the viewer shares with the firmware reader are CHECKED against
59 * the firmware policy headers, never silently copied: the viewer over-provisions
60 * for a host (it has RAM to spare), but a firmware sizing that grows past that
61 * headroom must break THIS build rather than let the viewer quietly preview a
62 * document the board would reject (or vice versa). The page index must hold every
63 * member the shared archive decoder will enumerate; the gzip/xz unwrap arena must
64 * hold the largest container the shared decompression policy admits; the xz
65 * scratch must exceed the firmware decoder-state reserve so a dictionary budget
66 * remains. (k_viewer_name_cap and k_viewer_arena_bytes are host-local sizing with
67 * no firmware counterpart, so they carry no such coupling.)
68 */
69static_assert((uint64_t)k_viewer_page_cap >= (uint64_t)k_ra8_decomp_def_max_entries,
70 "viewer page index must cover the firmware archive entry cap "
71 "(ra8_decomp_limits.h)");
72static_assert((uint64_t)k_viewer_unwrap_bytes >= (uint64_t)k_ra8_decomp_def_output_bytes,
73 "viewer unwrap arena must cover the firmware decode output cap "
74 "(ra8_decomp_limits.h)");
75static_assert((uint64_t)k_viewer_xz_scratch > (uint64_t)k_ra8_unarch_xz_state_reserve,
76 "viewer xz scratch must exceed the firmware xz decoder-state reserve "
77 "(ra8_unarch_xz.h)");
