139[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_kint_demo_arm(void)
140{
141 /* ICU bring-up is now performed in internal_kint_demo_setup_or_halt so the
142 * canonical CGC -> MSTP -> IOPORT -> TIME -> ICU -> peripheral order
143 * is preserved (see audit_init_order). */
144 ra8_err_t err = ra8_board_sw_init(k_ra8_board_sw1);
145 if (err != k_ra8_ok) {
146 return err;
147 }
148 const ra8_icu_irq_cfg_t cfg = {
149 .sense = k_ra8_icu_irqmd_falling,
150 .filter_div = k_ra8_icu_fclksel_pclkb_64,
151 .filter_en = true,
152 };
153 return ra8_icu_configure_irq_pin((uint8_t)k_kint_demo_irq_channel, &cfg);
154}

182 }

184 ra8_delay_ms((uint32_t)k_kint_demo_poll_ms);
