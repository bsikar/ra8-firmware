199static void c6_hosted_heartbeat(void)
200{
201 uint32_t beat = 0U;
202 while (1) {
203 c6_hosted_puts("c6_hosted_init: heartbeat n=");
204 c6_hosted_put_u32(beat);
205 c6_hosted_puts(" events=");
206 c6_hosted_put_u32(c6_hosted_event_count());
207 c6_hosted_puts("\r\n");
208 if (ra8_esp_hosted_port_is_ready()) {
209 c6_hosted_report_sideband();
210 } else {
211 c6_hosted_puts("c6_hosted_init: port down -- side-band cannot be read\r\n");
212 }
213 beat++;
214 tx_thread_sleep((ULONG)k_c6_hosted_heartbeat_ms);
215 }
216}
