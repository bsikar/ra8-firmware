4"""Generate epub_cover_fixture.h: a minimal EPUB3 with a real PNG cover.
5
6The manifest declares the cover with `properties="cover-image"`, and the
7on-device ereader_cover gate exercises the whole chain against it: open the
8blob in memory with `ra8_epub_open()`, pull the cover bytes with
9`ra8_epub_get_cover_image()`, decode, scale and blit with
10`ra8_img_decode_blit()`, then CRC-gate the framebuffer.
11
12The cover is a REAL RGB PNG, and that is the point of this fixture existing
13alongside tests/test_ra8_epub.c. That test uses a 4-byte stand-in, which
14exercises only the byte-copy path and never the decoder. Here stb_image has to
15actually decode something.
16
17Output is pure 7-bit ASCII -- a C array of the .epub bytes -- baked at 16 bytes
18per row so clang-format leaves it byte-identical rather than reflowing it.
19
20Changing the cover changes the framebuffer CRC. Re-run this, then read the CRC
21the board prints and update hil.conf in the SAME change, or the gate fails on a
22fixture that is perfectly correct.
23
24Usage:
25 python3 make_cover_fixture.py
26"""
27

36# --- Deterministic portrait cover: 96x144 (2:3, a real book-cover ratio) ----
37# Four horizontal colour bands. A real RGB PNG so stb_image actually decodes it
38# (unlike the 4-byte stand-in in tests/test_ra8_epub.c, which only exercises the
39# byte-copy path, not the decoder).
40COVER_W, COVER_H = 96, 144
41BANDS = [(0xC0, 0x10, 0x20), (0x18, 0x90, 0x30), (0x20, 0x40, 0xC0), (0xD0, 0xA0, 0x18)]

44def make_cover_png() -> bytes:
45 """Encode the deterministic 96x144 four-band portrait cover as PNG.
46
47 The 2:3 aspect is a real book-cover ratio, so the gate exercises the
48 scaler's aspect handling rather than a convenient square.
49
50 `optimize=False` is required, not a default: Pillow's optimizer picks filter
51 and compression settings that vary across builds, which would change the
52 baked bytes -- and therefore the pinned framebuffer CRC -- on a machine with
53 a different Pillow. Without it the fixture is not reproducible.
54
55 Returns:
56 Encoded PNG bytes, identical across runs and across Pillow builds.
57 """
58 img = Image.new("RGB", (COVER_W, COVER_H))
59 px = img.load()
60 band_h = COVER_H // len(BANDS)
61 for y in range(COVER_H):
62 idx = min(y // band_h, len(BANDS) - 1)
63 for x in range(COVER_W):
64 px[x, y] = BANDS[idx]
65 out = io.BytesIO()
66 # optimize=False keeps the encoder deterministic across Pillow builds.
67 img.save(out, format="PNG", optimize=False)
68 return out.getvalue()

111def make_epub() -> bytes:
112 """Assemble the minimal EPUB3 archive in memory and return its bytes.
113
114 Reproducible byte for byte: every entry carries a fixed 2026-01-01
115 timestamp and fixed permissions, so regenerating on a clean tree yields an
116 identical header and an empty diff.
117
118 Two entries are ZIP_STORED. `mimetype` must be first and uncompressed per
119 the EPUB spec -- readers check it at a fixed offset -- and the cover PNG is
120 stored because deflating already-compressed PNG data only grows it.
121
122 Returns:
123 The complete .epub archive as bytes.
124 """
125 cover_png = make_cover_png()
126 out = io.BytesIO()
127 fixed = (2026, 1, 1, 0, 0, 0)
128
129 def add(zf: zipfile.ZipFile, name: str, data: bytes | str, store: bool = False) -> None:
130 info = zipfile.ZipInfo(name, date_time=fixed)
131 info.compress_type = zipfile.ZIP_STORED if store else zipfile.ZIP_DEFLATED
132 info.external_attr = 0o600 << 16
133 zf.writestr(info, data)
134
135 with zipfile.ZipFile(out, "w") as zf:
136 # mimetype MUST be the first entry and stored (uncompressed).
137 add(zf, "mimetype", b"application/epub+zip", store=True)
138 add(zf, "META-INF/container.xml", CONTAINER_XML.encode("ascii"))
139 add(zf, "OEBPS/content.opf", CONTENT_OPF.encode("ascii"))
140 add(zf, "OEBPS/ch1.xhtml", CH1_XHTML.encode("ascii"))
141 add(zf, "OEBPS/cover.png", cover_png, store=True)
142 return out.getvalue()

145def bake_header(epub: bytes) -> str:
146 """Render the EPUB bytes as a C header with a `static const uint8_t` table.
147
148 The array is sized from a generated `enum : size_t`, so the declared length
149 and the data cannot drift apart.
150
151 The table sits inside a `clang-format off`/`on` guard at 16 bytes per row.
152 That is what keeps the formatter from reflowing thousands of bytes and
153 turning every regeneration into a large spurious diff.
154
155 Args:
156 epub: Archive bytes to bake, emitted as `0xNN`.
157
158 Returns:
159 The complete header source, pure 7-bit ASCII.
160 """
161 rows = []
162 for i in range(0, len(epub), 16):
163 chunk = epub[i : i + 16]
164 rows.append(" " + ", ".join(f"0x{b:02X}" for b in chunk) + ",")
165 body = "\n".join(rows)
166 return (
167 "/**\n"
168 " * @file epub_cover_fixture.h\n"
169 " * @brief Baked minimal EPUB3 with a real PNG cover (cover-image manifest item).\n"
170 " *\n"
171 " * @details A 96x144 four-band RGB PNG cover wrapped in a one-chapter EPUB3,\n"
172 " * byte-identical run to run. The ereader_cover gate opens it in memory\n"
173 " * (ra8_epub_open), extracts the cover (ra8_epub_get_cover_image), and\n"
174 " * decode+scale+blits it (ra8_img_decode_blit); the framebuffer hash in\n"
175 " * hil.conf pins the result. Generated by make_cover_fixture.py. Pure ASCII.\n"
176 " *\n"
177 " * @copyright Copyright (c) 2026 Brighton Sikarskie\n"
178 " * SPDX-License-Identifier: MIT\n"
179 " */\n"
180 "#pragma once\n"
181 "\n"
182 "#include <stddef.h>\n"
183 "#include <stdint.h>\n"
184 "\n"
185 "/** @brief Length of the baked cover EPUB blob, bytes. */\n"
186 f"enum : size_t {{ k_epub_cover_fixture_len = {len(epub)}U }};\n"
187 "\n"
188 "/** @brief Baked cover-art EPUB3 byte stream. */\n"
189 "static const uint8_t k_epub_cover_fixture[k_epub_cover_fixture_len] = {\n"
190 f"{body}\n"
191 "};\n"
192 )

195def main() -> int:
196 """Regenerate epub_cover_fixture.h in the CURRENT working directory.
197
198 The output path is relative, so run this from the app directory that owns
199 the fixture; anywhere else it writes a stray header instead of updating the
200 committed one.
201
202 Remember the two-step: this refreshes the fixture, but the framebuffer CRC
203 in hil.conf is pinned separately and must be re-read from the board and
204 updated in the same change.
205 """
206 epub = make_epub()
207 header = bake_header(epub)
208 with Path("epub_cover_fixture.h").open("w", encoding="ascii") as f:
209 f.write(header)
