1#!/usr/bin/env python3
2# SPDX-License-Identifier: MIT

4"""Gate: every ``scripts/...`` path mentioned anywhere in the tree resolves.
5
6``scripts/`` is referenced from workflows, Makefiles, CMake listfiles, the git
7hooks, C comments, and ~90 Markdown documents. Those references are plain
8text: nothing type-checks them, so a ``git mv`` inside ``scripts/`` breaks them
9silently. The repository has already been bitten by exactly that -- a rename
10left cross-references pointing at nothing and ``ci-fast`` did not notice,
11because a dead path in a doc link or a hook comment produces no build error and
12no test failure. It surfaces later as a workflow step that cannot find its
13driver.
14
15This gate closes that class. It reads every first-party text file, extracts
16every token that looks like a path under ``scripts/``, and requires it to
17resolve on disk. Run it before a restructuring and it is green; run it after
18and every reference you forgot to update is a named, located failure.
19
20Reference forms understood
21--------------------------
22``scripts/ci.sh`` repo-relative -- resolved against the repo root
23``../../scripts/dev/flash.sh`` relative -- resolved against the citing file
24``scripts/checks/`` trailing slash -- must be a directory
25``scripts/hil/*.sh`` glob -- must match at least one path
26``scripts/{flash,debug}.sh`` brace / ``$VAR`` interpolation -- the longest
27 literal directory prefix must exist
28
29The glob rule is the interesting one: it is what keeps a doc that says
30"``scripts/hil/*.sh``" honest after those scripts move, instead of leaving a
31pattern that matches nothing and reads as if it still does.
32
33Scope
34-----
35Deliberately limited to the ``scripts/`` prefix. Widening it to every
36top-level directory was measured first: 12445 path-like tokens tree-wide, of
37which 1223 distinct ones do not resolve -- build artifacts (``tests/build/``),
38illustrative globs (``tests/test_*.c``), and third-party prose. A gate that
39starts 1223 findings in the red cannot be landed, and grandfathering them would
40make it a gate that enforces nothing. ``scripts/`` is both the tree being
41restructured and the one whose inbound references are load-bearing (a broken
42``scripts/`` path in a workflow is a broken CI job), so it is where the rule
43pays for itself today. Extending the same machinery outward is tracked
44separately.
45
46Per-line opt-out: append ``PATHREF-OK: <reason>`` to a line to suppress it
47(mirrors the ``MAGIC-OK`` / ``CITES-OK`` / ``AI-OK`` family). It exists for
48prose that deliberately names a path that does not exist -- a checker docstring
49illustrating a hypothetical path, or a historical note. The marker has to sit
50on the same line as the token it waives, which is why the example below carries
51it inline rather than in the prose above:
52
53 a checker docstring naming ``scripts/foo`` PATHREF-OK: hypothetical
54
55Run::
56
57 check_script_references.py # gate
58 check_script_references.py --selftest # prove it fires and stays quiet
59
60Exit 0 if every reference resolves, 1 on findings, 2 on tool error.
61"""

140# A path token: an optional run of ``../`` segments, then ``scripts/``, then
141# path characters. The leading assertion keeps a ``myscripts/`` or
142# ``tools/scripts/`` prefix from matching, while still allowing a leading run
143# of parent-directory segments (whose match starts at the first dot, not at
144# the slash that follows it).
145_TOKEN_RE = re.compile(
146 r"(?:(?<=^)|(?<=[^A-Za-z0-9_/.-]))"
147 r"((?:\.\./)*" + re.escape(ROOT_PREFIX) + r"/[A-Za-z0-9_./*?{}$@,+\\-]+)"
148)
149
150# A sibling-directory reference built from segments rather than written as a
151# ``scripts/...`` path: ``. "$SCRIPT_DIR/../builders/select_host_compiler.sh"``.
152# _TOKEN_RE cannot see these -- there is no ``scripts/`` token anywhere in the
153# line -- and that blind spot took ``dev`` red once already. The incident: a
154# rename moved the builder directory, and the ``# shellcheck source=`` directive
155# one line above the sourcing line WAS rewritten, because it spells the path out
156# in full. The sourcing line itself kept its segment-built reference to the old
157# directory and failed only when the gate ran, in CI, after the merge.
158#
159# The variable is conventionally the citing script's own directory
160# (``SCRIPT_DIR``, ``SCRIPTDIR``, ``HERE``, ``DIR``), so the reference resolves
161# against the citing file's parent. That convention is the whole basis for
162# resolving these, so the variable name must match it -- an arbitrary
163# ``$SOMEWHERE_ELSE/../x`` is left alone rather than guessed at.
164_SIBLING_RE = re.compile(
165 r"\$\{?(SCRIPT_DIR|SCRIPTDIR|SCRIPT_ROOT|HERE|DIR)\}?/((?:\.\./)+[A-Za-z0-9_./-]+)"
166)
167
168# Trailing characters that are punctuation in the citing text, never part of a
169# filename. A trailing ``/`` is meaningful (directory) and is NOT stripped.
170_TRAILING_JUNK = ".,;:!?)`'\"|>"
171
172# Characters that mean the token is a shell / make interpolation rather than a
173# literal path.
174_INTERPOLATION_CHARS = ("$", "{", "}")
175
176# Characters that mean the token is a glob pattern.
177_GLOB_CHARS = ("*", "?")

195def _git_ls_files(root: Path) -> list[str]:
196 """Tracked plus untracked-but-not-ignored paths under `root`.
197
198 A filesystem walk would sweep in git-excluded local trees (``recon/``,
199 ``.claude/worktrees/``) that CI can never see, so the enumeration follows
200 git's own view of the tree -- the same choice the sibling checkers make.
201 """
202 git_tool = shutil.which("git")
203 if git_tool is None:
204 sys.stderr.write("check_script_references.py: FATAL -- `git` not on PATH\n")
205 sys.exit(2)
206 proc = subprocess.run( # noqa: S603 -- fixed argv, resolved tool path
207 [git_tool, "ls-files", "-z", "--cached", "--others", "--exclude-standard"],
208 cwd=root,
209 capture_output=True,
210 text=True,
211 check=False,
212 )
213 if proc.returncode != 0:
214 sys.stderr.write(proc.stderr)
215 sys.stderr.write("check_script_references.py: FATAL -- `git ls-files` failed\n")
216 sys.exit(2)
217 return [rel for rel in proc.stdout.split("\0") if rel]

228def _literal_prefix(token: str) -> str:
229 """The longest leading run of `token` segments free of glob / interpolation.
230
231 ``scripts/checks/{a,b}.py`` -> ``scripts/checks``. Used to salvage a
232 partial check from a token whose tail cannot be resolved literally: the
233 directory it lives in still has to exist, and that alone catches a moved
234 tree.
235 """
236 kept: list[str] = []
237 for segment in token.split("/"):
238 if any(ch in segment for ch in (*_INTERPOLATION_CHARS, *_GLOB_CHARS)):
239 break
240 kept.append(segment)
241 return "/".join(kept)

244def _resolve_base(root: Path, rel_file: str, token: str) -> tuple[Path, str]:
245 """Return the directory `token` is relative to, and the token minus ``../``.
246
247 A bare ``scripts/...`` is repo-relative by convention. A ``../../scripts/``
248 in a Markdown link is relative to the citing document, which is how the
249 qualification docs link to the tree.
250 """
251 if not token.startswith("../"):
252 return root, token
253 depth = 0
254 rest = token
255 while rest.startswith("../"):
256 depth += 1
257 rest = rest[3:]
258 base = (root / rel_file).parent
259 for _ in range(depth):
260 base = base.parent
261 return base, rest

264def _check_interpolated(base: Path, rest: str) -> str | None:
265 """A ``$VAR`` / brace token resolves as far as its literal prefix goes."""
266 prefix = _literal_prefix(rest)
267 if prefix and not (base / prefix).exists():
268 return f"interpolated path whose literal prefix {prefix!r} does not exist"
269 return None

272def _check_glob(base: Path, rest: str) -> str | None:
273 """A glob has to match something; one that matches nothing is a dead pattern."""
274 if not any(base.glob(rest.rstrip("/"))):
275 return "glob pattern matches nothing"
276 return None

279def _check_directory(base: Path, rest: str) -> str | None:
280 """A trailing slash asserts a directory, so a file of that name is still wrong."""
281 if not (base / rest.rstrip("/")).is_dir():
282 return "directory does not exist"
283 return None

286def _check_literal(base: Path, rest: str, token: str) -> str | None:
287 """The ordinary case: the path names something that must be on disk."""
288 if not (base / rest).exists():
289 return f"no such file (token {token!r})"
290 return None

293def _classify(base: Path, rest: str, token: str) -> str | None:
294 """Return a failure reason for an unresolved reference, or None if it is OK.
295
296 The four reference forms are checked by dedicated predicates so each one
297 reads as its own rule, and the order is significant: interpolation and
298 globs are recognised before the literal case, because a token carrying
299 either cannot be resolved as a plain path.
300 """
301 if any(ch in rest for ch in _INTERPOLATION_CHARS):
302 return _check_interpolated(base, rest)
303 if any(ch in rest for ch in _GLOB_CHARS):
304 return _check_glob(base, rest)
305 if rest.endswith("/"):
306 return _check_directory(base, rest)
307 return _check_literal(base, rest, token)

310def _unescape(token: str) -> str:
311 r"""Undo regex escaping, and cut the token where escaping stops meaning a dot.
312
313 ``check_ci_parity.py`` matches gate calls with a regex holding an escaped
314 ``.sh`` followed by ``\s+``. Left alone, the extracted token stops at the
315 first backslash and reads as a dangling directory reference; blindly
316 stripping every backslash instead welds the pattern's next atom onto the
317 filename. Both are noise on a correct file. So ``\.`` -- the only escape
318 that spells a character a real path can contain -- is unescaped, and the
319 token is cut at any other backslash, which is where the filename provably
320 ended.
321 """
322 token = token.replace("\\.", ".")
323 head, _, _ = token.partition("\\")
324 return head

327def _scan_text(rel_file: str, text: str, root: Path = REPO_ROOT) -> list[Finding]:
328 """Every unresolved ``scripts/...`` reference in one file's text."""
329 findings: list[Finding] = []
330 for line_no, line in enumerate(text.splitlines(), start=1):
331 if OPT_OUT in line:
332 continue
333 for match in _TOKEN_RE.finditer(line):
334 token = _unescape(match.group(1)).rstrip(_TRAILING_JUNK)
335 if not token or (token.endswith("/") and token.count("/") == 1):
336 continue # a bare "scripts/" mention, not a reference
337 base, rest = _resolve_base(root, rel_file, token)
338 reason = _classify(base, rest, token)
339 if reason is not None:
340 findings.append(Finding(rel_file, line_no, token, reason))
341 findings.extend(_scan_sibling_refs(rel_file, line, line_no, root))
342 return findings

345def _scan_sibling_refs(rel_file: str, line: str, line_no: int, root: Path) -> list[Finding]:
346 """Unresolved sibling-directory references in one line.
347
348 The shape matched is ``$SCRIPT_DIR/../somedir/file`` PATHREF-OK: shape.
349
350 Only files under ``scripts/`` are considered: the convention that the
351 variable holds the citing script's own directory is what makes the
352 reference resolvable, and it is a convention of this tree's scripts. A hit
353 elsewhere would be a guess.
354 """
355 if not rel_file.startswith(ROOT_PREFIX + "/"):
356 return []
357 findings: list[Finding] = []
358 citing_dir = (root / rel_file).parent
359 for match in _SIBLING_RE.finditer(line):
360 rest = match.group(2)
361 if any(ch in rest for ch in (*_INTERPOLATION_CHARS, *_GLOB_CHARS)):
362 continue
363 target = (citing_dir / rest).resolve()
364 if target.exists():
365 continue
366 try:
367 shown = target.relative_to(root.resolve()).as_posix()
368 except ValueError:
369 shown = target.as_posix()
370 findings.append(
371 Finding(
372 rel_file,
373 line_no,
374 f"${match.group(1)}/{rest}",
375 f"resolves to {shown}, which does not exist",
376 )
377 )
378 return findings

383 text = (root / rel).read_text(encoding="utf-8", errors="replace")
384 except OSError:
385 return []

389def scan_repo(root: Path = REPO_ROOT) -> list[Finding]:
390 """Every unresolved ``scripts/...`` reference in the first-party tree."""
391 findings: list[Finding] = []
392 for rel in _git_ls_files(root):
393 if _scannable(rel):
394 findings.extend(_scan_file(root, rel))
395 return findings

435def _selftest() -> int:
436 """Assert the detector fires on dead references and stays quiet on live ones.
437
438 A path checker that silently stopped matching would report a clean tree --
439 the failure mode that makes a gate worse than no gate. Both directions are
440 asserted here, and the gate body runs this before the real scan.
441 """
442 failures = 0
443 for body, expected, description in _SELFTEST_CASES:
444 got = len(_scan_text("docs/selftest_fixture.md", body))
445 if got != expected:
446 failures += 1
447 print(
448 f" FAIL {description}: expected {expected} finding(s), got {got}"
449 f" [{body.strip()}]",
450 file=sys.stderr,
451 )
452
453 # Relative-form resolution is file-position dependent, so it needs a real
454 # citing path rather than the shared fixture above.
455 rel_cases = (
456 ("docs/qualification/SDP.md", f"../../{_P}/ci.sh", 0),
457 ("docs/qualification/SDP.md", f"../../{_P}/nope_missing.sh", 1),
458 ("docs/qualification/SDP.md", f"../{_P}/ci.sh", 1),
459 )
460 for citing, token, expected in rel_cases:
461 got = len(_scan_text(citing, f"link to {token}\n"))
462 if got != expected:
463 failures += 1
464 print(
465 f" FAIL relative form {token} from {citing}: expected {expected}, got {got}",
466 file=sys.stderr,
467 )
468
469 # End-to-end: a real file in a real git tree must turn the whole gate red.
470 # The unit cases above exercise _scan_text; this proves the enumeration
471 # reaches a newly added file too, so a broken ls-files cannot report clean.
472 with tempfile.TemporaryDirectory() as tmp:
473 tmp_root = Path(tmp)
474 git_tool = shutil.which("git") or "git"
475 subprocess.run( # noqa: S603 -- fixed argv, resolved tool path
476 [git_tool, "init", "-q"], cwd=tmp_root, check=True, capture_output=True
477 )
478 (tmp_root / _P).mkdir()
479 (tmp_root / _P / "live.sh").write_text("#!/bin/sh\n")
480 (tmp_root / "README.md").write_text(f"ok: {_P}/live.sh\n")
481 if scan_repo(tmp_root):
482 failures += 1
483 print(" FAIL end-to-end: a live tree reported findings", file=sys.stderr)
484 (tmp_root / "README.md").write_text(f"bad: {_P}/dead.sh\n")
485 if len(scan_repo(tmp_root)) != 1:
486 failures += 1
487 print(" FAIL end-to-end: a dead reference was not reported", file=sys.stderr)
488
489 if failures:
490 print(f"check_script_references.py: --selftest FAILED ({failures})", file=sys.stderr)
491 return 1
492 total = len(_SELFTEST_CASES) + len(rel_cases) + 2
493 print(f"check_script_references.py: --selftest OK ({total} cases, both directions)")
494 return 0

497def main(argv: list[str]) -> int:
498 """Verify every ``scripts/`` path named anywhere in the tree still resolves.
499
500 A stale script reference fails only when someone runs it, which may be
501 months after the rename that broke it -- and in a Makefile or workflow
502 that is a broken build for whoever is unlucky, not for whoever moved the
503 file. This turns that into a build-time error at the moment of the move.
504
505 Returns 1 listing each dangling reference, 0 when all resolve.
506 """
507 if "--selftest" in argv[1:]:
508 return _selftest()
509
510 findings = scan_repo()
511 if not findings:
512 print(f"check_script_references.py: every {ROOT_PREFIX}/ reference resolves.")
513 return 0
514
515 print(
516 f"check_script_references.py: {len(findings)} unresolved {ROOT_PREFIX}/ reference(s):\n",
517 file=sys.stderr,
518 )
519 for finding in sorted(findings, key=lambda f: (f.rel_file, f.line_no)):
520 print(f" {finding}", file=sys.stderr)
521 print(
522 f"\nUpdate the reference, or -- if the path is deliberately "
523 f"hypothetical prose -- append `{OPT_OUT}: <reason>` to the line.",
524 file=sys.stderr,
525 )
526 return 1
