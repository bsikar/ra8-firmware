85emu_elf_io_result_t priv_emu_elf_source_close(emu_elf_source_t* source)
86{
87 if ((source == nullptr) || (source->fd < 0)) {
88 return internal_elf_result(k_emu_elf_io_invalid, 0U, 0U, 0);
89 }
90 const uint64_t length = source->length;
91 emu_io_file_t file = {.fd = source->fd, .size = (int64_t)source->length};
92 source->fd = -1;
93 source->length = 0U;
94 const emu_io_result_t closed = priv_emu_io_close(&file);
95 return (closed.status == k_emu_io_ok) ? internal_elf_result(k_emu_elf_io_ok, length, 0U, 0)
96 : internal_from_host(closed, length, 0U);
97}
