68static uint32_t priv_match_block_end(const ra8_reflow_t* engine, uint32_t start, uint32_t limit)
69{
70 const uint8_t tag = engine->tokens[start].tag;
71 int32_t depth = 1;
72 for (uint32_t i = start + 1U; i < limit; ++i) {
73 const ra8_reflow_token_t* tok = &engine->tokens[i];
74 if (tok->tag != tag) {
75 continue;
76 }
77 if (tok->kind == (uint8_t)k_ra8_reflow_tok_block_start) {
78 depth++;
79 } else if (tok->kind == (uint8_t)k_ra8_reflow_tok_block_end) {
80 depth--;
81 if (depth == 0) {
82 return i;
83 }
84 }
85 }
86 return limit;
87}

176RA8_INTERNAL
177static uint32_t priv_table_columns(const ra8_reflow_t* engine, uint32_t start, uint32_t end)
178{
179 uint32_t cols = 0U;
180 uint32_t i = start + 1U;
181 while (i < end) {
182 if (!priv_is_row_start(engine, i)) {
183 i++;
184 continue;
185 }
186 const uint32_t tr_end = priv_match_block_end(engine, i, end);
187 uint32_t cells = 0U;
188 for (uint32_t j = i + 1U; j < tr_end; ++j) {
189 if (priv_is_cell_start(engine, j)) {
190 cells++;
191 }
192 }
193 if (cells > cols) {
194 cols = cells;
195 }
196 i = tr_end + 1U;
197 }
198 return cols;
199}

237RA8_INTERNAL
238static void priv_cell_text(ra8_reflow_t* engine,
239 const stbtt_fontinfo* font,
240 const ra8_reflow_token_t* tok,
241 int32_t cell_x,
242 int32_t cell_right,
243 uint16_t font_px,
244 uint32_t color,
245 int32_t* cx,
246 int32_t* cy,
247 uint32_t* lines)
248{
249 const int32_t line_h = (int32_t)ra8_reflow_layout_line_height(font_px);
250 const uint8_t* base = engine->text_pool + tok->text_off;
251 const uint32_t len = tok->text_len;
252 uint32_t i = 0U;
253 while (i < len) {
254 if (base[i] == ' ') {
255 const int32_t adv = ra8_reflow_layout_glyph_advance(font, font_px, (int32_t)' ');
256 /* mcdc-deactivated: ra8_reflow_tok_stash_run collapses leading whitespace of every text run (last_ws starts true), so a cell text token never begins with a space, and the pen always advances past cell_x after a word is emitted; a space is therefore only ever seen with *cx > cell_x, making the (*cx > cell_x) false arm unreachable. */
257 if ((*cx > cell_x) && ((*cx + adv) <= cell_right)) {
258 (void)ra8_reflow_layout_push_glyph(engine, *cx, *cy, (int32_t)' ', font_px, 0U, color, 0U);
259 *cx += adv;
260 }
261 ++i;
262 continue;
263 }
264 uint32_t word_end = i;
265 int32_t word_w = 0;
266 while ((word_end < len) && (base[word_end] != ' ')) {
267 word_w += ra8_reflow_layout_glyph_advance(font, font_px, (int32_t)base[word_end]);
268 ++word_end;
269 }
270 if (((*cx + word_w) > cell_right) && (*cx > cell_x)) {
271 *cx = cell_x;
272 *cy += line_h;
273 (*lines)++;
274 }
275 while (i < word_end) {
276 const int32_t adv = ra8_reflow_layout_glyph_advance(font, font_px, (int32_t)base[i]);
277 (void)
278 ra8_reflow_layout_push_glyph(engine, *cx, *cy, (int32_t)base[i], font_px, 0U, color, 0U);
279 *cx += adv;
280 ++i;
281 }
282 }
283}

317static uint32_t priv_layout_cell(ra8_reflow_t* engine,
318 const stbtt_fontinfo* font,
319 int32_t cell_x,
320 int32_t cell_w,
321 int32_t top_y,
322 uint32_t color,
323 uint32_t tstart,
324 uint32_t tend)
325{
326 const int32_t cell_right = cell_x + cell_w;
327 const uint16_t font_px = engine->font_px;
328 int32_t cx = cell_x;
329 int32_t cy = top_y;
330 uint32_t lines = 1U;
331 for (uint32_t t = tstart; t < tend; ++t) {
332 const ra8_reflow_token_t* tok = &engine->tokens[t];
333 if (tok->kind == (uint8_t)k_ra8_reflow_tok_text) {
334 priv_cell_text(engine, font, tok, cell_x, cell_right, font_px, color, &cx, &cy, &lines);
335 }
336 }
337 return lines;
338}

369RA8_INTERNAL
370static uint32_t priv_row_cells(ra8_reflow_t* engine,
371 const stbtt_fontinfo* font,
372 uint32_t tr_start,
373 uint32_t tr_end,
374 int32_t col_w,
375 int32_t row_y)
376{
377 const int32_t pad = (int32_t)k_priv_cell_pad_px;
378 const uint32_t color = engine->body_color;
379 uint32_t rows = 1U;
380 uint32_t col = 0U;
381 uint32_t i = tr_start + 1U;
382 while (i < tr_end) {
383 if (!priv_is_cell_start(engine, i)) {
384 i++;
385 continue;
386 }
387 const uint32_t cell_end = priv_match_block_end(engine, i, tr_end);
388 const int32_t cell_x = (int32_t)k_ra8_reflow_margin_px + ((int32_t)col * col_w) + pad;
389 const int32_t cell_w = col_w - (2 * pad);
390 const uint32_t lines =
391 priv_layout_cell(engine, font, cell_x, cell_w, row_y, color, i + 1U, cell_end);
392 if (lines > rows) {
393 rows = lines;
394 }
395 col++;
396 i = cell_end + 1U;
397 }
398 return rows;
399}

431RA8_INTERNAL
432static uint32_t priv_layout_row(ra8_reflow_t* engine,
433 priv_cursor_t* cur,
434 const stbtt_fontinfo* font,
435 uint32_t tr_start,
436 uint32_t table_end,
437 int32_t col_w)
438{
439 const uint32_t tr_end = priv_match_block_end(engine, tr_start, table_end);
440 const int32_t line_h = (int32_t)ra8_reflow_layout_line_height(engine->font_px);
441 const int32_t bottom_limit = (int32_t)engine->viewport_h - (int32_t)k_ra8_reflow_margin_px;
442
443 const uint32_t glyphs_before = engine->glyph_count;
444 uint32_t row_lines = priv_row_cells(engine, font, tr_start, tr_end, col_w, cur->y);
445 int32_t row_h = (int32_t)row_lines * line_h;
446
447 if (((cur->y + row_h) > bottom_limit) && (cur->y > (int32_t)k_ra8_reflow_margin_px)) {
448 engine->glyph_count = glyphs_before; /* roll the row back ... */
449 (void)ra8_reflow_layout_finish_page(engine,
450 cur); /* ... flush the page (cur->y -> margin) ... */
451 /* ... re-lay it */
452 row_lines = priv_row_cells(engine, font, tr_start, tr_end, col_w, cur->y);
453 row_h = (int32_t)row_lines * line_h;
454 }
455 cur->y += row_h + (int32_t)k_priv_row_gap_px;
456 return tr_end + 1U;
457}

459ra8_err_t ra8_reflow_layout_table(ra8_reflow_t* engine,
460 priv_cursor_t* cur,
461 const stbtt_fontinfo* font,
462 uint32_t start,
463 uint32_t* out_next)
464{
465 if (cur->line_has_content != 0U) {
466 if (!ra8_reflow_layout_newline(engine, cur, false)) {
467 return k_ra8_err_no_mem;
468 }
469 }
470 const uint32_t end = priv_match_block_end(engine, start, engine->token_count);
471 const uint32_t cols = priv_table_columns(engine, start, end);
472 if (cols == 0U) {
473 *out_next = (end < engine->token_count) ? (end + 1U) : engine->token_count;
474 return k_ra8_ok;
475 }
476 const int32_t content_w = (int32_t)engine->viewport_w - (2 * (int32_t)k_ra8_reflow_margin_px);
477 const int32_t col_w = content_w / (int32_t)cols;
478
479 uint32_t i = start + 1U;
480 while (i < end) {
481 if (priv_is_row_start(engine, i)) {
482 i = priv_layout_row(engine, cur, font, i, end, col_w);
483 } else {
484 i++;
485 }
486 }
487
488 /* Resume linear flow below the table. */
489 cur->x = (int32_t)k_ra8_reflow_margin_px + (int32_t)cur->indent_px;
490 cur->align = (uint8_t)k_ra8_reflow_align_left;
491 cur->line_has_content = 0U;
492 cur->y += (int32_t)k_ra8_reflow_paragraph_gap_px;
493 cur->line_top = cur->y;
494 cur->line_first_glyph = engine->glyph_count;
495 *out_next = (end < engine->token_count) ? (end + 1U) : engine->token_count;
496 return k_ra8_ok;
497}
