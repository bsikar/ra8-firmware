178static bool pp_demo_cycle_modes(void)
179{
180 if (ra8_power_profile_mark_enter(k_ra8_power_profile_region_active) != k_ra8_ok) {
181 return false;
182 }
183 for (volatile uint32_t i = 0U; i < (uint32_t)k_pp_demo_busy_iters; i++) {
184 /* Busy spin so the analyser sees a clear ACTIVE shape. */
185 }
186 if (ra8_power_profile_mark_exit(k_ra8_power_profile_region_active) != k_ra8_ok) {
187 return false;
188 }
189
190 if (ra8_power_profile_mark_enter(k_ra8_power_profile_region_sleep) != k_ra8_ok) {
191 return false;
192 }
193 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_sleep) != k_ra8_ok) {
194 return false;
195 }
196 if (ra8_power_profile_mark_exit(k_ra8_power_profile_region_sleep) != k_ra8_ok) {
197 return false;
198 }
199
200 /* Deep Sleep and Software Standby skipped: in both modes the CPU
201 * clock (and therefore SysTick) is gated, so the demo has no wake
202 * source and ra8_lpm_enter_sleep hangs WFI indefinitely. Configuring
203 * AGT/RTC as a wake source would require deeper board-specific
204 * setup. The "pp: a=" stats line still prints with deep_standby and
205 * software_standby region times = 0. */
206 return true;
207}

220static uint8_t pp_demo_uint_to_dec(uint8_t* buf, uint64_t v)
221{
222 uint8_t tmp[k_pp_demo_uint_dec_max];
223 uint8_t n = 0U;
224 uint64_t r = v;
225 if (r == 0U) {
226 buf[0] = (uint8_t)k_pp_demo_ascii_zero;
227 return 1U;
228 }
229 while ((r > 0U) && (n < (uint8_t)k_pp_demo_uint_dec_max)) {
230 tmp[n] = (uint8_t)((uint8_t)k_pp_demo_ascii_zero + (uint8_t)(r % (uint64_t)k_pp_demo_dec_base));
231 r = r / (uint64_t)k_pp_demo_dec_base;
232 n++;
233 }
234 for (uint8_t i = 0U; i < n; i++) {
235 buf[i] = tmp[(uint8_t)(n - 1U - i)];
236 }
237 return n;
238}

254static uint32_t pp_demo_append(uint8_t* buf, uint32_t off, const char* tag, uint64_t val)
255{
256 uint32_t i = 0U;
257 while (tag[i] != '\0') {
258 buf[off++] = (uint8_t)tag[i];
259 i++;
260 }
261 off += pp_demo_uint_to_dec(&buf[off], val);
262 return off;

277static uint32_t pp_demo_format_line(uint8_t* out, const ra8_power_profile_stats_t* stats)
278{
279 uint32_t off = 0U;
280 off = pp_demo_append(out,
281 off,
282 "pp: a=",
283 stats->regions[k_ra8_power_profile_region_active].total_time_us);
284 off =
285 pp_demo_append(out, off, " s=", stats->regions[k_ra8_power_profile_region_sleep].total_time_us);
286 off = pp_demo_append(out,
287 off,
288 " d=",
289 stats->regions[k_ra8_power_profile_region_deep_standby].total_time_us);
290 off = pp_demo_append(out,
291 off,
292 " st=",
293 stats->regions[k_ra8_power_profile_region_software_standby].total_time_us);
294 const uint8_t suffix[] = " us\r\n";
295 for (uint32_t i = 0U; i < sizeof(suffix) - 1U; i++) {
296 out[off++] = suffix[i];
297 }
298 return off;
299}

316int32_t main(void)
317{
318 pp_demo_setup_or_halt();
319 ra8_isr_globals_enable();
320
321 while (1) {
322 bool profile_ok = true;
323 if (!pp_demo_cycle_modes()) {
324 profile_ok = false;
325 (void)ra8_board_led_on(k_ra8_board_led2);
326 /* Emit a FAIL banner over SCI so the HIL negative regex catches
327 * a silently-broken profile_mark_enter/exit path; without this
328 * line the test would pass on a chip where the LPM tracker was
329 * never actually transitioning regions. */
330 const uint8_t fail_banner[] = "pp: FAIL cycle_modes\r\n";
331 (void)ra8_board_uart_console_write(fail_banner, (size_t)(sizeof(fail_banner) - 1U));
332 }
333 ra8_power_profile_stats_t stats = {};
334 if (ra8_power_profile_get_stats(&stats) != k_ra8_ok) {
335 profile_ok = false;
336 (void)ra8_board_led_on(k_ra8_board_led2);
337 const uint8_t stats_fail[] = "pp: FAIL get_stats\r\n";
338 (void)ra8_board_uart_console_write(stats_fail, (size_t)(sizeof(stats_fail) - 1U));
339 }
340 uint8_t out[k_pp_demo_print_buf] = {};
341 const uint32_t off = pp_demo_format_line(out, &stats);
342 if (ra8_board_uart_console_write(out, (size_t)off) != k_ra8_ok) {
343 break;
344 }
345 /* Single-condition guard (no compound decision): the verdict prints
346 * only when neither failure branch above flipped profile_ok. */
347 if (profile_ok) {
348 (void)ra8_board_uart_console_write(k_pp_demo_pass_msg,
349 (size_t)(sizeof(k_pp_demo_pass_msg) - 1U));
350 }
351 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
352 break;
353 }
354 ra8_delay_ms(k_pp_demo_period_ms);
355 }
356
357 pp_demo_panic_halt();
358 return 0;
359}
