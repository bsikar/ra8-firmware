57ra8_err_t ra8_time_init(uint32_t cpu_hz)
58{
59 if (cpu_hz == 0U) {
60 ra8_log_error(s_tag, "cpu_hz must be non-zero");
61 return k_ra8_err_invalid_arg;
62 }
63
64 const uint32_t reload = (cpu_hz / k_ra8_ms_per_sec) - 1U;
65 if (reload == 0U) {
66 ra8_log_error(s_tag, "cpu_hz too low for 1kHz tick");
67 return k_ra8_err_invalid_arg;
68 }
69
70#ifndef RA8_OFF_TARGET
71 /* Arm SysTick (CPU clock + tick IRQ) and enable the DWT cycle counter
72 * through the shared timebase primitive, which owns every raw SYST_CSR /
73 * SYST_RVR / SYST_CVR / DEMCR / DWT access. On the host build these calls
74 * are skipped, exactly as the raw pokes were. */
75 const ra8_err_t cfg_err = ra8_systick_configure(reload, k_ra8_systick_clk_cpu, true);
76 if (cfg_err != k_ra8_ok) {
77 ra8_log_error(s_tag, "systick configure failed");
78 return cfg_err;
79 }
80 ra8_dwt_cyccnt_enable();
81#endif
82
83 s_cycles_per_ms = cpu_hz / k_ra8_ms_per_sec;
84 s_tick_ms = 0U;
85 ra8_log_info_val(s_tag, "systick reload", reload);
86 return k_ra8_ok;
87}

127void ra8_delay_ms(uint32_t ms)
128{
129#ifdef RA8_OFF_TARGET
130 (void)ms; /* No SysTick off-target -- s_tick_ms never advances. */
131#else
132 /* If PRIMASK is set the SysTick IRQ cannot dispatch and s_tick_ms */
133 /* never advances -- a wfi-loop on s_tick_ms would hang forever. */
134 /* Fall back to DWT_CYCCNT, which ticks every CPU cycle regardless */
135 /* of PRIMASK. Once IRQs are globally enabled, the cheaper */
136 /* SysTick path is used. */
137 uint32_t primask;
138 __asm__ volatile("mrs %0, primask" : "=r"(primask));
139 if ((primask & 1U) != 0U) {
140 const uint32_t target = ms * s_cycles_per_ms;
141 const uint32_t start = ra8_dwt_cyccnt_read();
142 while ((uint32_t)(ra8_dwt_cyccnt_read() - start) < target) {
143 __asm__ volatile("nop");
144 }
145 } else {
146 const uint32_t start = s_tick_ms;
147 while ((uint32_t)(s_tick_ms - start) < ms) {
148 __asm__ volatile("wfi");
149 }
150 }
151#endif
152}

173/*
174 * Per-subsystem tick callouts. These are declared as WEAK EXTERNS: if
175 * the firmware image links a subsystem that defines the symbol, the
176 * weak reference resolves to that strong defn and the function pointer
177 * is non-null; if the subsystem isn't linked the reference remains a
178 * null pointer and the call is skipped.
179 *
180 * - `_tx_timer_interrupt` lives in libthreadx.a. Advances
181 * the ThreadX time bases so
182 * tx_thread_sleep / TX_TIMER /
183 * semaphore-wait timeouts fire.
184 * GUARDED by `g_ra8_threadx_systick_ready`
185 * (also in libthreadx.a) -- the kernel
186 * timer state is not safe to call
187 * into until the project's
188 * `_tx_initialize_low_level` has run.
189 * - `ux_dcd_ra8_usb_irq_reenable` lives in port/usbx/src/ux_dcd_ra8_usb.c.
190 * Re-arms the USBFS NVIC line that
191 * the bridge's storm guard masks.
192 *
193 * Apps that need additional tick work can still publish a strong
194 * SysTick_Handler in their own main.c -- it wins over the weak default
195 * below. The weak-extern trick is what lets a per-app handler go away
196 * once its work is just "tick threadx and tick usb".
197 */
198/* The default SysTick_Handler below and its weak externs are firmware-only.
199 * The host unit-test build (RA8_OFF_TARGET) has no SysTick ISR, never calls
200 * this handler, and must not drag in the ThreadX / USB weak externs: the macOS
201 * test linker (ld64) does NOT resolve an undefined `weak` reference to NULL the
202 * way the ELF firmware linker does, so leaving them visible breaks every host
203 * test that links ra8_time.c. Compile the whole ISR + its externs for the
204 * firmware target only; on ELF the `weak` references still fold to NULL when a
205 * subsystem is absent and to its strong defn when present. */
206#ifndef RA8_OFF_TARGET
207
208/* NOLINTNEXTLINE(bugprone-reserved-identifier,cert-dcl37-c,cert-dcl51-cpp,readability-identifier-naming) -- vendor (ThreadX) symbol; we must use Eclipse ThreadX's actual entry-point name. */
209[[gnu::weak]] extern void _tx_timer_interrupt(void);
210[[gnu::weak]] extern void ux_dcd_ra8_usb_irq_reenable(void);
211
212/* When a non-ThreadX app is being built nothing pulls in the storage
213 * for this flag and the weak reference resolves to a null pointer at
214 * link time; the runtime check below short-circuits the conditional.
215 * When ThreadX IS linked the pointer is non-null and we read the
216 * underlying byte every SysTick. */
217[[gnu::weak]] extern volatile uint32_t g_ra8_threadx_systick_ready;
218
219/*
220 * Default SysTick_Handler -- weak so an app can still supply a strong
221 * override when needed. The default already routes ticks into ThreadX
222 * and USB via the weak externs above, so most apps no longer need a
223 * per-app SysTick_Handler at all.
224 *
225 * The vector_table.c weak alias to Default_Handler is overridden by
226 * this stronger weak symbol; an even stronger non-weak one in the
227 * application file wins above both.
228 */
229void SysTick_Handler(void);
230
231/* NOLINTNEXTLINE(misc-use-internal-linkage) -- linker symbol for vector table. */
232[[gnu::weak]] void SysTick_Handler(void)
233{
234 ra8_time_on_tick();
235 /* Only dispatch into the ThreadX timer ISR after the project's
236 * `_tx_initialize_low_level` has flagged the kernel as ready.
237 * Between `ra8_time_init()` (which arms SysTick at 1 kHz from C)
238 * and `tx_kernel_enter()` (which runs ThreadX init), the kernel
239 * timer state is uninitialised; an early call into
240 * `_tx_timer_interrupt` here HardFaults with UFSR.INVPC=1 -- the
241 * bench symptom that bricks `threadx_netx_tcp_echo` boot when its
242 * intervening `ra8_board_ethernet_init` runs long enough for
243 * SysTick to fire. Non-ThreadX apps see both externs as NULL
244 * weak references and the nested-if folds away. Written as
245 * nested ifs (no compound boolean operator) so the project's
246 * MC/DC gate does not demand an extra test vector for this
247 * ISR-only code path. */
248 if (&g_ra8_threadx_systick_ready != ((void*)0)) {
249 if (g_ra8_threadx_systick_ready != 0U) {
250 if (_tx_timer_interrupt != ((void*)0)) {
251 _tx_timer_interrupt();
252 }
253 }
254 }
255 if (ux_dcd_ra8_usb_irq_reenable != ((void*)0)) {
256 ux_dcd_ra8_usb_irq_reenable();
257 }
258}
259
260#else /* RA8_OFF_TARGET */

264 * and the macOS test linker (ld64) cannot resolve the firmware handler's
