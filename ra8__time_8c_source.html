143ra8_err_t ra8_time_init(uint32_t cpu_hz)
144{
145 if (cpu_hz == 0U) {
146 ra8_log_error(s_tag, "cpu_hz must be non-zero");
147 return k_ra8_err_invalid_arg;
148 }
149
150 const uint32_t reload = (cpu_hz / k_ra8_ms_per_sec) - 1U;
151 if (reload == 0U) {
152 ra8_log_error(s_tag, "cpu_hz too low for 1kHz tick");
153 return k_ra8_err_invalid_arg;
154 }
155
156#ifndef RA8_OFF_TARGET
157 /* Disable, programme reload, clear current count, re-enable with
158 * IRQ and CPU clock source. The SysTick registers live in the
159 * Cortex-M System Control Space and are not mapped on the host
160 * test build. */
161 *internal_csr() = 0U;
162 *internal_rvr() = reload;
163 *internal_cvr() = 0U;
164 *internal_csr() =
165 k_ra8_systick_csr_clksource | k_ra8_systick_csr_tickint | k_ra8_systick_csr_enable;
166
167 /* Enable DWT cycle counter as a PRIMASK-immune fallback for
168 * ra8_delay_ms. DEMCR.TRCENA unlocks DWT; DWT_CTRL.CYCCNTENA starts
169 * the free-running counter. */
170 volatile uint32_t* demcr = (volatile uint32_t*)k_ra8_dwt_demcr_addr;
171 volatile uint32_t* dwtctrl = (volatile uint32_t*)k_ra8_dwt_ctrl_addr;
172 *demcr |= (uint32_t)k_ra8_dwt_demcr_trcena;
173 *dwtctrl |= (uint32_t)k_ra8_dwt_ctrl_cyccntena;
174#endif
175
176 s_cycles_per_ms = cpu_hz / k_ra8_ms_per_sec;
177 s_tick_ms = 0U;
178 ra8_log_info_val(s_tag, "systick reload", reload);
179 return k_ra8_ok;
180}

220void ra8_delay_ms(uint32_t ms)
221{
222#ifdef RA8_OFF_TARGET
223 (void)ms; /* No SysTick off-target -- s_tick_ms never advances. */
224#else
225 /* If PRIMASK is set the SysTick IRQ cannot dispatch and s_tick_ms */
226 /* never advances -- a wfi-loop on s_tick_ms would hang forever. */
227 /* Fall back to DWT_CYCCNT, which ticks every CPU cycle regardless */
228 /* of PRIMASK. Once IRQs are globally enabled, the cheaper */
229 /* SysTick path is used. */
230 uint32_t primask;
231 __asm__ volatile("mrs %0, primask" : "=r"(primask));
232 if ((primask & 1U) != 0U) {
233 volatile const uint32_t* cyccnt = (volatile const uint32_t*)k_ra8_dwt_cyccnt_addr;
234 const uint32_t target = ms * s_cycles_per_ms;
235 const uint32_t start = *cyccnt;
236 while ((uint32_t)(*cyccnt - start) < target) {
237 __asm__ volatile("nop");
238 }
239 } else {
240 const uint32_t start = s_tick_ms;
241 while ((uint32_t)(s_tick_ms - start) < ms) {
242 __asm__ volatile("wfi");
243 }
244 }
245#endif
246}

267/*
268 * Per-subsystem tick callouts. These are declared as WEAK EXTERNS: if
269 * the firmware image links a subsystem that defines the symbol, the
270 * weak reference resolves to that strong defn and the function pointer
271 * is non-null; if the subsystem isn't linked the reference remains a
272 * null pointer and the call is skipped.
273 *
274 * - `_tx_timer_interrupt` lives in libthreadx.a. Advances
275 * the ThreadX time bases so
276 * tx_thread_sleep / TX_TIMER /
277 * semaphore-wait timeouts fire.
278 * GUARDED by `g_ra8_threadx_systick_ready`
279 * (also in libthreadx.a) -- the kernel
280 * timer state is not safe to call
281 * into until the project's
282 * `_tx_initialize_low_level` has run.
283 * - `ux_dcd_ra8_usb_irq_reenable` lives in port/usbx/src/ux_dcd_ra8_usb.c.
284 * Re-arms the USBFS NVIC line that
285 * the bridge's storm guard masks.
286 *
287 * Apps that need additional tick work can still publish a strong
288 * SysTick_Handler in their own main.c -- it wins over the weak default
289 * below. The weak-extern trick is what lets a per-app handler go away
290 * once its work is just "tick threadx and tick usb".
291 */
292/* The default SysTick_Handler below and its weak externs are firmware-only.
293 * The host unit-test build (RA8_OFF_TARGET) has no SysTick ISR, never calls
294 * this handler, and must not drag in the ThreadX / USB weak externs: the macOS
295 * test linker (ld64) does NOT resolve an undefined `weak` reference to NULL the
296 * way the ELF firmware linker does, so leaving them visible breaks every host
297 * test that links ra8_time.c. Compile the whole ISR + its externs for the
298 * firmware target only; on ELF the `weak` references still fold to NULL when a
299 * subsystem is absent and to its strong defn when present. */
300#ifndef RA8_OFF_TARGET
301
302/* NOLINTNEXTLINE(bugprone-reserved-identifier,cert-dcl37-c,cert-dcl51-cpp,readability-identifier-naming) -- vendor (ThreadX) symbol; we must use Eclipse ThreadX's actual entry-point name. */
303[[gnu::weak]] extern void _tx_timer_interrupt(void);
304[[gnu::weak]] extern void ux_dcd_ra8_usb_irq_reenable(void);
305
306/* When a non-ThreadX app is being built nothing pulls in the storage
307 * for this flag and the weak reference resolves to a null pointer at
308 * link time; the runtime check below short-circuits the conditional.
309 * When ThreadX IS linked the pointer is non-null and we read the
310 * underlying byte every SysTick. */
311[[gnu::weak]] extern volatile uint32_t g_ra8_threadx_systick_ready;
312
313/*
314 * Default SysTick_Handler -- weak so an app can still supply a strong
315 * override when needed. The default already routes ticks into ThreadX
316 * and USB via the weak externs above, so most apps no longer need a
317 * per-app SysTick_Handler at all.
318 *
319 * The vector_table.c weak alias to Default_Handler is overridden by
320 * this stronger weak symbol; an even stronger non-weak one in the
321 * application file wins above both.
322 */
323void SysTick_Handler(void);
324
325/* NOLINTNEXTLINE(misc-use-internal-linkage) -- linker symbol for vector table. */
326[[gnu::weak]] void SysTick_Handler(void)
327{
328 ra8_time_on_tick();
329 /* Only dispatch into the ThreadX timer ISR after the project's
330 * `_tx_initialize_low_level` has flagged the kernel as ready.
331 * Between `ra8_time_init()` (which arms SysTick at 1 kHz from C)
332 * and `tx_kernel_enter()` (which runs ThreadX init), the kernel
333 * timer state is uninitialised; an early call into
334 * `_tx_timer_interrupt` here HardFaults with UFSR.INVPC=1 -- the
335 * bench symptom that bricks `threadx_netx_tcp_echo` boot when its
336 * intervening `ra8_board_ethernet_init` runs long enough for
337 * SysTick to fire. Non-ThreadX apps see both externs as NULL
338 * weak references and the nested-if folds away. Written as
339 * nested ifs (no compound boolean operator) so the project's
340 * MC/DC gate does not demand an extra test vector for this
341 * ISR-only code path. */
342 if (&g_ra8_threadx_systick_ready != ((void*)0)) {
343 if (g_ra8_threadx_systick_ready != 0U) {
344 if (_tx_timer_interrupt != ((void*)0)) {
345 _tx_timer_interrupt();
346 }
347 }
348 }
349 if (ux_dcd_ra8_usb_irq_reenable != ((void*)0)) {
350 ux_dcd_ra8_usb_irq_reenable();
351 }
352}
353
354#else /* RA8_OFF_TARGET */

357 * symbol must exist -- but there is no ThreadX kernel or USB bridge to tick,
358 * and the macOS test linker (ld64) cannot resolve the firmware handler's
359 * undefined weak externs. Provide a minimal handler that does only the
360 * tick-counter work the test observes. */
361void SysTick_Handler(void);
