267static ra8_err_t demo_write_verify(uint32_t lbn, uint32_t tag, uint16_t* out_phys)
268{
269 RA8_CHECK_NULL_PTR(out_phys, s_tag, "out_phys must not be nullptr");
270 uint8_t blk[(size_t)k_demo_block];
271 for (uint32_t i = 0; i < (uint32_t)k_demo_block; ++i) {
272 blk[i] = (uint8_t)((i * (uint32_t)k_demo_seed_mul) + (lbn * (uint32_t)k_demo_lbn_mul) + tag);
273 }
274 RA8_RETURN_ON_ERROR(ra8_io_blockdev_write(&s_present, lbn, (uint32_t)k_demo_one_block, blk),
275 s_tag,
276 "write");
277 uint8_t got[(size_t)k_demo_block];
278 RA8_RETURN_ON_ERROR(ra8_io_blockdev_read(&s_present, lbn, (uint32_t)k_demo_one_block, got),
279 s_tag,
280 "read");
281 if (memcmp(got, blk, sizeof(blk)) != 0) {
282 return k_ra8_err_checksum_mismatch;
283 }
284 return ra8_ftl_phys_of(&s_ftl, lbn, out_phys);
285}

312static ra8_err_t demo_wear_phase(uint32_t* migrations, uint16_t* last_phys)
313{
314 RA8_CHECK_NULL_PTR(migrations, s_tag, "migrations must not be nullptr");
315 RA8_CHECK_NULL_PTR(last_phys, s_tag, "last_phys must not be nullptr");
316 RA8_RETURN_ON_ERROR(ra8_ftl_init(&s_ftl,
317 &s_bd,
318 s_map,
319 (uint32_t)k_demo_ftl_logical,
320 s_pblocks,
321 (uint32_t)k_demo_ftl_phys,
322 s_scratch),
323 s_tag,
324 "ftl init");
325 RA8_RETURN_ON_ERROR(ra8_ftl_as_blockdev(&s_ftl, &s_present), s_tag, "ftl as blockdev");
326 uint32_t moves = 0U;
327 uint16_t prev = (uint16_t)k_ra8_ftl_unmapped;
328 for (uint32_t k = 0; k < (uint32_t)k_demo_writes; ++k) {
329 uint16_t cur = 0U;
330 const ra8_err_t e =
331 demo_write_verify((uint32_t)k_demo_test_lbn, (uint32_t)k_demo_tag_base + k, &cur);
332 if (e != k_ra8_ok) {
333 return e;
334 }
335 demo_report_map((uint32_t)k_demo_test_lbn, cur);
336 if ((prev != (uint16_t)k_ra8_ftl_unmapped) && (cur != prev)) {
337 moves += 1U;
338 }
339 prev = cur;
340 }
341 *migrations = moves;
342 *last_phys = prev;
343 return k_ra8_ok;
344}

464static ra8_err_t demo_restore(uint32_t want_tag, uint16_t want_phys)
465{
466 RA8_RETURN_ON_ERROR(
467 ra8_io_blockdev_read(&s_bd, (uint32_t)k_demo_meta_lba, (uint32_t)k_demo_one_block, s_ckbuf),
468 s_tag,
469 "meta read");
470 RA8_RETURN_ON_ERROR(ra8_ftl_checkpoint_load(&s_ftl, s_ckbuf, (uint32_t)k_demo_block),
471 s_tag,
472 "checkpoint load");
473 uint8_t want[(size_t)k_demo_block] = {};
474 for (uint32_t i = 0; i < (uint32_t)k_demo_block; ++i) {
475 want[i] = (uint8_t)((i * (uint32_t)k_demo_seed_mul) +
476 ((uint32_t)k_demo_test_lbn * (uint32_t)k_demo_lbn_mul) + want_tag);
477 }
478 uint8_t got[(size_t)k_demo_block];
479 RA8_RETURN_ON_ERROR(
480 ra8_io_blockdev_read(&s_present, (uint32_t)k_demo_test_lbn, (uint32_t)k_demo_one_block, got),
481 s_tag,
482 "restored read");
483 if (memcmp(got, want, sizeof(want)) != 0) {
484 return k_ra8_err_checksum_mismatch;
485 }
486 uint16_t phys = 0U;
487 RA8_RETURN_ON_ERROR(ra8_ftl_phys_of(&s_ftl, (uint32_t)k_demo_test_lbn, &phys),
488 s_tag,
489 "restored map");
490 if (phys != want_phys) {
491 return k_ra8_err_invalid_state;
492 }
493 return k_ra8_ok;
494}

547static ra8_err_t demo_run(void)
548{
549 uint32_t migrations = 0U;
550 uint16_t last_phys = 0U;
551 RA8_RETURN_ON_ERROR(demo_wear_phase(&migrations, &last_phys), s_tag, "wear phase");
552 demo_report_wear();
553 if (migrations == 0U) {
554 return k_ra8_err_invalid_state; /* no relocation observed -- wear-levelling failed */
555 }
556 RA8_RETURN_ON_ERROR(demo_checkpoint(), s_tag, "checkpoint");
557 RA8_RETURN_ON_ERROR(demo_reopen_naive(), s_tag, "reopen naive");
558 demo_print("ra8_ftl_demo: naive re-open lost the mapping (reads erase value)\r\n");
559 RA8_RETURN_ON_ERROR(
560 demo_restore((uint32_t)k_demo_tag_base + (uint32_t)k_demo_writes - (uint32_t)k_demo_one_block,
561 last_phys),
562 s_tag,
563 "restore");
564 return k_ra8_ok;
565}
