149ra8_err_t c6_cam_camera_init(void)
150{
151 ra8_err_t err = ra8_board_camera_xclk_start((uint32_t)k_sw_cam_xclk_hz);
152 if (err != k_ra8_ok) {
153 return err;
154 }
155 err = ra8_board_camera_select_parallel();
156 if (err != k_ra8_ok) {
157 return err;
158 }
159 ra8_delay_ms((uint32_t)k_sw_cam_mode_settle_ms);
160 err = ra8_board_camera_reset();
161 if (err != k_ra8_ok) {
162 return err;
163 }
164 err = internal_c6_cam_sensor_bind();
165 if (err != k_ra8_ok) {
166 return err;
167 }
168 uint16_t chip_id = 0U;
169 if ((ra8_ov5640_probe(&s_camera_sensor, &chip_id) != k_ra8_ok) ||
170 (chip_id != (uint16_t)k_ra8_ov5640_chip_id)) {
171 return k_ra8_err_hw_init_failed;
172 }
173 err = ra8_ov5640_configure(&s_camera_sensor, k_ra8_ov5640_mode_vga_uyvy);
174 if (err != k_ra8_ok) {
175 return err;
176 }
177 err = ra8_board_camera_route_parallel_pins();
178 if (err != k_ra8_ok) {
179 return err;
180 }
181 const ra8_camera_source_ceu_cfg_t source_cfg = internal_c6_cam_source_config();
182 err = ra8_camera_source_ceu_init(&s_source, &s_source_state, &source_cfg);
183 if (err != k_ra8_ok) {
184 return err;
185 }
186 const ra8_camera_codec_jpeg_sw_cfg_t codec_cfg = {
187 .rgb_workspace = s_rgb,
188 .rgb_workspace_capacity = (uint32_t)sizeof(s_rgb),
189 .output_width = (uint16_t)k_sw_cam_stream_width,
190 .output_height = (uint16_t)k_sw_cam_stream_height,
191 .quality = (uint8_t)k_sw_cam_jpeg_quality,
192 };
193 return ra8_camera_codec_jpeg_sw_init(&s_codec, &s_codec_state, &codec_cfg);
194}

196ra8_err_t c6_cam_camera_capture_jpeg(const uint8_t** out_jpeg,
197 uint32_t* out_bytes,
198 c6_cam_frame_timing_t* out_timing)
199{
200 if ((out_jpeg == nullptr) || (out_bytes == nullptr) || (out_timing == nullptr)) {
201 return k_ra8_err_null_ptr;
202 }
203 *out_jpeg = nullptr;
204 *out_bytes = 0U;
205 *out_timing = (c6_cam_frame_timing_t){};
206 const uint32_t capture_start = ra8_time_ms();
207 out_timing->timestamp_ms = capture_start;
208 const ra8_camera_buffer_t capture_buffer = {
209 .data = s_capture,
210 .capacity = (uint32_t)sizeof(s_capture),
211 };
212 ra8_camera_frame_t captured = {};
213 ra8_err_t err = ra8_camera_source_capture(&s_source, &capture_buffer, &captured);
214 const uint32_t capture_end = ra8_time_ms();
215 out_timing->capture_ms = capture_end - capture_start;
216 if (err != k_ra8_ok) {
217 return err;
218 }
219 const ra8_camera_buffer_t encoded_buffer = {
220 .data = s_jpeg,
221 .capacity = (uint32_t)sizeof(s_jpeg),
222 };
223 ra8_camera_frame_t encoded = {};
224 err = ra8_camera_codec_encode(&s_codec, &captured, &encoded_buffer, &encoded);
225 out_timing->encode_ms = ra8_time_ms() - capture_end;
226 if (err != k_ra8_ok) {
227 return err;
228 }
229 *out_jpeg = encoded.data;
230 *out_bytes = encoded.bytes;
231 return k_ra8_ok;
232}
