166static bool internal_decode(ra8_mstp_t id, uint8_t* out_reg, uint8_t* out_bit)
167{
168 const ra8_mstp_reg_t reg_id = ra8_mstp_id_reg(id);
169 const uint8_t reg = (uint8_t)reg_id;
170 const uint8_t bit = ra8_mstp_id_bit(id);
171 if (reg >= k_ra8_mstp_reg_count) {
172 return false;
173 }
174 if (bit >= k_ra8_mstp_bit_count) {
175 return false;
176 }
177 *out_reg = reg;
178 *out_bit = bit;
179 return true;
180}

224static ra8_err_t internal_wait_readback(uint8_t reg, uint8_t bit, bool expected_stopped)
225{
226 volatile const uint32_t* p = internal_reg_ptr(reg);
227 const uint32_t mask = (uint32_t)1U << bit;
228 for (uint16_t i = 0U; i < k_ra8_mstp_readback_spin; ++i) {
229 const bool seen_stopped = (*p & mask) != 0U;
230 const bool settled = (seen_stopped == expected_stopped);
231#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
232 /* Host MMIO fault seam, keyed on the polled MSTPCR register. */
233 if (ra8_fake_mmio_wait_eval(p, (uint32_t)i, settled)) {
234 return k_ra8_ok;
235 }
236#else
237 if (settled) {
238 return k_ra8_ok;
239 }
240#endif
241 }
242 return k_ra8_err_hw_timeout;
243}

266uint32_t ra8_mstp_ns_mask_internal(uint8_t reg)
267{
268 /* MSTPCRA has no PSAR: always fully Secure-owned. */
269 static const uintptr_t k_psar_addr[k_ra8_mstp_reg_count] = {
270 0U,
271 (uintptr_t)k_ra8_mstp_psarb_addr,
272 (uintptr_t)k_ra8_mstp_psarc_addr,
273 (uintptr_t)k_ra8_mstp_psard_addr,
274 (uintptr_t)k_ra8_mstp_psare_addr,
275 };
276 if (k_psar_addr[reg] == 0U) {
277 return 0U;
278 }
279 /* Read PSARB..PSARE (reg 1..4); the Non-secure-attributed bit mask.
280 * HUM Ch 51.8.1 "PSARB : Peripheral Security Attribution Register B" p 3284 */
281 return *(volatile const uint32_t*)k_psar_addr[reg];
282}

289ra8_err_t ra8_mstp_init(void)
290{
291 ra8_log_info(s_tag, "ra8_mstp_init -- gating all modules");
292
293 /* Reset every ref count. */
294 for (uint8_t reg = 0U; reg < k_ra8_mstp_reg_count; ++reg) {
295 for (uint8_t bit = 0U; bit < k_ra8_mstp_bit_count; ++bit) {
296 s_refcount[reg][bit] = 0U;
297 }
298 }
299
300 /* Stop all peripherals. MSTPCRA bits 0..3 (SRAM0..3) must stay 0
301 * (running) or the CPU stack becomes inaccessible and causes a
302 * BusFault. All other registers use the full all-stopped pattern.
303 * HUM Ch 11.2.6..10 p 443..449. */
304 static const uint32_t k_init_vals[k_ra8_mstp_reg_count] = {
305 (uint32_t)k_ra8_mstp_safe_stopped_a, /* MSTPCRA: SRAM0-3 kept running */
306 (uint32_t)k_ra8_mstp_all_stopped, /* MSTPCRB */
307 (uint32_t)k_ra8_mstp_all_stopped, /* MSTPCRC */
308 (uint32_t)k_ra8_mstp_all_stopped, /* MSTPCRD */
309 (uint32_t)k_ra8_mstp_all_stopped, /* MSTPCRE */
310 };
311 /* HUM Ch 11.2.6 "MSTPCRA : Module Stop Control Register A", p 443 */
312 *internal_reg_ptr(0U) = k_init_vals[0U];
313 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B", p 444 */
314 *internal_reg_ptr(1U) = k_init_vals[1U];
315 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C", p 446 */
316 *internal_reg_ptr(2U) = k_init_vals[2U];
317 /* HUM Ch 11.2.9 "MSTPCRD : Module Stop Control Register D", p 448 */
318 *internal_reg_ptr(3U) = k_init_vals[3U];
319 /* HUM Ch 11.2.10 "MSTPCRE : Module Stop Control Register E", p 449 */
320 *internal_reg_ptr(4U) = k_init_vals[4U];
321
322 /* Read-back: each register must settle to the value written, but only for
323 * the bits the Secure side actually owns. A Non-secure-attributed module
324 * (PSAR bit set -- e.g. a USB controller delegated to a Non-secure world)
325 * ignores the Secure module-stop write, so its bit will not read back the
326 * commanded state; requiring it would spuriously fail boot on every
327 * TrustZone system that delegates a peripheral. Masking the read-back by the
328 * Secure-owned bits keeps a genuine stuck Secure module a hard failure while
329 * tolerating the by-design Non-secure ones. On a non-TrustZone system the
330 * mask is all-ones, so the check stays fully strict. */
331 for (uint8_t reg = 0U; reg < k_ra8_mstp_reg_count; ++reg) {
332 const uint32_t care = ~ra8_mstp_ns_mask_internal(reg);
333 if (ra8_mstp_wait_reg_settle_internal(reg, k_init_vals[reg], care) != k_ra8_ok) {
334 ra8_log_error_val(s_tag, "init read-back failed reg", (uint32_t)reg);
335 return k_ra8_err_hw_timeout;
336 }
337 }
338 return k_ra8_ok;
339}

341ra8_err_t ra8_mstp_enable(ra8_mstp_t id)
342{
343 uint8_t reg = 0U;
344 uint8_t bit = 0U;
345 if (!internal_decode(id, &reg, &bit)) {
346 ra8_log_error_val(s_tag, "enable: invalid id", (uint32_t)id);
347 return k_ra8_err_invalid_arg;
348 }
349
350 const uint8_t prev = s_refcount[reg][bit];
351 if (prev == UINT8_MAX) {
352 /* Defensive: 255 simultaneous users would mean a bug elsewhere. */
353 ra8_log_error_val(s_tag, "enable: refcount saturated", (uint32_t)id);
354 return k_ra8_err_invalid_state;
355 }
356 s_refcount[reg][bit] = (uint8_t)(prev + 1U);
357
358 if (prev != 0U) {
359 /* Already running. Nothing else to do. */
360 return k_ra8_ok;
361 }
362
363 /* HUM Ch 11.2.6 "MSTPCRA : Module Stop Control Register A", p 443
364 * -- bit clear == ungate the peripheral. */
365 volatile uint32_t* p = internal_reg_ptr(reg);
366 const uint32_t mask = (uint32_t)1U << bit;
367 *p = *p & ~mask;
368
369 const ra8_err_t err = internal_wait_readback(reg, bit, false);
370 if (err != k_ra8_ok) {
371 /* Roll the ref count back so a retry from the caller starts
372 * fresh rather than thinking the module is already on. */
373 s_refcount[reg][bit] = prev;
374 ra8_log_error_val(s_tag, "enable: read-back timeout id", (uint32_t)id);
375 return err;
376 }
377 return k_ra8_ok;
378}

380ra8_err_t ra8_mstp_disable(ra8_mstp_t id)
381{
382 uint8_t reg = 0U;
383 uint8_t bit = 0U;
384 if (!internal_decode(id, &reg, &bit)) {
385 ra8_log_error_val(s_tag, "disable: invalid id", (uint32_t)id);
386 return k_ra8_err_invalid_arg;
387 }
388
389 const uint8_t prev = s_refcount[reg][bit];
390 if (prev == 0U) {
391 ra8_log_error_val(s_tag, "disable: refcount underflow", (uint32_t)id);
392 return k_ra8_err_invalid_state;
393 }
394 s_refcount[reg][bit] = (uint8_t)(prev - 1U);
395
396 if (prev != 1U) {
397 /* Other users still active. Leave the bit cleared. */
398 return k_ra8_ok;
399 }
400
401 /* HUM Ch 11.2.6 "MSTPCRA : Module Stop Control Register A", p 443
402 * -- bit set == gate the peripheral. */
403 volatile uint32_t* p = internal_reg_ptr(reg);
404 const uint32_t mask = (uint32_t)1U << bit;
405 *p = *p | mask;
406
407 const ra8_err_t err = internal_wait_readback(reg, bit, true);
408 if (err != k_ra8_ok) {
409 /* Roll back so the next caller still sees the resource as in
410 * use and we don't lose track of pending ungates. */
411 s_refcount[reg][bit] = prev;
412 ra8_log_error_val(s_tag, "disable: read-back timeout id", (uint32_t)id);
413 return err;
414 }
415 return k_ra8_ok;
416}
