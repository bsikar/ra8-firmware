76RA8_INTERNAL static bool next_line(const char** pp, const char* end, char* buf, size_t cap)
77{
78 const char* p = *pp;
79 if (p >= end) {
80 return false;
81 }
82 size_t n = 0U;
83 while ((p < end) && (*p != '\n')) {
84 if ((n + 1U) < cap) {
85 buf[n] = *p;
86 ++n;
87 }
88 ++p;
89 }
90 if (p < end) {
91 ++p; /* consume the '\n' */
92 }
93 *pp = p;
94 buf[n] = '\0';

202RA8_INTERNAL static void harvest(const char* text,
203 size_t len,
204 const char* ua_token,
205 int target,
206 bool wildcard_target,
207 mdl_robots_t* out)
208{
209 const char* p = text;
210 const char* end = text + len;
211 bool prev_agent = false;
212 bool selected = false;
213 int grp_spec = (int)k_spec_none;
214 bool grp_wild = false;
215 char line[k_robots_line_max];
216 while (next_line(&p, end, line, sizeof(line))) {
217 char* field = nullptr;
218 char* value = nullptr;
219 if (!split_field(line, &field, &value)) {
220 continue;
221 }
222 if (ieq(field, "user-agent")) {
223 if (!prev_agent) {
224 grp_spec = (int)k_spec_none;
225 grp_wild = false;
226 }
227 const int spec = agent_spec(value, ua_token);
228 if (spec == 0) {
229 grp_wild = true;
230 } else if (spec > grp_spec) {
231 grp_spec = spec;
232 }
233 selected = group_selected(target, wildcard_target, grp_spec, grp_wild);
234 prev_agent = true;
235 } else {
236 prev_agent = false;
237 if (selected) {
238 apply_directive(out, field, value);
239 }
240 }
241 }

244void mdl_robots_parse(const char* text, size_t len, const char* ua_token, mdl_robots_t* out)
245{
246 if (out == nullptr) {
247 return;
248 }
249 *out = (mdl_robots_t){};
250 if ((text == nullptr) || (len == 0U) || (ua_token == nullptr)) {
251 return;
252 }
253 int best = (int)k_spec_none;
254 bool wild = false;
255 scan_spec(text, len, ua_token, &best, &wild);
256 if ((best < 0) && !wild) {
257 return; /* no group matches us -> no restrictions */
258 }
259 const bool wildcard_target = (best < 0) && wild;
260 harvest(text, len, ua_token, best, wildcard_target, out);
261}

264RA8_INTERNAL static bool
265glob_prefix(const char* pat, size_t patlen, const char* path, bool anchored)
266{
267 size_t pi = 0U;
268 const char* s = path;
269 size_t star = patlen + 1U; /* > patlen means "no star seen yet" */
270 const char* ss = nullptr;
271 while (*s != '\0') {
272 if ((pi < patlen) && (pat[pi] == '*')) {
273 star = pi;
274 ++pi;
275 ss = s;
276 } else if ((pi < patlen) && (pat[pi] == *s)) {
277 ++pi;
278 ++s;
279 } else if ((star <= patlen) && (ss != nullptr)) {
280 /* star <= patlen holds only after `ss = s` ran, so ss is non-NULL here;
281 * the explicit check makes that invariant visible to the analyser. */
282 pi = star + 1U;
283 ++ss;
284 s = ss;
285 } else {
286 return false;
287 }
288 if ((pi == patlen) && !anchored) {
289 return true; /* whole pattern consumed -> prefix match */
290 }
291 }
292 while ((pi < patlen) && (pat[pi] == '*')) {
293 ++pi;
294 }
295 return pi == patlen;
296}
