64RA8_INTERNAL static char* trim_ws(char* s)
65{
66 while ((*s == ' ') || (*s == '\t') || (*s == '\r')) {
67 ++s;
68 }
69 size_t n = strlen(s);
70 while ((n > 0U) && ((s[n - 1U] == ' ') || (s[n - 1U] == '\t') || (s[n - 1U] == '\r'))) {
71 s[n - 1U] = '\0';
72 --n;
73 }
74 return s;
75}

78RA8_INTERNAL static bool next_line(const char** pp, const char* end, char* buf, size_t cap)
79{
80 const char* p = *pp;
81 if (p >= end) {
82 return false;
83 }
84 size_t n = 0U;
85 while ((p < end) && (*p != '\n')) {
86 if ((n + 1U) < cap) {
87 buf[n] = *p;
88 ++n;
89 }
90 ++p;
91 }
92 if (p < end) {
93 ++p; /* consume the '\n' */
94 }
95 *pp = p;
96 buf[n] = '\0';
97 char* hash = strchr(buf, '#');
98 if (hash != nullptr) {
99 *hash = '\0';
100 }
101 return true;
102}

130RA8_INTERNAL static void
131scan_spec(const char* text, size_t len, const char* ua_token, int* best, bool* wild)
132{
133 *best = (int)k_spec_none;
134 *wild = false;
135 const char* p = text;
136 const char* end = text + len;
137 char line[k_robots_line_max];
138 while (next_line(&p, end, line, sizeof(line))) {
139 char* field = nullptr;
140 char* value = nullptr;
141 if (!split_field(line, &field, &value)) {
142 continue;
143 }
144 if (ieq(field, "user-agent")) {
145 const int spec = agent_spec(value, ua_token);
146 if (spec == 0) {
147 *wild = true;
148 } else if (spec > *best) {
149 *best = spec;
150 }
151 }
152 }
153}

163RA8_INTERNAL static void add_rule(mdl_robots_t* out, mdl_robots_rule_kind_t kind, const char* value)
164{
165 if ((value[0] == '\0') || (out->count >= (size_t)k_mdl_robots_max_rules)) {
166 return; /* empty pattern imposes no restriction; full table stops growing */
167 }
168 mdl_robots_rule_t* r = &out->rules[out->count];
169 r->kind = kind;
170 (void)snprintf(r->path, sizeof(r->path), "%s", value);
171 r->len = (uint16_t)strlen(r->path);
172 out->count++;
173}

176RA8_INTERNAL static void set_crawl(mdl_robots_t* out, const char* value)
177{
178 const double secs = strtod(value, nullptr);
179 if (secs <= 0.0) {
180 return;
181 }
182 uint32_t ms = (secs >= ((double)k_crawl_cap_ms / (double)k_ms_per_s))
183 ? (uint32_t)k_crawl_cap_ms
184 : (uint32_t)(secs * (double)k_ms_per_s);
185 if (!out->have_crawl_delay || (ms > out->crawl_delay_ms)) {
186 out->crawl_delay_ms = ms;
187 }
188 out->have_crawl_delay = true;
189}

192RA8_INTERNAL static void apply_directive(mdl_robots_t* out, const char* field, const char* value)
193{
194 if (ieq(field, "disallow")) {

198 } else if (ieq(field, "crawl-delay")) {

204RA8_INTERNAL static void harvest(const char* text,
205 size_t len,
206 const char* ua_token,
207 int target,
208 bool wildcard_target,
209 mdl_robots_t* out)
210{
211 const char* p = text;
212 const char* end = text + len;
213 bool prev_agent = false;
214 bool selected = false;
215 int grp_spec = (int)k_spec_none;
216 bool grp_wild = false;
217 char line[k_robots_line_max];
218 while (next_line(&p, end, line, sizeof(line))) {
219 char* field = nullptr;
220 char* value = nullptr;
221 if (!split_field(line, &field, &value)) {
222 continue;
223 }
224 if (ieq(field, "user-agent")) {
225 if (!prev_agent) {
226 grp_spec = (int)k_spec_none;
227 grp_wild = false;
228 }
229 const int spec = agent_spec(value, ua_token);
230 if (spec == 0) {
231 grp_wild = true;
232 } else if (spec > grp_spec) {
233 grp_spec = spec;
234 }
235 selected = group_selected(target, wildcard_target, grp_spec, grp_wild);
236 prev_agent = true;
237 } else {
238 prev_agent = false;
239 if (selected) {
240 apply_directive(out, field, value);
241 }
242 }
243 }
244}

246void mdl_robots_parse(const char* text, size_t len, const char* ua_token, mdl_robots_t* out)
247{
248 if (out == nullptr) {
249 return;
250 }
251 *out = (mdl_robots_t){};
252 if ((text == nullptr) || (len == 0U) || (ua_token == nullptr)) {
253 return;
254 }
255 int best = (int)k_spec_none;
256 bool wild = false;
257 scan_spec(text, len, ua_token, &best, &wild);
258 if ((best < 0) && !wild) {
259 return; /* no group matches us -> no restrictions */
260 }
261 const bool wildcard_target = (best < 0) && wild;
262 harvest(text, len, ua_token, best, wildcard_target, out);
263}

266RA8_INTERNAL static bool
267glob_prefix(const char* pat, size_t patlen, const char* path, bool anchored)
268{
269 size_t pi = 0U;
270 const char* s = path;
271 size_t star = patlen + 1U; /* > patlen means "no star seen yet" */
272 const char* ss = nullptr;
273 while (*s != '\0') {
274 if ((pi < patlen) && (pat[pi] == '*')) {
275 star = pi;
276 ++pi;
277 ss = s;
278 } else if ((pi < patlen) && (pat[pi] == *s)) {
279 ++pi;
280 ++s;
281 } else if ((star <= patlen) && (ss != nullptr)) {
282 /* star <= patlen holds only after `ss = s` ran, so ss is non-NULL here;
283 * the explicit check makes that invariant visible to the analyser. */
284 pi = star + 1U;
285 ++ss;
286 s = ss;
287 } else {
288 return false;
289 }
290 if ((pi == patlen) && !anchored) {
291 return true; /* whole pattern consumed -> prefix match */
292 }
293 }
294 while ((pi < patlen) && (pat[pi] == '*')) {
295 ++pi;
296 }
297 return pi == patlen;
298}

312RA8_INTERNAL static const mdl_robots_rule_t* best_matching_rule(const mdl_robots_t* robots,
313 const char* path)
314{
315 const mdl_robots_rule_t* best = nullptr;
316 int best_len = (int)k_spec_none;
317 for (size_t i = 0U; i < robots->count; ++i) {
318 const mdl_robots_rule_t* r = &robots->rules[i];
319 if (!robots_match(r->path, r->len, path)) {
320 continue;
321 }
322 const bool is_allow = (r->kind == k_mdl_rule_allow);
323 if (((int)r->len > best_len) || (((int)r->len == best_len) && is_allow)) {
324 best_len = (int)r->len;
325 best = r;
326 }
327 }
328 return best;
329}
