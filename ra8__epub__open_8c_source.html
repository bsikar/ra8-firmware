123RA8_INTERNAL
124static void priv_dirname(const char* path, char* dst, size_t cap)
125{
126 if (dst == nullptr || cap == 0U) {
127 return; /* GCOVR_EXCL_LINE -- passes opf_dir (never NULL), k_ra8_epub_max_path_len (never 0) */
128 }
129 dst[0] = '\0';
130 if (path == nullptr) {
131 return; /* GCOVR_EXCL_LINE -- callsite passes cres.opf_path (local array, never NULL) */
132 }
133 const char* slash = strrchr(path, '/');
134 if (slash == nullptr) {
135 return;
136 }
137 size_t len = (size_t)(slash - path) + 1U;
138 /* GCOVR_EXCL_START -- opf_path is bounded by cap; len < cap always holds */
139 if (len >= cap) {
140 len = cap - 1U;
141 }
142 /* GCOVR_EXCL_STOP */
143 priv_byte_copy((uint8_t*)dst, (const uint8_t*)path, len);
144 dst[len] = '\0';
145}

169static ra8_err_t
170priv_extract(mz_zip_archive* zip, const char* name, uint8_t* buf, size_t cap, size_t* got)
171{
172 *got = 0U;
173 int32_t idx = mz_zip_reader_locate_file(zip, name, nullptr, 0U);
174 if (idx < 0) {
175 return k_ra8_err_not_found;
176 }
177 mz_zip_archive_file_stat st;
178 if (mz_zip_reader_file_stat(zip, (mz_uint)idx, &st) == MZ_FALSE) {
179 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- stat cannot fail: locate found it */
180 }
181 const ra8_err_t gerr = ra8_epub_zip_guard_entry(&st);
182 if (gerr != k_ra8_ok) {
183 return gerr; /* lying header / declared bomb: reject before inflation */
184 }
185 if ((size_t)st.m_uncomp_size > cap) {
186 return k_ra8_err_no_mem;
187 }
188 if (mz_zip_reader_extract_to_mem(zip, (mz_uint)idx, buf, cap, 0U) == MZ_FALSE) {
189 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- cannot fail: stat validated it */
190 }
191 *got = (size_t)st.m_uncomp_size;
192 return k_ra8_ok;
193}

254RA8_INTERNAL
255static void priv_load_toc(mz_zip_archive* zip, ra8_epub_book_t* book, uint8_t* scratch, size_t cap)
256{
257 if (book->toc_kind == (uint8_t)k_ra8_epub_toc_none) {
258 return;
259 }
260 if (book->toc_path[0] == '\0') {
261 return;
262 }
263
264 char full_path[k_ra8_epub_max_path_len];
265 ra8_epub_internal_join_path(book->opf_dir, book->toc_path, full_path, sizeof(full_path));
266
267 size_t got = 0U;
268 ra8_err_t err = priv_extract(zip, full_path, scratch, cap, &got);
269 if (err == k_ra8_err_not_found) {
270 /* Some EPUBs store the nav/NCX href already rooted at the archive. */
271 err = priv_extract(zip, book->toc_path, scratch, cap, &got);
272 }
273 if (err != k_ra8_ok) {
274 return;
275 }
276
277 if (book->toc_kind == (uint8_t)k_ra8_epub_toc_nav) {
278 (void)ra8_epub_xml_parse_nav(scratch, got, book);
279 } else {
280 (void)ra8_epub_xml_parse_ncx(scratch, got, book);
281 }
282}

305RA8_INTERNAL
306static ra8_err_t priv_parse_archive(mz_zip_archive* zip,
307 ra8_epub_book_t* out_book,
308 uint8_t* opf_scratch,
309 size_t opf_cap)
310{
311 /* Static rather than auto: 4 KiB on the stack would blow the
312 * NASA-rule stack-usage budget. EPUB parser is single-threaded
313 * init-context-only, so the static scratch buffer is safe. */
314 static uint8_t s_container_buf[k_ra8_epub_container_xml_buf];
315 size_t got = 0U;
316 ra8_err_t err =
317 priv_extract(zip, k_container_path, s_container_buf, sizeof(s_container_buf), &got);
318 if (err != k_ra8_ok) {
319 return err;
320 }
321
322 ra8_epub_container_result_t cres = {};
323 err = ra8_epub_xml_parse_container(s_container_buf, got, &cres);
324 if (err != k_ra8_ok) {
325 return err;
326 }
327
328 size_t opf_got = 0U;
329 err = priv_extract(zip, cres.opf_path, opf_scratch, opf_cap, &opf_got);
330 if (err != k_ra8_ok) {
331 return err;
332 }
333
334 priv_dirname(cres.opf_path, out_book->opf_dir, k_ra8_epub_max_path_len);
335 err = ra8_epub_xml_parse_opf(opf_scratch, opf_got, out_book);
336 if (err != k_ra8_ok) {
337 return err;
338 }
339
340 /* Optional navigation document (NCX / nav.xhtml). Reuses opf_scratch
341 * now that the OPF has been parsed into the book. Best-effort. */
342 priv_load_toc(zip, out_book, opf_scratch, opf_cap);
343 return k_ra8_ok;
344}

404static size_t priv_stream_read(void* opaque, mz_uint64 file_ofs, void* buf, size_t n)
405{
406 const ra8_epub_stream_media_t* sm = (const ra8_epub_stream_media_t*)opaque;
407 if (sm == nullptr || sm->read == nullptr) {
408 return 0U; /* GCOVR_EXCL_LINE -- open binds validated media; never called with NULL opaque */
409 }

411 return 0U; /* GCOVR_EXCL_LINE -- miniz clamps reads within m_archive_size before calling */
412 }
413 const uint64_t avail = sm->size - file_ofs;
414 const size_t want = ((uint64_t)n > avail) ? (size_t)avail : n;
415 return sm->read(sm->ctx, (uint64_t)file_ofs, buf, want);

439RA8_INTERNAL
440static ra8_err_t priv_finish_open(mz_zip_archive* zip, ra8_epub_book_t* out_book)
441{
442 if (zip == nullptr || out_book == nullptr) {
443 return k_ra8_err_null_ptr; /* GCOVR_EXCL_LINE -- callers pass storage + non-NULL out_book */
444 }
445 /* Archive-level decompression-limits guard: an archive whose central
446 * directory floods the policy entry cap dies before any entry work. */
447 const ra8_err_t gerr = ra8_epub_zip_guard_archive(zip);
448 if (gerr != k_ra8_ok) {
449 priv_zip_destroy(zip);
450 return gerr;
451 }
452 /* Static (file-scope) OPF scratch keeps the firmware stack frame small --
453 * the OPF blob can be tens of KiB and would otherwise blow the per-thread
454 * stack budget. Single-threaded init-context use makes the static safe. */
455 static uint8_t s_opf_buf[k_ra8_epub_opf_xml_buf];
456 ra8_err_t err = priv_parse_archive(zip, out_book, s_opf_buf, sizeof(s_opf_buf));
457 if (err != k_ra8_ok) {
458 priv_zip_destroy(zip);
459 return err;
460 }
461 out_book->zip_archive_active = 1U;
462 out_book->in_use = 1U;
463 return k_ra8_ok;
464}

471ra8_err_t ra8_epub_open(const void* media, const char* path, ra8_epub_book_t* out_book)
472{
473 (void)path;
474 if (media == nullptr || out_book == nullptr) {
475 return k_ra8_err_null_ptr;
476 }
477 const ra8_epub_mem_media_t* mem = (const ra8_epub_mem_media_t*)media;
478 if (mem->data == nullptr || mem->size == 0U) {
479 return k_ra8_err_invalid_arg;
480 }
481
482 /* Zero-init the book up front so failure paths return a clean record.
483 * This also clears `zip_archive_storage` to a known state for miniz. */
484 priv_byte_zero((uint8_t*)out_book, sizeof(*out_book));
485
486 /* Place the mz_zip_archive directly in the book record's inline
487 * storage. No heap allocation -- NASA Rule 3 compliance. The byte-
488 * storage punning is intentional and documented in the header. */
489 void* const zip_storage = &out_book->zip_archive_storage[0];
490 mz_zip_archive* zip = (mz_zip_archive*)zip_storage;
491 priv_set_miniz_alloc(zip);
492 if (mz_zip_reader_init_mem(zip, mem->data, mem->size, 0U) == MZ_FALSE) {
493 return k_ra8_err_validation_failed;
494 }
495
496 const ra8_err_t err = priv_finish_open(zip, out_book);
497 if (err != k_ra8_ok) {
498 return err;
499 }
500 out_book->zip_bytes = mem->data;
501 out_book->zip_size = mem->size;
502 return k_ra8_ok;
503}

505ra8_err_t ra8_epub_open_streamed(const ra8_epub_stream_media_t* media,
506 const char* path,
507 ra8_epub_book_t* out_book)
508{
509 (void)path;
510 if (media == nullptr || out_book == nullptr) {
511 return k_ra8_err_null_ptr;
512 }
513 if (media->read == nullptr || media->size == 0U) {
514 return k_ra8_err_invalid_arg;
515 }
516
517 /* Zero-init the book up front (also clears the inline miniz storage). */
518 priv_byte_zero((uint8_t*)out_book, sizeof(*out_book));
519
520 /* The book owns the media descriptor so miniz's `m_pIO_opaque` has a stable
521 * address for the book's whole lifetime -- per-entry reads (chapters, cover,
522 * resources) happen long after this call returns and dereference it. */
523 out_book->stream_media = *media;
524
525 void* const zip_storage = &out_book->zip_archive_storage[0];
526 mz_zip_archive* zip = (mz_zip_archive*)zip_storage;
527 priv_set_miniz_alloc(zip);
528 zip->m_pRead = priv_stream_read;
529 zip->m_pIO_opaque = &out_book->stream_media;
530 /* User-read reader: reads only the ZIP tail (EOCD + central directory) now;
531 * each entry is inflated on demand later through `priv_stream_read`. */
532 if (mz_zip_reader_init(zip, (mz_uint64)media->size, 0U) == MZ_FALSE) {
533 return k_ra8_err_validation_failed;
534 }
535
536 const ra8_err_t err = priv_finish_open(zip, out_book);
537 if (err != k_ra8_ok) {
538 return err;
539 }
540 /* Streamed books hold no resident blob -- every read goes through the callback. */
541 out_book->zip_bytes = nullptr;
542 out_book->zip_size = (size_t)media->size;
543 return k_ra8_ok;
544}

546ra8_err_t ra8_epub_close(ra8_epub_book_t* book)
547{
548 if (book == nullptr) {
549 return k_ra8_err_null_ptr;
550 }
551 if (book->in_use == 0U) {
552 return k_ra8_err_not_initialized;
553 }
554 if (book->zip_archive_active != 0U) {
555 void* const zip_storage = &book->zip_archive_storage[0];
556 priv_zip_destroy((mz_zip_archive*)zip_storage);
557 book->zip_archive_active = 0U;
558 }
559 book->in_use = 0U;
560 book->chapter_count = 0U;
561 return k_ra8_ok;
