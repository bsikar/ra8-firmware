375static void fs_fmt_fill_payload(void)
376{
377 uint32_t state = (uint32_t)k_fs_fmt_prng_seed;
378 for (uint32_t i = 0U; i < (uint32_t)k_fs_fmt_payload_bytes; i++) {
379 state = (state * (uint32_t)k_fs_fmt_prng_mul) + (uint32_t)k_fs_fmt_prng_add;
380 s_payload[i] = (uint8_t)((state >> k_fs_fmt_prng_byte_shift) & k_fs_fmt_byte_mask);
381 }
382}

527[[nodiscard]] static ra8_err_t fs_fmt_run_one_type(ra8_fs_type_t type)
528{
529 ra8_fs_backend_t backend = {};
530 if (ra8_sdmmc_spi_bind_fs_backend(&backend) != k_ra8_ok) {
531 fs_fmt_print_line(type, k_msg_fail_fmt, (uint32_t)sizeof(k_msg_fail_fmt) - 1U);
532 return k_ra8_err_hw_error;
533 }
534 ra8_fs_format_opts_t opts = {};
535 opts.type = type;
536 opts.label = "RAFSFMT";
537 ra8_err_t err = ra8_fs_format(&backend, &opts);
538 if (err != k_ra8_ok) {
539 /* invalid_size == the card cannot hold this FAT type (too large/small): the
540 * caller treats that as a SKIP, not a FAIL, so do not print the FAIL line. */
541 if (err != k_ra8_err_invalid_size) {
542 fs_fmt_print_line(type, k_msg_fail_fmt, (uint32_t)sizeof(k_msg_fail_fmt) - 1U);
543 }
544 return err;
545 }
546 ra8_fs_mount_t* mount = nullptr;
547 err = ra8_fs_mount(&backend, &mount);
548 if (err != k_ra8_ok) {
549 fs_fmt_print_line(type, k_msg_fail_mount, (uint32_t)sizeof(k_msg_fail_mount) - 1U);
550 return err;
551 }
552 if (mount->type != type) {
553 fs_fmt_print_line(type, k_msg_fail_type, (uint32_t)sizeof(k_msg_fail_type) - 1U);
554 (void)ra8_fs_unmount(mount);
555 return k_ra8_err_validation_failed;
556 }
557 err = fs_fmt_write_cycle(mount, type);
558 if (err != k_ra8_ok) {
559 (void)ra8_fs_unmount(mount);
560 return err;
561 }
562 (void)ra8_fs_unmount(mount);
563 fs_fmt_print_line(type, k_msg_pass_suf, (uint32_t)sizeof(k_msg_pass_suf) - 1U);
564 return k_ra8_ok;
565}

703void main(void)
704{
705 uint32_t pclka_hz = 0U;
706 fs_fmt_setup_or_halt(&pclka_hz);
707 ra8_isr_globals_enable();
708 ra8_log_init();
709 FS_FMT_PUTS(k_msg_boot);
710
711 fs_fmt_init_card_or_halt(&pclka_hz);
712 fs_fmt_fill_payload();
713
714 static const ra8_fs_type_t k_types[k_fs_fmt_idx_count] = {
715 k_ra8_fs_type_fat12,
716 k_ra8_fs_type_fat16,
717 k_ra8_fs_type_fat32,
718 };
719 uint32_t passed = 0U;
720 for (uint32_t i = 0U; i < (uint32_t)k_fs_fmt_idx_count; i++) {
721 const ra8_err_t r = fs_fmt_run_one_type(k_types[i]);
722 if (r == k_ra8_ok) {
723 passed++;
724 } else if (r == k_ra8_err_invalid_size) {
725 /* This FAT type does not fit this card's capacity -- skip it, do not halt.
726 * A big card (e.g. 128 GB) takes only FAT32; a tiny one only FAT12/16. */
727 fs_fmt_print_line(k_types[i], k_msg_skip_size, (uint32_t)sizeof(k_msg_skip_size) - 1U);
728 } else {
729 fs_fmt_panic_halt(); /* a real failure (mount / write / verify). */
730 }
731 }
732 /* exFAT trial (full cycle: format + mount + create/write/rename/unlink); same
733 * SKIP semantics as the FAT types above. */
734 const ra8_err_t ex = fs_fmt_run_exfat();
735 if (ex == k_ra8_ok) {
736 passed++;
737 } else if (ex == k_ra8_err_invalid_size) {
738 fs_fmt_print_line(k_ra8_fs_type_exfat, k_msg_skip_size, (uint32_t)sizeof(k_msg_skip_size) - 1U);
739 } else {
740 fs_fmt_panic_halt();
741 }
742 if (passed == 0U) {
743 fs_fmt_panic_halt(); /* no FAT type fit the card -- unexpected. */
744 }
745 fs_fmt_print(k_msg_all_pass, (uint32_t)sizeof(k_msg_all_pass) - 1U);
746 fs_fmt_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
747
748 while (1) {
749 __asm__ volatile("wfi");
750 }
751}
