86static inline uint16_t internal_ra8_doc_run_16(uint16_t seed, uint16_t operand)
87{
88 volatile r_doc_regs_t* reg = ra8_doc();
89 volatile uint16_t* dodsr0 = (volatile uint16_t*)&reg->DODSR0;
90 volatile uint16_t* dodir = (volatile uint16_t*)&reg->DODIR;
91 /* Seed the accumulator first. */
92 /* HUM Ch 57.2.5 "DODSR0 : DOC Data Setting Register 0" p 3521 */
93 *dodsr0 = seed;
94 /* Writing DODIR triggers the operation. */
95 /* HUM Ch 57.2.4 "DODIR : DOC Data Input Register" p 3521 */
96 *dodir = operand;
97 /* Read back the accumulator the operation just updated. */
98 /* HUM Ch 57.2.5 "DODSR0 : DOC Data Setting Register 0" p 3521 */
99 return *dodsr0;
100}

110 /* HUM Ch 57.2.1 "DOCR : DOC Control Register" p 3519 */
111 reg->DOCR = 0U;

113 /* HUM Ch 57.2.3 "DOSCR : DOC Status Clear Register" p 3521 */

116 /* HUM Ch 57.2.4 "DODIR : DOC Data Input Register" p 3521 */

148[[nodiscard]] ra8_err_t
149ra8_doc_set_window(uint16_t lower, uint16_t upper, ra8_doc_window_polarity_t polarity)
150{
151 if (lower >= upper) {
152 ra8_log_error(s_tag, "set_window: lower must be strictly less than upper");
153 return k_ra8_err_invalid_arg;
154 }
155 if ((uint8_t)polarity > (uint8_t)k_ra8_doc_window_outside) {
156 ra8_log_error(s_tag, "set_window: polarity out of range");
157 return k_ra8_err_invalid_arg;
158 }
159
160 volatile r_doc_regs_t* reg = ra8_doc();
161 volatile uint16_t* dodsr0 = (volatile uint16_t*)&reg->DODSR0;
162 volatile uint16_t* dodsr1 = (volatile uint16_t*)&reg->DODSR1;
163 uint8_t dcsel;
164
165 if (polarity == k_ra8_doc_window_outside) {
166 dcsel = (uint8_t)k_ra8_doc_dcsel_outside;
167 } else {
168 dcsel = (uint8_t)k_ra8_doc_dcsel_inside;
169 }
170
171 /* OMS=00 (compare), DOBW=0 (16-bit), DCSEL=inside(4) or outside(5). */
172 /* HUM Ch 57.2.1 "DOCR : DOC Control Register" p 3519 */
173 reg->DOCR =
174 (uint8_t)((uint8_t)(dcsel << (uint8_t)k_ra8_doc_bit_dcsel) & (uint8_t)k_ra8_doc_mask_dcsel);
175 /* Lower threshold; the constraint DODSR1 > DODSR0 is caller-validated. */
176 /* HUM Ch 57.2.5 "DODSR0 : DOC Data Setting Register 0" p 3521 */
177 *dodsr0 = lower;
178 /* Upper threshold; only used in window comparison mode. */
179 /* HUM Ch 57.2.6 "DODSR1 : DOC Data Setting Register 1" p 3522 */
180 *dodsr1 = upper;
181 /* Clear any stale DOPCF so the first compare sees a fresh flag. */
182 /* HUM Ch 57.2.3 "DOSCR : DOC Status Clear Register" p 3521 */
183 reg->DOSCR = (uint8_t)k_ra8_doc_mask_dopcfcl;
184
185 ra8_log_info(s_tag, "set_window");
186 return k_ra8_ok;
187}

190[[nodiscard]] ra8_err_t ra8_doc_window_compare(uint16_t value, bool* out_flag)
191{
192 RA8_CHECK_NULL_PTR(out_flag, s_tag, "out_flag must not be nullptr");
193
194 volatile r_doc_regs_t* reg = ra8_doc();
195 /* Precondition: OMS must be 00 (compare mode); set_window() enforces this. */
196 /* HUM Ch 57.2.1 "DOCR : DOC Control Register" p 3519 */
197 if ((reg->DOCR & (uint8_t)k_ra8_doc_mask_oms) != 0U) {
198 ra8_log_error(s_tag, "window_compare: DOC not in compare mode");
199 return k_ra8_err_invalid_state;
200 }
201
202 volatile uint16_t* dodir = (volatile uint16_t*)&reg->DODIR;
203 /* Clear any prior flag before triggering a new comparison. */
204 /* HUM Ch 57.2.3 "DOSCR : DOC Status Clear Register" p 3521 */
205 reg->DOSCR = (uint8_t)k_ra8_doc_mask_dopcfcl;
206 /* Writing DODIR triggers the hardware comparison; DOSR.DOPCF is set.
207 * The RAM-backed host register file has no comparator engine, so on
208 * the unit-test build DOSR simply retains whatever the test staged
209 * before the call -- both flag legs below are driven that way. */
210 /* HUM Ch 57.2.4 "DODIR : DOC Data Input Register" p 3521 */
211 *dodir = value;
212
213 /* Read DOPCF -- set by the silicon comparator after the compare. */
214 /* HUM Ch 57.2.2 "DOSR : DOC Flag Status Register" p 3520 */
215 *out_flag = ((reg->DOSR & (uint8_t)k_ra8_doc_mask_dopcf) != 0U);
216 /* Clear DOPCF so the flag reflects only the last comparison. */
217 /* HUM Ch 57.2.3 "DOSCR : DOC Status Clear Register" p 3521 */
218 reg->DOSCR = (uint8_t)k_ra8_doc_mask_dopcfcl;
219
220 return k_ra8_ok;
221}
