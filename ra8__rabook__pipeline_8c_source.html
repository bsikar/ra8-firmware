90RA8_INTERNAL
91static const uint8_t* s_downscale_if_needed(const ra8_rabook_pipeline_scratch_t* scr,
92 const uint8_t* pixels,
93 uint16_t sw,
94 uint16_t sh,
95 uint16_t ow,
96 uint16_t oh)
97{
98 if (ow == sw && oh == sh) {
99 return pixels;

103 return nullptr;

105 if (ra8_rabook_gray4_downscale(pixels, sw, sh, scr->gray, ow, oh) != k_ra8_ok) {
106 return nullptr;
107 }
108 return scr->gray;
109}

179RA8_INTERNAL
180static uint32_t s_transcode_image(ra8_rabook_ctx_t* ctx,
181 const ra8_rabook_pipeline_scratch_t* scr,
182 uint32_t id_off,
183 size_t raw_len)
184{
185 int sw = 0;
186 int sh = 0;
187 int comp = 0;
188 ra8_img_arena_bind(scr->img_arena);
189 stbi_uc* pixels =
190 stbi_load_from_memory(scr->image_raw, /* alloc-allow: stb backed by ra8_img_arena */
191 (int)raw_len,
192 &sw,
193 &sh,
194 &comp,
195 (int)k_stbi_grey);
196 if (pixels == nullptr) {
197 ra8_img_arena_unbind();
198 ra8_log_error(s_tag, "stb_image decode failed");
199 return k_ra8_book_nil;
200 }
201
202 /* Downscale is opt-in: max_image_edge == 0 (the default) preserves the
203 * source resolution so zoomable content (manga pages) keeps every pixel. */
204 uint16_t ow = (uint16_t)sw;
205 uint16_t oh = (uint16_t)sh;
206 if (scr->max_image_edge != 0U) {
207 ra8_rabook_gray4_output_dims((uint16_t)sw, (uint16_t)sh, scr->max_image_edge, &ow, &oh);
208 }
209
210 const uint8_t* gray_src = s_downscale_if_needed(scr, pixels, (uint16_t)sw, (uint16_t)sh, ow, oh);
211 if (gray_src == nullptr) {
212 stbi_image_free(pixels); /* alloc-allow: stb backed by ra8_img_arena */
213 ra8_img_arena_unbind();
214 return k_ra8_book_nil;
215 }
216
217 uint32_t encoded_size = 0U;
218 ra8_err_t enc_err = s_encode_gray(scr, gray_src, ow, oh, &encoded_size);
219 stbi_image_free(pixels); /* alloc-allow: stb backed by ra8_img_arena */
220 ra8_img_arena_unbind();
221
222 if (enc_err != k_ra8_ok) {
223 return k_ra8_book_nil;
224 }
225
226 return ra8_rabook_add_image(ctx,
227 id_off,
228 ow,
229 oh,
230 (uint8_t)k_ra8_book_image_gray4,
231 scr->pixel_format,
232 scr->image_raw,
233 encoded_size);
234}

328RA8_INTERNAL
329static ra8_err_t s_compile_stylesheets(ra8_rabook_ctx_t* ctx,
330 const ra8_rabook_pipeline_scratch_t* scr,
331 ra8_epub_book_t* epub)
332{
333 const uint16_t count = ra8_epub_manifest_count(epub);
334 for (uint16_t i = 0U; i < count; i++) {
335 const ra8_epub_manifest_item_t* item = ra8_epub_manifest_item(epub, i);
336 if (item == nullptr) {
337 continue;
338 }
339 if (strcmp(item->media_type, "text/css") != 0) {
340 continue;
341 }
342 size_t got = 0U;
343 ra8_err_t err =
344 ra8_epub_get_resource(epub, item->href, (uint8_t*)scr->css, scr->css_cap - 1U, &got);
345 if (err == k_ra8_err_not_found) {
346 continue; /* desktop skips a css item absent from the archive */
347 }
348 if (err != k_ra8_ok) {
349 return err;
350 }
351 scr->css[got] = '\0';
352 const uint32_t source_off = ra8_rabook_intern(ctx, scr->css);
353 (void)ra8_rabook_add_stylesheet(ctx, source_off, (uint32_t)k_ra8_book_nil);
354 }
355 return k_ra8_ok;
356}

382RA8_INTERNAL
383static uint32_t s_add_manifest_image(ra8_rabook_ctx_t* ctx,
384 const ra8_rabook_pipeline_scratch_t* scr,
385 ra8_epub_book_t* epub,
386 const ra8_epub_manifest_item_t* item)
387{
388 const char* const img_prefix = "image/";
389 uint8_t fmt = (uint8_t)k_ra8_book_image_gray4;
390 if (strcmp(item->media_type, "image/svg+xml") == 0) {
391 fmt = (uint8_t)k_ra8_book_image_svg;
392 } else if (strncmp(item->media_type, img_prefix, strlen(img_prefix)) != 0) {
393 return (uint32_t)k_ra8_book_nil; /* not an image item */
394 }
395
396#ifdef RA8_RABOOK_NO_RASTER
397 if (fmt != (uint8_t)k_ra8_book_image_svg) {
398 ra8_log_error(s_tag, "raster image skipped (text/CSS/SVG-only build)");
399 return (uint32_t)k_ra8_book_nil;
400 }
401#endif
402
403 size_t got = 0U;
404 ra8_err_t err = ra8_epub_get_resource(epub, item->href, scr->image_raw, scr->image_cap, &got);
405 if (err != k_ra8_ok) {
406 return (uint32_t)k_ra8_book_nil; /* absent / unreadable -> skip, like the desktop */
407 }
408
409 const uint32_t id_off = ra8_rabook_intern(ctx, item->href);
410 if (fmt == (uint8_t)k_ra8_book_image_svg) {
411 return ra8_rabook_add_image(ctx,
412 id_off,
413 0U,
414 0U,
415 (uint8_t)k_ra8_book_image_svg,
416 (uint8_t)k_ra8_book_pixfmt_gray4, /* unused for SVG; store 0 */
417 scr->image_raw,
418 (uint32_t)got);
419 }
420#ifndef RA8_RABOOK_NO_RASTER
421 return s_transcode_image(ctx, scr, id_off, got);
422#else
423 return (uint32_t)k_ra8_book_nil; /* unreachable: raster returned nil above */
424#endif
425}

599RA8_INTERNAL
600static ra8_err_t s_compile_to_blob(ra8_epub_book_t* epub,
601 const ra8_rabook_buffers_t* bufs,
602 const ra8_rabook_pipeline_scratch_t* scr,
603 const void** out_blob,
604 uint32_t* out_len)
605{
606 ra8_rabook_ctx_t ctx = {};
607 ra8_err_t err = ra8_rabook_compile_init(&ctx, bufs);
608 if (err != k_ra8_ok) {
609 return err;
610 }
611 /* Emit in the desktop epub_compile.py order so the blob is byte-identical:
612 * stylesheets, images (cover resolved within), chapters, metadata interned LAST. */
613 err = s_compile_stylesheets(&ctx, scr, epub);
614 if (err != k_ra8_ok) {
615 return err;
616 }
617 uint32_t cover_image_index = (uint32_t)k_ra8_book_nil;
618 err = s_compile_images(&ctx, scr, epub, &cover_image_index);
619 if (err != k_ra8_ok) {
620 return err;
621 }
622 err = s_compile_chapters(epub, scr, &ctx);
623 if (err != k_ra8_ok) {
624 return err;
625 }
626 err = s_compile_metadata(&ctx, epub, cover_image_index);
627 if (err != k_ra8_ok) {
628 return err;
629 }
630 return ra8_rabook_finalize(&ctx, out_blob, out_len);
631}
