132 break;
133 }
134 g_dac_b_demo_tick += 1U;
135 if (up) {
136 if ((uint16_t)(code + (uint16_t)k_dac_b_demo_step) >= (uint16_t)k_dac_b_demo_full_scale) {
137 code = (uint16_t)k_dac_b_demo_full_scale;
138 up = false;
139 } else {
140 code = (uint16_t)(code + (uint16_t)k_dac_b_demo_step);
141 }
142 } else {
143 if (code <= (uint16_t)k_dac_b_demo_step) {
144 code = 0U;
145 up = true;
146 } else {
147 code = (uint16_t)(code - (uint16_t)k_dac_b_demo_step);
148 }
149 }
150 ra8_delay_ms((uint32_t)k_dac_b_demo_step_ms);
151 }
152 dac_b_demo_panic_halt();
153 return 0;
154}
