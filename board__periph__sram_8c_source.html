95static void sram_shadow_store(uint64_t off, unsigned size, uint64_t value)
96{
97 for (unsigned i = 0U; i < size; ++i) {
98 const uint64_t at = off + (uint64_t)i;
99 if (at < (uint64_t)k_sram_span) {
100 s_sram.shadow[at] = (uint8_t)(value >> (k_sram_byte_bits * i));
101 }
102 }
103}

106static uint64_t sram_shadow_load(uint64_t off, unsigned size)
107{
108 uint64_t v = 0U;
109 for (unsigned i = 0U; i < size; ++i) {
110 const uint64_t at = off + (uint64_t)i;
111 if (at < (uint64_t)k_sram_span) {
112 v |= (uint64_t)s_sram.shadow[at] << (k_sram_byte_bits * i);
113 }
114 }
115 return v;

119static void sram_latch_bank(uint32_t bank)
120{
121 s_sram.esr = (uint16_t)(s_sram.esr | sram_esr_bit(bank, 0U) | sram_esr_bit(bank, 1U));
122 const uint64_t ear0 = k_sram_off_ear0 + ((uint64_t)bank * (uint64_t)k_sram_ear_bank_str);
123 sram_shadow_store(ear0, sizeof(uint32_t), (uint64_t)k_sram_bank_data_off[bank]);
124 sram_shadow_store(ear0 + (uint64_t)k_sram_ear_slot_str,
125 sizeof(uint32_t),
126 (uint64_t)k_sram_bank_data_off[bank]);
127 s_sram.latch_count++;
128}

131static void sram_cr_write(uint32_t bank, uint8_t cr)
132{
133 bool was_bypass = (s_sram.last_cr[bank] == (uint8_t)k_sram_cr_bypass);
134 bool is_verify = (cr == (uint8_t)k_sram_cr_verify);
135 if (was_bypass) {
136 if (is_verify) {
137 sram_latch_bank(bank);
138 }
139 }
140 s_sram.last_cr[bank] = cr;
141}

144static void sram_clear(uint16_t mask)
145{
146 s_sram.esr = (uint16_t)(s_sram.esr & (uint16_t)~mask);
147 for (uint32_t bank = 0U; bank < k_sram_bank_count; ++bank) {
148 for (uint32_t slot = 0U; slot < k_sram_slots_bank; ++slot) {
149 if ((mask & sram_esr_bit(bank, slot)) != 0U) {
150 const uint64_t ear = k_sram_off_ear0 + ((uint64_t)bank * (uint64_t)k_sram_ear_bank_str) +
151 ((uint64_t)slot * (uint64_t)k_sram_ear_slot_str);
152 sram_shadow_store(ear, sizeof(uint32_t), 0U);
153 }
154 }
155 }

170static void sram_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
171{
172 (void)uc;
173 const uint64_t off = addr - (uint64_t)k_sram_base;
174 if (off == (uint64_t)k_sram_off_esclr) {
175 sram_clear((uint16_t)value);
176 return;
177 }
178 if (off == (uint64_t)k_sram_off_esr) {
179 s_sram.esr = (uint16_t)value;
180 return;
181 }
182 if ((off >= (uint64_t)k_sram_off_cr0) && (off <= (uint64_t)k_sram_off_cr3)) {
183 const uint64_t rel = off - (uint64_t)k_sram_off_cr0;
184 if ((rel % sizeof(uint32_t)) == 0U) {
185 sram_cr_write((uint32_t)(rel / sizeof(uint32_t)),
186 (uint8_t)(value & (uint64_t)k_sram_cr_mask));
187 }
188 }
189 sram_shadow_store(off, size, value);
190}
