215static ra8_err_t lin_hil_send_frame(uint8_t id)
216{
217 const ra8_err_t herr = ra8_sci_lin_send_header((uint8_t)k_lin_hil_channel, id);
218 if (herr != k_ra8_ok) {
219 return herr;
220 }
221 const uint8_t pid = ra8_sci_lin_pid(id);
222 return ra8_sci_lin_send_response((uint8_t)k_lin_hil_channel,
223 k_ra8_sci_lin_checksum_enhanced,
224 pid,
225 k_lin_hil_payload,
226 (uint8_t)k_lin_hil_data_len);

242int32_t main(void)
243{
244 lin_hil_setup_or_halt();
245 ra8_isr_globals_enable();
246 (void)ra8_board_uart_console_write(k_lin_hil_banner, (size_t)(sizeof(k_lin_hil_banner) - 1U));
247
248 uint8_t idx = 0U;
249 while (1) {
250 const uint8_t id = k_lin_hil_ids[idx];
251 const ra8_err_t err = lin_hil_send_frame(id);
252 if (err == k_ra8_ok) {
253 (void)ra8_board_uart_console_write(k_lin_hil_frame_msg,
254 (size_t)(sizeof(k_lin_hil_frame_msg) - 1U));
255 } else {
256 (void)ra8_board_uart_console_write(k_lin_hil_fault_msg,
257 (size_t)(sizeof(k_lin_hil_fault_msg) - 1U));
258 }
259
260 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
261 break;
262 }
263 idx = (uint8_t)((idx + 1U) % (uint8_t)k_lin_hil_id_count);
264 ra8_delay_ms((uint32_t)k_lin_hil_period_ms);
265 }
266
267 lin_hil_panic_halt();
268 return 0;
269}
