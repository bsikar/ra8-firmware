58static ra8_err_t
59s_table_check(const ra8_book_chunked_t* rd, const uint64_t* table_buf, uint64_t payload_len)
60{
61 if (table_buf[0] != 0U) {
62 return k_ra8_err_invalid_arg;
63 }
64 uint64_t max_clen = 0U;
65 for (uint32_t i = 0U; i < rd->chunk_count; ++i) { /* bound: validated chunk_count */
66 if (table_buf[i + 1U] <= table_buf[i]) {
67 return k_ra8_err_invalid_arg;
68 }
69 const uint64_t clen = table_buf[i + 1U] - table_buf[i];
70 if (clen > max_clen) {
71 max_clen = clen;
72 }
73 }
74 if (table_buf[rd->chunk_count] != payload_len) {
75 return k_ra8_err_invalid_arg;
76 }
77 if (max_clen > (uint64_t)rd->staging_cap) {
78 return k_ra8_err_invalid_size;
79 }
80 return k_ra8_ok;
81}

117static ra8_err_t s_load_table(ra8_book_chunked_t* rd,
118 uint64_t file_len,
119 uint64_t* table_buf,
120 uint32_t table_cap_entries)
121{
122 const uint64_t entries = (uint64_t)rd->chunk_count + 1U;
123 const uint64_t table_bytes = entries * k_ra8_book_container_entry_len;
124 if (entries > (uint64_t)table_cap_entries) {
125 return k_ra8_err_invalid_size;
126 }
127 if (table_bytes > (uint64_t)UINT32_MAX) {
128 return k_ra8_err_invalid_size;
129 }
130 if (file_len < ((uint64_t)k_ra8_book_container_header_len + table_bytes)) {
131 return k_ra8_err_invalid_size;
132 }
133 const ra8_err_t err = rd->file_read(rd->file_ctx,
134 (uint64_t)k_ra8_book_container_header_len,
135 (uint8_t*)table_buf,
136 (uint32_t)table_bytes);
137 if (err != k_ra8_ok) {
138 return err;
139 }
140 const uint64_t payload_off = (uint64_t)k_ra8_book_container_header_len + table_bytes;
141 const ra8_err_t shape = s_table_check(rd, table_buf, file_len - payload_off);
142 if (shape != k_ra8_ok) {
143 return shape;
144 }
145 rd->table = table_buf;
146 rd->payload_off = payload_off;
147 return k_ra8_ok;
148}

258s_stage_and_inflate(const ra8_book_chunked_t* rd, uint32_t idx, uint8_t* buf, uint32_t len)
259{
260 const uint64_t clen = rd->table[idx + 1U] - rd->table[idx];
261 ra8_err_t err =
262 rd->file_read(rd->file_ctx, rd->payload_off + rd->table[idx], rd->staging, (uint32_t)clen);
263 if (err != k_ra8_ok) {
264 return err;
265 }
266 size_t produced = 0U;

269 return err;
270 }
271 if (produced != (size_t)len) {

273 }
274 return k_ra8_ok;

277ra8_err_t ra8_book_chunked_read(void* ctx, uint64_t offset, uint8_t* buf, uint32_t len)
278{
279 RA8_CHECK_NULL_PTR(ctx, s_tag_chunked, "read: null ctx");
280 RA8_CHECK_NULL_PTR(buf, s_tag_chunked, "read: null buf");
281 const ra8_book_chunked_t* rd = (const ra8_book_chunked_t*)ctx;
282 if (rd->table == nullptr) {
283 return k_ra8_err_invalid_state;
284 }
285 if (offset >= rd->inflated_total) {
286 return k_ra8_err_out_of_range;
287 }
288 if ((offset % (uint64_t)rd->chunk_bytes) != 0U) {
289 return k_ra8_err_invalid_arg;
290 }
291 uint64_t expected = rd->inflated_total - offset;
292 if (expected > (uint64_t)rd->chunk_bytes) {
293 expected = rd->chunk_bytes;
294 }
295 if ((uint64_t)len != expected) {
296 return k_ra8_err_invalid_arg;
297 }
298 return s_stage_and_inflate(rd, (uint32_t)(offset / (uint64_t)rd->chunk_bytes), buf, len);
299}
