138RA8_INTERNAL static uint32_t internal_lcd_bringup_clocks(void)
139{
140 uint32_t cpuclk0_hz = 0U;
141 if (ra8_cgc_init() != k_ra8_ok) {
142 internal_lcd_panic_halt();
143 }
144 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
145 internal_lcd_panic_halt();
146 }
147 if (ra8_mstp_init() != k_ra8_ok) {
148 internal_lcd_panic_halt();
149 }
150 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
151 internal_lcd_panic_halt();
152 }
153 if (ra8_board_led_init(k_ra8_board_led_blue) != k_ra8_ok) {
154 internal_lcd_panic_halt();
155 }
156 if (ra8_board_led_init(k_ra8_board_led_red) != k_ra8_ok) {
157 internal_lcd_panic_halt();
158 }
159 ra8_isr_globals_enable();
160 return cpuclk0_hz;
161}

183RA8_INTERNAL static void internal_lcd_bringup_panel(void)
184{
185 /* Stabilization delays: PLLs, SDRAM, and the panel itself all need
186 * a few hundred ms after power-on to settle. Without these, the
187 * GLCDC sometimes starts before LCDCLK is stable and the panel
188 * comes up in its no-signal "white" state on cold boot. */
189 ra8_delay_ms(k_lcd_powerup_ms);
190
191 /* SDRAM is initialized so the framebuffer region at 0x68000000 is
192 * accessible for follow-on apps; this demo doesn't use it. */
193 if (ra8_sdramc_init() != k_ra8_ok) {
194 internal_lcd_panic_halt();
195 }
196 ra8_delay_ms(k_lcd_sdram_settle_ms);
197
198 if (ra8_board_lcd_panel_power_on() != k_ra8_ok) {
199 internal_lcd_panic_halt();
200 }
201 if (ra8_board_glcdc_init(k_ra8_board_glcdc_fmt_rgb888) != k_ra8_ok) {
202 internal_lcd_panic_halt();
203 }
204 ra8_delay_ms(k_lcd_pin_settle_ms); /* let pins settle in output mode */
205
206 /* GLCDC: BG plane drives the panel on its own with both graphics
207 * layers held invisible by the driver, so the framebuffer pointer
208 * is never dereferenced -- leave it null. */
209 const ra8_glcdc_config_t cfg = {
210 .framebuffer_addr = 0UL,
211 .width_px = (uint16_t)k_lcd_panel_w,
212 .height_px = (uint16_t)k_lcd_panel_h,
213 .format = k_ra8_glcdc_fmt_rgb565,
214 .timing = s_ra8_panel_ek_ra8d2_timing,
215 };
216 if (ra8_glcdc_init(&cfg) != k_ra8_ok) {
217 internal_lcd_panic_halt();
218 }
219 if (ra8_glcdc_set_background_color(k_bgc_red) != k_ra8_ok) {
220 internal_lcd_panic_halt();
221 }
222 if (ra8_glcdc_start(true) != k_ra8_ok) {
223 internal_lcd_panic_halt();
224 }
225}
