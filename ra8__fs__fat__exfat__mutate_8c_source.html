25/* `priv_exfat_bitmap_clear()`: see header for the documented contract. */
26ra8_err_t
27priv_exfat_bitmap_clear(const ra8_fs_mount_t* m, uint64_t bmp_lba, uint32_t clus, uint32_t count)
28{
29 uint64_t loaded = UINT64_MAX;
30 uint8_t* const sec = priv_sec_io();
31 for (uint32_t k = 0U; k < count; k++) {
32 const uint32_t idx = (clus - k_cluster_first_data) + k;
33 const uint64_t lba = bmp_lba + ((idx >> k_exfat_bit_shift) / priv_bps(m));
34 const uint32_t byte = (idx >> k_exfat_bit_shift) % priv_bps(m);
35 const uint32_t bit = idx & k_exfat_bit_mask;
36 ra8_err_t e = priv_exfat_bmp_switch(m, lba, &loaded, sec);
37 if (e != k_ra8_ok) {
38 return e; /* GCOVR_EXCL_LINE */
39 }
40 sec[byte] = (uint8_t)(sec[byte] & (uint8_t)~(uint8_t)(1U << bit));
41 /* Pull the scan hint back to the space just released, or the next create
42 * would step over it and only find it after a full rescan (#607). */
43 priv_alloc_hint_lower(m, idx + (uint32_t)k_cluster_first_data);
44 }
45 return priv_write_sector(m, loaded, sec);
46}

76RA8_INTERNAL
77static ra8_err_t priv_exfat_take_set(const ra8_fs_mount_t* m,
78 exfat_cursor_t* cur,
79 const uint16_t* name,
80 uint32_t nlen,
81 uint32_t sc,
82 exfat_setpos_t* pos,
83 uint8_t* strm_copy,
84 uint8_t* out_match)
85{
86 uint8_t matched = 1U;
87 for (uint32_t k = 0U; k < sc; k++) {
88 uint8_t se[k_exfat_entry_bytes] = {};
89 const ra8_err_t r = priv_exfat_next_entry(m, cur, se);
90 if (r != k_ra8_ok) {
91 return r; /* GCOVR_EXCL_LINE */
92 }
93 /* AFTER the read: ::priv_exfat_next_entry may cross a cluster boundary before
94 * reading, so the entry lives at the cursor's CURRENT cluster, one slot back
95 * (it has already advanced past what it read). Capturing before the call
96 * records (old cluster, one-past-end) for the first entry of a new cluster --
97 * harmless while directories were one cluster, a mis-retired slot once they
98 * grow (#677). */
99 const exfat_setpos_t sp = {.cluster = cur->cluster, .index = cur->entry_in_cluster - 1U};
100 pos[1U + k] = sp;
101 if (k == 0U) {
102 if (se[0] != (uint8_t)k_exfat_entry_stream) {
103 matched = 0U;
104 continue;
105 }
106 if ((uint32_t)se[k_exfat_strm_off_nlen] != nlen) {
107 matched = 0U;
108 continue;
109 }
110 priv_byte_copy(strm_copy, se, (uint32_t)k_exfat_entry_bytes);
111 continue;
112 }
113 if (matched == 0U) {
114 continue;
115 }
116 if (se[0] != (uint8_t)k_exfat_entry_name) {
117 matched = 0U;
118 continue;
119 }
120 const uint32_t cpos = (k - 1U) * (uint32_t)k_exfat_name_per_entry;
121 if (priv_exfat_name_chunk_eq(se, name, cpos, nlen) == 0U) {
122 matched = 0U;
123 }
124 }
125 *out_match = matched;
126 return k_ra8_ok;
127}

161static ra8_err_t priv_exfat_try_set(const ra8_fs_mount_t* m,
162 exfat_cursor_t* cur,
163 exfat_setpos_t at,
164 const uint8_t* e,
165 const uint16_t* need,
166 uint32_t nlen,
167 exfat_setpos_t* pos,
168 uint32_t max_pos,
169 uint32_t* out_count,
170 uint8_t* file_copy,
171 uint8_t* strm_copy)
172{
173 const uint32_t sc = (uint32_t)e[k_exfat_off_file_secnt];
174 const uint32_t total = 1U + sc;
175 if (total > max_pos) {
176 return k_ra8_err_no_mem;
177 }
178 pos[0] = at;
179 priv_byte_copy(file_copy, e, (uint32_t)k_exfat_entry_bytes);
180 uint8_t matched = 0U;
181 const ra8_err_t r = priv_exfat_take_set(m, cur, need, nlen, sc, pos, strm_copy, &matched);
182 if (r != k_ra8_ok) {
183 return r; /* GCOVR_EXCL_LINE */
184 }
185 if (matched == 1U) {
186 *out_count = total;
187 return k_ra8_ok;
188 }
189 return k_ra8_err_not_found;
190}

193ra8_err_t priv_exfat_find_set(const ra8_fs_mount_t* m,
194 const exfat_dir_t* dir,
195 const char* path,
196 exfat_setpos_t* pos,
197 uint32_t max_pos,
198 uint32_t* out_count,
199 uint8_t* file_copy,
200 uint8_t* strm_copy)
201{
202 /* Strip leading slashes so a "/name" path matches (#93), as FAT does. */
203 while (*path == '/') {
204 path++;
205 }
206 uint16_t need[k_exfat_name_cap] = {};
207 uint32_t nlen = 0U;
208 const ra8_err_t ne = priv_exfat_needle_units(m, path, need, &nlen, k_ra8_err_not_found);
209 if (ne != k_ra8_ok) {
210 return ne;
211 }
212 exfat_cursor_t cur = {};
213 priv_exfat_cursor_init(dir, &cur);
214 while (cur.scanned < (uint32_t)k_exfat_scan_limit) {
215 uint8_t e[k_exfat_entry_bytes] = {};
216 ra8_err_t r = priv_exfat_next_entry(m, &cur, e);
217 if (r != k_ra8_ok) {
218 return r; /* GCOVR_EXCL_LINE */
219 }
220 /* AFTER the read (see ::priv_exfat_take_set): the File entry lives at the
221 * cursor's current cluster, one slot back, so a set whose File entry is the
222 * first of a grown cluster is recorded where it truly is (#677). */
223 const exfat_setpos_t at = {.cluster = cur.cluster, .index = cur.entry_in_cluster - 1U};
224 if (e[0] == (uint8_t)k_exfat_entry_eod) {
225 return k_ra8_err_not_found;
226 }
227 if (e[0] != (uint8_t)k_exfat_entry_file) {
228 continue;
229 }
230 r =
231 priv_exfat_try_set(m, &cur, at, e, need, nlen, pos, max_pos, out_count, file_copy, strm_copy);
232 if (r != k_ra8_err_not_found) {
233 return r;
234 }
235 }
236 return k_ra8_err_not_found; /* GCOVR_EXCL_LINE */
237}

269ra8_err_t priv_exfat_free_clusters(const ra8_fs_mount_t* m, const uint8_t* strm)
270{
271 const uint32_t first = priv_rd32(&strm[k_exfat_strm_off_clus]);
272 /* The full 64-bit DataLength: reading only the low word here under-counted a
273 * >4 GiB file's clusters and leaked the rest on unlink/truncate (#676). */
274 const uint64_t size = priv_rd64(&strm[k_exfat_strm_off_dlen]);
275 if (first < k_cluster_first_data) {
276 return k_ra8_ok;
277 }
278 if (size == 0U) {
279 return k_ra8_ok;
280 }
281 uint64_t bmp_lba = 0U;
282 ra8_err_t e = priv_exfat_bitmap_lba(m, &bmp_lba);
283 if (e != k_ra8_ok) {
284 return e; /* GCOVR_EXCL_LINE */
285 }
286 const uint32_t cluster_bytes = priv_cluster_bytes(m);
287 const uint8_t nofat =
288 ((strm[k_exfat_strm_off_flags] & (uint8_t)k_exfat_secflag_no_fat) != 0U) ? 1U : 0U;
289 if (nofat == 1U) {
290 const uint32_t count = (uint32_t)((size + cluster_bytes - 1U) / cluster_bytes);
291 return priv_exfat_bitmap_clear(m, bmp_lba, first, count);
292 }
293 uint32_t clus = first;
294 for (uint32_t guard = 0U; guard < m->count_of_clusters; guard++) {
295 e = priv_exfat_bitmap_clear(m, bmp_lba, clus, 1U);
296 if (e != k_ra8_ok) {
297 return e; /* GCOVR_EXCL_LINE */
298 }
299 uint32_t next = 0U;
300 e = priv_fat_get(m, clus, &next);
301 if (e != k_ra8_ok) {
302 return e; /* GCOVR_EXCL_LINE */
303 }
304 if (priv_is_eoc(m, next) != 0U) {
305 return k_ra8_ok;
306 }
307 clus = next;
308 }
309 return k_ra8_ok; /* GCOVR_EXCL_LINE */
310}

312/* `priv_exfat_drop_set()`: see header for the documented contract. */
313ra8_err_t priv_exfat_drop_set(const ra8_fs_mount_t* m, const exfat_setpos_t* pos, uint32_t count)
314{
315 for (uint32_t k = 0U; k < count; k++) {
316 /* A single-entry read at a recorded position: contig_end is irrelevant
317 * because this cursor never advances past the entry it was aimed at. */
318 exfat_cursor_t one = {.cluster = pos[k].cluster,
319 .entry_in_cluster = pos[k].index,
320 .scanned = 0U,
321 .contig_end = 0U};
322 uint8_t entry[k_exfat_entry_bytes] = {};
323 ra8_err_t e = priv_exfat_next_entry(m, &one, entry);
324 if (e != k_ra8_ok) {
325 return e; /* GCOVR_EXCL_LINE */
326 }
327 entry[0] = (uint8_t)(entry[0] & (uint8_t)~(uint8_t)k_exfat_inuse_bit);
328 e = priv_exfat_put_entry(m, &pos[k], entry);
329 if (e != k_ra8_ok) {
330 return e; /* GCOVR_EXCL_LINE */
331 }
332 }
333 return k_ra8_ok;
334}

336/* `priv_exfat_unlink_at()`: see header for the documented contract. */
337ra8_err_t priv_exfat_unlink_at(const ra8_fs_mount_t* m, const exfat_dir_t* dir, const char* name)
338{
339 exfat_setpos_t pos[k_exfat_set_max_entries] = {};
340 uint32_t count = 0U;
341 uint8_t file_e[k_exfat_entry_bytes] = {};
342 uint8_t strm_e[k_exfat_entry_bytes] = {};
343 const ra8_err_t e = priv_exfat_find_set(m,
344 dir,
345 name,
346 pos,
347 (uint32_t)k_exfat_set_max_entries,
348 &count,
349 file_e,
350 strm_e);
351 if (e != k_ra8_ok) {
352 return e;
353 }
354 /* A directory's entry set looks like a file's, so without this the chain
355 * holding every child is handed to priv_exfat_free_clusters and the children
356 * become unreachable allocated clusters -- silent, unrecoverable loss (#604). */
357 if ((file_e[k_exfat_off_file_attr] & (uint8_t)k_exfat_attr_directory) != 0U) {
358 return k_ra8_err_invalid_arg;
359 }
360 /* Honor the read-only attribute (#681): a read-only file is refused before the
361 * entry set is dropped or its clusters freed, so a denied unlink is a no-op. */
362 if ((file_e[k_exfat_off_file_attr] & (uint8_t)k_exfat_attr_read_only) != 0U) {
363 return k_ra8_err_access_denied;
364 }
365 const ra8_err_t de = priv_exfat_drop_set(m, pos, count);
366 if (de != k_ra8_ok) {
367 return de; /* GCOVR_EXCL_LINE */
368 }
369 return priv_exfat_free_clusters(m, strm_e);
370}

417RA8_INTERNAL
418static uint32_t priv_exfat_build_rename_set(uint8_t* set,
419 const uint8_t* file_e,
420 const uint8_t* strm_e,
421 const uint16_t* new_name,
422 uint32_t new_len)
423{
424 const uint32_t name_entries =
425 (new_len + (uint32_t)k_exfat_name_per_entry - 1U) / (uint32_t)k_exfat_name_per_entry;
426 const uint32_t sec_count = 1U + name_entries;
427 const uint32_t total = (1U + sec_count) * (uint32_t)k_exfat_entry_bytes;
428 for (uint32_t i = 0U; i < total; i++) {
429 set[i] = 0U;
430 }
431 priv_byte_copy(&set[0], file_e, (uint32_t)k_exfat_entry_bytes);
432 priv_byte_copy(&set[k_exfat_entry_bytes], strm_e, (uint32_t)k_exfat_entry_bytes);
433 set[k_exfat_off_file_secnt] = (uint8_t)sec_count;
434 uint8_t* strm = &set[k_exfat_entry_bytes];
435 strm[k_exfat_strm_off_nlen] = (uint8_t)new_len;
436 priv_wr16(&strm[k_exfat_off_strm_hash], priv_exfat_name_hash(new_name, new_len));
437 for (uint32_t n = 0U; n < name_entries; n++) {
438 uint8_t* ne = &set[(size_t)(2U + n) * (size_t)k_exfat_entry_bytes];
439 ne[0] = (uint8_t)k_exfat_entry_name;
440 for (uint32_t c = 0U; c < (uint32_t)k_exfat_name_per_entry; c++) {
441 const uint32_t p = (n * (uint32_t)k_exfat_name_per_entry) + c;
442 if (p < new_len) {
443 priv_wr16(&ne[k_exfat_name_off + (c * 2U)], new_name[p]);
444 }
445 }
446 }
447 /* Access stamp only, and before the checksum that covers it. Same reasoning
448 * as the FAT rename: the name moved, the bytes did not, so LastModified must
449 * not move or every backup tool concludes the file changed (#601). */
450 priv_exfat_file_stamp_access(set);
451 priv_wr16(&set[k_exfat_off_file_csum], priv_exfat_set_checksum(set, total));
452 return total;
453}

550RA8_INTERNAL
551static ra8_err_t priv_exfat_place_rename(const ra8_fs_mount_t* m,
552 const exfat_dir_t* parent,
553 const exfat_setpos_t* pos,
554 uint32_t old_count,
555 const uint8_t* set,
556 uint32_t bytes)
557{
558 const uint32_t new_count = bytes / (uint32_t)k_exfat_entry_bytes;
559 if (new_count == old_count) {
560 for (uint32_t k = 0U; k < new_count; k++) {
561 const ra8_err_t e =
562 priv_exfat_put_entry(m, &pos[k], &set[(size_t)k * (size_t)k_exfat_entry_bytes]);
563 if (e != k_ra8_ok) {
564 return e; /* GCOVR_EXCL_LINE */
565 }
566 }
567 return k_ra8_ok;
568 }
569 uint32_t nclus = 0U;
570 uint32_t nidx = 0U;
571 const ra8_err_t se = priv_exfat_find_dir_space(m, parent, new_count, &nclus, &nidx);
572 if (se != k_ra8_ok) {
573 return se;
574 }
575 const ra8_err_t we = priv_exfat_write_dir_set(m, nclus, nidx, set, bytes);
576 if (we != k_ra8_ok) {
577 return we; /* GCOVR_EXCL_LINE */
578 }
579 return priv_exfat_drop_set(m, pos, old_count);

582/* `priv_exfat_rename()`: see header for the documented contract. */
583ra8_err_t priv_exfat_rename(const ra8_fs_mount_t* m, const char* old_path, const char* new_path)
584{
585 exfat_dir_t parent = {};
586 const char* old_name = nullptr;
587 const char* new_name = nullptr;
588 const ra8_err_t pe =
589 priv_exfat_rename_prepare(m, old_path, new_path, &parent, &old_name, &new_name);
590 if (pe != k_ra8_ok) {
591 return pe;
592 }
593 /* The new leaf in UTF-16 up front: the built Name entries and the NameHash
594 * both count CODE UNITS, and a UTF-8 argument of the same unit count can be
595 * three times as many bytes (#606). A name over the cap is an argument fault,
596 * not a full disk. The old leaf stays UTF-8: ::priv_exfat_find_set converts it
597 * itself, and the entry count it reports is all the placement below needs. */
598 uint16_t new_units[k_exfat_name_cap] = {};
599 uint32_t new_len = 0U;
600 const ra8_err_t nue =
601 priv_exfat_needle_units(m, new_name, new_units, &new_len, k_ra8_err_invalid_arg);
602 if (nue != k_ra8_ok) {
603 return nue;
604 }
605 uint8_t e_strm[k_exfat_entry_bytes] = {};
606 uint8_t e_attr = 0U;
607 if (priv_exfat_find(m, &parent, new_name, e_strm, &e_attr) == k_ra8_ok) {
608 return k_ra8_err_exists;
609 }
610 exfat_setpos_t pos[k_exfat_set_max_entries] = {};
611 uint32_t count = 0U;
612 uint8_t file_e[k_exfat_entry_bytes] = {};
613 uint8_t strm_e[k_exfat_entry_bytes] = {};
614 const ra8_err_t fe = priv_exfat_find_set(m,
615 &parent,
616 old_name,
617 pos,
618 (uint32_t)k_exfat_set_max_entries,
619 &count,
620 file_e,
621 strm_e);
622 if (fe != k_ra8_ok) {
623 return fe;
624 }
625 /* Honor the read-only attribute (#681). file_e is the File entry read back by
626 * priv_exfat_find_set above, so its FileAttributes low byte carries the
627 * read-only bit; a read-only file is refused before the rename set is rebuilt
628 * or placed. */
629 if ((file_e[k_exfat_off_file_attr] & (uint8_t)k_exfat_attr_read_only) != 0U) {
630 return k_ra8_err_access_denied;
631 }
632 uint8_t set[k_exfat_max_set_bytes] = {};
633 const uint32_t bytes = priv_exfat_build_rename_set(set, file_e, strm_e, new_units, new_len);
634 return priv_exfat_place_rename(m, &parent, pos, count, set, bytes);
635}

674RA8_INTERNAL
675static ra8_err_t priv_exfat_gather_name(const ra8_fs_mount_t* m,
676 exfat_cursor_t* cur,
677 uint32_t sc,
678 uint32_t nlen,
679 char* name,
680 uint32_t cap)
681{
682 uint16_t units[k_exfat_name_cap] = {};
683 uint32_t got = 0U;
684 name[0] = '\0';
685 for (uint32_t k = 1U; k < sc; k++) {
686 uint8_t ne[k_exfat_entry_bytes] = {};
687 const ra8_err_t r = priv_exfat_next_entry(m, cur, ne);
688 if (r != k_ra8_ok) {
689 return r; /* GCOVR_EXCL_LINE */
690 }
691 if (ne[0] != (uint8_t)k_exfat_entry_name) {
692 continue;
693 }
694 for (uint32_t c = 0U; c < (uint32_t)k_exfat_name_per_entry; c++) {
695 const uint32_t p = ((k - 1U) * (uint32_t)k_exfat_name_per_entry) + c;
696 if (p >= nlen) {
697 break;
698 }
699 if (got < (uint32_t)k_exfat_name_cap) {
700 units[got] = priv_rd16(&ne[k_exfat_name_off + (c * 2U)]);
701 got++;
702 }
703 }
704 }
705 if (priv_utf16_to_utf8(units, got, name, cap) != k_ra8_ok) {
706 name[0] = '\0';
707 }
708 return k_ra8_ok;
709}

735RA8_INTERNAL
736static ra8_err_t priv_exfat_list_emit(const ra8_fs_mount_t* m,
737 exfat_cursor_t* cur,
738 const uint8_t* e,
739 ra8_fs_listdir_cb_t cb,
740 void* ctx)
741{
742 const uint32_t sc = (uint32_t)e[k_exfat_off_file_secnt];
743 const uint8_t attr = e[k_exfat_off_file_attr];
744 uint8_t strm[k_exfat_entry_bytes] = {};
745 ra8_err_t r = priv_exfat_next_entry(m, cur, strm);
746 if (r != k_ra8_ok) {
747 return r; /* GCOVR_EXCL_LINE */
748 }
749 if (strm[0] != (uint8_t)k_exfat_entry_stream) {
750 return k_ra8_ok;
751 }
752 /* A directory's exFAT DataLength is its ALLOCATION, not a byte count, so
753 * reporting it as a size would tell every caller that an empty folder held
754 * a cluster's worth of bytes. FAT reports 0 for a directory and so does
755 * ra8_fs_stat(); this is the third place that has to agree (#605). */
756 uint64_t size = priv_rd64(&strm[k_exfat_strm_off_dlen]);
757 if ((attr & (uint8_t)k_exfat_attr_directory) != 0U) {
758 size = 0U;
759 }
760 const uint32_t nlen = (uint32_t)strm[k_exfat_strm_off_nlen];
761 char name[k_exfat_name_u8_cap] = {};
762 r = priv_exfat_gather_name(m, cur, sc, nlen, name, (uint32_t)k_exfat_name_u8_cap);
763 if (r != k_ra8_ok) {
764 return r; /* GCOVR_EXCL_LINE */
765 }
766 if (name[0] == '\0') {
767 return k_ra8_ok; /* units no UTF-8 string encodes: report nothing, not a lie */
768 }
769 cb(name, attr, size, ctx);
770 return k_ra8_ok;

773/* `priv_exfat_listdir()`: see header for the documented contract. */
774ra8_err_t priv_exfat_listdir(const ra8_fs_mount_t* m,
775 const exfat_dir_t* dir,
776 ra8_fs_listdir_cb_t cb,
777 void* ctx)
778{
779 exfat_cursor_t cur = {};
780 priv_exfat_cursor_init(dir, &cur);
781 while (cur.scanned < (uint32_t)k_exfat_scan_limit) {
782 uint8_t e[k_exfat_entry_bytes] = {};
783 ra8_err_t r = priv_exfat_next_entry(m, &cur, e);
784 if (r == k_ra8_err_not_found) {
785 return k_ra8_ok; /* the directory's run ended without a 0x00 marker */
786 }
787 if (r != k_ra8_ok) {
788 return r; /* GCOVR_EXCL_LINE */
789 }
790 if (e[0] == (uint8_t)k_exfat_entry_eod) {
791 return k_ra8_ok;
792 }
793 if (e[0] != (uint8_t)k_exfat_entry_file) {
794 continue;
795 }
796 r = priv_exfat_list_emit(m, &cur, e, cb, ctx);
797 if (r != k_ra8_ok) {
798 return r; /* GCOVR_EXCL_LINE */
799 }
800 }
801 return k_ra8_ok; /* GCOVR_EXCL_LINE */
802}
