67static ra8_err_t
68priv_exfat_bitmap_clear(const ra8_fs_mount_t* m, uint32_t bmp_lba, uint32_t clus, uint32_t count)
69{
70 uint32_t loaded = UINT32_MAX;
71 uint8_t sec[k_ra8_fs_bytes_per_sector] = {};
72 for (uint32_t k = 0U; k < count; k++) {
73 const uint32_t idx = (clus - k_cluster_first_data) + k;
74 const uint32_t lba = bmp_lba + ((idx >> k_exfat_bit_shift) / k_ra8_fs_bytes_per_sector);
75 const uint32_t byte = (idx >> k_exfat_bit_shift) % k_ra8_fs_bytes_per_sector;
76 const uint32_t bit = idx & k_exfat_bit_mask;
77 ra8_err_t e = priv_exfat_bmp_switch(m, lba, &loaded, sec);
78 if (e != k_ra8_ok) {
79 return e; /* GCOVR_EXCL_LINE */
80 }
81 sec[byte] = (uint8_t)(sec[byte] & (uint8_t)~(uint8_t)(1U << bit));
82 }
83 return priv_write_sector(m, loaded, sec);
84}

187RA8_INTERNAL
188static ra8_err_t priv_exfat_find_set(const ra8_fs_mount_t* m,
189 const char* path,
190 exfat_setpos_t* pos,
191 uint32_t max_pos,
192 uint32_t* out_count,
193 uint8_t* file_copy,
194 uint8_t* strm_copy)
195{
196 /* Strip leading slashes so a "/name" path matches (#93), as FAT does. */
197 while (*path == '/') {
198 path++;
199 }
200 exfat_cursor_t cur = {.cluster = m->root_cluster, .entry_in_cluster = 0U, .scanned = 0U};
201 const uint32_t nlen = priv_strlen(path);
202 while (cur.scanned < (uint32_t)k_exfat_scan_limit) {
203 const exfat_setpos_t at = {.cluster = cur.cluster, .index = cur.entry_in_cluster};
204 uint8_t e[k_exfat_entry_bytes] = {};
205 ra8_err_t r = priv_exfat_next_entry(m, &cur, e);
206 if (r != k_ra8_ok) {
207 return r; /* GCOVR_EXCL_LINE */
208 }
209 if (e[0] == (uint8_t)k_exfat_entry_eod) {
210 return k_ra8_err_not_found;
211 }
212 if (e[0] != (uint8_t)k_exfat_entry_file) {
213 continue;
214 }
215 const uint32_t sc = (uint32_t)e[k_exfat_off_file_secnt];
216 const uint32_t total = 1U + sc;
217 if (total > max_pos) {
218 return k_ra8_err_no_mem;
219 }
220 pos[0] = at;
221 priv_byte_copy(file_copy, e, (uint32_t)k_exfat_entry_bytes);
222 uint8_t matched = 0U;
223 r = priv_exfat_take_set(m, &cur, path, nlen, sc, pos, strm_copy, &matched);
224 if (r != k_ra8_ok) {
225 return r; /* GCOVR_EXCL_LINE */
226 }
227 if (matched == 1U) {
228 *out_count = total;
229 return k_ra8_ok;
230 }
231 }
232 return k_ra8_err_not_found; /* GCOVR_EXCL_LINE */
233}

285static ra8_err_t priv_exfat_free_clusters(const ra8_fs_mount_t* m, const uint8_t* strm)
286{
287 const uint32_t first = priv_rd32(&strm[k_exfat_strm_off_clus]);
288 const uint32_t size = priv_rd32(&strm[k_exfat_strm_off_dlen]);
289 if (first < k_cluster_first_data) {
290 return k_ra8_ok;
291 }
292 if (size == 0U) {
293 return k_ra8_ok;
294 }
295 uint32_t bmp_clus = 0U;
296 uint32_t bmp_len = 0U;
297 ra8_err_t e = priv_exfat_find_bitmap(m, &bmp_clus, &bmp_len);
298 if (e != k_ra8_ok) {
299 return e; /* GCOVR_EXCL_LINE */
300 }
301 const uint32_t bmp_lba = priv_cluster_to_lba(m, bmp_clus);
302 const uint32_t cluster_bytes = m->sectors_per_cluster * k_ra8_fs_bytes_per_sector;
303 const uint8_t nofat =
304 ((strm[k_exfat_strm_off_flags] & (uint8_t)k_exfat_secflag_no_fat) != 0U) ? 1U : 0U;
305 if (nofat == 1U) {
306 const uint32_t count = (size + cluster_bytes - 1U) / cluster_bytes;
307 return priv_exfat_bitmap_clear(m, bmp_lba, first, count);
308 }
309 uint32_t clus = first;
310 for (uint32_t guard = 0U; guard < m->count_of_clusters; guard++) {
311 e = priv_exfat_bitmap_clear(m, bmp_lba, clus, 1U);
312 if (e != k_ra8_ok) {
313 return e; /* GCOVR_EXCL_LINE */
314 }
315 uint32_t next = 0U;
316 e = priv_fat_get(m, clus, &next);
317 if (e != k_ra8_ok) {
318 return e; /* GCOVR_EXCL_LINE */
319 }
320 if (priv_is_eoc(m, next) != 0U) {
321 return k_ra8_ok;
322 }
323 clus = next;
324 }
325 return k_ra8_ok; /* GCOVR_EXCL_LINE */
326}

330{
331 exfat_setpos_t pos[k_exfat_set_max_entries] = {};
332 uint32_t count = 0U;
333 uint8_t file_e[k_exfat_entry_bytes] = {};
334 uint8_t strm_e[k_exfat_entry_bytes] = {};
335 ra8_err_t e =
336 priv_exfat_find_set(m, path, pos, (uint32_t)k_exfat_set_max_entries, &count, file_e, strm_e);
337 if (e != k_ra8_ok) {
338 return e;
339 }
340 for (uint32_t k = 0U; k < count; k++) {
341 exfat_cursor_t one = {.cluster = pos[k].cluster,
342 .entry_in_cluster = pos[k].index,
343 .scanned = 0U};
344 uint8_t entry[k_exfat_entry_bytes] = {};
345 e = priv_exfat_next_entry(m, &one, entry);
346 if (e != k_ra8_ok) {
347 return e; /* GCOVR_EXCL_LINE */
348 }
349 entry[0] = (uint8_t)(entry[0] & (uint8_t)~(uint8_t)k_exfat_inuse_bit);
350 e = priv_exfat_put_entry(m, &pos[k], entry);
351 if (e != k_ra8_ok) {
352 return e; /* GCOVR_EXCL_LINE */
353 }
354 }
355 return priv_exfat_free_clusters(m, strm_e);

413/* `priv_exfat_rename()`: see header for the documented contract. */
414ra8_err_t priv_exfat_rename(const ra8_fs_mount_t* m, const char* old_path, const char* new_path)
415{
416 /* Strip leading slashes on both paths so the old name matches and the new
417 * name is stored without a slash (#93), consistent with create/find. */
418 while (*old_path == '/') {
419 old_path++;
420 }
421 while (*new_path == '/') {
422 new_path++;
423 }
424 const uint32_t old_len = priv_strlen(old_path);
425 const uint32_t new_len = priv_strlen(new_path);
426 if (old_len > (uint32_t)k_exfat_name_per_entry) {
427 return k_ra8_err_not_supported;
428 }
429 if (new_len > (uint32_t)k_exfat_name_per_entry) {
430 return k_ra8_err_not_supported;
431 }
432 uint32_t e_first = 0U;
433 uint32_t e_size = 0U;
434 uint8_t e_nofat = 0U;
435 if (priv_exfat_find(m, new_path, &e_first, &e_size, &e_nofat) == k_ra8_ok) {
436 return k_ra8_err_exists;
437 }
438 exfat_setpos_t pos[k_exfat_set_max_entries] = {};
439 uint32_t count = 0U;
440 uint8_t set[k_exfat_rename_bytes] = {};
441 const ra8_err_t e = priv_exfat_find_set(m,
442 old_path,
443 pos,
444 (uint32_t)k_exfat_set_max_entries,
445 &count,
446 &set[0],
447 &set[k_exfat_entry_bytes]);
448 if (e != k_ra8_ok) {
449 return e;
450 }
451 if (count != (uint32_t)k_exfat_rename_entries) {
452 return k_ra8_err_not_supported;
453 }
454 return priv_exfat_apply_rename(m, pos, set, new_path, new_len);
455}
