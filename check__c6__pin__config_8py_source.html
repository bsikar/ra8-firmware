4"""Gate: the C6 Kconfig defaults agree with the pin map they are derived from.
5
6Why two files exist
7-------------------
8``coprocessor/esp32c6/pins.env`` is the SINGLE SOURCE OF TRUTH for the
9RA8-host <-> ESP32-C6 SPI wiring. It is a plain ``KEY=value`` fragment because
10its consumers are shell (``build.sh``, ``flash.sh``) and, on the RA8 side,
11humans reading one file to learn the pinout.
12
13``coprocessor/esp32c6/sdkconfig.defaults`` carries the SAME numbers again in
14Kconfig syntax, because that is the only form esp-idf reads. It is a DERIVED
15artifact, and it is deliberately byte-stable: the bench-proven C6 image was
16built from exactly these lines, so it is verified against pins.env rather than
17regenerated from it.
18
19Two files holding one fact drift silently, and a drift here is expensive: the
20build succeeds, the firmware flashes, and the SPI link simply never comes up
21because the C6 is driving a different pin than the RA8 is. Nothing downstream
22of the mistake can detect it -- which is what makes this a gate and not a
23convention.
24
25What is compared
26----------------
27Every SPI signal (CS, COPI/CIPO, SCK, DATA_READY, HANDSHAKE, RESET), the chip
28target, the flash size, and the dev-board preset. The last two are the
29interesting ones: esp-idf and esp-hosted-mcu encode both in the KEY
30(``CONFIG_ESPTOOLPY_FLASHSIZE_16MB=y``, ``CONFIG_ESP_HOST_DEV_BOARD_NONE=y``)
31rather than in the value, so each is parsed out of the key name and compared
32against ``C6_FLASH_SIZE`` / ``C6_DEV_BOARD``.
33
34The dev-board preset is load-bearing and easy to miss. esp-hosted-mcu ships
35presets for Espressif's own dev boards, and selecting one OVERRIDES the SPI pin
36leaves -- so with any board but ``NONE`` selected, every pin this gate compares
37is silently ignored by the build and the C6 comes up on the preset's pins
38instead of ours. Comparing the enabled symbol's SUFFIX rather than merely
39asserting the symbol is present catches the drift in both of its shapes: the
40selection vanishing, and the selection moving to a different board.
41
42Two of the data signals carry Espressif's legacy names in their Kconfig symbols.
43Those symbols belong to upstream and cannot be renamed from here; the mapping
44table is where they are tied to this project's COPI/CIPO vocabulary.
45
46What is checked on the RA8 side
47-------------------------------
48pins.env also records where each signal LANDS on the EK-RA8D2 (MCU pin and
49J26 hole) and which SW4 DIP positions the link needs. That map IS restated in
50one other place -- ``port/esp-hosted/inc/ra8_esp_hosted_pins.h``, which the
51esp-hosted port compiles against -- so the two are diffed here. They have
52already drifted once: the port header was first written from the probe's
53candidate list while the module was disconnected, and the rebuilt harness was
54later characterised with HANDSHAKE and DATA_READY the other way round. A pin
55map that only one file knows is a pin map that will be wrong again.
56
57Independently of that diff, the map is checked for the failures that actually
58happen to a pin map: a signal named at one end and not the other, a pin or
59hole name that is not a pin or hole name, two signals silently claiming the
60same pin after a copy-paste, and a missing SW4 position. That bank is not
61incidental: SW4-4 ON with SW4-3 OFF holds the
62Pmod1 bus switches open, so J26-1..J26-4 never reach the MCU, and misreading
63it cost a full bench day chasing a harness that was fine.
64
65This is a pure text comparison of two committed files -- no esp-idf, no
66toolchain, no hardware -- so it runs in CI on any box. ``build.sh`` invokes
67this same script before it builds, so the bench and CI apply one rule.
68
69Non-vacuity
70-----------
71``--selftest`` drives the comparator with crafted file bodies: an agreeing
72pair must be silent, and a disagreeing pair, a missing key on either side, a
73missing flash-size key, and a dev-board preset that either moved to another
74board or stopped being selected must each be reported.
75
76Run::
77
78 check_c6_pin_config.py # gate (fail on any drift)
79 check_c6_pin_config.py --selftest # prove the comparator both ways
80
81Exit 0 when the two files agree, 1 on drift or a failing selftest, 2 when a
82file cannot be read.
83"""

201 """Return {enumerator: board symbol} for every row of the port pin map."""

205def check_port_header(pins: dict[str, str], header: str) -> list[str]:
206 """Return one message per disagreement with the port's pin map.
207
208 Args:
209 pins: Parsed pins.env assignments (the source of truth).
210 header: Text of ``ra8_esp_hosted_pins.h``.
211
212 Returns:
213 Human-readable findings; empty when the header agrees with pins.env.
214 """
215 rows = parse_port_header(header)
216 findings: list[str] = []
217 for label, enum_name, pin_key in PORT_PIN_ROWS:
218 if enum_name not in rows:
219 findings.append(f"{label}: the port pin header does not define {enum_name}")
220 continue
221 symbol = rows[enum_name]
222 if symbol not in BOARD_SYMBOL_TO_PIN:
223 findings.append(
224 f"{label}: the port pin header names {symbol}, which this gate cannot "
225 f"resolve to an MCU pin; add it to BOARD_SYMBOL_TO_PIN"
226 )
227 continue
228 if pin_key not in pins:
229 findings.append(f"{label}: pins.env is missing {pin_key}")
230 continue
231 resolved = BOARD_SYMBOL_TO_PIN[symbol]
232 if resolved != pins[pin_key]:
233 findings.append(
234 f"{label}: the port pin header says {symbol} ({resolved}) "
235 f"but pins.env says {pin_key}={pins[pin_key]}"
236 )
237 return findings

242_ASSIGN_RE = re.compile(r"^\s*([A-Za-z_][A-Za-z0-9_]*)\s*=\s*(.*?)\s*$")

244# Shortest value that can carry a matched pair of surrounding quotes ("" is 2).
245_MIN_QUOTED_LEN = 2
246
247
248def parse_assignments(text: str) -> dict[str, str]:
249 """Parse ``KEY=value`` lines, ignoring comments and blanks.
250
251 Handles both file formats: the shell fragment and Kconfig defaults share
252 this syntax, and Kconfig's quoted string values are unquoted here so
253 ``CONFIG_IDF_TARGET="esp32c6"`` compares equal to ``ESP_TARGET=esp32c6``.
254
255 Args:
256 text: File contents.
257
258 Returns:
259 Mapping of key to unquoted value.
260 """
261 out: dict[str, str] = {}
262 for raw in text.splitlines():
263 line = raw.strip()
264 if not line or line.startswith("#"):
265 continue
266 match = _ASSIGN_RE.match(line)
267 if match is None:
268 continue
269 key, value = match.group(1), match.group(2)
270 if len(value) >= _MIN_QUOTED_LEN and value[0] == value[-1] and value[0] in "\"'":
271 value = value[1:-1]
272 out[key] = value
273 return out

310def _check_one_signal(label: str, c6: str, ra8_pin: str, ra8_hole: str) -> list[str]:
311 """Return findings for one signal's RA8-side entry.
312
313 Args:
314 label: Human-readable signal name.
315 c6: C6 GPIO number as written in pins.env.
316 ra8_pin: RA8 landing pin, or "none".
317 ra8_hole: J26 hole, or "none".
318
319 Returns:
320 Human-readable findings; empty when the entry is well-formed.
321 """
322 unwired_c6 = c6 == UNWIRED_C6
323 unwired_ra8 = ra8_pin == UNWIRED_RA8 and ra8_hole == UNWIRED_RA8
324 if unwired_c6 != unwired_ra8:
325 return [
326 f"{label}: half-recorded connection -- C6 side is "
327 f"{'disconnected' if unwired_c6 else c6} but RA8 side is "
328 f"{ra8_pin}/{ra8_hole}"
329 ]
330 if unwired_ra8:
331 return []
332
333 findings: list[str] = []
334 if _RA8_PIN_RE.match(ra8_pin) is None:
335 findings.append(f"{label}: RA8 pin {ra8_pin!r} is not a Pnnn pin name or 'none'")
336 if _RA8_HOLE_RE.match(ra8_hole) is None:
337 findings.append(f"{label}: J26 hole {ra8_hole!r} is not a J26-n hole or 'none'")
338 return findings

341def _check_uniqueness(pins: dict[str, str]) -> list[str]:
342 """Return findings for any RA8 pin or J26 hole claimed by two signals.
343
344 Args:
345 pins: Parsed pins.env assignments.
346
347 Returns:
348 Human-readable findings; empty when every wired entry is unique.
349 """
350 findings: list[str] = []
351 for what, index in (("RA8 pin", 2), ("J26 hole", 3)):
352 seen: dict[str, str] = {}
353 for triple in RA8_TRIPLES:
354 value = pins.get(triple[index], UNWIRED_RA8)
355 if value == UNWIRED_RA8:
356 continue
357 if value in seen:
358 findings.append(f"{what} {value} is claimed by both {seen[value]} and {triple[0]}")
359 seen[value] = triple[0]
360 return findings

363def check_ra8_side(pins: dict[str, str]) -> list[str]:
364 """Return one message per defect in the RA8-side half of the pin map.
365
366 The C6-side numbers are checked against sdkconfig.defaults by ``compare``;
367 nothing downstream can check the RA8 side that way, because no second file
368 restates it. What is checkable is that the map is COMPLETE and INTERNALLY
369 CONSISTENT: every signal names both ends or neither, the names are
370 well-formed, no two signals claim one pin or one hole, and the SW4
371 positions the link depends on are recorded.
372
373 Args:
374 pins: Parsed pins.env assignments.
375
376 Returns:
377 Human-readable findings; empty when the RA8 side is well-formed.
378 """
379 findings: list[str] = []
380 for label, c6_key, pin_key, hole_key in RA8_TRIPLES:
381 missing = [k for k in (c6_key, pin_key, hole_key) if k not in pins]
382 if missing:
383 findings.extend(f"{label}: pins.env is missing {k}" for k in missing)
384 continue
385 findings.extend(_check_one_signal(label, pins[c6_key], pins[pin_key], pins[hole_key]))
386
387 findings.extend(_check_uniqueness(pins))
388
389 for key in SW4_KEYS:
390 if key not in pins:
391 findings.append(f"SW4 positions: pins.env is missing {key}")
392 elif pins[key] not in SW4_VALUES:
393 findings.append(f"SW4 positions: {key}={pins[key]!r} is not one of {SW4_VALUES}")
394 return findings

407def parse_host_fw_version(text: str) -> str | None:
408 """Return the vendored host driver's own version as ``major.minor.patch``.
409
410 Args:
411 text: Body of ``host/esp_hosted_host_fw_ver.h`` from the vendor tree.
412
413 Returns:
414 The dotted version, or ``None`` when any of the three macros is absent
415 (which means the vendored header changed shape and this check can no
416 longer speak for it -- reported, never silently skipped).
417 """
418 parts: list[str] = []
419 for macro in _FW_VER_MACROS:
420 match = re.search(rf"^\s*#define\s+{re.escape(macro)}\s+(\d+)\s*$", text, re.MULTILINE)
421 if match is None:
422 return None
423 parts.append(match.group(1))
424 return ".".join(parts)

427def check_fw_version_lock(pins: dict[str, str], header_text: str) -> list[str]:
428 """Return findings when the co-processor image and the host driver disagree.
429
430 ``pins.env`` names the esp-hosted-mcu firmware the CO-PROCESSOR image is
431 built from; ``esp_hosted_host_fw_ver.h`` states the version of the HOST
432 driver vendored into this tree. Both come from one pinned upstream commit,
433 so they must be identical -- and upstream itself compares them at run time
434 and warns about "RPC timeouts" when they are not.
435
436 This is a gate rather than a convention because the mismatch is otherwise
437 silent in CI and expensive on the bench:
438 ``examples/ek_ra8d2/hw_validated/c6/c6_fw_version`` asserts the
439 co-processor's reported version against these very macros, so bumping the
440 vendor pin without reflashing the C6 turns a hardware test red with no
441 hint, from here, of why.
442
443 Args:
444 pins: Parsed pins.env assignments.
445 header_text: Body of the vendored host firmware-version header.
446
447 Returns:
448 Human-readable findings; empty when the two agree.
449 """
450 findings: list[str] = []
451 host = parse_host_fw_version(header_text)
452 if host is None:
453 findings.append(
454 "firmware version: could not read "
455 f"{', '.join(_FW_VER_MACROS)} out of {HOST_FW_VER.name} "
456 "-- the vendored header changed shape"
457 )
458 return findings
459 if FW_VERSION_KEY not in pins:
460 findings.append(f"firmware version: pins.env is missing {FW_VERSION_KEY}")
461 return findings
462 if pins[FW_VERSION_KEY] != host:
463 findings.append(
464 f"firmware version: pins.env {FW_VERSION_KEY}={pins[FW_VERSION_KEY]} "
465 f"but the vendored host driver is {host} "
466 "-- host and co-processor must come from one upstream commit"
467 )
468 return findings

471def compare(pins: dict[str, str], sdk: dict[str, str]) -> list[str]:
472 """Return one message per disagreement between the two parsed files.
473
474 Args:
475 pins: Parsed pins.env assignments (the source of truth).
476 sdk: Parsed sdkconfig.defaults assignments (the derived artifact).
477
478 Returns:
479 Human-readable findings; empty when the two agree.
480 """
481 findings: list[str] = []
482
483 for sdk_key, pin_key, label in PIN_PAIRS + VALUE_PAIRS:
484 if pin_key not in pins:
485 findings.append(f"{label}: pins.env is missing {pin_key}")
486 continue
487 if sdk_key not in sdk:
488 findings.append(f"{label}: sdkconfig.defaults is missing {sdk_key}")
489 continue
490 if pins[pin_key] != sdk[sdk_key]:
491 findings.append(
492 f"{label}: pins.env {pin_key}={pins[pin_key]} "
493 f"but sdkconfig.defaults {sdk_key}={sdk[sdk_key]}"
494 )
495
496 size = _flash_size(sdk)
497 if FLASH_SIZE_KEY not in pins:
498 findings.append(f"flash size: pins.env is missing {FLASH_SIZE_KEY}")
499 elif size is None:
500 findings.append(
501 "flash size: sdkconfig.defaults enables no CONFIG_ESPTOOLPY_FLASHSIZE_<n>MB symbol"
502 )
503 elif size != pins[FLASH_SIZE_KEY]:
504 findings.append(
505 f"flash size: pins.env {FLASH_SIZE_KEY}={pins[FLASH_SIZE_KEY]} "
506 f"but sdkconfig.defaults enables CONFIG_ESPTOOLPY_FLASHSIZE_{size}"
507 )
508
509 board = _dev_board(sdk)
510 if DEV_BOARD_KEY not in pins:
511 findings.append(f"dev board: pins.env is missing {DEV_BOARD_KEY}")
512 elif board is None:
513 findings.append(
514 "dev board: sdkconfig.defaults enables no CONFIG_ESP_HOST_DEV_BOARD_<X> "
515 "symbol -- without an explicit selection the pin leaves above may be "
516 "overridden by whatever preset upstream defaults to"
517 )
518 elif board != pins[DEV_BOARD_KEY]:
519 findings.append(
520 f"dev board: pins.env {DEV_BOARD_KEY}={pins[DEV_BOARD_KEY]} "
521 f"but sdkconfig.defaults enables CONFIG_ESP_HOST_DEV_BOARD_{board.upper()} "
522 "-- a dev-board preset OVERRIDES the SPI pin leaves, so every pin "
523 "compared above would be ignored by the build"
524 )
525
526 findings.extend(check_ra8_side(pins))
527
528 try:
529 header = PORT_PIN_HEADER.read_text(encoding="utf-8")
530 except OSError as exc:
531 findings.append(f"port pin header: cannot read {PORT_PIN_HEADER}: {exc}")
532 else:
533 findings.extend(check_port_header(pins, header))
534 return findings

908def selftest() -> int:
909 """Prove the comparator reports drift and stays silent on agreement.
910
911 Returns:
912 EXIT_OK when every crafted case yields the expected verdict, else
913 EXIT_FAIL.
914 """
915 failures: list[str] = []
916 for label, pins_text, sdk_text, expect in _selftest_cases():
917 findings = compare(parse_assignments(pins_text), parse_assignments(sdk_text))
918 if bool(findings) != expect:
919 verb = "reported nothing" if expect else f"reported {findings}"
920 failures.append(f" {label}: {verb}")
921 failures.extend(_selftest_port_failures())
922 failures.extend(_selftest_fw_version_failures())
923
924 if failures:
925 sys.stderr.write("check_c6_pin_config.py --selftest: FAILED\n\n")
926 sys.stderr.write("\n".join(failures) + "\n")
927 sys.stderr.write("\nThe comparator does not detect drift as claimed.\n")
928 return EXIT_FAIL
929
930 cases = _selftest_cases() + _selftest_cases_port_header()
931 cases += [(a, b, "", c) for a, b, c in _selftest_cases_fw_version()]
932 fires = sum(1 for c in cases if c[3])
933 print(
934 f"check_c6_pin_config.py --selftest: OK "
935 f"({len(cases)} cases: {fires} must fire, {len(cases) - fires} must stay quiet)."
936 )
937 return EXIT_OK

940def main(argv: list[str]) -> int:
941 """Compare the committed pin map against the committed Kconfig defaults.
942
943 Returns:
944 EXIT_OK when they agree, EXIT_FAIL on drift or a failing selftest,
945 EXIT_CONFIG when either file is missing.
946 """
947 if "--selftest" in argv[1:]:
948 return selftest()
949
950 for path in (PINS_ENV, SDKCONFIG, HOST_FW_VER):
951 if not path.is_file():
952 sys.stderr.write(f"check_c6_pin_config.py: FATAL -- missing {path}\n")
953 return EXIT_CONFIG
954
955 pins = parse_assignments(PINS_ENV.read_text(encoding="utf-8"))
956 sdk = parse_assignments(SDKCONFIG.read_text(encoding="utf-8"))
957 findings = compare(pins, sdk)
958 findings.extend(check_fw_version_lock(pins, HOST_FW_VER.read_text(encoding="utf-8")))
959
960 if not findings:
961 # The three settings compared outside the two PAIR tables: the flash
962 # size, the dev-board preset (both encoded in a symbol NAME) and the
963 # host / co-processor firmware version lock.
964 key_encoded_and_version = 3
965 checked = len(PIN_PAIRS) + len(VALUE_PAIRS) + key_encoded_and_version
966 ra8 = (len(RA8_TRIPLES) * 2) + len(SW4_KEYS)
967 print(
968 f"check_c6_pin_config.py: pins.env and sdkconfig.defaults agree "
969 f"({checked} settings); dev board "
970 f"{pins.get(DEV_BOARD_KEY, '?')} (no preset overriding our pins); "
971 f"RA8-side map well-formed ({ra8} entries); "
972 f"co-processor firmware {pins.get(FW_VERSION_KEY, '?')} matches the "
973 f"vendored host driver."
974 )

976
977 sys.stderr.write(f"check_c6_pin_config.py: {len(findings)} C6 config drift(s):\n")
978 for message in findings:
979 sys.stderr.write(f" {message}\n")
980 sys.stderr.write(
