347static UINT
348dfu_write(VOID* dfu, ULONG block_number, const UCHAR* data, ULONG length, ULONG* media_status)
349{
350 (void)dfu;
351 const ULONG off = block_number * (ULONG)k_dfu_xfer_size;
352 if ((length > 0UL) && (off + length <= (ULONG)sizeof(s_dfu_image))) {
353 (void)memcpy(&s_dfu_image[off], data, (size_t)length);
354 if ((uint32_t)(off + length) > s_dfu_image_len) {
355 s_dfu_image_len = (uint32_t)(off + length);
356 }
357 s_dbg_dev_writes++;
358 }
359 *media_status = (ULONG)UX_SLAVE_CLASS_DFU_MEDIA_STATUS_OK;
360 return UX_SUCCESS;
361}

380static UINT dfu_read(VOID* dfu, ULONG block_number, UCHAR* data, ULONG length, ULONG* actual_length)
381{
382 (void)dfu;
383 const ULONG off = block_number * (ULONG)k_dfu_xfer_size;
384 if (off >= (ULONG)s_dfu_image_len) {
385 *actual_length = 0UL;
386 return UX_SUCCESS;
387 }
388 ULONG remain = (ULONG)s_dfu_image_len - off;
389 if (remain > length) {
390 remain = length;
391 }
392 (void)memcpy(data, &s_dfu_image[off], (size_t)remain);
393 *actual_length = remain;
394 return UX_SUCCESS;
395}

525static VOID dfu_device_worker(ULONG arg)
526{
527 (void)arg;
528
529 UINT ux = dfu_usbx_stack_up();
530 if (ux != UX_SUCCESS) {
531 s_dbg_host_err = (uint32_t)ux;
532 return;
533 }
534 s_dbg_dev_step = 1U;
535 ux = dfu_class_register();
536 if (ux != UX_SUCCESS) {
537 s_dbg_host_err = (uint32_t)ux;
538 return;
539 }
540 s_dbg_dev_step = 2U;
541 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
542 if (e != k_ra8_ok) {
543 s_dbg_host_err = (uint32_t)e;
544 return;
545 }
546 s_dbg_dev_step = 3U;
547 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
548 if (e != k_ra8_ok) {
549 s_dbg_host_err = (uint32_t)e;
550 return;
551 }
552 s_dbg_dev_step = 4U;
553 while (1) {
554 tx_thread_sleep(k_dfu_idle_ticks);
555 }
556}

569static VOID dfu_host_worker(ULONG arg)
570{
571 (void)arg;
572
573 tx_thread_sleep(k_dfu_boot_wait_ticks);
574 for (;;) {
575 const ra8_err_t err = dfu_host_pass();
576 if (err == k_ra8_ok) {
577 break;
578 }
579 s_dbg_host_err = (uint32_t)err;
580 tx_thread_sleep(k_dfu_retry_ticks);
581 }
582 while (1) {
583 tx_thread_sleep(k_dfu_idle_ticks);
584 }
585}

707 dfu_panic_halt();

710 dfu_panic_halt();
