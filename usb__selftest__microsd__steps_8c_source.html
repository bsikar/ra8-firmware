251static UINT microsd_msc_read(VOID* storage,
252 ULONG lun,
253 UCHAR* data_pointer,
254 ULONG number_blocks,
255 ULONG lba,
256 ULONG* media_status)
257{
258 (void)storage;
259 (void)lun;
260 s_dbg_read_calls++;
261 if ((lba + number_blocks) > (ULONG)k_microsd_sectors) {
262 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
263 k_scsi_asc_lba_out_of_range,
264 k_scsi_ascq_none);
265 return UX_ERROR;
266 }
267 (void)memcpy(data_pointer,
268 &s_usb_selftest_microsd_sd_snapshot[lba * (ULONG)k_microsd_block_size],
269 (size_t)(number_blocks * (ULONG)k_microsd_block_size));
270 *media_status = 0UL;
271 (void)ra8_board_led_toggle(k_ra8_board_led1);
272 return UX_SUCCESS;
273}

523[[nodiscard]] static ra8_err_t microsd_verify_one(uint32_t lun)
524{
525 static uint8_t s_burst[k_microsd_burst_bytes] = {};
526
527 uint32_t block_count = 0U;
528 uint32_t block_size = 0U;
529 ra8_err_t err = ra8_usb_hmsc_read_capacity((uint8_t)lun, &block_count, &block_size);
530 if (err != k_ra8_ok) {
531 (void)microsd_print_fail("read_capacity", err);
532 return err;
533 }
534 if (block_count != (uint32_t)k_microsd_sectors) {
535 (void)microsd_print_fail("capacity mismatch", k_ra8_err_invalid_size);
536 return k_ra8_err_invalid_size;
537 }
538 if (s_usb_selftest_microsd_dbg_bootsig == 0U) {
539 (void)microsd_print_fail("no 0x55AA boot sig on SD LBA0", k_ra8_err_invalid_state);
540 return k_ra8_err_invalid_state;
541 }
542 for (uint32_t blk = 0U; blk < (uint32_t)k_microsd_sectors;
543 blk += (uint32_t)k_microsd_burst_blocks) {
544 err = ra8_usb_hmsc_read10((uint8_t)lun, blk, (uint16_t)k_microsd_burst_blocks, s_burst);
545 if (err != k_ra8_ok) {
546 (void)microsd_print_fail("READ(10)", err);
547 return err;
548 }
549 const uint32_t off = blk * (uint32_t)k_microsd_block_size;
550 if (memcmp(s_burst, &s_usb_selftest_microsd_sd_snapshot[off], (size_t)k_microsd_burst_bytes) !=
551 0) {
552 s_dbg_mismatch = blk;
553 (void)microsd_print_fail("SD data mismatch vs snapshot", k_ra8_err_invalid_state);
554 return k_ra8_err_invalid_state;
555 }
556 }
557 return k_ra8_ok;
558}

656[[nodiscard]] static ra8_err_t microsd_host_pass(void)
657{
658 s_dbg_phase = (uint32_t)k_microsd_phase_init;
659 ra8_err_t err = microsd_print("ra8d2 microsd: host up on USB-HS, probing the loop...\r\n");
660 if (err != k_ra8_ok) {
661 return err;
662 }
663 err = ra8_usb_hmsc_init(k_ra8_usb_speed_hs);
664 if (err != k_ra8_ok) {
665 (void)microsd_print_fail("host init", err);
666 return err;
667 }
668
669 ra8_usb_hmsc_device_t device = {};
670 err = microsd_host_enumerate(&device);
671 if (err != k_ra8_ok) {
672 return err;
673 }
674
675 s_dbg_phase = (uint32_t)k_microsd_phase_verify;
676 s_dbg_luns_ok = 0U;
677 for (uint32_t lun = 0U; lun < (uint32_t)k_microsd_count; lun++) {
678 err = microsd_verify_one(lun);
679 if (err != k_ra8_ok) {
680 (void)ra8_usb_hmsc_close();
681 return err;
682 }
683 s_dbg_luns_ok++;
684 err = microsd_print_lun_ok(lun);
685 if (err != k_ra8_ok) {
686 return err;
687 }
688 }
689
690 s_dbg_phase = (uint32_t)k_microsd_phase_pass;
691 s_dbg_pass_count++;
692 err = microsd_print("ra8d2 microsd: USB SELFTEST MICROSD PASS\r\n");
693 if (err != k_ra8_ok) {
694 return err;
695 }
696 (void)ra8_board_led_on(k_ra8_board_led2);
697 return k_ra8_ok;
698}
