43void board_usb_roles_swap(uc_engine* uc)
44{
45 (void)uc;
46 if (s_roles_swapped) {
47 return; /* one-shot: both triggers (FS DCFM=1, HS DPRPU=1) fire in Config B. */
48 }
49 /* The device firmware's PHY + module-init writes landed in the HS window's
50 * shadow while its role was still unknowable (the HS host bring-up drives the
51 * very same device-polarity PHY sequence first). Adopt that accumulated state
52 * as the device model's register file -- the models swap windows, the
53 * firmware-visible register contents must not change -- and reset the host
54 * model for its fresh life behind the FS window (which is still virgin: the
55 * FS host driver's first window write is the SYSCFG.DCFM=1 that triggers
56 * this swap, and the HS DPRPU=1 trigger fires before any FS host activity). */
57 (void)board_usbhs_host_shadow_handoff(s_usb.reg, (uint32_t)k_usb_reg_words);
58 s_roles_swapped = true;
59 s_dev_irq_event = (uint16_t)k_ra8_elc_event_usbhs_int_resume;
60 usb_log_line("role swap: FS window = host model, HS window = device model (Config B)");
61}

112uint16_t board_usb_bridge_dcp_in_take(uint8_t* buf, uint16_t cap)
113{
114 if (buf == nullptr) {
115 return 0U;
116 }
117 uint16_t n = s_usb.dcp_in.len;
118 if (n > cap) {
119 n = cap;
120 }
121 (void)memcpy(buf, s_usb.dcp_in.data, n);
122 usb_detect_class(s_usb.dcp_in.data, s_usb.dcp_in.len);
123 usb_log_count("bridge control-IN: device returned", (unsigned)s_usb.dcp_in.len);
124 if (s_trace) {
125 /* Payload head, --trace only: enough to identify a descriptor / status. */
126 char line[k_usb_log_width];
127 int w = 0;
128 const uint16_t limit =
129 (n < (uint16_t)k_usb_trace_dump_max) ? n : (uint16_t)k_usb_trace_dump_max;
130 for (uint16_t i = 0U; (i < limit) && (w >= 0) && ((size_t)w < sizeof(line)); i++) {
131 w += snprintf(&line[w], sizeof(line) - (size_t)w, "%02X ", (unsigned)s_usb.dcp_in.data[i]);
132 }
133 (void)fprintf(stderr, " [usb] bridge control-IN bytes: %s\n", line);
134 }
135 s_usb.dcp_in.len = 0U;
136 s_usb.dcp_in.valid = false;
137 return n;

159{

178bool board_usb_bridge_bulk_out_consumed(uint8_t dev_pipe)
179{
180 const usb_out_buf_t* b = &s_usb.pipe_out[dev_pipe % k_usb_pipe_count];
181 /* The device firmware drains the staged OUT packet through its CFIFO, which
182 * advances @c rd to @c len. Until then the packet is still in flight, so the
183 * host must not push the next one (it would overwrite this bank -- the exact
184 * failure that wedged a multi-packet WRITE(10) data stage). */
185 return b->ready && (b->rd >= b->len);
186}

214void board_usb_bridge_dcp_out(uc_engine* uc, const uint8_t* data, uint16_t len)
215{
216 (void)uc;
217 if (data == nullptr) {
218 return;
219 }
220 uint16_t n = len;
221 if (n > (uint16_t)sizeof(s_dcp_hold)) {
222 n = (uint16_t)sizeof(s_dcp_hold);
223 }
224 /* Hold the bytes as "in flight on the wire": the device has almost certainly
225 * not armed its DCP for OUT yet, and its arm-time BCLR would discard them.
226 * bridge_pump_device() lands them once the device is ready. */
227 (void)memcpy(s_dcp_hold, data, n);
228 s_dcp_hold_len = n;
229 s_dcp_hold_pending = true;
230}

257void bridge_pump_device(uc_engine* uc)
258{
259 if (!s_dcp_hold_pending) {
260 return;
261 }
262 const uint16_t dcpctr = s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_dcpctr)];
263 const uint16_t brdyenb = s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_brdyenb)];
264 const bool armed = ((dcpctr & (uint16_t)k_ra8_pid_mask) == (uint16_t)k_ra8_pid_buf) &&
265 ((brdyenb & (uint16_t)k_usb_dcp_pipe_bit) != 0U);
266 if (!armed) {
267 return; /* device has not armed its DCP for OUT yet; keep the bytes on the wire. */
268 }
269 /* The device is ready (its arm-time BCLR already ran on an empty bank): land
270 * the held control-OUT packet into the DCP and raise its BRDY so the device's
271 * receive ISR drains it, exactly as an OUT token the device finally ACKs. */
272 (void)memcpy(s_usb.dcp_out.data, s_dcp_hold, s_dcp_hold_len);
273 s_usb.dcp_out.len = s_dcp_hold_len;
274 s_usb.dcp_out.rd = 0U;
275 s_usb.dcp_out.ready = true;
276 s_dcp_hold_pending = false;
277 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_brdysts);
278 s_usb.reg[w] = (uint16_t)(s_usb.reg[w] | (uint16_t)k_usb_dcp_pipe_bit);
279 usb_intsts0_set((uint8_t)k_ra8_int0_bit_brdy);
280 usb_raise_irq(uc);
281}
