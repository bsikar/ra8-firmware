238RA8_INTERNAL
239static ra8_err_t internal_unwind(void)
240{
241 /* Both release steps run before either result is folded, so neither can be
242 skipped by an earlier failure. The fold itself lives in
243 `priv_ra8_esp_hosted_port_first_error()` because it is the only decision here
244 and reaching this function needs a brought-up port. */
245 const ra8_err_t spi_err = priv_ra8_esp_hosted_spi_close();
246 ra8_err_t first = priv_ra8_esp_hosted_port_first_error(k_ra8_ok, spi_err);
247 const ra8_err_t rtos_err = priv_ra8_esp_hosted_rtos_deinit();
248 first = priv_ra8_esp_hosted_port_first_error(first, rtos_err);
249
250 g_hosted_osi_funcs = (hosted_osi_funcs_t){};
251 s_ra8_esp_hosted_port_ready = false;
252 return first;
253}

305RA8_INTERNAL
306static ra8_err_t internal_bring_up(const ra8_esp_hosted_port_cfg_t* cfg)
307{
308 RA8_CHECK_NULL_PTR(cfg, s_esp_hosted_tag, "cfg");
309
310 ra8_err_t err = priv_ra8_esp_hosted_rtos_init();
311 if (err != k_ra8_ok) {
312 return err;
313 }
314
315 err = priv_ra8_esp_hosted_osi_bind_all(&g_hosted_osi_funcs);
316 if (err == k_ra8_ok) {
317 err = priv_ra8_esp_hosted_gpio_set_edge_poll_ms(cfg->edge_poll_ms);
318 }
319 if (err == k_ra8_ok) {
320 err = priv_ra8_esp_hosted_spi_open(cfg->sci_channel, cfg->pclk_hz, cfg->sck_hz);
321 }
322 if (err == k_ra8_ok) {
323 err = internal_configure_sideband();
324 }
325 return err;
326}

331ra8_err_t ra8_esp_hosted_port_init(const ra8_esp_hosted_port_cfg_t* cfg)
332{
333 const ra8_err_t cfg_err = priv_ra8_esp_hosted_port_cfg_check(cfg);
334 if (cfg_err != k_ra8_ok) {
335 return cfg_err;
336 }
337 if (s_ra8_esp_hosted_port_ready) {
338 return k_ra8_err_invalid_state;
339 }
340
341 const ra8_err_t pin_err = priv_ra8_esp_hosted_port_pins_check();
342 if (pin_err != k_ra8_ok) {
343 ESP_LOGE(s_esp_hosted_tag, "pin map is not self-consistent");
344 return pin_err;
345 }
346
347 const ra8_err_t err = internal_bring_up(cfg);
348 if (err != k_ra8_ok) {
349 (void)internal_unwind();
350 return err;
351 }
352
353 s_ra8_esp_hosted_port_ready = true;
354 ESP_LOGI(s_esp_hosted_tag,
355 "esp-hosted port up on SCI%u at %u Hz",
356 (unsigned int)cfg->sci_channel,
357 (unsigned int)cfg->sck_hz);
358 return k_ra8_ok;
359}
