134{

136 return k_ra8_err_invalid_arg;

144 case k_ra8_flash_irq_extra_ecc_dec:

247static uint32_t internal_dispatch_ecc(uint16_t status_off,
248 uint16_t ted_addr_off,
249 uint16_t dec_addr_off,
250 ra8_flash_irq_src_t ted_src,
251 ra8_flash_irq_src_t dec_src)
252{
253 uint32_t delivered = 0U;
254 const uint8_t status = *ra8_mram_reg8(status_off);
255 if ((status & k_ra8_mrcraes_mask_tederrc) != 0U) {
256 const uint32_t fa = *ra8_mram_reg32(ted_addr_off);
257 internal_deliver(ted_src, fa, (uint32_t)status);
258 delivered++;
259 }
260 if ((status & k_ra8_mrcraes_mask_decerrc) != 0U) {
261 const uint32_t fa = *ra8_mram_reg32(dec_addr_off);
262 internal_deliver(dec_src, fa, (uint32_t)status);
263 delivered++;
264 }
265 if (status != 0U) {
266 *ra8_mram_reg8(status_off) = 0U;
267 }
268 return delivered;
269}

271uint32_t ra8_flash_dispatch_isr(void)
272{
273 uint32_t delivered = 0U;
274
275 delivered += internal_dispatch_ecc(k_ra8_mram_off_mrcraes,
276 k_ra8_mram_off_mrcrtea,
277 k_ra8_mram_off_mrcrdea,
278 k_ra8_flash_irq_code_ecc_ted,
279 k_ra8_flash_irq_code_ecc_dec);
280 delivered += internal_dispatch_ecc(k_ra8_mram_off_mreraes,
281 k_ra8_mram_off_mrertea,
282 k_ra8_mram_off_mrerdea,
283 k_ra8_flash_irq_extra_ecc_ted,
284 k_ra8_flash_irq_extra_ecc_dec);
285
286 /* HUM Ch 59 "MRCPS : Code MRAM Program Status Register" p 3577 */
287 const uint8_t mrcps = *ra8_mram_reg8(k_ra8_mram_off_mrcps);
288 if ((mrcps & k_ra8_mrcps_mask_errors) != 0U) {
289 /* HUM Ch 59 "MRCPEA : Code MRAM Program Error Address" p 3579 */
290 const uint32_t fa = *ra8_mram_reg32(k_ra8_mram_off_mrcpea);
291 internal_deliver(k_ra8_flash_irq_program_err, fa, (uint32_t)mrcps);
292 /* W1C the program error bits. */
293 *ra8_mram_reg8(k_ra8_mram_off_mrcps) = k_ra8_mrcps_mask_errors;
294#ifdef RA8_OFF_TARGET
295 /* The host-test fake is dumb memory: emulate the W1C clear so the
296 * post-dispatch state matches real HW. */
297 *ra8_mram_reg8(k_ra8_mram_off_mrcps) &= (uint8_t)~k_ra8_mrcps_mask_errors;
298#endif
299 delivered++;
300 }
301
302 /* HUM Ch 59 "MASTAT : Extra MRAM Access Status Register" p 3562 */
303 const uint8_t mastat = *ra8_mram_reg8(k_ra8_mram_off_mastat);
304 if ((mastat & k_ra8_mastat_mask_mreae) != 0U) {
305 internal_deliver(k_ra8_flash_irq_extra_err, 0U, (uint32_t)mastat);
306 delivered++;
307 }
308 if ((mastat & k_ra8_mastat_mask_cmdlk) != 0U) {
309 internal_deliver(k_ra8_flash_irq_extra_cmdlk, 0U, (uint32_t)mastat);
310 delivered++;
311 }
312
313 /* HUM Ch 59 "MSTATR : Extra MRAM Status Register" p 3568 */
314 const uint32_t mstatr = *ra8_mram_reg32(k_ra8_mram_off_mstatr);
315 if ((mstatr & k_ra8_mstatr_mask_mrdy) != 0U) {
316 internal_deliver(k_ra8_flash_irq_extra_ready, 0U, mstatr);
317 delivered++;
318 }
319 return delivered;
320}

380static ra8_err_t internal_validate_range(uintptr_t address, uint64_t total_len)
381{
382 if ((address & (k_ra8_mram_block_size_bytes - 1U)) != 0U) {
383 return k_ra8_err_invalid_arg;
384 }
385 const uint64_t end_excl = (uint64_t)address + total_len;
386 if (address < k_ra8_flash_code_start) {

398ra8_err_t ra8_flash_erase(uintptr_t address, uint32_t num_blocks)
399{
400 RA8_VALIDATE_INIT(s_flash_rt.initialized, s_flash_tag, "flash_erase before init");
401 if (num_blocks == 0U) {
402 return k_ra8_err_invalid_arg;
403 }
404 const uint64_t total_bytes = (uint64_t)num_blocks * (uint64_t)k_ra8_mram_block_size_bytes;
405 const ra8_err_t v_err = internal_validate_range(address, total_bytes);
406 RA8_RETURN_ON_ERROR(v_err, s_flash_tag, "flash_erase: validate"); /* GCOVR_EXCL_BR_LINE */
407 /* internal_validate_range above rejects any address outside
408 * [k_ra8_flash_code_start, +k_ra8_flash_code_size), which is the default
409 * (non-secure) 0x02000000 code-MRAM view -- HUM Ch 59.1 "Address Map" p 3543.
410 * The secure alias therefore cannot reach here, so MRCPC0 is provably the
411 * right controller. A caller that needs the secure half calls
412 * ra8_flash_erase_block / ra8_flash_write_block directly with world_s. */
413 const ra8_flash_world_t world = k_ra8_flash_world_ns;
414 /* FSP r_mram.c L917 mram_erase_blocks loops one programming-size block at
415 * a time; we mirror that one-block-per-iteration cadence. NASA Rule 2:
416 * loop bound is the caller-supplied num_blocks, validated above. */
417 uintptr_t cur = address;
418 for (uint32_t i = 0U; i < num_blocks; ++i) {
419 const ra8_err_t err = ra8_flash_erase_block((uint32_t)cur, world);
420 if (err != k_ra8_ok) {
421 return err;
422 }
423 cur += k_ra8_mram_block_size_bytes;
424 }
425 return k_ra8_ok;
426}

428ra8_err_t ra8_flash_write(uintptr_t address, const uint8_t* src, uint32_t len)
429{
430 RA8_CHECK_NULL_PTR(src, s_flash_tag, "src must not be nullptr");
431 RA8_VALIDATE_INIT(s_flash_rt.initialized, s_flash_tag, "flash_write before init");
432 if (len == 0U || (len % k_ra8_mram_write_size_bytes) != 0U) {
433 return k_ra8_err_invalid_arg;
434 }
435 const ra8_err_t v_err = internal_validate_range(address, (uint64_t)len);
436 RA8_RETURN_ON_ERROR(v_err, s_flash_tag, "flash_write: validate"); /* GCOVR_EXCL_BR_LINE */
437 /* Same reasoning as ra8_flash_erase: the validated range lies inside the
438 * default non-secure code-MRAM view, so MRCPC0 is provably correct. */
439 const ra8_flash_world_t world = k_ra8_flash_world_ns;
440 /* FSP r_mram.c L861 mram_write_data chunks the request into 32-byte
441 * page programs; our wrapper re-uses ra8_flash_write_block for each
442 * page. NASA Rule 2: loop bound is len/page (validated above). */
443 const uint32_t pages = len / k_ra8_mram_write_size_bytes;
444 for (uint32_t i = 0U; i < pages; ++i) {
445 const uint32_t offset = i * k_ra8_mram_write_size_bytes;
446 const uintptr_t page_addr = address + offset;
447 const ra8_err_t err =
448 ra8_flash_write_block((uint32_t)page_addr, src + offset, k_ra8_mram_write_size_bytes, world);
449 if (err != k_ra8_ok) {
450 return err;
451 }
452 }
453 return k_ra8_ok;
454}

463ra8_err_t ra8_flash_blank_check(uintptr_t address, uint32_t len, bool* out_blank)
464{
465 RA8_CHECK_NULL_PTR(out_blank, s_flash_tag, "out_blank must not be nullptr");
466 if (len == 0U) {
467 return k_ra8_err_invalid_arg;
468 }
469 /* Accept either MRAM window. FSP r_mram.c L395 R_MRAM_BlankCheck returns
470 * UNSUPPORTED on RA8D2 because the silicon has no BlankCheck command;
471 * we implement the check as a direct read since 0xFF is the documented
472 * erased state (HUM Ch 59.4.2 p 3548). */
473 const uint64_t end_excl = (uint64_t)address + (uint64_t)len;
474 const bool in_code =
475 // mcdc-deactivated: ra8_flash_blank_check window-membership AND; start address and end_excl are derived from the same caller-supplied (address, len) pair, so the two inequalities are co-dependent -- any address within the window satisfies both, any address outside violates the first; the MC/DC vector that would flip the upper bound while keeping the lower bound true requires a window-spanning length that the public-API len-cap upstream rejects.
476 (address >= k_ra8_flash_code_start) &&
477 (end_excl <= (uint64_t)k_ra8_flash_code_start + (uint64_t)k_ra8_flash_code_size);
478 const bool in_extra =
479 // mcdc-deactivated: ra8_flash_blank_check extra-window membership AND; identical co-dependence rationale as the code-window decision above.
480 (address >= k_ra8_flash_extra_start) &&
481 (end_excl <= (uint64_t)k_ra8_flash_extra_start + (uint64_t)k_ra8_flash_extra_size);
482 /* HUM Ch 7 "Option-Setting Memory" p 278 also benefits from a blank-check
483 * (callers may want to verify a freshly-erased OFS slot before re-write). */
484 const bool in_ofs =
485 (address >= k_ra8_flash_ofs_start) &&
486 (end_excl <= (uint64_t)k_ra8_flash_ofs_start + (uint64_t)k_ra8_flash_ofs_size);
487 // mcdc-deactivated: ra8_flash_blank_check 3-way OR over disjoint flash windows; tests cover the four addressable outcomes (in-code, in-extra, in-ofs, out-of-range), but llvm-cov MC/DC requires a vector where exactly one of the three booleans flips while the others stay false -- the windows are mutually exclusive by construction so no such vector exists.
488 if (!in_code && !in_extra && !in_ofs) {
489 return k_ra8_err_invalid_arg;
490 }
491
492 const volatile uint8_t* p = (const volatile uint8_t*)address;
493 bool is_blank = true;
494 /* NASA Rule 2: bounded loop on caller-validated len. */
495 for (uint32_t i = 0U; i < len; ++i) {
496 if (p[i] != k_ra8_flash_blank_byte) {
497 is_blank = false;
498 break;
499 }
500 }
501 *out_blank = is_blank;
502 return k_ra8_ok;
503}

505ra8_err_t ra8_flash_status(ra8_flash_status_t* out)
506{
507 RA8_CHECK_NULL_PTR(out, s_flash_tag, "out must not be nullptr");
508
509 /* HUM Ch 59 "MRCPS : Code MRAM Program Status Register" p 3577 */
510 const uint8_t mrcps = *ra8_mram_reg8(k_ra8_mram_off_mrcps);
511 /* HUM Ch 59 "MASTAT : Extra MRAM Access Status Register" p 3562 */
512 const uint8_t mastat = *ra8_mram_reg8(k_ra8_mram_off_mastat);
513 /* HUM Ch 59 "MENTRYR : Extra MRAM Program-Mode Entry" p 3571 */
514 const uint16_t mentryr = *ra8_mram_reg16(k_ra8_mram_off_mentryr);
515 /* HUM Ch 59 "MSTATR : Extra MRAM Status Register" p 3568 */
516 const uint32_t mstatr = *ra8_mram_reg32(k_ra8_mram_off_mstatr);
517 /* HUM Ch 59 "MRCBPROT0 : Code MRAM Block Protection (NS)" p 3576 */
518 const uint16_t mrcbprot0 = *ra8_mram_reg16(k_ra8_mram_off_mrcbprot0);
519 /* HUM Ch 59 "MRCBPROT1 : Code MRAM Block Protection (S)" p 3577 */
520 const uint16_t mrcbprot1 = *ra8_mram_reg16(k_ra8_mram_off_mrcbprot1);
521
522 const bool busy =
523 ((mrcps & k_ra8_mrcps_mask_prgbsyc) != 0U) || ((mentryr & k_ra8_mentryr_mask_pe_mode) != 0U);
524 out->programming_busy = busy;
525 out->erase_busy = busy; /* MRAM has no separate erase. */
526 out->illegal_command =
527 ((mastat & k_ra8_mastat_mask_cmdlk) != 0U) || ((mstatr & k_ra8_mstatr_mask_ilgcomerr) != 0U);
528 out->voltage_error = ((mstatr & k_ra8_mstatr_mask_oterr) != 0U);
529 /* MRCBPROTx low bit = 1 means programming is permitted; bit clear
530 * (the keyed lock pattern with bit 0 == 0) means the block is
531 * write-protected. HUM Ch 59 p 3604..3605. */
532 out->sector_protected = ((mrcbprot0 & 0x0001U) == 0U) || ((mrcbprot1 & 0x0001U) == 0U);
533 out->program_error = ((mrcps & k_ra8_mrcps_mask_prgerrc) != 0U);
534 out->ecc_error = ((mrcps & k_ra8_mrcps_mask_eccerrc) != 0U);
535 return k_ra8_ok;
