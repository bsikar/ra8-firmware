111static uint64_t cb_rng(uint64_t* s)
112{
113 uint64_t x = *s;
114 x ^= x << (uint8_t)k_rng_shift_a;
115 x ^= x >> (uint8_t)k_rng_shift_b;
116 x ^= x << (uint8_t)k_rng_shift_c;
117 *s = x;
118 return x;

227static cb_trace_t cb_gen_mixed(void)
228{
229 cb_trace_t t = {};
230 if (!cb_alloc(&t, "mixed-session", k_cb_accesses)) {
231 return t;
232 }
233 uint64_t s = (uint64_t)k_rng_seed_mixed_a ^ (uint64_t)k_rng_seed_mixed_b;
234 uint32_t page = 0U;
235 uint32_t hot = 0U;
236 for (uint64_t i = 0U; i < t.n; ++i) {
237 const uint32_t phase =
238 (uint32_t)((i / (uint64_t)k_cb_mixed_phase) % 4U); /* alternate behaviours */
239 if (phase == 0U) {
240 page = (page + 1U) % k_cb_footprint; /* linear */
241 } else if (phase == 1U) {
242 page = hot + cb_rand_below(&s, k_cb_hot_pages); /* reread */
243 } else if (phase == 2U) {
244 page = cb_rand_below(&s, k_cb_footprint); /* random jumps */
245 hot = (page < k_cb_hot_pages) ? 0U : (page - k_cb_hot_pages);
246 } else {
247 t.keys[i] = (cb_key_t){.object_id = k_cb_obj_comic, .page = page % k_cb_tile_span};
248 continue;
249 }
250 t.keys[i] = (cb_key_t){.object_id = k_cb_obj_book, .page = page % k_cb_footprint};
251 }
252 t.footprint = k_cb_footprint;
253 return t;
254}

376static bool cb_parse_trace_line(const char* line, uint32_t* obj, uint32_t* pg)
377{

382 return false;

388 return false;

395cb_trace_t cb_trace_load(const char* path, const char* name)
396{
397 cb_trace_t t = {.name = name};
398 FILE* f = fopen(path, "r");
399 if (f == nullptr) {
400 /* cppcheck-suppress resourceLeak ; false positive: cppcheck 2.13 does not
401 * model the C23 nullptr keyword, so it cannot see f is NULL here. */
402 return t;
403 }
404 uint64_t cap = (uint64_t)k_cb_load_init;
405 t.keys = (cb_key_t*)calloc((size_t)cap, sizeof(cb_key_t));
406 if (t.keys == nullptr) {
407 (void)fclose(f);
408 return t;
409 }
410 char line[k_cb_line_max] = {};
411 while (fgets(line, (int)sizeof(line), f) != nullptr) {
412 uint32_t obj = 0U;
413 uint32_t pg = 0U;
414 if (!cb_parse_trace_line(line, &obj, &pg)) {
415 break; /* stop at the first malformed record, as fscanf did */
416 }
417 if (t.n == cap) {
418 cap *= 2U;
419 cb_key_t* grown = (cb_key_t*)realloc(t.keys, (size_t)cap * sizeof(cb_key_t));
420 if (grown == nullptr) {
421 break;
422 }
423 t.keys = grown;
424 }
425 t.keys[t.n] = (cb_key_t){.object_id = obj, .page = pg};
426 t.n++;
427 }
428 (void)fclose(f);
429 return t;
430}
