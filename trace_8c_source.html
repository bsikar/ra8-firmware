131static uint64_t cb_rng(uint64_t* s)
132{
133 uint64_t x = *s;
134 x ^= x << (uint8_t)k_rng_shift_a;
135 x ^= x >> (uint8_t)k_rng_shift_b;
136 x ^= x << (uint8_t)k_rng_shift_c;
137 *s = x;
138 return x;

394static cb_trace_t cb_gen_mixed(void)
395{
396 cb_trace_t t = {};
397 if (!cb_alloc(&t, "mixed-session", k_cb_accesses)) {
398 return t;
399 }
400 uint64_t s = (uint64_t)k_rng_seed_mixed_a ^ (uint64_t)k_rng_seed_mixed_b;
401 uint32_t page = 0U;
402 uint32_t hot = 0U;
403 for (uint64_t i = 0U; i < t.n; ++i) {
404 const uint32_t phase =
405 (uint32_t)((i / (uint64_t)k_cb_mixed_phase) % 4U); /* alternate behaviours */
406 if (phase == 0U) {
407 page = (page + 1U) % k_cb_footprint; /* linear */
408 } else if (phase == 1U) {
409 page = hot + cb_rand_below(&s, k_cb_hot_pages); /* reread */
410 } else if (phase == 2U) {
411 page = cb_rand_below(&s, k_cb_footprint); /* random jumps */
412 hot = (page < k_cb_hot_pages) ? 0U : (page - k_cb_hot_pages);
413 } else {
414 t.keys[i] = (cb_key_t){.object_id = k_cb_obj_comic, .page = page % k_cb_tile_span};
415 continue;
416 }
417 t.keys[i] = (cb_key_t){.object_id = k_cb_obj_book, .page = page % k_cb_footprint};
418 }
419 t.footprint = k_cb_footprint;
420 return t;
421}

596cb_trace_t cb_trace_load(const char* path, const char* name)
597{
598 cb_trace_t t = {.name = name};
599 FILE* f = fopen(path, "r");
600 if (f == nullptr) {
601 /* cppcheck-suppress resourceLeak ; false positive: cppcheck 2.13 does not
602 * model the C23 nullptr keyword, so it cannot see f is NULL here. */
603 return t;
604 }
605 uint64_t cap = (uint64_t)k_cb_load_init;
606 t.keys = (cb_key_t*)calloc((size_t)cap, sizeof(cb_key_t));
607 if (t.keys == nullptr) {
608 (void)fclose(f);
609 return t;
610 }
611 char line[k_cb_line_max] = {};
612 while (fgets(line, (int)sizeof(line), f) != nullptr) {
613 uint32_t obj = 0U;
614 uint32_t pg = 0U;
615 if (!cb_parse_trace_line(line, &obj, &pg)) {
616 break; /* stop at the first malformed record, as fscanf did */
617 }
618 if (t.n == cap) {
619 cap *= 2U;
620 cb_key_t* grown = (cb_key_t*)realloc(t.keys, (size_t)cap * sizeof(cb_key_t));
621 if (grown == nullptr) {
622 break;
623 }
624 t.keys = grown;
625 }
626 t.keys[t.n] = (cb_key_t){.object_id = obj, .page = pg};
627 t.n++;
628 }
629 (void)fclose(f);
630 return t;
631}
