234int32_t main(void)
235{
236 /* CGC bring-up FIRST. tx_initialize_low_level.S programs SysTick
237 * with a reload sized for the post-PLL CPUCLK0 = 1 GHz (see the
238 * threadx_blink/main.c rationale). Skipping ra8_cgc_init leaves the
239 * chip on the MOCO (~8.4 MHz) so the SysTick reload takes ~119 ms
240 * wallclock per tick -- the worker's tx_thread_sleep(1000) would
241 * then sleep for ~2 minutes and the HIL counter window would see
242 * zero advance. Bring up the PLL before tx_kernel_enter so the
243 * scheduler tick rate matches what tx_user.h declared. */
244 if (ra8_cgc_init() != k_ra8_ok) {
245 while (1) {
246 __asm__ volatile("wfi");
247 }
248 }
249
250 if (ra8_mpu_configure(&s_mpu_cfg) != k_ra8_ok) {
251 while (1) {
252 __asm__ volatile("wfi");
253 }
254 }
255
256 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
257 while (1) {
258 __asm__ volatile("wfi");
259 }
260 }
261
262#ifndef RA8_OFF_TARGET
263 tx_kernel_enter();
264#endif
265
266 while (1) {
267 __asm__ volatile("wfi");
268 }
