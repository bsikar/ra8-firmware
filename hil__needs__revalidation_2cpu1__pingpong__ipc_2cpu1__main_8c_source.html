171static void cpu1_sau_init(void)
172{
173 /* Region 0: peripherals NS alias (0x50000000-0x5FFFFFE0). */
174 *(volatile uint32_t*)k_cpu1_sau_rnr_addr = 0x00000000UL; /* SAU_RNR */
175 *(volatile uint32_t*)k_cpu1_sau_rbar_addr = (uint32_t)k_cpu1_sau_periph_ns_base; /* SAU_RBAR */
176 *(volatile uint32_t*)k_cpu1_sau_rlar_addr =
177 (uint32_t)k_cpu1_sau_periph_ns_limit; /* SAU_RLAR limit | enable */
178 /* Region 1: peripherals S alias (0x40000000-0x4FFFFFE0). */
179 *(volatile uint32_t*)k_cpu1_sau_rnr_addr = 0x00000001UL;
180 *(volatile uint32_t*)k_cpu1_sau_rbar_addr = (uint32_t)k_cpu1_sau_periph_s_base;
181 *(volatile uint32_t*)k_cpu1_sau_rlar_addr = (uint32_t)k_cpu1_sau_periph_s_limit;
182 /* Region 2: NS SRAM range (0x22100000-0x221FFFE0). The CPU1 SRAM bank
183 * (0x22190000-0x2219FFE0) now lives inside this window, so one NS region
184 * covers both the shared markers and the bank -- a separate bank region
185 * would overlap this one, and overlapping SAU regions resolve to Secure,
186 * which the permanent-NS M33 cannot reach. */
187 *(volatile uint32_t*)k_cpu1_sau_rnr_addr = 0x00000002UL;
188 *(volatile uint32_t*)k_cpu1_sau_rbar_addr = (uint32_t)k_cpu1_sau_ns_sram_base;
189 *(volatile uint32_t*)k_cpu1_sau_rlar_addr = (uint32_t)k_cpu1_sau_ns_sram_limit;
190 /* Region 3: MRAM_CPU1 code (0x020C0000-0x020FFFE0). */
191 *(volatile uint32_t*)k_cpu1_sau_rnr_addr = 0x00000003UL;
192 *(volatile uint32_t*)k_cpu1_sau_rbar_addr = (uint32_t)k_cpu1_sau_mram_base;
193 *(volatile uint32_t*)k_cpu1_sau_rlar_addr = (uint32_t)k_cpu1_sau_mram_limit;
194 /* Enable SAU (no ALLNS, so unprogrammed defaults to IDAU). */
195 *(volatile uint32_t*)k_cpu1_sau_ctrl_addr = 0x00000001UL;
196 __asm__ volatile("dsb 0xf" ::: "memory");
197 __asm__ volatile("isb 0xf" ::: "memory");
198 *(volatile uint32_t*)k_cpu1_probe_sau_addr = (uint32_t)k_cpu1_probe_sau_val; /* SAU configured */
199}

201[[noreturn]] static void cpu1_main(void)
202{
203 *(volatile uint32_t*)k_cpu1_probe_main_addr =
204 (uint32_t)k_cpu1_probe_main_val; /* cpu1_main entry */
205
206 cpu1_sau_init();
207
208 while (1) {
209 *(volatile uint32_t*)k_cpu1_probe_iter_addr += 1U; /* loop iter counter */
210 *(volatile uint32_t*)k_cpu1_probe_preread_addr =
211 (uint32_t)k_cpu1_probe_pre_val; /* pre-IPC-read marker */
212 /* TEST: read IPCSEM0 (the simplest IPC reg) first to isolate fault.
213 * IPCSEM0 is at 0x40020000 / NS alias 0x50020000.
214 * IPCSAR.SAIPCSEM0 (bit 0) was set NS too. */
215 const uint32_t sem = *(volatile uint32_t*)k_cpu1_ipcsem0_ns_addr;
216 *(volatile uint32_t*)k_cpu1_probe_sem_addr = sem; /* IPCSEM0 read survived */
217 /* Poll CH2 RX for ping. */
218 const uint32_t sta = *(volatile uint32_t*)(k_cpu1_ipc_ch2_addr + k_cpu1_ipc_off_sta);
219 *(volatile uint32_t*)k_cpu1_probe_sta_addr = sta; /* STA read survived */
220 if ((sta & (uint32_t)k_cpu1_ipc_sta_rdy) == 0U) {
221 continue;
222 }
223 const uint32_t got = *(volatile uint32_t*)(k_cpu1_ipc_ch2_addr + k_cpu1_ipc_off_rxd);
224 *(volatile uint32_t*)k_cpu1_probe_rxd_addr = got; /* most recent rxd */
225 if (got != (uint32_t)k_cpu1_pingpong_magic_ping) {
226 continue;
227 }
228 /* Push pong onto CH0 TX. */
229 *(volatile uint32_t*)(k_cpu1_ipc_ch0_addr + k_cpu1_ipc_off_txd) =
230 (uint32_t)k_cpu1_pingpong_magic_pong;
231 *(volatile uint32_t*)k_cpu1_probe_pong_addr += 1U; /* pong sent counter */
232 }

263[[noreturn]] void cpu1_reset_handler(void)
264{
265 /* CPU1 NS-alias-side marker. The M33 here has SECEXT disabled so it */ /* LEGACY-OK: ARMv8-M bus-architecture term */
266 /* is hardware-locked to the NS controller state; the dedicated SRAM_CPU1
267 * bank at 0x22190000 is its physical alias for these BSS reads, but
268 * CPU0's J-Link memprobe sees the same bytes through the standard
269 * 0x22190000 view. Bench tail: confirm 0xC0DEDEAD before the data
270 * copy starts -- it tells us reset_handler actually ran and the
271 * MRAM_CPU1 fetch worked. */
272 /* Markers placed in NS_SRAM (CPU0 J-Link memprobe can read this view; */ /* LEGACY-OK: ARMv8-M bus-architecture term */
273 /* CPU0's SAU NS_SRAM region 0x22100000-0x221FFFE0 maps NS, and CPU1
274 * as a permanent NS controller can write to the same physical bytes
275 * through the NS alias 0x32100200). The chip's two views of SRAM
276 * see the same backing store. */
277 *(volatile uint32_t*)k_cpu1_probe_reset_addr = (uint32_t)k_cpu1_probe_reset_val;
278 *(volatile uint32_t*)k_cpu1_probe_reset_s_addr = (uint32_t)k_cpu1_probe_reset_val;
279
280 /* Copy .data from MRAM_CPU1 load address into SRAM_CPU1. The linker
281 * defines ``g_ra8_ls_cpu1_data_load`` as LOADADDR(.data) so this
282 * works regardless of the absolute MRAM_CPU1 base. */
283 uint32_t* dst = &g_ra8_ls_cpu1_data_start;
284 uint32_t* src = &g_ra8_ls_cpu1_data_load;
285 while (dst < &g_ra8_ls_cpu1_data_end) {
286 *dst = *src;
287 dst++;
288 src++;
289 }
290 *(volatile uint32_t*)k_cpu1_probe_data_addr =
291 (uint32_t)k_cpu1_probe_data_val; /* survived .data copy */
292
293 /* Zero .bss in SRAM_CPU1. */
294 uint32_t* bss = &g_ra8_ls_cpu1_bss_start;
295 while (bss < &g_ra8_ls_cpu1_bss_end) {
296 *bss = 0U;
297 bss++;
298 }
299 *(volatile uint32_t*)k_cpu1_probe_bss_addr =
300 (uint32_t)k_cpu1_probe_bss_val; /* survived .bss zero */
301
302 cpu1_main();
303}
