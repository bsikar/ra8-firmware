286ra8_err_t ra8_usb_pmsc_feed_cbw(const uint8_t* cbw)
287{
288 RA8_CHECK_NULL_PTR(cbw, s_tag, "feed_cbw: cbw");
289 if (!s_usb_pmsc_state.initialized) {
290 return k_ra8_err_invalid_state;
291 }
292 if (!s_usb_pmsc_state.storage_attached) {
293 return k_ra8_err_invalid_state;
294 }
295
296 /* USB MSC BBB rev 1.0 sec 6.2.1 "Valid CBW" -- signature must be
297 * 'USBC' little-endian. */
298 const uint32_t signature = internal_unpack_u32_le(&cbw[k_ra8_pmsc_cbw_off_signature]);
299 if (signature != k_ra8_pmsc_cbw_signature) {
300 /* Spec sec 6.6.1 "CBW Not Valid": stall both bulk pipes and
301 * await reset recovery. The starter surfaces this as a CSW
302 * with status `phase_error` so the caller's state machine can
303 * proceed deterministically. */
304 s_usb_pmsc_state.bot_state = k_ra8_pmsc_state_csw_tx;
305 s_usb_pmsc_state.cbw_tag = internal_unpack_u32_le(&cbw[k_ra8_pmsc_cbw_off_tag]);
306 return k_ra8_err_invalid_arg;
307 }
308
309 s_usb_pmsc_state.cbw_tag = internal_unpack_u32_le(&cbw[k_ra8_pmsc_cbw_off_tag]);
310 s_usb_pmsc_state.cbw_data_length = internal_unpack_u32_le(&cbw[k_ra8_pmsc_cbw_off_data_length]);
311 s_usb_pmsc_state.cbw_dir_in = (cbw[k_ra8_pmsc_cbw_off_flags] & k_ra8_pmsc_cbw_flag_data_in) != 0U;
312 s_usb_pmsc_state.cbw_lun = (uint8_t)(cbw[k_ra8_pmsc_cbw_off_lun] & k_ra8_pmsc_lun_field_mask);
313 s_usb_pmsc_state.cbw_cdb_len =
314 (uint8_t)(cbw[k_ra8_pmsc_cbw_off_cdb_length] & k_ra8_pmsc_cdb_field_mask);
315 internal_copy_bytes(s_usb_pmsc_state.cbw_cdb,
316 &cbw[k_ra8_pmsc_cbw_off_cdb],
317 (uint32_t)k_ra8_pmsc_cdb_max_len);
318 s_usb_pmsc_state.bot_state = k_ra8_pmsc_state_cdb_decode;
319 s_usb_pmsc_state.last_data_len = 0U;
320 return k_ra8_ok;
321}

344RA8_INTERNAL
345static ra8_err_t internal_dispatch_scsi(uint8_t* data_buf,
346 uint32_t data_buf_capacity,
347 uint32_t* data_len,
348 ra8_usb_pmsc_csw_status_t* csw_status)
349{
350 const uint8_t opcode = s_usb_pmsc_state.cbw_cdb[k_ra8_pmsc_cdb_off_opcode];
351 switch (opcode) {
352 case k_ra8_pmsc_scsi_test_unit_ready:
353 /* Passing -- no data phase. */
354 return k_ra8_ok;
355 case k_ra8_pmsc_scsi_inquiry:
356 return internal_handle_inquiry(data_buf, data_buf_capacity, data_len);
357 case k_ra8_pmsc_scsi_read_capacity_10:
358 return internal_handle_read_capacity(data_buf, data_buf_capacity, data_len);
359 case k_ra8_pmsc_scsi_request_sense:
360 return internal_handle_request_sense(data_buf, data_buf_capacity, data_len);
361 case k_ra8_pmsc_scsi_mode_sense_6:
362 return internal_handle_mode_sense(data_buf, data_buf_capacity, data_len);
363 case k_ra8_pmsc_scsi_read_10:
364 return internal_handle_read10(data_buf, data_buf_capacity, data_len);
365 case k_ra8_pmsc_scsi_write_10:
366 return internal_handle_write10(data_buf, data_len);
367 default:
368 /* Unsupported opcode -> CSW status FAILED, no data. */
369 *csw_status = k_ra8_pmsc_csw_status_failed;
370 return k_ra8_ok;
371 }
372}

431ra8_err_t ra8_usb_pmsc_dispatch_command(uint8_t* data_buf,
432 uint32_t data_buf_capacity,
433 uint32_t* data_len,
434 ra8_usb_pmsc_csw_status_t* csw_status)
435{
436 RA8_CHECK_NULL_PTR(data_buf, s_tag, "dispatch: data_buf");
437 RA8_CHECK_NULL_PTR(data_len, s_tag, "dispatch: data_len");
438 RA8_CHECK_NULL_PTR(csw_status, s_tag, "dispatch: csw_status");
439 const ra8_err_t pre_err = internal_dispatch_preconditions(data_buf_capacity);
440 if (pre_err != k_ra8_ok) {
441 return pre_err;
442 }
443
444 *data_len = 0U;
445 *csw_status = k_ra8_pmsc_csw_status_passed;
446 const ra8_err_t err = internal_dispatch_scsi(data_buf, data_buf_capacity, data_len, csw_status);
447 if (err != k_ra8_ok) {
448 *csw_status = k_ra8_pmsc_csw_status_failed;
449 *data_len = 0U;
450 }
451 internal_dispatch_advance_state(*data_len);
452 return k_ra8_ok;
453}

472ra8_err_t ra8_usb_pmsc_step(void)
473{
474 if (!s_usb_pmsc_state.initialized) {
475 return k_ra8_err_invalid_state;
476 }
477 if (!s_usb_pmsc_state.storage_attached) {
478 return k_ra8_err_invalid_state;
479 }
480
481 /* Each invocation advances by exactly one phase. The IDLE phase
482 * is observable through the public state; in production the
483 * caller pumps `step` from the bulk-OUT-completion ISR until the
484 * machine returns to IDLE waiting for the next CBW. The CBW_RX
485 * and CDB_DECODE phases are caller-driven: the application drains
486 * the bulk-OUT FIFO and invokes `ra8_usb_pmsc_feed_cbw` /
487 * `ra8_usb_pmsc_dispatch_command` directly, so `step` leaves
488 * those phases untouched. */
489 switch (s_usb_pmsc_state.bot_state) {
490 case k_ra8_pmsc_state_idle:
491 /* Awaiting CBW. */
492 s_usb_pmsc_state.bot_state = k_ra8_pmsc_state_cbw_rx;
493 break;
494 case k_ra8_pmsc_state_cbw_rx:
495 case k_ra8_pmsc_state_cdb_decode:
496 /* Caller drives the CBW capture / CDB dispatch directly. */
497 break;
498 case k_ra8_pmsc_state_data_tx:
499 case k_ra8_pmsc_state_data_rx:
500 /* Caller pushes / pulls the data phase; transition to CSW. */
501 s_usb_pmsc_state.bot_state = k_ra8_pmsc_state_csw_tx;
502 break;
503 case k_ra8_pmsc_state_csw_tx:
504 default:
505 /* CSW sent (or unknown state); rewind to IDLE. */
506 s_usb_pmsc_state.bot_state = k_ra8_pmsc_state_idle;
507 break;
508 }
509 return k_ra8_ok;
510}

540ra8_err_t ra8_usb_pmsc_init(ra8_usb_speed_t speed)
541{
542 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
543 return k_ra8_err_invalid_arg;
544 }
545 const ra8_err_t usb_err = ra8_usb_device_init(speed);
546 if (usb_err != k_ra8_ok) {
547 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
548 return k_ra8_err_hw_init_failed;
549 }
550
551 s_usb_pmsc_state.speed = speed;
552 s_usb_pmsc_state.bot_state = k_ra8_pmsc_state_idle;
553 s_usb_pmsc_state.storage_attached = false;
554 s_usb_pmsc_state.storage = (ra8_usb_pmsc_storage_t){};
555 s_usb_pmsc_state.cbw_tag = k_ra8_pmsc_initial_tag;
556 s_usb_pmsc_state.cbw_data_length = 0U;
557 s_usb_pmsc_state.cbw_dir_in = false;
558 s_usb_pmsc_state.cbw_lun = 0U;
559 s_usb_pmsc_state.cbw_cdb_len = 0U;
560 s_usb_pmsc_state.last_data_len = 0U;
561 internal_zero_bytes(s_usb_pmsc_state.cbw_cdb, (uint32_t)k_ra8_pmsc_cdb_max_len);
562 s_usb_pmsc_state.initialized = true;
563
564 const ra8_err_t pipes_err = internal_configure_pipes();
565 if (pipes_err != k_ra8_ok) {
566 (void)ra8_usb_device_deinit(speed);
567 s_usb_pmsc_state.initialized = false;
568 return pipes_err;
569 }
570
571 ra8_log_info_val(s_tag, "device-MSC ready", (uint32_t)speed);
572 return k_ra8_ok;
573}
