133/* `priv_check_mark()`: see header for the documented contract. */
134bool priv_check_mark(ra8_fs_check_ctx_t* ctx, uint32_t cluster)
135{
136 const uint32_t idx = cluster - (uint32_t)k_cluster_first_data;
137 const uint32_t byte = idx >> (uint32_t)k_check_byte_shift;
138 const uint8_t mask = (uint8_t)(1U << (idx & (uint32_t)k_check_bit_mask));
139 if ((ctx->bitmap[byte] & mask) != 0U) {
140 return true;
141 }
142 ctx->bitmap[byte] = (uint8_t)(ctx->bitmap[byte] | mask);
143 return false;
144}

405RA8_INTERNAL
406static ra8_err_t priv_fat_entry(ra8_fs_check_ctx_t* ctx,
407 fat_dir_stack_t* stack,
408 const uint8_t* ent,
409 uint64_t lba,
410 uint32_t entry_off,
411 uint8_t* out_eod)
412{
413 *out_eod = 0U;
414 const uint8_t name0 = ent[k_dir_off_name];
415 const uint8_t attr = ent[k_dir_off_attr];
416 if (name0 == (uint8_t)k_dir_marker_free_perm) {
417 *out_eod = 1U;
418 return k_ra8_ok;
419 }
420 if (name0 == (uint8_t)k_dir_marker_free_used) {
421 return k_ra8_ok;
422 }
423 if (attr == (uint8_t)k_ra8_fs_attr_lfn) {
424 return k_ra8_ok;
425 }
426 if (name0 == (uint8_t)k_dir_marker_dot) {
427 return k_ra8_ok;
428 }
429 const uint8_t is_dir = ((attr & (uint8_t)k_ra8_fs_attr_directory) != 0U) ? 1U : 0U;
430 if ((is_dir == 0U) && ((attr & (uint8_t)k_ra8_fs_attr_volume_id) != 0U)) {
431 return k_ra8_ok; /* a volume-label entry owns no chain */
432 }
433 const uint32_t first = priv_entry_first_cluster(ent);
434 if ((first != 0U) && !priv_check_in_range(ctx, first)) {
435 ctx->rep->entries_bad++;
436 priv_check_fault(ctx, k_ra8_fs_check_fault_bad_dir_entry, first, lba, entry_off);
437 return k_ra8_ok;
438 }
439 if (is_dir != 0U) {
440 if (first != 0U) {
441 priv_fat_push(ctx, stack, first);
442 }
443 return k_ra8_ok;
444 }
445 ctx->rep->files_visited++;
446 if (first != 0U) {
447 return priv_fat_mark_chain(ctx, first);
448 }
449 return k_ra8_ok;
450}

614RA8_INTERNAL
615static ra8_err_t priv_fat_tree(ra8_fs_check_ctx_t* ctx)
616{
617 fat_dir_stack_t stack = {};
618 stack.items[0].is_root = 1U;
619 stack.items[0].cluster = 0U;
620 stack.top = 1U;
621 const bool fat32 = (ctx->m->type == k_ra8_fs_type_fat32);
622 while (stack.top > 0U) {
623 stack.top--;
624 const dir_loc_t loc = stack.items[stack.top];
625 ctx->rep->dirs_visited++;
626 ra8_err_t err = k_ra8_ok;
627 if ((loc.is_root != 0U) && !fat32) {
628 err = priv_fat_scan_fixed_root(ctx, &stack);
629 } else {
630 const uint32_t first = (loc.is_root != 0U) ? ctx->m->root_cluster : loc.cluster;
631 err = priv_fat_scan_cluster_dir(ctx, &stack, first);
632 }
633 if (err != k_ra8_ok) {
634 return err;
635 }
636 }
637 return k_ra8_ok;
638}

661RA8_INTERNAL
662static ra8_err_t priv_fat_diff(ra8_fs_check_ctx_t* ctx)
663{
664 const uint32_t marker = priv_fat_bad_marker(ctx->m);
665 const uint32_t last = (uint32_t)k_cluster_first_data + ctx->rep->clusters_total;
666 for (uint32_t c = (uint32_t)k_cluster_first_data; c < last; c++) {
667 uint32_t v = 0U;
668 const ra8_err_t err = priv_fat_get(ctx->m, c, &v);
669 if (err != k_ra8_ok) {
670 return err;
671 }
672 if ((v == (uint32_t)k_cluster_free) || (v == marker)) {
673 continue;
674 }
675 const uint32_t idx = c - (uint32_t)k_cluster_first_data;
676 const uint8_t mask = (uint8_t)(1U << (idx & (uint32_t)k_check_bit_mask));
677 if ((ctx->bitmap[idx >> (uint32_t)k_check_byte_shift] & mask) == 0U) {
678 ctx->rep->clusters_lost++;
679 priv_check_fault(ctx, k_ra8_fs_check_fault_lost_cluster, c, 0U, 0U);
680 }
681 }
682 return k_ra8_ok;
683}

704RA8_INTERNAL
705static ra8_err_t priv_fat_fsinfo(ra8_fs_check_ctx_t* ctx)
706{
707 if (ctx->m->type != k_ra8_fs_type_fat32) {
708 return k_ra8_ok;
709 }
710 uint8_t* const boot = priv_sec_walk();
711 const ra8_err_t be = priv_read_sector(ctx->m, 0U, boot);
712 if (be != k_ra8_ok) {
713 return be; /* GCOVR_EXCL_LINE -- boot-sector read failure */
714 }
715 const uint32_t lba = (uint32_t)priv_rd16(&boot[k_fmt_off_f32_fsinfo]);
716 if ((lba == 0U) || (lba >= ctx->m->reserved_sectors)) {
717 return k_ra8_ok;
718 }
719 uint8_t* const sec = priv_sec_walk();
720 const ra8_err_t se = priv_read_sector(ctx->m, lba, sec);
721 if (se != k_ra8_ok) {
722 return se; /* GCOVR_EXCL_LINE -- FSInfo-sector read failure */
723 }
724 if (priv_rd32(&sec[k_fmt_fsi_off_lead]) != (uint32_t)k_fmt_fsi_lead_sig) {
725 return k_ra8_ok;
726 }
727 if (priv_rd32(&sec[k_fmt_fsi_off_struct]) != (uint32_t)k_fmt_fsi_struct_sig) {
728 return k_ra8_ok;
729 }
730 if (priv_rd32(&sec[k_fmt_fsi_off_trail]) != (uint32_t)k_fmt_fsi_trail_sig) {
731 return k_ra8_ok;
732 }
733 const uint32_t stored = priv_rd32(&sec[k_fmt_fsi_off_free]);
734 if ((stored != (uint32_t)k_fs_free_unknown) && (stored != ctx->rep->clusters_free)) {
735 priv_check_fault(ctx, k_ra8_fs_check_fault_free_count_bad, 0U, lba, 0U);
736 }
737 return k_ra8_ok;
738}
