56static void mstp_report(void)
57{
58 const uint32_t r = board_mstp_gated_read_count();
59 const uint32_t w = board_mstp_gated_write_count();
60 if ((r == 0U) && (w == 0U)) {
61 return; /* Clean: every peripheral the firmware touched was clocked. */
62 }
63 /* Loud: firmware read/wrote a module-stopped peripheral. On silicon those
64 * reads return 0 and writes vanish -- the exact masked-pass this block ends. */
65 (void)fprintf(stderr,
66 " MSTP-GATING : DROPPED %u read(s)+%u write(s) to module-stopped "
67 "peripheral(s) (last: %s) -- firmware forgot to cancel module-stop\n",
68 r,
69 w,
70 board_mstp_last_gated_name());
71}
