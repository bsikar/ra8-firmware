226[[nodiscard]] static ra8_err_t sdmsc_card_up(uint32_t pclka_hz)
227{
228 const ra8_sdmmc_spi_sci_pins_t pins = {
229 .sck = (ra8_port_pin_t)k_ra8_board_pmod2_spi_sck,
230 .cipo = (ra8_port_pin_t)k_ra8_board_pmod2_spi_cipo,
231 .copi = (ra8_port_pin_t)k_ra8_board_pmod2_spi_copi,
232 .cs = (ra8_port_pin_t)k_ra8_board_pmod2_spi_cs,
233 };
234 ra8_sdmmc_spi_transport_t transport = {};
235
236 ra8_err_t err =
237 ra8_sdmmc_spi_transport_sci((uint8_t)k_sdmsc_sd_spi_channel, pclka_hz, &pins, &transport);
238 if (err != k_ra8_ok) {
239 s_dbg_sd_err = (uint32_t)err;
240 (void)sdmsc_print_fail("sd transport", err);
241 return err;
242 }
243 err = ra8_sdmmc_spi_init(&transport);
244 if (err != k_ra8_ok) {
245 s_dbg_sd_err = (uint32_t)err;
246 (void)sdmsc_print_fail("sd init (card seated?)", err);
247 return err;
248 }
249 uint32_t blocks = 0U;
250 err = ra8_sdmmc_spi_get_capacity(&blocks);
251 if (err != k_ra8_ok) {
252 s_dbg_sd_err = (uint32_t)err;
253 (void)sdmsc_print_fail("sd capacity", err);
254 return err;
255 }
256 s_usb_msc_sdcard_blocks = blocks;
257 s_dbg_sd_err = 0U;
258 return sdmsc_print_card_banner(blocks);
259}

309static void sdmsc_setup_or_halt(void)
310{
311 uint32_t cpuclk0_hz = 0U;
312 uint32_t pclka_hz = 0U;
313 if (ra8_cgc_init() != k_ra8_ok) {
314 sdmsc_panic_halt();
315 }
316 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
317 sdmsc_panic_halt();
318 }
319 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
320 sdmsc_panic_halt();
321 }
322 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
323 sdmsc_panic_halt();
324 }
325 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
326 sdmsc_panic_halt();
327 }
328 if (ra8_board_uart_console_init((uint32_t)k_sdmsc_baud) != k_ra8_ok) {
329 sdmsc_panic_halt();
330 }
331 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
332 sdmsc_panic_halt();
333 }
334 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
335 sdmsc_panic_halt();
336 }
337 /* Enumerate the card and latch its capacity before the kernel starts.
338 * On failure s_dbg_sd_err records it, the FAIL banner is on the wire,
339 * and the worker refuses to attach -- the board does not hang. */
340 (void)sdmsc_card_up(pclka_hz);
341 sdmsc_route_usb_or_halt();
342}
