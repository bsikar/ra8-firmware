175[[nodiscard]] static ra8_err_t aes_demo_one_round_trip(void)
176{
177 /* NOLINTBEGIN(clang-analyzer-optin.core.EnumCastOutOfRange) -- OR-combined PSA usage bits form a valid policy mask outside the enumerator list. */
178 /* The PSA usage enum is intentionally a bitfield -- combining
179 * encrypt + decrypt yields 0x0C, which is a valid policy mask but
180 * not a declared enumerator. */
181 const ra8_psa_key_attr_t attr = {
182 .type = k_ra8_psa_key_type_aes,
183 .alg = k_ra8_psa_alg_aes_gcm,
184 .usage = (ra8_psa_key_usage_t)k_aes_demo_usage_aead,
185 };
186 /* NOLINTEND(clang-analyzer-optin.core.EnumCastOutOfRange) */
187 ra8_psa_key_t key = nullptr;
188 ra8_err_t err = ra8_psa_key_import(&key, &attr, k_aes_demo_key, (size_t)k_aes_demo_key_bytes);
189 if (err != k_ra8_ok) {
190 return err;
191 }
192
193 uint8_t ct[k_aes_demo_plain_bytes + k_ra8_psa_gcm_tag_len] = {};
194 size_t ct_len = 0U;
195 err = ra8_psa_aead_encrypt(key,
196 k_ra8_psa_alg_aes_gcm,
197 k_aes_demo_nonce,
198 (size_t)k_ra8_psa_gcm_nonce_len,
199 k_aes_demo_aad,
200 (size_t)k_aes_demo_aad_bytes,
201 k_aes_demo_plain,
202 (size_t)k_aes_demo_plain_bytes,
203 ct,
204 sizeof(ct),
205 &ct_len);
206 if (err != k_ra8_ok) {
207 (void)ra8_psa_key_destroy(key);
208 return err;
209 }
210
211 uint8_t recovered[k_aes_demo_plain_bytes] = {};
212 size_t rec_len = 0U;
213 err = ra8_psa_aead_decrypt(key,
214 k_ra8_psa_alg_aes_gcm,
215 k_aes_demo_nonce,
216 (size_t)k_ra8_psa_gcm_nonce_len,
217 k_aes_demo_aad,
218 (size_t)k_aes_demo_aad_bytes,
219 ct,
220 ct_len,
221 recovered,
222 sizeof(recovered),
223 &rec_len);
224 (void)ra8_psa_key_destroy(key);
225 if (err != k_ra8_ok) {
226 return err;
227 }
228 if (rec_len != (size_t)k_aes_demo_plain_bytes) {
229 return k_ra8_err_invalid_size;
230 }
231 if (memcmp(recovered, k_aes_demo_plain, (size_t)k_aes_demo_plain_bytes) != 0) {
232 return k_ra8_err_crc_mismatch;
233 }
234 return k_ra8_ok;
235}

239int32_t main(void)
240{
241 aes_demo_setup_or_halt();
242 ra8_isr_globals_enable();
243
244 while (1) {
245 if (aes_demo_one_round_trip() == k_ra8_ok) {
246 (void)ra8_board_uart_console_write(k_aes_demo_msg_ok,
247 (size_t)(sizeof(k_aes_demo_msg_ok) - 1U));
248 (void)ra8_board_led_toggle(k_ra8_board_led1);

250 (void)ra8_board_uart_console_write(k_aes_demo_msg_fail,

252 (void)ra8_board_led_toggle(k_ra8_board_led2);
253 }
254 ra8_delay_ms(k_aes_demo_period_ms);
255 }
256 aes_demo_panic_halt();
257 return 0;
258}
