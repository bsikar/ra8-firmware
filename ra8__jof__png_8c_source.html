70RA8_INTERNAL
71static uint8_t png_paeth(uint8_t a, uint8_t b, uint8_t c)
72{
73 const int32_t p = (int32_t)a + (int32_t)b - (int32_t)c;
74 const int32_t pa = (p > (int32_t)a) ? (p - (int32_t)a) : ((int32_t)a - p);
75 const int32_t pb = (p > (int32_t)b) ? (p - (int32_t)b) : ((int32_t)b - p);
76 const int32_t pc = (p > (int32_t)c) ? (p - (int32_t)c) : ((int32_t)c - p);
77 if ((pa <= pb) && (pa <= pc)) {
78 return a;
79 }
80 if (pb <= pc) {
81 return b;
82 }
83 return c;
84}

106RA8_INTERNAL
107static ra8_err_t
108png_unfilter(uint8_t filter, uint8_t* row, const uint8_t* prev, uint32_t nbytes, uint8_t bpp)
109{
110 switch (filter) {
111 case k_ra8_png_filter_none:
112 return k_ra8_ok;
113 case k_ra8_png_filter_sub:
114 for (uint32_t i = (uint32_t)bpp; i < nbytes; i++) {
115 row[i] = (uint8_t)(row[i] + row[i - bpp]);
116 }
117 return k_ra8_ok;
118 case k_ra8_png_filter_up:
119 for (uint32_t i = 0U; i < nbytes; i++) {
120 row[i] = (uint8_t)(row[i] + prev[i]);
121 }
122 return k_ra8_ok;
123 case k_ra8_png_filter_avg:
124 for (uint32_t i = 0U; i < nbytes; i++) {
125 const uint32_t left = (i >= (uint32_t)bpp) ? (uint32_t)row[i - bpp] : 0U;
126 row[i] = (uint8_t)(row[i] + (uint8_t)((left + (uint32_t)prev[i]) / 2U));
127 }
128 return k_ra8_ok;
129 case k_ra8_png_filter_paeth:
130 for (uint32_t i = 0U; i < nbytes; i++) {
131 const uint8_t a = (i >= (uint32_t)bpp) ? row[i - bpp] : 0U;
132 const uint8_t c = (i >= (uint32_t)bpp) ? prev[i - bpp] : 0U;
133 row[i] = (uint8_t)(row[i] + png_paeth(a, prev[i], c));
134 }
135 return k_ra8_ok;
136 default:
137 return k_ra8_err_validation_failed;
138 }
139}

159RA8_INTERNAL
160static ra8_err_t png_translate(ra8_png_state_t* st, const uint8_t* row)
161{
162 const uint32_t w = (uint32_t)st->w;
163 if (st->color_type == (uint8_t)k_ra8_png_color_pal) {
164 for (uint32_t x = 0U; x < w; x++) {
165 const uint8_t idx = row[x];
166 if ((uint16_t)idx >= st->plte_count) {
167 return k_ra8_err_validation_failed;
168 }
169 const uint32_t src = (uint32_t)idx * (uint32_t)k_ra8_png_ch_3;
170 const uint32_t dst = x * (uint32_t)st->dst_ch;
171 st->xlat[dst] = st->palette[src];
172 st->xlat[dst + 1U] = st->palette[src + 1U];
173 st->xlat[dst + 2U] = st->palette[src + 2U];
174 if (st->dst_ch == (uint8_t)k_ra8_png_ch_4) {
175 st->xlat[dst + 3U] =
176 ((uint16_t)idx < st->trns_count) ? st->trns[idx] : (uint8_t)k_ra8_png_opaque;
177 }
178 }
179 return k_ra8_ok;
180 }
181 if (st->color_type == (uint8_t)k_ra8_png_color_ga) {
182 for (uint32_t x = 0U; x < w; x++) {
183 const uint8_t g = row[(size_t)x * (size_t)k_ra8_png_ch_2];
184 const uint8_t a = row[((size_t)x * (size_t)k_ra8_png_ch_2) + 1U];
185 const uint32_t dst = x * (uint32_t)k_ra8_png_ch_4;
186 st->xlat[dst] = g;
187 st->xlat[dst + 1U] = g;
188 st->xlat[dst + 2U] = g;
189 st->xlat[dst + 3U] = a;
190 }
191 return k_ra8_ok;
192 }
193 /* gray, RGB, RGBA: source layout == output layout. */
194 (void)memcpy(st->xlat, row, (size_t)w * (size_t)st->dst_ch);
195 return k_ra8_ok;
196}

218RA8_INTERNAL
219static ra8_err_t png_consume_rows(ra8_png_state_t* st, const uint8_t* src, uint32_t avail)
220{
221 uint32_t pos = 0U;
222 while (pos < avail) {
223 const uint32_t need = st->rowlen - st->rowfill;
224 const uint32_t left = avail - pos;
225 const uint32_t take = (left < need) ? left : need;
226 (void)memcpy(&st->rowbuf[st->rowfill], &src[pos], (size_t)take);
227 st->rowfill += take;
228 pos += take;
229 if (st->rowfill < st->rowlen) {
230 break; /* partial scanline; wait for more inflate output */
231 }
232 if (st->rows_done >= st->h) {
233 return k_ra8_err_validation_failed; /* more scanlines than IHDR said */
234 }
235 const uint32_t nbytes = st->rowlen - 1U;
236 ra8_err_t err = png_unfilter(st->rowbuf[0], &st->rowbuf[1], st->prevrow, nbytes, st->src_ch);
237 if (err != k_ra8_ok) {
238 return err;
239 }
240 (void)memcpy(st->prevrow, &st->rowbuf[1], (size_t)nbytes);
241 err = png_translate(st, &st->rowbuf[1]);
242 if (err != k_ra8_ok) {
243 return err;
244 }
245 err = st->on_rows(st->cb_ctx, st->xlat, st->w, st->rows_done, 1U, st->dst_ch);
246 if (err != k_ra8_ok) {
247 return err;
248 }
249 st->rows_done++;
250 st->rowfill = 0U;
251 }
252 return k_ra8_ok;
253}

279RA8_INTERNAL
280static ra8_err_t png_bind_geometry(ra8_png_state_t* st, ra8_jof_bump_t* bump)
281{
282 if (st->color_type == (uint8_t)k_ra8_png_color_pal) {
283 if (st->has_plte == 0U) {
284 return k_ra8_err_validation_failed;
285 }
286 st->dst_ch = (st->has_trns != 0U) ? (uint8_t)k_ra8_png_ch_4 : (uint8_t)k_ra8_png_ch_3;
287 } else if (st->color_type == (uint8_t)k_ra8_png_color_gray) {
288 st->dst_ch = (uint8_t)k_ra8_png_ch_1;
289 } else if (st->color_type == (uint8_t)k_ra8_png_color_ga) {
290 st->dst_ch = (uint8_t)k_ra8_png_ch_4;
291 } else {
292 st->dst_ch = st->src_ch; /* RGB -> 3, RGBA -> 4 */
293 }
294 const ra8_err_t err = st->on_geom(st->cb_ctx, st->w, st->h, st->dst_ch);
295 if (err != k_ra8_ok) {
296 return err;
297 }
298 st->rowlen = 1U + ((uint32_t)st->w * (uint32_t)st->src_ch);
299 /* The bump allocator returns 8-byte-aligned void* carves, which satisfies
300 * the inflate context's alignment. */
301 st->inflator = (tinfl_decompressor*)ra8_jof_priv_bump_take(bump, sizeof(tinfl_decompressor));
302 st->ring = ra8_jof_priv_bump_take(bump, (size_t)k_ra8_png_ring_bytes);
303 st->inbuf = ra8_jof_priv_bump_take(bump, (size_t)k_ra8_png_inbuf_bytes);
304 st->rowbuf = ra8_jof_priv_bump_take(bump, (size_t)st->rowlen);
305 st->prevrow = ra8_jof_priv_bump_take(bump, (size_t)(st->rowlen - 1U));
306 st->xlat = ra8_jof_priv_bump_take(bump, (size_t)st->w * (size_t)st->dst_ch);
307 if ((st->inflator == nullptr) || (st->ring == nullptr) || (st->inbuf == nullptr) ||
308 (st->rowbuf == nullptr) || (st->prevrow == nullptr) || (st->xlat == nullptr)) {
309 return k_ra8_err_invalid_size;
310 }
311 (void)memset(st->prevrow, 0, (size_t)(st->rowlen - 1U)); /* PNG sec 9: prior row = zeroes */
312 tinfl_init(st->inflator);
313 return k_ra8_ok;
314}

336static ra8_err_t png_refill_input(ra8_png_state_t* st, uint32_t* out_avail)
337{
338 *out_avail = 0U;
339 for (uint32_t guard = 0U; guard < (uint32_t)k_ra8_png_max_chunks; guard++) {
340 if (st->idat_rem > 0U) {
341 const uint32_t take = (st->idat_rem < (uint32_t)k_ra8_png_inbuf_bytes)
342 ? st->idat_rem
343 : (uint32_t)k_ra8_png_inbuf_bytes;
344 const ra8_err_t err = ra8_jof_png_priv_pull_exact(st, st->inbuf, take);
345 if (err != k_ra8_ok) {
346 return err;
347 }
348 st->idat_rem -= take;
349 *out_avail = take;
350 return k_ra8_ok;
351 }
352 /* Current IDAT exhausted: skip its CRC, look at the next chunk. */
353 ra8_err_t err = ra8_jof_png_priv_skip(st, (uint32_t)k_ra8_png_crc_bytes);
354 if (err != k_ra8_ok) {
355 return err;
356 }
357 uint32_t len = 0U;
358 uint32_t type = 0U;
359 err = ra8_jof_png_priv_chunk_hdr(st, &len, &type);
360 if (err != k_ra8_ok) {
361 return err;
362 }
363 if (type == (uint32_t)k_ra8_png_type_idat) {
364 st->idat_rem = len;
365 continue;
366 }
367 st->pending_len = len;
368 st->pending_type = type;
369 st->pending_valid = 1U;
370 st->source_done = 1U;
371 return k_ra8_ok; /* clean end of the compressed stream */
372 }
373 return k_ra8_err_protocol_error; /* chunk budget exhausted (hostile) */
374}

410RA8_INTERNAL
411static ra8_err_t png_inflate_step(ra8_png_state_t* st, ra8_png_iter_t* it)
412{
413 // mcdc-deactivated: refill-first loop structure; entering with a drained window after the source ended requires the source-ending iteration to return >= TINFL_STATUS_DONE without finishing, i.e. tinfl parked output mid-flush at the exact call the source ended -- for every constructible stream that call either completes (DONE exits the loop) or fails closed at the status check, so the (drained, source-done) entry cannot be flipped independently.
414 if ((it->in_pos == it->in_avail) && (st->source_done == 0U)) {
415 const ra8_err_t err = png_refill_input(st, &it->in_avail);
416 if (err != k_ra8_ok) {
417 return err;
418 }
419 it->in_pos = 0U;
420 }
421 size_t in_sz = (size_t)(it->in_avail - it->in_pos);
422 size_t out_sz = (size_t)((uint32_t)k_ra8_png_ring_bytes - st->ring_wr);
423 const mz_uint flags = (mz_uint)TINFL_FLAG_PARSE_ZLIB_HEADER |
424 ((st->source_done == 0U) ? (mz_uint)TINFL_FLAG_HAS_MORE_INPUT : 0U);
425 const tinfl_status status = tinfl_decompress(st->inflator,
426 &st->inbuf[it->in_pos],
427 &in_sz,
428 st->ring,
429 &st->ring[st->ring_wr],
430 &out_sz,
431 flags);
432 if (status < TINFL_STATUS_DONE) {
433 return k_ra8_err_protocol_error; /* corrupt deflate / bad zlib header */
434 }
435 // mcdc-deactivated: zero-progress stall guard, defense-in-depth against a decompressor that spins without failing; tinfl never returns >= TINFL_STATUS_DONE with neither input consumed nor output produced (a starved mid-stream call without HAS_MORE_INPUT fails closed at the status check above, and the refill-first structure guarantees input is present otherwise), so the stall arm is not constructible from any source stream.
436 if ((in_sz == 0U) && (out_sz == 0U)) {
437 it->stalls++;
438 // mcdc-deactivated: inner arm of the non-constructible stall guard above (same rationale); kept so a hypothetical spinning decompressor aborts after one repeat instead of looping.
439 if ((it->stalls > 1U) || (st->source_done != 0U)) {
440 return k_ra8_err_protocol_error; /* truncated stream / no progress */
441 }
442 } else {
443 it->stalls = 0U;
444 }
445 if (out_sz > 0U) {
446 const ra8_err_t err = png_consume_rows(st, &st->ring[st->ring_wr], (uint32_t)out_sz);
447 if (err != k_ra8_ok) {
448 return err;
449 }
450 st->ring_wr = (st->ring_wr + (uint32_t)out_sz) & ((uint32_t)k_ra8_png_ring_bytes - 1U);
451 }
452 it->in_pos += (uint32_t)in_sz;
453 it->done = (status == TINFL_STATUS_DONE) ? 1U : 0U;
454 return k_ra8_ok;
455}

477RA8_INTERNAL
478static ra8_err_t png_inflate_idat(ra8_png_state_t* st, uint32_t first_len)
479{
480 st->idat_rem = first_len;
481 ra8_png_iter_t it = {};
482 while (it.done == 0U) {
483 const ra8_err_t err = png_inflate_step(st, &it);
484 if (err != k_ra8_ok) {
485 return err;
486 }
487 }
488 /* Strict termination: every declared row arrived, nothing extra. */
489 if ((st->rows_done != st->h) || (st->rowfill != 0U)) {
490 return k_ra8_err_validation_failed;
491 }
492 if ((it.in_pos != it.in_avail) || (st->idat_rem != 0U)) {
493 return k_ra8_err_validation_failed; /* trailing garbage inside IDAT */
494 }
495 return k_ra8_ok;
496}

554static ra8_err_t
555png_walk_chunks(ra8_png_state_t* st, ra8_jof_bump_t* bump, uint16_t max_w, uint16_t max_h)
556{
557 ra8_err_t err = ra8_jof_png_priv_prologue(st, max_w, max_h);
558 if (err != k_ra8_ok) {
559 return err;
560 }
561 for (uint32_t guard = 0U; guard < (uint32_t)k_ra8_png_max_chunks; guard++) {
562 uint32_t len = 0U;
563 uint32_t type = 0U;
564 err = ra8_jof_png_priv_chunk_hdr(st, &len, &type);
565 if (err != k_ra8_ok) {
566 return err;
567 }
568 if (type == (uint32_t)k_ra8_png_type_idat) {
569 return png_run_idat(st, bump, len);
570 }
571 err = ra8_jof_png_priv_pre_idat(st, len, type);
572 if (err != k_ra8_ok) {
573 return err;
574 }
575 }
576 return k_ra8_err_protocol_error; /* chunk budget exhausted (hostile) */
577}
