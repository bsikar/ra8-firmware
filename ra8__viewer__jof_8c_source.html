104RA8_INTERNAL static ra8_err_t viewer_jof_blit(void* ctx,
105 const uint8_t* px,
106 uint16_t sw,
107 uint16_t sh,
108 uint8_t bpp,
109 int32_t dst_x,
110 int32_t dst_y)
111{
112 ra8_viewer_reader_t* r = (ra8_viewer_reader_t*)ctx;
113 const int32_t x_off = r->jof.x_off;
114 const int32_t tw = (int32_t)r->rt_w;
115 const int32_t th = (int32_t)r->rt_h;
116 for (int32_t row = 0; row < (int32_t)sh; ++row) {
117 const int32_t fy = dst_y + row;
118 if ((fy < 0) || (fy >= th)) {
119 continue;
120 }
121 for (int32_t col = 0; col < (int32_t)sw; ++col) {
122 const int32_t fx = dst_x + col + x_off;
123 if ((fx < 0) || (fx >= tw)) {
124 continue;
125 }
126 const uint8_t* sp = &px[(((size_t)row * (size_t)sw) + (size_t)col) * (size_t)bpp];
127 r->rt565[((size_t)fy * (size_t)tw) + (size_t)fx] = viewer_band_pixel(sp, bpp);
128 }
129 }
130 return k_ra8_ok;
131}

152RA8_INTERNAL static ra8_err_t viewer_jof_slurp(ra8_viewer_reader_t* r)
153{
154 /* The whole atlas is held resident in RAM, so a file larger than the per-unit
155 * output cap is refused before allocating -- the same working-set ceiling the
156 * board's SDRAM decode obeys, and the check that keeps `(size_t)file.size`
157 * from truncating a > 4 GiB length on the 32-bit target. */
158 if (r->file.size > r->limits.max_output_bytes) {
159 return k_ra8_err_decomp_output_cap;
160 }
161 const size_t sz = (size_t)r->file.size;
162 uint8_t* buf = (uint8_t*)malloc(sz);
163 if (buf == nullptr) {
164 return k_ra8_err_no_mem;
165 }
166 if (fseeko(r->file.fp, 0, SEEK_SET) != 0) {
167 free(buf);
168 return k_ra8_err_not_found;
169 }
170 if (fread(buf, 1U, sz, r->file.fp) != sz) {
171 free(buf);
172 return k_ra8_err_not_found;
173 }
174 r->jof.atlas = buf;
175 r->jof.store = (ra8_jof_memstore_t){.buf = buf, .cap = sz, .len = sz};
176 return k_ra8_ok;
177}

294RA8_INTERNAL static ra8_err_t viewer_jof_seek(viewer_jof_t* w, uint32_t page)
295{
296 const uint32_t top = page * w->viewport_h;
297 const uint32_t canvas = w->dctx.info.height;
298 const uint32_t remain = (canvas > top) ? (canvas - top) : 0U;
299 const uint32_t content = (remain < w->viewport_h) ? remain : w->viewport_h;
300 if (content == 0U) {
301 return k_ra8_err_out_of_range;
302 }
303 const ra8_err_t vrc =
304 ra8_longstrip_set_viewport(&w->strip, w->dctx.info.width, (uint16_t)content);
305 if (vrc != k_ra8_ok) {
306 return vrc;
307 }
308 const int32_t delta = (int32_t)top - w->strip.scroll_y;
309 return ra8_longstrip_scroll_by(&w->strip, delta);

312ra8_err_t viewer_open_jof(ra8_viewer_reader_t* r)
313{
314 viewer_jof_t* w = &r->jof;
315 ra8_err_t rc = viewer_jof_slurp(r);
316 if (rc != k_ra8_ok) {
317 return rc;
318 }
319
320 w->dctx.pread = ra8_jof_memstore_pread;
321 w->dctx.pread_ctx = &w->store;
322 rc = ra8_jof_parse(ra8_jof_memstore_pread, &w->store, r->file.size, &w->dctx.info);
323 if (rc != k_ra8_ok) {
324 return rc;
325 }
326 const ra8_jof_info_t info = w->dctx.info;
327 /* One decoded band is tile_w * tile_h * bpp bytes. tile_w/tile_h are untrusted
328 * 16-bit header fields that ra8_jof_parse does NOT bound against the image
329 * size, so form the product in 64 bits (u16*u16*u8 cannot wrap) and refuse it
330 * against the policy before narrowing to size_t: an absurd geometry -- e.g.
331 * 65535 x 65535 x 4 ~= 17 GiB -- can neither overflow the allocation size nor
332 * be attempted. */
333 const uint64_t band_bytes64 = (uint64_t)info.tile_w * (uint64_t)info.tile_h * (uint64_t)info.bpp;
334 if (band_bytes64 == 0U) {
335 return k_ra8_err_invalid_size;
336 }
337 rc = ra8_decomp_check_declared(&r->limits, r->file.size, band_bytes64);
338 if (rc != k_ra8_ok) {
339 return rc;
340 }
341 const size_t band_bytes = (size_t)band_bytes64; /* proven <= max_output_bytes */
342 uint32_t cell_count = info.tile_rows;
343 if (cell_count > (uint32_t)k_viewer_jof_max_cells) {
344 cell_count = (uint32_t)k_viewer_jof_max_cells;
345 }
346 if (cell_count == 0U) {
347 cell_count = 1U;
348 }
349 /* Bound the resident cache (cell_count * band_bytes) by the same output cap:
350 * keep at most as many bands resident as the budget admits, always >= 1 since
351 * one band already fits the cap. Fewer resident bands only means more
352 * decode-on-miss churn -- every tile still decodes correctly -- so this bounds
353 * the allocation without ever refusing a valid atlas. */
354 uint32_t max_resident = (uint32_t)(r->limits.max_output_bytes / band_bytes64);
355 if (max_resident == 0U) {
356 max_resident = 1U;
357 }
358 if (cell_count > max_resident) {
359 cell_count = max_resident;
360 }
361 rc = viewer_jof_alloc_cache(w, cell_count, band_bytes);
362 if (rc != k_ra8_ok) {
363 return rc;
364 }
365 return viewer_jof_wire(r, &info, band_bytes, cell_count);
366}

368ra8_err_t viewer_render_jof(ra8_viewer_reader_t* r, uint32_t page)
369{
370 viewer_jof_t* w = &r->jof;
371 /* Headless path: compose into the fixed framebuffer, recentring the
372 * native-width strip and clipping any overhang. */
373 r->rt565 = r->fb;
374 r->rt_w = (uint32_t)k_ra8_viewer_fb_width;
375 r->rt_h = (uint32_t)k_ra8_viewer_fb_height;
376 const int32_t vw = (int32_t)w->dctx.info.width;
377 w->x_off = ((int32_t)r->rt_w > vw) ? (((int32_t)r->rt_w - vw) / 2) : 0;
378 /* Clear to white first: a partial last page keeps a paper-white bottom margin.
379 * 0xFF bytes == 0xFFFF RGB565 == white. */
380 const size_t fb_pixels = (size_t)r->rt_w * (size_t)r->rt_h;
381 memset(r->fb, (int)k_viewer_white_byte, fb_pixels * sizeof(*r->fb));
382
383 const ra8_err_t rc = viewer_jof_seek(w, page);
384 if (rc != k_ra8_ok) {
385 return rc;
386 }
387 ra8_longstrip_render_stats_t st = {};
388 return ra8_longstrip_render(&w->strip, &st);
389}

391ra8_err_t
392viewer_tile_jof(ra8_viewer_reader_t* r, uint32_t i, uint32_t* w, uint32_t* h, uint16_t** out)
393{
394 viewer_jof_t* wt = &r->jof;
395 const uint32_t nw = r->tile_wpx[i];
396 const uint32_t nh = r->tile_hpx[i];
397 if ((nw == 0U) || (nh == 0U)) {
398 return k_ra8_err_invalid_size;
399 }
400 /* nw is the atlas width (bounded to k_ra8_jof_max_dim by ra8_jof_parse) and nh
401 * is one viewport band (<= the fixed framebuffer height), so this product is
402 * bounded by those prior policy checks and cannot wrap size_t. */
403 const size_t px = (size_t)nw * (size_t)nh;
404 uint16_t* buf = (uint16_t*)malloc(px * sizeof(uint16_t));
405 if (buf == nullptr) {
406 return k_ra8_err_no_mem;
407 }
408 /* Point the long-strip blit at this native-width band buffer (restored after). */
409 r->rt565 = buf;
410 r->rt_w = nw;
411 r->rt_h = nh;
412 wt->x_off = 0;
413 memset(buf, (int)k_viewer_white_byte, px * sizeof(uint16_t));
414
415 ra8_err_t rc = viewer_jof_seek(wt, i);
416 if (rc == k_ra8_ok) {
417 ra8_longstrip_render_stats_t st = {};
418 rc = ra8_longstrip_render(&wt->strip, &st);
419 }
420 r->rt565 = r->fb; /* restore the headless default target */
421 r->rt_w = (uint32_t)k_ra8_viewer_fb_width;
422 r->rt_h = (uint32_t)k_ra8_viewer_fb_height;
423 if (rc != k_ra8_ok) {
424 free(buf);
425 return rc;
426 }
427 *w = nw;
428 *h = nh;
429 *out = buf;
430 return k_ra8_ok;
431}
