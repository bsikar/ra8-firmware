143static void internal_pack_u32_be(uint32_t value, uint8_t* dst)
144{
145 dst[0] = (uint8_t)((value >> k_ra8_pmsc_shift_byte3) & k_ra8_pmsc_byte_mask);
146 dst[1] = (uint8_t)((value >> k_ra8_pmsc_shift_byte2) & k_ra8_pmsc_byte_mask);
147 dst[2] = (uint8_t)((value >> k_ra8_pmsc_shift_byte1) & k_ra8_pmsc_byte_mask);
148 dst[3] = (uint8_t)((value >> k_ra8_pmsc_shift_byte0) & k_ra8_pmsc_byte_mask);
149}

202ra8_err_t internal_handle_inquiry(uint8_t* data_buf, uint32_t capacity, uint32_t* out_len)
203{
204 if (capacity < k_ra8_pmsc_inquiry_resp_len) {
205 return k_ra8_err_invalid_size;
206 }
207 internal_zero_bytes(data_buf, (uint32_t)k_ra8_pmsc_inquiry_resp_len);
208
209 data_buf[k_ra8_pmsc_inq_off_dev_type] = k_ra8_pmsc_inq_byte_dev_type;
210 data_buf[k_ra8_pmsc_inq_off_removable] = k_ra8_pmsc_inq_byte_removable;
211 data_buf[k_ra8_pmsc_inq_off_version] = k_ra8_pmsc_inq_byte_version;
212 data_buf[k_ra8_pmsc_inq_off_resp_format] = k_ra8_pmsc_inq_byte_resp_format;
213 data_buf[k_ra8_pmsc_inq_off_addl_length] = k_ra8_pmsc_inq_byte_addl_length;
214
215 /* Pad the three string fields with SPACE first so a backend that
216 * writes fewer bytes still produces a spec-conforming response. */
217 internal_pad_space(&data_buf[k_ra8_pmsc_inq_off_vendor], (uint32_t)k_ra8_pmsc_inq_vendor_len);
218 internal_pad_space(&data_buf[k_ra8_pmsc_inq_off_product], (uint32_t)k_ra8_pmsc_inq_product_len);
219 internal_pad_space(&data_buf[k_ra8_pmsc_inq_off_revision], (uint32_t)k_ra8_pmsc_inq_revision_len);
220
221 const ra8_err_t err =
222 s_usb_pmsc_state.storage.get_inquiry(s_usb_pmsc_state.storage.ctx,
223 &data_buf[k_ra8_pmsc_inq_off_vendor],
224 &data_buf[k_ra8_pmsc_inq_off_product],
225 &data_buf[k_ra8_pmsc_inq_off_revision]);
226 if (err != k_ra8_ok) {
227 return err;
228 }
229 *out_len = (uint32_t)k_ra8_pmsc_inquiry_resp_len;
230 return k_ra8_ok;
231}

233ra8_err_t internal_handle_read_capacity(uint8_t* data_buf, uint32_t capacity, uint32_t* out_len)
234{
235 if (capacity < k_ra8_pmsc_read_capacity_resp_len) {
236 return k_ra8_err_invalid_size;
237 }
238 uint32_t block_count = 0U;
239 uint32_t block_size = 0U;
240 const ra8_err_t err =
241 s_usb_pmsc_state.storage.get_capacity(s_usb_pmsc_state.storage.ctx, &block_count, &block_size);
242 if (err != k_ra8_ok) {
243 return err;
244 }
245 /* Block count of zero is meaningless; degrade gracefully. */
246 const uint32_t last_lba = (block_count == 0U) ? 0U : (block_count - 1U);
247 internal_zero_bytes(data_buf, (uint32_t)k_ra8_pmsc_read_capacity_resp_len);
248 internal_pack_u32_be(last_lba, &data_buf[k_ra8_pmsc_cap_off_last_lba]);
249 internal_pack_u32_be(block_size, &data_buf[k_ra8_pmsc_cap_off_blk_size]);
250 *out_len = (uint32_t)k_ra8_pmsc_read_capacity_resp_len;
251 return k_ra8_ok;
252}

254ra8_err_t internal_handle_request_sense(uint8_t* data_buf, uint32_t capacity, uint32_t* out_len)
255{
256 if (capacity < k_ra8_pmsc_request_sense_resp_len) {
257 return k_ra8_err_invalid_size;
258 }
259 internal_zero_bytes(data_buf, (uint32_t)k_ra8_pmsc_request_sense_resp_len);
260 /* Byte 0 = response code (0x70 current error, valid bit clear). */

264 *out_len = (uint32_t)k_ra8_pmsc_request_sense_resp_len;

280ra8_err_t internal_handle_read10(uint8_t* data_buf, uint32_t capacity, uint32_t* out_len)
281{
282 uint32_t lba = 0U;
283 uint32_t block_count = 0U;
284 internal_decode_rw10(s_usb_pmsc_state.cbw_cdb, &lba, &block_count);
285 if (block_count == 0U) {
286 *out_len = 0U;
287 return k_ra8_ok;
288 }
289 uint32_t device_block_count = 0U;
290 uint32_t block_size = 0U;
291 const ra8_err_t cap_err = s_usb_pmsc_state.storage.get_capacity(s_usb_pmsc_state.storage.ctx,
292 &device_block_count,
293 &block_size);
294 if (cap_err != k_ra8_ok) {
295 return cap_err;
296 }
297 if (block_size == 0U) {
298 block_size = (uint32_t)k_ra8_pmsc_block_size_default;
299 }
300 const uint32_t bytes = block_count * block_size;
301 if (bytes > capacity) {
302 return k_ra8_err_invalid_size;
303 }
304 const ra8_err_t err =
305 s_usb_pmsc_state.storage.read_block(s_usb_pmsc_state.storage.ctx, lba, block_count, data_buf);
306 if (err != k_ra8_ok) {
307 return err;
308 }
309 *out_len = bytes;
310 return k_ra8_ok;
311}
