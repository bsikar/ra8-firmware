268static char* demo_append_byte(char* buf, uint8_t v)
269{
270 /* Tiny base-10 conversion; max 3 digits + NUL. */
271 char tmp[4] = {};
272 uint8_t i = 0U;
273 uint8_t value = v;
274 if (value == 0U) {
275 *buf = '0';
276 return buf + 1;
277 }
278 while (value > 0U && i < (uint8_t)sizeof(tmp)) {
279 tmp[i] = (char)('0' + (char)(value % k_demo_decimal_base));
280 value /= k_demo_decimal_base;
281 i++;
282 }
283 while (i > 0U) {
284 i--;
285 *buf = tmp[i];
286 buf++;
287 }
288 return buf;
289}

335static void demo_log_echo(ULONG n, ULONG peer_ip)
336{
337 char buf[k_demo_log_buf_bytes];
338 char* p = buf;
339 /* "echoed " literal. */
340 static const char k_prefix[] = "[netx] echoed ";
341 for (uint32_t i = 0U; i < (uint32_t)(sizeof(k_prefix) - 1U); i++) {
342 *p = k_prefix[i];
343 p++;
344 }
345 /* Decimal n (clipped to 5 digits which fits k_demo_packet_size). */
346 uint32_t v = (uint32_t)n;
347 if (v == 0U) {
348 *p = '0';
349 p++;
350 } else {
351 char dig[6];
352 uint8_t di = 0U;
353 while (v > 0U && di < (uint8_t)sizeof(dig)) {
354 dig[di] = (char)('0' + (char)(v % (uint32_t)k_demo_decimal_base));
355 v /= (uint32_t)k_demo_decimal_base;
356 di++;
357 }
358 while (di > 0U) {
359 di--;
360 *p = dig[di];
361 p++;
362 }
363 }
364 static const char k_mid[] = " bytes from ";
365 for (uint32_t i = 0U; i < (uint32_t)(sizeof(k_mid) - 1U); i++) {
366 *p = k_mid[i];
367 p++;
368 }
369 uint8_t a = (uint8_t)((peer_ip >> (ULONG)k_demo_ip_shift_a) & k_demo_octet_mask);
370 uint8_t b = (uint8_t)((peer_ip >> (ULONG)k_demo_ip_shift_b) & k_demo_octet_mask);
371 uint8_t c = (uint8_t)((peer_ip >> (ULONG)k_demo_ip_shift_c) & k_demo_octet_mask);
372 uint8_t d = (uint8_t)((peer_ip >> (ULONG)k_demo_ip_shift_d) & k_demo_octet_mask);
373 p = demo_append_byte(p, a);
374 *p = '.';
375 p++;
376 p = demo_append_byte(p, b);
377 *p = '.';
378 p++;
379 p = demo_append_byte(p, c);
380 *p = '.';
381 p++;
382 p = demo_append_byte(p, d);
383 *p = '\r';
384 p++;
385 *p = '\n';
386 p++;
387 *p = '\0';
388 demo_print(buf);
389}

405static UINT demo_netx_create_ip(void)
406{
407 UINT s = nx_packet_pool_create(&s_packet_pool,
408 (CHAR*)"ra8_eth_pool",
409 (ULONG)k_demo_packet_size,
410 s_pool_memory,
411 (ULONG)sizeof(s_pool_memory));
412 if (s != NX_SUCCESS) {
413 return s;
414 }
415
416 /* Hand the MAC to the link driver BEFORE nx_ip_create so it has it
417 * at INITIALIZE time. nx_ip_create runs INITIALIZE synchronously on
418 * the spawned IP thread, well before the main thread can call
419 * nx_ip_interface_physical_address_set; that ordering was the
420 * issue #1 RX-silent symptom on bench. */
421 nx_ether_driver_ra8_eth_set_mac(k_demo_mac);
422
423 ULONG ip_addr = demo_pack_ip(k_demo_ip);
424 ULONG ip_mask = demo_pack_ip(k_demo_mask);
425 s = nx_ip_create(&s_ip,
426 (CHAR*)"ra8_eth_ip",
427 ip_addr,
428 ip_mask,
429 &s_packet_pool,
430 nx_ether_driver_ra8_eth,
431 (VOID*)s_ip_stack,
432 (ULONG)sizeof(s_ip_stack),
433 (UINT)k_demo_ip_thread_pri);
434 if (s != NX_SUCCESS) {
435 return s;
436 }
437
438 /* Belt-and-suspenders: also push the MAC through the NetX API in
439 * case any subsystem reads it from the interface struct later. */
440 ULONG msw = 0U;
441 ULONG lsw = 0U;
442 demo_pack_mac(&msw, &lsw);
443 (void)nx_ip_interface_physical_address_set(&s_ip, 0U, msw, lsw, NX_TRUE);
444 return NX_SUCCESS;
445}

447static UINT demo_netx_bring_up(void)
448{
449 UINT s = demo_netx_create_ip();
450 if (s != NX_SUCCESS) {
451 return s;
452 }
453
454 s = nx_arp_enable(&s_ip, (VOID*)s_arp_cache, (ULONG)sizeof(s_arp_cache));
455 if (s != NX_SUCCESS) {
456 return s;
457 }
458 s = nx_tcp_enable(&s_ip);
459 if (s != NX_SUCCESS) {
460 return s;
461 }
462 (void)nx_icmp_enable(&s_ip);
463
464 /* Enable IP fragmentation so payloads larger than the capped 128-byte
465 * MTU are split into in-spec frames the ESWM egress transmits cleanly
466 * (the silicon corrupts single frames over ~512 bytes). Best-effort:
467 * if NetX was built with NX_DISABLE_FRAGMENTATION, TCP still works via
468 * the MSS clamp; only oversized ICMP/UDP would be affected. */
469 (void)nx_ip_fragment_enable(&s_ip);
470
471 s = nx_tcp_socket_create(&s_ip,
472 &s_echo_socket,
473 (CHAR*)"echo7",
474 NX_IP_NORMAL,
475 NX_FRAGMENT_OKAY,
476 (UINT)k_demo_socket_ttl,
477 (ULONG)k_demo_recv_window,
478 NX_NULL,
479 NX_NULL);
480 if (s != NX_SUCCESS) {
481 return s;
482 }
483
484 s = nx_tcp_server_socket_listen(&s_ip,
485 (UINT)k_demo_echo_port,
486 &s_echo_socket,
487 (UINT)k_demo_packet_count,
488 NX_NULL);
489 return s;
490}

500static void demo_echo_loop(void)
501{
502 while (1) {
503 UINT s = nx_tcp_server_socket_accept(&s_echo_socket, NX_WAIT_FOREVER);
504 if (s != NX_SUCCESS) {
505 demo_print("[netx] accept failed; relistening\r\n");
506 (void)nx_tcp_server_socket_unaccept(&s_echo_socket);
507 (void)nx_tcp_server_socket_relisten(&s_ip, (UINT)k_demo_echo_port, &s_echo_socket);
508 continue;
509 }
510
511 while (1) {
512 NX_PACKET* rx_pkt = NX_NULL;
513 s = nx_tcp_socket_receive(&s_echo_socket, &rx_pkt, NX_WAIT_FOREVER);
514 if (s != NX_SUCCESS || rx_pkt == NX_NULL) {
515 break;
516 }
517 ULONG peer_ip = 0U;
518 ULONG peer_port = 0U;
519 (void)nx_tcp_socket_peer_info_get(&s_echo_socket, &peer_ip, &peer_port);
520 ULONG length = rx_pkt->nx_packet_length;
521 /* Echo the packet straight back. */
522 UINT ts = nx_tcp_socket_send(&s_echo_socket, rx_pkt, NX_WAIT_FOREVER);
523 if (ts != NX_SUCCESS) {
524 (void)nx_packet_release(rx_pkt);
525 }
526 demo_log_echo(length, peer_ip);
527 }
528
529 (void)nx_tcp_socket_disconnect(&s_echo_socket, NX_NO_WAIT);
530 (void)nx_tcp_server_socket_unaccept(&s_echo_socket);
531 (void)nx_tcp_server_socket_relisten(&s_ip, (UINT)k_demo_echo_port, &s_echo_socket);
532 }
533}
