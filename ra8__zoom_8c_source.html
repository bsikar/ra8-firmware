59void ra8_zoom_priv_axis(int32_t anchor,
60 int32_t extent,
61 int32_t scale,
62 int32_t src_dim,
63 ra8_zoom_axis_t* out)
64{
65 const int32_t plane = src_dim * scale;
66 int32_t d0 = -anchor;
67 if (d0 < 0) {
68 d0 = 0;
69 }
70 int32_t d1 = plane - anchor;
71 if (d1 > extent) {
72 d1 = extent;
73 }
74 if (d1 < d0) {
75 d1 = d0;
76 }
77 out->d0 = d0;
78 out->d1 = d1;
79 out->s0 = 0U;
80 /* Beyond this point `anchor + d0 >= 0` by construction, so the source-index
81 * divisions never touch a negative numerator. */
82 if (d1 <= d0) {
83 out->count = 0U;
84 return;
85 }
86 const int32_t s0 = (anchor + d0) / scale;
87 const int32_t s1 = (anchor + d1 - 1) / scale;
88 /* `span` is a named object rather than the composite `(s1 - s0) + 1` inline,
89 * so the widening cast below converts a simple expression: casting the
90 * composite would change essential type mid-expression (MISRA 10.8). It is
91 * positive by construction -- s1 >= s0 whenever d1 > d0. */
92 const int32_t span = (s1 - s0) + 1;
93 out->s0 = (uint32_t)s0;
94 out->count = (uint32_t)span;
95}

169RA8_INTERNAL
170static int32_t zoom_clamp_anchor(int32_t anchor, int32_t plane, int32_t extent)
171{
172 if (plane <= extent) {
173 /* Image smaller than the viewport at this zoom: centre it, letting the
174 * anchor go negative. The render reads that as a letterbox. */
175 return -((extent - plane) / 2);
176 }
177 if (anchor < 0) {
178 return 0;
179 }
180 if (anchor > (plane - extent)) {
181 return plane - extent;
182 }
183 return anchor;
184}

313{

315 if ((cfg->scratch.row_cap < width) || (cfg->scratch.strip_cap < width)) {
316 ra8_log_error(s_tag, "scratch row/strip buffer is narrower than the viewport");

327 ra8_log_error(s_tag, "scratch packed buffer cannot hold one dithered strip");
328 return k_ra8_err_no_mem;

353RA8_INTERNAL
354static ra8_err_t
355zoom_cfg_ladder_ok(const ra8_zoom_view_cfg_t* cfg, uint8_t* out_scale, uint8_t* out_ceiling)
356{
357 /* The "0 means default" rewrite happens first and lifts both values to at
358 * least k_ra8_zoom_scale_min, so a `< min` half in either check below could
359 * never be true. Testing it anyway would be a condition that cannot vary --
360 * unreachable by construction, and dead weight in an MC/DC argument. Each
361 * check is therefore the one bound that can actually fail. */
362 const uint8_t ceiling = (cfg->scale_max == 0U) ? (uint8_t)k_ra8_zoom_scale_max : cfg->scale_max;
363 const uint8_t opening = (cfg->scale == 0U) ? (uint8_t)k_ra8_zoom_scale_min : cfg->scale;
364 if (ceiling > (uint8_t)k_ra8_zoom_scale_max) {
365 ra8_log_error(s_tag, "scale_max is outside the engine ladder");
366 return k_ra8_err_invalid_arg;
367 }
368 if (opening > ceiling) {
369 ra8_log_error(s_tag, "opening scale is outside the configured ladder");
370 return k_ra8_err_invalid_arg;
371 }
372 *out_scale = opening;
373 *out_ceiling = ceiling;
374 return k_ra8_ok;
375}

545uint8_t ra8_zoom_scale_cycle(uint8_t scale, uint8_t min, uint8_t max)
546{
547 const uint8_t floor_v = (min == 0U) ? (uint8_t)k_ra8_zoom_scale_min : min;
548 const uint8_t ceiling_v = (max < floor_v) ? floor_v : max;
549 if (scale >= ceiling_v) {
550 return floor_v;
551 }
552 const uint16_t doubled = (uint16_t)((uint16_t)scale * (uint16_t)k_zoom_scale_dbl);
553 if (doubled >= (uint16_t)ceiling_v) {
554 return ceiling_v;
555 }
556 if (doubled < (uint16_t)floor_v) {
557 return floor_v;
558 }
559 return (uint8_t)doubled;
560}

562ra8_err_t ra8_zoom_view_set_scale(ra8_zoom_view_t* v,
563 uint8_t scale,
564 int32_t focus_x,
565 int32_t focus_y,
566 uint32_t now_ms)
567{
568 RA8_CHECK_NULL_PTR(v, s_tag, "view must not be nullptr");
569 if (!v->active) {
570 ra8_log_error(s_tag, "set_scale on a closed view");
571 return k_ra8_err_invalid_state;
572 }
573 if ((scale < (uint8_t)k_ra8_zoom_scale_min) || (scale > v->scale_max)) {
574 ra8_log_error(s_tag, "requested scale is off the ladder");
575 return k_ra8_err_out_of_range;
576 }
577 if (scale == v->scale) {
578 return k_ra8_ok;
579 }
580 /* Keep the plane point under (focus_x, focus_y) fixed: rescale that plane
581 * coordinate by new/old, then re-derive the anchor from it. The intermediate
582 * product is 64-bit (a plane coordinate times the new scale can exceed int32
583 * before the divide brings it back), and each narrowing cast takes a NAMED
584 * 64-bit object rather than a composite expression -- casting the quotient
585 * in place would change essential type mid-expression (MISRA 10.8). */
586 const int32_t off_x = focus_x - v->dst.x;
587 const int32_t off_y = focus_y - v->dst.y;
588 const int64_t old = (int64_t)v->scale;
589 const int64_t plane_x = (int64_t)(v->anchor_x + off_x) * (int64_t)scale;
590 const int64_t plane_y = (int64_t)(v->anchor_y + off_y) * (int64_t)scale;
591 const int64_t scaled_x = plane_x / old;
592 const int64_t scaled_y = plane_y / old;
593 v->anchor_x = (int32_t)scaled_x - off_x;
594 v->anchor_y = (int32_t)scaled_y - off_y;
595 v->scale = scale;
596 zoom_reclamp(v);
597 zoom_touch(v, now_ms);
598 return k_ra8_ok;
599}

601ra8_err_t ra8_zoom_view_pan(ra8_zoom_view_t* v, int32_t dx, int32_t dy, uint32_t now_ms)
602{
603 RA8_CHECK_NULL_PTR(v, s_tag, "view must not be nullptr");
604 if (!v->active) {
605 ra8_log_error(s_tag, "pan on a closed view");
606 return k_ra8_err_invalid_state;
607 }
608 const int32_t was_x = v->anchor_x;
609 const int32_t was_y = v->anchor_y;
610 v->anchor_x = zoom_sat_add(v->anchor_x, dx);
611 v->anchor_y = zoom_sat_add(v->anchor_y, dy);
612 zoom_reclamp(v);
613 /* Decision: a pan that clamped to no movement owes the panel nothing. */
614 if ((v->anchor_x != was_x) || (v->anchor_y != was_y)) {
615 zoom_touch(v, now_ms);
616 }
617 return k_ra8_ok;
618}

620ra8_err_t ra8_zoom_view_pan_dir(ra8_zoom_view_t* v, ra8_zoom_pan_t dir, uint32_t now_ms)
621{
622 RA8_CHECK_NULL_PTR(v, s_tag, "view must not be nullptr");
623 if (!v->active) {
624 ra8_log_error(s_tag, "pan_dir on a closed view");
625 return k_ra8_err_invalid_state;
626 }
627 const int32_t step_x = v->dst.w - (v->dst.w / (int32_t)k_zoom_overlap_div);
628 const int32_t step_y = v->dst.h - (v->dst.h / (int32_t)k_zoom_overlap_div);
629 int32_t dx = 0;
630 int32_t dy = 0;
631 switch (dir) {
632 case k_ra8_zoom_pan_left:
633 dx = -step_x;
634 break;
635 case k_ra8_zoom_pan_right:
636 dx = step_x;
637 break;
638 case k_ra8_zoom_pan_up:
639 dy = -step_y;
640 break;
641 case k_ra8_zoom_pan_down:
642 dy = step_y;
643 break;
644 case k_ra8_zoom_pan_none:
645 default:
646 break;
647 }
648 return ra8_zoom_view_pan(v, dx, dy, now_ms);
649}

670bool ra8_zoom_view_tick(ra8_zoom_view_t* v, uint32_t now_ms)
671{
672 if (v == nullptr) {
673 return false;
674 }
675 /* Decision: only an open view with an armed settle can promote to quality. */
676 if (!v->active || !v->settle_armed) {
677 return false;
678 }
679 /* Unsigned difference, so a wrapping millisecond counter settles early
680 * rather than never. */
681 if ((now_ms - v->last_input_ms) < (uint32_t)v->settle_ms) {
682 return false;
683 }
684 v->settle_armed = false;
685 v->pending = true;
686 v->pending_kind = k_ra8_zoom_refresh_quality;
687 return true;
688}
