204RA8_INTERNAL
205static ra8_err_t internal_apply_instance(uint8_t instance, const ra8_cnecc_instance_cfg_t* cfg)
206{
207 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C", p 447 */
208 const ra8_err_t mst_err = ra8_mstp_enable(s_cnecc_mstp_table[instance]);
209 RA8_RETURN_ON_ERROR(mst_err, s_tag, "cnecc_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
210
211 volatile r_cnecc_regs_t* reg = ra8_cnecc(instance);
212 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE -- bounded by caller. */
213 return k_ra8_err_hw_init_failed; /* GCOVR_EXCL_LINE */
214 }
215
216 /* HUM Ch 42.2.1 "EC710CTL : ECC Control Register", p 2868
217 * Step 1: clear any latched 1-bit / 2-bit / overflow / address
218 * flags before turning judgment on, so we start from a clean
219 * slate (HUM 42.2.1 p 2870 ECER1C/ECER2C clearing notes). */
220 reg->EC710CTL = k_ra8_cnecc_mask_clear_all;
221
222 /* HUM Ch 42.2.1 "EC710CTL : ECC Control Register", p 2868
223 * Step 2: program IRQ enables + correction permission. The
224 * ``EC1ECP`` bit is "correction NOT executed" when SET, so the
225 * caller's "correct_1bit = true" maps to bit cleared. */
226 uint32_t ctl = 0U;
227 if (cfg->irq_1bit) {
228 ctl |= k_ra8_cnecc_mask_ec1edic;
229 }
230 if (cfg->irq_2bit) {
231 ctl |= k_ra8_cnecc_mask_ec2edic;
232 }
233 if (!cfg->correct_1bit) {
234 ctl |= k_ra8_cnecc_mask_ec1ecp;
235 }
236
237 /* HUM Ch 42.2.1 "EC710CTL : ECC Control Register" EMCA notes,
238 * p 2870: writes to ``ECERVF`` are ignored unless ``EMCA[1:0]``
239 * is ``01b`` in the same write. Combine the unlock pattern with
240 * the ECERVF bit (or omit it for "configured but not running"). */
241 ctl |= k_ra8_cnecc_mask_emca_unlock;
242 if (cfg->enable) {
243 ctl |= k_ra8_cnecc_mask_ecervf;
244 }
245
246 /* HUM Ch 42.2.1 "EC710CTL : ECC Control Register", p 2868 */
247 reg->EC710CTL = ctl;
248
249 /* Always leave fault-injection mode disabled at init -- the test
250 * register defaults to 0 anyway, but a previous test run could
251 * have left ECTMCE set (HUM Ch 42.2.2 "EC710TMC : ECC Test Mode
252 * Control Register", p 2871). */
253 reg->EC710TMC = k_ra8_cnecc_mask_test_disable;
254
255 s_cnecc_one_bit_count[instance] = 0U;
256 s_cnecc_two_bit_count[instance] = 0U;
257 s_cnecc_overflow_count[instance] = 0U;
258 if (s_cnecc_bbr_mirror[instance] != nullptr) {
259 s_cnecc_bbr_mirror[instance]->one_bit_count = 0U;
260 s_cnecc_bbr_mirror[instance]->two_bit_count = 0U;
261 s_cnecc_bbr_mirror[instance]->overflow_count = 0U;
262 }
263 return k_ra8_ok;
264}

291RA8_INTERNAL
292static void internal_ctl_rmw(volatile r_cnecc_regs_t* reg, uint32_t new_bits, uint32_t mask)
293{
294 /* HUM Ch 42.2.1 "EC710CTL : ECC Control Register", p 2868 */
295 uint32_t live = reg->EC710CTL;
296 /* Drop status / W1C bits (we never want to accidentally clear them
297 * on a config rewrite -- callers go through ra8_cnecc_clear_status
298 * for that). */
299 live &= k_ra8_cnecc_mask_ctl_writable;
300 /* Drop the W1C clear bits -- they read 0 but writing 1 latches a
301 * clear; we want this RMW to be neutral on the status flags. */
302 live &= ~(k_ra8_cnecc_mask_ecer1c | k_ra8_cnecc_mask_ecer2c);
303 /* Replace just the requested bits. */
304 live = ((live & ~mask) | (new_bits & mask));
305 /* Always re-assert the EMCA = 01b unlock pattern (HUM 42.2.1 p 2870
306 * EMCA notes). */
307 live = ((live & ~k_ra8_cnecc_mask_emca) | k_ra8_cnecc_mask_emca_unlock);
308 /* HUM Ch 42.2.1 "EC710CTL : ECC Control Register", p 2868 */
309 reg->EC710CTL = live;
310}

327{
328 /* If the ISR was attached, drop those slots first so they don't
329 * fire after the MSTP gate closes. */
330 if (s_cnecc_isr_attached) {
331 (void)ra8_cnecc_detach_isr();

334 for (uint8_t i = 0U; i < (uint8_t)k_ra8_cnecc_instance_count; ++i) {

347 s_cnecc_one_bit_count[i] = 0U;

548[[nodiscard]] ra8_err_t ra8_cnecc_inject_fault(uint8_t instance, const ra8_cnecc_inject_t* req)
549{
550 RA8_CHECK_NULL_PTR(req, s_tag, "req must not be nullptr");
551 if ((uint16_t)instance >= k_ra8_cnecc_instance_count) {
552 return k_ra8_err_invalid_arg;
553 }
554 volatile r_cnecc_regs_t* reg = ra8_cnecc(instance);
555 RA8_CHECK_NULL_PTR(reg, s_tag, "instance mapping failed");
556
557 /* HUM Ch 42.3.2 "ECC Decoder Testing", figure 42.2 p 2875.
558 * The procedure literal values come straight from the figure --
559 * 0x8000 / 0x8080 / 0x8082 -- and re-assert the ETMA = 10b unlock
560 * pattern on every write. */
561
562 /* Step 1: disable test mode (also clears ECDCS per HUM Ch 42.2.2
563 * "EC710TMC : ECC Test Mode Control Register" ECDCS notes, p 2872). */
564 reg->EC710TMC = k_ra8_cnecc_mask_test_disable;
565
566 /* Step 2: write the substitute value into EC710TED. The caller is
567 * responsible for picking a value that differs by 1 or 2 bits
568 * from the genuine MBRAM word at the target address; the
569 * one_bit_flip flag is recorded in the descriptor for traceability
570 * but does not change the register write itself. */
571 /* HUM Ch 42.2.3 "EC710TED : ECC Test Substitute Data Register", p 2872 */
572 reg->EC710TED = req->substitute;
573 (void)req->one_bit_flip; /* Tracked for caller bookkeeping. */
574
575 /* Step 3: enable test mode (ETMA unlock + ECTMCE). */
576 reg->EC710TMC = k_ra8_cnecc_mask_test_enable;
577
578 /* Step 4: select EC710TED as decoder input (sets ECDCS in addition
579 * to ECTMCE). After the next MBRAM read the decoder will see the
580 * substitute value and either ECER1F or ECER2F will latch
581 * depending on the Hamming distance the caller chose. */
582 reg->EC710TMC = k_ra8_cnecc_mask_test_subst;
583 return k_ra8_ok;
584}

655void ra8_cnecc_isr_handler(void* ctx)
656{
657 const uint8_t instance = (uint8_t)((uintptr_t)ctx & (uintptr_t)k_ra8_cnecc_isr_ctx_inst_mask);
658 if ((uint16_t)instance >= k_ra8_cnecc_instance_count) {
659 return; /* GCOVR_EXCL_LINE -- attach packs a valid index. */
660 }
661 volatile r_cnecc_regs_t* reg = ra8_cnecc(instance);
662 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
663 return; /* GCOVR_EXCL_LINE */
664 }
665
666 /* HUM Ch 42.3.1 figure 42.1 p 2874:
667 * - Confirm ECC error flags (ECER1F / ECER2F / ECOVFF).
668 * - Confirm ECC error address flags (ECSEDF0 / ECDEDF0).
669 * - Confirm ECC error address (EC710EAD0).
670 * - Clear ECC error flag (ECER1C = 1, ECER2C = 1). */
671 const uint32_t ctl = reg->EC710CTL;
672 const uint32_t ead = reg->EC710EAD0;
673 const uint16_t addr = (uint16_t)(ead & k_ra8_cnecc_mask_ecead);
674
675 const bool one_bit = (ctl & k_ra8_cnecc_mask_ecer1f) != 0U;
676 const bool two_bit = (ctl & k_ra8_cnecc_mask_ecer2f) != 0U;
677 const bool overflow = (ctl & k_ra8_cnecc_mask_ecovff) != 0U;
678
679 /* Dispatch each kind separately so the host callback sees one
680 * event per latched bit. 2-bit faults take priority for the
681 * "is_2bit" classification when both happen to be set in the
682 * same window. */
683 if (two_bit) {
684 ra8_cnecc_dispatch(instance, true, addr);
685 }
686 if (one_bit) {
687 ra8_cnecc_dispatch(instance, false, addr);
688 }
689 if (overflow) {
690 ra8_cnecc_dispatch_overflow(instance);
691 }
692
693 /* HUM Ch 42.3.1 "ECC Function Setting", p 2874 */
694 reg->EC710CTL = k_ra8_cnecc_mask_clear_all;
695}

698void ra8_cnecc_dispatch(uint8_t instance, bool is_2bit, uint16_t err_addr)
699{
700 if ((uint16_t)instance >= k_ra8_cnecc_instance_count) {
701 return;
702 }
703 if (is_2bit) {
704 ++s_cnecc_two_bit_count[instance];
705 if (s_cnecc_bbr_mirror[instance] != nullptr) {
706 ++s_cnecc_bbr_mirror[instance]->two_bit_count;

708 } else {

710 if (s_cnecc_bbr_mirror[instance] != nullptr) {
711 ++s_cnecc_bbr_mirror[instance]->one_bit_count;
712 }
713 }
714 const ra8_cnecc_error_fn_t fn = s_cnecc_fn;
715 void* const ctx = s_cnecc_ctx;
716 if (fn != nullptr) {
717 fn(ctx, instance, is_2bit, err_addr);
718 }
719}

733/* =============================================================================
734 * Code-flash ECC compute / verify (software fallback)
735 *
736 * The CNECC hardware (HUM Ch 42) only describes runtime read-side ECC for
737 * the CANFD MBRAM. The bootloader's anti-rollback and code-flash auditing
738 * paths still want a deterministic ECC tag for arbitrary memory regions
739 * (e.g. the active firmware slot in MRAM). We expose a small CRC32-style
740 * accumulator so callers can keep using the CNECC vocabulary without
741 * minting a separate driver.
742 * =============================================================================
743 */
744
749typedef enum : uint32_t {
750 k_ra8_cnecc_crc_seed = 0xFFFFFFFFUL,
751 k_ra8_cnecc_crc_xorout = 0xFFFFFFFFUL,
752 k_ra8_cnecc_crc_poly = 0xEDB88320UL,
753} ra8_cnecc_compute_const_t;
754
759typedef enum : uint8_t {
760 k_ra8_cnecc_compute_align = 4U,
761 k_ra8_cnecc_compute_byte_bits = 8U,
762 k_ra8_cnecc_compute_byte_mask = 0xFFU,
763} ra8_cnecc_compute_align_t;
764
783RA8_INTERNAL
784static uint32_t internal_crc32(const uint8_t* data, uint32_t bytes)
785{
786 uint32_t crc = (uint32_t)k_ra8_cnecc_crc_seed;
787 for (uint32_t i = 0U; i < bytes; ++i) {
788 crc ^= (uint32_t)data[i];
789 for (uint8_t b = 0U; b < (uint8_t)k_ra8_cnecc_compute_byte_bits; ++b) {
790 const uint32_t mask = (crc & 1UL) != 0UL ? (uint32_t)k_ra8_cnecc_crc_poly : 0UL;
791 crc = (crc >> 1U) ^ mask;
792 }
793 }
794 return crc ^ (uint32_t)k_ra8_cnecc_crc_xorout;
795}

812[[nodiscard]] ra8_err_t ra8_cnecc_compute(uint32_t addr, uint32_t len, uint32_t* out_ecc)
813{
814 if (out_ecc == nullptr) {
815 return k_ra8_err_null_ptr;
816 }
817 if (addr == 0U) {
818 return k_ra8_err_null_ptr;
819 }
820 if ((addr % (uint32_t)k_ra8_cnecc_compute_align) != 0U) {
821 return k_ra8_err_invalid_arg;
822 }
823 if (len < (uint32_t)k_ra8_cnecc_compute_align) {
824 return k_ra8_err_invalid_arg;
825 }
826 /* Round down to the nearest 4-byte boundary so the computation is
827 * deterministic regardless of trailing bytes. */
828 const uint32_t aligned_len = len & ~((uint32_t)k_ra8_cnecc_compute_align - 1U);
829 *out_ecc = internal_crc32((const uint8_t*)(uintptr_t)addr, aligned_len);
830 ra8_log_info_val(s_tag, "compute ecc", *out_ecc);
831 return k_ra8_ok;
832}
833
834[[nodiscard]] ra8_err_t ra8_cnecc_verify(uint32_t addr, uint32_t len, uint32_t expected_ecc)
835{
836 uint32_t got = 0U;
837 const ra8_err_t err = ra8_cnecc_compute(addr, len, &got);
838 RA8_RETURN_ON_ERROR(err, s_tag, "verify: compute failed"); /* GCOVR_EXCL_BR_LINE */
839 if (got != expected_ecc) {
840 ra8_log_error_val(s_tag, "verify mismatch", got);
841 return k_ra8_err_crc_mismatch;
842 }
843 return k_ra8_ok;
844}
