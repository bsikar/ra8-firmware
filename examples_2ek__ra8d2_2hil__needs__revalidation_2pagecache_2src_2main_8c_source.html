289RA8_INTERNAL static uint32_t internal_pc_crc32(const uint8_t* data, size_t len)
290{
291 uint32_t crc = (uint32_t)k_pc_crc_init;
292 for (size_t i = 0U; i < len; i++) {
293 crc ^= (uint32_t)data[i];
294 for (uint32_t bit = 0U; bit < (uint32_t)k_pc_crc_bits; bit++) {
295 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
296 crc = (crc >> 1U) ^ ((uint32_t)k_pc_crc_poly & mask);
297 }
298 }
299 return ~crc;
300}

700{

764void main(void)
765{
766 uint32_t pclka_hz = 0U;
767 internal_pc_setup_or_halt(&pclka_hz);
768 ra8_isr_globals_enable();
769 ra8_log_init();
770 internal_pc_print(s_msg_boot, (uint32_t)sizeof(s_msg_boot) - 1U);
771
772 internal_pc_init_card_or_halt(&pclka_hz);
773 ra8_fs_mount_t* const mount = internal_pc_mount_or_halt();
774
775 /* Reset-survival: a prior boot's persisted cache file is the hit signal. */
776 g_pc_hit = (internal_pc_read_cache(mount, s_read_blob, (uint32_t)k_pc_blob_cap) != 0U) ? 1U : 0U;
777
778 uint32_t font_len = 0U;
779 internal_pc_font_or_halt(mount, &font_len);
780
781 size_t n = 0U;
782 internal_pc_live_layout_or_halt(font_len, &n);
783 internal_pc_persist_and_verify_or_halt(mount, n);
784 internal_pc_reload_check_or_halt(font_len, n);
785 internal_pc_invalidate_check_or_halt(font_len, n);
786
787 internal_pc_print(s_msg_pass, (uint32_t)sizeof(s_msg_pass) - 1U);
788 internal_pc_print((const uint8_t*)(g_pc_hit != 0U ? "1" : "0"), 1U);
789 internal_pc_print(s_msg_ok, (uint32_t)sizeof(s_msg_ok) - 1U);
790
791 /* Idle with a heartbeat that advances ONLY here -- every failure path parks
792 * in WFI without bumping it -- so a memprobe gate proves the full SD -> ra8_fs
793 * -> reflow_cache round-trip + invalidation ran clean. */
794 while (1) {
795 ra8_delay_ms(k_pc_frame_ms);
796 g_pc_heartbeat++;
797 }
798}
