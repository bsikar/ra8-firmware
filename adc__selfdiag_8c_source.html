215static void
216internal_program_ext_channel(uint8_t vch, uint8_t physical_ch, uint8_t group, bool differential)
217{
218 volatile uint32_t* reg = ra8_adc_b_adchcr(vch);
219 if (reg == nullptr) {
220 return;
221 }
222 const uint32_t cnvcs =
223 ((uint32_t)physical_ch << (uint32_t)k_ra8_adchcr_bit_cnvcs) & k_ra8_adchcr_mask_cnvcs;
224 const uint32_t sgsel =
225 ((uint32_t)group << (uint32_t)k_ra8_adchcr_bit_sgsel) & k_ra8_adchcr_mask_sgsel;
226 const uint32_t ainmd =
227 differential ? (((uint32_t)1UL << (uint32_t)k_ra8_adchcr_bit_ainmd) & k_ra8_adchcr_mask_ainmd)
228 : 0UL;
229 /* HUM Ch 53.2.3.1 "ADCHCRn : A/D Conversion Channel Configuration Register n" p 3335 */
230 *reg = (cnvcs | sgsel | ainmd);
231}

248static void internal_set_data_format(uint8_t vch, uint8_t adprc, uint8_t signsel)
249{
250 volatile uint32_t* reg = ra8_adc_b_addopcrc(vch);
251 if (reg == nullptr) {
252 return;
253 }
254 const uint32_t adprc_field =
255 ((uint32_t)adprc << (uint32_t)k_ra8_addopcrc_bit_adprc) & k_ra8_addopcrc_mask_adprc;
256 const uint32_t sign_field =
257 ((uint32_t)signsel << (uint32_t)k_ra8_addopcrc_bit_signsel) & k_ra8_addopcrc_mask_signsel;
258 /* HUM Ch 53.2.3.4 "ADDOPCRCn : A/D Conversion Data Operation Control C Register n" p 3339 */
259 const uint32_t cleared = *reg & ~(k_ra8_addopcrc_mask_adprc | k_ra8_addopcrc_mask_signsel);
260 *reg = (cleared | adprc_field | sign_field);
261}

281static ra8_err_t internal_start_and_wait(uint8_t group)
282{
283 volatile uint32_t* adstr = ra8_adc_b_adstr(group);
284 if (adstr == nullptr) {
285 return k_ra8_err_out_of_range;
286 }
287 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
288 *adstr = k_ra8_adstr_mask_adst;
289 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
290 for (uint32_t i = 0U; i < k_ra8_adc_busy_wait_limit; ++i) { /* GCOVR_EXCL_BR_LINE */
291 if ((*ra8_adc_b_adsr() & k_ra8_adsr_mask_adact0) == 0U) { /* GCOVR_EXCL_BR_LINE */
292 return k_ra8_ok;
293 }
294 }
295 return k_ra8_err_hw_timeout;

321static ra8_err_t internal_selfdiag_run(uint32_t diagval)
322{
323 internal_program_ext_channel((uint8_t)k_ra8_adc_diag_vchan,
324 (uint8_t)k_ra8_adc_b_chan_selfdiag_adc0,
325 (uint8_t)k_ra8_adc_diag_group,
326 true);
327 internal_set_data_format((uint8_t)k_ra8_adc_diag_vchan,
328 (uint8_t)k_ra8_addopcrc_adprc_16bit,
329 (uint8_t)k_ra8_addopcrc_signsel_signed);
330
331 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
332 *ra8_adc_b_adsger() |= (uint32_t)(1UL << (uint32_t)k_ra8_adc_diag_group);
333
334 /* HUM Ch 53.2.4.1 "ADSGDCRn : Scan Group Diagnosis Function Control Register n" p 3340 */
335 volatile uint32_t* sgdcr = ra8_adc_b_adsgdcr((uint8_t)k_ra8_adc_diag_group);
336 *sgdcr = (*sgdcr & ~k_ra8_adsgdcr_mask_diagval) |
337 ((diagval << (uint32_t)k_ra8_adsgdcr_bit_diagval) & k_ra8_adsgdcr_mask_diagval);
338
339 const ra8_err_t run_err = internal_start_and_wait((uint8_t)k_ra8_adc_diag_group);
340
341 /* Always return DIAGVAL to off so later normal scans on this group are
342 * not stuck in diagnosis mode (HUM Ch 53.2.4.1 p 3340). */
343 /* HUM Ch 53.2.4.1 "ADSGDCRn : Scan Group Diagnosis Function Control Register n" p 3340 */
344 *sgdcr = *sgdcr & ~k_ra8_adsgdcr_mask_diagval;
345
346 return run_err;
347}

349ra8_err_t ra8_adc_self_diagnose(ra8_adc_selfdiag_mode_t mode, uint16_t* out_code, bool* out_pass)
350{
351 RA8_CHECK_NULL_PTR(out_code, s_tag, "out_code must not be nullptr");
352 RA8_CHECK_NULL_PTR(out_pass, s_tag, "out_pass must not be nullptr");
353 *out_code = 0U;
354 *out_pass = false;
355
356 uint32_t diagval = 0U;
357 if (!internal_diagval_for_mode(mode, &diagval)) {
358 return k_ra8_err_invalid_arg;
359 }
360
361 const ra8_err_t run_err = internal_selfdiag_run(diagval);
362 RA8_RETURN_ON_ERROR(run_err, s_tag, "self_diagnose: conversion"); /* GCOVR_EXCL_BR_LINE */
363
364 /* HUM Ch 53.2.13.2 "ADEXDRn : A/D Extended Analog Data Register n" p 3391 */
365 const uint32_t exd =
366 *ra8_adc_b_adexdr((uint8_t)k_ra8_adc_b_chan_selfdiag_adc0 - (uint8_t)k_ra8_adc_b_ext_chan_base);
367 const uint16_t data = (uint16_t)(exd & k_ra8_adexdr_mask_data);
368 const bool err_flag = (exd & k_ra8_adexdr_mask_err) != 0U;
369
370 const int32_t actual = (int32_t)(int16_t)data; /* signed format (SIGNSEL=0). */
371 const int32_t expected = internal_selfdiag_expected(mode);
372 const bool in_band = internal_selfdiag_in_band(actual - expected);
373
374 *out_code = data;
375 *out_pass = (!err_flag) && in_band;
376 return k_ra8_ok;
377}

379ra8_err_t ra8_adc_read_internal_channel(ra8_adc_internal_chan_t chan, uint16_t* out_raw)
380{
381 RA8_CHECK_NULL_PTR(out_raw, s_tag, "out_raw must not be nullptr");
382 *out_raw = 0U;
383 if (!internal_is_supported_ext_chan(chan)) {
384 return k_ra8_err_invalid_arg;
385 }
386
387 /* Single-ended, 12-bit unsigned so the code lines up with the 12-bit
388 * scale used by the TSN two-point calibration (HUM Ch 55.2.2 p 3498). */
389 internal_program_ext_channel((uint8_t)k_ra8_adc_diag_vchan,
390 (uint8_t)chan,
391 (uint8_t)k_ra8_adc_diag_group,
392 false);
393 internal_set_data_format((uint8_t)k_ra8_adc_diag_vchan,
394 (uint8_t)k_ra8_addopcrc_adprc_12bit,
395 (uint8_t)k_ra8_addopcrc_signsel_unsigned);
396
397 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
398 *ra8_adc_b_adsger() |= (uint32_t)(1UL << (uint32_t)k_ra8_adc_diag_group);
399
400 const ra8_err_t run_err = internal_start_and_wait((uint8_t)k_ra8_adc_diag_group);
401 RA8_RETURN_ON_ERROR(run_err, s_tag, "read_internal: conversion"); /* GCOVR_EXCL_BR_LINE */
402
403 const uint8_t idx = ra8_adc_b_adexdr_index_for_chan((uint8_t)chan);
404 volatile uint32_t* exd = ra8_adc_b_adexdr(idx);
405 if (exd == nullptr) {
406 return k_ra8_err_out_of_range;
407 }
408 /* HUM Ch 53.2.13.2 "ADEXDRn : A/D Extended Analog Data Register n" p 3391 */
409 *out_raw = (uint16_t)(*exd & k_ra8_adexdr_mask_data);
410 return k_ra8_ok;
