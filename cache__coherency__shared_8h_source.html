100/* The data-carry invariant the two cores rely on: CPU1 adds the fixed
101 * delta to the ping payload, and that must land exactly on the value
102 * CPU0 expects (pong_base + r == (ping_base + r) + delta). */

104 (uint32_t)k_cache_coherency_pong_base,

107/* The shared struct must start on a 32-byte (cache-line) boundary so an
