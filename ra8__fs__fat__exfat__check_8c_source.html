513static ra8_err_t priv_exchk_set(ra8_fs_check_ctx_t* ctx,
514 exfat_cursor_t* cur,
515 const uint8_t* file,
516 exfat_dir_stack_t* stack)
517{
518 const uint32_t count = 1U + (uint32_t)file[k_exfat_off_file_secnt];
519 const uint32_t bic = (cur->entry_in_cluster - 1U) * (uint32_t)k_exfat_entry_bytes;
520 const uint64_t lba = priv_cluster_to_lba(ctx->m, cur->cluster) + (bic / priv_bps(ctx->m));
521 const uint32_t off = bic % priv_bps(ctx->m);
522 if ((count < 2U) || (count > (uint32_t)k_exfat_set_max_entries)) {
523 ctx->rep->entries_bad++;
524 priv_check_fault(ctx, k_ra8_fs_check_fault_bad_dir_entry, 0U, lba, off);
525 return k_ra8_ok;
526 }
527 uint8_t set[(uint32_t)k_exfat_set_max_entries * (uint32_t)k_exfat_entry_bytes] = {};
528 priv_byte_copy(set, file, (uint32_t)k_exfat_entry_bytes);
529 for (uint32_t k = 1U; k < count; k++) {
530 const ra8_err_t e =
531 priv_exfat_next_entry(ctx->m, cur, &set[(size_t)k * (size_t)k_exfat_entry_bytes]);
532 if (e == k_ra8_err_not_found) {
533 ctx->rep->entries_bad++;
534 priv_check_fault(ctx, k_ra8_fs_check_fault_bad_dir_entry, 0U, lba, off);
535 return k_ra8_ok;
536 }
537 if (e != k_ra8_ok) {
538 return e;
539 }
540 }
541 priv_exchk_verify_set(ctx, set, count, lba, off);
542 return priv_exchk_set_clusters(ctx, set, file, stack, lba, off);

563RA8_INTERNAL
564static ra8_err_t
565priv_exchk_scan_dir(ra8_fs_check_ctx_t* ctx, const exfat_dir_t* dir, exfat_dir_stack_t* stack)
566{
567 exfat_cursor_t cur = {};
568 priv_exfat_cursor_init(dir, &cur);
569 while (cur.scanned < (uint32_t)k_exfat_scan_limit) {
570 uint8_t e[k_exfat_entry_bytes] = {};
571 const ra8_err_t r = priv_exfat_next_entry(ctx->m, &cur, e);
572 if (r == k_ra8_err_not_found) {
573 /* GCOVR_EXCL_START -- a directory run ending exactly on a cluster edge */
574 return k_ra8_ok; /* run ended without an end-of-directory marker */
575 /* GCOVR_EXCL_STOP */
576 }
577 if (r != k_ra8_ok) {
578 return r;
579 }
580 if (e[0] == (uint8_t)k_exfat_entry_eod) {
581 return k_ra8_ok;
582 }
583 if ((e[0] & (uint8_t)k_exfat_inuse_bit) == 0U) {
584 continue; /* a deleted remnant */
585 }
586 if ((e[0] == (uint8_t)k_exfat_entry_bitmap) || (e[0] == (uint8_t)k_exfat_entry_upcase)) {
587 priv_exchk_system_run(ctx, e);
588 continue;
589 }
590 if (e[0] == (uint8_t)k_exfat_entry_file) {
591 const ra8_err_t se = priv_exchk_set(ctx, &cur, e, stack);
592 if (se != k_ra8_ok) {
593 return se;
594 }
595 }
596 }
597 return k_ra8_ok; /* GCOVR_EXCL_LINE -- k_exfat_scan_limit (65536) entries required */
598}

616RA8_INTERNAL
617static ra8_err_t priv_exchk_tree(ra8_fs_check_ctx_t* ctx)
618{
619 /* Explicit DFS worklist (~2 KiB) kept in module-static storage so this frame
620 * stays within the stack-usage budget; the walk is iterative (no recursion)
621 * and single-threaded under the fs lock, so the shared buffer never overlaps. */
622 static exfat_dir_stack_t s_worklist;
623 s_worklist.top = 0U;
624 s_worklist.truncated = 0U;
625 priv_exfat_dir_root(ctx->m, &s_worklist.items[0]);
626 s_worklist.top = 1U;
627 while (s_worklist.top > 0U) {
628 s_worklist.top--;
629 const exfat_dir_t dir = s_worklist.items[s_worklist.top];
630 ctx->rep->dirs_visited++;
631 ra8_err_t err = priv_exchk_mark_dir_alloc(ctx, &dir);
632 if (err != k_ra8_ok) {
633 return err;
634 }
635 err = priv_exchk_scan_dir(ctx, &dir, &s_worklist);
636 if (err != k_ra8_ok) {
637 return err;
638 }
639 }
640 return k_ra8_ok;
641}

670RA8_INTERNAL
671static void priv_exchk_diff_byte(ra8_fs_check_ctx_t* ctx, uint8_t b, uint32_t base)
672{
673 for (uint32_t j = 0U; j < (uint32_t)k_exchk_bits_per_byte; j++) {
674 const uint32_t idx = base + j;
675 if (idx >= ctx->rep->clusters_total) {
676 return;
677 }
678 const uint8_t alloc = (uint8_t)(((uint32_t)b >> j) & 1U);
679 const uint8_t refd =
680 (uint8_t)(((uint32_t)ctx->bitmap[idx >> k_exchk_byte_shift] >> (idx & k_exchk_bit_mask)) &
681 1U);
682 if ((alloc != 0U) && (refd == 0U)) {
683 ctx->rep->clusters_lost++;
684 priv_check_fault(ctx,
685 k_ra8_fs_check_fault_lost_cluster,
686 idx + (uint32_t)k_cluster_first_data,
687 0U,
688 0U);
689 }
690 if ((alloc == 0U) && (refd != 0U)) {
691 ctx->rep->bitmap_mismatches++;
692 priv_check_fault(ctx,
693 k_ra8_fs_check_fault_bitmap_ref_unset,
694 idx + (uint32_t)k_cluster_first_data,
695 0U,
696 0U);
697 }
698 }

724RA8_INTERNAL
725static ra8_err_t priv_exchk_bitmap_pass(ra8_fs_check_ctx_t* ctx, uint64_t bmp_lba)
726{
727 const uint32_t total = ctx->rep->clusters_total;
728 const uint32_t full_bytes = total >> (uint32_t)k_exchk_byte_shift;
729 const uint32_t rem_bits = total & (uint32_t)k_exchk_bit_mask;
730 const uint32_t nbytes = full_bytes + ((rem_bits != 0U) ? 1U : 0U);
731 uint32_t used = 0U;
732 uint64_t loaded = UINT64_MAX;
733 uint8_t* const sec = priv_sec_io();
734 for (uint32_t bi = 0U; bi < nbytes; bi++) {
735 const uint64_t lba = bmp_lba + (bi / priv_bps(ctx->m));
736 if (lba != loaded) {
737 const ra8_err_t err = priv_read_sector(ctx->m, lba, sec);
738 if (err != k_ra8_ok) {
739 return err;
740 }
741 loaded = lba;
742 }
743 uint8_t b = sec[bi % priv_bps(ctx->m)];
744 if (bi == full_bytes) {
745 b = (uint8_t)(b & (uint8_t)((1U << rem_bits) - 1U));
746 }
747 used += priv_exchk_popcount8(b);
748 if (ctx->bitmap != nullptr) {
749 priv_exchk_diff_byte(ctx, b, bi << (uint32_t)k_exchk_byte_shift);
750 }
751 }
752 ctx->rep->clusters_used = used;
753 ctx->rep->clusters_free = total - used;
754 return k_ra8_ok;
755}
