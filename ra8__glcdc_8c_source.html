93/* =============================================================================
94 * RA8D2-specific power + clock bring-up that FSP's R_GLCDC_Open expects to
95 * have been done by the BSP before it touches any GLCDC register. Without
96 * these two steps the entire GLCDC register window at 0x40342000 silently
97 * drops reads and writes (the chip's Graphics power domain is off and
98 * LCDCLK is parked on the default MOCO).
99 *
100 * PDCTRGD.PDDE = 0 power on the Graphics power domain
101 * LCDCKCR = PLL2P LCDCLK = 240 MHz on PLL2P
102 *
103 * Both writes are PRCR-protected (LPM and CGC groups respectively).
104 * =============================================================================
105 */
106

197RA8_INTERNAL
198static void internal_graphics_domain_power_on(void)
199{
200 volatile uint16_t* const prcr = (volatile uint16_t*)k_addr_prcr;
201 volatile uint8_t* const pdctrgd = (volatile uint8_t*)k_addr_pdctrgd;
202 *prcr = (uint16_t)k_prcr_unlock_lpm;
203
204 /* Wait for "fully gated" state before issuing the power-on request. */
205 for (uint32_t i = 0U; i < k_glcdc_clut_size; i++) {
206#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
207 /* Host MMIO fault seam: tests drive the retry / exhaustion legs. */
208 if (ra8_fake_mmio_wait_eval(
209 pdctrgd,
210 i,
211 ((*pdctrgd & (uint8_t)k_pdctrgd_status) == (uint8_t)k_pdctrgd_pdpgsf))) {
212 break;
213 }
214#else
215 if ((*pdctrgd & (uint8_t)k_pdctrgd_status) == (uint8_t)k_pdctrgd_pdpgsf) {
216 break;
217 }
218#endif
219 }
220 *pdctrgd = (uint8_t)k_pdctrgd_on;
221 /* Wait for "fully on" state -- both PDCSF and PDPGSF must clear. */
222 for (uint32_t i = 0U; i < k_glcdc_clut_size; i++) {
223#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
224 if (ra8_fake_mmio_wait_eval(pdctrgd, i, ((*pdctrgd & (uint8_t)k_pdctrgd_status) == 0U))) {
225 break;
226 }
227#else
228 if ((*pdctrgd & (uint8_t)k_pdctrgd_status) == 0U) {
229 break;
230 }
231#endif
232 }
233 *prcr = (uint16_t)k_prcr_relock;
234}

320/* =============================================================================
321 * Bit positions / shifts that FSP's r_glcdc.c uses verbatim. Pulled from
322 * the FSP private header constants ("GLCDC_PRV_*_SHIFT" / "_MASK") -- not
323 * copied as code, just referenced for the register field layout.
324 *
325 * OUT_SET.FORMAT[13:12] output pixel format
326 * OUT_PDTHA.FORM[17:16] dither output format (must match OUT_SET)
327 * FLM6.FORMAT[30:28] framebuffer pixel format
328 * FLM3.LNOFF[31:16] line stride in bytes
329 * FLM5.LNNUM[26:16] lines per frame minus 1
330 * FLM5.DATANUM[15:0] 64-byte AXI bursts per line minus 1
331 * AB3.GRCHS[26:16]/W[10:0] graphics layer H position + width
332 * AB2.GRCVS[26:16]/W[10:0] graphics layer V position + height
333 * AB7.ARCDEF[23:16] constant alpha for alpha-rect
334 * BG_HSIZE.HP[26:16] BG active-pixel start position
335 * BG_VSIZE.VP[26:16] BG active-line start position
336 * BG_SYNC HP / VP min = 1 (HUM 63 p 3760)
337 * =============================================================================
338 */

427RA8_INTERNAL
428static void internal_program_tcon(const ra8_glcdc_panel_timing_t* t)
429{
430 *ra8_glcdc_reg32(k_ra8_glcdc_off_out_clkphase) = 0U; /* all rising edge */
431 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_tim) = 0U;
432
433 /* STVA -> LCD_TCON0 (VSYNC, active-low: INV=1). */
434 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_stva1) = t->v_sync;
435 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_stva2) = (uint32_t)k_glcdc_sel_stva | (1UL << 4);
436
437 /* STHA pulse width, DE -> LCD_TCON2 (active-high: INV=0). */
438 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_stha1) = t->h_sync;
439 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_stha2) = (uint32_t)k_glcdc_sel_de;
440
441 /* STVB -> LCD_TCON1 (HSYNC routed via STVB2.SEL = STHA, active-low). */
442 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_stvb1) =
443 ((uint32_t)(t->v_sync + t->v_back) << k_glcdc_shift_high) | t->v_active;
444 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_stvb2) = (uint32_t)k_glcdc_sel_stha | (1UL << 4);
445
446 /* STHB encodes horizontal DE strobe; TCON3 unused on this board. */
447 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_sthb1) =
448 ((uint32_t)(t->h_sync + t->h_back) << k_glcdc_shift_high) | t->h_active;
449 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_sthb2) = 0U;
450 *ra8_glcdc_reg32(k_ra8_glcdc_off_tcon_de) = 0U;
451}

510RA8_INTERNAL
511static void internal_program_gr1(const ra8_glcdc_panel_timing_t* t,
512 uintptr_t fb_addr,
513 uint16_t fb_format,
514 uint16_t fb_w,
515 uint16_t fb_h)
516{
517 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_fmt) = ((uint32_t)fb_format << k_glcdc_shift_flm6_fmt);
518 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_saddr) = (uint32_t)fb_addr;
519
520 /* FLM3.LNOFF (line stride) + FLM5 (active line / AXI-burst count)
521 * track the FRAMEBUFFER dimensions, not the panel. This lets a
522 * small SRAM framebuffer occupy only a portion of the panel
523 * (rest is filled by the BG plane via BG_BGC). */
524 const uint32_t line_bytes = (uint32_t)fb_w * (uint32_t)k_glcdc_bpp_rgb565;
525 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_flm3) = (line_bytes << k_glcdc_shift_high);
526
527 const uint32_t datanum = (line_bytes / (uint32_t)k_glcdc_axi_burst_bytes) - 1U;
528 const uint32_t lnnum = (uint32_t)fb_h - 1U;
529 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_line) = (lnnum << k_glcdc_shift_high) | datanum;
530
531 /* Layer position on the panel: start at the back-porch corner
532 * (top-left of the active area) so a small framebuffer renders
533 * at the panel's origin and the rest stays BG-plane. */
534 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_size) = (t->h_back << k_glcdc_shift_high) | (uint32_t)fb_w;
535 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_ab2) = (t->v_back << k_glcdc_shift_high) | (uint32_t)fb_h;
536 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_ab4) = (t->v_back << k_glcdc_shift_high) | (uint32_t)fb_h;
537 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_ab5) = (t->h_back << k_glcdc_shift_high) | (uint32_t)fb_w;
538
539 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_ab1) = (uint32_t)k_glcdc_dispsel_transparent;
540 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_flmrd) = 0U;
541 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_ab7) = 0UL;
542}

663/* HUM Ch 63 "Graphics LCD Controller (GLCDC)" p 3744 */
664ra8_err_t ra8_glcdc_init(const ra8_glcdc_config_t* cfg)
665{
666 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
667
668 /* HUM Ch 11.2.14 "PDCTRGD" p 452 + HUM Ch 9.2.58 "LCDCKCR" p 373.
669 * Power on the Graphics domain and route LCDCLK to PLL2P. Without
670 * this the GLCDC peripheral bus is unreachable. */
671 internal_graphics_power_on();
672
673 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C", p 446 */
674 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_glcdc);
675 RA8_RETURN_ON_ERROR(mst_err, s_tag, "glcdc_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
676
677 /* HUM Ch 63 "BG_EN.SWRST" p 3759 + FSP r_glcdc_open step 2. Single
678 * write of SWRST=1. FSP uses a bit-field write (read-modify-write
679 * leaving other bits at their reset values, which are all 0) -- a
680 * plain `*reg = SWRST_MASK` is equivalent on a register that was
681 * just out of reset. */
682 *ra8_glcdc_reg32(k_ra8_glcdc_off_bg_en) = (uint32_t)(1UL << 16);
683
684 /* HUM Ch 63 "SYSCNT_PANEL_CLK" p 3813. CLKSEL=LCDCLK, CLKEN=1,
685 * DCDR=/5 -> 240 MHz / 5 = 48 MHz pixel clock. */
686 enum : uint32_t {
687 k_panel_clk_clksel_lcdclk = 1U << 8,
688 k_panel_clk_clken = 1U << 6,
689 k_panel_clk_dcdr_div2 = 0x02U,
690 };
691 /* LCDCLK = 100 MHz (PLL1R / 4), DCDR /2 -> pixel clock = 50 MHz.
692 * Matches the LVGL EK-RA8D2 reference for the ER-TFT070-6 panel. */
693 *ra8_glcdc_reg32(k_ra8_glcdc_off_panel_clk) = (uint32_t)k_panel_clk_clksel_lcdclk |
694 (uint32_t)k_panel_clk_clken |
695 (uint32_t)k_panel_clk_dcdr_div2;
696
697 /* Programme every other panel-side register in the FSP-documented order. */
698 internal_panel_program(cfg->framebuffer_addr,
699 (uint16_t)cfg->format,
700 cfg->width_px,
701 cfg->height_px,
702 &cfg->timing);
703
704 ra8_log_info_val(s_tag, "glcdc_init fb", cfg->framebuffer_addr);
705 return k_ra8_ok;
706}

708ra8_err_t ra8_glcdc_start(bool enable)
709{
710 enum : uint32_t {
711 k_bit_en = 1U << 0,
712 k_bit_ven = 1U << 8,
713 k_bit_swrst = 1U << 16,
714 };
715
716 if (!enable) {
717 /* Drop everything to disabled but keep SWRST high so register
718 * reads keep working (writing SWRST=0 re-enters reset). */
719 *ra8_glcdc_reg32(k_ra8_glcdc_off_bg_en) = (uint32_t)k_bit_swrst;
720 *ra8_glcdc_reg32(k_ra8_glcdc_off_sys_cfg) = 0U;
721 return k_ra8_ok;
722 }
723
724 /* Match FSP's R_GLCDC_Start exactly: bit-field-style sequential writes
725 * so SWRST stays high while we add VEN then EN. Using read-modify-
726 * write rather than full-word writes avoids momentarily clearing
727 * SWRST or putting the controller in an undefined transition. */
728 uint32_t bg_en = *ra8_glcdc_reg32(k_ra8_glcdc_off_bg_en);
729
730 /* Step 1: VEN = 1 -- enable register value reflection on next VS. */
731 bg_en |= (uint32_t)k_bit_ven;
732 *ra8_glcdc_reg32(k_ra8_glcdc_off_bg_en) = bg_en;
733
734 /* Step 2: EN = 1 -- background plane operation enable. */
735 bg_en |= (uint32_t)k_bit_en;
736 *ra8_glcdc_reg32(k_ra8_glcdc_off_bg_en) = bg_en;
737
738 /* Step 3: SYSCNT.DTCTEN.VPOSDTC = 1 -- line-detect IRQ source. */
739 *ra8_glcdc_reg32(k_ra8_glcdc_off_sys_cfg) = 1U;
740
741 /* Step 4: OUT_VLATCH.VEN = 1 -- commit the OUT_SET / OUT_PDTHA /
742 * brightness / contrast shadow registers to the output stage's
743 * working registers. Without this the output-bus format we wrote
744 * in init never reaches the pixel pipeline. */
745 *ra8_glcdc_reg32(k_ra8_glcdc_off_out_vlatch) = 1U;
746
747 /* Step 5: GR1.VEN + GR2.VEN -- commit graphics-layer shadow regs.
748 *
749 * Both layers MUST be VEN-asserted, even if only one carries a
750 * framebuffer. See bring-up finding in internal_panel_program:
751 * without GR2's shadow registers committed, the output pipeline's
752 * lower-layer slot stays in reset and the panel sees solid black.
753 * FSP r_glcdc_open writes PVEN on every layer it configured. */
754 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr1_en) = 1U;
755 *ra8_glcdc_reg32(k_ra8_glcdc_off_gr2_en) = 1U;
756
757 return k_ra8_ok;
758}
