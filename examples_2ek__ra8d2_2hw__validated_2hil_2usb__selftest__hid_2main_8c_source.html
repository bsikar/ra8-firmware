396void hid_fill_report_body(uint8_t* out, uint32_t len)
397{
398 for (uint32_t i = (uint32_t)k_hid_body_idx; i < len; i++) {
399 const uint32_t v = (i * (uint32_t)k_hid_pat_idx_mul) + (uint32_t)k_hid_pat_bias;
400 out[i] = (uint8_t)(v & (uint32_t)k_hid_byte_mask);
401 }

578static void hid_send_iter(uint32_t* seq)
579{
580 if (_ux_system_slave != UX_NULL) {
581 _ux_system_slave->ux_system_slave_device.ux_slave_device_state =
582 (unsigned long)UX_DEVICE_CONFIGURED;
583 }
584 UX_SLAVE_CLASS_HID_EVENT ev;
585 (void)memset(&ev, 0, sizeof(ev));
586 ev.ux_device_class_hid_event_length = (ULONG)k_hid_report_len;
587 ev.ux_device_class_hid_event_report_id = 0UL;
588 ev.ux_device_class_hid_event_report_type = (ULONG)UX_DEVICE_CLASS_HID_REPORT_TYPE_INPUT;
589 hid_fill_report_body(ev.ux_device_class_hid_event_buffer, (uint32_t)k_hid_report_len);
590 ev.ux_device_class_hid_event_buffer[k_hid_seq_idx] = (UCHAR)(*seq & (uint32_t)k_hid_byte_mask);
591 if (_ux_device_class_hid_event_set(s_hid_class, &ev) == UX_SUCCESS) {
592 (*seq)++;
593 s_dbg_dev_sent++;
594 (void)ra8_board_led_toggle(k_ra8_board_led1);
595 }
596 tx_thread_sleep(1U);
597}

617static VOID hid_device_worker(ULONG arg)
618{
619 (void)arg;
620
621 UINT ux = hid_usbx_stack_up();
622 if (ux != UX_SUCCESS) {
623 s_dbg_dev_err = (uint32_t)ux;
624 return;
625 }
626 s_dbg_dev_step = (uint32_t)k_hid_dev_step_stack;
627 ux = hid_class_register();
628 if (ux != UX_SUCCESS) {
629 s_dbg_dev_err = (uint32_t)ux;
630 return;
631 }
632 s_dbg_dev_step = (uint32_t)k_hid_dev_step_class;
633 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
634 if (e != k_ra8_ok) {
635 s_dbg_dev_err = (uint32_t)e;
636 return;
637 }
638 s_dbg_dev_step = (uint32_t)k_hid_dev_step_dcd;
639 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
640 if (e != k_ra8_ok) {
641 s_dbg_dev_err = (uint32_t)e;
642 return;
643 }
644 s_dbg_dev_step = (uint32_t)k_hid_dev_step_attach;
645
646 uint32_t seq = 0U;
647 while (1) {
648 if (s_hid_class == UX_NULL) {
649 (void)tx_semaphore_get(&s_hid_active_sem, TX_WAIT_FOREVER);
650 continue;
651 }
652 s_dbg_dev_step = (uint32_t)k_hid_dev_step_send;
653 hid_send_iter(&seq);
654 }
655}
