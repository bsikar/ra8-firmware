137[[nodiscard]] static ra8_err_t doc_demo_hw_sum(uint16_t* out_sum)
138{
139 uint16_t acc = k_doc_demo_operands[0];
140 for (uint8_t i = 1U; i < (uint8_t)k_doc_demo_table_len; ++i) {
141 uint16_t partial = 0U;
142 const ra8_err_t err = ra8_doc_add16(acc, k_doc_demo_operands[i], &partial);
143 if (err != k_ra8_ok) {
144 return err;
145 }
146 acc = partial;
147 }
148 *out_sum = acc;

193[[nodiscard]] static ra8_err_t doc_demo_one_iter(uint8_t* out_match)
194{
195 uint16_t hw = 0U;
196 const ra8_err_t err = doc_demo_hw_sum(&hw);
197 if (err != k_ra8_ok) {
198 return err;
199 }
200 const uint16_t sw = doc_demo_sw_sum();
201 *out_match = (hw == sw) ? 1U : 0U;
202 return k_ra8_ok;
203}
