175{
176 volatile uint32_t* mem = (volatile uint32_t*)k_ra8_sdram_base_addr;
177 const uint32_t start_ms = ra8_time_ms();
178 for (uint32_t i = 0U; i < (uint32_t)k_sdram_demo_block_words; i++) {

197static bool sdram_demo_read_check(uint32_t base, uint32_t* elapsed, sdram_demo_diag_t* diag)
198{
199 volatile uint32_t* mem = (volatile uint32_t*)k_ra8_sdram_base_addr;
200 const uint32_t start_ms = ra8_time_ms();
201 diag->count = 0U;
202 for (uint32_t i = 0U; i < (uint32_t)k_sdram_demo_block_words; i++) {
203 const uint32_t got = mem[i];
204 if (got != base + i) {
205 if (diag->count == 0U) {
206 diag->first_idx = i;
207 diag->expected = base + i;
208 diag->got = got;
209 }
210 diag->count++;
211 }
212 }
213 *elapsed = ra8_time_ms() - start_ms;
214 return diag->count == 0U;
215}

228static uint8_t sdram_demo_uint_to_hex(uint8_t* buf, uint32_t v)
229{
230 for (uint8_t i = 0U; i < (uint8_t)k_sdram_demo_hex_digits; i++) {
231 const uint8_t shift =
232 (uint8_t)(((uint8_t)k_sdram_demo_hex_digits - 1U - i) * (uint8_t)k_sdram_demo_nyb_bits);
233 const uint8_t nyb = (uint8_t)((v >> shift) & (uint32_t)k_sdram_demo_nyb_mask);
234 buf[i] = (nyb < (uint8_t)k_sdram_demo_dec_base)
235 ? (uint8_t)((uint8_t)k_sdram_demo_ascii_zero + nyb)
236 : (uint8_t)((uint8_t)k_sdram_demo_ascii_a +
237 (uint8_t)(nyb - (uint8_t)k_sdram_demo_dec_base));
238 }
239 return (uint8_t)k_sdram_demo_hex_digits;
240}

253static uint8_t sdram_demo_uint_to_dec(uint8_t* buf, uint32_t v)
254{
255 uint8_t tmp[k_sdram_demo_uint_dec_max];
256 uint8_t n = 0U;
257 uint32_t r = v;
258 if (r == 0U) {
259 buf[0] = (uint8_t)k_sdram_demo_ascii_zero;
260 return 1U;
261 }
262 while ((r > 0U) && (n < (uint8_t)k_sdram_demo_uint_dec_max)) {
263 tmp[n] =
264 (uint8_t)((uint8_t)k_sdram_demo_ascii_zero + (uint8_t)(r % (uint32_t)k_sdram_demo_dec_base));
265 r = r / (uint32_t)k_sdram_demo_dec_base;
266 n++;
267 }
268 for (uint8_t i = 0U; i < n; i++) {
269 buf[i] = tmp[(uint8_t)(n - 1U - i)];
270 }
271 return n;
272}

287static uint32_t sdram_demo_format_line(uint8_t* out, uint32_t w_mbps, uint32_t r_mbps)
288{
289 uint32_t off = 0U;

291 const uint8_t mid[] = " MB/s r=";
292 const uint8_t suffix[] = " MB/s\r\n";

295 }

298 out[off++] = mid[i];
299 }
300 off += sdram_demo_uint_to_dec(&out[off], r_mbps);
301 for (uint32_t i = 0U; i < sizeof(suffix) - 1U; i++) {
302 out[off++] = suffix[i];
303 }
304 return off;

339 for (uint32_t i = 0U; i < sizeof(got_tag) - 1U; i++) {

341 }
342 off += sdram_demo_uint_to_hex(&out[off], diag->got);
343 for (uint32_t i = 0U; i < sizeof(n_tag) - 1U; i++) {
344 out[off++] = n_tag[i];
345 }
346 off += sdram_demo_uint_to_dec(&out[off], diag->count);
347 for (uint32_t i = 0U; i < sizeof(suffix) - 1U; i++) {
348 out[off++] = suffix[i];
349 }

368int32_t main(void)
369{
370 sdram_demo_setup_or_halt();
371 ra8_isr_globals_enable();
372
373 uint32_t base = (uint32_t)k_sdram_demo_pattern_seed;
374 while (1) {
375 const uint32_t w_ms = sdram_demo_write_pass(base);
376 uint32_t r_ms = 0U;
377 sdram_demo_diag_t diag = {};
378 const bool match = sdram_demo_read_check(base, &r_ms, &diag);
379 const uint32_t w_mbps = sdram_demo_mbps((uint32_t)k_sdram_demo_block_bytes, w_ms);
380 const uint32_t r_mbps = sdram_demo_mbps((uint32_t)k_sdram_demo_block_bytes, r_ms);
381
382 uint8_t out[k_sdram_demo_print_buf] = {};
383 const uint32_t off = sdram_demo_format_line(out, w_mbps, r_mbps);
384 if (!match) {
385 (void)ra8_board_led_on(k_ra8_board_led2);
386 /* Emit a FAIL diagnostic so the HIL negative regex catches a
387 * read-back mismatch and a log reader sees the corruption
388 * signature (first index, expected vs got, total count). */
389 uint8_t fail[k_sdram_demo_fail_buf] = {};
390 const uint32_t fail_off = sdram_demo_format_fail(fail, &diag);
391 (void)ra8_board_uart_console_write(fail, (size_t)fail_off);
392 }
393 if (ra8_board_uart_console_write(out, (size_t)off) != k_ra8_ok) {
394 break;
395 }
396 /* Success-only verdict for the HIL scrape: a clean read-back pass. */
397 if (match) {
398 (void)ra8_board_uart_console_write(k_sdram_demo_pass_msg,
399 (size_t)(sizeof(k_sdram_demo_pass_msg) - 1U));
400 }
401 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
402 break;
403 }
404 base += 1U;
405 ra8_delay_ms(k_sdram_demo_period_ms);
406 }
407
408 sdram_demo_panic_halt();
409 return 0;
410}
