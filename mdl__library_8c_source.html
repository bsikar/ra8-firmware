47RA8_INTERNAL static bool
48visit_entry(const char* out_dir, const char* name, mdl_library_fn cb, void* ctx, ra8_err_t* rc)
49{
50 char series[PATH_MAX];
51 char state[PATH_MAX];
52 const int sn = snprintf(series, sizeof(series), "%s/%s", out_dir, name);
53 const int tn = snprintf(state, sizeof(state), "%s/%s/%s", out_dir, name, state_basename());
54 if ((sn < 0) || ((size_t)sn >= sizeof(series)) || (tn < 0) || ((size_t)tn >= sizeof(state))) {
55 return true; /* path too long: skip, keep walking */
56 }
57 struct stat st;
58 if ((stat(series, &st) != 0) || !S_ISDIR(st.st_mode) || !is_file(state)) {
59 return true; /* not a tracked-series directory */
60 }
61 *rc = cb(series, state, ctx);
62 return (*rc == k_ra8_ok);
63}

65ra8_err_t mdl_library_for_each(const char* out_dir, mdl_library_fn cb, void* ctx)
66{
67 if ((out_dir == nullptr) || (cb == nullptr)) {
68 return k_ra8_err_invalid_arg;
69 }
70 DIR* dir = opendir(out_dir);
71 if (dir == nullptr) {
72 return k_ra8_ok; /* an absent library tracks nothing */
73 }
74 ra8_err_t rc = k_ra8_ok;
75 const struct dirent* ent;
76 uint32_t seen = 0U;
77 while (((ent = readdir(dir)) != nullptr) && (seen < (uint32_t)k_lib_max_series)) {
78 ++seen;
79 if (!is_real_entry(ent->d_name)) {
80 continue;
81 }
82 if (!visit_entry(out_dir, ent->d_name, cb, ctx, &rc)) {
83 break; /* callback asked to stop */
84 }
85 }
86 (void)closedir(dir);
87 return rc;
88}

100ra8_err_t mdl_library_remove_tree(const char* dir)
101{
102 if ((dir == nullptr) || (dir[0] == '\0')) {
103 return k_ra8_err_invalid_arg;
104 }
105 struct stat st;
106 if (stat(dir, &st) != 0) {
107 return k_ra8_ok; /* already gone */
108 }
109 /* FTW_DEPTH: children before parents; FTW_PHYS: never follow a symlink out. */
110 if (nftw(dir, rm_entry, (int)k_lib_nftw_max_fds, FTW_DEPTH | FTW_PHYS) != 0) {
111 return k_ra8_fail;
112 }
113 return k_ra8_ok;
114}
