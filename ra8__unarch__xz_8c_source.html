78bool ra8_unarch_xz_magic(const uint8_t* sig, size_t sig_len)
79{
80 if (sig == nullptr) {
81 return false;
82 }
83 if (sig_len < (size_t)k_ra8_unarch_xz_sig_len) {
84 return false;
85 }
86 const uint8_t want[k_ra8_unarch_xz_sig_len] = {
87 (uint8_t)k_xz_magic_b0,
88 (uint8_t)k_xz_magic_b1,
89 (uint8_t)k_xz_magic_b2,
90 (uint8_t)k_xz_magic_b3,
91 (uint8_t)k_xz_magic_b4,
92 (uint8_t)k_xz_magic_b5,
93 };
94 return memeq(sig, want, sizeof(want));
95}

98ra8_unarch_xz_stream_begin(ra8_unarch_xz_stream_t* xs, void* scratch, uint32_t scratch_len)
99{
100 RA8_CHECK_NULL_PTR(xs, s_tag_xz, "begin: null session");
101 *xs = (ra8_unarch_xz_stream_t){};
102 RA8_CHECK_NULL_PTR(scratch, s_tag_xz, "begin: null scratch");
103 if (scratch_len <= (uint32_t)k_ra8_unarch_xz_state_reserve) {
104 return k_ra8_err_invalid_size;
105 }
106 const ra8_err_t perr = ra8_unarch_xz_pool_install(scratch, scratch_len);
107 if (perr != k_ra8_ok) {
108 return perr;
109 }
110 /* The CRC lookup tables must exist before any other xz_* call; both
111 * initialisers are idempotent, so calling per-session is safe. */
112 xz_crc32_init();
113 xz_crc64_init();
114 const uint32_t dict_max = scratch_len - (uint32_t)k_ra8_unarch_xz_state_reserve;
115 xs->dec = xz_dec_init(XZ_PREALLOC, dict_max);
116 if (xs->dec == nullptr) {
117 ra8_unarch_xz_pool_reset();
118 return k_ra8_err_no_mem;
119 }
120 xs->live = true;
121 return k_ra8_ok;
122}

171 const uint8_t* in,
172 size_t in_len,
173 size_t* in_used,
174 uint8_t* out,
175 size_t out_cap,
176 size_t* out_used,
177 bool* end)
178{
179 RA8_CHECK_NULL_PTR(xs, s_tag_xz, "run: null session");
180 RA8_CHECK_NULL_PTR(in_used, s_tag_xz, "run: null in_used");
181 RA8_CHECK_NULL_PTR(out, s_tag_xz, "run: null out");
182 RA8_CHECK_NULL_PTR(out_used, s_tag_xz, "run: null out_used");
183 RA8_CHECK_NULL_PTR(end, s_tag_xz, "run: null end");
184 *in_used = 0U;
185 *out_used = 0U;
186 *end = false;
187 if (in_len > 0U) {
188 RA8_CHECK_NULL_PTR(in, s_tag_xz, "run: null in with input");
189 }
190 if (!xs->live) {
191 return k_ra8_err_invalid_state;
192 }
193 struct xz_buf b = {
194 .in = in,
195 .in_pos = 0U,
196 .in_size = in_len,
197 .out = out,
198 .out_pos = 0U,
199 .out_size = out_cap,
200 };
201 const enum xz_ret ret = xz_dec_run(xs->dec, &b);
202 *in_used = b.in_pos;
203 *out_used = b.out_pos;
204 if (ret == XZ_STREAM_END) {
205 *end = true;
206 return k_ra8_ok;
207 }
208 if (ret == XZ_OK) {
209 return k_ra8_ok;
210 }
211 return s_xz_map_err(ret);
212}

214void ra8_unarch_xz_stream_end(ra8_unarch_xz_stream_t* xs)
215{
216 if (xs == nullptr) {
217 return; /* teardown paths call unconditionally */
218 }
219 if (!xs->live) {
220 return; /* never begun (or already ended): nothing to release */
221 }
222 xz_dec_end(xs->dec); /* frees are no-ops; the pool reset reclaims all */
223 ra8_unarch_xz_pool_reset();
224 xs->dec = nullptr;
225 xs->live = false;
226}

305RA8_INTERNAL
306static ra8_err_t s_unwrap_pass(xz_unwrap_state_t* st, bool* end)
307{
308 const ra8_err_t ierr = ra8_decomp_budget_charge_iter(&st->budget);
309 if (ierr != k_ra8_ok) {
310 return ierr;
311 }
312 const uint64_t remain = st->size - st->in_off;
313 const size_t want = (remain > (uint64_t)k_xz_chunk) ? (size_t)k_xz_chunk : (size_t)remain;
314 size_t got = 0U;
315 if (want > 0U) {
316 got = st->read(st->ctx, st->in_off, st->chunk, want);
317 }
318 if (got > want) {
319 got = want; /* clamp a misbehaving backing to the request */
320 }
321 if (got == 0U) {
322 return k_ra8_err_validation_failed; /* backing truncated under the stream */
323 }
324 const size_t space = st->out_cap - st->total_out;
325 const size_t owin = (space > (size_t)k_xz_out_window) ? (size_t)k_xz_out_window : space;
326 size_t used = 0U;
327 size_t produced = 0U;
328 const ra8_err_t rerr = ra8_unarch_xz_stream_run(&st->xs,
329 st->chunk,
330 got,
331 &used,
332 &st->out[st->total_out],
333 owin,
334 &produced,
335 end);
336 st->in_off += (uint64_t)used;
337 st->total_out += produced;
338 if (rerr != k_ra8_ok) {
339 if (rerr == k_ra8_err_validation_failed) {
340 if (st->total_out == st->out_cap) {
341 return k_ra8_err_no_mem; /* stuck only because the arena is full */
342 }
343 }
344 return rerr;
345 }
346 return ra8_decomp_budget_charge_output(&st->budget, st->in_off, (uint64_t)produced);
347}

349ra8_err_t ra8_unarch_xz_unwrap(ra8_unarch_read_fn read,
350 void* ctx,
351 uint64_t size,
352 uint8_t* out,
353 size_t out_cap,
354 void* scratch,
355 uint32_t scratch_len,
356 const ra8_decomp_limits_t* limits,
357 size_t* out_len)
358{
359 const ra8_err_t nerr = s_unwrap_reject_null(read, out, scratch, out_len);
360 if (nerr != k_ra8_ok) {
361 return nerr;
362 }
363 *out_len = 0U;
364 if (size == 0U) {
365 return k_ra8_err_invalid_size;
366 }
367 if (out_cap == 0U) {
368 return k_ra8_err_invalid_size;
369 }
370 xz_unwrap_state_t st = {};
371 st.read = read;
372 st.ctx = ctx;
373 st.size = size;
374 st.out = out;
375 st.out_cap = out_cap;
376 const ra8_err_t berr = ra8_decomp_budget_init(&st.budget, limits);
377 if (berr != k_ra8_ok) {
378 return berr;
379 }
380 const ra8_err_t serr = ra8_unarch_xz_stream_begin(&st.xs, scratch, scratch_len);
381 if (serr != k_ra8_ok) {
382 return serr;
383 }
384 bool end = false;
385 ra8_err_t perr = k_ra8_ok;
386 while (!end) { /* bound: every pass charges the policy iteration budget */
387 perr = s_unwrap_pass(&st, &end);
388 if (perr != k_ra8_ok) {
389 break;
390 }
391 }
392 ra8_unarch_xz_stream_end(&st.xs);
393 if (perr != k_ra8_ok) {
394 return perr;
395 }
396 if (st.in_off < size) {
397 return k_ra8_err_validation_failed; /* trailing bytes after the footer */
398 }
399 *out_len = st.total_out;
400 return k_ra8_ok;
401}
