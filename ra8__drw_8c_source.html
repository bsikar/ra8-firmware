491[[nodiscard]] ra8_err_t ra8_drw_init(const ra8_drw_config_t* cfg)
492{
493 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
494
495 /* The DRW lives in the GRAPHICS POWER DOMAIN, which is gated OFF out of
496 * reset (PDCTRGD = 0x81: PDPGSF = 1, PDDE = 1 -- HUM Ch 11.2.14 p 452;
497 * domain contents in HUM Ch 11.5.1 Table 11.7 p 480). Cancelling
498 * module-stop is NOT sufficient: with the domain dark every DRW register
499 * reads 0 and every write is lost, which is exactly why the engine never
500 * rasterised a pixel on silicon (#247) while ra8_emulator -- modelling no
501 * power domain -- rendered happily. Power the domain first.
502 * Bench evidence: HWREVISION reads 0x00000000 before this call and
503 * 0x0FBE0107 after it. */
504 const ra8_err_t pd_err = ra8_lpm_graphics_power_on((uint32_t)k_ra8_lpm_pd_timeout_default);
505 RA8_RETURN_ON_ERROR(pd_err, s_tag, "drw_init: graphics power"); /* GCOVR_EXCL_BR_LINE */
506
507 /* HUM Ch 11.2.8 "MSTPCRC: Module Stop Control Register C", p 446 */
508 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_drw);
509 RA8_RETURN_ON_ERROR(mst_err, s_tag, "drw_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
510
511 /* HUM Ch 62.2.3 "IRQCTL: Interrupt Control Register", p 3693 */
512 /* Clear all pending IRQs and leave them masked. */
513 *ra8_drw_reg32(k_ra8_drw_off_irqctl) = k_ra8_drw_irqctl_all_clr;
514
515 /* HUM Ch 62.2.31 "ORIGIN: Framebuffer Base Address Register", p 3705 */
516 /* Cache the framebuffer base: writing ORIGIN is the render TRIGGER, so
517 * every primitive re-writes it as its final step (see ra8_drw_draw.c).
518 * This init write fires a trigger too, but the limiters/SIZE are still
519 * zero so no pixel is produced. */
520 s_drw_origin = (uint32_t)cfg->framebuffer_addr;
521 *ra8_drw_reg32(k_ra8_drw_off_origin) = s_drw_origin;
522
523 s_drw_pitch_px = (uint32_t)cfg->pitch_px;
524 s_drw_bytes_px = internal_bytes_per_px(cfg->format);
525
526 /* HUM Ch 62.2.30 "PITCH: Framebuffer Pitch and Spanstore Delay
527 * Register", p 3705. Spanstore delay (SSD) field [31:16] kept 0. */
528 *ra8_drw_reg32(k_ra8_drw_off_pitch) = (uint32_t)cfg->pitch_px;
529
530 /* HUM Ch 62.2.2 "CONTROL2: Surface Control Register", p 3690 */
531 s_drw_control2 = internal_pack_surface_defaults(cfg->format);
532 *ra8_drw_reg32(k_ra8_drw_off_control2) = s_drw_control2;
533 s_drw_color1 = 0U;
534
535 internal_drw_program_cache_options(cfg);
536
537 ra8_log_info_val(s_tag, "drw_init fb", (uint32_t)cfg->framebuffer_addr);
538 return k_ra8_ok;
539}

628{
629 if (poll_budget == 0UL) {
630 return k_ra8_err_invalid_arg;
631 }
632 /* NASA Rule 2: bounded loop. */
633 for (uint32_t i = 0UL; i < poll_budget; ++i) { /* GCOVR_EXCL_BR_LINE */
634 /* HUM Ch 62.2.5 "STATUS: Status Control Register", p 3695 */
635 const uint32_t s = *ra8_drw_reg32(k_ra8_drw_off_status);
636 if ((s & k_ra8_drw_status_busy_mask) == 0UL) { /* GCOVR_EXCL_BR_LINE */
637 return k_ra8_ok;
638 }
639 }
640 ra8_log_error(s_tag, "drw_wait_idle: timeout");
641 return k_ra8_err_hw_timeout;
642}
