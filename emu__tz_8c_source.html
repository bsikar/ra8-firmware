86static void on_blxns(uc_engine* uc, uint64_t address, uint32_t size, void* user)
87{
88 (void)address;
89 (void)size;
90 (void)user;
91 /* Prefer the live VTOR_NS: ra8_tz_secure_boot_jump_ns stores the NS vector
92 * base to the 0xE002ED08 alias (plain PPB RAM here) right before its BLXNS,
93 * so this resolves the NS table wherever the app placed it (MRAM-resident
94 * 0x02080000, SRAM2 run alias 0x32100000, or OSPI XIP) with no per-app
95 * knowledge. Fall back to ::s_ns_vector_base when the app never wrote it. */
96 uint32_t vector_base = 0U;
97 (void)uc_mem_read(uc, (uint64_t)k_scb_vtor_ns_addr, &vector_base, sizeof(vector_base));
98 if (vector_base == 0U) {
99 vector_base = s_ns_vector_base;
100 }
101 uint32_t ns_msp = 0U;
102 uint32_t ns_reset = 0U;
103 (void)uc_mem_read(uc, (uint64_t)vector_base, &ns_msp, sizeof(ns_msp));
104 (void)uc_mem_read(uc,
105 (uint64_t)vector_base + (uint64_t)sizeof(uint32_t),
106 &ns_reset,
107 sizeof(ns_reset));
108 const uint32_t ns_pc = ns_reset & ~1U; /* mask the Thumb bit for the PC write */
109 (void)uc_reg_write(uc, UC_ARM_REG_SP, &ns_msp);
110 (void)uc_reg_write(uc, UC_ARM_REG_PC, &ns_pc);
111 (void)uc_emu_stop(uc);
112}

134static uint32_t emu_tz_find_blxns(uc_engine* uc, uint32_t from, uint32_t size)
135{
136 for (uint32_t a = from; (a + (uint32_t)k_thumb_hw_bytes) <= (from + size);
137 a += (uint32_t)k_thumb_hw_bytes) {
138 uint16_t hw = 0U;
139 (void)uc_mem_read(uc, (uint64_t)a, &hw, sizeof(hw));
140 if (((uint32_t)hw & (uint32_t)k_blxns_mask) == (uint32_t)k_blxns_match) {
141 return a;
142 }
143 }
144 return 0U;
145}

148void emu_tz_install(uc_engine* uc, const uint8_t* elf, long elf_len)
149{
150 /* TrustZone S->NS boot seams -- armed whenever the firmware links the secure
151 * boot's ra8_tz_secure_boot_jump_ns: a two-image --ns app, OR a single-image
152 * app whose NS half is embedded at its MRAM LMA (cpu1_pingpong_ipc). The
153 * Secure boot bails to its fallback main() unless SAU_TYPE.SREGION >= 4/5;
154 * ra8_emulator maps the PPB as plain RAM (SAU_TYPE reads 0), so seed the M85's
155 * 8-region count to let the real SAU programming + NS-image copy + BLXNS
156 * run. Firmware without the symbol keeps its current (all-Secure) path. */
157 uint32_t jn_size = 0U;
158 const uint32_t jump_ns = elf_sym_addr(elf, elf_len, "ra8_tz_secure_boot_jump_ns", &jn_size);
159 if ((jump_ns == 0U) || (jn_size < (uint32_t)k_thumb_hw_bytes)) {
160 return; /* Not a TrustZone image: keep the all-Secure path. */
161 }
162 const uint32_t sau_type = (uint32_t)k_sau_type_regs;
163 (void)uc_mem_write(uc, (uint64_t)k_sau_type_addr, &sau_type, sizeof(sau_type));
164
165 /* Hand-emulate the Secure->NS BLXNS in ra8_tz_secure_boot_jump_ns.
166 * Unicorn's all-Secure M33 cannot really switch worlds, so resolve the
167 * BLXNS site from the Secure symtab (scan the function for the BLXNS
168 * opcode) and hook it to enter NS manually (see on_blxns). Without this
169 * the BLXNS stalls and the NS world never runs. */
170 const uint32_t blxns_at = emu_tz_find_blxns(uc, jump_ns, jn_size);
171 if (blxns_at == 0U) {
172 (void)fprintf(stderr,
173 "ra8_emulator: TZ warning: no BLXNS found in ra8_tz_secure_boot_jump_ns\n");
174 return;
175 }
176 uc_hook h_blxns;
177 (void)uc_hook_add(uc,
178 &h_blxns,
179 UC_HOOK_CODE,
180 (void*)on_blxns,
181 nullptr,
182 (uint64_t)blxns_at,
183 (uint64_t)blxns_at);
184 (void)fprintf(stderr, "ra8_emulator: TZ BLXNS seam armed @ 0x%08X\n", blxns_at);
185}

188void emu_tz_patch_cmse(uc_engine* uc, const uint8_t* elf, long elf_len)
189{
190 /* TrustZone NSC pointer validation. The Non-Secure-Callable veneers guard
191 * their pointer args with cmse_check_address_range(), which issues Armv8-M
192 * `TT`/`TTA` (Test Target) instructions to read an address's security/MPU
193 * attribution and then checks the Non-Secure read/write bit. Unicorn's M33
194 * has no SAU/IDAU configured (ra8_emulator maps the PPB as plain RAM, so the
195 * core's internal SAU stays at its reset all-Secure state); a native TT thus
196 * reports every address as Secure, the NS range-check fails, and the veneer
197 * returns k_ra8_err_invalid_arg -- stalling CGC/SD bring-up. ra8_emulator collapses
198 * the Secure/Non-Secure split into one flat, fully-accessible domain, so every
199 * NS pointer the veneers pass (each already null-checked before the range check)
200 * is valid. Model that by patching the routine's entry to `BX LR`: r0 still
201 * holds the first argument (the pointer `p`) at entry, so an immediate return
202 * yields p != NULL == "address OK". This is a one-time 2-byte memory patch
203 * (the function image is already copied into Unicorn memory by load_elf), not a
204 * UC_HOOK_CODE -- a code hook forces Unicorn to single-step the whole run
205 * (~10x slower), whereas the patch has zero steady-state cost. Absent in
206 * non-TZ firmware (symbol not found -> no patch). */
207 const uint32_t cmse_check_addr = elf_sym_addr(elf, elf_len, "cmse_check_address_range", nullptr);
208 if (cmse_check_addr != 0U) {
209 const uint16_t bx_lr = (uint16_t)k_thumb_bx_lr;
210 (void)uc_mem_write(uc, (uint64_t)cmse_check_addr, &bx_lr, sizeof(bx_lr));
211 }
212}
