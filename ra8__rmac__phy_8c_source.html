148RA8_INTERNAL
149static ra8_err_t internal_reset_and_wait(uint16_t poll_max)
150{
151 ra8_err_t err = s_state.io.write(s_state.io.ctx,
152 s_state.phy_address,
153 k_ra8_rmac_phy_reg_control,
154 k_ra8_rmac_phy_bmcr_reset);
155 if (err != k_ra8_ok) {
156 return err;
157 }
158 for (uint16_t i = 0U; i < poll_max; ++i) { /* GCOVR_EXCL_BR_LINE */
159 uint16_t reg = 0U;
160 err = s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_rmac_phy_reg_control, &reg);
161 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
162 return err;
163 }
164 if ((reg & k_ra8_rmac_phy_bmcr_reset) == 0U) {
165 return k_ra8_ok;
166 }
167 }
168 return k_ra8_err_hw_timeout;
169}

251ra8_err_t ra8_rmac_phy_open(const ra8_rmac_phy_cfg_t* cfg)
252{
253 const ra8_err_t varg = internal_open_validate(cfg);
254 if (varg != k_ra8_ok) {
255 return varg;
256 }
257 if (s_state.opened) {
258 return k_ra8_err_exists;
259 }
260 s_state.opened = true;
261 s_state.phy_address = cfg->phy_address;
262 s_state.lsi_type = cfg->lsi_type;
263 s_state.local_advertise = cfg->local_advertise;
264 s_state.gbit_advertise = cfg->gbit_advertise;
265 s_state.io = cfg->io;
266 s_state.last_bmsr = 0U;
267
268 const uint16_t poll_max =
269 (cfg->reset_poll_max == 0U) ? k_ra8_rmac_phy_reset_poll_max : cfg->reset_poll_max;
270 ra8_err_t err = internal_reset_and_wait(poll_max);
271 if (err != k_ra8_ok) {
272 s_state.opened = false;
273 return err;
274 }
275
276 err = internal_program_advertise();
277 if (err != k_ra8_ok) {
278 s_state.opened = false;
279 return err;
280 }
281 ra8_log_info_val(s_tag, "rmac phy lsi", (uint32_t)cfg->lsi_type);
282 return k_ra8_ok;
283}

441RA8_INTERNAL
442static void internal_resolve_speed(ra8_rmac_phy_link_t* out)
443{
444 ra8_err_t err = k_ra8_ok;
445 /* 1000T status first; if no gbit, fall back to LPA. */
446 if (s_state.gbit_advertise != 0U) {
447 uint16_t msr = 0U;
448 err =
449 s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_rmac_phy_reg_1000t_status, &msr);
450 if (ra8_rmac_phy_internal_speed_ok(err, msr, (uint16_t)k_ra8_rmac_phy_msr_1000full)) {
451 out->speed = k_ra8_rmac_phy_speed_1000f;
452 return;
453 }
454 if (ra8_rmac_phy_internal_speed_ok(err, msr, (uint16_t)k_ra8_rmac_phy_msr_1000half)) {
455 out->speed = k_ra8_rmac_phy_speed_1000h;
456 return;
457 }
458 }
459 uint16_t lpa = 0U;
460 err = s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_rmac_phy_reg_an_partner, &lpa);
461 if (err == k_ra8_ok) {
462 out->partner_ability = lpa;
463 if ((lpa & k_ra8_rmac_phy_lpa_100full) != 0U) {
464 out->speed = k_ra8_rmac_phy_speed_100f;
465 } else if ((lpa & k_ra8_rmac_phy_lpa_100half) != 0U) {
466 out->speed = k_ra8_rmac_phy_speed_100h;
467 } else if ((lpa & k_ra8_rmac_phy_lpa_10full) != 0U) {
468 out->speed = k_ra8_rmac_phy_speed_10f;
469 } else if ((lpa & k_ra8_rmac_phy_lpa_10half) != 0U) {
470 out->speed = k_ra8_rmac_phy_speed_10h;
471 }
472 }
473}

500ra8_err_t ra8_rmac_phy_link_status_get(ra8_rmac_phy_link_t* out)
501{
502 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
503 if (!s_state.opened) {
504 return k_ra8_err_not_initialized;
505 }
506 uint16_t bmsr = 0U;
507 ra8_err_t err =
508 s_state.io.read(s_state.io.ctx, s_state.phy_address, k_ra8_rmac_phy_reg_status, &bmsr);
509 if (err != k_ra8_ok) {
510 return err;
511 }
512 s_state.last_bmsr = bmsr;
513 out->bmsr = bmsr;
514 out->link_up = (uint8_t)(((bmsr & k_ra8_rmac_phy_bmsr_link_up) != 0U) ? 1U : 0U);
515 out->auto_neg_done = (uint8_t)(((bmsr & k_ra8_rmac_phy_bmsr_an_complete) != 0U) ? 1U : 0U);
516 out->speed = k_ra8_rmac_phy_speed_no_link;
517 out->partner_ability = 0U;
518
519 if ((out->link_up != 0U) && (out->auto_neg_done != 0U)) {
520 internal_resolve_speed(out);
521 }
522 return k_ra8_ok;
523}
