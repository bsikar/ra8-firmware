62RA8_INTERNAL static bool
63internal_decode_entity(const char* s, size_t len, size_t* i, char* out_ch, bool* space)
64{
65 static const mdl_entity_t table[] = {
66 {"amp;", '&'},
67 {"lt;", '<'},
68 {"gt;", '>'},
69 {"quot;", '"'},
70 {"#39;", '\''},
71 {"apos;", '\''},
72 {"nbsp;", '\0'},
73 };
74 const size_t start = *i + 1U; /* byte after '&' */
75 for (size_t e = 0U; e < (sizeof(table) / sizeof(table[0])); ++e) {
76 const size_t nlen = strlen(table[e].name);
77 if (((start + nlen) <= len) && (memcmp(s + start, table[e].name, nlen) == 0)) {
78 *i = start + nlen - 1U; /* leave *i on the ';'; the loop's ++ steps past */
79 *space = (table[e].ch == '\0');
80 *out_ch = table[e].ch;
81 return true;
82 }
83 }
84 return false;
85}

135RA8_INTERNAL static bool
136internal_clean_inner_text(const char* text, size_t len, char* out, size_t cap)
137{
138 size_t n = 0U;
139 bool pending = false; /* a space is owed before the next visible char */
140 bool fits = true;
141 for (size_t i = 0U; i < len; ++i) {
142 const char c = text[i];
143 if (c == '<') {
144 while ((i < len) && (text[i] != '>')) {
145 ++i; /* skip a nested tag such as <span> */
146 }
147 continue;
148 }
149 char emit = c;
150 bool space = internal_is_ws(c);
151 if (c == '&') {
152 char decoded = '\0';
153 if (internal_decode_entity(text, len, &i, &decoded, &space)) {
154 emit = decoded;
155 }
156 }
157 if (space) {
158 pending = (n > 0U); /* collapse; never a leading space */
159 continue;
160 }
161 if (pending) {
162 internal_append_out_char(out, &n, cap, &fits, ' ');
163 }
164 pending = false;
165 internal_append_out_char(out, &n, cap, &fits, emit);
166 }
167 out[n] = '\0';
168 return fits;
169}

185RA8_INTERNAL static void internal_url_slug(const char* url, char* out, size_t cap)
186{
187 const char* q = strpbrk(url, "?#");
188 const size_t end = (q == nullptr) ? strlen(url) : (size_t)(q - url);
189 size_t beg = 0U;
190 size_t fin = end;
191 while ((fin > 0U) && (url[fin - 1U] == '/')) {
192 --fin; /* drop trailing slashes */
193 }
194 beg = fin;
195 while ((beg > 0U) && (url[beg - 1U] != '/')) {
196 --beg; /* walk back to the segment start */
197 }
198 size_t n = 0U;
199 for (size_t i = beg; (i < fin) && ((n + 1U) < cap); ++i) {
200 out[n] = url[i];
201 ++n;
202 }
203 out[n] = '\0';
204}

340RA8_INTERNAL static ra8_err_t internal_emit_hit(const char* html,
341 size_t html_len,
342 const char* base_url,
343 const char* keep,
344 const char* tag,
345 size_t tag_len,
346 size_t gt_off,
347 mdl_hit_list_t* out,
348 bool* real)
349{
350 char raw[k_mdl_url_max];
351 if (!internal_find_attr_value(tag, tag_len, "href", raw, sizeof(raw))) {
352 return k_ra8_ok; /* an <a> with no href is not a link */
353 }
354 char abs[k_mdl_url_max];
355 if (!mdl_extract_resolve_url(base_url, raw, abs, sizeof(abs))) {
356 return k_ra8_ok; /* unresolvable (e.g. javascript:, fragment): not counted
357 */
358 }
359 out->anchors_seen++;
360 if (!internal_contains_ok(abs, keep)) {
361 return k_ra8_ok; /* a real link, but not a result: counted, not stored */
362 }
363 char title[k_mdl_hit_title_max];
364 const bool title_real =
365 internal_anchor_title(html, html_len, tag, tag_len, gt_off, title, sizeof(title));
366 if (!title_real) {
367 internal_url_slug(abs, title, sizeof(title));
368 }
369 return internal_merge_hit(out, real, abs, title, title_real);
370}

433RA8_INTERNAL static bool internal_class_has_token(const char* classes, const char* value)
434{
435 const size_t value_len = strlen(value);
436 const char* at = classes;
437 while ((at = strstr(at, value)) != nullptr) {

440 const bool right_ok = (after == '\0') || internal_is_ws(after);
