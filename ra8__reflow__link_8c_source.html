36ra8_err_t ra8_reflow_hit_test_link(const ra8_reflow_t* engine,
37 uint32_t page_idx,
38 int32_t x,
39 int32_t y,
40 uint32_t* out_href_off,
41 uint32_t* out_href_len)
42{
43 RA8_CHECK_NULL_PTR(engine, s_tag_link, "hit: null engine");
44 RA8_CHECK_NULL_PTR(out_href_off, s_tag_link, "hit: null out_href_off");
45 RA8_CHECK_NULL_PTR(out_href_len, s_tag_link, "hit: null out_href_len");
46
47 for (uint32_t i = 0U; i < engine->link_rect_count; ++i) {
48 const ra8_reflow_link_rect_t* rect = &engine->link_rects[i];
49 if (rect->page_index != page_idx) {
50 continue;
51 }
52 if ((x >= rect->x) && (x < (rect->x + rect->w)) && (y >= rect->y) &&
53 (y < (rect->y + rect->h))) {
54 const ra8_reflow_link_target_t* target = &engine->link_targets[rect->target];
55 *out_href_off = target->href_off;
56 *out_href_len = target->href_len;
57 return k_ra8_ok;
58 }
59 }
60 return k_ra8_err_not_found;
61}

63ra8_err_t ra8_reflow_hit_test_image(const ra8_reflow_t* engine,
64 uint32_t page_idx,
65 int32_t x,
66 int32_t y,
67 uint32_t* out_index)
68{
69 RA8_CHECK_NULL_PTR(engine, s_tag_link, "img hit: null engine");
70 RA8_CHECK_NULL_PTR(out_index, s_tag_link, "img hit: null out_index");
71
72 for (uint32_t i = 0U; i < engine->image_box_count; ++i) {
73 const ra8_reflow_image_box_t* box = &engine->image_boxes[i];
74 if (box->page_index != page_idx) {
75 continue;
76 }
77 /* Decision: the point is inside the box (4 conditions, half-open on the
78 * right/bottom edges so abutting figures never both claim a tap). */
79 if ((x >= box->x) && (x < (box->x + box->w)) && (y >= box->y) && (y < (box->y + box->h))) {
80 *out_index = i;
81 return k_ra8_ok;
82 }
83 }
84 return k_ra8_err_not_found;
85}

87ra8_err_t ra8_reflow_find_anchor(const ra8_reflow_t* engine,
88 const char* id,
89 uint32_t id_len,
90 uint32_t* out_page)
91{
92 RA8_CHECK_NULL_PTR(engine, s_tag_link, "anchor: null engine");
93 RA8_CHECK_NULL_PTR(id, s_tag_link, "anchor: null id");
94 RA8_CHECK_NULL_PTR(out_page, s_tag_link, "anchor: null out_page");
95 if (id_len == 0U) {
96 return k_ra8_err_invalid_arg;
97 }
98
99 for (uint32_t i = 0U; i < engine->anchor_count; ++i) {
100 const ra8_reflow_anchor_t* anchor = &engine->anchors[i];
101 if ((anchor->id_len == id_len) &&
102 (memcmp(&engine->text_pool[anchor->id_off], id, (size_t)id_len) == 0)) {
103 *out_page = anchor->page_index;
104 return k_ra8_ok;
105 }
106 }
107 return k_ra8_err_not_found;
108}

182RA8_INTERNAL
183static void priv_href_classify(const char* href,
184 uint32_t len,
185 ra8_reflow_href_kind_t* out_kind,
186 uint32_t* out_path_len,
187 uint32_t* out_frag_off,
188 uint32_t* out_frag_len)
189{
190 uint32_t hash = len;
191 for (uint32_t i = 0U; i < len; ++i) {
192 if (href[i] == '#') {
193 hash = i;
194 break;
195 }
196 }
197 if (priv_has_scheme(href, hash)) {
198 *out_kind = k_ra8_reflow_href_external;
199 *out_path_len = len;
200 return;
201 }
202 const bool has_frag = (hash < len);
203 *out_path_len = hash;
204 if (has_frag) {
205 *out_frag_off = hash + 1U;
206 *out_frag_len = len - hash - 1U;
207 }
208 if (hash == 0U) {
209 *out_kind = has_frag ? k_ra8_reflow_href_fragment : k_ra8_reflow_href_empty;
210 } else {
211 *out_kind = has_frag ? k_ra8_reflow_href_chapter_fragment : k_ra8_reflow_href_chapter;
212 }
213}
