205static void tz_sram_ns_boundary(void)
206{
207 /* Open PRC4 so the CPSCU SRAMSABAR writes below land. */
208 /* HUM Ch 9.2.4 "PRCR_S" p 397 */
209 *(volatile uint16_t*)k_tz_prcr_s_addr = (uint16_t)k_tz_prcr_s_open;
210
211 /* HUM Ch 58.2 "SRAMSABARn : SRAM Security Attribute Boundary Address
212 * Register" p 3527 -- boundary = start address of the NS region; below =
213 * Secure, at/above = Non-secure. */
214 tz_write32(k_tz_sramsabar0_addr + (0U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar0_val);
215 tz_write32(k_tz_sramsabar0_addr + (1U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar1_val);
216 tz_write32(k_tz_sramsabar0_addr + (2U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar2_val);
217 tz_write32(k_tz_sramsabar0_addr + (3U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar3_val);
218
219 /* Re-lock PRC4 (restore CPSCU write-protect). */
220 /* HUM Ch 9.2.4 "PRCR_S" p 397 */
221 *(volatile uint16_t*)k_tz_prcr_s_addr = (uint16_t)k_tz_prcr_s_close;
222}

242static ra8_err_t tz_sau_program(void)
243{
244 /* HUM Ch 51.3.3.3 "Secure Attribution Unit (SAU)" p 3266 -- need at least 3
245 * implemented regions for this layout. */
246 const uint32_t sau_type = tz_read32(k_tz_sau_type_addr);
247 if ((sau_type & (uint32_t)k_tz_sau_type_mask) < (uint32_t)k_tz_sau_min_regions) {
248 return k_ra8_err_not_supported;
249 }
250
251 tz_sau_set_ns_region((uint8_t)k_tz_region_ns_code,
252 (uint32_t)k_tz_ns_code_base,
253 (uint32_t)k_tz_ns_code_limit);
254 tz_sau_set_ns_region((uint8_t)k_tz_region_ns_sram,
255 (uint32_t)k_tz_ns_sram_base,
256 (uint32_t)k_tz_ns_sram_limit);
257 tz_sau_set_ns_region((uint8_t)k_tz_region_ns_per,
258 (uint32_t)k_tz_ns_per_base,
259 (uint32_t)k_tz_ns_per_limit);
260
261 __asm__ volatile("dsb 0xF" ::: "memory");
262 /* Enable the SAU with ALLNS = 0 (default-deny). */
263 /* HUM Ch 51.3.3.3 "Secure Attribution Unit (SAU)" p 3266 */
264 tz_write32(k_tz_sau_ctrl_addr, (uint32_t)k_tz_sau_ctrl_enable);
265 __asm__ volatile("dsb 0xF" ::: "memory");
266 __asm__ volatile("isb" ::: "memory");
267 return k_ra8_ok;
268}

302void ra8_trustzone_init(void)
303{
304#ifdef RA8_TRUSTZONE_ENABLE
305 /* 1. Carve the SRAM2 NS aperture via the runtime SRAMSABAR boundary. */
306 tz_sram_ns_boundary();
307
308 /* 2. Programme the bit[28] SAU (3 IDAU-NS ranges), enable default-deny. */
309 if (tz_sau_program() != k_ra8_ok) {
310 return; /* Leave the SAU disabled; main() will find no NS liveness. */
311 }
312
313 /* 3. Copy the NS image (body + RoT trailer) into the now-Non-secure SRAM
314 * alias. main() authenticates it there and BLXNS-es after crypto is up. */
315 tz_copy_ns_image();
316#endif
317}
