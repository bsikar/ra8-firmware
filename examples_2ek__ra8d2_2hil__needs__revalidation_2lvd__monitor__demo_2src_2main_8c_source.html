68/* PRCR write values to (un)lock the PVD protection group. Not a Doxygen block:
69 * it documents constants that live in ra8_lvd's headers, not the declaration
70 * that follows it.
71 *
72 * The PVD control registers (PVD1CMPCR / PVD1CR0 / PVD1CR1 / PVD1SR /
73 * PVD1FCR) are write-protected by PRCR.PRC3. Each PVD register page carries
74 * the note "Set the PRCR.PRC3 bit to 1 (write enabled) before rewriting this
75 * register" (HUM Ch 8.2.2 p 303, Ch 8.2.4 p 305, Ch 8.2.6 p 307, Ch 8.2.7
76 * p 307). PRCR is R_SYSTEM + 0x3FA; bits 15:8 are the mandatory 0xA5 key and
77 * k_ra8_prcr_grp3_pvd (0x0008) is the PRC3 group bit, so
78 * k_ra8_prcr_unlock_pvd (0xA508) enables writes and k_ra8_prcr_lock_all (the
79 * key alone) relocks. */

209[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_lvd_demo_configure(void)
210{
211 const ra8_lvd_cfg_t cfg = {
212 .threshold = k_ra8_lvd_pvdlvl_2_80v,
213 .edge = k_ra8_lvd_edge_both,
214 .irq_type = k_ra8_lvd_irq_maskable,
215 .response = k_ra8_lvd_response_none, /* SAFE: flags only, no reset/NMI. */
216 .negate = k_ra8_lvd_negate_after_voltage,
217 .hysteresis = k_ra8_lvd_hysteresis_lvd,
218 .filter_div = k_ra8_lvd_loco_div_8,
219 .filter_en = false,
220 .irq_enable = false,
221 .clear_status = true,
222 };
223
224 /* HUM Ch 8.2.4 "PVDmCR0" p 305: "Set the PRCR.PRC3 bit to 1 (write
225 * enabled) before rewriting this register." PRCR = R_SYSTEM + 0x3FA,
226 * key 0xA5 in [15:8], PRC3 = bit 3. */
227 *ra8_sys_prcr() = (uint16_t)k_ra8_prcr_unlock_pvd;
228 const ra8_err_t err = ra8_lvd_channel_init(k_ra8_lvd_ch1, &cfg);
229 ra8_sys_prcr_lock_all();
230
231 g_lvd_cfg_err = (err == k_ra8_ok) ? 0U : (uint32_t)err;
232 return err;
233}

259[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_lvd_demo_sample(uint8_t* out_ok)
260{
261 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
262
263 ra8_lvd_status_t st = {.crossed = false, .above = false};
264 const ra8_err_t err = ra8_lvd_get_status(k_ra8_lvd_ch1, &st);
265 if (err != k_ra8_ok) {
266 return err;
267 }
268 g_lvd_mon_above = st.above ? 1U : 0U;
269 g_lvd_det = st.crossed ? 1U : 0U;
270 *out_ok = (st.above && !st.crossed) ? 1U : 0U;
271 return k_ra8_ok;
272}

274void main(void)
275{
276 internal_lvd_demo_setup_or_halt();
277 /* Clear PRIMASK so SysTick can dispatch: ra8_delay_ms() uses the
278 * SysTick tick path only when IRQs are globally enabled, otherwise it
279 * busy-waits on DWT_CYCCNT. No NVIC sources are armed (response=none). */
280 ra8_isr_globals_enable();
281 if (internal_lvd_demo_configure() != k_ra8_ok) {
282 internal_lvd_demo_panic_halt();
283 }
284 ra8_delay_ms((uint32_t)k_lvd_demo_stab_ms);
285
286 while (1) {
287 uint8_t ok = 0U;
288 const ra8_err_t err = internal_lvd_demo_sample(&ok);
289 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
290 g_lvd_ok = (uint32_t)good;
291 if (good != 0U) {
292 (void)ra8_board_uart_console_write(s_lvd_demo_ok_msg,
293 (size_t)(sizeof(s_lvd_demo_ok_msg) - 1U));
294 (void)ra8_board_led_toggle(k_ra8_board_led1);
295 } else {
296 (void)ra8_board_uart_console_write(s_lvd_demo_bad_msg,
297 (size_t)(sizeof(s_lvd_demo_bad_msg) - 1U));
298 (void)ra8_board_led_toggle(k_ra8_board_led2);
299 }
300 ++g_lvd_heartbeat;
301 ra8_delay_ms(k_lvd_demo_period_ms);
302 }
303 internal_lvd_demo_panic_halt();
304}
