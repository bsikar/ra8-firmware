168static void gptp_write_u64(uint64_t val)
169{
170 uint8_t buf[k_gptp_dec_u64_max];
171 uint8_t tmp[k_gptp_dec_u64_max];
172 uint32_t n = 0U;
173 uint64_t v = val;
174 if (v == 0U) {
175 buf[0] = (uint8_t)'0';
176 gptp_write(buf, 1U);
177 return;
178 }
179 while (v != 0U) {
180 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint64_t)k_gptp_radix));
181 v = v / (uint64_t)k_gptp_radix;
182 n++;
183 }
184 for (uint32_t i = 0U; i < n; i++) {
185 buf[i] = tmp[n - 1U - i];
186 }
187 gptp_write(buf, n);
188}

305static bool gptp_advance_ok(uint64_t advance_ns, uint32_t elapsed_ms)
306{
307 if (elapsed_ms == 0U) {
308 return false;
309 }
310 const uint64_t expected_ns = (uint64_t)elapsed_ms * (uint64_t)k_gptp_ns_per_ms;
311 const uint64_t band_ns =
312 (expected_ns * (uint64_t)k_gptp_tolerance_pct) / (uint64_t)k_gptp_pct_full;
313 if (advance_ns > (expected_ns + band_ns)) {
314 return false;
315 }
316 return advance_ns >= (expected_ns - band_ns);
317}

334static bool gptp_measure(void)
335{
336 uint64_t sec0 = 0U;
337 uint32_t nsec0 = 0U;
338 uint64_t sec1 = 0U;
339 uint32_t nsec1 = 0U;
340 bool ok = true;
341
342 const uint32_t ms0 = ra8_time_ms();
343 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec0, &nsec0) != k_ra8_ok) {
344 ok = false;
345 }
346 ra8_delay_ms((uint32_t)k_gptp_window_ms);
347 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec1, &nsec1) != k_ra8_ok) {
348 ok = false;
349 }
350 const uint32_t elapsed_ms = ra8_time_ms() - ms0;
351
352 gptp_log_time(sec0, nsec0);
353 gptp_log_time(sec1, nsec1);
354
355 const uint64_t ns0 = gptp_flatten_ns(sec0, nsec0);
356 const uint64_t ns1 = gptp_flatten_ns(sec1, nsec1);
357 const uint64_t advance_ns = (ns1 > ns0) ? (ns1 - ns0) : 0ULL;
358
359 gptp_write(k_gptp_adv_prefix, (uint32_t)(sizeof(k_gptp_adv_prefix) - 1U));
360 gptp_write_u64(advance_ns);
361 gptp_write(k_gptp_sys_prefix, (uint32_t)(sizeof(k_gptp_sys_prefix) - 1U));
362 gptp_write_u64((uint64_t)elapsed_ms);
363 gptp_write(k_gptp_crlf, (uint32_t)(sizeof(k_gptp_crlf) - 1U));
364
365 if (!gptp_advance_ok(advance_ns, elapsed_ms)) {
366 ok = false;
367 }
368 return ok;
369}

464[[nodiscard]] static ra8_err_t gptp_arm(void)
465{
466 uint32_t eswclk_hz = 0U;
467 const ra8_err_t hz_err = ra8_cgc_eswclk_hz(&eswclk_hz);
468 if (hz_err != k_ra8_ok) {
469 return hz_err;
470 }
471 const ra8_eth_gptp_cfg_t cfg = {.clk_hz = eswclk_hz};
472 const ra8_err_t init_err = ra8_eth_gptp_init(&cfg);
473 if (init_err != k_ra8_ok) {
474 return init_err;
475 }
476 const ra8_err_t off_err =
477 ra8_eth_gptp_set_offset(k_ra8_gptp_timer_0, (uint64_t)k_gptp_base_sec, 0U);
478 if (off_err != k_ra8_ok) {
479 return off_err;
480 }
481 return ra8_eth_gptp_timer_enable(k_ra8_gptp_timer_0);
482}

486int32_t main(void)
487{
488 gptp_setup_or_halt();
489 ra8_isr_globals_enable();
490
491 if (gptp_arm() != k_ra8_ok) {
492 gptp_panic_halt();
493 }
494
495 while (1) {
496 const bool healthy = gptp_run_cycle();
497 if (healthy) {
498 gptp_write(k_gptp_verdict_pass, (uint32_t)(sizeof(k_gptp_verdict_pass) - 1U));
499 } else {
500 gptp_write(k_gptp_verdict_fail, (uint32_t)(sizeof(k_gptp_verdict_fail) - 1U));
501 }
502 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
503 break;
504 }
505 ra8_delay_ms((uint32_t)k_gptp_period_ms);
506 }
507 gptp_panic_halt();
508 return 0;
509}
