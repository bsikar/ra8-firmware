163static void gptp_write_u64(uint64_t val)
164{
165 uint8_t buf[k_gptp_dec_u64_max];
166 uint8_t tmp[k_gptp_dec_u64_max];
167 uint32_t n = 0U;
168 uint64_t v = val;
169 if (v == 0U) {
170 buf[0] = (uint8_t)'0';
171 gptp_write(buf, 1U);
172 return;
173 }
174 while (v != 0U) {
175 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint64_t)k_gptp_radix));
176 v = v / (uint64_t)k_gptp_radix;
177 n++;
178 }
179 for (uint32_t i = 0U; i < n; i++) {
180 buf[i] = tmp[n - 1U - i];
181 }
182 gptp_write(buf, n);
183}

300static bool gptp_advance_ok(uint64_t advance_ns, uint32_t elapsed_ms)
301{
302 if (elapsed_ms == 0U) {
303 return false;
304 }
305 const uint64_t expected_ns = (uint64_t)elapsed_ms * (uint64_t)k_gptp_ns_per_ms;
306 const uint64_t band_ns =
307 (expected_ns * (uint64_t)k_gptp_tolerance_pct) / (uint64_t)k_gptp_pct_full;
308 if (advance_ns > (expected_ns + band_ns)) {
309 return false;
310 }
311 return advance_ns >= (expected_ns - band_ns);
312}

329static bool gptp_measure(void)
330{
331 uint64_t sec0 = 0U;
332 uint32_t nsec0 = 0U;
333 uint64_t sec1 = 0U;
334 uint32_t nsec1 = 0U;
335 bool ok = true;
336
337 const uint32_t ms0 = ra8_time_ms();
338 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec0, &nsec0) != k_ra8_ok) {
339 ok = false;
340 }
341 ra8_delay_ms((uint32_t)k_gptp_window_ms);
342 if (ra8_eth_gptp_get_time(k_ra8_gptp_timer_0, &sec1, &nsec1) != k_ra8_ok) {
343 ok = false;
344 }
345 const uint32_t elapsed_ms = ra8_time_ms() - ms0;
346
347 gptp_log_time(sec0, nsec0);
348 gptp_log_time(sec1, nsec1);
349
350 const uint64_t ns0 = gptp_flatten_ns(sec0, nsec0);
351 const uint64_t ns1 = gptp_flatten_ns(sec1, nsec1);
352 const uint64_t advance_ns = (ns1 > ns0) ? (ns1 - ns0) : 0ULL;
353
354 gptp_write(k_gptp_adv_prefix, (uint32_t)(sizeof(k_gptp_adv_prefix) - 1U));
355 gptp_write_u64(advance_ns);
356 gptp_write(k_gptp_sys_prefix, (uint32_t)(sizeof(k_gptp_sys_prefix) - 1U));
357 gptp_write_u64((uint64_t)elapsed_ms);
358 gptp_write(k_gptp_crlf, (uint32_t)(sizeof(k_gptp_crlf) - 1U));
359
360 if (!gptp_advance_ok(advance_ns, elapsed_ms)) {
361 ok = false;
362 }
363 return ok;
364}

459[[nodiscard]] static ra8_err_t gptp_arm(void)
460{
461 uint32_t eswclk_hz = 0U;
462 const ra8_err_t hz_err = ra8_cgc_eswclk_hz(&eswclk_hz);
463 if (hz_err != k_ra8_ok) {
464 return hz_err;
465 }
466 const ra8_eth_gptp_cfg_t cfg = {.clk_hz = eswclk_hz};
467 const ra8_err_t init_err = ra8_eth_gptp_init(&cfg);
468 if (init_err != k_ra8_ok) {
469 return init_err;
470 }
471 const ra8_err_t off_err =
472 ra8_eth_gptp_set_offset(k_ra8_gptp_timer_0, (uint64_t)k_gptp_base_sec, 0U);
473 if (off_err != k_ra8_ok) {
474 return off_err;
475 }
476 return ra8_eth_gptp_timer_enable(k_ra8_gptp_timer_0);
477}

481int32_t main(void)
482{
483 gptp_setup_or_halt();
484 ra8_isr_globals_enable();
485
486 if (gptp_arm() != k_ra8_ok) {
487 gptp_panic_halt();
488 }
489
490 while (1) {
491 const bool healthy = gptp_run_cycle();
492 if (healthy) {
493 gptp_write(k_gptp_verdict_pass, (uint32_t)(sizeof(k_gptp_verdict_pass) - 1U));
494 } else {
495 gptp_write(k_gptp_verdict_fail, (uint32_t)(sizeof(k_gptp_verdict_fail) - 1U));
496 }
497 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
498 break;
499 }
500 ra8_delay_ms((uint32_t)k_gptp_period_ms);
501 }
502 gptp_panic_halt();
503 return 0;
504}
