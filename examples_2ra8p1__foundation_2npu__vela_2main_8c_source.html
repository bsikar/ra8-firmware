210static bool npu_vela_verify(const ra8_npu_job_t* job, uint32_t* out_check)
211{
212 const uint8_t* input = (const uint8_t*)(uintptr_t)job->region_base[k_ra8_npu_region_1];
213 const uint8_t* output = (const uint8_t*)(uintptr_t)job->region_base[k_ra8_npu_region_2];
214 const uint32_t konst = npu_vela_cmd_word(job, k_ra8_npu_fake_word_const);
215 const uint32_t count = npu_vela_cmd_word(job, k_ra8_npu_fake_word_count);
216 uint32_t check = (uint32_t)k_npu_vela_fnv_offset;
217 bool ok = true;
218 for (uint32_t i = 0U; i < count; i++) {
219 const uint8_t expected = (uint8_t)((input[i] + konst) & (uint32_t)k_npu_vela_byte_mask);
220 if (output[i] != expected) {
221 ok = false;
222 }
223 check = (check ^ (uint32_t)output[i]) * (uint32_t)k_npu_vela_fnv_prime;
224 }
225 *out_check = check;
226 return ok;
227}

239static ra8_err_t npu_vela_run_job(ra8_npu_job_t* out_job)
240{
241 const ra8_npu_arena_t arena = {.base = s_npu_vela_arena, .bytes = (uint32_t)k_npu_vela_arena};
242 const ra8_err_t ld =
243 ra8_npu_load(ra8_npu_model_addk_fake_blob(), ra8_npu_model_addk_fake_bytes(), &arena, out_job);
244 if (ld != k_ra8_ok) {
245 return ld;
246 }
247 const ra8_err_t sub = ra8_npu_submit(out_job);
248 if (sub != k_ra8_ok) {
249 return sub;
250 }
251 const ra8_err_t run = ra8_npu_run();
252 if (run != k_ra8_ok) {
253 return run;
254 }
255 return ra8_npu_wait();
256}

299int32_t main(void)
300{
301 npu_vela_setup_or_halt();
302
303 bool load_ok = false;
304 bool run_ok = false;
305 uint32_t id = 0U;
306 uint32_t check = 0U;
307 bool pass = false;
308
309 if (ra8_npu_init() == k_ra8_ok) {
310 (void)ra8_npu_read_id(&id);
311 g_npu_vela_id = id;
312
313 ra8_npu_job_t job = {};
314 const ra8_err_t rc = npu_vela_run_job(&job);
315 load_ok = (job.cmd_stream != nullptr); /* load leaves it null on failure */
316 run_ok = (rc == k_ra8_ok);
317
318 /* Gate verify on load_ok as well: a successful run implies the load
319 * populated cmd_stream, but stating it explicitly lets the static analyzer
320 * see that npu_vela_verify() never dereferences a null cmd_stream. */
321 const bool matched = run_ok && load_ok && npu_vela_verify(&job, &check);
322 g_npu_vela_check = check;
323 pass = load_ok && run_ok && matched && (id != 0U);
324 }
325
326 g_npu_vela_pass = pass ? 1U : 0U;
327 npu_vela_emit(id, load_ok, run_ok, check, pass);
328
329 npu_vela_park();
330 return 0;
331}
