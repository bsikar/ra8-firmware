235[[nodiscard]] static ra8_err_t lpi_wait_count_started(uint8_t channel)
236{
237 const uint8_t tcstf_mask = (uint8_t)(1U << (uint8_t)k_ra8_ulpt_bit_tcstf);
238 for (uint32_t i = 0U; i < (uint32_t)k_lpi_tcstf_poll_limit; ++i) {
239 uint8_t status = 0U;
240 ra8_err_t err = ra8_ulpt_get_status(channel, &status);
241 if (err != k_ra8_ok) {
242 return err;
243 }
244 if ((status & tcstf_mask) != 0U) {
245 return k_ra8_ok;
246 }
247 }
248 return k_ra8_err_hw_timeout;
249}

406[[nodiscard]] static ra8_err_t lpi_idle_one_period(void)
407{
408 ra8_err_t err = ra8_ulpt_start((uint8_t)k_lpi_channel, (uint32_t)k_lpi_period_ticks);
409 if (err != k_ra8_ok) {
410 return err;
411 }
412 err = lpi_wait_count_started((uint8_t)k_lpi_channel);
413 if (err != k_ra8_ok) {
414 return err;
415 }
416 err = ra8_lpm_enter_sleep(k_ra8_sleep_mode_software_std);
417 if (err != k_ra8_ok) {
418 return err;
419 }
420 return ra8_ulpt_stop((uint8_t)k_lpi_channel);
421}

425int32_t main(void)
426{
427 lpi_setup_or_halt();
428
429 if (lpi_arm_wake() != k_ra8_ok) {
430 lpi_panic_halt();
431 }
432 ra8_isr_globals_enable();
433
434 if (ra8_board_uart_console_write(k_lpi_boot_msg, (size_t)(sizeof(k_lpi_boot_msg) - 1U)) !=
435 k_ra8_ok) {
436 lpi_panic_halt();
437 }
438
439 /* Bounded periodic loop (NASA P10 Rule 2): k_lpi_period_count periods
440 * of work-then-standby, each woken by the ULPT0 underflow. */
441 for (uint32_t period = 0U; period < (uint32_t)k_lpi_period_count; ++period) {
442 if (lpi_do_work() != k_ra8_ok) {
443 lpi_panic_halt();
444 }
445 if (lpi_idle_one_period() != k_ra8_ok) {
446 lpi_panic_halt();
447 }
448 }
449
450 if (ra8_board_uart_console_write(k_lpi_pass_msg, (size_t)(sizeof(k_lpi_pass_msg) - 1U)) !=
451 k_ra8_ok) {
452 lpi_panic_halt();
453 }
454
455 lpi_panic_halt();
456 return 0;
457}
