234{
235 s_ro_buffer[0] = (uint8_t)k_mpu_simple_probe_byte;
236 /* cppcheck-suppress knownConditionTrueFalse -- host: the write succeeds; silicon: the MemManage handler skipped the store. */

289RA8_INTERNAL [[gnu::used]] static void mpu_simple_fault_recover(uint32_t* frame);
290RA8_INTERNAL [[gnu::used]] static void mpu_simple_fault_recover(uint32_t* frame)
291{
292 /* Advance the stacked PC past the faulting store. The compiler
293 * emits ``strb r3, [r2]`` for the probe, which is 16 bits on
294 * Thumb-1 / 32 bits on Thumb-2 depending on register allocation.
295 * Reading the half-word at the stacked PC tells us which one we
296 * landed on: low 5 bits 0b11101/0b11110/0b11111 mark a 32-bit
297 * encoding (Armv8-M ARM A6.3.1 "Thumb instruction set encoding"),
298 * everything else is 16-bit. */
299 const uint16_t* insn = (const uint16_t*)frame[k_mpu_simple_frame_pc_idx];
300 const uint16_t first_halfwd = *insn;
301 const uint16_t hi5 = (uint16_t)((first_halfwd >> k_thumb_hi5_shift) & k_thumb_hi5_mask);
302 const bool is_thumb2_32b = (hi5 == k_thumb2_prefix_11101) || (hi5 == k_thumb2_prefix_11110) ||
303 (hi5 == k_thumb2_prefix_11111);
304 frame[k_mpu_simple_frame_pc_idx] += (is_thumb2_32b ? 4U : 2U);
305
306 /* NOTE: we deliberately do NOT clear the MemManage half-word of
307 * CFSR. The HIL alive-mode probe runs with HIL_FAULT_EXPECTED=1
308 * and explicitly REQUIRES a non-zero CFSR at probe time to prove
309 * the configurable fault actually fired (the failure mode the
310 * flag guards against is "app booted, looped, but never tripped
311 * the MPU because the partition was misconfigured"). Leaving the
312 * MMFSR bits latched gives the probe both proofs: CFSR != 0 says
313 * the fault fired, and the running CycleCnt + match-counter +
314 * UART banner say recovery succeeded. The bits self-clear on the
315 * next chip reset; firmware never reads CFSR. */
316 g_mpu_simple_fault_count += 1U;
317 s_fault_pending = 1U;
318}

321void MemManage_Handler(void)
322{
323 __asm__ volatile("tst lr, #4 \n"
324 "ite eq \n"
325 "mrseq r0, msp \n"
326 "mrsne r0, psp \n"
327 "push {lr} \n"
328 "bl mpu_simple_fault_recover \n"
329 "pop {lr} \n"
330 "bx lr \n");

361int32_t main(void)
362{
363 mpu_simple_setup_or_halt();
364 ra8_isr_globals_enable();
365
366 if (ra8_mpu_configure(&s_cfg) != k_ra8_ok) {
367 mpu_simple_panic_halt();
368 }
369
370 /* Probe drives the deliberate fault. On silicon the recovering
371 * handler advances the stacked PC past the offending store and
372 * resumes here; the readback then sees the original 0x00 (handler
373 * skipped the write that would have written 0x42), so the probe
374 * returns k_ra8_err_hw_error. On the host fake the store just
375 * succeeds and the probe returns k_ra8_ok. */
376 if (mpu_simple_probe() == k_ra8_ok) {
377 /* Host path: MPU did not arm (or fake). Latch LED3 as a
378 * diagnostic but keep the main loop spinning so the HIL probe
379 * sees forward progress. */
380 (void)ra8_board_led_on(k_ra8_board_led3);
381 } else {
382 (void)ra8_board_led_on(k_ra8_board_led1);
383 }
384
385 while (1) {
386 if (s_fault_pending != 0U) {
387 s_fault_pending = 0U;
388 mpu_simple_emit_banner();
389 }
390 g_mpu_simple_match += 1U;
391 (void)ra8_board_led_toggle(k_ra8_board_led1);
392 ra8_delay_ms((uint32_t)k_mpu_simple_period_ms);
393 }
394 mpu_simple_panic_halt();
395 return 0;
396}
