3"""Turning a parsed translation unit into a symbol table and a call list.
4
5One pass over the AST fills three things the rules then reason over: the
6USR-keyed symbol table, every call site, and the set of functions a hardware
7vector table names.
8
9Identity is the whole game here
10-------------------------------
11Symbols are keyed by USR (:func:`symbol_key`), never by name. C lets every
12translation unit have its own ``static`` helper with the same name and this
13repo does exactly that -- three unrelated ``internal_zero_bytes`` live in
14net_pal, usb_hmsc and usb_pmsc. A name-keyed table merges them into one entry
15whose annotations are the union of all three, so an annotation on one namesake
16gets enforced against another's callers. That is not hypothetical: it took
17``dev`` red, and the selftest still reproduces the exact walk order that did it.
18
19Call sites record the USR on *both* ends for the same reason. A callee-keyed
20lookup that used names would attribute one module's RA8_PRIV tag to another
21module's file-local helper of the same name.
22"""
23

35#: A section name the linker script maps to a hardware vector table.
36#: ``.vectors`` for the M85 images, ``.cpu1_vectors`` for the M33 ones.
37_VECTOR_SECTION_RE = re.compile(r'section\s*\‍(\s*"(\.[a-z0-9_]*vectors)"')
38
39#: How many lines above a declaration to search for its section attribute.
40VECTOR_ATTR_LOOKBACK_LINES = 3

43def collect_annotations(cursor: cindex.Cursor) -> list[str]:
44 """Return every ra8_* string attached to a cursor via AnnotateAttr."""
45 out: list[str] = []
46 for child in cursor.get_children():
47 if child.kind == cindex.CursorKind.ANNOTATE_ATTR:
48 text = child.spelling or child.displayname or ""
49 if text.startswith(ANNOTATION_PREFIXES):
50 out.append(text)
51 return out

54def usr_of(cursor: cindex.Cursor) -> str:
55 """Return the canonical USR of ``cursor``, or '' when unavailable."""
56 with contextlib.suppress(Exception):
57 return cursor.canonical.get_usr()
58 return ""

61def symbol_key(cursor: cindex.Cursor) -> str:
62 """Return the symbol-table key for a function cursor.
63
64 The USR when libclang can produce one -- it encodes the defining file
65 for internal-linkage symbols, which is the only thing that separates
66 same-named ``static`` helpers in different modules. A libclang build
67 that cannot produce USRs degrades to name plus file, which keeps the
68 namesakes apart for the common case rather than merging them.
69 """
70 usr = usr_of(cursor)
71 if usr:
72 return usr
73 where = str(cursor.location.file) if cursor.location.file else ""
74 return f"{cursor.spelling}@{where}"

77def _file_of(cursor: cindex.Cursor) -> str:
78 """Return the path holding ``cursor``, or '' when it has no location."""
79 if cursor.location.file is None:
80 return ""
81 return str(cursor.location.file)

84def _has_vector_section(cursor: cindex.Cursor) -> bool:
85 """True when ``cursor``'s declaration carries a ``*vectors`` section."""
86 if cursor.location.file is None:
87 return False
88 lines = source_lines(str(cursor.location.file))
89 if not lines:
90 return False
91 # The attribute sits on the declaration line or just above it; C23
92 # attributes may be split across a couple of lines by the formatter.
93 first = max(0, cursor.extent.start.line - VECTOR_ATTR_LOOKBACK_LINES)
94 return bool(_VECTOR_SECTION_RE.search("\n".join(lines[first : cursor.extent.start.line])))

97def is_vector_table(cursor: cindex.Cursor) -> bool:
98 """True when ``cursor`` declares a hardware exception vector table.
99
100 Recognised by what makes it a vector table, never by name. Either:
101
102 * a file-scope array of ``const`` pointers to functions -- the shape
103 the Cortex-M core reads through VTOR, which nothing else in this
104 tree has; or
105 * a file-scope ``const`` array the linker places in a ``*vectors``
106 section. The two-entry copy-to-run payload table and the M33
107 ``.cpu1_vectors`` table are ``const uintptr_t[]``, so the type alone
108 does not identify them, but the section placement does -- that
109 placement is the whole reason the array exists.
110
111 The section is read out of the declaration's source text because
112 ``CursorKind.SECTION_ATTR`` is absent from the libclang 18.1.x wheels
113 the runners install, so an AST lookup would silently find nothing.
114 """
115 if cursor.kind != cindex.CursorKind.VAR_DECL:
116 return False
117 parent = cursor.semantic_parent
118 if parent is None or parent.kind != cindex.CursorKind.TRANSLATION_UNIT:
119 return False
120 array = cursor.type
121 if array.kind not in (cindex.TypeKind.CONSTANTARRAY, cindex.TypeKind.INCOMPLETEARRAY):
122 return False
123 element = array.element_type
124 if not element.is_const_qualified():
125 return False
126 canonical = element.get_canonical()
127 if (
128 canonical.kind == cindex.TypeKind.POINTER
129 and canonical.get_pointee().get_canonical().kind == cindex.TypeKind.FUNCTIONPROTO
130 ):
131 return True
132 return _has_vector_section(cursor)

135class _Walker:
136 """One traversal, accumulating into the caller's collections.
137
138 A class rather than a nest of closures so each construct the walk knows
139 about -- function declaration, call, address-of, vector table -- is a
140 separately readable method instead of another branch in one long visitor.
141 """
142
143 def __init__(
144 self,
145 symbols: dict[str, AnnotatedSymbol],
146 calls: list[CallSite],
147 stats: ParseStats,
148 vector_entries: set[str],
149 ) -> None:
150 self.symbols = symbols
151 self.calls = calls
152 self.stats = stats
153 self.vector_entries = vector_entries
154
155 def _record_vector_entries(self, node: cindex.Cursor) -> None:
156 """Collect the USR of every function named in a vector table."""
157 if (
158 node.kind == cindex.CursorKind.DECL_REF_EXPR
159 and node.referenced is not None
160 and node.referenced.kind == cindex.CursorKind.FUNCTION_DECL
161 ):
162 self.vector_entries.add(symbol_key(node.referenced))
163 for child in node.get_children():
164 self._record_vector_entries(child)
165
166 def _symbol_for(self, node: cindex.Cursor) -> AnnotatedSymbol:
167 """Return (creating if needed) the table entry for a function cursor."""
168 key = symbol_key(node)
169 return self.symbols.setdefault(
170 key,
171 AnnotatedSymbol(
172 name=node.spelling, file=_file_of(node), line=node.location.line, usr=key
173 ),
174 )
175
176 @staticmethod
177 def _update_declaration(sym: AnnotatedSymbol, node: cindex.Cursor) -> None:
178 """Fold one declaration or definition of a function into its entry."""
179 for a in collect_annotations(node):
180 if a not in sym.annotations:
181 sym.annotations.append(a)
182 sym.is_static = node.linkage == cindex.LinkageKind.INTERNAL or sym.is_static
183 sym.return_type = node.result_type.spelling
184 if any(arg.type.kind == cindex.TypeKind.POINTER for arg in node.get_arguments()):
185 sym.has_pointer_param = True
186 if node.is_definition():
187 sym.is_defined = True
188 sym.file = _file_of(node)
189 sym.line = node.location.line
190 sym.end_line = node.extent.end.line
191 else:
192 sym.decl_files.add(_file_of(node))
193 # libclang exposes inline-ness via the cursor's tokens
194 tokens = [t.spelling for t in node.get_tokens()]
195 if "inline" in tokens or "__inline__" in tokens:
196 sym.has_inline = True
197 # Section attribute (only when this libclang exposes the kind).
198 if SECTION_ATTR_KIND is not None:
199 for child in node.get_children():
200 if child.kind == SECTION_ATTR_KIND:
201 sym.section = child.spelling
202
203 def _visit_function(self, node: cindex.Cursor, current_func: cindex.Cursor | None) -> None:
204 """Record a function declaration, then descend into its body."""
205 self._update_declaration(self._symbol_for(node), node)
206 inner = node if node.is_definition() else current_func
207 for child in node.get_children():
208 self.visit(child, inner)
209
210 def _record_call(self, node: cindex.Cursor, current_func: cindex.Cursor) -> None:
211 """Record one direct call, and whether its callee resolved."""
212 self.stats.calls_seen += 1
213 callee = node.referenced
214 if callee is None:
215 return
216 self.stats.calls_resolved += 1
217 self.calls.append(
218 CallSite(
219 callee_name=callee.spelling,
220 caller_name=current_func.spelling,
221 caller_file=_file_of(node),
222 caller_line=node.location.line,
223 callee_usr=symbol_key(callee),
224 caller_usr=symbol_key(current_func),
225 )
226 )
227
228 def _record_address_of(self, node: cindex.Cursor, current_func: cindex.Cursor | None) -> None:
229 """Record ``&foo`` references, which are uses but not direct calls."""
230 tokens = [t.spelling for t in node.get_tokens()]
231 if not tokens or tokens[0] != "&":
232 return
233 self.calls.extend(
234 CallSite(
235 callee_name=child.referenced.spelling,
236 caller_name=(current_func.spelling if current_func else ""),
237 caller_file=_file_of(node),
238 caller_line=node.location.line,
239 callee_usr=symbol_key(child.referenced),
240 caller_usr=(symbol_key(current_func) if current_func else ""),
241 in_address_of=True,
242 )
243 for child in node.get_children()
244 if (
245 child.kind == cindex.CursorKind.DECL_REF_EXPR
246 and child.referenced
247 and child.referenced.kind == cindex.CursorKind.FUNCTION_DECL
248 )
249 )
250
251 def visit(self, node: cindex.Cursor, current_func: cindex.Cursor | None) -> None:
252 """Dispatch one cursor, then recurse into its children."""
253 if node.kind == cindex.CursorKind.FUNCTION_DECL:
254 self._visit_function(node, current_func)
255 return
256
257 if is_vector_table(node):
258 self._record_vector_entries(node)
259
260 if node.kind == cindex.CursorKind.CALL_EXPR and current_func is not None:
261 self._record_call(node, current_func)
262
263 if node.kind == cindex.CursorKind.UNARY_OPERATOR:
264 self._record_address_of(node, current_func)
265
266 for child in node.get_children():
267 self.visit(child, current_func)
