175static ra8_err_t mk_populate(ra8_fs_mount_t* mount)
176{
177 uint8_t payload[k_mk_payload];
178 for (uint32_t i = 0U; i < (uint32_t)k_mk_payload; i++) {
179 payload[i] = (uint8_t)('A' + (i % (uint32_t)k_mk_alphabet));
180 }
181
182 for (size_t i = 0U; i < (sizeof(k_entries) / sizeof(k_entries[0])); i++) {
183 const mk_entry_t* e = &k_entries[i];
184 const ra8_err_t err = e->is_dir
185 ? ra8_fs_mkdir(mount, e->path)
186 : ra8_fs_write_file(mount, e->path, payload, (uint32_t)k_mk_payload);
187 if (err != k_ra8_ok) {
188 (void)fprintf(stderr, "exfat_mkimage: cannot create %s (%d)\n", e->meaning, (int)err);
189 return err;
190 }
191 (void)printf(" created %-44s %s\n", e->path, e->meaning);
192 }
193 return k_ra8_ok;
194}

206static bool mk_dump(const char* out_path, size_t total)
207{
208 FILE* fp = fopen(out_path, "wb");
209 if (fp == nullptr) {
210 (void)fprintf(stderr, "exfat_mkimage: cannot open %s for writing\n", out_path);
211 return false;
212 }
213 const size_t wrote = fwrite(s_disk.bytes, 1U, total, fp);
214 (void)fclose(fp);
215 if (wrote != total) {
216 (void)fprintf(stderr, "exfat_mkimage: short write to %s\n", out_path);
217 return false;
218 }
219 return true;

234int main(int argc, char** argv)
235{
236 const char* out_path = (argc > 1) ? argv[1] : "exfat_showcase.img";
237 const size_t total = (size_t)k_mk_block_count * (size_t)k_mk_block_size;
238
239 s_disk.bytes = calloc(1U, total);
240 s_disk.blocks = k_mk_block_count;
241 if (s_disk.bytes == nullptr) {
242 (void)fprintf(stderr, "exfat_mkimage: cannot allocate a %zu byte card\n", total);
243 return 1;
244 }
245
246 ra8_fs_backend_t backend = {};
247 backend.read_block = mk_read;
248 backend.write_block = mk_write;
249 backend.get_capacity = mk_capacity;
250
251 ra8_fs_format_opts_t opts = {};
252 opts.type = k_ra8_fs_type_exfat;
253 opts.label = "RA8IMAGE";
254
255 ra8_fs_mount_t* mount = nullptr;
256 ra8_err_t err = ra8_fs_format(&backend, &opts);
257 if (err == k_ra8_ok) {
258 err = ra8_fs_mount(&backend, &mount);
259 }
260 if (err != k_ra8_ok) {
261 (void)fprintf(stderr, "exfat_mkimage: format/mount failed (%d)\n", (int)err);
262 free(s_disk.bytes);
263 return 1;
264 }
265
266 err = mk_populate(mount);
267 (void)ra8_fs_unmount(mount);
268 if ((err != k_ra8_ok) || !mk_dump(out_path, total)) {
269 free(s_disk.bytes);
270 return 1;
271 }
272
273 free(s_disk.bytes);
274 (void)printf("wrote %s (%zu bytes) -- the card exactly as ra8_fs left it\n", out_path, total);
275 return 0;
276}
