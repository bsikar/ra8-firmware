62const ra8_fmt_desc_t* ra8_fmt_find(const char* name)
63{
64 if (name == nullptr) {
65 return nullptr;
66 }
67 size_t n = 0U;
68 const ra8_fmt_desc_t* regs = ra8_fmt_registry(&n);
69 for (size_t i = 0U; i < n; ++i) { /* bounded by the registry size */
70 if (strcmp(regs[i].name, name) == 0) {
71 return &regs[i];
72 }
73 }
74 return nullptr;
75}
