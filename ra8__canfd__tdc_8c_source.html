38ra8_err_t ra8_canfd_set_tdc(uint8_t channel, const ra8_canfd_tdc_cfg_t* cfg)
39{
40 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
41 volatile r_canfd_t* reg = ra8_canfd(channel);
42 RA8_CHECK_NULL_PTR(reg, s_tag, "channel out of range");
43 if (cfg->offset > (uint8_t)k_ra8_canfd_tdc_offset_max) {
44 return k_ra8_err_invalid_arg;
45 }
46
47 /* FDCFG is only writable in CH_RESET or CH_HALT; use CH_RESET as
48 * the immediate abort path (mirrors ra8_canfd_set_bitrate). */
49 /* HUM Ch 41 "CFDCnFDCFG" p 2788 */
50 const ra8_err_t rst_err = ra8_canfd_internal_set_channel_mode(reg, k_ra8_chmdc_reset);
51 if (rst_err != k_ra8_ok) {
52 return rst_err;
53 }
54
55 /* Read-modify-write FDCFG: preserve FDOE, REFE, CLOE, ESIC, EOCCFG.
56 * Clear TDE, TDCOC, and the TDCO slot before re-stamping. */
57 /* HUM Ch 41 "CFDCnFDCFG" p 2788 */
58 uint32_t fdcfg = reg->CFDC2[0].FDCFG;
59 const uint32_t tdco_slot = (uint32_t)k_ra8_fdcfg_mask_tdco << (uint32_t)k_ra8_fdcfg_shift_tdco;
60 fdcfg &= ~(k_ra8_fdcfg_mask_tde | k_ra8_fdcfg_mask_tdcoc | tdco_slot);
61 if (cfg->enable) {
62 fdcfg |= k_ra8_fdcfg_mask_tde;
63 fdcfg |= ((uint32_t)cfg->offset & (uint32_t)k_ra8_fdcfg_mask_tdco)
64 << (uint32_t)k_ra8_fdcfg_shift_tdco;
65 if (cfg->manual) {
66 fdcfg |= k_ra8_fdcfg_mask_tdcoc;
67 }
68 }
69 /* HUM Ch 41 "CFDCnFDCFG" p 2788 */
70 reg->CFDC2[0].FDCFG = fdcfg;
71
72 const ra8_err_t op_err = ra8_canfd_internal_set_channel_mode(reg, k_ra8_chmdc_operation);
73 if (op_err != k_ra8_ok) {
74 return op_err;
75 }
76 ra8_log_info(s_tag, "tdc configured");
77 return k_ra8_ok;
78}
