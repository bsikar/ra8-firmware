93RA8_INTERNAL static uint32_t internal_u16_to_dec(uint8_t* buf, uint16_t val)
94{
95 if (val == 0U) {
96 buf[0] = (uint8_t)'0';
97 return 1U;
98 }
99 uint8_t tmp[k_adc_b_demo_dec_digits];
100 uint32_t n = 0U;
101 uint16_t v = val;
102 while (v != 0U) {
103 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint16_t)k_adc_b_demo_radix));
104 v = (uint16_t)(v / (uint16_t)k_adc_b_demo_radix);
105 n++;
106 }
107 for (uint32_t i = 0U; i < n; i++) {
108 buf[i] = tmp[n - 1U - i];
109 }
110 return n;
111}

201void main(void)
202{
203 internal_setup_or_halt();
204 ra8_isr_globals_enable();
205
206 if (internal_arm() != k_ra8_ok) {
207 internal_panic_halt();
208 }
209
210 while (1) {
211 uint16_t raw = 0U;
212 if (ra8_adc_read_channel((uint8_t)k_adc_b_demo_channel, &raw) != k_ra8_ok) {
213 break;
214 }
215 const uint16_t mv = (uint16_t)(((uint32_t)raw * (uint32_t)k_adc_b_demo_vref_mv) /
216 (uint32_t)k_adc_b_demo_full_scale);
217
218 uint8_t line[k_adc_b_demo_line_max];
219 uint32_t pos = 0U;
220 for (uint32_t i = 0U; i < (sizeof(s_adc_b_demo_log_prefix) - 1U); i++) {
221 line[pos++] = s_adc_b_demo_log_prefix[i];
222 }
223 pos += internal_u16_to_dec(&line[pos], raw);
224 for (uint32_t i = 0U; i < (sizeof(s_adc_b_demo_mv_sep) - 1U); i++) {
225 line[pos++] = s_adc_b_demo_mv_sep[i];
226 }
227 pos += internal_u16_to_dec(&line[pos], mv);
228 line[pos++] = (uint8_t)k_adc_b_demo_cr;
229 line[pos++] = (uint8_t)k_adc_b_demo_lf;
230
231 (void)ra8_board_uart_console_write(line, (size_t)pos);
232 (void)ra8_board_uart_console_write(s_adc_b_demo_pass_msg,
233 (size_t)(sizeof(s_adc_b_demo_pass_msg) - 1U));
234 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
235 break;
236 }
237 ra8_delay_ms((uint32_t)k_adc_b_demo_period_ms);
238 }
239 internal_panic_halt();
240}
