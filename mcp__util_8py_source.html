3"""Shared plumbing for the MCP server: subprocess capture, file reads, app discovery.
4
5Everything here is deliberately dependency-free -- the server ships with zero
6third-party packages so it runs against a bare ``python3`` -- and everything
7that reaches the outside world is bounded: :data:`MAX_OUTPUT_CHARS` caps what a
8runaway build log can hand back to a client, and :func:`run_command` always
9carries a timeout.
10"""
11

20# Repo root = two parents up from this file (tools/mcp/<this>).
21REPO_ROOT = Path(__file__).resolve().parents[2]

23# Upper bound on captured subprocess output returned to the client, so a
24# runaway build log can never blow up the assistant's context window.
25MAX_OUTPUT_CHARS = 12000

32 """Write a diagnostic line to stderr (never stdout, which is the wire)."""
33 print(f"[ra8d2-mcp] {message}", file=sys.stderr, flush=True)

36def truncate(text: str, limit: int = MAX_OUTPUT_CHARS) -> str:
37 """Clamp ``text`` to ``limit`` characters, keeping the tail (most recent)."""
38 if len(text) <= limit:
39 return text
40 head = "[... output truncated, showing the last bytes ...]\n"
41 return head + text[-(limit - len(head)) :]

44def run_command(argv: list[str], timeout: int) -> str:
45 """Run ``argv`` from the repo root and return a formatted result block.
46
47 The combined stdout+stderr is captured and truncated. The returned string
48 leads with the exact command and its exit status so the assistant always
49 sees whether the step actually succeeded.
50 """
51 pretty = " ".join(argv)
52 try:
53 proc = subprocess.run( # noqa: S603 # trusted: fixed make/tool argv from internal registry
54 argv,
55 cwd=str(REPO_ROOT),
56 stdout=subprocess.PIPE,
57 stderr=subprocess.STDOUT,
58 text=True,
59 timeout=timeout,
60 check=False,
61 )
62 except FileNotFoundError as exc:
63 return f"$ {pretty}\n[not run] executable not found: {exc}"
64 except subprocess.TimeoutExpired:
65 return f"$ {pretty}\n[timeout] exceeded {timeout}s and was killed"
66 status = "ok" if proc.returncode == 0 else f"FAILED (exit {proc.returncode})"
67 body = truncate(proc.stdout or "")
68 return f"$ {pretty}\n[status] {status}\n\n{body}".rstrip() + "\n"

71def read_text(path: Path, max_lines: int = 0) -> str:
72 """Read a UTF-8 text file, optionally clamped to the first ``max_lines``."""
73 text = path.read_text(encoding="utf-8", errors="replace")
74 if max_lines > 0:
75 lines = text.splitlines()
76 if len(lines) > max_lines:
77 kept = "\n".join(lines[:max_lines])
78 return f"{kept}\n[... {len(lines) - max_lines} more lines ...]\n"
79 return text

83# Firmware app catalogue (mirrors the top-level Makefile discovery rules)
84# ---------------------------------------------------------------------------
85_DESCRIPTION_RX = re.compile(r'DESCRIPTION\s+"([^"]*)"')
86
87
88def discover_apps() -> dict[str, dict[str, str]]:
89 """Scan ``examples/`` for firmware apps the same way ``make apps`` does.
90
91 An app is any directory holding a ``main.c`` one to three levels under a
92 tier directory (``examples/<tier>/.../<app>/main.c``). The app name is the
93 directory basename -- the exact token ``make <app>`` / ``make flash-<app>``
94 expects. The description is parsed from the app's ``CMakeLists.txt``
95 ``ra8_add_app(... DESCRIPTION "...")`` clause.
96 """
97 apps: dict[str, dict[str, str]] = {}
98 examples = REPO_ROOT / "examples"
99 if not examples.is_dir():
100 return apps
101 for main in sorted(examples.glob("*/**/main.c")):
102 rel = main.relative_to(examples)
103 if rel.parts and rel.parts[0] == "shared":
104 continue
105 if "build" in rel.parts:
106 continue
107 app_dir = main.parent
108 name = app_dir.name
109 group = str(app_dir.relative_to(examples).parent)
110 description = ""
111 cml = app_dir / "CMakeLists.txt"
112 if cml.is_file():
113 match = _DESCRIPTION_RX.search(cml.read_text(encoding="utf-8", errors="replace"))
114 if match:
115 description = match.group(1)
116 apps[name] = {
117 "name": name,
118 "group": group,
119 "dir": str(app_dir.relative_to(REPO_ROOT)),
120 "description": description,
121 }
122 return apps

125def require_app(name: str) -> dict[str, str]:
126 """Resolve a firmware app name or raise ``ValueError`` if it is unknown.
127
128 Validating against the discovered set keeps an arbitrary string from ever
129 reaching ``make`` as a target, so a tool call cannot inject a foreign rule.
130 """
131 apps = discover_apps()
132 app = apps.get(name)
133 if app is None:
134 sample = ", ".join(sorted(apps)[:12])
135 msg = f"unknown app '{name}'. Use the list_apps tool. Examples: {sample} ..."
136 raise ValueError(msg)
137 return app
