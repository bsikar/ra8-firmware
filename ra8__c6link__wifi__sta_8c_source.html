98ra8_err_t ra8_c6link_sta_cfg_set(ra8_c6link_sta_cfg_t* cfg, const char* ssid, const char* pass)
99{
100 if ((cfg == nullptr) || (ssid == nullptr)) {
101 return k_ra8_err_null_ptr;
102 }
103 *cfg = (ra8_c6link_sta_cfg_t){};
104
105 const uint8_t ssid_len = ra8_c6link_sta_len(ssid, (uint8_t)sizeof cfg->ssid);
106 if ((ssid_len == 0U) || (ssid_len > (uint8_t)k_ra8_c6link_ssid_max)) {
107 return k_ra8_err_invalid_size;
108 }
109 const uint8_t pass_len =
110 (pass == nullptr) ? 0U : ra8_c6link_sta_len(pass, (uint8_t)sizeof cfg->pass);
111 if (pass_len > (uint8_t)k_ra8_c6link_pass_max) {
112 return k_ra8_err_invalid_size;
113 }
114
115 for (uint8_t i = 0U; i < ssid_len; i++) {
116 cfg->ssid[i] = ssid[i];
117 }
118 for (uint8_t i = 0U; i < pass_len; i++) {
119 cfg->pass[i] = pass[i];
120 }
121 cfg->ssid_len = ssid_len;
122 cfg->pass_len = pass_len;
123 return k_ra8_ok;
124}
