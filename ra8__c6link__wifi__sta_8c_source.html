99ra8_err_t ra8_c6link_sta_cfg_set(ra8_c6link_sta_cfg_t* cfg, const char* ssid, const char* pass)
100{
101 if ((cfg == nullptr) || (ssid == nullptr)) {
102 return k_ra8_err_null_ptr;
103 }
104 *cfg = (ra8_c6link_sta_cfg_t){};
105
106 const uint8_t ssid_len = ra8_c6link_sta_len(ssid, (uint8_t)sizeof cfg->ssid);
107 if ((ssid_len == 0U) || (ssid_len > (uint8_t)k_ra8_c6link_ssid_max)) {
108 return k_ra8_err_invalid_size;
109 }
110 const uint8_t pass_len =
111 (pass == nullptr) ? 0U : ra8_c6link_sta_len(pass, (uint8_t)sizeof cfg->pass);
112 if (pass_len > (uint8_t)k_ra8_c6link_pass_max) {
113 return k_ra8_err_invalid_size;
114 }
115
116 for (uint8_t i = 0U; i < ssid_len; i++) {
117 cfg->ssid[i] = ssid[i];
118 }
119 for (uint8_t i = 0U; i < pass_len; i++) {
120 cfg->pass[i] = pass[i];
121 }
122 cfg->ssid_len = ssid_len;
123 cfg->pass_len = pass_len;
124 return k_ra8_ok;
125}

280 }
281 return ra8_c6link_priv_bare_req(link, (uint32_t)RPC_ID__Req_WifiConnect);
