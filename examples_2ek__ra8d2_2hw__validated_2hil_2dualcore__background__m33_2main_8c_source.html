267int main(void)
268{
269 ra8_log_init();
270 ra8_log_info("M85", "==== RA8D2 dualcore_background_m33 demo ====");
271 ra8_log_info("M85", "Cortex-M85 primary core online");
272 ra8_log_info("M85", "shared block in SRAM at 0x22100000");
273
274 /* Bring up the VCOM console so the success path can emit the HIL banner.
275 * Best-effort: a failure is narrated over ITM and the demo continues. */
276 if (!dualcore_bg_hil_console_init()) {
277 ra8_log_info("M85", "VCOM console init failed -- HIL banner unavailable");
278 }
279
280 volatile dualcore_bg_t* bg = dualcore_bg();
281 zero_bg(bg);
282
283 ra8_log_info("M85", "releasing Cortex-M33 secondary core ...");
284 const ra8_err_t err = ra8_cpu1_release(&g_ra8_ls_cpu1_mram_start, &g_ra8_ls_cpu1_stack_top);
285 ra8_log_info_val("M85", "ra8_cpu1_release rc (0 = ok)", (uint32_t)err);
286 if (err != k_ra8_ok) {
287 ra8_log_info("M85", "release FAILED -- halting");
288 park_forever();
289 }
290
291 if (wait_for_m33_sig(bg)) {
292 ra8_log_info("M85", "M33 is alive");
293 } else {
294 ra8_log_info("M85", "M33 signature not seen -- did it boot?");
295 }
296
297 ra8_log_info("M85", "M85 yielding -- waiting for M33 to finish counting ...");
298 if (!wait_for_done(bg)) {
299 ra8_log_info("M85", "M33 done flag not seen -- timed out");
300 park_forever();
301 }
302
303 /* Consumer-side data memory barrier: order the load of `counter` AFTER the
304 * observation of `done == 1`, pairing with the producer's pre-`done` `dsb`
305 * in cpu1_main.c so the M85 reads the final, settled count. Mirrors the
306 * publish/consume barrier discipline used by dualcore_mailbox (ping_once)
307 * and cpu1_pingpong (main). */
308 __asm volatile("dmb" ::: "memory");
309 const uint32_t final_count = bg->counter;
310 ra8_log_info_val("M85", "M33 counted to", final_count);
311
312 if (final_count == (uint32_t)k_bg_target_count) {
313 ra8_log_info("M85", "dualcore_background_m33 PASS");
314 /* Additive HIL banner: the M33 counted autonomously to its target, so
315 * mirror the verdict to VCOM for the Pi rig's uart_scrape to gate. */
316 dualcore_bg_hil_emit_pass();
317 } else {
318 ra8_log_info("M85", "dualcore_background_m33 FAIL -- wrong count");
319 }
320
321 park_forever();
322}
