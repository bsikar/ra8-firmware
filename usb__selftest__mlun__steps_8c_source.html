65void mlun_pattern_fill(uint32_t lun, uint32_t lba, UCHAR* out)
66{
67 for (uint32_t i = 0U; i < (uint32_t)k_mlun_block_size; i++) {
68 const uint32_t v = (lun * (uint32_t)k_mlun_pat_lun_mul) + (lba * (uint32_t)k_mlun_pat_lba_mul) +
69 i + (uint32_t)k_mlun_pat_bias;
70 out[i] = (UCHAR)(v & (uint32_t)k_mlun_byte_mask);
71 }
72}

199[[nodiscard]] static ra8_err_t mlun_print_dec(uint32_t value)
200{
201 uint8_t scratch[k_mlun_dec_chars_u32] = {};
202 uint8_t out[k_mlun_dec_chars_u32] = {};
203 uint8_t count = 0U;
204 uint32_t v = value;
205 if (v == 0U) {
206 out[0] = (uint8_t)'0';
207 return mlun_sci_write(out, 1U);
208 }
209 while (v != 0U) {
210 if (count >= (uint8_t)k_mlun_dec_chars_u32) {
211 break;
212 }
213 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_mlun_dec_radix));
214 v = v / k_mlun_dec_radix;
215 count++;
216 }
217 for (uint8_t i = 0U; i < count; i++) {
218 out[i] = scratch[count - 1U - i];
219 }
220 return mlun_sci_write(out, (uint32_t)count);
221}

242[[nodiscard]] static ra8_err_t mlun_print_hex(uint32_t value, uint8_t digits)
243{
244 uint8_t out[k_mlun_hex_chars_u32] = {};
245 uint8_t width = digits;
246 if (width > (uint8_t)k_mlun_hex_chars_u32) {
247 width = (uint8_t)k_mlun_hex_chars_u32;
248 }
249 for (uint8_t i = 0U; i < width; i++) {
250 const uint8_t shift = (uint8_t)((width - 1U - i) * k_mlun_nibble_bits);
251 out[i] = mlun_nibble_to_hex((value >> shift) & k_mlun_nibble_mask);
252 }
253 return mlun_sci_write(out, (uint32_t)width);
254}

322[[nodiscard]] static ra8_err_t mlun_verify_one(uint32_t lun)
323{
324 static uint8_t s_burst[k_mlun_burst_bytes] = {};
325 static uint8_t s_expect[k_mlun_block_size] = {};
326
327 uint32_t block_count = 0U;
328 uint32_t block_size = 0U;
329 ra8_err_t err = ra8_usb_hmsc_read_capacity((uint8_t)lun, &block_count, &block_size);
330 if (err != k_ra8_ok) {
331 (void)mlun_print_fail("read_capacity", err);
332 return err;
333 }
334 if (block_count != (uint32_t)k_mlun_sectors) {
335 (void)mlun_print_fail("capacity mismatch", k_ra8_err_invalid_size);
336 return k_ra8_err_invalid_size;
337 }
338 for (uint32_t blk = 0U; blk < (uint32_t)k_mlun_sectors; blk += (uint32_t)k_mlun_burst_blocks) {
339 err = ra8_usb_hmsc_read10((uint8_t)lun, blk, (uint16_t)k_mlun_burst_blocks, s_burst);
340 if (err != k_ra8_ok) {
341 (void)mlun_print_fail("READ(10)", err);
342 return err;
343 }
344 for (uint32_t s = 0U; s < (uint32_t)k_mlun_burst_blocks; s++) {
345 mlun_pattern_fill(lun, blk + s, s_expect);
346 const uint32_t boff = s * (uint32_t)k_mlun_block_size;
347 if (memcmp(&s_burst[boff], s_expect, (size_t)k_mlun_block_size) != 0) {
348 s_dbg_mismatch = (lun << (uint32_t)k_mlun_mismatch_lun_shift) | (blk + s);
349 (void)mlun_print_fail("LUN data mismatch", k_ra8_err_invalid_state);
350 return k_ra8_err_invalid_state;
351 }
352 }
353 }
354 return k_ra8_ok;
355}

453[[nodiscard]] static ra8_err_t mlun_host_pass(void)
454{
455 s_dbg_phase = (uint32_t)k_mlun_phase_init;
456 ra8_err_t err = mlun_print("ra8d2 mlun: host up on USB-HS, probing the loop...\r\n");
457 if (err != k_ra8_ok) {
458 return err;
459 }
460 err = ra8_usb_hmsc_init(k_ra8_usb_speed_hs);
461 if (err != k_ra8_ok) {
462 (void)mlun_print_fail("host init", err);
463 return err;
464 }
465
466 ra8_usb_hmsc_device_t device = {};
467 err = mlun_host_enumerate(&device);
468 if (err != k_ra8_ok) {
469 return err;
470 }
471
472 s_dbg_phase = (uint32_t)k_mlun_phase_verify;
473 s_dbg_luns_ok = 0U;
474 for (uint32_t lun = 0U; lun < (uint32_t)k_mlun_count; lun++) {
475 err = mlun_verify_one(lun);
476 if (err != k_ra8_ok) {
477 (void)ra8_usb_hmsc_close();
478 return err;
479 }
480 s_dbg_luns_ok++;
481 err = mlun_print_lun_ok(lun);
482 if (err != k_ra8_ok) {
483 return err;
484 }
485 }
486
487 s_dbg_phase = (uint32_t)k_mlun_phase_pass;
488 s_dbg_pass_count++;
489 err = mlun_print("ra8d2 mlun: USB SELFTEST MULTI-LUN PASS\r\n");
490 if (err != k_ra8_ok) {
491 return err;
492 }
493 (void)ra8_board_led_on(k_ra8_board_led2);
494 return k_ra8_ok;
495}

497VOID mlun_host_worker(ULONG arg)
498{
499 (void)arg;
500
501 tx_thread_sleep(k_mlun_boot_wait_ticks);
502 for (;;) {
503 const ra8_err_t err = mlun_host_pass();
504 if (err == k_ra8_ok) {
505 break;
506 }
507 tx_thread_sleep(k_mlun_retry_ticks);
508 }
509 while (1) {
510 tx_thread_sleep(k_mlun_idle_ticks);
511 }
512}
