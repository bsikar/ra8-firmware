232static inline volatile uint8_t* ra8_icu_irqcr(uint8_t irq_num)
233{
234 if (irq_num >= (uint8_t)k_ra8_icu_num_irqs) {
235 return nullptr;
236 }
237 /* HUM Ch 14.2.12 "IRQCRi : IRQ Control Register", p 535: IRQCRa
238 * occupies bytes 0..15, NMICR + reserved fill 16..19, IRQCRb starts
239 * at 20. Match FSP ICU_IRQCR_CH(c) = c + ((c >> 4) << 2). */
240 const uintptr_t gap_bytes = ((uintptr_t)irq_num >= (uintptr_t)k_ra8_icu_irqcrb_first)
241 ? (uintptr_t)k_ra8_icu_irqcrb_offset
242 : (uintptr_t)0U;
243 return (volatile uint8_t*)((uintptr_t)k_ra8_icu_base_addr + (uintptr_t)k_ra8_icu_off_irqcra0 +
244 (uintptr_t)irq_num + gap_bytes);
245}
