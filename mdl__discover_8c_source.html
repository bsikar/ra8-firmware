45RA8_INTERNAL static bool discover_url(const mdl_discover_req_t* req, char* out, size_t cap, int* rc)
46{
47 *rc = 2;
48 if (req->mode == k_mdl_discover_browse) {
49 if (req->site->browse_url[0] == '\0') {
50 (void)fprintf(stderr,
51 "media_dl: this site descriptor has no browse_url; browsing is "
52 "unavailable\n");
53 return false;
54 }
55 const int n = snprintf(out, cap, "%s", req->site->browse_url);
56 if ((n < 0) || ((size_t)n >= cap)) {
57 (void)fprintf(stderr, "media_dl: browse_url is too long\n");
58 return false;
59 }
60 return true;
61 }
62 if (req->site->search_url[0] == '\0') {
63 (void)fprintf(stderr,
64 "media_dl: this site descriptor has no search_url; searching is "
65 "unavailable\n");
66 return false;
67 }
68 if ((req->term == nullptr) || (req->term[0] == '\0')) {
69 (void)fprintf(stderr, "media_dl: --search needs a non-empty TERM\n");
70 return false;
71 }
72 char encoded[k_disc_term_max];
73 if (!mdl_query_encode(req->term, encoded, sizeof(encoded))) {
74 (void)fprintf(stderr, "media_dl: search term is too long to encode\n");
75 return false;
76 }
77 if (!mdl_search_build_url(req->site->search_url, encoded, out, cap)) {
78 (void)fprintf(stderr,
79 "media_dl: search_url '%s' has no %s placeholder (or is too long)\n",
80 req->site->search_url,
81 mdl_search_placeholder());
82 return false;
83 }
84 return true;
85}

92RA8_INTERNAL static ra8_err_t
93discover_fetch(const mdl_discover_req_t* req, const char* url, size_t* out_len, long* out_status)
94{
95 uint32_t crawl = 0U;
96 if (!mdl_session_url_allowed(req->session, url, &crawl)) {
97 return k_ra8_fail; /* robots refused (message already printed) */
98 }
99 char hostbuf[k_mdl_gov_host_max];
100 const char* host = mdl_url_host(url, hostbuf, sizeof(hostbuf)) ? hostbuf : nullptr;
101 const uint32_t jmin = max_u32(req->site->chapter_delay_min, crawl);
102 const uint32_t jmax = max_u32(req->site->chapter_delay_max, crawl);
103 const mdl_net_req_t nreq = {.user_agent = req->session->user_agent,
104 .referer = nullptr,
105 .timeout_ms = req->timeout_ms};
106 ra8_err_t rc = k_ra8_fail;
107 for (uint8_t attempt = 0U; attempt < (uint8_t)k_disc_max_attempts; ++attempt) {
108 if (mdl_governor_acquire(req->gov, host, jmin, jmax) == k_ra8_err_would_block) {
109 rc = k_ra8_err_would_block;
110 continue; /* no slot reserved, so nothing to release */
111 }
112 mdl_net_resp_t resp = {};
113 rc =
114 mdl_net_get_buf(req->session->net, url, &nreq, req->page_buf, req->page_cap, out_len, &resp);
115 mdl_governor_observe(req->gov, host, resp.status, resp.retry_after);
116 mdl_governor_release(req->gov, host);
117 *out_status = resp.status;
118 if (!mdl_fetch_is_retryable(rc)) {
119 break;
120 }
121 }
122 return rc;
123}

176RA8_INTERNAL static bool
177select_hit(const mdl_hit_t* chosen, size_t pick, char* out_url, size_t out_cap)
178{
179 const int n = snprintf(out_url, out_cap, "%s", chosen->url);
180 if ((n < 0) || ((size_t)n >= out_cap)) {
181 (void)fprintf(stderr, "media_dl: selected URL is too long to hand off\n");
182 if (out_cap > 0U) {
183 out_url[0] = '\0';
184 }
185 return false;
186 }
187 (void)printf("selected [%zu] %s\n %s\n", pick, chosen->title, chosen->url);
188 return true;

192RA8_INTERNAL static int present_and_pick(const mdl_discover_req_t* req,
193 mdl_search_outcome_t outcome,
194 size_t pick,
195 char* out_url,
196 size_t out_cap)
197{
198 const mdl_hit_list_t* hits = req->hits;
199 if (outcome == k_mdl_search_have_results) {
200 print_header(req, hits->count);
201 print_hits(hits);
202 (void)printf(" (add --pick N to download %s N, e.g. --pick 1 --format cbz)\n",
203 (req->mode == k_mdl_discover_search) ? "result" : "update");
204 } else if (outcome == k_mdl_search_zero_results) {
205 print_zero(req);
206 } else {
207 print_markup_changed(req);
208 }
209
210 if (pick == 0U) {
211 return (outcome == k_mdl_search_markup_changed) ? 1 : 0;
212 }
213 if ((out_url == nullptr) || (out_cap == 0U)) {
214 (void)fprintf(stderr, "media_dl: --pick needs an output buffer to hand off the selection\n");
215 return 2; /* precondition: a pick requires somewhere to return the URL */
216 }
217 if (outcome != k_mdl_search_have_results) {
218 (void)fprintf(stderr,
219 "media_dl: --pick %zu but the %s returned no results\n",
220 pick,
221 mode_word(req->mode));
222 return 1;
223 }
224 if (pick > hits->count) {
225 (void)fprintf(stderr, "media_dl: --pick %zu out of range (1..%zu)\n", pick, hits->count);
226 return 2;
227 }
228 return select_hit(&hits->hits[pick - 1U], pick, out_url, out_cap) ? 0 : 1;
229}

231int mdl_discover_run(const mdl_discover_req_t* req, size_t pick, char* out_url, size_t out_cap)
232{
233 if ((out_url != nullptr) && (out_cap > 0U)) {
234 out_url[0] = '\0';
235 }
236 if ((req == nullptr) || (req->session == nullptr) || (req->gov == nullptr) ||
237 (req->site == nullptr) || (req->page_buf == nullptr) || (req->hits == nullptr)) {
238 (void)fprintf(stderr, "media_dl: discovery called with missing dependencies\n");
239 return 2;
240 }
241 char url[k_disc_url_max];
242 int rc = 0;
243 if (!discover_url(req, url, sizeof(url), &rc)) {
244 return rc;
245 }
246 size_t len = 0U;
247 long status = 0;
248 const ra8_err_t frc = discover_fetch(req, url, &len, &status);
249 if (frc != k_ra8_ok) {
250 char reason[k_mdl_reason_max];
251 mdl_fetch_reason(frc, status, reason, sizeof(reason));
252 (void)fprintf(stderr, "media_dl: %s request failed -- %s\n", mode_word(req->mode), reason);
253 return 1;
254 }
255 (void)mdl_extract_hits(req->page_buf, len, url, req->site->search_result_contains, req->hits);
256 const mdl_search_outcome_t outcome = mdl_search_classify(req->hits);
257 return present_and_pick(req, outcome, pick, out_url, out_cap);
258}
