280[[gnu::section(".ns_text")]] RA8_INTERNAL static void
281internal_ns_ipc_channel_reset(uintptr_t ch_base)
282{
283 /* HUM Ch 3.2.14 "IPC0CLR0" p 216 -- RST resets the FIFO and clears
284 * RDY/FULL atomically; RCLR/FCLR clear the sticky FIFO error bits;
285 * CLR7..CLR0 drop any latent IRQ event bits. */
286 internal_ns_write32(ch_base + (uintptr_t)k_ns_ipc_off_clr, (uint32_t)k_ns_ipc_clr_all);
287}

304[[gnu::section(".ns_text")]] RA8_INTERNAL static void internal_ns_ipc_send(uintptr_t ch_base,
305 uint32_t msg)
306{
307 /* HUM Ch 3.2.12 "IPC0TXD0" p 215-216 -- writes push the word onto
308 * the FIFO; FERR is raised separately and is cleared by
309 * internal_ns_ipc_channel_reset on init. */
310 internal_ns_write32(ch_base + (uintptr_t)k_ns_ipc_off_txd, msg);
311}

331[[gnu::section(".ns_text")]] RA8_INTERNAL static uint32_t internal_ns_ipc_recv(uintptr_t ch_base,
332 uint32_t* out_word)
333{
334 if (out_word == ((void*)0)) {
335 return 1U;
336 }
337 for (uint32_t i = 0U; i < (uint32_t)k_ns_recv_poll_max; ++i) {
338 const uint32_t sta = internal_ns_read32(ch_base + (uintptr_t)k_ns_ipc_off_sta);
339 if ((sta & (uint32_t)k_ns_ipc_sta_rdy) != 0U) {
340 /* HUM Ch 3.2.13 "IPC0RXD0" p 216 -- reading RXD pops one word
341 * from the FIFO and re-evaluates RDY for the next iteration. */
342 const uint32_t w = internal_ns_read32(ch_base + (uintptr_t)k_ns_ipc_off_rxd);
343 *out_word = w;
344 return 0U;
345 }
346 }
347 return 1U;
348}

384[[gnu::section(".ns_text"), noreturn]] void ns_reset_handler(void)
385{
386 /* Zero the NS BSS region. NOLOAD means the linker does not populate
387 * initial values from MRAM, and cold-boot SRAM contents are
388 * undefined. The address comparison is via uintptr_t because the
389 * two linker-provided extern symbols are different objects from a
390 * static-analysis perspective even though the linker guarantees
391 * they bracket a contiguous BSS run. */
392 const uintptr_t bss_start = (uintptr_t)&g_ra8_ls_ns_bss_start;
393 const uintptr_t bss_end = (uintptr_t)&g_ra8_ls_ns_bss_end;
394 for (uintptr_t a = bss_start; a < bss_end; a += sizeof(uint32_t)) {
395 *(volatile uint32_t*)a = 0U;
396 }
397
398 /* Stamp the entry marker AFTER the bss zero so the bench can tell
399 * "BLXNS reached NS" apart from "marker happened to land back at 0". */
400 g_ns_pingpong_entry_marker = (uint32_t)k_ns_entry_marker;
401
402 g_ns_pingpong_step = (uint32_t)k_ns_step_bss_zeroed;
403
404 /* Cold-init the CPU0-owned channels. */
405 internal_ns_ipc_channel_reset((uintptr_t)k_ns_ipc_ch0_addr);
406 internal_ns_ipc_channel_reset((uintptr_t)k_ns_ipc_ch2_addr);
407 g_ns_pingpong_step = (uint32_t)k_ns_step_channels_up;
408
409 for (;;) {
410 g_ns_pingpong_step = (uint32_t)k_ns_step_tx_start;
411 internal_ns_ipc_send((uintptr_t)k_ns_ipc_ch2_addr, (uint32_t)k_ns_magic_ping);
412 g_ns_pingpong_step = (uint32_t)k_ns_step_tx_done;
413
414 uint32_t got = 0U;
415 if (internal_ns_ipc_recv((uintptr_t)k_ns_ipc_ch0_addr, &got) != 0U) {
416 g_ns_pingpong_step = (uint32_t)k_ns_step_rx_timeout;
417 g_ns_pingpong_mismatch += 1U;
418 continue;
419 }
420 g_ns_pingpong_step = (uint32_t)k_ns_step_rx_word;
421 g_ns_pingpong_last_rxd = got;
422 if (got != (uint32_t)k_ns_magic_pong) {
423 g_ns_pingpong_mismatch += 1U;
424 continue;
425 }
426 g_ns_pingpong_step = 4U;
427 g_ns_pingpong_match += 1U;
428 }
429}
