97static uint32_t crc_step_reflected(uint32_t crc, uint8_t byte, uint32_t poly)
98{
99 crc ^= (uint32_t)byte;
100 for (uint32_t i = 0U; i < 8U; i++) {
101 crc = ((crc & 1U) != 0U) ? ((crc >> 1) ^ poly) : (crc >> 1);
102 }
103 return crc;
104}

107static uint32_t crc_step_forward(uint32_t crc, uint8_t byte, uint32_t poly, uint32_t width)
108{
109 const uint32_t top = 1U << (width - 1U);
110 const uint32_t msk = (width >= 32U) ? (uint32_t)k_u32_all : ((1U << width) - 1U);
111 crc ^= (uint32_t)byte << (width - 8U);
112 for (uint32_t i = 0U; i < 8U; i++) {
113 const bool set = (crc & top) != 0U;
114 crc = (crc << 1) & msk;
115 if (set) {
116 crc ^= poly;
117 }
118 }
119 return crc & msk;
120}

123static void crc_fold_byte(uint8_t byte)
124{
125 switch ((uint32_t)(s_crc.cr0 & (uint8_t)k_crc_gps_mask)) {
126 case (uint32_t)k_crc_gps_8:
127 s_crc.dor = crc_step_forward(s_crc.dor, byte, (uint32_t)k_crc_poly_8_fwd, 8U);
128 break;
129 case (uint32_t)k_crc_gps_16:
130 s_crc.dor = crc_step_reflected(s_crc.dor, byte, (uint32_t)k_crc_poly_16_refl);
131 break;
132 case (uint32_t)k_crc_gps_ccitt:
133 s_crc.dor = crc_step_forward(s_crc.dor, byte, (uint32_t)k_crc_poly_ccitt_fwd, 16U);
134 break;
135 case (uint32_t)k_crc_gps_32:
136 s_crc.dor = crc_step_reflected(s_crc.dor, byte, (uint32_t)k_crc_poly_32_refl);
137 break;
138 case (uint32_t)k_crc_gps_32c:
139 s_crc.dor = crc_step_reflected(s_crc.dor, byte, (uint32_t)k_crc_poly_32c_refl);
140 break;
141 default:
142 break; /* GPS=0: no calculation. */
143 }
144 s_crc.bytes++;
145}

157static uint64_t crc_read(uc_engine* uc, uint64_t addr, unsigned size)
158{
159 (void)uc;
160 (void)size;
161 const uint64_t off = addr - (uint64_t)k_crc_base;
162 if (off == (uint64_t)k_crc_off_cr0) {
163 return s_crc.cr0;
164 }
165 if (off == (uint64_t)k_crc_off_cr1) {
166 return s_crc.cr1;
167 }
168 if ((off >= (uint64_t)k_crc_off_dor) && (off < (uint64_t)k_crc_off_dor + 4UL)) {
169 /* CRCDOR and its 16/8-bit aliases share +0x08; serve the low bytes. */
170 const uint32_t shift = (uint32_t)(off - (uint64_t)k_crc_off_dor) * 8U;
171 return (uint64_t)(s_crc.dor >> shift);
172 }
173 return 0U;
174}

177static void crc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
178{
179 (void)uc;
180 const uint64_t off = addr - (uint64_t)k_crc_base;
181 if (off == (uint64_t)k_crc_off_cr0) {
182 s_crc.cr0 = (uint8_t)value;
183 if (((uint8_t)value & (uint8_t)k_crc_dorclr) != 0U) {
184 s_crc.dor = 0U; /* DORCLR clears the running remainder. */
185 }
186 return;
187 }
188 if (off == (uint64_t)k_crc_off_cr1) {
189 s_crc.cr1 = (uint8_t)value;
190 return;
191 }
192 if (off == (uint64_t)k_crc_off_dir) {
193 /* Fold each byte of the access LSB-first: a 32-bit word feed (CRC-32/32C)
194 * delivers four bytes in the order ra8_crc's word packing produces, an
195 * 8-bit feed (CRC-8/16/CCITT) delivers one. */
196 unsigned n = size;
197 if (size > 4U) {
198 n = 4U;
199 } else if (size == 0U) {
200 n = 1U;
201 }
202 for (unsigned i = 0U; i < n; i++) {
203 crc_fold_byte((uint8_t)((value >> (8U * i)) & (uint32_t)k_byte_mask));
204 }
205 return;
206 }
207 if ((off >= (uint64_t)k_crc_off_dor) && (off < (uint64_t)k_crc_off_dor + 4UL)) {
208 /* Seed / overwrite CRCDOR (driver pre-seeds 0xFFFFFFFF for CRC-32/32C). */
209 if (size >= 4U) {
210 s_crc.dor = (uint32_t)value;
211 } else {
212 const uint32_t shift = (uint32_t)(off - (uint64_t)k_crc_off_dor) * 8U;
213 const uint32_t mask = ((size == 2U) ? (uint32_t)k_u16_mask : (uint32_t)k_byte_mask) << shift;
214 s_crc.dor = (s_crc.dor & ~mask) | (((uint32_t)value << shift) & mask);
215 }
216 }
217}
