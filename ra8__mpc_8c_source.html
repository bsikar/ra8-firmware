121ra8_err_t ra8_mpc_route_peripheral(ra8_port_t port, ra8_pin_t pin, ra8_mpc_psel_t psel)
122{
123 const ra8_err_t err = internal_check(port, pin);
124 if (err != k_ra8_ok) {
125 return err;
126 }
127 const uint32_t psel_bits = ((uint32_t)psel << k_ra8_pfs_bit_psel0) & k_ra8_pfs_mask_psel;
128 const uint32_t pmr = k_ra8_pfs_mask_pmr;
129
130 /* HUM Ch 20.2.4 "Notes on the PmnPFS Register Setting" p 859 -- PMR
131 * must be cleared before a new PSEL is programmed. Three-step write:
132 * 1. PMR := 0 (return pin to GPIO mode)
133 * 2. PFS := new PSEL (programme function while PMR is still 0)
134 * 3. PFS := PSEL | PMR (hand pin to the peripheral)
135 * Single combined write is what FSP r_ioport_pfs_write avoids; doing
136 * it would let the pin briefly drive whatever the *previous* PSEL
137 * decoded to with PMR already 1.
138 * @note PMSAR resets to all-Secure on RA8D2; we hold the unlock
139 * across the whole sequence so no extra PWPR write storms. */
140 volatile uint32_t* pfs = ra8_pfs_pmn(port, pin);
141 if (pfs == nullptr) { /* GCOVR_EXCL_BR_LINE -- bounds already validated */
142 return k_ra8_err_gpio_invalid_pin;
143 }
144 ra8_pfs_pwpr_unlock();
145 /* HUM Ch 20.2.4 "Notes on the PmnPFS Register Setting" p 859 */
146 *pfs = 0U; /* step 1: PMR = 0. */
147 *pfs = psel_bits; /* step 2: PSEL, PMR=0. */
148 *pfs = psel_bits | pmr; /* step 3: PSEL | PMR=1. */
149 ra8_pfs_pwpr_lock();
150
151 ra8_log_info_val(s_tag, "route", (uint32_t)psel);
152 return k_ra8_ok;
153}
