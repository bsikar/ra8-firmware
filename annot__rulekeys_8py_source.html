3"""The annotation vocabulary, and the proof that it matches what the macros emit.
4
5This module is small on purpose. It holds the set of rule keys the checker
6dispatches on, and one cross-check -- :func:`check_rule_keys` -- that compares
7that set against the strings ``ra8_attributes.h`` actually writes.
8
9That cross-check is the reason the vocabulary is not just spelled inline where
10it is used. A rule keyed on a spelling no macro produces matches zero symbols
11and reports zero violations forever, which is indistinguishable from a clean
12tree. Four rules in this checker were in exactly that state at once. Keeping
13the keys and their proof in one file makes the two impossible to edit apart.
14"""
15

26#: the scan roots it is never re-pointed at a synthetic tree.

67def parse_annotation(ann: str) -> tuple[str, str]:
68 """Split ``ra8_max_stack:512`` -> (``ra8_max_stack``, ``512``)."""
69 if ":" in ann:
70 rule, _, arg = ann.partition(":")
71 return rule.strip(), arg.strip()
72 return ann.strip(), ""

75def emitted_annotation_keys() -> set[str]:
76 """Return every annotation string ``ra8_attributes.h`` can emit.
77
78 Read straight out of the header rather than restated here, because a
79 restatement is what goes stale. Each macro expands through
80 ``RA8_INTERNAL_ANNOTATE("ra8_<rule>...")``; the rule key is the text
81 up to the first colon.
82 """
83 try:
84 text = ATTRIBUTES_HEADER.read_text(errors="ignore")
85 except OSError:
86 return set()
87 return {m.group(1) for m in re.finditer(r'RA8_INTERNAL_ANNOTATE\‍(\s*"(ra8_[a-z0-9_]+)', text)}

90def check_rule_keys() -> list[Violation]:
91 """Fail when a rule keys on a string no annotation macro emits.
92
93 This is the failure mode that looks exactly like success. A rule
94 keyed on a spelling nothing produces matches zero symbols and reports
95 zero violations for as long as nobody checks, and four rules in this
96 file were in that state at once: RA8_HW_REGISTER_ACCESS emits
97 "ra8_hw_register_access" but rule 6 looked for "ra8_hw_mmio", and the
98 NASA-rule-3, stack-budget and latency-budget rules each looked for a
99 key their macro never wrote. Cross-checking both directions against
100 the header makes the whole class impossible to reintroduce silently.
101 """
102 emitted = emitted_annotation_keys()
103 if not emitted:
104 return [
105 Violation(
106 "ra8_rule_keys",
107 str(ATTRIBUTES_HEADER),
108 0,
109 "no RA8_INTERNAL_ANNOTATE() strings found -- the annotation "
110 "header moved or changed shape, so every rule key is unverified",
111 )
112 ]
113 known = set(ANNOTATION_PREFIXES)
114 out: list[Violation] = []
115 out.extend(
116 Violation(
117 "ra8_rule_keys",
118 str(ATTRIBUTES_HEADER),
119 0,
120 f"rule key '{key}' is not emitted by any macro in ra8_attributes.h; "
121 f"the rule keyed on it can never match",
122 )
123 for key in sorted(known - emitted)
124 )
125 out.extend(
126 Violation(
127 "ra8_rule_keys",
128 str(ATTRIBUTES_HEADER),
129 0,
130 f"annotation '{key}' is emitted by a macro but no rule recognises it; "
131 f"every use of that macro is silently ignored",
132 )
133 for key in sorted(emitted - known)
134 )
135 return out
