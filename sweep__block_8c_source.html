133static int
134cbs_cache_open(cbs_cache_t* c, cbs_backend_t* be, uint32_t blob_bytes, uint32_t block_bytes)
135{
136 if ((c == nullptr) || (be == nullptr) || (be->read == nullptr) || (block_bytes == 0U)) {
137 return 1;
138 }
139 *c = (cbs_cache_t){};
140 c->frames = (uint32_t)k_cbs_cache_bytes / block_bytes;
141 if (c->frames == 0U) {
142 c->frames = 1U;
143 }
144 uint32_t nbuckets = cbs_pow2_ceil(c->frames * 2U);
145 if (nbuckets < (uint32_t)k_cbs_bucket_min) {
146 nbuckets = (uint32_t)k_cbs_bucket_min;
147 }
148 c->frame_mem = (uint8_t*)malloc((size_t)c->frames * (size_t)block_bytes);
149 c->meta = (ra8_vmem_frame_t*)calloc((size_t)c->frames, sizeof(ra8_vmem_frame_t));
150 c->keys = (ra8_vmem_key_t*)calloc((size_t)c->frames, sizeof(ra8_vmem_key_t));
151 c->buckets = (int32_t*)malloc((size_t)nbuckets * sizeof(int32_t));
152 if ((c->frame_mem == nullptr) || (c->meta == nullptr) || (c->keys == nullptr) ||
153 (c->buckets == nullptr)) {
154 cbs_cache_close(c);
155 return 1;
156 }
157 c->meter = (cbs_meter_t){.inner = be->read, .inner_ctx = be->read_ctx};
158 if (ra8_vsource_init(&c->vs, c->objs, 1U) != k_ra8_ok) {
159 cbs_cache_close(c);
160 return 1;
161 }
162 if (ra8_vsource_add_paged(&c->vs,
163 cbs_priv_meter_read,
164 &c->meter,
165 0U,
166 blob_bytes,
167 &c->object_id) != k_ra8_ok) {
168 cbs_cache_close(c);
169 return 1;
170 }
171 const ra8_vmem_cfg_t cfg = {.frame_mem = c->frame_mem,
172 .frame_bytes = block_bytes,
173 .frame_count = c->frames,
174 .meta = c->meta,
175 .keys = c->keys,
176 .buckets = c->buckets,
177 .bucket_count = nbuckets,
178 .loader = ra8_vsource_loader,
179 .loader_ctx = &c->vs};
180 if (ra8_vmem_init(&c->vm, &cfg) != k_ra8_ok) {
181 cbs_cache_close(c);
182 return 1;
183 }
184 return 0;
185}

214static int cbs_drive(cbs_cache_t* c,
215 const uint8_t* blob,
216 uint64_t n_reads,
217 uint64_t wrap_bytes,
218 cbs_row_t* row)
219{
220 if ((c == nullptr) || (blob == nullptr) || (row == nullptr) || (wrap_bytes == 0U)) {
221 return 1;
222 }
223 const uint64_t req = (uint64_t)k_cbs_req_bytes;
224 const uint64_t block = (uint64_t)c->vm.cfg.frame_bytes;
225 uint64_t bad = 0U;
226 const uint64_t t0 = cbs_priv_now_ns();
227 for (uint64_t i = 0U; i < n_reads; ++i) {
228 const uint64_t off = (i * req) % wrap_bytes;
229 void* page = nullptr;
230 if (ra8_vmem_get(&c->vm, c->object_id, off, &page) != k_ra8_ok) {
231 bad++;
232 continue;
233 }
234 const uint8_t* piece = &((const uint8_t*)page)[off % block];
235 if (memcmp(piece, &blob[off], (size_t)req) != 0) {
236 bad++;
237 }
238 if (ra8_vmem_put(&c->vm, page) != k_ra8_ok) {
239 bad++;
240 }
241 }
242 row->wall_ns = cbs_priv_now_ns() - t0;
243 row->reads = n_reads;
244 uint32_t hits = 0U;
245 uint32_t miss = 0U;
246 uint32_t evic = 0U;
247 if (ra8_vmem_stats(&c->vm, &hits, &miss, &evic) != k_ra8_ok) {
248 return 1;
249 }
250 row->hits = (uint64_t)hits;
251 row->misses = (uint64_t)miss;
252 row->evictions = (uint64_t)evic;
253 if (bad != 0U) {
254 (void)fprintf(stderr, "sweep-block: %llu bad accesses\n", (unsigned long long)bad);
255 return 1;
256 }
257 return 0;
258}

300static int cbs_run_block(cbs_backend_t* be,
301 const uint8_t* blob,
302 uint32_t block_bytes,
303 cbs_row_t* rows,
304 uint32_t* nrows)
305{
306 if ((be == nullptr) || (blob == nullptr) || (rows == nullptr) || (nrows == nullptr)) {
307 return 1;
308 }
309 if (be->setup(be, blob, (uint32_t)k_cbs_blob_bytes, block_bytes) != 0) {
310 return 1;
311 }
312 int rc = 0;
313 for (uint8_t leg = 0U; leg < (uint8_t)k_cbs_leg_count; ++leg) {
314 cbs_cache_t c = {};
315 if (cbs_cache_open(&c, be, (uint32_t)k_cbs_blob_bytes, block_bytes) != 0) {
316 rc = 1;
317 break;
318 }
319 const uint64_t src0 = (be->src_bytes != nullptr) ? *be->src_bytes : 0U;
320 const bool seq = (leg == (uint8_t)k_cbs_leg_seq);
321 const uint64_t n =
322 seq ? ((uint64_t)k_cbs_seq_passes * (uint64_t)k_cbs_blob_bytes / (uint64_t)k_cbs_req_bytes)
323 : (uint64_t)k_cbs_hot_reads;
324 const uint64_t wrap = seq ? (uint64_t)k_cbs_blob_bytes : (uint64_t)block_bytes;
325 cbs_row_t* row = &rows[*nrows];
326 *row = (cbs_row_t){.backend = be->name,
327 .leg = s_cbs_leg_names[leg],
328 .block_bytes = block_bytes,
329 .frames = c.frames,
330 .backing_bytes = be->backing_bytes};
331 const int drc = cbs_drive(&c, blob, n, wrap, row);
332 row->be_calls = c.meter.calls;
333 row->be_bytes = c.meter.bytes;
334 row->src_bytes = (be->src_bytes != nullptr) ? (*be->src_bytes - src0) : c.meter.bytes;
335 cbs_cache_close(&c);
336 if (drc != 0) {
337 rc = 1;
338 break;
339 }
340 (*nrows)++;
341 cbs_priv_print_row(row);
342 }
343 be->teardown(be);
344 return rc;
345}

363int cb_sweep_block(void)
364{
365 uint8_t* blob = (uint8_t*)malloc((size_t)k_cbs_blob_bytes);
366 if (blob == nullptr) {
367 (void)fprintf(stderr, "sweep-block: payload allocation failed\n");
368 /* cppcheck-suppress memleak ; false positive: cppcheck 2.13 does not
369 * model the C23 nullptr keyword, so it cannot see blob is NULL here. */
370 return 1;
371 }
372 cbs_priv_fill_text(blob, (uint32_t)k_cbs_blob_bytes);
373
374 (void)printf("# #208 block/frame-size sweep\n\n");
375 (void)printf("payload=%u cache_budget=%u req=%u seq_passes=%u hot_reads=%u zlib_level=%d "
376 "cache=ra8_vmem(SLRU)\n\n",
377 (unsigned)k_cbs_blob_bytes,
378 (unsigned)k_cbs_cache_bytes,
379 (unsigned)k_cbs_req_bytes,
380 (unsigned)k_cbs_seq_passes,
381 (unsigned)k_cbs_hot_reads,
382 MZ_BEST_COMPRESSION);
383
384 static cbs_row_t s_rows[k_cbs_max_rows] = {};
385 uint32_t nrows = 0U;
386 int rc = 0;
387 uint32_t nbe = 0U;
388 cbs_backend_t* backends = cbs_priv_backends(&nbe);
389 const uint32_t nblocks = (uint32_t)(sizeof(s_cbs_blocks) / sizeof(s_cbs_blocks[0]));
390 for (uint32_t b = 0U; (b < nbe) && (rc == 0); ++b) {
391 for (uint32_t s = 0U; (s < nblocks) && (rc == 0); ++s) {
392 rc = cbs_run_block(&backends[b], blob, s_cbs_blocks[s], s_rows, &nrows);
393 }
394 }
395 if (rc == 0) {
396 (void)printf("\n## Summary (payload 8 MiB, cache budget 1 MiB, %u B reader requests)\n",
397 (unsigned)k_cbs_req_bytes);
398 for (uint32_t b = 0U; b < nbe; ++b) {
399 cbs_priv_print_seq_table(s_rows, nrows, backends[b].name, s_cbs_blocks, nblocks);
400 cbs_priv_print_hot_table(s_rows, nrows, backends[b].name, s_cbs_blocks, nblocks);
401 }
402 cbs_priv_print_crossover(s_rows, nrows, s_cbs_blocks, nblocks);
403 } else {
404 (void)fprintf(stderr, "sweep-block: sweep aborted\n");
405 }
406 free(blob);
407 return rc;
408}
