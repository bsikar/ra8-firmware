236static void usb_audio_u32_to_ascii(uint32_t value, char* buf, uint32_t cap)
237{
238 enum : uint32_t {
239 k_dec_radix = 10U,
240 k_max_digits = 10U,
241 };
242 if (buf == nullptr || cap == 0U) {
243 return;
244 }
245 char tmp[k_max_digits + 1U];
246 uint32_t n = 0U;
247 if (value == 0U) {
248 tmp[n] = '0';
249 n++;
250 } else {
251 while (value > 0U && n < k_max_digits) {
252 const uint32_t digit = value % k_dec_radix;
253 tmp[n] = (char)('0' + (char)digit);
254 value /= k_dec_radix;
255 n++;
256 }
257 }
258 uint32_t out = 0U;
259 while (n > 0U && out < (cap - 1U)) {
260 n--;
261 buf[out] = tmp[n];
262 out++;
263 }
264 buf[out] = '\0';
265}

412int32_t main(void)
413{
414 usb_audio_setup_or_halt();
415 ra8_isr_globals_enable();
416 usb_audio_log("ra8d2: USB Audio device ready (UAC1 48 kHz / 16-bit / stereo)\r\n");
417
418 uint32_t frames = 0U;
419 while (1) {
420 if (usb_audio_send_one_frame() != k_ra8_ok) {
421 break;
422 }
423 frames++;
424 if ((frames % k_usb_audio_log_period) == 0U) {
425 usb_audio_log_frames(frames);
426 }
427 ra8_delay_ms(k_usb_audio_idle_step_ms);
428 }
429
430 usb_audio_panic_halt();
431 return 0;
432}
