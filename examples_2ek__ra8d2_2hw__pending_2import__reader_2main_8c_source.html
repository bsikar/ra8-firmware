257static void imp_print_uint(uint32_t value)
258{
259 uint8_t buf[k_imp_dec_base];
260 uint32_t n = 0U;
261 if (value == 0U) {
262 buf[n] = '0';
263 n++;
264 }
265 while ((value > 0U) && (n < (uint32_t)k_imp_dec_base)) {
266 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_imp_dec_base));
267 n++;
268 value /= (uint32_t)k_imp_dec_base;
269 }
270 for (uint32_t i = 0U; i < n; i++) {
271 imp_print(&buf[n - 1U - i], 1U);
272 }
273}

511[[nodiscard]] static ra8_err_t
512imp_read_cache(ra8_fs_mount_t* mount, const char* path, uint32_t* out_len)
513{
514 ra8_fs_file_t* file = nullptr;
515 ra8_err_t err = ra8_fs_open(mount, path, k_ra8_fs_mode_read, &file);
516 if (err != k_ra8_ok) {
517 return err;
518 }
519 uint32_t size = 0U;
520 err = ra8_fs_size(file, &size);
521 if (err != k_ra8_ok) {
522 (void)ra8_fs_close(file);
523 return err;
524 }
525 if (size > (uint32_t)sizeof(s_imp_readback)) {
526 (void)ra8_fs_close(file);
527 return k_ra8_err_no_mem;
528 }
529 uint32_t got = 0U;
530 err = ra8_fs_read(file, s_imp_readback, (uint32_t)sizeof(s_imp_readback), &got);
531 (void)ra8_fs_close(file);
532 if (err != k_ra8_ok) {
533 return err;
534 }
535 *out_len = got;
536 return k_ra8_ok;
537}

553[[nodiscard]] static ra8_err_t imp_walk_book(uint32_t len)
554{
555 const void* base = (const void*)s_imp_readback;
556 ra8_err_t err = ra8_book_validate(base, (size_t)len);
557 if (err != k_ra8_ok) {
558 return err;
559 }
560 const ra8_book_header_t* hdr = ra8_book_header(base);
561 IMP_PUTS(k_msg_title);
562 imp_print_cstr(ra8_book_string(base, hdr->title_off));
563 IMP_PUTS(k_msg_chaps);
564 imp_print_uint(hdr->chapter_count);
565 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
566
567 for (uint32_t ci = 0U; ci < hdr->chapter_count; ci++) {
568 size_t tlen = 0U;
569 ra8_err_t cerr =
570 ra8_book_chapter_text(base, ci, (char*)s_imp_xhtml, sizeof(s_imp_xhtml), &tlen);
571 if (cerr != k_ra8_ok) {
572 return cerr;
573 }
574 IMP_PUTS(k_msg_ch);
575 imp_print_uint(ci);
576 IMP_PUTS(k_msg_textlen);
577 imp_print_uint((uint32_t)tlen);
578 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
579 }
580 return k_ra8_ok;
581}

599static void imp_run(ra8_fs_mount_t* mount)
600{
601 ra8_rabook_import_cfg_t cfg = imp_make_cfg(mount);
602 ra8_rabook_import_outcome_t out = k_ra8_rabook_import_hit;
603 char path1[k_imp_path_cap] = {};
604 char path2[k_imp_path_cap] = {};
605
606 /* First open: cache miss -> compile + write-through. */
607 if (ra8_rabook_import_open(&cfg, k_imp_epub_path, path1, (uint32_t)sizeof(path1), &out) !=
608 k_ra8_ok) {
609 imp_panic("import compile");
610 }
611 if (out != k_ra8_rabook_import_compiled) {
612 imp_panic("expected miss");
613 }
614 IMP_PUTS(k_msg_miss);
615 imp_print_cstr(path1);
616 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
617
618 /* The .rabook now exists on the card. */
619 uint32_t cache_len = 0U;
620 if (imp_read_cache(mount, path1, &cache_len) != k_ra8_ok) {
621 imp_panic("cache read");
622 }
623 IMP_PUTS(k_msg_cbytes);
624 imp_print_uint(cache_len);
625 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
626
627 /* Second open: fresh cache -> hit, the compiler seam is NOT invoked. */
628 if (ra8_rabook_import_open(&cfg, k_imp_epub_path, path2, (uint32_t)sizeof(path2), &out) !=
629 k_ra8_ok) {
630 imp_panic("import reopen");
631 }
632 if (out != k_ra8_rabook_import_hit) {
633 imp_panic("expected hit");
634 }
635 IMP_PUTS(k_msg_hit);
636 imp_print_cstr(path2);
637 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
638
639 /* Read the cached book back and walk it. */
640 if (imp_read_cache(mount, path2, &cache_len) != k_ra8_ok) {
641 imp_panic("cache reread");
642 }
643 if (imp_walk_book(cache_len) != k_ra8_ok) {
644 imp_panic("book walk");
645 }
646}
