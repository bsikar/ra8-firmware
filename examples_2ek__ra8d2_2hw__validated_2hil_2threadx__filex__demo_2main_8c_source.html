180static bool demo_fx_format(void)
181{
182 if (lx_filex_adapter_bind(&s_nor_flash) != FX_SUCCESS) {
183 demo_print("[filex] lx_filex_adapter_bind failed\r\n");

185 }
186 ULONG total_sectors = lx_filex_adapter_get_total_sectors();

189 return false;

191 UINT status = fx_media_format(&s_media,
192 fx_media_driver_ra8_levelx,
193 LX_NULL,
194 s_media_memory,
195 (UINT)sizeof(s_media_memory),
196 s_fx_volume_name,
197 1U, /* number of FATs */
198 32U, /* root directory entries */
199 0U, /* hidden sectors */
200 total_sectors, /* total logical sectors */
201 (UINT)k_demo_sector_bytes, /* bytes per sector */
202 1U, /* sectors per cluster */
203 1U, /* heads (non-rotating) */
204 1U); /* sectors per track */
205 if (status != FX_SUCCESS) {
206 demo_print("[filex] fx_media_format failed\r\n");
207 return false;
208 }
209 if (fx_media_open(&s_media,
210 s_fx_volume_name,
211 fx_media_driver_ra8_levelx,
212 LX_NULL,
213 s_media_memory,
214 (ULONG)sizeof(s_media_memory)) != FX_SUCCESS) {
215 demo_print("[filex] fx_media_open failed\r\n");
216 return false;
217 }
218 return true;

231static bool demo_write_file(char* path, char* text)
232{
233 UINT status = fx_file_create(&s_media, path);
234 if (status != FX_SUCCESS && status != FX_ALREADY_CREATED) {
235 demo_print("[filex] fx_file_create failed\r\n");
236 return false;
237 }
238 if (fx_file_open(&s_media, &s_file, path, FX_OPEN_FOR_WRITE) != FX_SUCCESS) {
239 demo_print("[filex] fx_file_open(write) failed\r\n");
240 return false;
241 }
242 bool ok = fx_file_write(&s_file, text, (ULONG)strlen(text)) == FX_SUCCESS;
243 (void)fx_file_close(&s_file);
244 (void)fx_media_flush(&s_media);
245 if (!ok) {
246 demo_print("[filex] fx_file_write failed\r\n");
247 }
248 return ok;

261static bool demo_read_verify(char* path, const char* expected)
262{
263 if (fx_file_open(&s_media, &s_file, path, FX_OPEN_FOR_READ) != FX_SUCCESS) {
264 demo_print("[filex] fx_file_open(read) failed\r\n");
265 return false;
266 }
267 CHAR buf[k_demo_file_chunk];
268 ULONG actual = 0U;
269 uint32_t off = 0U;
270 bool ok = true;
271 do {
272 actual = 0U;
273 UINT status = fx_file_read(&s_file, buf, (ULONG)k_demo_file_chunk, &actual);
274 if (actual > 0U) {
275 if (memcmp(buf, &expected[off], (size_t)actual) != 0) {
276 ok = false;
277 }
278 off += (uint32_t)actual;
279 }
280 if (status != FX_SUCCESS) {
281 break;
282 }
283 } while (actual > 0U);
284 (void)fx_file_close(&s_file);
285 ok = ok && (off == (uint32_t)strlen(expected));
286 return ok;
287}

338static void demo_file_ops(void)
339{
340 if (!demo_write_file(s_readme_path, s_readme_text)) {
341 return;
342 }
343 if (!demo_write_file(s_scratch_path, s_readme_text)) {
344 return;
345 }
346 demo_list_root(&s_media);
347 if (!demo_read_verify(s_readme_path, s_readme_text)) {
348 demo_print("[filex] readback mismatch\r\n");
349 return;
350 }
351 demo_print("[filex] readback verified\r\n");
352 if (fx_file_delete(&s_media, s_scratch_path) != FX_SUCCESS) {
353 demo_print("[filex] fx_file_delete failed\r\n");
354 return;
355 }
356 (void)fx_media_flush(&s_media);
357 if (fx_file_open(&s_media, &s_file, s_scratch_path, FX_OPEN_FOR_READ) == FX_SUCCESS) {
358 (void)fx_file_close(&s_file);
359 demo_print("[filex] delete did not remove the file\r\n");
360 return;
361 }
362 demo_print("[filex] ospi FAT roundtrip ok\r\n");
363}
