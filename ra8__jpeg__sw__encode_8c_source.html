89static void fwd_dct_1d_norm(const int32_t* in, int32_t* out)
90{
91 for (uint8_t k = 0U; k < (uint8_t)k_ra8_jpeg_block_dim; k++) {
92 int64_t s = 0;
93 for (uint8_t n = 0U; n < (uint8_t)k_ra8_jpeg_block_dim; n++) {
94 s += (int64_t)in[n] * (int64_t)s_dct_cos_q14[k][n];
95 }
96 /* s is Q14*Q0 = Q14; multiply by Q14 weight -> Q28; round to Q0. */
97 int64_t r =
98 ((s * (int64_t)s_dct_w_q14[k]) + (1LL << k_jpeg_idct_p1_bias_sh)) >> k_jpeg_idct_p1_sh;
99 out[k] = (int32_t)r;
100 }
101}

119static void fdct8x8(int32_t* block)
120{
121 int32_t tmp[(uint32_t)k_ra8_jpeg_block_size];
122 int32_t row[(uint32_t)k_ra8_jpeg_block_dim];
123 int32_t col[(uint32_t)k_ra8_jpeg_block_dim];
124 int32_t out[(uint32_t)k_ra8_jpeg_block_dim];
125 for (uint8_t r = 0U; r < (uint8_t)k_ra8_jpeg_block_dim; r++) {
126 for (uint8_t c = 0U; c < (uint8_t)k_ra8_jpeg_block_dim; c++) {
127 row[c] = block[(r * (uint8_t)k_ra8_jpeg_block_dim) + c];

131 tmp[(r * (uint8_t)k_ra8_jpeg_block_dim) + c] = out[c];

135 for (uint8_t r = 0U; r < (uint8_t)k_ra8_jpeg_block_dim; r++) {

228static uint8_t enc_bits_needed(int32_t v)
229{
230 uint32_t a = (v < 0) ? (uint32_t)(-v) : (uint32_t)v;
231 uint8_t n = 0U;
232 while (a != 0U) {
233 a >>= 1U;
234 n++;
235 }

261static int32_t enc_quantize(int32_t v, uint8_t qv)
262{
263 int32_t q = (int32_t)qv;
264 if (q == 0) {
265 q = 1;
266 }
267 if (v < 0) {
268 int32_t a = -v + (q >> 1);
269 return -(a / q);
270 }
271 return (v + (q >> 1)) / q;
272}

299static void enc_block_ac(ra8_jpeg_enc_ctx_t* e,
300 const int32_t* z,
301 const uint16_t* ac_codes,
302 const uint8_t* ac_sizes)
303{
304 uint8_t run = 0U;
305 for (uint8_t k = 1U; k < (uint8_t)k_ra8_jpeg_block_size; k++) {
306 if (z[k] == 0) {
307 run++;
308 continue;
309 }
310 while (run >= 16U) {
311 /* ZRL = 0xF0. */
312 ra8_jpeg_sw_priv_enc_put_bits(e,
313 (uint32_t)ac_codes[k_jpeg_zrl_symbol],
314 ac_sizes[k_jpeg_zrl_symbol]);
315 run = (uint8_t)(run - 16U);
316 }
317 uint8_t mb = enc_bits_needed(z[k]);
318 uint8_t sym = (uint8_t)((run << k_ra8_jpeg_nibble_shift) | mb);
319 ra8_jpeg_sw_priv_enc_put_bits(e, (uint32_t)ac_codes[sym], ac_sizes[sym]);
320 int32_t v = (z[k] < 0) ? (z[k] - 1) : z[k];
321 ra8_jpeg_sw_priv_enc_put_bits(e, (uint32_t)v & ((1U << mb) - 1U), mb);
322 run = 0U;
323 }
324 if (run > 0U) {
325 /* EOB = 0x00. */
326 ra8_jpeg_sw_priv_enc_put_bits(e, (uint32_t)ac_codes[0U], ac_sizes[0U]);
327 }
328}

354static void enc_block(ra8_jpeg_enc_ctx_t* e,
355 const int32_t* spatial,
356 const uint8_t* qtab,
357 uint8_t comp_idx,
358 const uint16_t* dc_codes,
359 const uint8_t* dc_sizes,
360 const uint16_t* ac_codes,
361 const uint8_t* ac_sizes)
362{
363 int32_t blk[(uint32_t)k_ra8_jpeg_block_size];
364 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_block_size; i++) {
365 blk[i] = spatial[i];
366 }
367 fdct8x8(blk);
368
369 /* Quantize in zig-zag order. */
370 int32_t z[(uint32_t)k_ra8_jpeg_block_size];
371 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_block_size; i++) {
372 z[i] = enc_quantize(blk[s_zigzag[i]], qtab[s_zigzag[i]]);
373 }
374
375 /* DC differential. */
376 int32_t diff = z[0] - e->dc_pred[comp_idx];
377 e->dc_pred[comp_idx] = z[0];
378 uint8_t nb = enc_bits_needed(diff);
379 ra8_jpeg_sw_priv_enc_put_bits(e, (uint32_t)dc_codes[nb], dc_sizes[nb]);
380 if (nb != 0U) {
381 int32_t v = (diff < 0) ? (diff - 1) : diff;
382 ra8_jpeg_sw_priv_enc_put_bits(e, (uint32_t)v & ((1U << nb) - 1U), nb);
383 }
384
385 /* AC. */
386 enc_block_ac(e, z, ac_codes, ac_sizes);
387}

414static void enc_rgb_to_ycc_row(const uint8_t* rgb, uint16_t n, int32_t* y, int32_t* cb, int32_t* cr)
415{
416 for (uint16_t i = 0U; i < n; i++) {
417 uint32_t px = (uint32_t)i * (uint32_t)k_ra8_jpeg_rgb_components;
418 int32_t r = (int32_t)rgb[px];
419 int32_t g = (int32_t)rgb[px + 1U];
420 int32_t b = (int32_t)rgb[px + 2U];
421 y[i] = (((int32_t)k_ra8_jpeg_yr * r) + ((int32_t)k_ra8_jpeg_yg * g) +
422 ((int32_t)k_ra8_jpeg_yb * b)) >>

426 k_ra8_jpeg_yuv_shift) +

508{
509 int32_t s = 0;

523 }
524 }

713static ra8_err_t
714enc_run(ra8_jpeg_enc_ctx_t* e, const uint8_t* rgb, uint16_t w, uint16_t h, uint8_t quality)
715{
716 /* Build quantization tables. */
717 uint16_t qs = enc_quality_scale(quality);
718 enc_scale_qtab(s_quant_luma, e->qy, qs);
719 enc_scale_qtab(s_quant_chroma, e->qc, qs);
720
721 /* Build Huffman code tables from K.3.3 BITS/VALUES. */
722 uint16_t total_dc_l;
723 uint16_t total_ac_l;
724 uint16_t total_dc_c;
725 uint16_t total_ac_c;
726 ra8_jpeg_sw_priv_enc_build_huff(e, &total_dc_l, &total_ac_l, &total_dc_c, &total_ac_c);
727
728 ra8_jpeg_sw_priv_enc_headers(e, w, h, total_dc_l, total_ac_l, total_dc_c, total_ac_c);
729
730 /* Convert source rows to YCbCr 16-row strips. Static buffers
731 * keep the host stack small while still respecting NASA Rule 3
732 * (no heap). The encoder is single-threaded, so file-scope
733 * sharing is safe. */
734 uint16_t pad_w = (uint16_t)((w + k_jpeg_mcu_align) & ~k_jpeg_mcu_align);
735 uint16_t pad_h = (uint16_t)((h + k_jpeg_mcu_align) & ~k_jpeg_mcu_align);
736 if ((uint32_t)pad_w > (uint32_t)k_ra8_jpeg_enc_max_w) {
737 return k_ra8_err_invalid_arg;
738 }
739 static int32_t s_y_strip[16U * (uint32_t)k_ra8_jpeg_enc_max_w];
740 static int32_t s_cb_strip[16U * (uint32_t)k_ra8_jpeg_enc_max_w];
741 static int32_t s_cr_strip[16U * (uint32_t)k_ra8_jpeg_enc_max_w];
742 static uint8_t s_tmp_rgb[3U * (uint32_t)k_ra8_jpeg_enc_max_w];
743
744 for (uint16_t mby = 0U; mby < pad_h; mby = (uint16_t)(mby + 16U)) {
745 enc_convert_strip_to_ycc(rgb, w, h, pad_w, mby, s_y_strip, s_cb_strip, s_cr_strip, s_tmp_rgb);
746 enc_encode_mcu_row(e, pad_w, s_y_strip, s_cb_strip, s_cr_strip);
747 }
748
749 ra8_jpeg_sw_priv_enc_flush_bits(e);
750 ra8_jpeg_sw_priv_enc_emit_u16(e, (uint16_t)k_ra8_jpeg_marker_eoi);
751
752 if (e->overflow) {
753 return k_ra8_err_invalid_size;
754 }
755 return k_ra8_ok;
756}

758ra8_err_t ra8_jpeg_sw_encode(const uint8_t* rgb_buf,
759 uint16_t width,
760 uint16_t height,
761 uint8_t quality,
762 uint8_t* out_buf,
763 uint32_t out_buf_len,
764 uint32_t* out_len)
765{
766 RA8_CHECK_NULL_PTR(rgb_buf, s_tag, "rgb_buf is NULL");
767 RA8_CHECK_NULL_PTR(out_buf, s_tag, "out_buf is NULL");
768 RA8_CHECK_NULL_PTR(out_len, s_tag, "out_len is NULL");
769 *out_len = 0U;
770 if (width == 0U || height == 0U) {
771 return k_ra8_err_invalid_arg;
772 }
773 if (quality < (uint8_t)k_ra8_jpeg_sw_quality_min ||
774 quality > (uint8_t)k_ra8_jpeg_sw_quality_max) {
775 return k_ra8_err_invalid_arg;
776 }
777
778 /* Encoder context contains 2KiB of Huffman code/size LUTs; allocate
779 * static to avoid the project's stack-usage budget firing. */
780 static ra8_jpeg_enc_ctx_t s_e;
781 ra8_jpeg_enc_ctx_t* e = &s_e;
782 memset(e, 0, sizeof(*e));
783 e->dst = out_buf;
784 e->cap = out_buf_len;
785
786 ra8_err_t err = enc_run(e, rgb_buf, width, height, quality);
787 if (err != k_ra8_ok) {
788 return err;
789 }
790 *out_len = e->pos;
791 return k_ra8_ok;
792}
