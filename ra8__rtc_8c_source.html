85static uint8_t internal_bcd_to_bin(uint8_t bcd)
86{
87 const uint8_t high = (uint8_t)((bcd >> k_ra8_bcd_digit_shift) & k_ra8_bcd_digit_mask);
88 const uint8_t low = (uint8_t)(bcd & k_ra8_bcd_digit_mask);
89 return (uint8_t)((high * k_ra8_bcd_digit_base) + low);
90}

207static ra8_err_t internal_start_count_source(ra8_rtc_clk_src_t src)
208{
209 if (src == k_ra8_rtc_clk_loco) {
210 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
211 {
212 /* HUM Ch 9.2.15 "LOCOCR : Low-Speed On-Chip Oscillator Control
213 * Register" p 339 -- LCSTP = 0 keeps the internal LOCO running. */
214 *ra8_sys_lococr() = 0U;
215 }
216 ra8_delay_ms((uint32_t)k_ra8_rtc_clk_stab_loco_ms);
217 if ((*ra8_sys_lococr() & k_ra8_lococr_lcstp_mask) != 0U) { /* GCOVR_EXCL_BR_LINE */
218 return k_ra8_err_hw_init_failed; /* GCOVR_EXCL_LINE */
219 }
220 return k_ra8_ok;
221 }
222
223 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
224 {
225 /* HUM Ch 9.2.29 "SOMCR : Sub-Clock Oscillator Mode Control Register"
226 * p 351 -- set the crystal drive (standard, SOSEL = resonator) while
227 * SOSTP is still 1. */
228 *ra8_sys_somcr() = (uint8_t)k_ra8_somcr_drv_standard;
229 /* HUM Ch 9.2.14 "SOSCCR : Sub-Clock Oscillator Control Register"
230 * p 339 -- SOSTP = 0 starts the 32.768 kHz crystal. */
231 *ra8_sys_sosccr() = 0U;
232 }
233 ra8_delay_ms((uint32_t)k_ra8_rtc_clk_stab_sub_ms);
234 if ((*ra8_sys_sosccr() & k_ra8_sosccr_sostp_mask) != 0U) { /* GCOVR_EXCL_BR_LINE */
235 return k_ra8_err_hw_init_failed; /* GCOVR_EXCL_LINE */
236 }
237 return k_ra8_ok;

240ra8_err_t ra8_rtc_clock_init(ra8_rtc_clk_src_t src)
241{
242 if (src > k_ra8_rtc_clk_loco) {
243 return k_ra8_err_invalid_arg;
244 }
245
246 const ra8_err_t osc_err = internal_start_count_source(src);
247 if (osc_err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
248 ra8_log_error(s_tag, "rtc count source not running"); /* GCOVR_EXCL_LINE */
249 return osc_err; /* GCOVR_EXCL_LINE */
250 }
251
252 volatile r_rtc_regs_t* rtc = ra8_rtc();
253
254 /* HUM Ch 26.2.23 "RCR4 : RTC Control Register 4" p 1236 -- RCKSEL
255 * selects the count source (0 = sub-clock, 1 = LOCO). It must be set
256 * once before the initial RTC register settings. */
257 rtc->RCR4 = (uint8_t)src;
258 /* HUM Ch 26.3.2 "Clock and Count Mode Setting Procedure" p 1243 --
259 * supply at least 6 clocks of the count source after writing RCKSEL. */
260 ra8_delay_ms((uint32_t)k_ra8_rtc_clk_six_clocks_ms);
261
262 /* HUM Ch 26.2.21 "RCR2 : RTC Control Register 2" p 1232 -- stop the
263 * prescaler (START = 0) before the frequency register and software
264 * reset, and wait for the bit to fall. */
265 rtc->RCR2 = (uint8_t)(rtc->RCR2 & (uint8_t)~(1U << k_ra8_rcr2_bit_start));
266 internal_wait_bit(&rtc->RCR2, (uint8_t)(1U << k_ra8_rcr2_bit_start), 0U);
267
268 if (src == k_ra8_rtc_clk_loco) {
269 /* HUM Ch 26.2.25 "RFRH : Frequency Register H" p 1237 -- clear RFRH
270 * before RFRL on a cold start. */
271 rtc->RFRH = (uint16_t)k_ra8_rtc_rfrh_cold;
272 /* HUM Ch 26.2.24 "RFRL : Frequency Register L" p 1236 -- (LOCO / 128)
273 * - 1; 0x00FF for a 32.768 kHz LOCO. */
274 rtc->RFRL = (uint16_t)k_ra8_rtc_rfrl_32768;
275 }
276
277 /* HUM Ch 26.2.21 "RCR2 : RTC Control Register 2" p 1233 -- RESET = 1
278 * initializes the prescaler and count registers against the live count
279 * source; the bit auto-clears when the reset completes. */
280 rtc->RCR2 = (uint8_t)(1U << k_ra8_rcr2_bit_reset);
281 ra8_delay_ms((uint32_t)k_ra8_rtc_clk_reset_ms);
282 internal_wait_bit(&rtc->RCR2, (uint8_t)(1U << k_ra8_rcr2_bit_reset), 0U);
283
284 ra8_log_info_val(s_tag, "rtc clock init src", (uint32_t)src);
285 return k_ra8_ok;
286}

288ra8_err_t ra8_rtc_init(void)
289{
290 volatile r_rtc_regs_t* rtc = ra8_rtc();
291
292 /* HUM Ch 26.2.4 "RCR2 : RTC Control Register 2" p 1230 -- stop the
293 * counter (START=0) and clear CNTMD so we are in calendar mode.
294 * FSP r_rtc.c r_rtc_software_reset writes RCR2 = 0 and waits for
295 * CNTMD == 0 to confirm the mode change. */
296 rtc->RCR2 = 0U;
297 internal_wait_bit(&rtc->RCR2, (uint8_t)(1U << k_ra8_rcr2_bit_cntmd), 0U);
298
299 /* HUM Ch 26.2.3 "RCR1 : RTC Control Register 1" p 1229 -- mask
300 * every IRQ source (AIE/CIE/PIE) and clear PES. FSP waits for the
301 * write to land. */
302 rtc->RCR1 = 0U;
303 internal_wait_bit(&rtc->RCR1, k_ra8_rtc_byte_mask_all, 0U);
304
305 /* HUM Ch 26.2.4 "RCR2 : RTC Control Register 2" p 1230 -- HR24=1
306 * selects 24-hour mode. FSP also polls until HR24 reads back 1. */
307 rtc->RCR2 = (uint8_t)(1U << k_ra8_rcr2_bit_hr24);
308 internal_wait_bit(&rtc->RCR2,
309 (uint8_t)(1U << k_ra8_rcr2_bit_hr24),
310 (uint8_t)(1U << k_ra8_rcr2_bit_hr24));
311
312 /* HUM Ch 26.2.4 "RCR2 : RTC Control Register 2" p 1230 -- START=1
313 * starts the counter. */
314 rtc->RCR2 = (uint8_t)((1U << k_ra8_rcr2_bit_hr24) | (1U << k_ra8_rcr2_bit_start));
315 internal_wait_bit(&rtc->RCR2,
316 (uint8_t)(1U << k_ra8_rcr2_bit_start),
317 (uint8_t)(1U << k_ra8_rcr2_bit_start));
318
319 ra8_log_info(s_tag, "rtc_init (24h calendar)");
320 return k_ra8_ok;
321}

323ra8_err_t ra8_rtc_set(const ra8_rtc_datetime_t* dt)
324{
325 RA8_CHECK_NULL_PTR(dt, s_tag, "dt must not be nullptr");
326 if (dt->year < k_ra8_rtc_year_base) {
327 return k_ra8_err_invalid_arg;
328 }
329
330 volatile r_rtc_regs_t* rtc = ra8_rtc();
331
332 /* HUM Ch 26.2.4 "RCR2 : RTC Control Register 2" p 1230 -- count
333 * registers must only be written while START=0. Clear the bit and
334 * wait for the hardware to honour it (FSP's r_rtc_start_bit_update). */
335 const uint8_t saved = rtc->RCR2;
336 rtc->RCR2 = (uint8_t)(saved & (uint8_t)~(1U << k_ra8_rcr2_bit_start));
337 internal_wait_bit(&rtc->RCR2, (uint8_t)(1U << k_ra8_rcr2_bit_start), 0U);
338
339 /* HUM Ch 26.2.5 "RSECCNT : Second Counter" p 1232 */
340 rtc->RSECCNT = internal_bin_to_bcd(dt->second);
341 /* HUM Ch 26.2.6 "RMINCNT : Minute Counter" p 1232 */
342 rtc->RMINCNT = internal_bin_to_bcd(dt->minute);
343 /* HUM Ch 26.2.7 "RHRCNT : Hour Counter" p 1233 */
344 rtc->RHRCNT = internal_bin_to_bcd(dt->hour);
345 /* HUM Ch 26.2.8 "RWKCNT : Day-of-Week Counter" p 1234 */
346 rtc->RWKCNT = dt->weekday;
347 /* HUM Ch 26.2.9 "RDAYCNT : Day Counter" p 1234 */
348 rtc->RDAYCNT = internal_bin_to_bcd(dt->day);
349 /* HUM Ch 26.2.10 "RMONCNT : Month Counter" p 1235 */
350 rtc->RMONCNT = internal_bin_to_bcd(dt->month);
351 /* HUM Ch 26.2.11 "RYRCNT : Year Counter" p 1235 */
352 rtc->RYRCNT = (uint16_t)internal_bin_to_bcd((uint8_t)(dt->year - k_ra8_rtc_year_base));
353
354 /* HUM Ch 26.2.4 "RCR2 : RTC Control Register 2" p 1230 -- restore
355 * START to its prior value (always 1 if init has run) and wait. */
356 rtc->RCR2 = (uint8_t)(saved | (1U << k_ra8_rcr2_bit_start));
357 internal_wait_bit(&rtc->RCR2,
358 (uint8_t)(1U << k_ra8_rcr2_bit_start),
359 (uint8_t)(1U << k_ra8_rcr2_bit_start));
360
361 ra8_log_info_val(s_tag, "rtc_set year", (uint32_t)dt->year);
362 return k_ra8_ok;
363}

417ra8_err_t ra8_rtc_set_alarm(const ra8_rtc_datetime_t* alarm)
418{
419 RA8_CHECK_NULL_PTR(alarm, s_tag, "alarm must not be nullptr");
420 if (alarm->hour > k_ra8_rtc_alarm_max_hr || alarm->minute > k_ra8_rtc_alarm_max_min ||
421 alarm->second > k_ra8_rtc_alarm_max_sec) {
422 return k_ra8_err_invalid_arg;
423 }
424
425 volatile r_rtc_regs_t* rtc = ra8_rtc();
426 const uint8_t enb = (uint8_t)(1U << k_ra8_rtc_alarm_enb_bit);
427
428 /* HUM Ch 26.2.13 "RSECAR" / 26.2.14 "RMINAR" / 26.2.15 "RHRAR" --
429 * encode the BCD match value, OR in the ENB bit to enable the field. */
430 rtc->RSECAR = (uint8_t)(internal_bin_to_bcd(alarm->second) | enb);
431 rtc->RMINAR = (uint8_t)(internal_bin_to_bcd(alarm->minute) | enb);
432 rtc->RHRAR = (uint8_t)(internal_bin_to_bcd(alarm->hour) | enb);
433
434 /* Wildcard the date / weekday / month / year alarms so the alarm
435 * fires on the next hh:mm:ss match regardless of date. */
436 rtc->RWKAR = 0U;
437 rtc->RDAYAR = 0U;
438 rtc->RMONAR = 0U;
439 rtc->RYRAR = 0U;
440 rtc->RYRAREN = 0U;
441
442 return k_ra8_ok;
