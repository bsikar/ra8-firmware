291static ra8_err_t npu_smoke_run_job(void)
292{
293 ra8_npu_job_t job = {};
294 job.cmd_stream = s_npu_cmd_stream;
295 job.cmd_stream_bytes = (uint32_t)sizeof(s_npu_cmd_stream);
296 job.region_count = (uint8_t)k_npu_smoke_region_count;
297 job.region_base[k_npu_smoke_region_weights] = (uint64_t)(uintptr_t)s_npu_weights;
298 job.region_base[k_npu_smoke_region_input] = (uint64_t)(uintptr_t)s_npu_input;
299 job.region_base[k_npu_smoke_region_output] = (uint64_t)(uintptr_t)s_npu_output;
300
301 const ra8_err_t sub = ra8_npu_submit(&job);
302 if (sub != k_ra8_ok) {
303 return sub;
304 }
305 const ra8_err_t run = ra8_npu_run();
306 if (run != k_ra8_ok) {
307 return run;
308 }
309 return ra8_npu_wait();

322static bool npu_smoke_verify(uint32_t* out_check)
323{
324 uint32_t check = (uint32_t)k_npu_smoke_fnv_offset;
325 bool ok = true;
326 for (uint32_t i = 0U; i < (uint32_t)k_npu_smoke_arena_bytes; i++) {
327 const uint8_t expected =
328 (uint8_t)((s_npu_input[i] + (uint32_t)k_npu_smoke_addk) & (uint32_t)k_npu_smoke_byte_mask);
329 if (s_npu_output[i] != expected) {
330 ok = false;
331 }
332 check = (check ^ (uint32_t)s_npu_output[i]) * (uint32_t)k_npu_smoke_fnv_prime;
333 }
334 *out_check = check;
335 return ok;
336}

350static void npu_smoke_emit(uint32_t id, bool run_ok, uint32_t check, bool pass)
351{
352 char line[k_npu_smoke_line_cap];
353 const int n = snprintf(line,
354 sizeof(line),
355 "npu: id=0x%08X run=%s out=0x%08X verdict=%s\r\n",
356 (unsigned)id,
357 run_ok ? "OK" : "FAIL",
358 (unsigned)check,
359 pass ? "PASS" : "FAIL");
360 if (n > 0) {
361 (void)ra8_board_uart_console_write((const uint8_t*)line, (size_t)n);
362 (void)ra8_board_uart_console_flush();
363 }
364}

377int32_t main(void)
378{
379 npu_smoke_setup_or_halt();
380
381 bool run_ok = false;
382 uint32_t id = 0U;
383 uint32_t check = 0U;
384 bool pass = false;
385
386 if (ra8_npu_init() == k_ra8_ok) {
387 (void)ra8_npu_read_id(&id);
388 g_npu_smoke_id = id;
389
390 npu_smoke_build_stream();
391 npu_smoke_seed_arenas();
392 run_ok = (npu_smoke_run_job() == k_ra8_ok);
393
394 ra8_npu_status_t st = {};
395 if (ra8_npu_read_status(&st) == k_ra8_ok) {
396 g_npu_smoke_status = st.raw;
397 }
398 const bool matched = npu_smoke_verify(&check);
399 g_npu_smoke_check = check;
400 pass = run_ok && matched && (id != 0U);
401 }
402
403 g_npu_smoke_pass = pass ? 1U : 0U;
404 npu_smoke_emit(id, run_ok, check, pass);
405
406 npu_smoke_park();
407 return 0;
408}
