142[[nodiscard]] static ra8_err_t rtt_write(const uint8_t* data, uint32_t len)
143{
144 if (data == nullptr) {
145 return k_ra8_err_null_ptr;
146 }
147 rtt_buf_t* up = &s_rtt_cb.up[0];
148 for (uint32_t i = 0U; i < len; i++) {
149 uint32_t next = up->wr_off + 1U;
150 if (next == up->size) {
151 next = 0U;
152 }
153 if (next == up->rd_off) {
154 break; /* host has not drained -- drop the tail. */
155 }
156 up->buf[up->wr_off] = data[i];
157 up->wr_off = next;
158 }
159 return k_ra8_ok;
160}

169static uint8_t rtt_format_line(uint32_t counter, char* buf)
170{
171 static const char prefix[] = "rtt_log_demo: ";
172 uint8_t n = 0U;
173 for (size_t k = 0U; k < (sizeof prefix - 1U); k++) {
174 buf[n++] = prefix[k];
175 }
176 char dig[k_rtt_max_digits];
177 uint8_t d = 0U;
178 uint32_t tmp = counter;
179 do {
180 dig[d++] = (char)('0' + (uint8_t)(tmp % (uint32_t)k_rtt_decimal_base));
181 tmp /= (uint32_t)k_rtt_decimal_base;
182 } while (tmp != 0U && d < (uint8_t)k_rtt_max_digits);
183 while (d > 0U) {
184 buf[n++] = dig[--d];
185 }
186 buf[n++] = '\r';
187 buf[n++] = '\n';
188 return n;
189}

194static void rtt_demo_setup_or_halt(void)

198 rtt_demo_panic_halt();

201 rtt_demo_panic_halt();
202 }
203 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
204 rtt_demo_panic_halt();
205 }
206 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
207 rtt_demo_panic_halt();
208 }
209 rtt_init();
210}

231 }
