123[[nodiscard]] ra8_err_t sdmsc_print_dec(uint32_t value)
124{
125 uint8_t scratch[k_sdmsc_dec_chars_u32] = {};
126 uint8_t out[k_sdmsc_dec_chars_u32] = {};
127 uint8_t count = 0U;
128 uint32_t v = value;
129 if (v == 0U) {
130 out[0] = (uint8_t)'0';
131 return sdmsc_sci_write(out, 1U);
132 }
133 while (v != 0U) {
134 if (count >= (uint8_t)k_sdmsc_dec_chars_u32) {
135 break;
136 }
137 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_sdmsc_dec_radix));
138 v = v / k_sdmsc_dec_radix;
139 count++;
140 }
141 for (uint8_t i = 0U; i < count; i++) {
142 out[i] = scratch[count - 1U - i];
143 }
144 return sdmsc_sci_write(out, (uint32_t)count);
145}

148[[nodiscard]] ra8_err_t sdmsc_print_hex(uint32_t value, uint8_t digits)
149{
150 uint8_t out[k_sdmsc_hex_chars_u32] = {};
151 uint8_t width = digits;
152 if (width > (uint8_t)k_sdmsc_hex_chars_u32) {
153 width = (uint8_t)k_sdmsc_hex_chars_u32;
154 }
155 for (uint8_t i = 0U; i < width; i++) {
156 const uint8_t shift = (uint8_t)((width - 1U - i) * k_sdmsc_nibble_bits);
157 out[i] = sdmsc_nibble_to_hex((value >> shift) & k_sdmsc_nibble_mask);
158 }
159 return sdmsc_sci_write(out, (uint32_t)width);
160}
