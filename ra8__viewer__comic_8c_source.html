60RA8_INTERNAL static void viewer_fit_centered(int32_t src_w, int32_t src_h, viewer_fit_box_t* box)
61{
62 const int64_t fb_w = (int64_t)k_ra8_viewer_fb_width;
63 const int64_t fb_h = (int64_t)k_ra8_viewer_fb_height;
64 int64_t fit_w = fb_w;
65 int64_t fit_h = ((int64_t)src_h * fb_w) / (int64_t)src_w;
66 if (fit_h > fb_h) {
67 fit_h = fb_h;
68 fit_w = ((int64_t)src_w * fb_h) / (int64_t)src_h;
69 }
70 if (fit_w < 1) {
71 fit_w = 1;
72 }
73 if (fit_h < 1) {
74 fit_h = 1;
75 }
76 box->x = (int32_t)((fb_w - fit_w) / 2);
77 box->y = (int32_t)((fb_h - fit_h) / 2);
78 box->w = (int32_t)fit_w;
79 box->h = (int32_t)fit_h;
80}

99RA8_INTERNAL static void viewer_cap_render(uint32_t nw, uint32_t nh, uint32_t* rw, uint32_t* rh)
100{
101 const uint32_t k = (uint32_t)k_ra8_gfx_max_dim;
102 uint32_t w = nw;
103 uint32_t h = nh;
104 if (w > k) {
105 h = (uint32_t)(((uint64_t)h * k) / w);
106 w = k;
107 }
108 if (h > k) {
109 w = (uint32_t)(((uint64_t)w * k) / h);
110 h = k;
111 }
112 *rw = (w < 1U) ? 1U : w;
113 *rh = (h < 1U) ? 1U : h;
114}

134RA8_INTERNAL static ra8_err_t
135viewer_read_page_bytes(ra8_viewer_reader_t* r, uint32_t page, size_t* got)
136{
137 uint64_t raw_size = 0U;
138 ra8_err_t rc = ra8_comic_page_info(&r->comic, page, nullptr, 0U, nullptr, &raw_size, nullptr);
139 if (rc != k_ra8_ok) {
140 return rc;
141 }
142 /* Refuse a lying or bomb-ratio declaration before any allocation. The whole
143 * archive length is a sound upper bound on this entry's compressed bytes, so
144 * a declared uncompressed size over the output cap -- or implausible against
145 * that compressed bound -- is rejected here (the CBT/tar walker has no
146 * open-time size guard, so this is the viewer's own line of defence for it). */
147 rc = ra8_decomp_check_declared(&r->limits, r->file.size, raw_size);
148 if (rc != k_ra8_ok) {
149 return rc;
150 }
151 rc = viewer_reserve_page_buf(r, (size_t)raw_size);
152 if (rc != k_ra8_ok) {
153 return rc;
154 }
155 *got = 0U;
156 return ra8_comic_page_read(&r->comic, page, r->page_buf, r->page_cap, got);
157}

186{
187 size_t got = 0U;
188 ra8_err_t rc = viewer_read_page_bytes(r, page, &got);
189 if (rc != k_ra8_ok) {
190 return rc;
191 }
192
193 rc = ra8_gfx_init(r->fb, k_ra8_viewer_fb_width, k_ra8_viewer_fb_height, k_ra8_gfx_format_rgb565);
194 if (rc != k_ra8_ok) {
195 return rc;
196 }
197 (void)ra8_gfx_clear((uint32_t)k_viewer_bg);
198
199 /* stb's allocator is routed through ra8_img_arena; a JPEG header probe needs a
200 * bound arena to allocate its decode context (PNG does not), so bind around the
201 * probe. ra8_img_decode_blit binds its own arena below. */
202 int32_t src_w = 0;
203 int32_t src_h = 0;
204 ra8_img_arena_t probe_arena = {.base = r->arena_mem, .cap = (size_t)k_viewer_arena_bytes};
205 ra8_img_arena_bind(&probe_arena);
206 rc = ra8_img_probe_size(r->page_buf, got, &src_w, &src_h);
207 ra8_img_arena_unbind();
208 if (rc != k_ra8_ok) {
209 return rc;
210 }
211 viewer_fit_box_t box = {.x = 0, .y = 0, .w = k_ra8_viewer_fb_width, .h = k_ra8_viewer_fb_height};
212 viewer_fit_centered(src_w, src_h, &box);
213
214 ra8_img_arena_t arena = {.base = r->arena_mem, .cap = (size_t)k_viewer_arena_bytes};
215 return ra8_img_decode_blit(&arena,
216 r->page_buf,
217 got,
218 box.x,
219 box.y,
220 box.w,
221 box.h,
222 nullptr,
223 nullptr);

226ra8_err_t
227viewer_tile_comic(ra8_viewer_reader_t* r, uint32_t i, uint32_t* w, uint32_t* h, uint16_t** out)
228{
229 const uint32_t nw = r->tile_wpx[i];
230 const uint32_t nh = r->tile_hpx[i];
231 if ((nw == 0U) || (nh == 0U)) {
232 return k_ra8_err_not_supported; /* page that failed to probe at open */
233 }
234 uint32_t rw = 0U;
235 uint32_t rh = 0U;
236 viewer_cap_render(nw, nh, &rw, &rh);
237
238 size_t got = 0U;
239 ra8_err_t rc = viewer_read_page_bytes(r, i, &got);
240 if (rc != k_ra8_ok) {
241 return rc;
242 }
243 /* rw/rh are already clamped to k_ra8_gfx_max_dim, but size the destination
244 * through a widened multiply and the output cap so the allocation is
245 * provably bounded (and cannot wrap size_t) whatever the render cap becomes. */
246 const uint64_t buf_bytes = (uint64_t)rw * (uint64_t)rh * (uint64_t)sizeof(uint16_t);
247 if (buf_bytes > r->limits.max_output_bytes) {
248 return k_ra8_err_decomp_output_cap;
249 }
250 uint16_t* buf = (uint16_t*)malloc((size_t)buf_bytes);
251 if (buf == nullptr) {
252 return k_ra8_err_no_mem;
253 }
254 rc = ra8_gfx_init(buf, (uint16_t)rw, (uint16_t)rh, k_ra8_gfx_format_rgb565);
255 if (rc == k_ra8_ok) {
256 (void)ra8_gfx_clear((uint32_t)k_viewer_bg);
257 ra8_img_arena_t arena = {.base = r->arena_mem, .cap = (size_t)k_viewer_arena_bytes};
258 rc = ra8_img_decode_blit(&arena,
259 r->page_buf,
260 got,
261 0,
262 0,
263 (int32_t)rw,
264 (int32_t)rh,
265 nullptr,
266 nullptr);
267 }
268 if (rc != k_ra8_ok) {
269 free(buf);
270 return rc;
271 }
272 *w = rw;
273 *h = rh;
274 *out = buf;
275 return k_ra8_ok;
276}

279{
