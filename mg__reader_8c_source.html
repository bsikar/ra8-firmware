119static void mg_region(const mg_reader_t* r, int32_t* x0, int32_t* y0, int32_t* x1, int32_t* y1)
120{
121 const int32_t s = mg_scale(r);
122 *x0 = r->view_x;
123 *y0 = r->view_y;
124 int32_t xe = r->view_x + (r->fb_w * s);
125 int32_t ye = r->view_y + (mg_content_h(r) * s);
126 if (xe > (int32_t)r->page_w) {
127 xe = (int32_t)r->page_w;
128 }
129 if (ye > (int32_t)r->page_h) {
130 ye = (int32_t)r->page_h;
131 }
132 *x1 = xe;
133 *y1 = ye;
134}

137static void mg_offsets(const mg_reader_t* r, int32_t* ox, int32_t* oy)
138{
139 int32_t x0 = 0;
140 int32_t y0 = 0;
141 int32_t x1 = 0;
142 int32_t y1 = 0;
143 mg_region(r, &x0, &y0, &x1, &y1);
144 const int32_t s = mg_scale(r);
145 const int32_t disp_w = ((x1 - x0) + s - 1) / s;
146 const int32_t disp_h = ((y1 - y0) + s - 1) / s;
147 int32_t px = (r->fb_w - disp_w) / 2;
148 int32_t py = (mg_content_h(r) - disp_h) / 2;
149 *ox = (px > 0) ? px : 0;
150 *oy = (py > 0) ? py : 0;
151}

154static bool mg_map(const mg_reader_t* r,
155 int32_t s,
156 int32_t ox,
157 int32_t oy,
158 int32_t sx,
159 int32_t sy,
160 int32_t* dx,
161 int32_t* dy)
162{
163 const int32_t rx = sx - r->view_x;
164 const int32_t ry = sy - r->view_y;
165 if (((rx % s) != 0) || ((ry % s) != 0)) {
166 return false;
167 }
168 const int32_t px = ox + (rx / s);
169 const int32_t py = (int32_t)k_mg_statusbar_h + oy + (ry / s);
170 if ((px < 0) || (px >= r->fb_w)) {
171 return false;
172 }
173 if ((py < (int32_t)k_mg_statusbar_h) || (py >= r->fb_h)) {
174 return false;
175 }
176 *dx = px;
177 *dy = py;
178 return true;
179}

182static uint16_t mg_g565(uint8_t g)
183{
184 const uint16_t r5 = (uint16_t)(g >> k_mg_r_drop);
185 const uint16_t g6 = (uint16_t)(g >> k_mg_g_drop);
186 const uint16_t b5 = (uint16_t)(g >> k_mg_r_drop);
187 return (uint16_t)(((uint16_t)(r5 << k_mg_r_sh)) | ((uint16_t)(g6 << k_mg_g_sh)) | b5);
188}

191static void mg_clamp(mg_reader_t* r)
192{
193 if (r->zoom == (uint8_t)k_mg_zoom_fit) {
194 r->view_x = 0;
195 r->view_y = 0;
196 return;
197 }
198 int32_t maxx = (int32_t)r->page_w - r->fb_w;
199 int32_t maxy = (int32_t)r->page_h - mg_content_h(r);
200 if (maxx < 0) {
201 maxx = 0;
202 }
203 if (maxy < 0) {
204 maxy = 0;
205 }
206 if (r->view_x < 0) {
207 r->view_x = 0;
208 } else if (r->view_x > maxx) {
209 r->view_x = maxx;
210 }
211 if (r->view_y < 0) {
212 r->view_y = 0;
213 } else if (r->view_y > maxy) {
214 r->view_y = maxy;
215 }

223static void mg_blit_tile(mg_reader_t* r,
224 const ra8_tile_t* t,
225 uint16_t tx,
226 uint16_t ty,
227 int32_t s,
228 int32_t ox,
229 int32_t oy,
230 const int32_t reg[4])
231{
232 const int32_t px0 = (int32_t)tx * (int32_t)r->tile_w;
233 const int32_t py0 = (int32_t)ty * (int32_t)r->tile_h;
234 for (int32_t rr = 0; rr < (int32_t)t->height; ++rr) {
235 const int32_t sy = py0 + rr;
236 if ((sy < reg[1]) || (sy >= reg[3])) {
237 continue;
238 }
239 for (int32_t cc = 0; cc < (int32_t)t->width; ++cc) {
240 const int32_t sx = px0 + cc;
241 if ((sx < reg[0]) || (sx >= reg[2])) {
242 continue;
243 }
244 int32_t dx = 0;
245 int32_t dy = 0;
246 if (!mg_map(r, s, ox, oy, sx, sy, &dx, &dy)) {
247 continue;
248 }
249 const uint8_t g = t->pixels[(rr * (int32_t)t->width) + cc];
250 r->fb[(dy * r->fb_w) + dx] = mg_g565(g);
251 }
252 }
253}

256static void mg_tiles_of_region(const mg_reader_t* r, const int32_t reg[4], uint16_t out[4])
257{

261 out[3] = (uint16_t)((reg[3] - 1) / (int32_t)r->tile_h); /* ty1 */

265static ra8_err_t mg_render_page(mg_reader_t* r)
266{
267 int32_t reg[4] = {};
268 mg_region(r, &reg[0], &reg[1], &reg[2], &reg[3]);
269 int32_t s = mg_scale(r);
270 int32_t ox = 0;
271 int32_t oy = 0;
272 mg_offsets(r, &ox, &oy);
273 uint16_t tiles[4] = {};
274 mg_tiles_of_region(r, reg, tiles);
275 for (uint16_t ty = tiles[1]; ty <= tiles[3]; ++ty) {
276 for (uint16_t tx = tiles[0]; tx <= tiles[2]; ++tx) {
277 ra8_tile_key_t key = {.image_id = r->image_id, .tile_x = tx, .tile_y = ty, .zoom = 0U};
278 ra8_tile_t tile = {};
279 const ra8_err_t e = ra8_tile_cache_get(r->cache, &key, &tile);
280 if (e != k_ra8_ok) {
281 return e;
282 }
283 mg_blit_tile(r, &tile, tx, ty, s, ox, oy, reg);
284 const ra8_err_t p = ra8_tile_cache_put(r->cache, tile.pixels);
285 if (p != k_ra8_ok) {
286 return p;
287 }
288 }
289 }
290 return k_ra8_ok;
291}

352ra8_err_t mg_reader_prefetch(mg_reader_t* r)
353{
354 RA8_CHECK_NULL_PTR(r, k_mg_tag, "reader");
355 RA8_CHECK_NULL_PTR(r->cache, k_mg_tag, "cache");
356 if (r->last_pan == (ra8_tile_pan_dir_t)k_ra8_tile_pan_none) {
357 /* Last action was not a moving pan (initial frame, zoom toggle, or a pan
358 * clamped at the page edge): nothing ahead of the viewport to warm. A pan is
359 * a no-op at fit-zoom, so this also covers the whole-page-shown case. */
360 return k_ra8_ok;
361 }
362 int32_t reg[4] = {};
363 mg_region(r, &reg[0], &reg[1], &reg[2], &reg[3]);
364 uint16_t tiles[4] = {};
365 mg_tiles_of_region(r, reg, tiles);
366 /* Budget the prefetch to the cache's spare cells (capacity minus the tiles the
367 * current frame occupies) so warming can never evict an on-screen tile. */
368 const uint32_t frame =
369 ((uint32_t)(tiles[2] - tiles[0]) + 1U) * ((uint32_t)(tiles[3] - tiles[1]) + 1U);
370 const uint32_t cap = ra8_tile_cache_capacity(r->cache);
371 const uint32_t spare = (cap > frame) ? (cap - frame) : 0U;
372 const uint16_t budget = (spare > (uint32_t)UINT16_MAX) ? (uint16_t)UINT16_MAX : (uint16_t)spare;
373 const ra8_tile_prefetch_req_t req = {
374 .image_id = r->image_id,
375 .view = {.tx0 = tiles[0], .ty0 = tiles[1], .tx1 = tiles[2], .ty1 = tiles[3]},
376 .tile_cols = r->tile_cols,
377 .tile_rows = r->tile_rows,
378 .zoom = 0U,
379 .max_tiles = budget,
380 .dir = r->last_pan,
381 };
382 return ra8_tile_cache_prefetch_pan(r->cache, &req, nullptr);
383}

405static bool mg_pan(mg_reader_t* r, int32_t dx, int32_t dy)
406{
407 if (r->zoom == (uint8_t)k_mg_zoom_fit) {
408 return false; /* the whole page is shown -- nothing to pan */
409 }
410 const int32_t ox = r->view_x;
411 const int32_t oy = r->view_y;
412 r->view_x += dx;
413 r->view_y += dy;
414 mg_clamp(r);
415 const bool moved = (r->view_x != ox) || (r->view_y != oy);
416 r->last_pan = moved ? mg_pan_dir(dx, dy) : (ra8_tile_pan_dir_t)k_ra8_tile_pan_none;
417 return moved;
418}

420mg_zone_t mg_zone_hit(const mg_reader_t* r, int32_t x, int32_t y)
421{
422 if (r == nullptr) {
423 return k_mg_zone_none;
424 }
425 if (x < 0) {
426 return k_mg_zone_none;
427 }
428 if (x >= r->fb_w) {
429 return k_mg_zone_none;
430 }
431 if (y < (int32_t)k_mg_statusbar_h) {
432 return k_mg_zone_none; /* status bar */
433 }
434 if (y >= r->fb_h) {
435 return k_mg_zone_none;
436 }
437 const int32_t band = (int32_t)k_mg_edge_band;
438 if (y < ((int32_t)k_mg_statusbar_h + band)) {
439 return k_mg_zone_pan_up;
440 }
441 if (y >= (r->fb_h - band)) {
442 return k_mg_zone_pan_down;
443 }
444 if (x < band) {
445 return k_mg_zone_pan_left;
446 }
447 if (x >= (r->fb_w - band)) {
448 return k_mg_zone_pan_right;
449 }
450 return k_mg_zone_zoom;
451}

464bool mg_reader_tap(mg_reader_t* r, int32_t x, int32_t y)
465{
466 if (r == nullptr) {
467 return false;
468 }
469 switch (mg_zone_hit(r, x, y)) {

491static uint32_t mg_append_uint(char* buf, uint32_t cap, uint32_t pos, int32_t val)
492{
493 char tmp[k_mg_dec_max] = {};
494 uint32_t n = 0U;
495 uint32_t v = (val < 0) ? 0U : (uint32_t)val;
496 if (v == 0U) {
497 tmp[n] = '0';
498 n++;
499 }
500 while ((v > 0U) && (n < (uint32_t)k_mg_dec_max)) {
501 tmp[n] = (char)('0' + (v % (uint32_t)k_mg_dec_ten));
502 n++;
503 v /= (uint32_t)k_mg_dec_ten;
504 }
505 for (uint32_t i = 0U; (i < n) && (pos < (cap - 1U)); ++i) {
506 buf[pos] = tmp[n - 1U - i];
507 pos++;
508 }
509 buf[pos] = '\0';
510 return pos;
511}

540uint32_t mg_fnv1a(const void* buf, uint32_t len)
541{
542 const uint8_t* p = (const uint8_t*)buf;
543 if (p == nullptr) {
544 return (uint32_t)k_mg_fnv_offset;
545 }
546 uint32_t hsh = (uint32_t)k_mg_fnv_offset;
547 for (uint32_t i = 0U; i < len; ++i) {
548 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_mg_fnv_prime;
549 }
550 return hsh;
551}

554static uint8_t mg_fit_factor(const mg_reader_t* r)
555{
556 const int32_t cw = r->fb_w;
557 const int32_t ch = mg_content_h(r);
558 const int32_t fx = ((int32_t)r->page_w + cw - 1) / cw;
559 const int32_t fy = ((int32_t)r->page_h + ch - 1) / ch;
560 int32_t f = (fx > fy) ? fx : fy;
561 if (f < 1) {
562 f = 1;
563 }
564 return (uint8_t)f;

601static ra8_err_t mg_reader_check_geometry(const mg_reader_cfg_t* cfg)
602{
603 if ((cfg->fb_w <= 0) || (cfg->fb_h <= (int32_t)k_mg_statusbar_h)) {
604 return k_ra8_err_invalid_size;
605 }
606 if ((cfg->info->width == 0U) || (cfg->info->height == 0U) || (cfg->info->tile_w == 0U) ||
607 (cfg->info->tile_h == 0U)) {
608 return k_ra8_err_invalid_size;
609 }
610 /* The blit path is gray8-only; a colour atlas (bpp 3/4) would be mis-read one
611 * byte per pixel and rendered as garbage. Fail closed instead (#339). */
612 if (cfg->info->bpp != (uint8_t)k_mg_gray8_bpp) {
613 ra8_log_error(k_mg_tag, "JOF atlas bpp unsupported: reader is gray8-only");
614 return k_ra8_err_not_supported;
615 }
616 return k_ra8_ok;

619ra8_err_t mg_reader_init(mg_reader_t* r, const mg_reader_cfg_t* cfg)
