192static void sfr_panic_halt(uint32_t stage)
193{
194 g_sfr_stage = stage;
195 (void)ra8_board_led_on(k_ra8_board_led_red);
196 const uint8_t lvl =
197 (uint8_t)((uint32_t)k_sfr_diag_lvl_base +
198 ((stage & (uint32_t)k_sfr_diag_lvl_mask) * (uint32_t)k_sfr_diag_lvl_step));
199 const uint32_t argb = (uint32_t)k_sfr_diag_argb_opaque |
200 ((uint32_t)lvl << (uint32_t)k_sfr_diag_shift_r) |
201 ((uint32_t)lvl << (uint32_t)k_sfr_diag_shift_g) | (uint32_t)lvl;
202 (void)ra8_gfx_clear(argb);
203 while (1) {
204 __asm__ volatile("wfi");
205 }
206}

209static void sfr_bringup_clocks(void)
210{
211 uint32_t cpuclk0_hz = 0U;
212 if (ra8_cgc_init() != k_ra8_ok) {
213 sfr_panic_halt(k_sfr_stage_boot);
214 }
215 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
216 sfr_panic_halt(k_sfr_stage_boot);
217 }
218 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &s_pclka_hz) != k_ra8_ok) {
219 sfr_panic_halt(k_sfr_stage_boot);
220 }
221 if (ra8_mstp_init() != k_ra8_ok) {
222 sfr_panic_halt(k_sfr_stage_boot);
223 }
224 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
225 sfr_panic_halt(k_sfr_stage_boot);
226 }
227 if (ra8_board_led_init(k_ra8_board_led_blue) != k_ra8_ok) {
228 sfr_panic_halt(k_sfr_stage_boot);
229 }
230 if (ra8_board_led_init(k_ra8_board_led_red) != k_ra8_ok) {
231 sfr_panic_halt(k_sfr_stage_boot);
232 }
233 ra8_isr_globals_enable();
234}

247 sfr_panic_halt(k_sfr_stage_boot);

253 (void)ra8_gfx_clear((uint32_t)k_sfr_paper_argb);
254 g_sfr_stage = k_sfr_stage_panel_ok;

264static void sfr_load_font_or_halt(void)
265{
266 const ra8_sdfont_cfg_t cfg = {
267 .spi_channel = (uint8_t)k_sfr_spi_chan,
268 .sck = k_sfr_pin_sck,
269 .cipo = k_sfr_pin_cipo,

271 .cs = k_sfr_pin_cs,

273 .filename = "FONT.OTF",
274 .provision_blob = g_ra8_font_literata_latin1,

276 };
277 uint32_t got = 0U;

279 const ra8_err_t err = ra8_sdfont_load(&cfg, s_font_buf, (uint32_t)k_sfr_font_cap, &got, &src);
280 g_sfr_err = (uint32_t)err;

285 g_sfr_source = (uint32_t)src;
286 g_sfr_stage = k_sfr_stage_font_ok;

290static void sfr_render_or_halt(void)
291{
292 if (ra8_reflow_init((uint16_t)k_sfr_fb_w,
293 (uint16_t)k_sfr_fb_h,
294 s_font_buf,
295 g_sfr_font_len,
296 (uint16_t)k_sfr_font_px,
297 (uint32_t)k_sfr_ink_argb,
298 (uint32_t)k_sfr_link_argb,
299 &s_engine) != k_ra8_ok) {
300 sfr_panic_halt(k_sfr_stage_reflow_fail);
301 }
302 uint32_t pages = 0U;
303 if (ra8_reflow_layout_chapter(&s_engine,
304 (const uint8_t*)k_sfr_body,
305 (uint32_t)(sizeof(k_sfr_body) - 1U),
306 &pages) != k_ra8_ok) {
307 sfr_panic_halt(k_sfr_stage_reflow_fail);
308 }
309 g_sfr_pages = pages;
310 g_sfr_stage = k_sfr_stage_layout_ok;
311 if (ra8_reflow_render_page(&s_engine, 0U, nullptr) != k_ra8_ok) {
312 sfr_panic_halt(k_sfr_stage_reflow_fail);
313 }
314
315 /* Count inked pixels so a HIL probe / emulator can prove text actually landed. */
316 uint32_t ink = 0U;
317 const uint16_t paper565 = (uint16_t)k_sfr_paper_rgb565;
318 for (size_t i = 0U; i < (size_t)k_sfr_fb_w * (size_t)k_sfr_fb_h; i++) {
319 if (s_framebuffer[i] != paper565) {
320 ink++;
321 }
322 }
323 g_sfr_ink = ink;
324 g_sfr_stage = k_sfr_stage_render_ok;
325 (void)ra8_board_led_on(k_ra8_board_led_blue);
326}

334int32_t main(void)
335{
336 sfr_bringup_clocks();
337 sfr_bringup_panel();
338 sfr_load_font_or_halt();
339 sfr_render_or_halt();
340
341 /* The page is rendered; GLCDC scans out the framebuffer continuously.
342 * ra8_emulator snapshots it via --ppm. Idle with a heartbeat that advances only
343 * on this success path -- the panic-halt loop does not bump it -- so a J-Link
344 * memprobe gate proves the full SD-font render pipeline ran end to end. */
345 while (1) {
346 ra8_delay_ms((uint32_t)k_sfr_frame_ms);
347 g_sfr_heartbeat++;
348 }
349}
