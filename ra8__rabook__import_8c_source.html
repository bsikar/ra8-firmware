89static uint32_t s_crc32_block(uint32_t crc, const uint8_t* data, uint32_t len)
90{
91 if (data == nullptr) {
92 return crc;
93 }
94 if (len == 0U) {
95 return crc;
96 }
97 for (uint32_t i = 0U; i < len; i++) {
98 crc ^= data[i];
99 for (uint8_t b = 0U; b < (uint8_t)k_crc32_bits; b++) {
100 bool lsb = (crc & 1U) != 0U;
101 crc >>= 1U;
102 if (lsb) {
103 crc ^= (uint32_t)k_crc32_poly;
104 }
105 }
106 }
107 return crc;
108}

130RA8_INTERNAL
131static ra8_err_t
132s_crc_stream(ra8_fs_file_t* file, uint8_t* buf, uint32_t cap, uint32_t* out_size, uint32_t* out_crc)
133{
134 RA8_CHECK_NULL_PTR(file, s_tag, "file");
135 RA8_CHECK_NULL_PTR(buf, s_tag, "buf");
136
137 uint32_t crc = (uint32_t)k_crc32_seed;
138 uint32_t total = 0U;
139 for (uint32_t i = 0U; i < (uint32_t)k_import_max_chunks; i++) {
140 uint32_t got = 0U;
141 ra8_err_t err = ra8_fs_read(file, buf, cap, &got);
142 if (err != k_ra8_ok) {
143 return err;
144 }
145 if (got == 0U) {
146 break;
147 }
148 crc = s_crc32_block(crc, buf, got);
149 total += got;
150 }
151 *out_size = total;
152 *out_crc = crc ^ (uint32_t)k_crc32_seed;
153 return k_ra8_ok;
154}

591

601 return err;

606 char mark_name[k_ra8_rabook_import_name_cap];

616 .source_size = src_size,
