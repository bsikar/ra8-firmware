74static uint32_t s_crc32_block(uint32_t crc, const uint8_t* data, uint32_t len)
75{
76 if (data == nullptr) {
77 return crc;
78 }
79 if (len == 0U) {
80 return crc;
81 }
82 for (uint32_t i = 0U; i < len; i++) {
83 crc ^= data[i];
84 for (uint8_t b = 0U; b < (uint8_t)k_crc32_bits; b++) {
85 bool lsb = (crc & 1U) != 0U;
86 crc >>= 1U;
87 if (lsb) {
88 crc ^= (uint32_t)k_crc32_poly;
89 }
90 }
91 }
92 return crc;
93}

116static ra8_err_t
117s_crc_stream(ra8_fs_file_t* file, uint8_t* buf, uint32_t cap, uint32_t* out_size, uint32_t* out_crc)
118{
119 RA8_CHECK_NULL_PTR(file, s_tag, "file");
120 RA8_CHECK_NULL_PTR(buf, s_tag, "buf");
121
122 uint32_t crc = (uint32_t)k_crc32_seed;
123 uint32_t total = 0U;
124 for (uint32_t i = 0U; i < (uint32_t)k_import_max_chunks; i++) {
125 uint32_t got = 0U;
126 ra8_err_t err = ra8_fs_read(file, buf, cap, &got);
127 if (err != k_ra8_ok) {
128 return err;
129 }
130 if (got == 0U) {
131 break;
132 }
133 crc = s_crc32_block(crc, buf, got);
134 total += got;
135 }
136 *out_size = total;
137 *out_crc = crc ^ (uint32_t)k_crc32_seed;
138 return k_ra8_ok;
139}

214RA8_INTERNAL
215static ra8_err_t s_derive_stem(const char* epub_path, char* out, uint32_t cap)
216{
217 RA8_CHECK_NULL_PTR(epub_path, s_tag, "epub_path");
218 RA8_CHECK_NULL_PTR(out, s_tag, "out");
219
220 size_t start = 0U;
221 size_t i = 0U;
222 for (; epub_path[i] != '\0'; i++) {
223 if (epub_path[i] == '/') {
224 start = i + 1U;
225 }
226 }
227 size_t len = i - start;
228 for (size_t j = len; j > 1U; j--) {
229 if (epub_path[start + j - 1U] == '.') {
230 len = j - 1U;
231 break;
232 }
233 }
234 if (len == 0U) {
235 return k_ra8_err_invalid_arg;
236 }
237 if (len >= (size_t)cap) {
238 return k_ra8_err_invalid_size;
239 }
240 (void)memcpy(out, &epub_path[start], len);
241 out[len] = '\0';
242 return k_ra8_ok;
243}

621ra8_err_t ra8_rabook_import_open(const ra8_rabook_import_cfg_t* cfg,
622 const char* epub_path,
623 char* out_cache_path,
624 uint32_t cache_path_cap,
625 ra8_rabook_import_outcome_t* out_outcome)
626{
627 ra8_err_t err = s_validate_open_args(cfg, epub_path, out_cache_path, cache_path_cap, out_outcome);
628 if (err != k_ra8_ok) {
629 return err;
630 }
631
632 /* Derive the cache names from the source's own basename FIRST, so a bad name
633 * or a too-small output buffer is refused before the source is streamed or
634 * the compiler is invoked (a compile must never run for a doomed request). */
635 char stem[k_ra8_rabook_import_name_cap];
636 err = s_derive_stem(epub_path, stem, (uint32_t)k_ra8_rabook_import_name_cap);
637 if (err != k_ra8_ok) {
638 return err;
639 }
640 char cache_name[k_ra8_rabook_import_name_cap];
641 char tmp_name[k_ra8_rabook_import_name_cap];
642 char mark_name[k_ra8_rabook_import_name_cap];
643 err = s_derive_names(stem, cache_name, tmp_name, mark_name);
644 if (err != k_ra8_ok) {
645 return err;
646 }
647 if ((strlen(cache_name) + 1U) > (size_t)cache_path_cap) {
648 return k_ra8_err_invalid_size;
649 }
650
651 uint32_t src_size = 0U;
652 uint32_t src_crc = 0U;
653 err = s_compute_source_key(cfg->mount,
654 epub_path,
655 cfg->scratch,
656 cfg->scratch_cap,
657 &src_size,
658 &src_crc);
659 if (err != k_ra8_ok) {
660 return err;
661 }
662
663 const ra8_rabook_import_stamp_t want = {
664 .magic = (uint32_t)k_ra8_rabook_import_stamp_magic,
665 .format_version = cfg->format_version,
666 .importer_version = cfg->importer_version,
667 .source_size = src_size,
668 .source_crc32 = src_crc,
669 };
670
671 if (s_cache_is_fresh(cfg->mount, mark_name, &want)) {
672 if (s_file_exists(cfg->mount, cache_name)) {
673 *out_outcome = k_ra8_rabook_import_hit;
674 return s_copy_name(out_cache_path, cache_path_cap, cache_name);
675 }
676 }
677
678 err = s_compile_and_cache(cfg, epub_path, cache_name, tmp_name, mark_name, &want);
679 if (err != k_ra8_ok) {
680 return err;
681 }
682 *out_outcome = k_ra8_rabook_import_compiled;
683 return s_copy_name(out_cache_path, cache_path_cap, cache_name);
684}
