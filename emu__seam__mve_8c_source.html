29/* ============================================================================
30 * Minimal MVE (Helium) emulation. The RA8D2 is Cortex-M85 (Armv8.1-M, has MVE),
31 * but the closest core Unicorn offers is M33 (Armv8-M, NO MVE), so the Helium
32 * instructions GCC's auto-vectoriser emits trap here as "invalid". GCC vectorises
33 * the memset / struct-zero idiom with two forms, which this handles:
34 * VMOV.I32 Qd, #imm -- set all four 32-bit lanes of Qd to imm
35 * VSTRW.32 Qd, [Rn{, #off}] -- store the 16-byte Qd to memory (no write-back)
36 * The Q registers alias the FPU D registers (Qn == D[2n]:D[2n+1]) -- the M33 core
37 * has those (the firmware uses the FPU) -- so the vector state is read/written
38 * through Unicorn's D registers. Any other MVE form falls through and still
39 * reports as invalid, so nothing is silently mis-executed. capstone (already
40 * linked, and already decoding these for the error path) provides the operands;
41 * the engine handle is opened once and reused. On real silicon all of this just
42 * runs natively on Helium -- this only makes the M33-based emulator faithful to it.
43 * ==========================================================================*/

96/* ---------------------------------------------------------------------------
97 * MVE contiguous load/store family (VLDRB/VLDRH/VLDRW, VSTRB/VSTRH/VSTRW with
98 * an immediate offset), serviced from the NoCP UsageFault.
99 *
100 * Armv8.1-M reallocates coprocessor space 0b1110 / 0b1111 to MVE, so this
101 * family reuses the legacy STC/LDC encodings byte for byte. `arm-none-eabi-as
102 * -march=armv8.1-m.main+mve` assembles `stc p15, c7, [r0, #196]` and
103 * `vstrw.32 q3, [r0, #196]` to the SAME word ED80 7F31, and objdump
104 * `-m armv8.1-m.main` renders that word as the MVE form. capstone renders it as
105 * the legacy `stc p15`, so this family must never be decoded through capstone --
106 * the fields below are taken straight from the encoding.
107 *
108 * Unicorn's M33 implements neither MVE nor coprocessor 14/15, so it does NOT
109 * trap these as invalid instructions: it raises a NoCP UsageFault (QEMU
110 * EXCP_NOCP, reported as int_no 17) through UC_HOOK_INTR with PC still at the
111 * faulting instruction. Verified against libunicorn standalone: on a cold
112 * engine the invalid-instruction hook is never called for any form in this
113 * family, so the NoCP fault is the only way in.
114 *
115 * That covers the FIRST instruction of a run. Servicing it means writing PC and
116 * calling `uc_emu_stop` from inside the interrupt hook, and once that has
117 * happened the NEXT instruction of the same run arrives at the
118 * invalid-instruction hook instead -- also verified standalone, with two
119 * consecutive `vstrw.32`. GCC emits exactly such runs (the struct-zero idiom),
120 * so BOTH arrival paths are wired to the same decode via ::mve_mem_try: the
121 * NoCP hook through ::emu_mve_nocp_emulate, and the invalid-instruction
122 * dispatcher through ::emulate_mve. Wiring only one of them leaves every run
123 * of two or more MVE accesses faulting on its second instruction.
124 *
125 * Field layout, confirmed by assembling each form and reading the bytes back:
126 * hw1 = 1110 1101 0 U W L Rn (0xED.., hw1[6] the legacy 'D' bit is 0)
127 * hw2 = Qd[2:0] 1 1 1 1 size[1:0] imm7
128 * with size 0b00 byte (imm7 scaled by 1), 0b01 halfword (by 2), 0b10 word
129 * (by 4) and 0b11 unallocated. Every form transfers the whole 16-byte vector;
130 * `size` sets the lane width and the immediate scale only, so for a contiguous
131 * little-endian access the data movement is a plain 16-byte copy.
132 *
133 * Aliasing: the single- and double-precision FP stores share hw1 exactly and
134 * differ only in hw2[11:9] -- `vstr s14, [r0, #196]` is ED80 7A31 and
135 * `vstr d7, [r0, #196]` is ED80 7B31, both hw2[12:9] == 0b1101 / 0b1010.
136 * Unicorn executes both natively and correctly, so the hw2[12:9] == 0b1111
137 * guard is load-bearing: widening it would hijack working FP stores and turn
138 * them into silent 16-byte writes.
139 * ===========================================================================
140 */

207static bool mve_mem_decode(uint16_t hw1, uint16_t hw2, mve_mem_op_t* op)
208{
209 if (((hw1 & (uint16_t)k_mve_mem_h1_mask) != (uint16_t)k_mve_mem_h1_val) ||
210 ((hw2 & (uint16_t)k_mve_mem_h2_mask) != (uint16_t)k_mve_mem_h2_val)) {
211 return false;
212 }
213 const uint32_t size =
214 ((uint32_t)hw2 >> (uint32_t)k_mve_mem_size_sh) & (uint32_t)k_mve_mem_size_msk;
215 uint32_t scale;
216 switch (size) {
217 case (uint32_t)k_mve_mem_sz_byte:
218 scale = (uint32_t)k_mve_mem_scl_byte;
219 break;
220 case (uint32_t)k_mve_mem_sz_half:
221 scale = (uint32_t)k_mve_mem_scl_half;
222 break;
223 case (uint32_t)k_mve_mem_sz_word:
224 scale = (uint32_t)k_mve_mem_scl_word;
225 break;
226 default:
227 return false; /* size 0b11 is unallocated -- never emulate it. */
228 }
229 op->qd = ((uint32_t)hw2 >> (uint32_t)k_mve_qd_shift) & (uint32_t)k_mve_qd_mask;
230 op->rn = (uint32_t)hw1 & (uint32_t)k_mve_rn_mask;
231 op->off = ((uint32_t)hw2 & (uint32_t)k_mve_imm7_mask) * scale;
232 op->load = (hw1 & (uint16_t)k_mve_mem_bit_l) != 0U;
233 op->wback = (hw1 & (uint16_t)k_mve_mem_bit_w) != 0U;
234 op->add = (hw1 & (uint16_t)k_mve_mem_bit_u) != 0U;
235 return true;
236}

261static void mve_mem_exec(uc_engine* uc, const mve_mem_op_t* op)
262{
263 uint32_t base = 0U;
264 (void)uc_reg_read(uc, k_arm_reg_id[op->rn], &base);
265 const uint32_t addr = op->add ? (base + op->off) : (base - op->off);
266 const int d_lo = (int)UC_ARM_REG_D0 + (int)(2U * op->qd);
267 uint8_t buf[k_mve_q_bytes];
268 uint64_t lo = 0U;
269 uint64_t hi = 0U;
270 if (op->load) {
271 (void)uc_mem_read(uc, (uint64_t)addr, buf, (size_t)k_mve_q_bytes);
272 (void)memcpy(&lo, buf, sizeof(lo));
273 (void)memcpy(&hi, buf + sizeof(lo), sizeof(hi));
274 (void)uc_reg_write(uc, d_lo, &lo);
275 (void)uc_reg_write(uc, d_lo + 1, &hi);
276 } else {
277 (void)uc_reg_read(uc, d_lo, &lo);
278 (void)uc_reg_read(uc, d_lo + 1, &hi);
279 (void)memcpy(buf, &lo, sizeof(lo));
280 (void)memcpy(buf + sizeof(lo), &hi, sizeof(hi));
281 (void)uc_mem_write(uc, (uint64_t)addr, buf, (size_t)k_mve_q_bytes);
282 }
283 if (op->wback) {
284 (void)uc_reg_write(uc, k_arm_reg_id[op->rn], &addr);
285 }
286}

312static bool mve_mem_try(uc_engine* uc, const uint8_t code[4])
313{
314 const uint16_t hw1 = (uint16_t)(code[0] | ((uint16_t)code[1] << (uint16_t)k_byte_bits));
315 const uint16_t hw2 = (uint16_t)(code[2] | ((uint16_t)code[3] << (uint16_t)k_byte_bits));
316 mve_mem_op_t op = {};
317 if (!mve_mem_decode(hw1, hw2, &op)) {
318 return false;
319 }
320 mve_mem_exec(uc, &op);
321 return true;
322}

332static bool mve_exec_one(uc_engine* uc, const cs_insn* insn)
333{
334 const cs_arm* d = &insn->detail->arm;
335 const bool op0_q = (d->op_count == 2) && (d->operands[0].type == ARM_OP_REG) &&
336 (d->operands[0].reg >= ARM_REG_Q0) && (d->operands[0].reg <= ARM_REG_Q7);
337 if (!op0_q) {
338 return false;
339 }
340 /* VMOV.I32 Qd, #imm -> replicate the 32-bit immediate into all four lanes. */
341 if ((insn->id == ARM_INS_VMOV) && (d->operands[1].type == ARM_OP_IMM) &&
342 (strstr(insn->mnemonic, ".i32") != nullptr)) {
343 const uint32_t imm = (uint32_t)d->operands[1].imm;
344 const uint64_t pair = ((uint64_t)imm << (uint64_t)k_mve_lane_shift) | (uint64_t)imm;
345 (void)uc_reg_write(uc, mve_q_d(d->operands[0].reg, false), &pair);
346 (void)uc_reg_write(uc, mve_q_d(d->operands[0].reg, true), &pair);
347 return true;
348 }
349 return false;
350}

380bool emulate_mve(uc_engine* uc, uint32_t pc0, const uint8_t code0[4])
381{
382 csh* cs = mve_capstone();
383 if (cs == nullptr) {
384 return false;
385 }
386 uint32_t pc = pc0;
387 uint8_t code[4];
388 (void)memcpy(code, code0, sizeof(code));
389 uint32_t handled = 0U;
390 while (handled < (uint32_t)k_mve_max_run) {
391 /* Contiguous load/store first: capstone renders this family as a legacy
392 * `stc p15`, so it must be decoded from the raw encoding, not through the
393 * disassembler below. */
394 if (mve_mem_try(uc, code)) {
395 handled++;
396 pc += (uint32_t)k_mve_insn_len;
397 if (uc_mem_read(uc, (uint64_t)pc, code, sizeof(code)) != UC_ERR_OK) {
398 break;
399 }
400 continue;
401 }
402 cs_insn* insn = nullptr;
403 const size_t n = cs_disasm(*cs, code, (size_t)k_mve_insn_len, pc, 1, &insn);
404 if (n != 1U) {
405 break;
406 }
407 const bool ok = mve_exec_one(uc, &insn[0]);
408 cs_free(insn, n);
409 if (!ok) {
410 break; /* first non-MVE (valid) instruction -- relaunch resumes here. */
411 }
412 handled++;
413 pc += (uint32_t)k_mve_insn_len;
414 if (uc_mem_read(uc, (uint64_t)pc, code, sizeof(code)) != UC_ERR_OK) {
415 break;
416 }
417 }
418 if (handled > 0U) {
419 (void)uc_reg_write(uc, UC_ARM_REG_PC, &pc);
420 s_mve_emulated += handled;
421 }
422 return handled > 0U;
423}

425bool emu_mve_nocp_emulate(uc_engine* uc, uint32_t pc)
426{
427 uint8_t code[k_mve_insn_len] = {};
428 if (uc_mem_read(uc, (uint64_t)pc, code, sizeof(code)) != UC_ERR_OK) {
429 return false;
430 }
431 if (!mve_mem_try(uc, code)) {
432 return false;
433 }
434 const uint32_t next = pc + (uint32_t)k_mve_insn_len;
435 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
436 s_mve_emulated++;
437 s_mve_nocp_handled = true;
438 s_mve_resume_pc = next;
439 s_mve_resume_armed = true;
440 return true;
441}
