105static uint32_t ec_framebuffer_hash(void)
106{
107 const uint8_t* p = (const uint8_t*)s_framebuffer;
108 const size_t n = sizeof(s_framebuffer);
109 uint32_t hsh = (uint32_t)k_ec_fnv_offset;
110 for (size_t i = 0U; i < n; i++) {
111 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_ec_fnv_prime;
112 }
113 return hsh;
114}

117static void ec_print_hex(uint32_t value)
118{
119 uint8_t buf[k_ec_hex_nibbles];
120 for (uint32_t i = 0U; i < (uint32_t)k_ec_hex_nibbles; i++) {
121 const uint32_t shift = ((uint32_t)k_ec_hex_nibbles - 1U - i) * (uint32_t)k_ec_nibble_bits;
122 const uint32_t nib = (value >> shift) & (uint32_t)k_ec_nibble_mask;
123 buf[i] = (uint8_t)((nib < (uint32_t)k_ec_dec_ten) ? ('0' + nib) : ('A' + (nib - k_ec_dec_ten)));
124 }
125 ec_print(buf, (uint32_t)k_ec_hex_nibbles);
126}

129static void ec_print_uint(uint32_t value)
130{
131 uint8_t buf[k_ec_dec_ten];
132 uint32_t n = 0U;
133 if (value == 0U) {
134 buf[n] = '0';
135 n++;
136 }
137 while ((value > 0U) && (n < (uint32_t)k_ec_dec_ten)) {
138 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_ec_dec_ten));
139 n++;
140 value /= (uint32_t)k_ec_dec_ten;
141 }
142 for (uint32_t i = 0U; i < n; i++) {
143 ec_print(&buf[n - 1U - i], 1U);
144 }
145}

181static size_t ec_render_cover_or_halt(int32_t* out_w, int32_t* out_h)
182{
183 const ra8_epub_mem_media_t media = {.data = k_epub_cover_fixture,
184 .size = (size_t)k_epub_cover_fixture_len};
185 if (ra8_epub_open(&media, "book.epub", &s_book) != k_ra8_ok) {
186 ec_panic_halt(k_msg_open, (uint32_t)sizeof(k_msg_open) - 1U);
187 }
188 size_t cover_len = 0U;
189 if ((ra8_epub_get_cover_image(&s_book, s_cover, (size_t)k_ec_cover_cap, &cover_len) !=
190 k_ra8_ok) ||
191 (cover_len == 0U)) {
192 ec_panic_halt(k_msg_cov, (uint32_t)sizeof(k_msg_cov) - 1U);
193 }
194
195 ra8_img_arena_t arena = {.base = s_img_arena,
196 .cap = (size_t)k_ec_arena_bytes,
197 .offset = 0U,
198 .live = 0U};
199 if (ra8_img_decode_blit(&arena,
200 s_cover,
201 cover_len,
202 0,
203 0,
204 (int32_t)k_ec_fb_w,
205 (int32_t)k_ec_fb_h,
206 out_w,
207 out_h) != k_ra8_ok) {
208 ec_panic_halt(k_msg_derr, (uint32_t)sizeof(k_msg_derr) - 1U);
209 }
210 return cover_len;
211}

252 __asm__ volatile("wfi");
