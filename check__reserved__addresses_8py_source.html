4"""Reject first-party address constants that point into a Reserved memory window.
5
6WHY THIS EXISTS
7===============
8Three times now a driver has named an address that describes hardware which is
9not there, compiled clean, and failed only on silicon:
10
11 * the `ra8_rsip` crypto family -- invented registers, whole driver rewritten;
12 * `ra8_ptp` -- addressed a reserved window, and its `gptp: clock PASS` was a
13 reserved aperture echoing back a write (#498);
14 * `ra8_wdt_regs.h` -- read OFS0/OFS3 at 0x03001E04 / 0x03001E20, which appear
15 nowhere in either Hardware User's Manual and land in a Reserved area, and
16 dereferenced them at runtime (#545).
17
18Nothing catches this class. `check_hum_register_map.py` (#540) cross-checks
19register SYMBOLS and struct/window OFFSETS against the manual's tables, but an
20absolute-address enumerator like `k_ra8_wdt_ofs0_addr` is neither, so it falls
21outside that gate's rules. `check_linker_scripts.py` rule LD007 guards the
22phantom data-flash base on the LINKER side only -- the C side was unguarded,
23which is exactly how #545 survived.
24
25WHAT IT CHECKS
26==============
27 RA001 an enumerator in a `uintptr_t`-typed enum resolves into a window the
28 HUM memory map marks Reserved.
29
30The rule is deliberately narrow. `uintptr_t` enums are, by the CLAUDE.md
31constants hierarchy, THE way this tree spells a hardware address ("Use
32`uintptr_t` for any enum whose values are hardware memory-mapped addresses"),
33so the scan surface is precisely the set of things that claim to be addresses.
34A value that is not an address does not belong in one of these enums, and an
35address that lands in a Reserved window is wrong by construction -- reading it
36is undefined: it may fault, or return garbage the caller then trusts.
37
38WHAT IT DELIBERATELY DOES NOT CHECK
39===================================
40It does not try to validate that a non-reserved address is CORRECT -- that
41needs the manual's register tables, which is #540's job. This gate answers the
42much cheaper question that #540 cannot: "is this address inside a hole?"
43"""
44

56# Reserved windows of the RA8 address map. Both supported parts agree
57# line-for-line: RA8D2 HUM R01UH1065EJ0130 and RA8P1 HUM R01UH1064EJ0130,
58# Ch 3 "Address Space" memory map.
59#
60# Keep this table SMALL and primary-sourced. A window listed here must be
61# reserved on every supported part, because a hit is a hard failure.
62RESERVED_WINDOWS: list[tuple[int, int, str]] = [
63 # Between the Extra MRAM option-setting region and the SiP-Flash area.
64 # This is the window #545's invented OFS addresses landed in.
65 (0x03000000, 0x07FFFFFF, "Reserved area between Extra MRAM and SiP-Flash (HUM Ch 3 map)"),
66 # The same window through the non-secure alias (BASE_MC 0x1200_0000). The
67 # option-setting words have a full non-secure mirror -- OFS3 is genuinely
68 # read at 0x12C9_F4C4 -- so the hole above it is just as reachable by a
69 # wrong constant, and omitting it would leave the alias half unguarded.
70 (0x13000000, 0x17FFFFFF, "Reserved area, non-secure alias of 0x0300_0000 (HUM Ch 3 map)"),
71 # The conventional RA-family data-flash base. The RA8 has no such array;
72 # it faults on this silicon (#397). LD007 guards the linker side.
73 (0x27000000, 0x27FFFFFF, "phantom data-flash -- the RA8 has no data-flash array (#397)"),
74]
75
76# Directories holding hand-written first-party code. `libs/third_party` is SOUP
77# and exempt per CLAUDE.md; it is filtered below rather than listed here.
78SCAN_ROOTS = ("libs", "src", "examples", "port", "tools", "tests")
79
80# Below this many `uintptr_t` enumerators the scan has plainly stopped
81# matching -- a syntax change in how the tree spells address enums would
82# otherwise leave this gate quietly enforcing nothing. Measured 2026-07-28:
83# 151 enum blocks across 136 files, well over a thousand enumerators. The
84# floor sits far under a healthy tree but far above a collapsed scan.
85ENUMERATOR_FLOOR = 300
86
87ENUM_OPEN = re.compile(r"\benum\s*:\s*uintptr_t\s*\{", re.MULTILINE)
88ENUMERATOR = re.compile(r"(\bk_[A-Za-z0-9_]+)\s*=\s*(0[xX][0-9A-Fa-f_]+)")

91@dataclass(frozen=True)
92class Finding:
93 """One RA001 hit: an address enumerator inside a Reserved window."""

95 path: pathlib.Path
96 line: int
97 symbol: str
98 value: int
99 why: str

103 return (

109def strip_comments(text: str) -> str:
110 """Blank out block and line comments, preserving line structure.
111
112 Newlines are kept so reported line numbers stay true to the source. A
113 commented-out address is documentation, not a dereference, so it must not
114 be flagged -- both remediated headers cite the old bad values in prose.
115 """
116 text = re.sub(r"/\*.*?\*/", lambda m: re.sub(r"[^\n]", " ", m.group(0)), text, flags=re.DOTALL)
117 return re.sub(r"//[^\n]*", "", text)
118
119
120def enum_bodies(code: str) -> list[tuple[int, str]]:
121 """Yield `(offset, body)` for every `enum : uintptr_t { ... }` block.
122
123 Brace-matched rather than regex-terminated so a nested initialiser cannot
124 truncate the body and silently drop the enumerators after it.
125 """
126 out: list[tuple[int, str]] = []
127 for m in ENUM_OPEN.finditer(code):
128 start = m.end()
129 depth = 1
130 i = start
131 while i < len(code) and depth > 0:
132 if code[i] == "{":
133 depth += 1
134 elif code[i] == "}":
135 depth -= 1
136 i += 1
137 out.append((start, code[start : i - 1]))
138 return out
139
140
141def reserved_hit(value: int) -> str | None:
142 """Return the description of the Reserved window containing `value`, if any."""
143 for lo, hi, why in RESERVED_WINDOWS:
144 if lo <= value <= hi:
145 return why
146 return None

149def check_text(path: pathlib.Path, text: str) -> tuple[list[Finding], int]:
150 """Apply RA001 to one file; also return how many enumerators were seen.
151
152 The enumerator count feeds the vacuity floor, and is produced by the same
153 pass that applies the rule so the two cannot drift.
154 """
155 code = strip_comments(text)
156 findings: list[Finding] = []
157 seen = 0
158 for offset, body in enum_bodies(code):
159 for m in ENUMERATOR.finditer(body):
160 seen += 1
161 value = int(m.group(2).replace("_", ""), 16)
162 why = reserved_hit(value)
163 if why is not None:
164 line = code[: offset + m.start()].count("\n") + 1
165 findings.append(Finding(path, line, m.group(1), value, why))
166 return findings, seen

181def tracked_sources(root: pathlib.Path) -> list[pathlib.Path]:
182 """List tracked first-party C sources and headers, excluding SOUP."""
183 out = subprocess.run( # noqa: S603 # fixed argv, no shell
184 ["git", "ls-files", "--", *(f"{d}/**/*.[ch]" for d in SCAN_ROOTS)], # noqa: S607
185 capture_output=True,
186 text=True,

188 cwd=root,
189 ).stdout.split()
190 return [root / p for p in out if not p.startswith("libs/third_party/")]

193def scan(paths: list[pathlib.Path]) -> tuple[list[Finding], int]:
194 """Run RA001 over every path, returning findings and the enumerator count."""
195 findings: list[Finding] = []
196 total = 0
197 for p in paths:
198 try:
199 text = p.read_text(encoding="utf-8", errors="replace")
200 except OSError:
201 continue
202 f, seen = check_text(p, text)
203 findings.extend(f)
204 total += seen
205 return findings, total

208def floor_breached(seen: int) -> bool:
209 """Report and fail when the scan matched too few enumerators to mean anything."""
210 if seen >= ENUMERATOR_FLOOR:
211 return False
212 print(
213 f"ERROR: matched only {seen} uintptr_t enumerator(s), below the floor of "
214 f"{ENUMERATOR_FLOOR}. Either the tree stopped spelling addresses as "
215 f"`enum : uintptr_t` (RA001 now enforces nothing) or the scan scope "
216 f"collapsed. Refusing to report success.",
217 file=sys.stderr,
218 )
219 return True

222# Values the selftest fixtures below encode, named so the assertions do not
223# repeat bare literals.
224_FIXTURE_BAD_OFS0 = 0x03001E04
225_FIXTURE_BAD_DATAFLASH = 0x27000000
226_FIXTURE_QUIET_ENUMERATORS = 3
227
228_MUST_FIRE = """
229typedef enum : uintptr_t {
230 k_ra8_bad_addr = 0x03001E04UL,
231} bad_t;
232"""
233
234_MUST_FIRE_DATAFLASH = """
235typedef enum : uintptr_t {
236 k_ra8_df_addr = 0x27000000UL,
237} df_t;
238"""
239
240_MUST_STAY_QUIET = """
241/* A commented-out 0x03001E04 is prose, not a dereference. */
242typedef enum : uintptr_t {
243 k_ra8_ofs0_addr = 0x02C9F040UL,
244 k_ra8_ofs3_addr = 0x12C9F4C4UL,
245 k_ra8_wdt_base_addr = 0x40202600UL,
246} good_t;
247
248/* Not a uintptr_t enum: a mask that happens to look like a reserved address. */
249typedef enum : uint32_t {
250 k_ra8_mask_thing = 0x03001E04UL,
251} mask_t;
252"""

255def selftest() -> int:
256 """Assert RA001 fires on a reserved address and stays quiet on a real one.
257
258 Both directions, driving `check_text` -- the same entry point `scan` uses.
259 A checker that only proves it can fire is indistinguishable from one whose
260 scope has collapsed to nothing.
261 """
262 rc = 0
263 fake = pathlib.Path("selftest.h")
264
265 hits, seen = check_text(fake, _MUST_FIRE)
266 if len(hits) != 1 or hits[0].value != _FIXTURE_BAD_OFS0:
267 print("SELFTEST FAIL: RA001 did not fire on the #545 reserved address", file=sys.stderr)
268 rc = 1
269 if seen != 1:
270 print(f"SELFTEST FAIL: expected 1 enumerator, counted {seen}", file=sys.stderr)
271 rc = 1
272
273 hits, _ = check_text(fake, _MUST_FIRE_DATAFLASH)
274 if len(hits) != 1 or hits[0].value != _FIXTURE_BAD_DATAFLASH:
275 print("SELFTEST FAIL: RA001 did not fire on phantom data-flash", file=sys.stderr)
276 rc = 1
277
278 hits, seen = check_text(fake, _MUST_STAY_QUIET)
279 if hits:
280 print(
281 f"SELFTEST FAIL: RA001 over-fired on legitimate addresses: "
282 f"{[h.render() for h in hits]}",
283 file=sys.stderr,
284 )
285 rc = 1
286 if seen != _FIXTURE_QUIET_ENUMERATORS:
287 print(
288 f"SELFTEST FAIL: expected {_FIXTURE_QUIET_ENUMERATORS} uintptr_t "
289 f"enumerators in the quiet fixture, counted {seen} -- the uint32_t "
290 f"enum must not be scanned",
291 file=sys.stderr,
292 )
293 rc = 1
294
295 # The floor must itself be able to fail, or it is decoration. Its diagnostic
296 # is swallowed here so a PASSING selftest never prints a line starting
297 # "ERROR:" -- a gate whose success output reads like a failure costs the
298 # next reader real time.
299 sink = io.StringIO()
300 with contextlib.redirect_stderr(sink):
301 floor_bites = floor_breached(0)
302 if not floor_bites:
303 print("SELFTEST FAIL: vacuity floor accepted an empty scan", file=sys.stderr)
304 rc = 1
305
306 if rc == 0:
307 print("check_reserved_addresses selftest: OK (fires, stays quiet, floor bites)")
308 return rc

311def main() -> int:
312 """Scan first-party C for address enumerators inside Reserved windows."""
313 ap = argparse.ArgumentParser(description=__doc__)
314 ap.add_argument("--selftest", action="store_true", help="assert both directions")
315 ap.add_argument("--list-files", action="store_true", help="print the scanned file list")
316 ap.add_argument("paths", nargs="*", help="files to check (default: all tracked)")
317 args = ap.parse_args()
318
319 if args.selftest:
320 return selftest()
321
322 root = repo_root()
323 paths = [pathlib.Path(p) for p in args.paths] if args.paths else tracked_sources(root)
324
325 if args.list_files:
326 for p in paths:
327 print(p)
328 return 0
329
330 findings, seen = scan(paths)
331 for f in findings:
332 print(f.render(), file=sys.stderr)
333
334 # Only floor a whole-tree run; a positional subset legitimately sees few.
335 if not args.paths and floor_breached(seen):
336 return 1
337
338 if findings:
339 print(
340 f"\n{len(findings)} address constant(s) point into a Reserved window. "
341 f"An address the manual does not define is not a citation problem -- "
342 f"reading it is undefined behaviour on silicon.",
343 file=sys.stderr,
344 )
345 return 1
346 return 0
