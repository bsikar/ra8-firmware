427static VOID demo_cdc_activate(VOID* cdc_instance)
428{
429 s_cdc_acm = (UX_SLAVE_CLASS_CDC_ACM*)cdc_instance;
430 /* USBX writes ux_slave_device_state = CONFIGURED in
431 * _ux_device_stack_configuration_set just before invoking this
432 * activate callback. Pin it here so any concurrent IRQ/poll-driven
433 * write in the dispatch worker observes the chapter-9 result and
434 * does not demote it back to ATTACHED, which would break the
435 * subsequent cdc_acm_read state gate. */
436 if (_ux_system_slave != UX_NULL) {
437 _ux_system_slave->ux_system_slave_device.ux_slave_device_state =
438 (unsigned long)UX_DEVICE_CONFIGURED;
439 }
440 /* Wake the demo loop -- it blocks on s_cdc_active_sem instead of
441 * polling s_cdc_acm with tx_thread_sleep, which never returned on
442 * this hardware. */
443 (void)tx_semaphore_put(&s_cdc_active_sem);
444 /* CDC bulk endpoints: EP2 OUT -> pipe 2, EP1 IN -> pipe 1. Enable
445 * the bridge's ISR-side auto-echo, which mirrors OUT data back on
446 * the IN pipe directly inside the ISR -- ~10x faster than the
447 * worker-thread _read/_write loop. demo_worker's _read/_write call
448 * is suppressed under the matching s_dcd_auto_echo_enable check so
449 * the two paths cannot race on the bulk-OUT pipe. */
450 ux_dcd_ra8_usb_auto_echo_enable(2U, 1U);
451}

570static void demo_echo_iter(UCHAR* buf, ULONG cap)
571{
572 demo_pin_configured_state();
573 ULONG n = 0UL;
574 s_demo_diag.loop_pre_read++;
575 UINT read_status = _ux_device_class_cdc_acm_read(s_cdc_acm, buf, cap, &n);
576 s_demo_diag.loop_post_read++;
577 if (read_status != UX_SUCCESS) {
578 tx_thread_sleep(k_demo_idle_ticks);
579 return;
580 }
581 s_demo_diag.loop_read_ok++;
582 if (n == 0UL) {
583 s_demo_diag.loop_read_zero++;
584 return;
585 }
586 s_demo_diag.loop_pre_write++;
587 if (_ux_device_class_cdc_acm_write(s_cdc_acm, buf, n, &n) != UX_SUCCESS) {
588 return;
589 }
590 s_demo_diag.loop_post_write++;
591 for (ULONG i = 0UL; i < n; i++) {
592 (void)ra8_board_led_toggle(k_ra8_board_led1);
593 }
594}

607static VOID demo_worker(ULONG arg)
608{
609 (void)arg;
610
611 if (demo_usbx_stack_up() != UX_SUCCESS) {
612 return;
613 }
614 if (demo_cdc_class_register() != UX_SUCCESS) {
615 return;
616 }
617 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs) != k_ra8_ok) {
618 return;
619 }
620 if (ra8_usb_device_attach(k_ra8_usb_speed_fs, true) != k_ra8_ok) {
621 return;
622 }
623
624 UCHAR buf[k_demo_echo_buf_bytes];
625 (void)buf; /* Auto-echo owns the data path; buf reserved for future fallback. */
626 (void)demo_echo_iter; /* Kept for the optional non-auto-echo fallback path. */
627 while (1) {
628 s_demo_diag.loop_iter++;
629 if (s_cdc_acm == UX_NULL) {
630 s_demo_diag.loop_cdc_null++;
631 /* Block until demo_cdc_activate posts the semaphore. Avoids
632 * tx_thread_sleep, which we observed never returning on this
633 * silicon under polled-dispatch worker load. */
634 (void)tx_semaphore_get(&s_cdc_active_sem, TX_WAIT_FOREVER);
635 continue;
636 }
637 /* Auto-echo handles bulk OUT -> IN mirroring inside the ISR.
638 * Invoking demo_echo_iter here would race with auto-echo on the
639 * IN pipe and deliver out-of-order data for MPS-aligned packets. */
640 tx_thread_sleep(k_demo_idle_ticks);
641 }
642}

709[[nodiscard]] static ra8_err_t demo_pins_init(void)
710{
711 ra8_err_t err = ra8_pfs_route_peripheral(k_demo_pin_vbus, k_ra8_psel_usb_fs, "usb_cdc.vbus");
712 if (err != k_ra8_ok) {
713 return err;
714 }
715 /* P500 is the EK-RA8D2 USB-FS role-select GPIO per the v1 User's
716 * Manual section "USB Full Speed" (J11 connector): drive LOW for
717 * device mode, HIGH for host mode. Routing P500 to the USBFS
718 * peripheral function lets the USB module drive it as VBUSEN
719 * (host-mode VBUS supply enable, default HIGH) which keeps the chip
720 * in HOST mode and prevents enumeration. For this device-mode demo
721 * we instead claim P500 as a GPIO output and drive it LOW so the
722 * board's role-select circuitry presents a USB device to the host. */
723 err = ra8_gpio_output_init(k_demo_pin_vbusen, k_ra8_level_low);
724 if (err != k_ra8_ok) {
725 return err;
726 }
727 err = ra8_pfs_route_peripheral(k_demo_pin_dp, k_ra8_psel_usb_fs, "usb_cdc.dp");
728 if (err != k_ra8_ok) {
729 return err;
730 }
731 return ra8_pfs_route_peripheral(k_demo_pin_dm, k_ra8_psel_usb_fs, "usb_cdc.dm");
732}

749int32_t main(void)
750{
751 uint32_t cpuclk0_hz = 0U;
752
753 if (ra8_cgc_init() != k_ra8_ok) {
754 demo_panic_halt();
755 }
756
757 /* Bring up PLL2 -> USBCKCR / USBCKDIVCR so USBFS sees a spec-compliant
758 * 48 MHz reference (PLL2P 240 MHz / 5). Must run BEFORE any caller
759 * releases MSTPB11 (USBFS) -- the SREQ -> SRDY handshake silently
760 * hangs otherwise (HUM Ch 9 "Clock selection switching procedure"
761 * step 1). Without this the SIE never sees a 48 MHz clock and the
762 * host never enumerates the device. The init-order audit
763 * (scripts/checks/audit_init_order.py) requires CGC bring-up to
764 * land BEFORE peripheral inits like ra8_log_init, so the RTT
765 * heart-beat moves down to right after the time/board bring-up. */
766 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
767 demo_panic_halt();
768 }
769 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
770 demo_panic_halt();
771 }
772 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
773 demo_panic_halt();
774 }
775 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
776 demo_panic_halt();
777 }
778 if (demo_pins_init() != k_ra8_ok) {
779 demo_panic_halt();
780 }
781 ra8_log_init();
782 ra8_log_info(s_demo_tag, "threadx_usbx_cdc_demo boot, CGC PLL2 enable OK");
783
784 ra8_isr_globals_enable();
785
786#ifndef RA8_OFF_TARGET
787 /* tx_kernel_enter is __noreturn -- it never comes back. */
788 tx_kernel_enter();
789#endif
790
791 demo_panic_halt();
792 return 0;
793}
