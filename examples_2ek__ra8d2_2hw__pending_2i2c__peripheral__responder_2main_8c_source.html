122static void riic_target_on_event(void* vctx, ra8_i2c_peripheral_event_t event)
123{
124 riic_target_ctx_t* ctx = (riic_target_ctx_t*)vctx;
125 if (ctx == nullptr) {
126 return;
127 }
128 if (event == k_ra8_i2c_peripheral_event_write) {
129 uint32_t got = 0U;
130 (void)ra8_i2c_peripheral_receive(ctx->channel, ctx->rx, (uint32_t)k_riic_target_buf_len, &got);
131 if (got > 0U) {
132 ctx->rx_len = got;
133 }
134 (void)ra8_board_uart_console_write(k_riic_target_msg_rx, sizeof(k_riic_target_msg_rx) - 1U);
135 (void)ra8_board_led_toggle(k_ra8_board_led1);
136 return;
137 }
138 if (event == k_ra8_i2c_peripheral_event_read) {
139 uint32_t sent = 0U;
140 (void)ra8_i2c_peripheral_transmit(ctx->channel, ctx->rx, ctx->rx_len, &sent);
141 (void)ra8_board_uart_console_write(k_riic_target_msg_tx, sizeof(k_riic_target_msg_tx) - 1U);
142 (void)ra8_board_led_toggle(k_ra8_board_led1);
143 }
144}

211[[nodiscard]] static ra8_err_t riic_target_arm(riic_target_ctx_t* ctx)
212{
213 if (ctx == nullptr) {
214 return k_ra8_err_null_ptr;
215 }
216 const ra8_err_t attached =
217 ra8_i2c_peripheral_attach_handler((uint8_t)k_riic_target_channel, riic_target_on_event, ctx);
218 if (attached != k_ra8_ok) {
219 return attached;
220 }
221 const ra8_i2c_peripheral_cfg_t cfg = {
222 .own_addr_7b = (uint8_t)k_riic_target_own_addr,
223 .slot = k_ra8_i2c_peripheral_slot_0,
224 .general_call = false,
225 .clock_stretch = true,
226 .irq_enable = false, /* polled dispatch, no NVIC wiring */
227 };
228 return ra8_i2c_peripheral_init((uint8_t)k_riic_target_channel, &cfg);
229}

245{
246 riic_target_setup_or_halt();

253 };

255 riic_target_panic_halt();

258 k_ra8_ok) {
259 riic_target_panic_halt();
260 }
261
262 while (1) {
263 ra8_i2c_peripheral_dispatch((uint8_t)k_riic_target_channel);
264 }
265
266 riic_target_panic_halt();
267 return 0;
