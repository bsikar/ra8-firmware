72static ra8_err_t mem_read(void* ctx, uint32_t lba, uint32_t count, uint8_t* buf)

107static void put16(uint8_t* p, uint32_t off, uint16_t v)
108{
109 p[off] = (uint8_t)(v & (uint16_t)k_byte_lo_mask);
110 p[off + 1U] = (uint8_t)((v >> (uint16_t)k_byte_shift) & (uint16_t)k_byte_lo_mask);
111}

114static void build_fat16(uint8_t* b)
115{
116 put16(b, (uint32_t)k_bpb_off_bytspersec, (uint16_t)k_block_size);
117 b[k_bpb_off_secperclus] = (uint8_t)k_bpb_secperclus;

182static int dump_image(const char* image_out)
183{
184 const size_t image_bytes = (size_t)k_blocks_fat16 * (size_t)k_block_size;
185 FILE* fout = fopen(image_out, "wb");
186 if (fout == nullptr) {
187 (void)fprintf(stderr, "mkfontimg: cannot write %s\n", image_out);
188 return 1;
189 }
190 const size_t wrote = fwrite(s_disk.bytes, 1U, image_bytes, fout);
191 (void)fclose(fout);
192 if (wrote != image_bytes) {
193 (void)fprintf(stderr, "mkfontimg: short write to %s\n", image_out);
194 return 1;
195 }
196 return 0;
197}

208static int
209build_and_dump(const char* image_out, const uint8_t* font, size_t font_len, const char* dest_name)
210{
211 s_disk.block_count = (uint32_t)k_blocks_fat16;
212 s_disk.bytes = (uint8_t*)calloc(1U, (size_t)k_blocks_fat16 * (size_t)k_block_size);
213 if (s_disk.bytes == nullptr) {
214 (void)fprintf(stderr, "mkfontimg: out of memory (disk)\n");
215 return 1;
216 }
217 build_fat16(s_disk.bytes);
218
219 const ra8_fs_backend_t backend = {.read_block = mem_read,
220 .write_block = mem_write,
221 .get_capacity = mem_cap,
222 .ctx = &s_disk};
223
224 /* Write the font (if any) through the real ra8_fs so the on-card layout
225 * matches exactly what the firmware app will read. */
226 ra8_fs_mount_t* mnt = nullptr;
227 if (ra8_fs_mount(&backend, &mnt) != k_ra8_ok) {
228 (void)fprintf(stderr, "mkfontimg: ra8_fs_mount failed\n");
229 free(s_disk.bytes);
230 return 1;
231 }
232 if (write_font_file(mnt, font, font_len, dest_name) != 0) {
233 free(s_disk.bytes);
234 return 1;
235 }
236 (void)ra8_fs_unmount(mnt);
237
238 const int rc = dump_image(image_out);
239 free(s_disk.bytes);
240 return rc;
241}

263static uint8_t* slurp_font(const char* font_in, size_t* font_len)
264{
265 uint8_t* font = (uint8_t*)malloc((size_t)k_font_cap);
266 if (font == nullptr) {
267 (void)fprintf(stderr, "mkfontimg: out of memory\n");
268 return nullptr;
269 }
270 FILE* fin = fopen(font_in, "rb");
271 if (fin == nullptr) {
272 (void)fprintf(stderr, "mkfontimg: cannot open %s\n", font_in);
273 free(font);
274 return nullptr;
275 }
276 const size_t len = fread(font, 1U, (size_t)k_font_cap, fin);
277 (void)fclose(fin);
278 if (len < (size_t)k_font_min_bytes) {
279 (void)fprintf(stderr, "mkfontimg: %s too small (%zu bytes)\n", font_in, len);
280 free(font);
281 return nullptr;
282 }
283 *font_len = len;
284 return font;
285}

334static int run_font(int argc, char** argv)
335{
336 if (argc < 3) {
337 (void)fprintf(stderr, "usage: %s <font-in> <image-out> [dest-name]\n", argv[0]);
338 return 2;
339 }
340 const char* font_in = argv[1];
341 const char* image_out = argv[2];
342 const char* dest_name = (argc > 3) ? argv[3] : "FONT.OTF";
343
344 size_t font_len = 0U;
345 uint8_t* font = slurp_font(font_in, &font_len);
346 if (font == nullptr) {
347 return 1;
348 }
349 const int rc = build_and_dump(image_out, font, font_len, dest_name);
350 free(font);
351 if (rc != 0) {
352 return 1;
353 }
354 (void)fprintf(stderr, "mkfontimg: wrote %s (%s = %zu bytes)\n", image_out, dest_name, font_len);
355 return 0;
356}
