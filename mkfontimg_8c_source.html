196static void put16(uint8_t* p, uint32_t off, uint16_t v)
197{
198 p[off] = (uint8_t)(v & (uint16_t)k_byte_lo_mask);
199 p[off + 1U] = (uint8_t)((v >> (uint16_t)k_byte_shift) & (uint16_t)k_byte_lo_mask);

301static int dump_image(const char* image_out)
302{
303 const size_t image_bytes = (size_t)k_blocks_fat16 * (size_t)k_block_size;
304 FILE* fout = fopen(image_out, "wb");
305 if (fout == nullptr) {
306 (void)fprintf(stderr, "mkfontimg: cannot write %s\n", image_out);
307 return 1;
308 }
309 const size_t wrote = fwrite(s_disk.bytes, 1U, image_bytes, fout);
310 (void)fclose(fout);
311 if (wrote != image_bytes) {
312 (void)fprintf(stderr, "mkfontimg: short write to %s\n", image_out);
313 return 1;
314 }
315 return 0;

345static int
346build_and_dump(const char* image_out, const uint8_t* font, size_t font_len, const char* dest_name)
347{
348 s_disk.block_count = (uint32_t)k_blocks_fat16;
349 s_disk.bytes = (uint8_t*)calloc(1U, (size_t)k_blocks_fat16 * (size_t)k_block_size);
350 if (s_disk.bytes == nullptr) {
351 (void)fprintf(stderr, "mkfontimg: out of memory (disk)\n");
352 return 1;
353 }
354 build_fat16(s_disk.bytes);
355
356 const ra8_fs_backend_t backend = {.read_block = mem_read,
357 .write_block = mem_write,
358 .get_capacity = mem_cap,
359 .ctx = &s_disk};
360
361 /* Write the font (if any) through the real ra8_fs so the on-card layout
362 * matches exactly what the firmware app will read. */
363 ra8_fs_mount_t* mnt = nullptr;
364 if (ra8_fs_mount(&backend, &mnt) != k_ra8_ok) {
365 (void)fprintf(stderr, "mkfontimg: ra8_fs_mount failed\n");
366 free(s_disk.bytes);
367 return 1;
368 }
369 if (write_font_file(mnt, font, font_len, dest_name) != 0) {
370 free(s_disk.bytes);
371 return 1;
372 }
373 (void)ra8_fs_unmount(mnt);
374
375 const int rc = dump_image(image_out);
376 free(s_disk.bytes);
377 return rc;
378}

400static uint8_t* slurp_font(const char* font_in, size_t* font_len)
401{
402 uint8_t* font = (uint8_t*)malloc((size_t)k_font_cap);
403 if (font == nullptr) {
404 (void)fprintf(stderr, "mkfontimg: out of memory\n");
405 return nullptr;
406 }
407 FILE* fin = fopen(font_in, "rb");
408 if (fin == nullptr) {
409 (void)fprintf(stderr, "mkfontimg: cannot open %s\n", font_in);
410 free(font);
411 return nullptr;
412 }
413 const size_t len = fread(font, 1U, (size_t)k_font_cap, fin);
414 (void)fclose(fin);
415 if (len < (size_t)k_font_min_bytes) {
416 (void)fprintf(stderr, "mkfontimg: %s too small (%zu bytes)\n", font_in, len);
417 free(font);
418 return nullptr;
419 }
420 *font_len = len;
421 return font;
422}

483static int run_font(int argc, char** argv)
484{
485 if (argc < 3) {
486 (void)fprintf(stderr, "usage: %s <font-in> <image-out> [dest-name]\n", argv[0]);
487 return 2;
488 }
489 const char* font_in = argv[1];
490 const char* image_out = argv[2];
491 const char* dest_name = (argc > 3) ? argv[3] : "FONT.OTF";
492
493 size_t font_len = 0U;
494 uint8_t* font = slurp_font(font_in, &font_len);
495 if (font == nullptr) {
496 return 1;
497 }
498 const int rc = build_and_dump(image_out, font, font_len, dest_name);
499 free(font);
500 if (rc != 0) {
501 return 1;
502 }
503 (void)fprintf(stderr, "mkfontimg: wrote %s (%s = %zu bytes)\n", image_out, dest_name, font_len);
504 return 0;
505}
