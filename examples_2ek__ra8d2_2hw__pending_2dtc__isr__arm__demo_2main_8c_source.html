355static void dtc_arm_program_ti(void)
356{
357 const uint8_t mra = (uint8_t)(((uint8_t)k_dtc_arm_md_block << k_dtc_arm_mra_md_pos) |
358 ((uint8_t)k_dtc_arm_sz_word << k_dtc_arm_mra_sz_pos) |
359 ((uint8_t)k_dtc_arm_sm_inc << k_dtc_arm_mra_sm_pos));
360 const uint8_t mrb = (uint8_t)((uint8_t)k_dtc_arm_dm_inc << k_dtc_arm_mrb_dm_pos);
361 /* MR[31:24]=MRA, MR[23:16]=MRB, MR[15:8]=MRC(0); HUM Ch 18.2.2 p 786 /
362 * 18.2.3 p 787 / 18.2.4 p 789, Figure 18.4 p 799. SRAM (not MMIO). */
363 s_dtc_ti.MR =
364 ((uint32_t)mra << k_dtc_arm_mra_byte_pos) | ((uint32_t)mrb << k_dtc_arm_mrb_byte_pos);
365 s_dtc_ti.SAR = (uint32_t)(uintptr_t)s_src;
366 s_dtc_ti.DAR = (uint32_t)(uintptr_t)s_dst;
367 s_dtc_ti.CRB = (uint16_t)k_dtc_arm_crb_one_block;
368 s_dtc_ti.CRA = (uint16_t)k_dtc_arm_cra_block_256;
369}

384static void dtc_arm_bringup_or_halt(void)
385{
386 if (ra8_dtc_init(s_dtc_vt) != k_ra8_ok) {
387 dtc_arm_panic_halt();
388 }
389 if (ra8_dtc_attach_handler(dtc_arm_complete_cb, nullptr) != k_ra8_ok) {
390 dtc_arm_panic_halt();
391 }
392 if (ra8_isr_register((ra8_elc_event_t)k_dtc_arm_event_swevt0,
393 dtc_arm_swevt_isr,
394 nullptr,
395 (uint8_t)k_dtc_arm_isr_prio,
396 &s_dtc_slot) != k_ra8_ok) {
397 dtc_arm_panic_halt();
398 }
399 if (s_dtc_slot >= (uint16_t)k_dtc_arm_vt_entries) {
400 dtc_arm_panic_halt();
401 }
402 /* DTCVBR + slot*4 holds the 16-byte-aligned TI start address; bit0 is the
403 * privilege attribution (0 = privileged). HUM Ch 18.3.1 p 796 + Figure 18.3
404 * p 798. SRAM vector-table write (not MMIO). */
405 s_dtc_vt[s_dtc_slot] = (uint32_t)(uintptr_t)&s_dtc_ti;
406 if (ra8_dtc_enable() != k_ra8_ok) {
407 dtc_arm_panic_halt();
408 }

476[[nodiscard]] static ra8_err_t dtc_arm_run_armed(uint8_t* out_ok)
477{
478 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
479
480 dtc_arm_fill(0U);
481 dtc_arm_program_ti();
482 const ra8_err_t arm = ra8_isr_set_dtc(s_dtc_slot, true);
483 if (arm != k_ra8_ok) {
484 return arm;
485 }
486 const ra8_err_t trig = ra8_elc_software_trigger((uint8_t)k_dtc_arm_swevt_index);
487 if (trig != k_ra8_ok) {
488 return trig;
489 }
490 ++g_dtc_activations;
491
492 for (uint32_t i = 0U; i < (uint32_t)k_dtc_arm_poll_limit; ++i) {
493 if (s_dst[k_dtc_arm_buf_words - 1U] == s_src[k_dtc_arm_buf_words - 1U]) {
494 break;
495 }
496 }
497
498 *out_ok = dtc_arm_all_match_src();
499 return k_ra8_ok;
500}

520[[nodiscard]] static ra8_err_t dtc_arm_run_disarmed(uint8_t* out_ok)
521{
522 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
523
524 dtc_arm_fill((uint32_t)k_dtc_arm_dst_sentinel);
525 dtc_arm_program_ti();
526 const ra8_err_t disarm = ra8_isr_set_dtc(s_dtc_slot, false);
527 if (disarm != k_ra8_ok) {
528 return disarm;
529 }
530 const ra8_err_t trig = ra8_elc_software_trigger((uint8_t)k_dtc_arm_swevt_index);
531 if (trig != k_ra8_ok) {
532 return trig;
533 }
534 ++g_dtc_activations;
535
536 /* No copy is expected, so there is nothing to poll for; give the (gated)
537 * DTC a bounded settle window, then confirm the destination is untouched.
538 * The volatile read keeps the loop from being optimised away. */
539 for (uint32_t i = 0U; i < (uint32_t)k_dtc_arm_settle; ++i) {
540 if (s_dst[k_dtc_arm_buf_words - 1U] != (uint32_t)k_dtc_arm_dst_sentinel) {
541 break;
542 }
543 }
544
545 *out_ok = dtc_arm_all_equal((uint32_t)k_dtc_arm_dst_sentinel);
546 return k_ra8_ok;
547}

551int32_t main(void)
552{
553 dtc_arm_setup_or_halt();
554 dtc_arm_bringup_or_halt();
555 /* Completion is detected by polling (armed) or a settle window (disarmed),
556 * so the global interrupt is intentionally left masked -- as in
557 * dtc_transfer_demo, enabling the DTC-complete CPU interrupt lets its ISR
558 * (which writes DTCSTS) race the in-flight transfer on silicon. The IELSR
559 * slot armed by ra8_isr_set_dtc still activates the DTC; we simply do not
560 * take the completion IRQ. */
561
562 while (1) {
563 uint8_t armed_ok = 0U;
564 uint8_t disarmed_ok = 0U;
565 const ra8_err_t e_arm = dtc_arm_run_armed(&armed_ok);
566 const ra8_err_t e_dis = dtc_arm_run_disarmed(&disarmed_ok);
567
568 g_dtc_armed_ok = (e_arm == k_ra8_ok && armed_ok != 0U) ? 1U : 0U;
569 g_dtc_disarmed_ok = (e_dis == k_ra8_ok && disarmed_ok != 0U) ? 1U : 0U;
570
571 /* The DTCE primitive is correct only if arming COPIED and disarming did
572 * NOT -- both must hold. This compound decision is the demo's verdict. */
573 const uint8_t good = (g_dtc_armed_ok != 0U && g_dtc_disarmed_ok != 0U) ? 1U : 0U;
574
575 if (good != 0U) {
576 (void)ra8_board_uart_console_write(k_dtc_arm_ok_msg, (size_t)(sizeof(k_dtc_arm_ok_msg) - 1U));
577 (void)ra8_board_led_toggle(k_ra8_board_led1);
578 } else {
579 (void)ra8_board_uart_console_write(k_dtc_arm_bad_msg,
580 (size_t)(sizeof(k_dtc_arm_bad_msg) - 1U));
581 (void)ra8_board_led_toggle(k_ra8_board_led2);
582 }
583 ++g_dtc_heartbeat;
584 ra8_delay_ms(k_dtc_arm_period_ms);
585 }
586 return 0;
587}
