169uint8_t ra8_sdmmc_spi_crc7(const uint8_t* data, uint32_t len)
170{
171 uint8_t crc = 0U;
172 if (data == nullptr) {
173 return 0U;
174 }
175 for (uint32_t i = 0U; i < len; i++) {
176 uint8_t byte = data[i];
177 for (uint8_t b = 0U; b < (uint8_t)k_sd_bit_byte; b++) {
178 const uint8_t top = (uint8_t)((crc & (uint8_t)k_sd_crc7_msb_test) ^
179 ((byte & (uint8_t)k_sd_crc7_byte_msb) >> 1U));
180 crc = (uint8_t)((crc << 1U) & (uint8_t)k_sd_crc7_register_mask);
181 if (top != 0U) {
182 crc ^= (uint8_t)k_sd_crc7_poly_low7;
183 }
184 byte = (uint8_t)(byte << 1U);
185 }
186 }
187 return crc;
188}

190uint16_t ra8_sdmmc_spi_crc16(const uint8_t* data, uint32_t len)
191{
192 uint16_t crc = 0U;
193 if (data == nullptr) {
194 return 0U;
195 }
196 for (uint32_t i = 0U; i < len; i++) {
197 crc ^= (uint16_t)((uint16_t)data[i] << k_sd_bit_byte);
198 for (uint8_t b = 0U; b < (uint8_t)k_sd_bit_byte; b++) {
199 if ((crc & (uint16_t)k_sd_crc16_msb_test) != 0U) {
200 crc = (uint16_t)((crc << 1U) ^ (uint16_t)k_sd_crc16_poly);
201 } else {
202 crc = (uint16_t)(crc << 1U);
203 }
204 }
205 }
206 return crc;
207}

270/* Build a 6-byte command frame in ``out_frame`` -- see implementation for details. */
271static void internal_build_frame(sd_cmd_t cmd, uint32_t arg, uint8_t* out_frame)
272{
273 out_frame[k_sd_frame_idx_cmd] = (uint8_t)cmd;
274 out_frame[k_sd_frame_idx_arg_msb] =
275 (uint8_t)((arg >> k_sd_bit_three_byte) & (uint32_t)k_sd_mask_byte);
276 out_frame[k_sd_frame_idx_arg_byte2] =
277 (uint8_t)((arg >> k_sd_bit_two_byte) & (uint32_t)k_sd_mask_byte);
278 out_frame[k_sd_frame_idx_arg_byte1] =
279 (uint8_t)((arg >> k_sd_bit_byte) & (uint32_t)k_sd_mask_byte);
280 out_frame[k_sd_frame_idx_arg_lsb] = (uint8_t)(arg & (uint32_t)k_sd_mask_byte);
281 if (cmd == k_sd_cmd_go_idle_state) {
282 out_frame[k_sd_frame_idx_crc] = (uint8_t)k_sd_crc7_cmd0_byte;
283 } else if ((cmd == k_sd_cmd_send_if_cond) && (arg == (uint32_t)k_sd_cmd8_arg_check_pattern)) {
284 out_frame[k_sd_frame_idx_crc] = (uint8_t)k_sd_crc7_cmd8_byte;
285 } else {
286 const uint8_t crc7 = ra8_sdmmc_spi_crc7(out_frame, k_sd_cmd_frame_len);
287 out_frame[k_sd_frame_idx_crc] = (uint8_t)(((uint32_t)crc7 << 1U) | 1U);
288 }
289}

291/* Wait for the R1 token (first byte without the sentinel bit set) -- see implementation for details. */
292static ra8_err_t internal_read_r1(uint8_t* out_r1)
293{
294 for (uint32_t i = 0U; i < (uint32_t)k_sd_max_r1_wait_bytes; i++) {
295 uint8_t byte = 0U;
296 ra8_err_t err = ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &byte);
297 if (err != k_ra8_ok) {
298 return err;
299 }
300 if ((byte & (uint8_t)k_sd_r1_sentinel) == 0U) {
301 *out_r1 = byte;
302 return k_ra8_ok;
303 }
304 }
305 return k_ra8_err_hw_timeout;
306}

308/* Send a command frame and capture the R1 response byte -- see implementation for details. */

336/* Send CMD12 (STOP_TRANSMISSION) with the stuff-byte skip before R1 -- see implementation for details. */
337ra8_err_t ra8_sdmmc_spi_send_stop_transmission(uint8_t* out_r1)
338{
339 uint8_t frame[k_ra8_sdmmc_spi_cmd_frame_bytes];
340 internal_build_frame(k_sd_cmd_stop_transmission, 0U, frame);
341 uint8_t rx_dummy[k_ra8_sdmmc_spi_cmd_frame_bytes];
342 ra8_err_t err = s_sdmmc_spi_state.transport.xfer(s_sdmmc_spi_state.transport.ctx,
343 frame,
344 rx_dummy,
345 (uint32_t)k_ra8_sdmmc_spi_cmd_frame_bytes);
346 if (err != k_ra8_ok) {
347 return err;
348 }
349 /* The byte immediately following the CMD12 frame is a stuff byte: it
350 * carries the tail of the interrupted read stream and its value is
351 * undefined, so it must be discarded before the R1 poll -- a bit7-clear
352 * garbage byte would otherwise be misread as the response. This is why
353 * CMD12 cannot go through ra8_sdmmc_spi_send_command, which polls for R1
354 * straight after the frame. */
355 err = ra8_sdmmc_spi_send_idle(1U);
356 if (err != k_ra8_ok) {
357 return err;
358 }
359 return internal_read_r1(out_r1);
360}

367/* Drain the four trailing bytes of an R3 / R7 response -- see implementation for details. */
368static ra8_err_t internal_read_r3_or_r7_tail(uint32_t* out_word)
369{
370 uint8_t bytes[4] = {};
371 ra8_err_t err =
372 s_sdmmc_spi_state.transport.xfer(s_sdmmc_spi_state.transport.ctx, nullptr, bytes, 4U);
373 if (err != k_ra8_ok) {
374 /* Some transports do not allow NULL tx; fall back to per-byte. */
375 for (uint32_t i = 0U; i < 4U; i++) {
376 err = ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &bytes[i]);
377 if (err != k_ra8_ok) {
378 return err;
379 }
380 }
381 }
382 *out_word = ((uint32_t)bytes[0] << k_sd_bit_three_byte) |
383 ((uint32_t)bytes[1] << k_sd_bit_two_byte) | ((uint32_t)bytes[2] << k_sd_bit_byte) |
384 (uint32_t)bytes[3];
385 return k_ra8_ok;
386}

436/* Compute capacity (in 512-byte blocks) from a 16-byte CSD register -- see implementation for details. */
437static uint32_t internal_csd_to_blocks(const uint8_t* csd)
438{
439 const uint8_t version = (uint8_t)((csd[k_sd_csd_byte_version] >> k_sd_csd_version_shift) & 0x03U);
440 if (version == 1U) {
441 /* CSD v2: ``capacity = (C_SIZE + 1) * 512 * 1024 / 512``. */
442 const uint32_t c_size =
443 (((uint32_t)csd[k_sd_csd_v2_byte_csize_msb] & k_sd_csize_msb_mask) << k_sd_bit_two_byte) |
444 ((uint32_t)csd[k_sd_csd_v2_byte_csize_mid] << k_sd_bit_byte) |
445 (uint32_t)csd[k_sd_csd_v2_byte_csize_lsb];
446 return (c_size + 1U) << (uint32_t)k_sd_csd_v2_blocks_shift;
447 }
448 if (version == 0U) {
449 /* CSD v1 (SD spec PHY v9 section 5.3.2):
450 * capacity = (C_SIZE+1) * 2^(C_SIZE_MULT+2) * 2^READ_BL_LEN / 512.
451 * C_SIZE is a 12-bit field spanning csd[6][1:0] (high 2 bits),
452 * csd[7] (middle 8 bits), csd[8][7:6] (low 2 bits). */
453 const uint8_t read_bl_len = (uint8_t)(csd[5] & k_sd_read_bl_mask);
454 const uint32_t c_size = (((uint32_t)csd[6] & 0x03U) << k_sd_csize_shift) |
455 ((uint32_t)csd[7] << 2U) |
456 ((uint32_t)(csd[8] & k_sd_csize_lo_mask) >> 6U);
457 const uint8_t c_size_mult =
458 (uint8_t)((((uint8_t)csd[9] & 0x03U) << 1U) |
459 (((uint8_t)csd[10] & k_sd_mult_lo_mask) >> k_sd_mult_shift));
460 const uint32_t mult = (uint32_t)1U << ((uint32_t)c_size_mult + 2U);
461 const uint32_t block_len = (uint32_t)1U << (uint32_t)read_bl_len;
462 /* Multiply in 64 bits to avoid overflow before dividing by 512. */
463 const uint64_t bytes = (uint64_t)(c_size + 1U) * (uint64_t)mult * (uint64_t)block_len;
464 return (uint32_t)(bytes / (uint64_t)k_ra8_sdmmc_spi_block_size);
465 }
466 return 0U;
467}

517static void internal_recover_stuck_card(void)
518{
519 /* Phase 1: drain busy + reset byte framing with CS released. */
520 (void)s_sdmmc_spi_state.transport.cs(s_sdmmc_spi_state.transport.ctx, false);
521 (void)ra8_sdmmc_spi_send_idle((uint32_t)k_sd_recover_idle_bytes);
522
523 /* Phase 2: complete/abort a wedged data phase with CS asserted. */
524 if (s_sdmmc_spi_state.transport.cs(s_sdmmc_spi_state.transport.ctx, true) != k_ra8_ok) {
525 return;
526 }
527 (void)ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_stop_multi, nullptr);
528 (void)ra8_sdmmc_spi_send_idle((uint32_t)k_sd_recover_flush_bytes);
529 (void)ra8_sdmmc_spi_wait_not_busy();
530
531 /* Phase 3: explicit STOP_TRANSMISSION for any open multi-block transfer. */
532 uint8_t r1 = 0U;
533 (void)ra8_sdmmc_spi_send_command(k_sd_cmd_stop_transmission, 0U, &r1);
534 (void)ra8_sdmmc_spi_wait_not_busy();
535 (void)s_sdmmc_spi_state.transport.cs(s_sdmmc_spi_state.transport.ctx, false);
536
537 /* Phase 4: settle framing before the caller's wake/CMD0. */
538 (void)ra8_sdmmc_spi_send_idle((uint32_t)k_sd_recover_idle_bytes);

648/* Send CMD9 (SEND_CSD) and parse capacity -- see implementation for details. */
649static ra8_err_t internal_read_csd(uint32_t* out_blocks)
650{
651 ra8_err_t err = ra8_sdmmc_spi_cs_assert();
652 if (err != k_ra8_ok) {
653 return err;
654 }
655 uint8_t r1 = 0U;
656 err = ra8_sdmmc_spi_send_command(k_sd_cmd_send_csd, 0U, &r1);
657 if (err != k_ra8_ok) {
658 (void)ra8_sdmmc_spi_cs_release();
659 return err;
660 }
661 if (r1 != 0U) {
662 (void)ra8_sdmmc_spi_cs_release();

664 }
665 err = ra8_sdmmc_spi_wait_data_token();

667 (void)ra8_sdmmc_spi_cs_release();

671 for (uint32_t i = 0U; i < (uint32_t)k_ra8_sdmmc_spi_csd_response_len; i++) {

673 if (err != k_ra8_ok) {
674 (void)ra8_sdmmc_spi_cs_release();
675 return err;
676 }
677 }
678 /* Drain trailing CRC16 (2 bytes). */
679 uint8_t crc_bytes[2] = {};
680 (void)ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &crc_bytes[0]);
681 (void)ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &crc_bytes[1]);
682 (void)ra8_sdmmc_spi_cs_release();

686 }

722/* Drive the CMD0..CMD8..ACMD41..CMD58 probe sequence (no CSD/CMD16) -- see implementation for details. */
723static ra8_err_t internal_probe_card(bool* out_is_v2, bool* out_is_hc)
724{
725 ra8_err_t err = internal_wake_card();
726 if (err == k_ra8_ok) {
727 err = internal_send_cmd0();
728 }
729 /* CMD0 failed: the card may be stranded mid-write from an interrupted
730 * transaction. Flush it and retry, escalating across a few attempts. A
731 * healthy card answers CMD0 on the first pass and never enters this loop, so
732 * the recovery cannot disturb a normal bring-up. */
733 for (uint32_t attempt = 0U; (err != k_ra8_ok) && (attempt < (uint32_t)k_sd_max_recover_attempts);
734 attempt++) {
735 internal_recover_stuck_card();
736 err = internal_wake_card();
737 if (err == k_ra8_ok) {
738 err = internal_send_cmd0();
739 }
740 }
741 if (err != k_ra8_ok) {
742 return err;
743 }
744 err = internal_send_cmd8(out_is_v2);
745 if (err != k_ra8_ok) {
746 return err;
747 }
748 err = internal_acmd41_loop(*out_is_v2);
749 if (err != k_ra8_ok) {
750 return err;
751 }
752 *out_is_hc = false;
753 if (*out_is_v2) {
754 err = internal_read_ocr(out_is_hc);
755 }
756 return err;
757}

759/* Walk the full SD identification sequence and learn capacity / type -- see implementation for details. */
760ra8_err_t ra8_sdmmc_spi_run_init_sequence(void)
761{
762 bool is_v2 = false;
763 bool is_hc = false;
764 ra8_err_t err = internal_probe_card(&is_v2, &is_hc);
765 if (err != k_ra8_ok) {
766 return err;
767 }
768 uint32_t blocks = 0U;
769 err = internal_read_csd(&blocks);
770 RA8_RETURN_ON_ERROR(err, s_tag, "CMD9 SEND_CSD");
771 /* SDSC (v1.x or v2.x SDSC) needs an explicit block-size set; HC cards
772 * accept it as a no-op but the spec says we should still issue it. */
773 err = internal_set_block_len();
774 RA8_RETURN_ON_ERROR(err, s_tag, "CMD16 SET_BLOCKLEN");
775 s_sdmmc_spi_state.card_type = internal_classify_card(is_v2, is_hc);
776 s_sdmmc_spi_state.capacity_blocks = blocks;
777 return k_ra8_ok;
