176RA8_INTERNAL static bool internal_wait_ovf(void)
177{
178 enum : uint32_t { k_poll_budget = 200000U };
179 for (uint32_t i = 0U; i < k_poll_budget; ++i) {
180 uint32_t status = 0U;
181 if (ra8_gpt_get_status((uint8_t)k_gpt_os_demo_channel, &status) != k_ra8_ok) {
182 return false;
183 }
184 if ((status & (uint32_t)k_ra8_gpt_status_overflow) != 0U) {
185 (void)ra8_gpt_clear_status((uint8_t)k_gpt_os_demo_channel,
186 (uint32_t)k_ra8_gpt_status_overflow);
187 return true;
188 }
189 }
190 return false;
191}

207void main(void)
208{
209 internal_setup_or_halt();
210 ra8_isr_globals_enable();
211
212 if (internal_arm() != k_ra8_ok) {
213 internal_panic_halt();
214 }
215
216 while (1) {
217 if (ra8_gpt_start_free_run((uint8_t)k_gpt_os_demo_channel, (uint32_t)k_gpt_os_demo_period) !=
218 k_ra8_ok) {
219 g_gpt_one_shot_mismatch += 1U;
220 ra8_delay_ms((uint32_t)k_gpt_os_demo_rearm_delay_ms);
221 continue;
222 }
223 if (internal_wait_ovf()) {
224 g_gpt_one_shot_match += 1U;
225 (void)ra8_board_led_toggle(k_ra8_board_led1);
226 } else {
227 g_gpt_one_shot_mismatch += 1U;
228 }
229 ra8_delay_ms((uint32_t)k_gpt_os_demo_rearm_delay_ms);
230 }
231 internal_panic_halt();
