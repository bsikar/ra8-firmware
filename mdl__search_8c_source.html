32RA8_INTERNAL static bool is_unreserved(unsigned char c)
33{
34 const bool alpha = ((c >= 'A') && (c <= 'Z')) || ((c >= 'a') && (c <= 'z'));
35 const bool digit = (c >= '0') && (c <= '9');
36 const bool mark = (c == '-') || (c == '.') || (c == '_') || (c == '~');
37 return alpha || digit || mark;
38}

47bool mdl_query_encode(const char* term, char* out, size_t cap)
48{
49 if ((term == nullptr) || (out == nullptr) || (cap == 0U)) {
50 if ((out != nullptr) && (cap > 0U)) {
51 out[0] = '\0';
52 }
53 return false;
54 }
55 size_t n = 0U;
56 for (size_t i = 0U; term[i] != '\0'; ++i) {
57 const unsigned char c = (unsigned char)term[i];
58 if (is_unreserved(c)) {
59 if ((n + 1U) >= cap) {
60 out[0] = '\0';
61 return false;
62 }
63 out[n] = (char)c;
64 ++n;
65 } else {
66 if ((n + (size_t)k_triplet_len) >= cap) {
67 out[0] = '\0';
68 return false;
69 }
70 out[n] = '%';
71 out[n + 1U] = hex_digit((unsigned)c >> (unsigned)k_hi_nibble_shift);
72 out[n + 2U] = hex_digit((unsigned)c);
73 n += (size_t)k_triplet_len;
74 }
75 }
76 out[n] = '\0';
77 return true;
78}

80bool mdl_search_build_url(const char* tmpl, const char* encoded_term, char* out, size_t cap)
81{
82 if ((tmpl == nullptr) || (encoded_term == nullptr) || (out == nullptr) || (cap == 0U)) {
83 if ((out != nullptr) && (cap > 0U)) {
84 out[0] = '\0';
85 }
86 return false;
87 }
88 const size_t plen = strlen(s_placeholder);
89 const size_t tlen = strlen(encoded_term);
90 size_t n = 0U;
91 bool hit = false;
92 size_t i = 0U;
93 while (tmpl[i] != '\0') {
94 if (strncmp(tmpl + i, s_placeholder, plen) == 0) {
95 if ((n + tlen) >= cap) {
96 out[0] = '\0';
97 return false;
98 }
99 memcpy(out + n, encoded_term, tlen);
100 n += tlen;
101 i += plen;
102 hit = true;
103 } else {
104 if ((n + 1U) >= cap) {
105 out[0] = '\0';
106 return false;
107 }
108 out[n] = tmpl[i];
109 ++n;
110 ++i;
111 }
112 }
113 if (!hit) {
114 out[0] = '\0';
115 return false; /* a template with no {q} could never carry the term */
116 }
117 out[n] = '\0';
118 return true;
119}
