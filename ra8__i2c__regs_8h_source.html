355RA8_HW_REGISTER_ACCESS
356static inline volatile r_i2c_regs_t* ra8_i2c_regs(uint8_t channel)
357{
358 if ((uint16_t)channel >= (uint16_t)k_ra8_i2c_channel_count) {
359 return nullptr;
360 }
361 const uintptr_t base =
362 (uintptr_t)k_ra8_i2c0_base_addr + ((uintptr_t)channel * (uintptr_t)k_ra8_i2c_channel_stride);
363 return (volatile r_i2c_regs_t*)base;
364}
