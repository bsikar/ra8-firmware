60static void internal_kbd_key(const ra8_widget_keyboard_t* kbd, const ra8_widget_key_info_t* info)
61{
62 const ra8_widget_paint_t* p = kbd->paint;
63 ra8_widget_priv_fill_box(p, &info->rect, kbd->key_face, kbd->key_border, kbd->border_w);
64 if (p->draw_text == nullptr) {
65 return;
66 }
67 char glyph_buf[k_ra8_widget_kbd_glyph_len] = {};
68 const char* text = info->label;
69 if (info->glyph != (char)k_ra8_widget_kbd_no_glyph) {
70 glyph_buf[k_ra8_widget_kbd_glyph_char] = info->glyph;
71 glyph_buf[k_ra8_widget_kbd_glyph_nul] = '\0';
72 text = glyph_buf;
73 }
74 if (text == nullptr) {
75 return;
76 }
77 int32_t tx = 0;
78 int32_t ty = 0;
79 ra8_widget_priv_text_pos(p, &info->rect, text, (int16_t)0, k_ra8_widget_align_center, &tx, &ty);
80 p->draw_text(p->user, tx, ty, text, kbd->key_fg, kbd->key_face);
81}

99static void internal_kbd_render(ra8_widget_t* w)
100{
101 ra8_widget_keyboard_t* kbd = (ra8_widget_keyboard_t*)w->ctx;
102 if (kbd == nullptr) {
103 return;
104 }
105 if (kbd->paint == nullptr) {
106 return;
107 }
108 ra8_widget_priv_fill_box(kbd->paint, &w->rect, kbd->bg, kbd->bg, (int16_t)0);
109 if ((kbd->ops == nullptr) || (kbd->ops->count == nullptr) || (kbd->ops->key_info == nullptr)) {
110 return;
111 }
112 const uint8_t n = kbd->ops->count(kbd->ops->user);
113 for (uint8_t i = 0U; i < n; ++i) {
114 ra8_widget_key_info_t info = {};
115 kbd->ops->key_info(kbd->ops->user, i, &info);
116 internal_kbd_key(kbd, &info);
117 }
118}

139RA8_INTERNAL
140static bool internal_kbd_on_input(ra8_widget_t* w, const ra8_widget_event_t* ev)
141{
142 ra8_widget_keyboard_t* kbd = (ra8_widget_keyboard_t*)w->ctx;
143 if (kbd == nullptr) {
144 return false;
145 }
146 if (ev->kind != k_ra8_widget_ev_touch) {
147 return false;
148 }
149 if ((kbd->ops == nullptr) || (kbd->ops->hit == nullptr) || (kbd->ops->apply == nullptr)) {
150 return true;
151 }
152 const uint8_t key = kbd->ops->hit(kbd->ops->user, ev->x, ev->y);
153 if (key == (uint8_t)k_ra8_widget_key_no_hit) {
154 return true;
155 }
156 const bool committed = kbd->ops->apply(kbd->ops->user, key);
157 (void)ra8_widget_invalidate(w, k_ra8_widget_refresh_quality);
158 if (committed && (kbd->on_commit != nullptr)) {
159 kbd->on_commit(w);
160 }
161 return true;
162}
