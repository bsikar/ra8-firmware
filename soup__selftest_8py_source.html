3"""Both-directions selftest for the vendored-SOUP upstream provenance gate.
4
5It lives beside ``check_soup_upstream.py`` rather than inside it only because
6the checker crossed the project file-size cap; ``--selftest`` on that script is
7still the entry point, and the ``soup-upstream`` gate runs it before the scan.
8
9Two properties are asserted, because this claim -- "byte-identical to upstream"
10-- was stated in three places and checked by nothing, so every tree passed it
11and a drifted one would have too:
12
13 * **it fires.** A mutated blob, a changed file mode, a lost file, a ghost
14 manifest row, an edit on top of a reviewed patch, an undeclared deviation,
15 a stale declaration, a pin that disagrees with the SBOM's, and every
16 vacuity floor -- each is provoked one at a time against a REAL scratch git
17 repository and driven through ``run_check()``, the function CI calls.
18 * **it stays quiet.** The untouched fixture verifies clean, every record kind
19 round-trips through the manifest format, and the shipped floors are real
20 numbers rather than zero-with-a-comment.
21"""

65FIXTURE_PATH = "libs/third_party/fixture"
66FIXTURE_FLOORS = (1, 1, 1)
67# The fixture manifest has two `ok`/`moved` rows among its four records; the
68# other two are the declared patch and the declared local file.
69FIXTURE_VERIFIED_ROWS = 2
70# The shipped floors must be real numbers, not 0-with-a-comment. A floor of
71# zero passes for a scan that covered nothing, which is the failure this whole
72# family of constants exists to prevent.
73FLOOR_SANITY_MIN = 1000
74_FIXTURE_FILES = {
75 "src/a.c": b"int a;\n",
76 "src/b.c": b"int b;\n",
77 "LICENSE": b"MIT\n",
78 "patched.c": b"int patched; /* local */\n",
79 "generated.h": b"/* generated here, not upstream */\n",
80}

83def _quiet(func: Callable[..., int], *args: object) -> int:
84 """Call `func` with its diagnostics captured, returning only its status.
85
86 The must-fire cases below deliberately provoke real failures; letting their
87 error text through would bury the pass/fail report the selftest exists to
88 print. The status is what is asserted, so only the status is kept.
89 """
90 sink = io.StringIO()
91 with contextlib.redirect_stdout(sink), contextlib.redirect_stderr(sink):
92 return func(*args)

95def _fixture_component(**overrides: object) -> Component:
96 """Build the selftest's synthetic registry entry."""
97 fields: dict[str, object] = {
98 "key": "fixture",
99 "name": "fixture",
100 "version": "0",
101 "ctype": "library",
102 "url": "https://example.invalid/fixture",
103 "path": FIXTURE_PATH,
104 "provenance": "commit-pinned-sha256",
105 "description": "selftest fixture",
106 "upstream_commit": "0" * 40,

126def _write_fixture_repo(root: Path) -> dict[str, tuple[str, str]]:
127 """Materialise a REAL git repository holding the fixture vendored tree.
128
129 A real repository, not a stub: the blob ids the gate compares come out of a
130 git index, so a fixture that faked them would exercise a different code
131 path from the one CI runs.
132
133 Args:
134 root: Scratch directory to initialise as a repository.
135
136 Returns:
137 ``{rel path: (mode, blob)}`` exactly as `git_ls_files` will report it.
138 """
139 _git(["init", "-q", "-b", "main", "."], root)
140 for rel, data in _FIXTURE_FILES.items():
141 target = root / FIXTURE_PATH / rel
142 target.parent.mkdir(parents=True, exist_ok=True)
143 target.write_bytes(data)
144 _git(["add", "-A"], root)
145 return git_ls_files(FIXTURE_PATH, (), root)

148def _write_fixture_manifest(root: Path, ours: dict[str, tuple[str, str]], **mutate: str) -> None:
149 """Write the fixture's manifest, optionally corrupting one field.
150
151 Args:
152 root: Scratch repository root.
153 ours: The fixture's ``{rel path: (mode, blob)}``.
154 mutate: ``kind``/``blob``/``mode``/``drop``/``extra`` knobs the
155 must-fire cases use to break exactly one thing.
156 """
157 entries: list[Entry] = []
158 for rel, (mode, blob) in sorted(ours.items()):
159 if rel == mutate.get("drop"):
160 continue
161 if rel == "patched.c":
162 entries.append(Entry(KIND_PATCH, mode, rel, upstream_blob="1" * 40, local_blob=blob))
163 elif rel == "generated.h":
164 entries.append(Entry(KIND_LOCAL, mode, rel, local_blob=blob))
165 else:
166 entries.append(Entry(KIND_OK, mode, rel, upstream_blob=blob))
167 entries = [
168 Entry(
169 e.kind,
170 mutate["mode"] if e.rel_path == mutate.get("mode_of") else e.mode,
171 e.rel_path,
172 upstream_blob=("2" * 40 if e.rel_path == mutate.get("blob_of") else e.upstream_blob),
173 local_blob=("3" * 40 if e.rel_path == mutate.get("local_of") else e.local_blob),
174 upstream_path=e.upstream_path,
175 )
176 for e in entries
177 ]
178 if mutate.get("extra"):
179 entries.append(Entry(KIND_OK, "100644", mutate["extra"], upstream_blob="4" * 40))
180 header = {"upstream-url": "https://example.invalid", "ref": "v0", "commit": "0" * 40}
181 out = root / manifest_path("fixture")
182 out.parent.mkdir(parents=True, exist_ok=True)
183 out.write_text(format_manifest("fixture", header, entries), encoding="utf-8")

186def _selftest_manifest_cases(
187 root: Path, ours: dict[str, tuple[str, str]]
188) -> list[tuple[str, bool]]:
189 """Break the MANIFEST one way at a time and assert `run_check`'s verdict.
190
191 Args:
192 root: The scratch repository from `_write_fixture_repo`.
193 ours: That fixture's ``{rel path: (mode, blob)}``.
194
195 Returns:
196 One ``(label, passed)`` pair per assertion.
197 """
198 comps = (_fixture_component(),)
199
200 def verdict(**mutate: str) -> int:
201 _write_fixture_manifest(root, ours, **mutate)
202 return _quiet(run_check, comps, root, FIXTURE_FLOORS)
203
204 return [
205 ("MUST NOT FIRE: an untouched fixture verifies clean", verdict() == EXIT_OK),
206 (
207 "MUST FIRE: a vendored file that is not the upstream blob",
208 verdict(blob_of="src/a.c") == EXIT_FAIL,
209 ),
210 (
211 "MUST FIRE: a mode that disagrees with upstream",
212 verdict(mode_of="src/b.c", mode="100755") == EXIT_FAIL,
213 ),
214 (
215 "MUST FIRE: an edit on top of a reviewed patch",
216 verdict(local_of="patched.c") == EXIT_FAIL,
217 ),
218 (
219 "MUST FIRE: a tracked file absent from the manifest",
220 verdict(drop="src/b.c") == EXIT_FAIL,
221 ),
222 (
223 "MUST FIRE: a manifest record with no file in the tree",
224 verdict(extra="ghost.c") == EXIT_FAIL,
225 ),
226 ]

229def _selftest_registry_cases(
230 root: Path, ours: dict[str, tuple[str, str]]
231) -> list[tuple[str, bool]]:
232 """Hold the manifest correct and break the REGISTRY's declarations instead.
233
234 Args:
235 root: The scratch repository from `_write_fixture_repo`.
236 ours: That fixture's ``{rel path: (mode, blob)}``.
237
238 Returns:
239 One ``(label, passed)`` pair per assertion.
240 """
241 _write_fixture_manifest(root, ours)
242 comps = (_fixture_component(),)
243
244 def verdict(**overrides: object) -> int:
245 return _quiet(run_check, (_fixture_component(**overrides),), root, FIXTURE_FLOORS)
246
247 return [
248 (
249 "MUST FIRE: a patch/local record the registry does not declare",
250 verdict(patched_files=(), local_files=()) == EXIT_FAIL,
251 ),
252 (
253 "MUST FIRE: a patched file on a component recording modified=False",
254 verdict(modified=False) == EXIT_FAIL,
255 ),
256 (
257 "MUST FIRE: a declaration for a file that is byte-identical to upstream (stale)",
258 verdict(
259 patched_files=(("patched.c", "why"), ("src/a.c", "a patch that no longer exists"))
260 )
261 == EXIT_FAIL,
262 ),
263 (
264 "MUST FIRE: the registry pin and the verified pin are different revisions",
265 verdict(upstream_commit="7" * 40) == EXIT_FAIL,
266 ),
267 (
268 "MUST FIRE: a vendored component with no manifest at all",
269 verdict(key="absent") == EXIT_VACUOUS,
270 ),
271 (
272 "MUST FIRE: a manifest whose rows prove nothing against upstream",
273 _quiet(run_check, comps, root, (1, 1, 99)) == EXIT_VACUOUS,
274 ),
275 (
276 "MUST FIRE: a scan covering fewer components than the floor",
277 _quiet(run_check, comps, root, (99, 1, 1)) == EXIT_VACUOUS,
278 ),
279 ]

311def _selftest_resolve_cases() -> list[tuple[str, bool]]:
312 """Assert `_resolve_entry`'s mappings, and its refusal to invent a deviation."""
313 comp = _fixture_component()
314 tree = {"src/a.c": ("100644", "a" * 40), "moved/here.c": ("100644", "c" * 40)}
315 cases: list[tuple[str, bool]] = [
316 (
317 "MUST NOT FIRE: a byte-identical file resolves as 'ok'",
318 _resolve_entry(comp, "src/a.c", ("100644", "a" * 40), tree).kind == KIND_OK,
319 ),
320 (
321 "MUST NOT FIRE: a relocated but identical file resolves as 'moved'",
322 _resolve_entry(comp, "flat.c", ("100644", "c" * 40), tree).upstream_path
323 == "moved/here.c",
324 ),
325 (
326 "MUST NOT FIRE: a declared patch keeps upstream's hash alongside ours",
327 _resolve_entry(
328 comp, "patched.c", ("100644", "d" * 40), {"patched.c": ("100644", "e" * 40)}
329 ).upstream_blob
330 == "e" * 40,
331 ),
332 ]
333 for label, path, tree_arg in _REFUSAL_CASES:
334 fired = False
335 try:
336 _resolve_entry(comp, path, ("100644", "9" * 40), tree_arg)
337 except VacuousScanError:
338 fired = True
339 cases.append((f"MUST FIRE: --refresh refuses {label}", fired))
340 return cases

343def _selftest_format_cases() -> list[tuple[str, bool]]:
344 """Assert the manifest format round-trips and rejects malformed records."""
345 entries = [
346 Entry(KIND_OK, "100644", "a.c", upstream_blob="a" * 40),
347 Entry(KIND_MOVED, "120000", "b.c", upstream_blob="b" * 40, upstream_path="up/b.c"),
348 Entry(KIND_PATCH, "100644", "c.c", upstream_blob="c" * 40, local_blob="d" * 40),
349 Entry(KIND_LOCAL, "100755", "d.c", local_blob="e" * 40),
350 ]
351 text = format_manifest("fixture", {"commit": "0" * 40}, entries)
352 parsed = parse_manifest("fixture", text, Path("fixture"))
353 cases = [
354 ("MUST NOT FIRE: every record kind round-trips", list(parsed.entries) == entries),
355 (
356 "MUST NOT FIRE: only ok/moved rows count as upstream-verified",
357 parsed.verified_count() == FIXTURE_VERIFIED_ROWS,
358 ),
359 ]
360 for label, bad in (
361 ("an unknown record kind", "bogus 100644 " + "a" * 40 + " a.c"),
362 ("a truncated blob id", "ok 100644 abc a.c"),
363 ("a non-octal file mode", "ok 10x644 " + "a" * 40 + " a.c"),
364 ("a duplicate path", "ok 100644 " + "a" * 40 + " a.c\nok 100644 " + "b" * 40 + " a.c"),
365 ):
366 fired = False
367 try:
368 parse_manifest("fixture", f"# component: fixture\n{bad}\n", Path("fixture"))
369 except ManifestError:
370 fired = True
371 cases.append((f"MUST FIRE: the parser rejects {label}", fired))
372 cases.append(
373 (
374 "MUST FIRE: the parser rejects a manifest naming another component",
375 _raises_manifest_error("other", "# component: fixture\n"),
376 )
377 )
378 return cases

390def run_selftest() -> int:
391 """Prove the gate fires on every provenance defect and stays quiet otherwise.
392
393 Both directions are asserted because only one of them has ever been true of
394 this claim: "byte-identical to upstream" was stated in three places and
395 checked nowhere, so every tree passed and a corrupted one would have too.
396
397 Returns:
398 ``EXIT_OK`` when every case holds, ``EXIT_VACUOUS`` otherwise.
399 """
400 with tempfile.TemporaryDirectory() as tmp:
401 root = Path(tmp)
402 ours = _write_fixture_repo(root)
403 cases = _selftest_manifest_cases(root, ours)
404 cases.extend(_selftest_registry_cases(root, ours))
405 cases.extend(_selftest_resolve_cases())
406 cases.extend(_selftest_format_cases())
407 cases.append(
408 (
409 "MUST NOT FIRE: the shipped floors are below the live tree, not zero",
410 MIN_COMPONENTS > 1
411 and MIN_ENTRIES > FLOOR_SANITY_MIN
412 and MIN_UPSTREAM_VERIFIED > FLOOR_SANITY_MIN,
413 )
414 )
415 failed = [label for label, ok in cases if not ok]
416 for label, ok in cases:
417 print(f" {'ok ' if ok else 'FAIL'} {label}")
418 if failed:
419 print(f"check_soup_upstream: selftest FAILED ({len(failed)} case(s))", file=sys.stderr)
420 return EXIT_VACUOUS
421 print(f"check_soup_upstream: selftest passed ({len(cases)} cases, both directions).")
422 return EXIT_OK
