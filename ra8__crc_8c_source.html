117static inline void internal_crc_feed_words(const uint8_t* data, uint32_t len)
118{
119 volatile r_crc_regs_t* reg = ra8_crc();
120 const uint32_t word_count = len >> 2U;
121 for (uint32_t i = 0U; i < word_count; i++) {
122 const uint32_t base = i << 2U;
123 const uint32_t packed = (uint32_t)data[base + 0U] | ((uint32_t)data[base + 1U] << 8U) |
124 ((uint32_t)data[base + 2U] << 16U) |
125 ((uint32_t)data[base + 3U] << k_crc_shift_byte3);
126 /* HUM Ch 48.2.3 "CRCDIR : CRC Data Input Register" p 3182 */
127 reg->CRCDIR = packed;
128 }
129}

157ra8_err_t ra8_crc_init(ra8_crc_poly_t poly)
158{
159 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C", p 446 */
160 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_crc);
161 RA8_RETURN_ON_ERROR(mst_err, s_tag, "crc_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
162
163 volatile r_crc_regs_t* reg = ra8_crc();
164 const uint8_t crccr0 = (uint8_t)((uint8_t)poly | k_ra8_crccr0_dorclr);
165 /* HUM Ch 48.2.1 "CRCCR0 : CRC Control Register 0" p 3181.
166 * Mirror FSP `R_CRC_Open`: write GPS plus DORCLR=1 in a single
167 * store so any stale CRCDOR value from a prior session is cleared
168 * atomically with the polynomial select. LMS defaults to 0
169 * (LSB-first); use ra8_crc_set_bit_order() to change. */
170 reg->CRCCR0 = crccr0;
171 /* HUM Ch 48.2.2 "CRCCR1 : CRC Control Register 1" p 3182 */ /* snoop off. */
172 reg->CRCCR1 = 0U;
173 ra8_log_info_val(s_tag, "crc_init poly", (uint32_t)poly);
174 return k_ra8_ok;
175}

177void ra8_crc_reset(void)
178{
179 volatile r_crc_regs_t* reg = ra8_crc();
180 /* HUM Ch 48.2.1 "CRCCR0 : CRC Control Register 0" p 3181 -- DORCLR
181 * is a write-only bit that clears CRCDOR and auto-clears itself on
182 * the real chip. We read-modify-write so the GPS/LMS bits are
183 * preserved. */
184 reg->CRCCR0 = (uint8_t)(reg->CRCCR0 | k_ra8_crccr0_dorclr);
185}

187ra8_err_t ra8_crc_compute(const uint8_t* data, uint32_t len, uint32_t* out_crc)
188{
189 RA8_CHECK_NULL_PTR(data, s_tag, "data must not be nullptr");
190 RA8_CHECK_NULL_PTR(out_crc, s_tag, "out_crc must not be nullptr");
191
192 volatile r_crc_regs_t* reg = ra8_crc();
193 /* HUM Ch 48.2.1 "CRCCR0 : CRC Control Register 0" p 3181 -- GPS lives
194 * in the low 3 bits of CRCCR0. */
195 const ra8_crc_poly_t poly = (ra8_crc_poly_t)(reg->CRCCR0 & k_ra8_crccr0_gps_mask);
196 const bool is_32_bit = ra8_crc_is_32bit_poly(poly);
197
198 /* HUM Ch 48.2.4 "CRCDOR : CRC Data Output Register" p 3182 documents
199 * CRCDOR as a 32-bit read/write register; "Because its initial value
200 * is 0x00000000, rewrite the CRCDOR ... register to perform the
201 * calculations using a value other than the initial value." Standard
202 * CRC-32 / CRC-32C use init = 0xFFFFFFFF and xor-out = 0xFFFFFFFF; the
203 * hardware applies neither, so the driver pre-seeds CRCDOR and XORs
204 * the readback on the way out for the 32-bit polynomials. CRC-8 / 16
205 * / CCITT keep the chip's natural init = 0 (HUM example p 3185 shows
206 * the same flow). */
207 if (is_32_bit) {
208 /* HUM Ch 48.2.4 "CRCDOR : CRC Data Output Register" p 3182 -- pre-seed
209 * CRCDOR with the CRC-32 init value before clocking data through (see
210 * the decision comment above for the init / xor-out rationale). */
211 reg->CRCDOR = (uint32_t)k_ra8_crc_32bit_seed;
212 internal_crc_feed_words(data, len);
213 /* HUM Ch 48.2.4 "CRCDOR : CRC Data Output Register" p 3182 -- read the
214 * running result back and apply the CRC-32 xor-out. */
215 *out_crc = reg->CRCDOR ^ (uint32_t)k_ra8_crc_32bit_seed;
216 return k_ra8_ok;
217 }
218
219 internal_crc_feed_bytes(data, len);
220 /* HUM Ch 48.2.4 p 3182 -- the full 32-bit register holds the running
221 * result; lower bits mirror `CRCDOR_HA` / `CRCDOR_BY` aliases.
222 * Reading the wide register matches FSP `crc_calculated_value_get`
223 * and is a superset of the narrower aliases. */
224 *out_crc = reg->CRCDOR;
225 return k_ra8_ok;
226}

237 reg->CRCCR0 = 0U;

243ra8_err_t ra8_crc_set_poly(ra8_crc_poly_t poly)
244{
245 volatile r_crc_regs_t* reg = ra8_crc();
246 /* HUM Ch 48.2.1 "CRCCR0 : CRC Control Register 0" p 3181 -- pulse
247 * DORCLR alongside the new GPS so the running result doesn't bleed
248 * into the next polynomial's calculation (mirrors FSP Open). */
249 reg->CRCCR0 = (uint8_t)((uint8_t)poly | k_ra8_crccr0_dorclr);
250 return k_ra8_ok;
