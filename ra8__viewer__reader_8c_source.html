67RA8_INTERNAL static bool internal_align_up(size_t value, size_t alignment, size_t* out)
68{
69 const size_t mask = alignment - 1U;
70 if ((alignment == 0U) || ((alignment & mask) != 0U) || (value > (SIZE_MAX - mask))) {
71 return false;
72 }
73 *out = (value + mask) & ~mask;
74 return true;
75}

93RA8_INTERNAL static void* internal_take(viewer_layout_t* layout, size_t bytes, size_t alignment)
94{
95 size_t start = 0U;
96 if (!layout->valid || !internal_align_up(layout->used, alignment, &start) ||
97 (start > SIZE_MAX - bytes)) {
98 layout->valid = false;
99 return nullptr;
100 }
101 const size_t end = start + bytes;
102 if ((layout->base != nullptr) && (end > layout->capacity)) {
103 layout->valid = false;
104 return nullptr;
105 }
106 void* result = (layout->base == nullptr) ? nullptr : &layout->base[start];
107 layout->used = end;
108 return result;
109}

191RA8_INTERNAL static bool internal_ends_with(const char* path, const char* suffix)
192{
193 const size_t path_len = strlen(path);
194 const size_t suffix_len = strlen(suffix);
195 if (path_len < suffix_len) {
196 return false;
197 }
198 const char* tail = &path[path_len - suffix_len];
199 for (size_t index = 0U; index < suffix_len; ++index) {
200 char value = tail[index];
201 if ((value >= 'A') && (value <= 'Z')) {
202 value = (char)(value + ('a' - 'A'));
203 }
204 if (value != suffix[index]) {
205 return false;
206 }
207 }
208 return true;
209}

224RA8_INTERNAL static viewer_fmt_t internal_classify(const char* path)
225{
226 if (internal_ends_with(path, ".jof")) {
227 return k_viewer_fmt_jof;
228 }
229 if (internal_ends_with(path, ".cbz") || internal_ends_with(path, ".cbr") ||
230 internal_ends_with(path, ".cbt")) {
231 return k_viewer_fmt_comic;
232 }
233 if (internal_ends_with(path, ".gz") || internal_ends_with(path, ".xz")) {
234 return k_viewer_fmt_comic_wrap;
235 }
236 if (internal_ends_with(path, ".epub") || internal_ends_with(path, ".epb") ||
237 internal_ends_with(path, ".rabook") || internal_ends_with(path, ".rbk")) {
238 return k_viewer_fmt_reflow;
239 }
240 return k_viewer_fmt_unknown;
241}

254RA8_INTERNAL static void internal_stderr_write(const char* text)
255{
256 size_t offset = 0U;
257 const size_t length = strlen(text);
258 while (offset < length) {
259 const ssize_t written = write(STDERR_FILENO, &text[offset], length - offset);
260 if ((written < 0) && (errno == EINTR)) {
261 continue;
262 }
263 if (written <= 0) {
264 return;
265 }
266 offset += (size_t)written;
267 }

316RA8_INTERNAL static ra8_err_t internal_file_open(viewer_file_ctx_t* file, const char* path)
317{
318 const int descriptor = open(path, O_RDONLY | O_CLOEXEC);
319 if (descriptor < 0) {
320 return k_ra8_err_not_found;
321 }
322 struct stat metadata = {};
323 if ((fstat(descriptor, &metadata) != 0) || !S_ISREG(metadata.st_mode) ||
324 (metadata.st_size <= 0)) {
325 (void)close(descriptor);
326 return k_ra8_err_not_found;
327 }
328 *file =
329 (viewer_file_ctx_t){.size = (uint64_t)metadata.st_size, .fd = descriptor, .is_open = true};
330 return k_ra8_ok;
331}

333ra8_err_t
334priv_viewer_pread(void* ctx, uint64_t offset, uint8_t* buffer, size_t length, size_t* out_read)
335{
336 viewer_file_ctx_t* file = (viewer_file_ctx_t*)ctx;
337 if (out_read == nullptr) {
338 return k_ra8_err_null_ptr;
339 }
340 *out_read = 0U;
341 if ((file == nullptr) || (buffer == nullptr)) {
342 return k_ra8_err_null_ptr;
343 }
344 if (!file->is_open || (offset > (uint64_t)INT64_MAX)) {
345 return k_ra8_err_invalid_state;
346 }
347 if (offset >= file->size) {
348 return k_ra8_ok;
349 }
350 const uint64_t available = file->size - offset;
351 size_t requested = ((uint64_t)length > available) ? (size_t)available : length;
352 size_t completed = 0U;
353 while (completed < requested) {
354 size_t pass = requested - completed;
355 if (pass > (size_t)SSIZE_MAX) {
356 pass = (size_t)SSIZE_MAX;
357 }
358 const ssize_t got =
359 pread(file->fd, &((uint8_t*)buffer)[completed], pass, (off_t)(offset + (uint64_t)completed));
360 if ((got < 0) && (errno == EINTR)) {
361 continue;
362 }
363 if (got <= 0) {
364 if (got == 0) {
365 *out_read = completed;
366 return k_ra8_ok;
367 }
368 return k_ra8_err_not_found;
369 }
370 completed += (size_t)got;
371 }
372 *out_read = completed;
373 return k_ra8_ok;
374}

487RA8_INTERNAL static ra8_err_t internal_require_jof(viewer_file_ctx_t* file,
488 ra8_viewer_reader_requirements_t* out)
489{
490 jof_info_t info = {};
491 ra8_err_t error = jof_parse(priv_viewer_pread, file, file->size, &info);
492 const uint64_t band = (uint64_t)info.tile_w * (uint64_t)info.tile_h * (uint64_t)info.bpp;
493 if ((error == k_ra8_ok) && (band == 0U)) {
494 error = k_ra8_err_invalid_size;
495 }
496 const ra8_decomp_limits_t limits = ra8_decomp_limits_default();
497 if (error == k_ra8_ok) {
498 error = ra8_decomp_check_declared(&limits, file->size, band);
499 }
500 if (error != k_ra8_ok) {
501 return error;
502 }
503 const uint32_t tiles = ((uint32_t)info.height + (uint32_t)k_ra8_viewer_fb_height - 1U) /
504 (uint32_t)k_ra8_viewer_fb_height;
505 out->dimensions_bytes = (size_t)tiles * 2U * sizeof(uint32_t);
506 out->cell_bytes = (size_t)band;
507 out->scratch_bytes =
508 (info.codec == (uint8_t)k_jof_codec_raw) ? 1U : (size_t)jof_stored_bound((uint32_t)band);
509 out->tile_count = tiles;
510 out->engine = (uint32_t)k_ra8_viewer_engine_jof;
511 return k_ra8_ok;
512}

537ra8_err_t ra8_viewer_reader_requirements(const char* path, ra8_viewer_reader_requirements_t* out)
538{
539 if ((path == nullptr) || (out == nullptr)) {
540 return k_ra8_err_null_ptr;
541 }
542 *out = (ra8_viewer_reader_requirements_t){};
543 const viewer_fmt_t format = internal_classify(path);
544 if ((format != k_viewer_fmt_jof) && (format != k_viewer_fmt_comic)) {
545 return internal_reject(format, path);
546 }
547 viewer_file_ctx_t file = {.fd = -1};
548 ra8_err_t error = internal_file_open(&file, path);
549 if (error == k_ra8_ok) {
550 out->required_alignment = alignof(max_align_t);
551 out->framebuffer_bytes =
552 (size_t)k_ra8_viewer_fb_width * (size_t)k_ra8_viewer_fb_height * sizeof(uint16_t);
553 out->layout_version = (uint32_t)k_viewer_layout_version;
554 if (format == k_viewer_fmt_jof) {
555 error = internal_require_jof(&file, out);
556 } else {
557 internal_require_comic(out);
558 }
559 }
560 if (file.is_open) {
561 (void)close(file.fd);
562 }
563 if (error != k_ra8_ok) {
564 *out = (ra8_viewer_reader_requirements_t){};
565 return error;
566 }
567 const viewer_layout_t layout = internal_layout(out, nullptr, 0U);
568 if (!layout.valid) {
569 *out = (ra8_viewer_reader_requirements_t){};
570 return k_ra8_err_invalid_size;
571 }
572 out->required_bytes = layout.used;
573 return k_ra8_ok;
574}

701ra8_err_t ra8_viewer_open(ra8_viewer_reader_t* reader, const char* path)
702{
703 if ((reader == nullptr) || (path == nullptr)) {
704 return k_ra8_err_null_ptr;
705 }
706 if (!reader->is_bound || reader->is_open || reader->file.is_open) {
707 return k_ra8_err_invalid_state;
708 }
709 const viewer_fmt_t format = internal_classify(path);
710 if (!internal_engine_matches(reader, format)) {
711 return k_ra8_err_invalid_state;
712 }
713 ra8_err_t error = internal_file_open(&reader->file, path);
714 if (error == k_ra8_ok) {
715 error = internal_open_selected(reader);
716 }
717 if (error == k_ra8_ok) {
718 reader->is_open = true;
719 return k_ra8_ok;
720 }
721 if (reader->engine == (uint32_t)k_ra8_viewer_engine_comic) {
722 (void)comic_close(&reader->comic.archive);
723 }
724 if (reader->file.is_open) {
725 (void)close(reader->file.fd);
726 }
727 reader->file = (viewer_file_ctx_t){.fd = -1};
728 reader->tile_n = 0U;
729 return error;
730}
