47size_t viewer_read(void* ctx, uint64_t offset, void* buf, size_t len)
48{
49 viewer_file_ctx_t* fc = (viewer_file_ctx_t*)ctx;
50 if ((fc == nullptr) || (fc->fp == nullptr) || (buf == nullptr)) {
51 return 0U;
52 }
53 if (offset >= fc->size) {
54 return 0U;
55 }
56 if (fseeko(fc->fp, (off_t)offset, SEEK_SET) != 0) {
57 return 0U;
58 }
59 return fread(buf, 1U, len, fc->fp);
60}

62ra8_err_t viewer_reserve_page_buf(ra8_viewer_reader_t* r, size_t need)
63{
64 /* Fail closed before touching the allocator: `need` is the encoded-image
65 * length declared in the archive header, so an attacker-chosen value must be
66 * rejected against the per-unit output cap rather than handed to realloc. */
67 if ((uint64_t)need > r->limits.max_output_bytes) {
68 return k_ra8_err_decomp_output_cap;
69 }
70 if (r->page_cap >= need) {
71 return k_ra8_ok;
72 }
73 uint8_t* grown = (uint8_t*)realloc(r->page_buf, need);
74 if (grown == nullptr) {
75 return k_ra8_err_no_mem;
76 }
77 r->page_buf = grown;
78 r->page_cap = need;
79 return k_ra8_ok;
80}

127RA8_INTERNAL static bool viewer_ends_with(const char* path, const char* ext)
128{
129 const size_t pl = strlen(path);
130 const size_t el = strlen(ext);
131 if (pl < el) {
132 return false;
133 }
134 const char* tail = path + (pl - el);
135 for (size_t i = 0U; i < el; ++i) {
136 char c = tail[i];
137 if ((c >= 'A') && (c <= 'Z')) {
138 c = (char)(c + ('a' - 'A'));
139 }
140 if (c != ext[i]) {
141 return false;
142 }
143 }
144 return true;
145}

155RA8_INTERNAL static viewer_fmt_t viewer_classify(const char* path)
156{
157 if (viewer_ends_with(path, ".gz") || viewer_ends_with(path, ".xz")) {
158 return k_vfmt_comic_wrap;
159 }
160 if (viewer_ends_with(path, ".cbz") || viewer_ends_with(path, ".cbr") ||
161 viewer_ends_with(path, ".cbt")) {
162 return k_vfmt_comic;
163 }
164 if (viewer_ends_with(path, ".epub") || viewer_ends_with(path, ".epb")) {
165 return k_vfmt_epub;
166 }
167 if (viewer_ends_with(path, ".jof")) {
168 return k_vfmt_jof;
169 }
170 if (viewer_ends_with(path, ".rabook") || viewer_ends_with(path, ".rbk")) {
171 return k_vfmt_rabook;
172 }
173 return k_vfmt_unsupported;
174}

183RA8_INTERNAL static ra8_err_t viewer_alloc_core(ra8_viewer_reader_t** out)
184{
185 ra8_viewer_reader_t* r = (ra8_viewer_reader_t*)calloc(1U, sizeof(*r));
186 if (r == nullptr) {
187 return k_ra8_err_no_mem;
188 }
189 const size_t fb_pixels = (size_t)k_ra8_viewer_fb_width * (size_t)k_ra8_viewer_fb_height;
190 r->fb = (uint16_t*)calloc(fb_pixels, sizeof(uint16_t));
191 if (r->fb == nullptr) {
192 viewer_free(r);
193 return k_ra8_err_no_mem;
194 }
195 /* Default JOF blit target is the fixed framebuffer (headless path); the tile
196 * path retargets it per render. */
197 r->rt565 = r->fb;
198 r->rt_w = (uint32_t)k_ra8_viewer_fb_width;
199 r->rt_h = (uint32_t)k_ra8_viewer_fb_height;
200 /* One owner-approved decompression policy governs every allocation the viewer
201 * sizes from untrusted archive fields (page buffers, the JOF atlas slurp, the
202 * JOF band cache). It is the same 64 MiB output cap / 1024:1 ratio bound the
203 * firmware reader core enforces, so the viewer rejects exactly what the board
204 * would. */
205 r->limits = ra8_decomp_limits_default();
206 *out = r;
207 return k_ra8_ok;
208}

264RA8_INTERNAL static ra8_err_t viewer_reject_fmt(viewer_fmt_t fmt, const char* path)
265{
266 if ((fmt == k_vfmt_epub) || (fmt == k_vfmt_rabook)) {
267 /* Recognised, but the epub (ra8_epub) and rabook (ra8_book) reflow render
268 * paths -- font face + image loader over ra8_reflow -- are not wired into the
269 * viewer yet. A clear message beats a silent error code. */
270 (void)fprintf(stderr,
271 "ra8_viewer: '%s' recognised but its reflow reader engine "
272 "(font + image loader) is not wired into the viewer yet "
273 "(comics and JOF long strips render today)\n",
274 path);
275 return k_ra8_err_not_supported;
276 }
277 (void)fprintf(stderr, "ra8_viewer: unsupported file type: %s\n", path);
278 return k_ra8_err_not_supported;
279}

312RA8_INTERNAL static ra8_err_t viewer_compute_tiles(ra8_viewer_reader_t* r)
313{
314 const uint32_t n = ra8_viewer_page_count(r);
315 r->tile_n = n;
316 if (n == 0U) {
317 return k_ra8_ok;
318 }
319 r->tile_wpx = (uint32_t*)calloc(n, sizeof(uint32_t));
320 r->tile_hpx = (uint32_t*)calloc(n, sizeof(uint32_t));
321 if ((r->tile_wpx == nullptr) || (r->tile_hpx == nullptr)) {
322 return k_ra8_err_no_mem;
323 }
324 if (r->fmt == k_vfmt_jof) {
325 viewer_size_jof_tiles(r, n);
326 return k_ra8_ok;
327 }
328 ra8_img_arena_t arena = {.base = r->arena_mem, .cap = (size_t)k_viewer_arena_bytes};
329 for (uint32_t i = 0U; i < n; ++i) {
330 viewer_probe_comic_tile(r, i, &arena);
331 }
332 return k_ra8_ok;
333}

335ra8_err_t ra8_viewer_open(ra8_viewer_reader_t** out, const char* path)
336{
337 if ((out == nullptr) || (path == nullptr)) {
338 return k_ra8_err_null_ptr;
339 }
340 *out = nullptr;
341
342 const viewer_fmt_t fmt = viewer_classify(path);
343 if ((fmt == k_vfmt_epub) || (fmt == k_vfmt_rabook) || (fmt == k_vfmt_unsupported)) {
344 return viewer_reject_fmt(fmt, path);
345 }
346
347 ra8_viewer_reader_t* r = nullptr;
348 ra8_err_t rc = viewer_alloc_core(&r);
349 if (rc != k_ra8_ok) {
350 return rc;
351 }
352 r->fmt = fmt;
353 if ((fmt == k_vfmt_comic) || (fmt == k_vfmt_comic_wrap)) {
354 rc = viewer_alloc_comic(r);
355 if (rc != k_ra8_ok) {
356 viewer_free(r);
357 return rc;
358 }
359 }
360 rc = viewer_open_file(r, path);
361 if (rc == k_ra8_ok) {
362 rc = viewer_dispatch_open(r, fmt);
363 }
364 if (rc == k_ra8_ok) {
365 rc = viewer_compute_tiles(r);
366 }
367 if (rc != k_ra8_ok) {
368 viewer_free(r);
369 return rc;
370 }
371 *out = r;
372 return k_ra8_ok;
373}

375uint32_t ra8_viewer_page_count(const ra8_viewer_reader_t* r)
376{
377 if (r == nullptr) {
378 return 0U;
379 }
380 if (r->fmt == k_vfmt_jof) {
381 /* Paginate the tall strip into framebuffer-height windows (>= 1 page). */
382 const uint32_t h = r->jof.dctx.info.height;
383 const uint32_t vh = r->jof.viewport_h;
384 if ((vh == 0U) || (h == 0U)) {
385 return 1U;
386 }
387 return (h + vh - 1U) / vh;
388 }
389 return ra8_comic_page_count(&r->comic);
390}
