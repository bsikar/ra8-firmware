46size_t viewer_read(void* ctx, uint64_t offset, void* buf, size_t len)
47{
48 viewer_file_ctx_t* fc = (viewer_file_ctx_t*)ctx;
49 if ((fc == nullptr) || (fc->fp == nullptr) || (buf == nullptr)) {
50 return 0U;
51 }

53 return 0U;

61ra8_err_t viewer_reserve_page_buf(ra8_viewer_reader_t* r, size_t need)
62{
63 /* Fail closed before touching the allocator: `need` is the encoded-image
64 * length declared in the archive header, so an attacker-chosen value must be
65 * rejected against the per-unit output cap rather than handed to realloc. */
66 if ((uint64_t)need > r->limits.max_output_bytes) {
67 return k_ra8_err_decomp_output_cap;
68 }
69 if (r->page_cap >= need) {
70 return k_ra8_ok;
71 }
72 uint8_t* grown = (uint8_t*)realloc(r->page_buf, need);
73 if (grown == nullptr) {
74 return k_ra8_err_no_mem;
75 }
76 r->page_buf = grown;
77 r->page_cap = need;
78 return k_ra8_ok;
79}

144RA8_INTERNAL static bool viewer_ends_with(const char* path, const char* ext)
145{
146 const size_t pl = strlen(path);
147 const size_t el = strlen(ext);
148 if (pl < el) {
149 return false;
150 }
151 const char* tail = path + (pl - el);
152 for (size_t i = 0U; i < el; ++i) {
153 char c = tail[i];
154 if ((c >= 'A') && (c <= 'Z')) {
155 c = (char)(c + ('a' - 'A'));
156 }
157 if (c != ext[i]) {
158 return false;
159 }
160 }
161 return true;
162}

178RA8_INTERNAL static viewer_fmt_t viewer_classify(const char* path)
179{
180 if (viewer_ends_with(path, ".gz") || viewer_ends_with(path, ".xz")) {
181 return k_vfmt_comic_wrap;
182 }
183 if (viewer_ends_with(path, ".cbz") || viewer_ends_with(path, ".cbr") ||
184 viewer_ends_with(path, ".cbt")) {
185 return k_vfmt_comic;
186 }

188 return k_vfmt_epub;
189 }

193 if (viewer_ends_with(path, ".rabook") || viewer_ends_with(path, ".rbk")) {
194 return k_vfmt_rabook;
195 }
196 return k_vfmt_unsupported;
197}

216RA8_INTERNAL static ra8_err_t viewer_alloc_core(ra8_viewer_reader_t** out)
217{
218 ra8_viewer_reader_t* r = (ra8_viewer_reader_t*)calloc(1U, sizeof(*r));
219 if (r == nullptr) {
220 return k_ra8_err_no_mem;
221 }
222 const size_t fb_pixels = (size_t)k_ra8_viewer_fb_width * (size_t)k_ra8_viewer_fb_height;
223 r->fb = (uint16_t*)calloc(fb_pixels, sizeof(uint16_t));
224 if (r->fb == nullptr) {
225 viewer_free(r);
226 return k_ra8_err_no_mem;
227 }
228 /* Default JOF blit target is the fixed framebuffer (headless path); the tile
229 * path retargets it per render. */
230 r->rt565 = r->fb;
231 r->rt_w = (uint32_t)k_ra8_viewer_fb_width;
232 r->rt_h = (uint32_t)k_ra8_viewer_fb_height;
233 /* One owner-approved decompression policy governs every allocation the viewer
234 * sizes from untrusted archive fields (page buffers, the JOF atlas slurp, the
235 * JOF band cache). It is the same 64 MiB output cap / 1024:1 ratio bound the
236 * firmware reader core enforces, so the viewer rejects exactly what the board
237 * would. */
238 r->limits = ra8_decomp_limits_default();
239 *out = r;
240 return k_ra8_ok;
241}

326RA8_INTERNAL static ra8_err_t viewer_reject_fmt(viewer_fmt_t fmt, const char* path)
327{
328 if ((fmt == k_vfmt_epub) || (fmt == k_vfmt_rabook)) {
329 /* Recognised, but the epub (ra8_epub) and rabook (ra8_book) reflow render
330 * paths -- font face + image loader over ra8_reflow -- are not wired into the
331 * viewer yet. A clear message beats a silent error code. */
332 (void)fprintf(stderr,
333 "ra8_viewer: '%s' recognised but its reflow reader engine "
334 "(font + image loader) is not wired into the viewer yet "
335 "(comics and JOF long strips render today)\n",
336 path);
337 return k_ra8_err_not_supported;
338 }
339 (void)fprintf(stderr, "ra8_viewer: unsupported file type: %s\n", path);
340 return k_ra8_err_not_supported;
341}

388RA8_INTERNAL static ra8_err_t viewer_compute_tiles(ra8_viewer_reader_t* r)
389{
390 const uint32_t n = ra8_viewer_page_count(r);
391 r->tile_n = n;
392 if (n == 0U) {
393 return k_ra8_ok;
394 }
395 r->tile_wpx = (uint32_t*)calloc(n, sizeof(uint32_t));
396 r->tile_hpx = (uint32_t*)calloc(n, sizeof(uint32_t));
397 if ((r->tile_wpx == nullptr) || (r->tile_hpx == nullptr)) {
398 return k_ra8_err_no_mem;
399 }
400 if (r->fmt == k_vfmt_jof) {
401 viewer_size_jof_tiles(r, n);
402 return k_ra8_ok;
403 }
404 ra8_img_arena_t arena = {.base = r->arena_mem, .cap = (size_t)k_viewer_arena_bytes};
405 for (uint32_t i = 0U; i < n; ++i) {
406 viewer_probe_comic_tile(r, i, &arena);
407 }
408 return k_ra8_ok;
409}

411ra8_err_t ra8_viewer_open(ra8_viewer_reader_t** out, const char* path)
412{
413 if ((out == nullptr) || (path == nullptr)) {
414 return k_ra8_err_null_ptr;
415 }
416 *out = nullptr;
417
418 const viewer_fmt_t fmt = viewer_classify(path);
419 if ((fmt == k_vfmt_epub) || (fmt == k_vfmt_rabook) || (fmt == k_vfmt_unsupported)) {
420 return viewer_reject_fmt(fmt, path);
421 }
422
423 ra8_viewer_reader_t* r = nullptr;
424 ra8_err_t rc = viewer_alloc_core(&r);
425 if (rc != k_ra8_ok) {
426 return rc;
427 }
428 r->fmt = fmt;
429 if ((fmt == k_vfmt_comic) || (fmt == k_vfmt_comic_wrap)) {
430 rc = viewer_alloc_comic(r);
431 if (rc != k_ra8_ok) {
432 viewer_free(r);
433 return rc;
434 }
435 }
436 rc = viewer_open_file(r, path);
437 if (rc == k_ra8_ok) {
438 rc = viewer_dispatch_open(r, fmt);
439 }
440 if (rc == k_ra8_ok) {
441 rc = viewer_compute_tiles(r);
442 }
443 if (rc != k_ra8_ok) {
444 viewer_free(r);
445 return rc;
446 }
447 *out = r;
448 return k_ra8_ok;
449}

451uint32_t ra8_viewer_page_count(const ra8_viewer_reader_t* r)
452{
453 if (r == nullptr) {
454 return 0U;
455 }
456 if (r->fmt == k_vfmt_jof) {
457 /* Paginate the tall strip into framebuffer-height windows (>= 1 page). */
458 const uint32_t h = r->jof.dctx.info.height;
459 const uint32_t vh = r->jof.viewport_h;
460 if ((vh == 0U) || (h == 0U)) {
461 return 1U;
462 }
463 return (h + vh - 1U) / vh;
464 }
465 return ra8_comic_page_count(&r->comic);
466}
