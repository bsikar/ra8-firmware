374static void fs_fmt_fill_payload(void)
375{
376 uint32_t state = (uint32_t)k_fs_fmt_prng_seed;
377 for (uint32_t i = 0U; i < (uint32_t)k_fs_fmt_payload_bytes; i++) {
378 state = (state * (uint32_t)k_fs_fmt_prng_mul) + (uint32_t)k_fs_fmt_prng_add;
379 s_payload[i] = (uint8_t)((state >> k_fs_fmt_prng_byte_shift) & k_fs_fmt_byte_mask);
380 }
381}

398[[nodiscard]] static ra8_err_t fs_fmt_create_and_verify(ra8_fs_mount_t* mount, const char* path)
399{
400 ra8_err_t err = ra8_fs_write_file(mount, path, s_payload, (uint32_t)k_fs_fmt_payload_bytes);
401 if (err != k_ra8_ok) {
402 return err;
403 }
404 ra8_fs_file_t* f = nullptr;
405 err = ra8_fs_open(mount, path, k_ra8_fs_mode_read, &f);
406 if (err != k_ra8_ok) {
407 return err;
408 }
409 memset(s_readback, 0, sizeof(s_readback));
410 uint32_t got = 0U;
411 err = ra8_fs_read(f, s_readback, (uint32_t)k_fs_fmt_payload_bytes, &got);
412 (void)ra8_fs_close(f);
413 if (err != k_ra8_ok) {
414 return err;
415 }
416 if (got != (uint32_t)k_fs_fmt_payload_bytes) {
417 return k_ra8_err_validation_failed;
418 }
419 if (memcmp(s_payload, s_readback, (size_t)k_fs_fmt_payload_bytes) != 0) {
420 return k_ra8_err_validation_failed;
421 }
422 return k_ra8_ok;

526[[nodiscard]] static ra8_err_t fs_fmt_run_one_type(ra8_fs_type_t type)
527{
528 ra8_fs_backend_t backend = {};
529 if (ra8_sdmmc_spi_bind_fs_backend(&backend) != k_ra8_ok) {
530 fs_fmt_print_line(type, k_msg_fail_fmt, (uint32_t)sizeof(k_msg_fail_fmt) - 1U);
531 return k_ra8_err_hw_error;
532 }
533 ra8_fs_format_opts_t opts = {};
534 opts.type = type;
535 opts.label = "RAFSFMT";
536 ra8_err_t err = ra8_fs_format(&backend, &opts);
537 if (err != k_ra8_ok) {
538 /* invalid_size == the card cannot hold this FAT type (too large/small): the
539 * caller treats that as a SKIP, not a FAIL, so do not print the FAIL line. */
540 if (err != k_ra8_err_invalid_size) {
541 fs_fmt_print_line(type, k_msg_fail_fmt, (uint32_t)sizeof(k_msg_fail_fmt) - 1U);
542 }
543 return err;
544 }
545 ra8_fs_mount_t* mount = nullptr;
546 err = ra8_fs_mount(&backend, &mount);
547 if (err != k_ra8_ok) {
548 fs_fmt_print_line(type, k_msg_fail_mount, (uint32_t)sizeof(k_msg_fail_mount) - 1U);
549 return err;
550 }
551 if (mount->type != type) {
552 fs_fmt_print_line(type, k_msg_fail_type, (uint32_t)sizeof(k_msg_fail_type) - 1U);
553 (void)ra8_fs_unmount(mount);
554 return k_ra8_err_validation_failed;
555 }
556 err = fs_fmt_write_cycle(mount, type);
557 if (err != k_ra8_ok) {
558 (void)ra8_fs_unmount(mount);
559 return err;
560 }
561 (void)ra8_fs_unmount(mount);
562 fs_fmt_print_line(type, k_msg_pass_suf, (uint32_t)sizeof(k_msg_pass_suf) - 1U);
563 return k_ra8_ok;
564}

706int32_t main(void)
707{
708 uint32_t pclka_hz = 0U;
709 fs_fmt_setup_or_halt(&pclka_hz);
710 ra8_isr_globals_enable();
711 ra8_log_init();
712 FS_FMT_PUTS(k_msg_boot);
713
714 fs_fmt_init_card_or_halt(&pclka_hz);
715 fs_fmt_fill_payload();
716
717 static const ra8_fs_type_t k_types[k_fs_fmt_idx_count] = {
718 k_ra8_fs_type_fat12,
719 k_ra8_fs_type_fat16,
720 k_ra8_fs_type_fat32,
721 };
722 uint32_t passed = 0U;
723 for (uint32_t i = 0U; i < (uint32_t)k_fs_fmt_idx_count; i++) {
724 const ra8_err_t r = fs_fmt_run_one_type(k_types[i]);
725 if (r == k_ra8_ok) {
726 passed++;
727 } else if (r == k_ra8_err_invalid_size) {
728 /* This FAT type does not fit this card's capacity -- skip it, do not halt.
729 * A big card (e.g. 128 GB) takes only FAT32; a tiny one only FAT12/16. */
730 fs_fmt_print_line(k_types[i], k_msg_skip_size, (uint32_t)sizeof(k_msg_skip_size) - 1U);
731 } else {
732 fs_fmt_panic_halt(); /* a real failure (mount / write / verify). */
733 }
734 }
735 /* exFAT trial (full cycle: format + mount + create/write/rename/unlink); same
736 * SKIP semantics as the FAT types above. */
737 const ra8_err_t ex = fs_fmt_run_exfat();
738 if (ex == k_ra8_ok) {
739 passed++;
740 } else if (ex == k_ra8_err_invalid_size) {
741 fs_fmt_print_line(k_ra8_fs_type_exfat, k_msg_skip_size, (uint32_t)sizeof(k_msg_skip_size) - 1U);
742 } else {
743 fs_fmt_panic_halt();
744 }
745 if (passed == 0U) {
746 fs_fmt_panic_halt(); /* no FAT type fit the card -- unexpected. */
747 }
748 fs_fmt_print(k_msg_all_pass, (uint32_t)sizeof(k_msg_all_pass) - 1U);
749 fs_fmt_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
750
751 while (1) {
752 __asm__ volatile("wfi");
753 }
754}
