168static ra8_err_t bd_read_head(ra8_io_blockdev_vsource_ctx_t* ctx,
169 ra8_io_bd_vsource_cursor_t* cur,
170 uint64_t block_bytes)
171{
172 const uint32_t head_in_block = (uint32_t)(cur->cur % block_bytes);
173 if (head_in_block == 0U) {
174 return k_ra8_ok;
175 }
176 uint32_t head_n = (uint32_t)(block_bytes - (uint64_t)head_in_block);
177 if (head_n > cur->remain) {
178 head_n = cur->remain;
179 }
180 const ra8_err_t err =
181 bd_copy_slice(ctx, (uint32_t)(cur->cur / block_bytes), head_in_block, head_n, cur->out);
182 if (err != k_ra8_ok) {
183 return err;
184 }
185 cur->cur += (uint64_t)head_n;
186 cur->out += head_n;
187 cur->remain -= head_n;
188 return k_ra8_ok;
189}

283ra8_err_t ra8_io_blockdev_vsource_read(void* ctx, uint64_t offset, uint8_t* buf, uint32_t len)
284{
285 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
286 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
287 ra8_io_blockdev_vsource_ctx_t* c = (ra8_io_blockdev_vsource_ctx_t*)ctx;
288
289 const uint64_t block_bytes = (uint64_t)k_ra8_io_block_size_bytes;
290 if (offset > (UINT64_MAX - (uint64_t)len)) {
291 return k_ra8_err_out_of_range;
292 }
293
294 ra8_io_bd_vsource_cursor_t cur = {.cur = offset, .out = buf, .remain = len};
295
296 /* Unaligned head: the byte offset does not start on a sector boundary. */
297 ra8_err_t err = bd_read_head(c, &cur, block_bytes);
298 if (err != k_ra8_ok) {
299 return err;
300 }
301
302 /* Aligned middle: whole sectors copied straight into the caller buffer. */
303 err = bd_read_middle(c, &cur, block_bytes);
304 if (err != k_ra8_ok) {
305 return err;
306 }
307
308 /* Unaligned tail: a partial final sector. */
309 return bd_read_tail(c, &cur, block_bytes);
310}
