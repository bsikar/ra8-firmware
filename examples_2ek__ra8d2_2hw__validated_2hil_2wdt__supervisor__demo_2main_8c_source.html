128static void worker_a_entry(ULONG arg)
129{
130 (void)arg;
131 /* Supervisor bring-up runs from worker A's first action -- the
132 * supervisor's register / start APIs need a real thread context
133 * (tx_mutex_get returns NOT_SYSTEM from tx_application_define). */

187static void wdt_sup_demo_bring_up(void)
188{
189 /* Bring up the underlying WWDT first -- supervisor's refresh hook
190 * calls ra8_wdt_refresh_deferred. NMI on expiry so a supervisor
191 * bug surfaces as a missing-tick rather than a reset loop. The
192 * 25..75 % window restriction is intentionally disabled for the
193 * supervisor demo (window_start=100, window_end=0) -- the
194 * supervisor's refresh cadence is on the 50 ms supervisor tick,
195 * not synchronized to the WWDT counter, so an artificial window
196 * would just look like a flaky test. */
197 const ra8_wdt_cfg_t wdt_cfg = {
198 .timeout = k_ra8_wdt_timeout_1024,
199 .clock_div = k_ra8_wdt_clkdiv_4,
200 .window_start = k_ra8_wdt_window_start_100,
201 .window_end = k_ra8_wdt_window_end_0,
202 .on_expiry = k_ra8_wdt_on_expiry_nmi,
203 .stop_in_sleep = k_ra8_wdt_sleep_stop_count,
204 };
205 ra8_err_t err = ra8_wdt_init(&wdt_cfg);
206 g_wdt_supervisor_demo_last_err = (uint32_t)err;
207 if (err != k_ra8_ok) {
208 wdt_sup_demo_halt();
209 }
210 g_wdt_supervisor_demo_step = k_wdt_sup_step_wdt_init;
211 const ra8_wdt_sup_cfg_t sup_cfg = {
212 .stack = s_sup_stack,
213 .stack_size_bytes = (uint32_t)k_wdt_sup_demo_stack_bytes,
214 .priority = (uint32_t)k_wdt_sup_demo_priority + 1U,
215 .refresh_period_ms = (uint32_t)k_wdt_sup_demo_sup_period_ms,
216 };
217 err = ra8_wdt_supervisor_init(&sup_cfg);
218 g_wdt_supervisor_demo_last_err = (uint32_t)err;
219 if (err != k_ra8_ok) {
220 wdt_sup_demo_halt();
221 }
222 g_wdt_supervisor_demo_step = k_wdt_sup_step_supervisor_init;
223 err = ra8_wdt_supervisor_register_thread("worker_a",
224 (uint32_t)k_wdt_sup_demo_worker_a_deadline_ms,
225 &s_handle_a);
226 g_wdt_supervisor_demo_last_err = (uint32_t)err;
227 if (err != k_ra8_ok) {
228 wdt_sup_demo_halt();
229 }
230 g_wdt_supervisor_demo_step = k_wdt_sup_step_worker_a_registered;
231 err = ra8_wdt_supervisor_register_thread("worker_b",
232 (uint32_t)k_wdt_sup_demo_worker_b_deadline_ms,
233 &s_handle_b);
234 g_wdt_supervisor_demo_last_err = (uint32_t)err;
235 if (err != k_ra8_ok) {
236 wdt_sup_demo_halt();
237 }
238 g_wdt_supervisor_demo_step = k_wdt_sup_step_worker_b_registered;
239 err = ra8_wdt_supervisor_start();
240 g_wdt_supervisor_demo_last_err = (uint32_t)err;
241 if (err != k_ra8_ok) {
242 wdt_sup_demo_halt();
243 }
244 g_wdt_supervisor_demo_step = k_wdt_sup_step_supervisor_started;
245}
