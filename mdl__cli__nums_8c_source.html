53RA8_INTERNAL static bool internal_cli_parse_ul(const char* s, unsigned long* out)
54{
55 if (s == nullptr) {
56 return false;
57 }
58 if (s[0] < '0') {
59 return false;
60 }
61 if (s[0] > '9') {
62 return false;
63 }
64 errno = 0;
65 char* end = nullptr;
66 const unsigned long v = strtoul(s, &end, k_cli_dec_base);
67 if (errno != 0) {
68 return false;
69 }
70 if (*end != '\0') {
71 return false;
72 }
73 *out = v;
74 return true;

133RA8_INTERNAL static bool internal_cli_parse_chapter(const char* s, double* out)
134{
135 if ((s == nullptr) || (out == nullptr) || (s[0] == '\0')) {
136 return false;
137 }
138 errno = 0;
139 char* end = nullptr;
140 const double v = strtod(s, &end);
141 if ((errno != 0) || (end == s) || (*end != '\0') || !isfinite(v)) {
142 return false;
143 }
144 *out = v;
145 return true;

391RA8_INTERNAL static ra8_err_t internal_cli_validate_num_ranges(const mdl_args_t* a,
392 ra8_io_stream_t* diagnostic,
393 const mdl_cli_raw_nums_t* raw)
394{
395 if ((raw->timeout == 0UL) || (raw->timeout > (unsigned long)UINT32_MAX)) {
396 return internal_cli_reject_limit(diagnostic, "mdl: --timeout must be in 1..", UINT32_MAX, "\n");
397 }
398 if ((raw->chapters == 0UL) || ((uintmax_t)raw->chapters > (uintmax_t)SIZE_MAX)) {
399 return internal_cli_reject_limit(diagnostic,
400 "mdl: --chapters must be in 1..",
401 (uint64_t)SIZE_MAX,
402 "\n");
403 }
404 if (raw->max_imgs > (unsigned long)UINT32_MAX) {
405 return internal_cli_reject_limit(diagnostic, "mdl: --max must not exceed ", UINT32_MAX, "\n");
406 }
407 if ((a->max_bytes != nullptr) && (raw->max_bytes == 0U)) {
408 const char* const parts[] = {"mdl: --max-bytes must be greater than zero\n"};
409 return priv_mdl_cli_reject_parts(diagnostic, parts, 1U);
410 }
411 if ((a->pick != nullptr) &&
412 ((raw->pick == 0UL) || ((uintmax_t)raw->pick > (uintmax_t)SIZE_MAX))) {
413 return internal_cli_reject_limit(diagnostic,
414 "mdl: --pick must be in 1..",
415 (uint64_t)SIZE_MAX,
416 "\n");
417 }
418 return k_ra8_ok;
419}

421ra8_err_t mdl_cli_parse_nums(const mdl_args_t* a, ra8_io_stream_t* diagnostic, mdl_nums_t* n)
422{
423 if ((a == nullptr) || (diagnostic == nullptr) || (n == nullptr)) {
424 return k_ra8_err_null_ptr;
425 }
426 mdl_nums_t candidate = {};
427 mdl_cli_raw_nums_t raw = {};
428 ra8_err_t err = internal_cli_parse_num_values(a, diagnostic, &candidate, &raw);
429 if (err == k_ra8_ok) {
430 err = internal_cli_parse_chapter_pick(a, diagnostic, &candidate, &raw);
431 }
432 if (err == k_ra8_ok) {
433 err = internal_cli_validate_num_ranges(a, diagnostic, &raw);
434 }
435 if (err == k_ra8_ok) {
436 candidate.timeout = (uint32_t)raw.timeout;
437 candidate.chapters = (size_t)raw.chapters;
438 candidate.max_imgs = (uint32_t)raw.max_imgs;
439 candidate.pick = (size_t)raw.pick;
440 *n = candidate;
441 }
442 return err;
