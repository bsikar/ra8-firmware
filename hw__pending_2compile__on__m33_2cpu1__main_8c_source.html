213static bool compile_fixture(const void** out_blob, uint32_t* out_len)
214{
215 if (out_blob == nullptr) {
216 return false;
217 }
218 if (out_len == nullptr) {
219 return false;
220 }
221 /* The M85 staged the source .epub into shared SRAM and posted its base/len in
222 * the request; the M33 compiles whatever was staged (not a baked-in fixture),
223 * so the same image serves any book the M85 dispatches. */
224 const volatile com33_mailbox_t* mb = com33_mailbox();
225 memset(&s_book, 0, sizeof(s_book));
226 const ra8_epub_mem_media_t media = {
227 .data = (const uint8_t*)(uintptr_t)mb->epub_base,
228 .size = (size_t)mb->epub_len,
229 };
230 if (ra8_epub_open(&media, "rabook.epub", &s_book) != k_ra8_ok) {
231 return false;
232 }
233
234 ra8_rabook_buffers_t buf = {};
235 ra8_rabook_pipeline_scratch_t scr = {};
236 if (!bind_compile(&buf, &scr)) {
237 (void)ra8_epub_close(&s_book);
238 return false;
239 }
240
241 const ra8_err_t rc =
242 ra8_rabook_compile_from_epub_to_buffer(&s_book, &buf, &scr, out_blob, out_len);
243 (void)ra8_epub_close(&s_book);
244 return rc == k_ra8_ok;
245}

271static void publish_result(volatile com33_mailbox_t* mb, const void* blob, uint32_t len, bool ok)
272{
273 if (mb == nullptr) {
274 return;
275 }
276 bool good = ok;
277 if (blob == nullptr) {
278 good = false;
279 }
280 if (len == 0U) {
281 good = false;
282 }
283 if (good) {
284 const ra8_book_header_t* hdr = ra8_book_header(blob);
285 mb->blob_base = (uint32_t)(uintptr_t)blob;
286 mb->blob_len = len;
287 mb->blob_crc = hdr->crc32;
288 mb->chapter_count = hdr->chapter_count;
289 __asm volatile("dsb" ::: "memory");
290 mb->status = (uint32_t)k_com33_status_ok;
291 } else {
292 mb->status = (uint32_t)k_com33_status_build_fail;
293 }
294 __asm volatile("dsb" ::: "memory");
295 mb->done = 1U;
296}

374[[noreturn]] static void cpu1_run_compile(void)
375{
376 volatile com33_mailbox_t* mb = com33_mailbox();
377 mb->m33_sig = (uint32_t)k_com33_m33_sig;
378 __asm volatile("dsb" ::: "memory");
379
380 /* The M85 stages the .epub + buffers and posts req_magic BEFORE releasing this
381 * core, so the job is ready on the first read; the dmb orders that read after
382 * observing the boot handshake. A missing request is reported, not compiled. */
383 __asm volatile("dmb" ::: "memory");
384 if (mb->req_magic != (uint32_t)k_com33_req_magic) {
385 mb->status = (uint32_t)k_com33_status_no_request;
386 __asm volatile("dsb" ::: "memory");
387 mb->done = 1U;
388 __asm volatile("dsb" ::: "memory");
389 notify_m85();
390 cpu1_park();
391 }
392
393 const void* blob = nullptr;
394 uint32_t len = 0U;
395 const bool ok = compile_fixture(&blob, &len);
396
397 publish_result(mb, blob, len, ok);
398 notify_m85();
399 cpu1_park();
400}

420[[noreturn]] void cpu1_reset_handler(void)
421{
422 uint32_t* dst = &g_ra8_ls_cpu1_data_start;
423 uint32_t* src = &g_ra8_ls_cpu1_data_load;
424 RA8_LOOP_BOUND_RUNTIME(g_ra8_ls_cpu1_data_end);
425 while (dst < &g_ra8_ls_cpu1_data_end) {
426 *dst = *src;
427 dst++;
428 src++;
429 }
430
431 uint32_t* bss = &g_ra8_ls_cpu1_bss_start;
432 RA8_LOOP_BOUND_RUNTIME(g_ra8_ls_cpu1_bss_end);
433 while (bss < &g_ra8_ls_cpu1_bss_end) {
434 *bss = 0U;
435 bss++;
436 }
437
438 cpu1_run_compile();
439}
