4"""Gate: every RA8_NSC_VENEER declared in ra8_nsc.h shall have a definition.
5
6An ``RA8_NSC_VENEER`` (a ``cmse_nonsecure_entry`` Secure-Gateway veneer) is the
7ONLY path a Non-Secure caller has into the Secure world. A declaration in the
8public header ``libs/ra8_nsc/inc/ra8_nsc.h`` with no matching definition under
9``libs/ra8_nsc/src/`` advertises an NS->S entry point that does not exist: it
10misleads Non-Secure integrators about the available trust-boundary surface, and
11because a veneer with no caller also has no link error, such a phantom decl can
12sit in the header indefinitely (this is how ``ra8_nsc_key_import`` and
13``ra8_nsc_trng_read`` accumulated -- see security-trustzone-audit F16/A4).
14
15This gate parses every ``RA8_NSC_VENEER`` function declaration out of the header
16and fails if any lacks a definition in the ``ra8_nsc`` sources. There is no
17allowlist: either implement the veneer or delete the declaration.
18
19Run::
20
21 check_nsc_veneer_defs.py
22
23Exit status: 0 if every declared veneer has a definition, 1 otherwise.
24"""
25
26import re
27import sys
28from pathlib import Path
29
30# Repo root: this file is scripts/checks/check_nsc_veneer_defs.py .
31REPO_ROOT = Path(__file__).resolve().parents[2]
32HEADER = REPO_ROOT / "libs" / "ra8_nsc" / "inc" / "ra8_nsc.h"
33SRC_DIR = REPO_ROOT / "libs" / "ra8_nsc" / "src"
34
35# A declaration: [[nodiscard]] RA8_NSC_VENEER ra8_err_t ra8_nsc_foo(...
36# We only need the function name that follows the RA8_NSC_VENEER return type.
37DECL_RE = re.compile(r"RA8_NSC_VENEER\s+\w[\w\s\*]*?\b(ra8_nsc_\w+)\s*\‍(")
38
39
40def declared_veneers(header: Path) -> list[str]:
41 """Return the veneer function names declared in the header, in order."""
42 text = header.read_text(encoding="utf-8")
43 seen: list[str] = []
44 for name in DECL_RE.findall(text):
45 if name not in seen:
46 seen.append(name)
47 return seen
48
49
50def is_defined(name: str, sources: list[Path]) -> bool:
51 """Whether a definition of ``name`` exists in any ra8_nsc source file.
52
53 A definition in a .c is the function signature followed (eventually) by a
54 body; a bare declaration lives only in the header, so any ``name(`` at
55 statement start in a source file is a definition.
56 """
57 # Match the definition head by its `RA8_NSC_VENEER` prefix -- the same anchor
58 # DECL_RE uses on the header -- so the return type is irrelevant: the
59 # `ra8_err_t` veneers and the `void` ra8_nsc_wdt_refresh are found alike. The
60 # `[\w\s\*]` operand class cannot cross a `(` / `;` / `{`, so the span never
61 # reaches from one veneer's prefix to another's name, and a call site (which
62 # carries no RA8_NSC_VENEER prefix) never matches.
63 def_re = re.compile(r"RA8_NSC_VENEER\s+\w[\w\s\*]*?\b" + re.escape(name) + r"\s*\‍(")
64 return any(def_re.search(src.read_text(encoding="utf-8")) for src in sources)
65
66
67def main() -> int:
68 """Fail when a declared RA8_NSC_VENEER has no definition in the NSC sources.
69
70 The asymmetry is the point: an undefined veneer is a TrustZone entry point
71 the public header advertises to the non-secure world and the secure world
72 cannot service. That is a trust hazard, not a link error to be discovered
73 later, so it is checked at the declaration rather than left to the linker.
74
75 A missing header exits 1 rather than 0 -- with nothing to parse there are
76 zero declarations, and "zero declared, zero missing" would pass while
77 verifying nothing.
78
79 Returns 0 when every declaration has a definition, 1 on a phantom veneer
80 or a missing header.
81 """
82 if not HEADER.is_file():
83 print(f"check_nsc_veneer_defs.py: header not found: {HEADER}", file=sys.stderr)
84 return 1
85 sources = sorted(SRC_DIR.glob("*.c"))
86 veneers = declared_veneers(HEADER)
87 missing = [name for name in veneers if not is_defined(name, sources)]
88
89 hdr = HEADER.relative_to(REPO_ROOT)
90 src = SRC_DIR.relative_to(REPO_ROOT)
91 if missing:
92 print("check_nsc_veneer_defs.py: RA8_NSC_VENEER declared without a definition:")
93 for name in missing:
94 print(f" {name}: declared in {hdr}, no definition in {src}/")
95 print("Fix each at the root -- implement the veneer, or delete the declaration.")
96 print("A phantom NS->S entry point in the public header is a trust hazard.")
97 return 1
98
99 count = len(veneers)
100 print(f"check_nsc_veneer_defs.py: PASS -- all {count} RA8_NSC_VENEER declaration(s) defined.")
101 return 0
102
103
104if __name__ == "__main__":
105 sys.exit(main())
