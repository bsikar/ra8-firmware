58long mdl_urlname_chapter_number(const char* url)
59{
60 if (url == nullptr) {
61 return 0L;
62 }
63 const size_t len = strlen(url);
64 size_t s = 0U;
65 size_t e = 0U;
66 bool found = false;
67 for (size_t i = 0U; i < len; ++i) {
68 if ((url[i] >= '0') && (url[i] <= '9')) {
69 size_t j = i;
70 while ((j < len) && (url[j] >= '0') && (url[j] <= '9')) {
71 ++j;
72 }
73 s = i;
74 e = j;
75 found = true;
76 i = j; /* skip the run just captured */
77 }
78 }

110void mdl_urlname_ext(const char* url, char* out, size_t cap)
111{
112 if ((out == nullptr) || (cap == 0U)) {
113 return;
114 }
115 (void)snprintf(out, cap, "%s", "jpg");
116 if (url == nullptr) {
117 return;
118 }
119 const size_t pend = path_end(url);
120 size_t start = pend;
121 while ((start > 0U) && (url[start - 1U] != '/')) {
122 --start; /* isolate the last path segment */
123 }
124 size_t dot = pend;
125 for (size_t i = start; i < pend; ++i) {
126 if (url[i] == '.') {
127 dot = i;
128 }
129 }
130 if (dot >= pend) {
131 return; /* no '.' in the segment -> keep the default */
132 }
133 char ext[8];
134 size_t n = 0U;
135 for (size_t i = dot + 1U; (i < pend) && (n + 1U < sizeof(ext)); ++i) {
136 ext[n++] = to_lower_ascii(url[i]);
137 }
138 ext[n] = '\0';
139 if (is_known_ext(ext)) {
