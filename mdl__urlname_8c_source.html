120RA8_INTERNAL static size_t internal_urlname_path_start(const char* url, size_t end)
121{
122 const char* scheme = strstr(url, "://");
123 if ((scheme == nullptr) || ((size_t)(scheme - url) >= end)) {
124 return 0U; /* Relative URL: its entire prefix is a path. */
125 }
126 const size_t authority = (size_t)(scheme - url) + strlen("://");
127 const char* slash = memchr(url + authority, '/', end - authority);
128 return (slash == nullptr) ? end : (size_t)(slash - url);

170RA8_INTERNAL static bool internal_urlname_parse_chapter_digits(const char* start,
171 const char* end,
172 double* out,
173 const char** out_next)
174{
175 if ((start >= end) || (*start < '0') || (*start > '9')) {
176 return false;
177 }
178 uint32_t whole = 0U;
179 while ((start < end) && (*start >= '0') && (*start <= '9')) {
180 const uint32_t digit = (uint32_t)(*start - '0');
181 if (whole > (((uint32_t)k_chapter_whole_max - digit) / (uint32_t)k_chapter_radix)) {
182 return false;
183 }
184 whole = (whole * (uint32_t)k_chapter_radix) + digit;
185 ++start;
186 }
187 double value = (double)whole;
188 if ((start < end) && ((*start == '-') || (*start == '.')) && ((start + 1) < end) &&
189 (start[1] >= '0') && (start[1] <= '9')) {
190 ++start;
191 double scale = s_chapter_fraction_step;
192 uint32_t digits = 0U;
193 while ((start < end) && (*start >= '0') && (*start <= '9')) {
194 if (digits >= (uint32_t)k_chapter_frac_max) {
195 return false;
196 }
197 value += (double)(*start - '0') * scale;
198 scale *= s_chapter_fraction_step;
199 ++digits;
200 ++start;
201 }
202 }
203 *out = value;
204 if (out_next != nullptr) {
205 *out_next = start;
206 }
207 return true;
208}

210bool mdl_urlname_chapter_parse(const char* url, double* out)
211{
212 if (out != nullptr) {
213 *out = 0.0;
214 }
215 if ((url == nullptr) || (out == nullptr)) {
216 return false;
217 }
218 const size_t bounded = strnlen(url, (size_t)k_urlname_scan_max);
219 if (bounded == (size_t)k_urlname_scan_max) {
220 return false;
221 }
222 const size_t end = internal_urlname_path_end(url);
223 const size_t begin = internal_urlname_path_start(url, end);
224 static const char* const markers[] = {"chapter-", "/ch-", "/ep"};
225 const char* found = nullptr;
226 size_t skip = 0U;
227 for (size_t i = 0U; i < (sizeof(markers) / sizeof(markers[0])); ++i) {
228 const char* candidate = internal_urlname_last_path_marker(url, begin, end, markers[i]);
229 if ((candidate != nullptr) && ((found == nullptr) || (candidate > found))) {
230 found = candidate;
231 skip = strlen(markers[i]);
232 }
233 }
234 return (found != nullptr) &&
235 internal_urlname_parse_chapter_digits(found + skip, url + end, out, nullptr);
236}

401RA8_INTERNAL static bool
402internal_sniff_magic_image(const void* buf, size_t buf_len, const char** ext, const char** mime)
403{
404 if ((buf == nullptr) || (buf_len < 2U)) {
405 return false;
406 }
407 const uint8_t* b = (const uint8_t*)buf;
408 if ((buf_len >= 3U) && (b[0] == (uint8_t)k_jpeg_marker_byte) &&
409 (b[1] == (uint8_t)k_jpeg_soi_byte) && (b[2] == (uint8_t)k_jpeg_marker_byte)) {
410 *ext = "jpg";
411 *mime = "image/jpeg";
412 return true;
413 }
414 if ((b[0] == (uint8_t)'B') && (b[1] == (uint8_t)'M')) {
415 *ext = "bmp";
416 *mime = "image/bmp";
417 return true;
418 }
419 if ((buf_len >= 4U) && (b[0] == (uint8_t)k_png_lead_byte) && (b[1] == (uint8_t)'P') &&
420 (b[2] == (uint8_t)'N') && (b[3] == (uint8_t)'G')) {
421 *ext = "png";
422 *mime = "image/png";
423 return true;
424 }
425 if ((buf_len >= (size_t)k_webp_sig_bytes) && (b[0] == 'R') && (b[1] == 'I') && (b[2] == 'F') &&
426 (b[3] == 'F') && (b[8] == 'W') && (b[k_webp_e_offset] == 'E') &&
427 (b[k_webp_b_offset] == 'B') && (b[k_webp_p_offset] == 'P')) {
428 *ext = "webp";
429 *mime = "image/webp";
430 return true;
431 }
432 if ((buf_len >= 6U) && (b[0] == 'G') && (b[1] == 'I') && (b[2] == 'F') && (b[3] == '8') &&
433 ((b[4] == '7') || (b[4] == '9')) && (b[k_gif_a_offset] == 'a')) {
434 *ext = "gif";
435 *mime = "image/gif";
436 return true;
437 }
438 return false;
439}
