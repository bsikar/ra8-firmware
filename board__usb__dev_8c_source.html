23uint16_t usb_intsts0(void)
24{
25 uint16_t v = s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_intsts0)];
26 v &= (uint16_t)((1U << k_ra8_int0_bit_brdy) | (1U << k_ra8_int0_bit_nrdy) |
27 (1U << k_ra8_int0_bit_bemp) | (1U << k_ra8_int0_bit_ctrt) |
28 (1U << k_ra8_int0_bit_dvst) | (1U << k_ra8_int0_bit_vbse));
29 v = (uint16_t)(v | (s_usb.ctsq & (uint16_t)k_ra8_intsts0_mask_ctsq));
30 v = (uint16_t)(v | (s_usb.dvsq & (uint16_t)k_ra8_intsts0_mask_dvsq));
31 v = (uint16_t)(v | (uint16_t)k_ra8_intsts0_mask_vbsts); /* VBUS always present. */
32 if (s_usb.setup_valid) {
33 v = (uint16_t)(v | (uint16_t)k_ra8_intsts0_mask_valid);
34 }
35 return v;
36}

151uint32_t usb_reg_read(uint64_t off, unsigned size)
152{
153 switch ((uint16_t)off) {
154 case (uint16_t)k_ra8_usb_off_intsts0:
155 return usb_intsts0();
156 case (uint16_t)k_ra8_usb_off_cfifoctr:
157 return cfifoctr_read();
158 case (uint16_t)k_ra8_usb_off_cfifo:
159 case (uint16_t)((uint16_t)k_ra8_usb_off_cfifo + (uint16_t)k_usb_cfifo_h): /* CFIFOH tail. */
160 case (uint16_t)((uint16_t)k_ra8_usb_off_cfifo + (uint16_t)k_usb_cfifo_hh): /* CFIFOHH tail. */
161 /* Full access width: the HS instance drains its CFIFO 32 bits at a time
162 * (MBW=32) with 16/8-bit residual reads through the CFIFOH / CFIFOHH
163 * aliases (HUM Ch 37.2.7 "CFIFO, DnFIFO : FIFO Port Register" p 2069-2070); the FS
164 * instance uses 16-bit accesses at the base port. */
165 return cfifo_read_port(size);
166 case (uint16_t)k_ra8_usb_off_syssts0:
167 return (uint16_t)0x0003U; /* LNST = J-state: device pull-up seen. */
168 case (uint16_t)k_ra8_usb_off_dvstctr0: {
169 /* HUM Ch 36.2.5 "DVSTCTR0 : Device State Control Register 0", p 1971:
170 * RHST[2:0] reports the settled link speed once the host's bus reset
171 * completes; the device firmware (internal_dvst_track_speed in the DCD)
172 * re-aims the USBX framework at it. The modelled loop is a full-speed
173 * link (the FS controller's ceiling caps it in either polarity), so
174 * report FS from the moment the device has left Powered. */
175 uint16_t v = s_usb.reg[usb_word(off)];
176 if (s_usb.dvsq != (uint16_t)k_ra8_dvsq_powered) {
177 v = (uint16_t)(v | (uint16_t)k_usb_rhst_fs);
178 }
179 return v;
180 }
181 case (uint16_t)k_ra8_usb_off_frmnum:
182 return s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_frmnum)];
183 default:
184 return s_usb.reg[usb_word(off)];
185 }
186}

189void usb_reg_write(uint64_t off, uint32_t value, unsigned size)
190{
191 const uint16_t v16 = (uint16_t)value;
192 switch ((uint16_t)off) {
193 case (uint16_t)k_ra8_usb_off_cfifo:
194 case (uint16_t)((uint16_t)k_ra8_usb_off_cfifo + (uint16_t)k_usb_cfifo_h): /* CFIFOH tail. */
195 case (uint16_t)((uint16_t)k_ra8_usb_off_cfifo + (uint16_t)k_usb_cfifo_hh): /* CFIFOHH tail. */
196 /* Full access width: the HS instance fills its CFIFO 32 bits at a time
197 * (MBW=32) with 16/8-bit residual writes through the CFIFOH / CFIFOHH
198 * aliases (HUM Ch 37.2.7 "CFIFO, DnFIFO : FIFO Port Register" p 2069-2070); truncating to
199 * 16 bits here zeroed bytes 2-3 of every descriptor word in Config B. */
200 cfifo_write_port(value, size);
201 return;
202 case (uint16_t)k_ra8_usb_off_cfifoctr:
203 cfifoctr_write(v16);
204 return;
205 case (uint16_t)k_ra8_usb_off_intsts0:
206 /* W0C on the event bits: a written 0 clears, a 1 preserves. */
207 s_usb.reg[usb_word(off)] &= v16;
208 if ((v16 & (uint16_t)k_ra8_intsts0_mask_valid) == 0U) {
209 s_usb.setup_valid = false;
210 }
211 return;
212 default:
213 s_usb.reg[usb_word(off)] = v16;
214 return;
215 }
216}

233void board_usb_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value, bool* handled)
234{
235 if ((addr < (uint64_t)k_usb_base) || (addr >= ((uint64_t)k_usb_base + (uint64_t)k_usb_span))) {
236 *handled = false;
237 return;
238 }
239 *handled = true;
240 const uint64_t off = addr - (uint64_t)k_usb_base;
241 /* Role declaration (HUM Ch 36.2.1 SYSCFG.DCFM, p 1966): the firmware writes
242 * DCFM=1 to select controller (host) function on THIS instance. Under the
243 * self-loop bridge that pins the FS window to the host model -- Config B. */
244 if (s_external_host && !s_roles_swapped && (off == (uint64_t)k_ra8_usb_off_syscfg) &&
245 (((uint16_t)value & (uint16_t)(1U << k_ra8_syscfg_bit_dcfm)) != 0U)) {
246 board_usb_roles_swap(uc);
247 }
248 if (s_roles_swapped) {
249 board_usbhs_host_reg_write(uc, off, size, value);
250 return;
251 }
252 usb_reg_write(off, (uint32_t)value, size);
