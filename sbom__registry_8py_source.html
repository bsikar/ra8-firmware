3"""The curated third-party component registry behind the ra8-firmware SBOM.
4
5This module holds the *data*: what each vendored SOUP component under
6``libs/third_party/`` (plus the bundled font asset and the one not-vendored
7co-processor firmware) is, where it came from, and under what license.
8``gen_sbom.py`` holds the *logic* that renders and cross-checks it.
9
10They are separate modules so the registry can grow with the tree without
11dragging the generator over the project file-size cap, and so a component
12edit reads as a data change in review rather than a change to the gate.
13
14Editing rules:
15
16 * Versions are cross-checked against the tree where a header macro exists;
17 do NOT edit a version here without the matching vendored source changing.
18 * Per-component qualification lives under ``docs/SOUP/``; the aggregated
19 human license inventory is ``THIRD_PARTY_LICENSES.md``. A registry edit
20 that does not update those is half a change.
21 * A newly vendored directory with no entry here fails the SBOM gate.
22 * There is NO integrity-hash field, deliberately. ``aggregate_sha256`` used
23 to live here as a hand-transcribed literal on four of the twenty-three
24 entries, which meant ``gen_sbom.py --check`` compared a constant against
25 itself and a mutated vendored byte reported clean (#538). The digest is
26 now DERIVED from the tree on every run by ``gen_sbom.tree_digest()``. Do
27 not re-introduce a stored copy: a transcribed value never disagrees with
28 itself.
29 * The ``upstream_*`` / ``patched_files`` / ``local_files`` fields are the
30 machine-readable form of what ``docs/SOUP/*.md`` states in prose: which
31 upstream ref the tree claims to be, and every file that deliberately
32 departs from it. ``scripts/checks/check_soup_upstream.py`` compares the
33 tree against upstream blob hashes fetched from the upstream project, so a
34 subset rule or a patch that lives only in prose is not enough -- an
35 undeclared deviation fails the gate (#548).
36"""

42# Provenance classes, most-to-least trustworthy. Recorded per component in a
43# CycloneDX property so an auditor can see, at a glance, which components are
44# pinned versus inferred.
45PROV_COMMIT_PINNED = "commit-pinned-sha256" # upstream commit + per-file upstream blob hashes
46PROV_ARCHIVE_PINNED = "archive-pinned-sha256" # upstream release artifact pinned by SHA-256
47PROV_VERSION_HEADER = "version-header" # version from an in-tree header macro only
48PROV_NOT_VENDORED = "not-vendored" # documented but absent from the tree
49PROV_PROPRIETARY = "proprietary-unresolved" # license not cleared; see notes
50PROV_OPEN_ASSET = "open-asset-versioned" # cleared open asset (OFL font); version from name table
51
52# How the pinned upstream revision is fetched by check_soup_upstream.py.
53# GIT is a ref (tag or commit) in the upstream repository; ARCHIVE is a release
54# artifact that never existed in the upstream git tree -- miniz publishes its
55# single-file amalgamation only as a release zip, so a git ref cannot express
56# what we actually vendored.
57UPSTREAM_GIT = "git"
58UPSTREAM_ARCHIVE = "archive"
59
60# The five Eclipse ThreadX trees share one deliberate deviation, so its
61# justification is written once. Repeating a justification string is how five
62# copies of it drift into four different claims.
63GITATTRIBUTES_PATCH = (
64 "Attribute-macro blocks ([attr]our-c-style, [attr]generated) removed: git "
65 "honours [attr] definitions only in the top-level .gitattributes and printed "
66 "a 'not allowed' warning on every git operation across five vendored trees. "
67 "The macro USES left behind reference undefined attributes, which git ignores "
68 "silently, so no vendored file's checkout behaviour changes."
69)

72@dataclass(frozen=True)
73class Component:
74 """One SOUP or bundled-asset record.
75
76 The required fields describe the component; the optional fields carry the
77 provenance and license detail that the SBOM and the cross-checks consume.
78
79 The ``upstream_*`` group answers "what is this tree supposed to BE?", which
80 the version and integrity fields cannot: a digest re-derived from our own
81 tree proves only that we have not changed it since we last looked
82 (#538/#548). ``upstream_ref`` names the revision the vendored subset is
83 claimed to come from; ``patched_files`` and ``local_files`` enumerate every
84 file that deliberately departs from it, so "modified" and "corrupted" are
85 distinguishable by a machine rather than by reading prose.
86 """
