92static void slru_unlink(slru_t* l, int32_t f, int32_t* head, int32_t* tail)
93{
94 if (l->prev[f] != -1) {
95 l->next[l->prev[f]] = l->next[f];
96 } else if (*head == f) {
97 *head = l->next[f];
98 }
99 if (l->next[f] != -1) {
100 l->prev[l->next[f]] = l->prev[f];
101 } else if (*tail == f) {
102 *tail = l->prev[f];
103 }
104}

126static void slru_push_head(slru_t* l, int32_t f, int32_t* head, int32_t* tail)
127{
128 l->prev[f] = -1;
129 l->next[f] = *head;
130 if (*head != -1) {
131 l->prev[*head] = f;
132 }
133 *head = f;
134 if (*tail == -1) {
135 *tail = f;
136 }

163static int slru_init(cb_cache_t* c)
164{
165 slru_t* l = (slru_t*)calloc(1U, sizeof(slru_t));
166 /* cppcheck-suppress memleak ; false positive: cppcheck 2.13 does not model
167 * the C23 nullptr keyword, so it cannot see l is NULL on this path. */
168 if (l == nullptr) {
169 return 1;
170 }
171 l->prev = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
172 l->next = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
173 l->pb_head = -1;
174 l->pb_tail = -1;
175 l->pt_head = -1;
176 l->pt_tail = -1;
177 l->pt_cap = (c->capacity * (uint32_t)k_slru_protected_pct) / (uint32_t)k_slru_pct_full_scale;
178 if ((l->prev == nullptr) || (l->next == nullptr)) {
179 /* The replay harness never deinits a policy whose init failed, so a
180 * partial allocation must be released here, not left on policy_data. */
181 free(l->prev);
182 free(l->next);
183 free(l);
184 return 1;
185 }
186 c->policy_data = l;
187 return 0;
188}

261static void slru_access(cb_cache_t* c, uint32_t frame)
262{
263 slru_t* l = (slru_t*)c->policy_data;
264 const int32_t f = (int32_t)frame;
265 if (c->frames[frame].meta[0] == (uint8_t)k_slru_protected) {
266 slru_unlink(l, f, &l->pt_head, &l->pt_tail);
267 slru_push_head(l, f, &l->pt_head, &l->pt_tail);
268 return;
269 }
270 /* Promote probationary -> protected; demote protected LRU if over cap. */
271 slru_unlink(l, f, &l->pb_head, &l->pb_tail);
272 if ((l->pt_cap > 0U) && (l->pt_count == l->pt_cap)) {
273 const int32_t d = l->pt_tail;
274 slru_unlink(l, d, &l->pt_head, &l->pt_tail);
275 c->frames[d].meta[0] = (uint8_t)k_slru_probation;
276 slru_push_head(l, d, &l->pb_head, &l->pb_tail);
277 l->pt_count--;
278 }
279 c->frames[frame].meta[0] = (uint8_t)k_slru_protected;
280 slru_push_head(l, f, &l->pt_head, &l->pt_tail);
281 l->pt_count++;
282}

306static uint32_t slru_victim(cb_cache_t* c, uint32_t* scanned)
307{
308 slru_t* l = (slru_t*)c->policy_data;
309 *scanned = 1U;
310 if (l->pb_tail != -1) {
311 const int32_t f = l->pb_tail;
312 slru_unlink(l, f, &l->pb_head, &l->pb_tail);
313 return (uint32_t)f;
314 }
315 const int32_t f = l->pt_tail;
316 slru_unlink(l, f, &l->pt_head, &l->pt_tail);
317 l->pt_count--;
318 return (uint32_t)f;

454static uint32_t srrip_victim(cb_cache_t* c, uint32_t* scanned)
455{
456 uint32_t* hand = (uint32_t*)c->policy_data;
457 uint32_t seen = 0U;
458 for (;;) {
459 const uint32_t f = *hand;
460 *hand = (*hand + 1U) % c->capacity;
461 seen++;
462 if (c->frames[f].meta[0] == (uint8_t)k_rrip_max) {
463 *scanned = seen;
464 return f;
465 }
466 /* Age toward the max so a victim is guaranteed within two ring passes. */
467 if (c->frames[f].meta[0] < (uint8_t)k_rrip_max) {
468 c->frames[f].meta[0]++;
469 }
470 }
471}
