72static void slru_unlink(slru_t* l, int32_t f, int32_t* head, int32_t* tail)
73{
74 if (l->prev[f] != -1) {
75 l->next[l->prev[f]] = l->next[f];
76 } else if (*head == f) {
77 *head = l->next[f];
78 }
79 if (l->next[f] != -1) {
80 l->prev[l->next[f]] = l->prev[f];
81 } else if (*tail == f) {
82 *tail = l->prev[f];
83 }
84}

87static void slru_push_head(slru_t* l, int32_t f, int32_t* head, int32_t* tail)
88{
89 l->prev[f] = -1;
90 l->next[f] = *head;
91 if (*head != -1) {
92 l->prev[*head] = f;
93 }
94 *head = f;
95 if (*tail == -1) {
96 *tail = f;
97 }
98}

100static int slru_init(cb_cache_t* c)
101{
102 slru_t* l = (slru_t*)calloc(1U, sizeof(slru_t));
103 /* cppcheck-suppress memleak ; false positive: cppcheck 2.13 does not model
104 * the C23 nullptr keyword, so it cannot see l is NULL on this path. */
105 if (l == nullptr) {
106 return 1;
107 }
108 l->prev = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
109 l->next = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
110 l->pb_head = -1;
111 l->pb_tail = -1;
112 l->pt_head = -1;
113 l->pt_tail = -1;
114 l->pt_cap = (c->capacity * (uint32_t)k_slru_protected_pct) / (uint32_t)k_slru_pct_full_scale;
115 if ((l->prev == nullptr) || (l->next == nullptr)) {
116 /* The replay harness never deinits a policy whose init failed, so a
117 * partial allocation must be released here, not left on policy_data. */
118 free(l->prev);
119 free(l->next);
120 free(l);
121 return 1;
122 }
123 c->policy_data = l;
124 return 0;
125}

144static void slru_access(cb_cache_t* c, uint32_t frame)
145{
146 slru_t* l = (slru_t*)c->policy_data;
147 const int32_t f = (int32_t)frame;
148 if (c->frames[frame].meta[0] == (uint8_t)k_slru_protected) {
149 slru_unlink(l, f, &l->pt_head, &l->pt_tail);
150 slru_push_head(l, f, &l->pt_head, &l->pt_tail);
151 return;
152 }
153 /* Promote probationary -> protected; demote protected LRU if over cap. */
154 slru_unlink(l, f, &l->pb_head, &l->pb_tail);
155 if ((l->pt_cap > 0U) && (l->pt_count == l->pt_cap)) {
156 const int32_t d = l->pt_tail;
157 slru_unlink(l, d, &l->pt_head, &l->pt_tail);
158 c->frames[d].meta[0] = (uint8_t)k_slru_probation;
159 slru_push_head(l, d, &l->pb_head, &l->pb_tail);
160 l->pt_count--;
161 }
162 c->frames[frame].meta[0] = (uint8_t)k_slru_protected;
163 slru_push_head(l, f, &l->pt_head, &l->pt_tail);
164 l->pt_count++;
165}

167static uint32_t slru_victim(cb_cache_t* c, uint32_t* scanned)
168{
169 slru_t* l = (slru_t*)c->policy_data;
170 *scanned = 1U;
171 if (l->pb_tail != -1) {
172 const int32_t f = l->pb_tail;
173 slru_unlink(l, f, &l->pb_head, &l->pb_tail);
174 return (uint32_t)f;
175 }
176 const int32_t f = l->pt_tail;
177 slru_unlink(l, f, &l->pt_head, &l->pt_tail);
178 l->pt_count--;
179 return (uint32_t)f;
180}

220static uint32_t srrip_victim(cb_cache_t* c, uint32_t* scanned)
221{
222 uint32_t* hand = (uint32_t*)c->policy_data;
223 uint32_t seen = 0U;
224 for (;;) {
225 const uint32_t f = *hand;
226 *hand = (*hand + 1U) % c->capacity;
227 seen++;
228 if (c->frames[f].meta[0] == (uint8_t)k_rrip_max) {
229 *scanned = seen;
230 return f;
231 }
232 /* Age toward the max so a victim is guaranteed within two ring passes. */
233 if (c->frames[f].meta[0] < (uint8_t)k_rrip_max) {
234 c->frames[f].meta[0]++;
235 }
236 }
237}
