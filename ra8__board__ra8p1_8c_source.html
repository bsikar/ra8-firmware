183ra8_err_t ra8_board_sw_read(ra8_board_sw_id_t sw, ra8_board_sw_state_t* out_pressed)
184{
185 if (out_pressed == nullptr) {
186 return k_ra8_err_invalid_arg;
187 }
188 ra8_port_pin_t pin = k_ra8_pin_none;
189 ra8_err_t err = ra8_board_sw_pin(sw, &pin);
190 if (err != k_ra8_ok) {
191 return err;
192 }
193 ra8_level_t lvl = k_ra8_level_high;
194 err = ra8_gpio_read(pin, &lvl);
195 if (err != k_ra8_ok) {
196 /* ra8_gpio_read returns k_ra8_ok for valid port-0 SW pins off-target. */
197 return err; /* GCOVR_EXCL_LINE */
198 }
199 /* Buttons are active-low: low level == pressed. */
200 *out_pressed = (lvl == k_ra8_level_low) ? k_ra8_board_sw_pressed : k_ra8_board_sw_released;
201 return k_ra8_ok;
202}

204ra8_err_t ra8_board_sw_attach_irq(ra8_board_sw_id_t sw, ra8_board_sw_irq_cb_t cb, void* ctx)
205{
206 if (cb == nullptr) {
207 return k_ra8_err_invalid_arg;
208 }
209 if ((uint8_t)sw >= (uint8_t)k_ra8_board_sw_count) {
210 return k_ra8_err_invalid_arg;
211 }
212 /* Step 1: configure the ICU IRQ pin for falling-edge detection so a button
213 * press (active-low) latches the IRQCR channel. The digital filter samples at
214 * PCLKB to debounce contact bounce (RA8P1 ICU is byte-identical to the RA8D2,
215 * HUM R01UH1064EJ Ch 14 "Interrupt Controller Unit"). */
216 const ra8_icu_irq_cfg_t cfg = {
217 .sense = k_ra8_icu_irqmd_falling,
218 .filter_div = k_ra8_icu_fclksel_pclkb,
219 .filter_en = true,
220 };
221 ra8_err_t err = ra8_icu_configure_irq_pin(s_sw_irq_nums[sw], &cfg);
222 if (err != k_ra8_ok) {
223 /* ra8_icu_configure_irq_pin succeeds for IRQ12/13 (well within 32-channel limit). */
224 return err; /* GCOVR_EXCL_LINE */
225 }
226
227 /* Step 2: route the ELC event for the IRQ channel through an IELSR slot and
228 * enable the matching NVIC line. SW1 -> IRQ13, SW2 -> IRQ12 (provisional). */
229 const ra8_elc_event_t evt =
230 (sw == k_ra8_board_sw1) ? k_ra8_elc_event_icu_irq13 : k_ra8_elc_event_icu_irq12;
231 return ra8_isr_register(evt, (ra8_isr_handler_t)cb, ctx, k_ra8_isr_prio_default, nullptr);
232}

271ra8_err_t ra8_board_uart_console_init(uint32_t baud)
272{
273 if (baud == 0U) {
274 return k_ra8_err_invalid_arg;
275 }
276
277 /* SCI8's baud generator is fed by PCLKA. Read it from CGC at runtime so the
278 * BRR tracks whatever the application's CGC tree settled on instead of an
279 * assumed constant (the classic 2x-too-fast UART bug). */
280 uint32_t pclka_hz = 0U;
281 ra8_err_t err = ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz);
282 if (err != k_ra8_ok) {
283 /* ra8_cgc_get_clock_hz with a valid clock-id and non-null out always returns k_ra8_ok. */
284 return err; /* GCOVR_EXCL_LINE */
285 }
286 if (pclka_hz < (uint32_t)k_ra8_board_uart_console_min_pclka_hz) {
287 return k_ra8_err_not_initialized;
288 }
289
290 /* Route PD02 -> TXD8, PD03 -> RXD8 (PSEL = SCI async). */
291 /* NOLINTBEGIN(clang-analyzer-optin.core.EnumCastOutOfRange) -- RA8_PIN()-packed board pin is valid ra8_port_pin_t data outside the enumerator list. */
292 err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_ra8_board_uart_console_pin_txd,
293 k_ra8_psel_sci_async,
294 "ra8_board.uart.console.txd");
295 if (err != k_ra8_ok) {
296 return err;
297 }
298 err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_ra8_board_uart_console_pin_rxd,
299 k_ra8_psel_sci_async,
300 "ra8_board.uart.console.rxd");
301 /* NOLINTEND(clang-analyzer-optin.core.EnumCastOutOfRange) */
302 if (err != k_ra8_ok) {
303 return err;
304 }
305
306 const ra8_sci_cfg_t cfg = {
307 .baud = baud,
308 .data_bits = k_ra8_sci_data_8,
309 .parity = k_ra8_sci_parity_none,
310 .stop_bits = k_ra8_sci_stop_1,
311 .pclk_hz = pclka_hz,
312 };
313 err = ra8_sci_init((uint8_t)k_ra8_board_uart_console_sci_channel, &cfg);
314 if (err != k_ra8_ok) {
315 /* ra8_sci_init on a valid channel with a non-null cfg always returns k_ra8_ok in RA8_OFF_TARGET. */
316 return err; /* GCOVR_EXCL_LINE */
317 }
318 s_uart_console_initialized = true;
319 return k_ra8_ok;
320}

322ra8_err_t ra8_board_uart_console_write(const uint8_t* data, size_t len)
323{
324 if (len == 0U) {
325 return k_ra8_ok;
326 }
327 if (data == nullptr) {
328 return k_ra8_err_invalid_arg;
329 }
330 if (!s_uart_console_initialized) {
331 return k_ra8_err_not_initialized;
332 }
333 return ra8_sci_write_polling((uint8_t)k_ra8_board_uart_console_sci_channel, data, (uint32_t)len);
334}
335
336ra8_err_t ra8_board_uart_console_read(uint8_t* out, size_t cap, size_t* out_len)
337{
338 if (out_len == nullptr) {
339 return k_ra8_err_invalid_arg;
340 }
341 *out_len = 0U;
342 if (cap == 0U) {
343 return k_ra8_ok;
344 }
345 if (out == nullptr) {
346 return k_ra8_err_invalid_arg;
347 }
348 if (!s_uart_console_initialized) {
349 return k_ra8_err_not_initialized;
350 }
351
352 /* Polled non-blocking drain: pull bytes while data is available, stop the
353 * moment ra8_sci_getc_polling reports nothing. The cap bounds the loop
354 * (NASA Rule 2). */
355 for (size_t i = 0U; i < cap; ++i) {
356 uint8_t byte = 0U;
357 const ra8_err_t err =
358 ra8_sci_getc_polling((uint8_t)k_ra8_board_uart_console_sci_channel, &byte);
359 if (err != k_ra8_ok) {
360 /* No byte available yet -- treat as a non-blocking stop. */
361 return k_ra8_ok;
362 }
363 out[i] = byte;
364 *out_len += 1U;
365 }
366 return k_ra8_ok;
367}
368
369ra8_err_t ra8_board_uart_console_flush(void)
370{
371 if (!s_uart_console_initialized) {
372 return k_ra8_err_not_initialized;
373 }
374 /* Defer to ra8_sci_flush, which spins on CSR.TEND for the SCI8 channel and
375 * carries the register-access HUM citation. Lets a panic handler drain the
376 * failure log before WFI gates the SCI clock. */
377 return ra8_sci_flush((uint8_t)k_ra8_board_uart_console_sci_channel);
378}
