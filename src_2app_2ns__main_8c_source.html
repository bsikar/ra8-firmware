234static void ui_thread_entry(ULONG thread_input)
235{
236 (void)thread_input;
237 (void)ra8_nsc_log_emit("UI", "UI thread started");
238
239 uint32_t frame = 0U;
240 for (;;) {
241 frame++;
242 g_ns_ui_frames = frame; /* HIL liveness probe (read via J-Link). */
243 /* Prove liveness to the watchdog supervisor once per frame. */
244 (void)ra8_wdt_supervisor_checkin(s_ui_wdt_handle);
245 if ((frame % (uint32_t)k_ns_ui_heartbeat_frames) == 0U) {
246 (void)ra8_nsc_log_emit("UI", "UI loop: frame heartbeat");
247 }
248 tx_thread_sleep((uint32_t)k_ns_ui_frame_ticks); /* ~16 ms; yields to the others. */
249 }

305static void ns_wdt_setup(void)
306{
307 if (ra8_nsc_wdt_start() != k_ra8_ok) {
308 ns_panic_halt();
309 }
310 const ra8_wdt_sup_cfg_t sup_cfg = {
311 .stack = s_sup_stack,
312 .stack_size_bytes = (uint32_t)k_ns_wdt_sup_stack_size,
313 .priority = (uint32_t)k_ns_wdt_sup_priority,
314 .refresh_period_ms = (uint32_t)k_ns_wdt_refresh_ms,
315 };
316 if (ra8_wdt_supervisor_init(&sup_cfg) != k_ra8_ok) {
317 ns_panic_halt();
318 }
319 /* Route refresh through the NSC veneer -- the WDT is Secure-owned, so the
320 * library's default direct ra8_wdt_refresh_deferred would fault in NS. */
321 if (ra8_wdt_supervisor_set_refresh_hook(ra8_nsc_wdt_refresh) != k_ra8_ok) {
322 ns_panic_halt();
323 }
324 if (ra8_wdt_supervisor_register_thread("ui",
325 (uint32_t)k_ns_wdt_ui_deadline_ms,
326 &s_ui_wdt_handle) != k_ra8_ok) {
327 ns_panic_halt();
328 }
329 if (ra8_wdt_supervisor_register_thread("sys",
330 (uint32_t)k_ns_wdt_sys_deadline_ms,
331 &s_sys_wdt_handle) != k_ra8_ok) {
332 ns_panic_halt();
333 }
334 if (ra8_wdt_supervisor_start() != k_ra8_ok) {
335 ns_panic_halt();
336 }
337}

342void tx_application_define(void* first_unused_memory)
343{
344 (void)first_unused_memory;
345
346 /* Arm the WDT + start the check-in supervisor before any worker runs. */
347 ns_wdt_setup();
348
349 /* Create the UI thread. A failed create is unrecoverable this early: the
350 * WDT supervisor already registered the "ui"/"sys" deadlines, so a missing
351 * worker would only surface later as a silent watchdog boot-loop -- panic
352 * now instead, like every other NS bring-up failure in this file. */
353 /* (CHAR*)(uintptr_t): the vendored ThreadX API takes a non-const CHAR* for
354 * the thread name; the uintptr_t hop launders the string-literal const
355 * without tripping -Wcast-qual (same pattern as ra8_wdt_supervisor). */
356 if (tx_thread_create(&s_ui_thread,
357 (CHAR*)(uintptr_t)"UI Thread",
358 ui_thread_entry,
359 0UL,
360 s_ui_thread_stack,
361 k_ns_ui_thread_stack_size,
362 k_ns_ui_priority, /* Priority */
363 k_ns_ui_priority, /* Preemption threshold */
364 TX_NO_TIME_SLICE,
365 TX_AUTO_START) != TX_SUCCESS) {
366 ns_panic_halt();
367 }
368
369 /* Create the System/Storage thread */
370 if (tx_thread_create(&s_sys_thread,
371 (CHAR*)(uintptr_t)"System Thread",
372 sys_thread_entry,
373 0UL,
374 s_sys_thread_stack,
375 k_ns_sys_thread_stack_size,
376 k_ns_sys_priority, /* Priority */
377 k_ns_sys_priority, /* Preemption threshold */
378 TX_NO_TIME_SLICE,
379 TX_AUTO_START) != TX_SUCCESS) {
380 ns_panic_halt();
381 }

403[[noreturn]] void ns_reset_handler(void)
404{
405#ifdef RA8_EREADER_NS_XIP
406 /* XIP: code/rodata execute in place from OSPI, but .data must be writable, so
407 * copy its init image from OSPI (LMA) into SRAM (VMA) before any initialised
408 * global is read. Must run first -- nothing below may touch a .data global. */
409 const uintptr_t data_src = (uintptr_t)&g_ra8_ls_ns_data_load_start;
410 const uintptr_t data_dst = (uintptr_t)&g_ra8_ls_ns_data_start;
411 const uintptr_t data_end = (uintptr_t)&g_ra8_ls_ns_data_end;
412 for (uintptr_t off = 0U; (data_dst + off) < data_end; off += sizeof(uint32_t)) {
413 *(volatile uint32_t*)(data_dst + off) = *(const volatile uint32_t*)(data_src + off);
414 }
415#endif
416
417 /* Zero the NS BSS section */
418 const uintptr_t bss_start = (uintptr_t)&g_ra8_ls_ns_bss_start;
419 const uintptr_t bss_end = (uintptr_t)&g_ra8_ls_ns_bss_end;
420 for (uintptr_t addr = bss_start; addr < bss_end; addr += sizeof(uint32_t)) {
421 *(volatile uint32_t*)addr = 0U;
422 }
423
424 /* Set the NS VTOR so exceptions vector correctly to NS handlers */
425 *(volatile uint32_t*)k_ns_scb_vtor_addr = (uint32_t)(uintptr_t)&g_ra8_ls_ns_run_start;
426
427 /* Call Secure-side substrate initialization via NSC veneer gateway */
428 if (ra8_nsc_periph_init() != k_ra8_ok) {
429 ns_panic_halt();
430 }
431
432 /* Announce the Non-Secure world is live before handing off to ThreadX.
433 * Logs flow NS -> ra8_nsc_log_emit veneer -> Secure ra8_log_info -> ITM
434 * (visible on the J-Link SWO console). */
435 (void)ra8_nsc_log_emit("BOOT", "ra8d2-ereader: Non-Secure world online");
436
437 /* Enter ThreadX RTOS kernel (never returns) */
438 tx_kernel_enter();
439
440 ns_panic_halt();
441}
