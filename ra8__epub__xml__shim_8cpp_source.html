434void ncx_walk(const XMLElement* parent, ra8_epub_book_t* book, std::uint8_t base_depth)
435{
436 const XMLElement* np = find_child(parent, "navPoint");
437 std::uint8_t depth = base_depth;
438 while (np != nullptr) {
439 const XMLElement* label = find_child(np, "navLabel");
440 const XMLElement* text = (label != nullptr) ? find_child(label, "text") : nullptr;
441 const XMLElement* content = find_child(np, "content");
442 const char* title = (text != nullptr) ? text->GetText() : nullptr;
443 const char* src = (content != nullptr) ? content->Attribute("src") : nullptr;
444 toc_emit(book, title, src, depth);
445
446 const XMLElement* child = find_child(np, "navPoint");
447 if (child != nullptr) {
448 np = child;
449 depth = static_cast<std::uint8_t>(depth + 1U);
450 continue;
451 }
452 const XMLElement* sibling = next_local_sibling(np, "navPoint");
453 while (sibling == nullptr && depth > base_depth) {
454 const tinyxml2::XMLNode* up = np->Parent();
455 np = (up != nullptr) ? up->ToElement() : nullptr;
456 depth = static_cast<std::uint8_t>(depth - 1U);
457 if (np == nullptr) {
458 break;
459 }
460 sibling = next_local_sibling(np, "navPoint");
461 }
462 np = sibling;
463 }

493void nav_walk(const XMLElement* ordered_list, ra8_epub_book_t* book, std::uint8_t base_depth)
494{
495 const XMLElement* li = find_child(ordered_list, "li");
496 std::uint8_t depth = base_depth;
497 while (li != nullptr) {
498 const XMLElement* anchor = find_child(li, "a");
499 const XMLElement* label = (anchor != nullptr) ? anchor : find_child(li, "span");
500 const char* title = (label != nullptr) ? label->GetText() : nullptr;
501 const char* href = (anchor != nullptr) ? anchor->Attribute("href") : nullptr;
502 toc_emit(book, title, href, depth);
503
504 const XMLElement* nested = find_child(li, "ol");
505 const XMLElement* child = (nested != nullptr) ? find_child(nested, "li") : nullptr;
506 if (child != nullptr) {
507 li = child;
508 depth = static_cast<std::uint8_t>(depth + 1U);
509 continue;
510 }
511 const XMLElement* sibling = next_local_sibling(li, "li");
512 while (sibling == nullptr && depth > base_depth) {
513 const tinyxml2::XMLNode* ol_node = li->Parent();
514 const tinyxml2::XMLNode* up_li = (ol_node != nullptr) ? ol_node->Parent() : nullptr;
515 li = (up_li != nullptr) ? up_li->ToElement() : nullptr;
516 depth = static_cast<std::uint8_t>(depth - 1U);
517 if (li == nullptr) {
518 break;
519 }
520 sibling = next_local_sibling(li, "li");
521 }
522 li = sibling;
523 }

626extern "C" ra8_err_t ra8_epub_xml_parse_container(const uint8_t* xml_bytes,
627 size_t xml_len,
628 ra8_epub_container_result_t* out)
629{
630 if (xml_bytes == nullptr || out == nullptr) {
631 return k_ra8_err_null_ptr;
632 }
633 if (xml_len == 0U) {
634 return k_ra8_err_invalid_size;
635 }
636
637 out->opf_path[0] = '\0';
638
639 XMLDocument doc;
640 if (doc.Parse(reinterpret_cast<const char*>(xml_bytes), xml_len) != tinyxml2::XML_SUCCESS) {
641 return k_ra8_err_validation_failed;
642 }
643
644 const XMLElement* root = doc.RootElement();
645 if (root == nullptr) {
646 return k_ra8_err_validation_failed;
647 }
648
649 const XMLElement* rootfiles = find_descendant(root, "rootfiles");
650 const XMLElement* rootfile =
651 (rootfiles != nullptr) ? find_child(rootfiles, "rootfile") : find_descendant(root, "rootfile");
652 if (rootfile == nullptr) {
653 return k_ra8_err_validation_failed;
654 }
655
656 const char* full_path = rootfile->Attribute("full-path");
657 if (full_path == nullptr || full_path[0] == '\0') {
658 return k_ra8_err_validation_failed;
659 }
660 copy_bounded(out->opf_path, k_ra8_epub_max_path_len, full_path);
661 return k_ra8_ok;
662}

664extern "C" ra8_err_t
665ra8_epub_xml_parse_opf(const uint8_t* xml_bytes, size_t xml_len, ra8_epub_book_t* book)
666{
667 if (xml_bytes == nullptr || book == nullptr) {
668 return k_ra8_err_null_ptr;
669 }
670 if (xml_len == 0U) {
671 return k_ra8_err_invalid_size;
672 }
673
674 XMLDocument doc;
675 if (doc.Parse(reinterpret_cast<const char*>(xml_bytes), xml_len) != tinyxml2::XML_SUCCESS) {
676 return k_ra8_err_validation_failed;
677 }
678 const XMLElement* package = doc.RootElement();
679 if (package == nullptr) {
680 return k_ra8_err_validation_failed;
681 }
682
683 /* ---- metadata block (Dublin Core) ----------------------------------- */
684 const XMLElement* metadata = find_child(package, "metadata");
685 if (metadata != nullptr) {
686 parse_metadata(package, metadata, book);
687 }
688
689 /* ---- manifest ------------------------------------------------------- */
690 const XMLElement* manifest = find_child(package, "manifest");
691 const XMLElement* spine = find_child(package, "spine");
692 if (manifest == nullptr || spine == nullptr) {
693 return k_ra8_err_validation_failed;
694 }
695
696 parse_cover(metadata, manifest, book);
697
698 /* ---- embedded fonts (#109): record manifest font hrefs -------------- */
699 collect_font_items(manifest, book);
700
701 /* ---- manifest items (#151): record id/href/media-type in OPF order --- */
702 collect_manifest_items(manifest, book);
703
704 /* ---- spine (chapters in document order) ----------------------------- */
705 const ra8_err_t spine_err = parse_spine(manifest, spine, book);
706 if (spine_err != k_ra8_ok) {
707 return spine_err;
708 }
709
710 /* ---- table-of-contents source --------------------------------------- */
711 parse_toc_source(manifest, spine, book);
712 return k_ra8_ok;
713}
