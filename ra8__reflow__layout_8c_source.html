305static void priv_justify_glyphs(ra8_reflow_t* engine, uint32_t lo, uint32_t hi, int32_t slack)
306{
307 uint32_t spaces = 0U;
308 for (uint32_t i = lo; i < hi; ++i) {
309 if (engine->glyphs[i].cp == (int32_t)' ') {
310 spaces++;
311 }
312 }
313 if (spaces == 0U) {
314 return;
315 }
316 const int32_t per = slack / (int32_t)spaces;
317 const int32_t rem = slack % (int32_t)spaces;
318 int32_t added = 0;
319 uint32_t seen = 0U;
320 for (uint32_t i = lo; i < hi; ++i) {
321 engine->glyphs[i].x += added;
322 if (engine->glyphs[i].cp == (int32_t)' ') {
323 added += per;
324 if ((int32_t)seen < rem) {
325 added += 1;
326 }
327 seen++;
328 }
329 }
330}

357RA8_INTERNAL
358static void priv_finish_line(ra8_reflow_t* engine, priv_cursor_t* cur, bool allow_justify)
359{
360 if (cur->align == (uint8_t)k_ra8_reflow_align_left) {
361 return;
362 }
363 const uint32_t lo = cur->line_first_glyph;
364 uint32_t hi = engine->glyph_count;
365 if (hi <= lo) {
366 return;
367 }
368 int32_t content_right = cur->x;
369 if (engine->glyphs[hi - 1U].cp == (int32_t)' ') {
370 content_right = engine->glyphs[hi - 1U].x;
371 hi--;
372 }
373 const int32_t right_limit = (int32_t)engine->viewport_w - (int32_t)k_ra8_reflow_margin_px;
374 const int32_t slack = right_limit - content_right;
375 if ((hi <= lo) || (slack <= 0)) {
376 return;
377 }
378 if (cur->align == (uint8_t)k_ra8_reflow_align_justify) {
379 if (allow_justify) {
380 priv_justify_glyphs(engine, lo, hi, slack);
381 }
382 return;
383 }
384 const int32_t offset = (cur->align == (uint8_t)k_ra8_reflow_align_center) ? (slack / 2) : slack;
385 for (uint32_t i = lo; i < hi; ++i) {
386 engine->glyphs[i].x += offset;
387 }
388}

442static ra8_err_t priv_emit_char(ra8_reflow_t* engine,
443 priv_cursor_t* cur,
444 const stbtt_fontinfo* font,
445 int32_t cp,
446 uint32_t color,
447 uint8_t link_id)
448{
449 const int32_t advance = ra8_reflow_layout_glyph_advance(font, cur->active_font_px, cp);
450 const int32_t advance_clamped =
451 (advance < (int32_t)k_priv_min_word_w_px) ? (int32_t)k_priv_min_word_w_px : advance;
452
453 const int32_t right_limit = (int32_t)engine->viewport_w - (int32_t)k_ra8_reflow_margin_px;
454 if (ra8_reflow_internal_right_overflow_break(cur->x,
455 advance_clamped,
456 right_limit,
457 cur->line_has_content)) {
458 if (!ra8_reflow_layout_newline(engine, cur, true)) { /* wrap -> previous line may justify */
459 return k_ra8_err_no_mem;
460 }
461 }
462 if (!ra8_reflow_layout_push_glyph(engine,
463 cur->x,
464 cur->y,
465 cp,
466 cur->active_font_px,
467 cur->active_style,
468 color,
469 link_id)) {
470 return k_ra8_err_no_mem;
471 }
472 cur->x += advance_clamped;
473 cur->line_has_content = 1U;
474 return k_ra8_ok;
475}

495RA8_INTERNAL
496static ra8_err_t priv_layout_text(ra8_reflow_t* engine,
497 priv_cursor_t* cur,
498 const stbtt_fontinfo* font,
499 const ra8_reflow_token_t* tok)
500{
501 const uint8_t* base = engine->text_pool + tok->text_off;
502 const uint32_t len = tok->text_len;
503 const uint8_t link_id = tok->reserved; /* 1-based `<a>` link id (0 = none) */
504 /* Colour precedence: an explicit CSS `color` (#140) wins; otherwise a link run
505 * uses the link colour and ordinary text the body colour. Link colour is keyed
506 * on actual link membership, not the underline style bit, so CSS
507 * `text-decoration: underline` (#111) on non-link text keeps the body colour. */
508 const uint32_t fallback = (link_id != 0U) ? engine->link_color : engine->body_color;
509 const uint32_t color =
510 (tok->color != (uint32_t)k_ra8_reflow_color_inherit) ? tok->color : fallback;
511
512 /* Pre-scan to find each word boundary. Greedy: if word_w + cursor_x
513 * exceeds the right margin AND the line already has content, break
514 * before emitting the word. */
515 uint32_t i = 0U;
516 while (i < len) {
517 /* Emit any leading whitespace as one space (already collapsed by
518 * the parser). */
519 if (base[i] == ' ') {
520 ra8_err_t err = priv_emit_char(engine, cur, font, ' ', color, link_id);
521 if (err != k_ra8_ok) {
522 return err;
523 }
524 ++i;
525 continue;
526 }
527
528 /* Word: bytes up to the next space. Pre-measure for greedy break. */
529 uint32_t word_end = i;
530 int32_t word_w = 0;
531 while (word_end < len && base[word_end] != ' ') {
532 word_w += ra8_reflow_layout_glyph_advance(font, cur->active_font_px, (int32_t)base[word_end]);
533 ++word_end;
534 }
535 const int32_t right_limit = (int32_t)engine->viewport_w - (int32_t)k_ra8_reflow_margin_px;
536 if (ra8_reflow_internal_right_overflow_break(cur->x,
537 word_w,
538 right_limit,
539 cur->line_has_content)) {
540 if (!ra8_reflow_layout_newline(engine, cur, true)) { /* wrap -> previous line may justify */
541 return k_ra8_err_no_mem;
542 }
543 }
544 while (i < word_end) {
545 ra8_err_t err = priv_emit_char(engine, cur, font, (int32_t)base[i], color, link_id);
546 if (err != k_ra8_ok) {
547 return err;
548 }
549 ++i;
550 }
551 }
552 return k_ra8_ok;
553}

572RA8_INTERNAL
573static bool priv_open_block(ra8_reflow_t* engine, priv_cursor_t* cur, const ra8_reflow_token_t* tok)
574{
575 if (cur->line_has_content != 0U) {
576 if (!ra8_reflow_layout_newline(engine, cur, false)) {
577 return false;
578 }
579 }
580 /* The block's text alignment travels in the block-start token's reserved byte
581 * (set by the tokenizer from `style="text-align:..."`). */
582 cur->align = tok->reserved;
583 /* Record an `id=` anchor (carried in the block-start token's slice) at this
584 * block's page + top, for same-chapter `#fragment` jumps. */
585 if ((tok->text_len > 0U) && (engine->anchor_count < (uint32_t)k_ra8_reflow_max_anchors)) {
586 ra8_reflow_anchor_t* anchor = &engine->anchors[engine->anchor_count];
587 anchor->id_off = tok->text_off;
588 anchor->id_len = tok->text_len;
589 anchor->page_index = engine->page_count;
590 anchor->y = cur->y;
591 engine->anchor_count++;
592 }
593 /* CSS `font-size` (#140) on the block-start token wins; else the UA default
594 * (body size or heading scale). 0 = no CSS font, so unstyled content is
595 * byte-identical. */
596 cur->active_font_px = (tok->css_font_px != 0U)
597 ? tok->css_font_px
598 : priv_block_font_px(engine->font_px, (ra8_reflow_html_tag_t)tok->tag);
599 cur->line_height_px = ra8_reflow_layout_line_height(cur->active_font_px);
600 if (ra8_reflow_internal_is_indent_tag((uint8_t)tok->tag)) {
601 cur->indent_px = k_ra8_reflow_indent_px;
602 cur->x = (int32_t)k_ra8_reflow_margin_px + (int32_t)cur->indent_px;
603 }
604 return true;
605}

733void ra8_reflow_layout_build_link_rects(ra8_reflow_t* engine, const stbtt_fontinfo* font)
734{
735 for (uint32_t p = 0U; p < engine->page_count; ++p) {
736 const ra8_reflow_page_t* page = &engine->pages[p];
737 const uint32_t base = page->glyph_first;
738 uint32_t k = 0U;
739 while (k < page->glyph_count) {
740 const uint8_t link = engine->glyphs[base + k].reserved;
741 if (link == 0U) {
742 ++k;
743 continue;
744 }
745 const int32_t y = engine->glyphs[base + k].y;
746 uint32_t e = k + 1U;
747 while ((e < page->glyph_count) && (engine->glyphs[base + e].reserved == link) &&
748 (engine->glyphs[base + e].y == y)) {
749 ++e;
750 }
751 priv_emit_link_rect(engine, font, base + k, base + e, link, p);
752 k = e;
753 }

758ra8_err_t ra8_reflow_layout_apply_token(ra8_reflow_t* engine,
759 priv_cursor_t* cur,
760 const stbtt_fontinfo* font,
761 const ra8_reflow_token_t* tok)
762{
763 /* Pack the per-run embedded-face index (stamped on the token by the cascade,
764 * #109) into the free high nibble of the style stamp; bits 0-2 keep the
765 * bold/italic/underline emphasis. Single-face content carries face 0, so the
766 * style byte is unchanged. */
767 cur->active_style =
768 (uint8_t)(tok->style | (uint8_t)(((uint32_t)tok->reserved16 & (uint32_t)k_ra8_reflow_face_mask)
769 << (uint32_t)k_ra8_reflow_face_shift));
770 switch (tok->kind) {
771 case k_ra8_reflow_tok_block_start:
772 return priv_open_block(engine, cur, tok) ? k_ra8_ok : k_ra8_err_no_mem;
773 case k_ra8_reflow_tok_block_end:
774 return priv_close_block(engine, cur, tok) ? k_ra8_ok : k_ra8_err_no_mem;
775 case k_ra8_reflow_tok_text:
776 return priv_layout_text(engine, cur, font, tok);
777 case k_ra8_reflow_tok_break:
778 return ra8_reflow_layout_newline(engine, cur, false) ? k_ra8_ok : k_ra8_err_no_mem;
779 case k_ra8_reflow_tok_rule:
780 return priv_apply_rule(engine, cur) ? k_ra8_ok : k_ra8_err_no_mem;
781 case k_ra8_reflow_tok_image:
782 return ra8_reflow_layout_apply_image(engine, cur, tok) ? k_ra8_ok : k_ra8_err_no_mem;
783 default:
784 return k_ra8_ok;
785 }
