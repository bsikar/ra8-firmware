159int32_t main(void)
160{
161 i2c_demo_setup_or_halt();
162 ra8_isr_globals_enable();
163
164 while (1) {
165 bool acked = false;
166 const ra8_err_t err =
167 ra8_i2c_scan((uint8_t)k_i2c_demo_iic_channel, (uint8_t)k_i2c_demo_probe_addr, &acked);
168 const uint8_t* msg = k_i2c_demo_msg_err;
169 uint32_t msg_len = (uint32_t)(sizeof(k_i2c_demo_msg_err) - 1U);
170 if (err == k_ra8_ok) {
171 if (acked) {
172 msg = k_i2c_demo_msg_ack;
173 msg_len = (uint32_t)(sizeof(k_i2c_demo_msg_ack) - 1U);
174 } else {
175 msg = k_i2c_demo_msg_nack;
176 msg_len = (uint32_t)(sizeof(k_i2c_demo_msg_nack) - 1U);
177 }
178 } else {
179 (void)ra8_board_led_on(k_ra8_board_led2);
180 }
181 if (ra8_board_uart_console_write(msg, (size_t)msg_len) != k_ra8_ok) {
182 break;
183 }
184 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
185 break;
186 }
187 ra8_delay_ms(k_i2c_demo_period_ms);
188 }
189
190 i2c_demo_panic_halt();
191 return 0;
192}
