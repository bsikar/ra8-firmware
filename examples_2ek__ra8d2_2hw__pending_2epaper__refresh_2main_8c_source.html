362static uint32_t ep_u32_to_dec(uint8_t* buf, uint32_t val)
363{
364 if (val == 0U) {
365 buf[0] = (uint8_t)'0';
366 return 1U;
367 }
368 uint8_t tmp[k_ep_dec_digits];
369 uint32_t n = 0U;
370 uint32_t v = val;
371 while ((v != 0U) && (n < (uint32_t)k_ep_dec_digits)) {
372 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint32_t)k_ep_radix));
373 v = v / (uint32_t)k_ep_radix;
374 n++;
375 }
376 for (uint32_t i = 0U; i < n; i++) {
377 buf[i] = tmp[n - 1U - i];
378 }
379 return n;
380}

698static void ep_report_stage(const char* label, uint32_t dt_ms, uint32_t pixels)
699{
700 static const uint8_t k_prefix[] = "epaper: ";
701 static const uint8_t k_ms_sep[] = " ms=";
702 static const uint8_t k_px_sep[] = " px=";
703 uint8_t line[k_ep_line_max];
704 uint32_t pos = 0U;
705 for (uint32_t i = 0U; i < (sizeof(k_prefix) - 1U); i++) {
706 line[pos++] = k_prefix[i];
707 }
708 for (uint32_t i = 0U; (label[i] != '\0') && (pos < (uint32_t)k_ep_line_max); i++) {
709 line[pos++] = (uint8_t)label[i];
710 }
711 for (uint32_t i = 0U; i < (sizeof(k_ms_sep) - 1U); i++) {
712 line[pos++] = k_ms_sep[i];
713 }
714 pos += ep_u32_to_dec(&line[pos], dt_ms);
715 for (uint32_t i = 0U; i < (sizeof(k_px_sep) - 1U); i++) {
716 line[pos++] = k_px_sep[i];
717 }
718 pos += ep_u32_to_dec(&line[pos], pixels);
719 line[pos++] = (uint8_t)k_ep_cr;
720 line[pos++] = (uint8_t)k_ep_lf;
721 ep_emit(line, pos);
722}

773static void ep_run_refresh_cycle(ra8_err_t* full_err, ra8_err_t* part_err)
774{
775 /* Full refresh: checkerboard + GC16 (quality). */
776 ep_paint_checker();
777 const uint32_t t0_full = ra8_now_ms();
778 *full_err = display_flush(s_display, display_full_rect(s_display), k_display_refresh_quality);
779 ep_report_stage("full-refresh",
780 ra8_now_ms() - t0_full,
781 (uint32_t)k_ep_panel_w * (uint32_t)k_ep_panel_h);
782
783 /* Partial refresh: blacken a sub-rect + A2 (fast), bounds-checked first. */
784 *part_err = k_ra8_err_invalid_arg;
785 if (ep_region_fits((uint16_t)k_ep_box_x,
786 (uint16_t)k_ep_box_y,
787 (uint16_t)k_ep_box_w,
788 (uint16_t)k_ep_box_h,
789 (uint16_t)k_ep_panel_w,
790 (uint16_t)k_ep_panel_h)) {
791 ep_paint_region_black();
792 const display_rect_t box = {
793 .x = (uint16_t)k_ep_box_x,
794 .y = (uint16_t)k_ep_box_y,
795 .w = (uint16_t)k_ep_box_w,
796 .h = (uint16_t)k_ep_box_h,
797 };
798 const uint32_t t0_part = ra8_now_ms();
799 *part_err = display_flush(s_display, box, k_display_refresh_fast);
800 ep_report_stage("partial-refresh",
801 ra8_now_ms() - t0_part,
802 (uint32_t)k_ep_box_w * (uint32_t)k_ep_box_h);
803 }
804}

821int32_t main(void)
822{
823 uint32_t pclka_hz = 0U;
824 ep_bringup_core(&pclka_hz);
825 ra8_isr_globals_enable();
826 ra8_delay_ms((uint32_t)k_ep_powerup_ms);
827 ep_bringup_panel_bus(pclka_hz);
828
829 const display_cfg_t cfg = {
830 .iface = &k_display_backend_eink_it8951,
831 .framebuffer = s_framebuffer,
832 .framebuffer_bytes = sizeof(s_framebuffer),
833 .width_px = (uint16_t)k_ep_panel_w,
834 .height_px = (uint16_t)k_ep_panel_h,
835 .pixfmt = k_display_pixfmt_rgb565,
836 .panel_timing = &s_epaper_cfg,
837 };
838
839 const ra8_err_t init_err = display_init(&cfg, &s_display);
840 if ((init_err != k_ra8_ok) || (display_get_caps(s_display, &s_caps) != k_ra8_ok) ||
841 (display_get_framebuffer(s_display, &s_fb) != k_ra8_ok)) {
842 ep_emit(k_ep_msg_fail, (uint32_t)(sizeof(k_ep_msg_fail) - 1U));
843 ep_panic_halt();
844 }
845 ep_report_init();
846
847 /* INV-VCOM-1: nothing gets refreshed until this panel's own bias has
848 * been resolved and confirmed. Deliberately a halt with its own banner
849 * rather than a best-effort draw -- a blank screen is a support call,
850 * a DC-biased panel is landfill. The driver would refuse the refresh
851 * anyway; failing here says why. */
852 if (!ep_calibrate_vcom()) {
853 ep_emit(k_ep_msg_no_vcom, (uint32_t)(sizeof(k_ep_msg_no_vcom) - 1U));
854 ep_panic_halt();
855 }
856
857 ra8_err_t full_err = k_ra8_err_invalid_arg;
858 ra8_err_t part_err = k_ra8_err_invalid_arg;
859 ep_run_refresh_cycle(&full_err, &part_err);
860
861 (void)display_deinit(s_display); /* sleep the panel */
862
863 if (!ep_flush_ok(init_err, full_err, part_err)) {
864 ep_emit(k_ep_msg_fail, (uint32_t)(sizeof(k_ep_msg_fail) - 1U));
865 ep_panic_halt();
866 }
867 ep_emit(k_ep_msg_pass, (uint32_t)(sizeof(k_ep_msg_pass) - 1U));
868
869 while (1) {
870 (void)ra8_board_led_toggle(k_ra8_board_led_blue);
871 ra8_delay_ms((uint32_t)k_ep_heartbeat_ms);
872 }
873 return 0;
874}
