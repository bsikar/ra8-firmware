165static uint8_t rtc_bin_to_bcd(uint8_t bin)
166{
167 return (uint8_t)(((uint8_t)(bin / (uint8_t)k_bcd_base) << (uint8_t)k_bcd_shift) |
168 (uint8_t)(bin % (uint8_t)k_bcd_base));
169}

172static uint8_t rtc_bcd_to_bin(uint8_t bcd)
173{
174 return (uint8_t)(((uint8_t)((bcd >> (uint8_t)k_bcd_shift) & (uint8_t)k_nibble_mask) *
175 (uint8_t)k_bcd_base) +
176 (uint8_t)(bcd & (uint8_t)k_nibble_mask));
177}

180static void rtc_publish_calendar(void)

182 s_rtc.reg[(uint64_t)k_rtc_off_r64cnt] = (uint8_t)(s_rtc.r64 & (uint8_t)k_rtc_r64_mask);

209static uint64_t rtc_read(uc_engine* uc, uint64_t addr, unsigned size)
210{
211 (void)uc;
212 const uint64_t off = addr - (uint64_t)k_rtc_base;
213 uint64_t v = 0U;
214 for (unsigned i = 0U; i < size; i++) {
215 const uint64_t b = off + (uint64_t)i;
216 if (b < (uint64_t)k_rtc_span) {
217 v |= (uint64_t)s_rtc.reg[b] << (8U * i);
218 }
219 }
220 return v;
221}

224static bool rtc_is_calendar_off(uint64_t off)
225{
226 return (off == (uint64_t)k_rtc_off_rseccnt) || (off == (uint64_t)k_rtc_off_rmincnt) ||
227 (off == (uint64_t)k_rtc_off_rhrcnt) || (off == (uint64_t)k_rtc_off_rdaycnt) ||
228 (off == (uint64_t)k_rtc_off_rmoncnt) || (off == (uint64_t)k_rtc_off_ryrcnt);
229}

232static void rtc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
233{
234 (void)uc;
235 const uint64_t off = addr - (uint64_t)k_rtc_base;
236 bool touched_cal = false;
237 for (unsigned i = 0U; i < size; i++) {
238 const uint64_t b = off + (uint64_t)i;
239 if (b >= (uint64_t)k_rtc_span) {
240 continue;
241 }
242 s_rtc.reg[b] = (uint8_t)(value >> (8U * i));
243 if (rtc_is_calendar_off(b)) {
244 touched_cal = true;
245 }
246 }
247 /* A calendar-count write (ra8_rtc_set, performed with START cleared) reseeds
248 * the running time mirror so the clock continues from the new value. */
249 if (touched_cal) {
250 rtc_latch_calendar();
251 }
252}

260static void rtc_advance_one_second(void)
261{
262 s_rtc.sec++;
263 if (s_rtc.sec < (uint8_t)k_rtc_secs_per_min) {
264 return;
265 }
266 s_rtc.sec = 0U;
267 s_rtc.min++;
268 if (s_rtc.min < (uint8_t)k_rtc_mins_per_hour) {
269 return;
270 }
271 s_rtc.min = 0U;
272 s_rtc.hour++;
273 if (s_rtc.hour < (uint8_t)k_rtc_hours_per_day) {
274 return;
275 }
276 s_rtc.hour = 0U;
277 s_rtc.day++;
278 if (s_rtc.day <= (uint8_t)k_rtc_days_per_mon) {
279 return;
280 }
281 s_rtc.day = 1U;
282 s_rtc.mon++;
283 if (s_rtc.mon <= (uint8_t)k_rtc_mons_per_year) {
284 return;
285 }
286 s_rtc.mon = 1U;
287 s_rtc.year++;

291static bool rtc_alarm_matches(void)
292{
293 const uint8_t sar = s_rtc.reg[(uint64_t)k_rtc_off_rsecar];
294 const uint8_t mar = s_rtc.reg[(uint64_t)k_rtc_off_rminar];
295 const uint8_t har = s_rtc.reg[(uint64_t)k_rtc_off_rhrar];
296 /* Alarm is meaningful only if at least one field has its ENB bit set. */
297 if (((sar | mar | har) & (uint8_t)k_rtc_alarm_enb) == 0U) {
298 return false;
299 }
300 const bool sec_ok = ((sar & (uint8_t)k_rtc_alarm_enb) == 0U) ||
301 (rtc_bcd_to_bin((uint8_t)(sar & (uint8_t)k_rtc_alarm_val)) == s_rtc.sec);
302 const bool min_ok = ((mar & (uint8_t)k_rtc_alarm_enb) == 0U) ||
303 (rtc_bcd_to_bin((uint8_t)(mar & (uint8_t)k_rtc_alarm_val)) == s_rtc.min);
304 const bool hr_ok = ((har & (uint8_t)k_rtc_alarm_enb) == 0U) ||
305 (rtc_bcd_to_bin((uint8_t)(har & (uint8_t)k_rtc_alarm_val)) == s_rtc.hour);
306 return sec_ok && min_ok && hr_ok;
307}

310static void rtc_raise_events(uc_engine* uc)
311{
312 const uint8_t rcr1 = s_rtc.reg[(uint64_t)k_rtc_off_rcr1];
313 if (rtc_alarm_matches()) {
314 s_rtc.alarms++;
315 if ((rcr1 & (uint8_t)k_rtc_rcr1_aie) != 0U) {
316 board_periph_icu_raise_event(uc, (uint16_t)k_rtc_event_alarm);
317 }
318 if (board_periph_trace()) {
319 (void)fprintf(stderr,
320 " RTC : alarm match %02u:%02u:%02u\n",
321 s_rtc.hour,
322 s_rtc.min,
323 s_rtc.sec);
324 }
325 }
326 if ((rcr1 & (uint8_t)k_rtc_rcr1_pie) != 0U) {
327 s_rtc.periodics++;
328 board_periph_icu_raise_event(uc, (uint16_t)k_rtc_event_periodic);
329 }
330}

333static void rtc_tick(uc_engine* uc)
334{
335 if ((s_rtc.reg[(uint64_t)k_rtc_off_rcr2] & (uint8_t)k_rtc_rcr2_start) == 0U) {
336 return; /* stopped: time holds */
337 }
338 s_rtc.r64 = (uint8_t)(s_rtc.r64 + 1U);
339 s_rtc.tick_frac++;
340 if (s_rtc.tick_frac < (uint32_t)k_rtc_ticks_per_sec) {
341 rtc_publish_calendar();
342 return;
343 }
344 s_rtc.tick_frac = 0U;
345 rtc_advance_one_second();
346 rtc_publish_calendar();
347 rtc_raise_events(uc);
348}
