95priv_webp_pull_all(ra8_jof_prefix_pull_t* pfx, uint8_t* dst, size_t cap, size_t* out_len)
96{
97 size_t len = 0U;
98 while (len < cap) {
99 size_t got = 0U;
100 const ra8_err_t err = ra8_jof_priv_prefix_pull(pfx, &dst[len], cap - len, &got);
101 if (err != k_ra8_ok) {
102 return err;
103 }
104 if (got == 0U) {
105 *out_len = len;
106 return k_ra8_ok;
107 }
108 len += got;
109 }

112 return k_ra8_err_invalid_size;

159RA8_PRIV ra8_err_t ra8_jof_priv_webp_transcode(ra8_jof_prod_state_t* st, ra8_jof_prefix_pull_t* pfx)
160{
161 uint8_t* const arena = st->cfg->webp_work;
162 if (arena == nullptr) {
163 return k_ra8_err_not_supported; /* WebP source but no whole-frame arena */
164 }
165 const size_t acap = st->cfg->webp_work_cap;
166 size_t src_len = 0U;
167 ra8_err_t err = priv_webp_pull_all(pfx, arena, acap, &src_len);
168 if (err != k_ra8_ok) {
169 return err;
170 }
171 uint32_t w = 0U;
172 uint32_t h = 0U;
173 err = ra8_webp_get_info(arena, src_len, &w, &h);
174 if (err != k_ra8_ok) {
175 return err;
176 }
177 err = ra8_jof_priv_on_geom(st, (uint16_t)w, (uint16_t)h, (uint8_t)k_ra8_jof_webp_bpp);
178 if (err != k_ra8_ok) {
179 return err;
180 }
181 const size_t mask = (size_t)k_ra8_jof_bump_align - 1U;
182 const size_t used = (src_len + mask) & ~mask;
183 const uint64_t frame64 = (uint64_t)w * (uint64_t)h * (uint64_t)k_ra8_jof_webp_bpp;
184 if ((used >= acap) || (frame64 >= (uint64_t)(acap - used))) {
185 return k_ra8_err_invalid_size; /* no room for frame + at least some scratch */
186 }
187 const size_t frame_n = (size_t)frame64;
188 uint8_t* const frame = &arena[used];
189 const size_t scr_off = used + frame_n;
190 ra8_webp_arena_t wa = {.base = &arena[scr_off], .cap = acap - scr_off, .offset = 0U, .live = 0U};
191 err = ra8_webp_decode_rgba(arena,
192 src_len,
193 &wa,
194 frame,
195 (size_t)w * (size_t)k_ra8_jof_webp_bpp,
196 frame_n,
197 nullptr,
198 nullptr);
199 if (err != k_ra8_ok) {
200 return err;
201 }
202 return priv_webp_feed(st, frame, (uint16_t)w, (uint16_t)h);
203}
