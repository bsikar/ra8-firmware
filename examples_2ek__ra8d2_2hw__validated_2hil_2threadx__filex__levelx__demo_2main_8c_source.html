225static void demo_fx_format_or_panic(void)
226{
227 UINT status = lx_filex_adapter_bind(&s_nor_flash);
228 if (status != FX_SUCCESS) {
229 demo_print("[fxlx] lx_filex_adapter_bind failed\r\n");
230 demo_panic_halt();
231 }
232 ULONG total_sectors = lx_filex_adapter_get_total_sectors();
233 if (total_sectors == 0U) {
234 demo_print("[fxlx] adapter reports zero sectors\r\n");
235 demo_panic_halt();
236 }
237 status = fx_media_format(&s_media,
238 fx_media_driver_ra8_levelx,
239 LX_NULL,
240 s_media_memory,
241 (UINT)sizeof(s_media_memory),
242 s_fx_volume_name,
243 1U, /* number of FATs */
244 32U, /* root directory entries */
245 0U, /* hidden sectors */
246 total_sectors, /* total logical sectors */
247 (UINT)k_demo_sector_bytes, /* bytes per sector */
248 1U, /* sectors per cluster */
249 1U, /* heads (non-rotating) */
250 1U); /* sectors per track */
251 if (status != FX_SUCCESS) {
252 demo_print("[fxlx] fx_media_format failed\r\n");
253 demo_panic_halt();
254 }
255 status = fx_media_open(&s_media,
256 s_fx_volume_name,
257 fx_media_driver_ra8_levelx,
258 LX_NULL,
259 s_media_memory,

311static void demo_read_test_file(void)
312{
313 UINT status = fx_file_open(&s_media, &s_file, s_demo_file_path, FX_OPEN_FOR_READ);
314 if (status != FX_SUCCESS) {
315 demo_print("[fxlx] fx_file_open(read) failed\r\n");
316 return;
317 }
318 demo_print("[fxlx] readback: ");
319 CHAR buf[k_demo_file_chunk];
320 ULONG actual = 0U;
321 do {
322 actual = 0U;
323 status = fx_file_read(&s_file, buf, (ULONG)k_demo_file_chunk, &actual);
324 if (actual > 0U) {
325 (void)ra8_board_uart_console_write((const uint8_t*)buf, (size_t)actual);
326 }
327 } while (status == FX_SUCCESS && actual > 0U);
328 demo_print("\r\n");
329 (void)fx_file_close(&s_file);
330}

342static void demo_thread_entry(ULONG thread_input)
343{
344 (void)thread_input;
345
346 demo_print("[fxlx] booting xSPI flash\r\n");
347 /* The LevelX NOR driver owns OCTA bus bring-up: lx_nor_flash_format ->
348 * lx_nor_driver_ra8_xspi_initialize -> priv_bus_init_once routes the
349 * pins, runs the 8D/1S software-reset recovery, calls ra8_xspi_init,
350 * and probes RDID exactly once. Doing it here as well double-routes
351 * the PFS pins (the validator rejects the second route with
352 * k_ra8_err_gpio_conflict), so the driver's initialize bails before
353 * wiring the sector buffer and format returns LX_NO_MEMORY. Let the
354 * driver be the single owner. */
355 demo_print("[fxlx] formatting + opening LevelX partition\r\n");
356 demo_lx_open_or_panic();
357
358 demo_print("[fxlx] formatting + opening FAT volume on LevelX\r\n");
359 demo_fx_format_or_panic();
360
361 demo_write_test_file();
362 demo_read_test_file();
363
364 (void)fx_media_close(&s_media);
365 (void)lx_nor_flash_close(&s_nor_flash);
366 demo_print("[fxlx] done\r\n");
367}
