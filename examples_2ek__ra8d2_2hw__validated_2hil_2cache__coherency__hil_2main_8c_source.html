263static void
264cache_coherency_round(volatile cache_coherency_shared_t* shared, uint32_t round, uint32_t* next_seq)
265{
266 if ((shared == nullptr) || (next_seq == nullptr)) {
267 return;
268 }
269 const uint32_t expect = (uint32_t)k_cache_coherency_pong_base + round;
270
271 shared->ping_payload = (uint32_t)k_cache_coherency_ping_base + round;
272 __asm volatile("dsb" ::: "memory");
273 shared->ping_seq = *next_seq;
274
275 bool ok = wait_for_pong(shared, *next_seq);
276 if (ok) {
277 ok = (shared->pong_payload == expect);
278 }
279 if (ok) {
280 g_cache_coherency_match += 1U;
281 } else {
282 g_cache_coherency_mismatch += 1U;
283 }
284 *next_seq += 1U;
285}

300int main(void)
301{
302 ra8_log_init();
303 ra8_log_info("M85", "cache_coherency_hil: D-cache ON, non-cacheable shared SRAM");
304
305 volatile cache_coherency_shared_t* shared = cache_coherency_shared();
306 /* CPU0 owns initialisation of the shared block. Zero it before
307 * releasing CPU1 so CPU1 starts from a known clean state. */
308 shared->ping_seq = 0U;
309 shared->pong_seq = 0U;
310 shared->ping_payload = 0U;
311 shared->pong_payload = 0U;
312 __asm volatile("dsb" ::: "memory");
313
314 /* Best-effort VCOM console (banner) + LED1 (self-test heartbeat). */
315 const bool console_up = cache_coherency_console_init();
316 if (!console_up) {
317 ra8_log_info("M85", "VCOM console init failed -- banner over ITM only");
318 }
319 const bool led_up = (ra8_board_led_init(k_ra8_board_led1) == k_ra8_ok);
320
321 const ra8_err_t err = ra8_cpu1_release(&g_ra8_ls_cpu1_mram_start, &g_ra8_ls_cpu1_stack_top);
322 ra8_log_info_val("M85", "ra8_cpu1_release rc (0 = ok)", (uint32_t)err);
323 if (err != k_ra8_ok) {
324 ra8_log_info("M85", "release FAILED -- halting");
325 park_forever();
326 }
327
328 uint32_t next_seq = 1U;
329 uint32_t round = 0U;
330 bool banner_sent = false;
331
332 while (1) {
333 /* Self-test heartbeat: one IOPORT write per round keeps ra8_emulator's
334 * MMIO-keyed idle detector awake through the pure-SRAM hand-off so the
335 * banner surfaces under RA8_EMU_IDLE_STOP=1 (and the LED visibly
336 * blinks on the bench). Dropped once the self-test has passed, so the
337 * steady phase below is pure SRAM and IDLE_STOP can stop the fake. */
338 if (led_up && (round < (uint32_t)k_cache_coherency_rounds)) {
339 (void)ra8_board_led_toggle(k_ra8_board_led1);
340 }
341
342 cache_coherency_round(shared, round, &next_seq);
343
344 if (!banner_sent && (g_cache_coherency_match >= (uint32_t)k_cache_coherency_rounds)) {
345 cache_coherency_emit_pass();
346 banner_sent = true;
347 }
348 round += 1U;
349 }
