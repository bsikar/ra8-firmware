151RA8_INTERNAL
152static ra8_usb_dev_state_t internal_decode_dvsq(uint16_t intsts0)
153{
154 const uint16_t dvsq = (uint16_t)(intsts0 & k_ra8_intsts0_mask_dvsq);
155 /* Suspend bit (0x40) wins over the lower three states. */
156 if ((dvsq & k_ra8_dvsq_suspend) != 0U) {
157 return k_ra8_usb_dev_state_suspended;
158 }
159 if (dvsq == k_ra8_dvsq_default) {
160 return k_ra8_usb_dev_state_default;
161 }
162 if (dvsq == k_ra8_dvsq_address) {
163 return k_ra8_usb_dev_state_address;
164 }
165 if (dvsq == k_ra8_dvsq_configured) {
166 return k_ra8_usb_dev_state_configured;
167 }
168 return k_ra8_usb_dev_state_powered;
169}

186ra8_err_t ra8_usb_device_init(ra8_usb_speed_t speed)
187{
188 volatile r_usb_regs_t* reg = internal_pick(speed);
189 if (reg == nullptr) {
190 return k_ra8_err_invalid_arg;
191 }
192
193 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B", p 444 */
194 const ra8_err_t mst_err = ra8_mstp_enable(internal_mstp(speed));
195 RA8_RETURN_ON_ERROR(mst_err, s_tag, "usb_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
196
197 if (speed == k_ra8_usb_speed_hs) {
198 const ra8_err_t phy_err = internal_usbhs_phy_bringup(reg);
199 RA8_RETURN_ON_ERROR(phy_err, s_tag, "usb_init: HS PHY bring-up"); /* GCOVR_EXCL_BR_LINE */
200 } else {
201 const ra8_err_t fs_err = internal_usbfs_module_bringup(reg);
202 RA8_RETURN_ON_ERROR(fs_err, s_tag, "usb_init: FS module bring-up"); /* GCOVR_EXCL_BR_LINE */
203 }
204
205 internal_usb_init_common(reg);
206
207 /* Bisect probes: capture SYSCFG/LPSTS state at the END of device-init,
208 * BEFORE control returns to ra8_board_usbhs_device_init / the DCD
209 * bridge. HUM Ch 37.2.1 SYSCFG p 2060, HUM Ch 37.2.43 LPSTS p 2111. */
210 if (speed == k_ra8_usb_speed_hs) {
211 s_syscfg_after_phy_bringup = reg->SYSCFG;
212 s_lpsts_after_phy_bringup = *ra8_usbhs_lpsts();
213 }
214
215 ra8_log_info_val(s_tag, "usb device init speed", (uint32_t)speed);
216 return k_ra8_ok;
217}

265ra8_err_t ra8_usb_device_attach(ra8_usb_speed_t speed, bool attached)
266{
267 volatile r_usb_regs_t* reg = internal_pick(speed);
268 if (reg == nullptr) {
269 return k_ra8_err_invalid_arg;
270 }
271
272 /* HUM Ch 36.2.1 "SYSCFG : System Configuration Control Register", p 1967 */
273 /* HUM Ch 37.2.1 "SYSCFG : System Configuration Control Register", p 2061 */
274 const uint16_t dprpu = (uint16_t)(1U << k_ra8_syscfg_bit_dprpu);
275 const uint16_t cnen = (uint16_t)(1U << k_ra8_syscfg_bit_cnen);
276
277 if (attached) {
278 /* HUM Ch 37.2.1 SYSCFG, "CNEN bit (Single-ended Receiver Enable)"
279 * p 2062: "In device controller mode, set this bit to 1 when VBUS
280 * is detected because of a VBUS interrupt, and set it to 0 when
281 * the VBUS line is removed."
282 *
283 * Without CNEN=1 the HS PHY's single-end receivers are powered
284 * down and SYSSTS0.LNST[1:0] reads 00b (SE0) regardless of what
285 * the host's 1.5 kohm-or-15 kohm pull-down arrangement and the
286 * device's DPRPU 1.5 kohm pull-up are doing on the wires.
287 *
288 * Programme CNEN BEFORE DPRPU so the receivers latch the line
289 * state the moment DPRPU pulls D+ high; this is the FS-attach
290 * sequence sketched in HUM Ch 37.3.3 Figure 37.2 p 2121
291 * ("Resistor control" step comes after USBE/CNEN are settled).
292 * The FS instance has no CNEN gate (HUM Ch 36.2.1 SYSCFG p 1966),
293 * so CNEN is HS-only here. */
294 if (speed == k_ra8_usb_speed_hs) {
295 internal_rmw16(&reg->SYSCFG, cnen, 0U);
296 s_syscfg_before_dprpu = reg->SYSCFG;
297 }
298 internal_rmw16(&reg->SYSCFG, dprpu, 0U);
299 } else {
300 internal_rmw16(&reg->SYSCFG, 0U, dprpu);
301 if (speed == k_ra8_usb_speed_hs) {
302 /* HUM Ch 37.2.1 p 2062: clear CNEN when VBUS is removed /
303 * detach is requested, to avoid through-current on floating
304 * D+/D- when the cable is unplugged. */
305 internal_rmw16(&reg->SYSCFG, 0U, cnen);
306 }
307 }
308
309 /* Bisect probes: capture SYSCFG/LPSTS/SYSSTS0 at end of attach. HUM
310 * Ch 37.2.1 SYSCFG p 2060, HUM Ch 37.2.3 SYSSTS0 p 2063, HUM
311 * Ch 37.2.43 LPSTS p 2111. */
312 if (speed == k_ra8_usb_speed_hs) {
313 s_syscfg_after_attach = reg->SYSCFG;
314 s_lpsts_after_attach = *ra8_usbhs_lpsts();
315 s_syssts0_after_attach = reg->SYSSTS0;
316 }
317 return k_ra8_ok;
318}

476ra8_err_t ra8_usb_device_busreset_rearm(ra8_usb_speed_t speed)
477{
478 volatile r_usb_regs_t* reg = internal_pick(speed);
479 if (reg == nullptr) {
480 return k_ra8_err_invalid_arg;
481 }
482
483 /* HUM Ch 36.2.27 "PIPECTR : Pipe n Control Register", p 2007.
484 * Clear PID for every non-control pipe so any half-completed pre-
485 * reset transfer is forgotten. The class driver re-issues queue_in /
486 * queue_out which will set PID=BUF when ready. */
487 for (uint8_t i = 0U; i < (uint8_t)k_ra8_usb_pipectr_count; i++) {
488 reg->PIPECTR[i] = 0U;
489 }
490
491 /* HUM Ch 36.2.13/14 BRDYSTS / NRDYSTS / BEMPSTS, p 1983-1985 (W0C).
492 * Drop any per-pipe status bits left over from before the reset so
493 * stale BRDY edges don't fire spurious BRDY callbacks during the next
494 * enumeration. Writing 0 clears every bit; writing 1 preserves. */
495 reg->BRDYSTS = 0U;
496 reg->NRDYSTS = 0U;
497 reg->BEMPSTS = 0U;
498
499 /* DCP defaults + FIFO clear (see helper). HUM Ch 36.2.19 / 37.2.29
500 * DCPCFG p 1989 / 2091 and HUM Ch 36.2.20 / 37.2.30 DCPMAXP p 1990
501 * / 2092. DCPCTR is intentionally NOT written here: HUM Ch 36.2.21
502 * / 37.2.31 (p 1991 / 2093) states the IP auto-defaults the
503 * writable DCPCTR fields on bus reset (PID=NAK, CCPL=0). */
504 internal_dcp_reset_defaults(reg);
505
506 /* HUM Ch 36.2.10 "INTENB0 : Interrupt Enable Register 0", p 1980.
507 * Some RA silicon revisions clear individual INTENB0 bits across a
508 * bus reset (notably CTRT). Re-apply the post-init mask defensively
509 * so the next SETUP raises CTRT as expected. Mirrors the mask in
510 * internal_usb_init_common. */
511 /* SOFR and RSME are intentionally NOT enabled here:
512 * - SOFR fires every 125us on HS (= 8 kHz);
513 * - RSME stays asserted on USBHS via the PHY's USBR signal
514 * while the host holds the bus in resume signalling;
515 * Both starve PendSV and the demo worker thread never gets
516 * scheduled. NRDY is still useful (drives the bridge's per-pipe
517 * NAK re-arm) and does not cause a storm on its own. RSME / SOFR
518 * status is still readable from INTSTS0 for poll-style drivers. */
519 reg->INTENB0 = (uint16_t)((1U << k_ra8_int0_bit_bemp) | (1U << k_ra8_int0_bit_brdy) |
520 (1U << k_ra8_int0_bit_nrdy) | (1U << k_ra8_int0_bit_ctrt) |
521 (1U << k_ra8_int0_bit_dvst) | (1U << k_ra8_int0_bit_vbse));
522
523 return k_ra8_ok;
524}

592static void
593internal_pipe_finalize(volatile r_usb_regs_t* reg, uint8_t pipe_num, ra8_usb_ep_dir_t dir)
594{
595 const uint8_t ctr_idx = (uint8_t)(pipe_num - 1U);
596 const uint16_t pipe_bit = (uint16_t)(1U << pipe_num);
597 internal_rmw16(&reg->PIPECTR[ctr_idx], k_ra8_pipectr_sqclr, 0U);
598 internal_rmw16(&reg->PIPECTR[ctr_idx], k_ra8_pipectr_aclrm, 0U);
599 internal_rmw16(&reg->PIPECTR[ctr_idx], 0U, k_ra8_pipectr_aclrm);
600 reg->BRDYSTS = (uint16_t)(~pipe_bit);
601 reg->BEMPSTS = (uint16_t)(~pipe_bit);
602 internal_pipe_pid(reg, pipe_num, (dir == k_ra8_usb_ep_dir_out) ? k_ra8_pid_buf : k_ra8_pid_nak);
603}

659ra8_err_t ra8_usb_configure_endpoint(ra8_usb_speed_t speed,
660 uint8_t pipe_num,
661 uint8_t ep_addr,
662 ra8_usb_ep_dir_t dir,
663 ra8_usb_ep_type_t type,
664 uint16_t max_packet)
665{
666 volatile r_usb_regs_t* reg = internal_pick(speed);
667 if (reg == nullptr) {
668 return k_ra8_err_invalid_arg;
669 }
670 const ra8_err_t arg_err = internal_check_ep_args(pipe_num, ep_addr, dir, type, max_packet);
671 if (arg_err != k_ra8_ok) {
672 return arg_err;
673 }
674
675 internal_pipe_quiesce(reg, pipe_num);
676
677 /* HUM Ch 36.2.23 "PIPESEL : Pipe Window Select Register", p 2002 */
678 reg->PIPESEL = pipe_num;
679 /* HUM Ch 36.2.24 "PIPECFG : Pipe Configuration Register", p 2003 */
680 reg->PIPECFG = internal_pipecfg_word(ep_addr, dir, type, false);
681 /* HUM Ch 37.2.35 "PIPEBUF : Pipe Buffer Register", p 2100. PIPEBUF is a
682 * USBHS-only register; the USBFS instance (Ch 36) has no equivalent.
683 * Bulk pipes get a single MPS-sized bank (DBLB clear). Interrupt/iso
684 * pipes keep the reset default. */
685 if (type == k_ra8_usb_ep_type_bulk) {
686 reg->PIPEBUF = internal_pipebuf_word(pipe_num, max_packet);
687 }
688 /* HUM Ch 36.2.26 "PIPEMAXP : Pipe Maximum Packet Size Register", p 2005 */
689 reg->PIPEMAXP = max_packet;
690 reg->PIPEPERI = 0U;
691 /* Deselect window (FIT step 4) so a later stray PIPECFG write does
692 * not land on this pipe. */
693 reg->PIPESEL = 0U;
694
695 internal_pipe_finalize(reg, pipe_num, dir);
696 internal_pipe_arm_irq(reg, pipe_num, dir);
697 return k_ra8_ok;
698}
