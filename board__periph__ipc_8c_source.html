170static bool ipc_decode(uint64_t off, uint32_t* ch, uint32_t* reg)
171{
172 if (off < (uint64_t)k_ipc_ch0_off) {
173 return false;
174 }
175 const uint64_t rel = off - (uint64_t)k_ipc_ch0_off;
176 const uint32_t channel = (uint32_t)(rel / (uint64_t)k_ipc_ch_stride);
177 if (channel >= (uint32_t)k_ipc_ch_count) {
178 return false;
179 }
180 *ch = channel;
181 *reg = (uint32_t)(rel % (uint64_t)k_ipc_ch_stride);
182 return true;
183}

207static uint32_t ipc_sta_value(uint32_t ch)
208{
209 const ipc_fifo_t* f = &s_ipc.fifo[ch];
210 uint32_t sta = s_ipc.sta[ch];
211 if (f->count > 0U) {
212 sta |= (uint32_t)k_ipc_sta_rdy;
213 }
214 if (f->count >= (uint32_t)k_ipc_fifo_depth) {
215 sta |= (uint32_t)k_ipc_sta_full;
216 }
217 if (f->rerr) {
218 sta |= (uint32_t)k_ipc_sta_rerr;
219 }
220 if (f->ferr) {
221 sta |= (uint32_t)k_ipc_sta_ferr;
222 }
223 return sta;
224}

242static uint32_t ipc_fifo_pop(uint32_t ch)
243{
244 ipc_fifo_t* f = &s_ipc.fifo[ch];
245 if (f->count == 0U) {
246 f->rerr = true; /* read-while-empty: latch RERR, data undefined (0). */
247 return 0U;
248 }
249 const uint32_t w = f->word[f->rd];
250 f->rd = (f->rd + 1U) % (uint32_t)k_ipc_fifo_depth;
251 f->count--;
252 s_ipc.pops++;
253 return w;
254}

257static uint64_t ipc_read(uc_engine* uc, uint64_t addr, unsigned size)
258{
259 (void)uc;
260 (void)size;
261 const uint64_t off = addr - (uint64_t)k_ipc_base;
262 uint32_t ch = 0U;
263 uint32_t reg = 0U;
264 if (ipc_decode(off, &ch, &reg)) {
265 if (reg == (uint32_t)k_ipc_reg_sta) {
266 return (uint64_t)ipc_sta_value(ch);
267 }
268 if (reg == (uint32_t)k_ipc_reg_rxd) {
269 return (uint64_t)ipc_fifo_pop(ch);
270 }
271 }
272 /* Semaphores and the NMI window are not modelled: read as 0. */
273 return 0U;
274}

293static void ipc_fifo_push(uint32_t ch, uint32_t value)
294{
295 ipc_fifo_t* f = &s_ipc.fifo[ch];
296 if (f->count >= (uint32_t)k_ipc_fifo_depth) {
297 f->ferr = true; /* write-while-full: latch FERR, drop the word. */
298 return;
299 }
300 f->word[(f->rd + f->count) % (uint32_t)k_ipc_fifo_depth] = value;
301 f->count++;
302 s_ipc.pushes++;
303}

321static void ipc_clr_write(uint32_t ch, uint32_t value)
322{
323 s_ipc.sta[ch] &= ~(value & (uint32_t)k_ipc_sta_irq_mask);
324 ipc_fifo_t* f = &s_ipc.fifo[ch];
325 if ((value & (uint32_t)k_ipc_clr_rst) != 0U) {
326 f->rd = 0U;
327 f->count = 0U; /* FIFO reset also drops RDY + FULL (they are computed). */
328 }
329 if ((value & (uint32_t)k_ipc_clr_rclr) != 0U) {
330 f->rerr = false;
331 }
332 if ((value & (uint32_t)k_ipc_clr_fclr) != 0U) {
333 f->ferr = false;
334 }
335}

338static void ipc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
339{
340 (void)size;
341 const uint64_t off = addr - (uint64_t)k_ipc_base;
342 uint32_t ch = 0U;
343 uint32_t reg = 0U;
344 if (!ipc_decode(off, &ch, &reg)) {
345 return; /* semaphore / NMI window write: benign no-op */
346 }
347 if (reg == (uint32_t)k_ipc_reg_iset) {
348 const uint32_t bits = (uint32_t)value & (uint32_t)k_ipc_sta_irq_mask;
349 if (bits == 0U) {
350 return;
351 }
352 s_ipc.sta[ch] |= bits;
353 s_ipc.sends++;
354 /* Raise the receiving core's ELC event: the core's ICU walks its IELSR
355 * shadow, pends the mapped NVIC line, and the M85 takes it on its next
356 * instruction boundary (cpu1 never drains, so an IPC1 raise is inert). */
357 const uint16_t event =
358 (ch < (uint32_t)k_ipc_unit_split) ? (uint16_t)k_ipc0_irq_event : (uint16_t)k_ipc1_irq_event;
359 board_periph_icu_raise_event(uc, event);
360 s_ipc.wakes++;
361 if (board_periph_trace()) {
362 (void)fprintf(stderr,
363 " IPC : ch%u ISET 0x%02X -> raise 0x%03X\n",
364 ch,
365 bits,
366 (unsigned)event);
367 }
368 return;
369 }
370 if (reg == (uint32_t)k_ipc_reg_txd) {
371 ipc_fifo_push(ch, (uint32_t)value);
372 return;
373 }
374 if (reg == (uint32_t)k_ipc_reg_clr) {
375 ipc_clr_write(ch, (uint32_t)value);
376 }
377}

391static void ipc_report(void)
392{
393 if ((s_ipc.sends == 0U) && (s_ipc.wakes == 0U) && (s_ipc.pushes == 0U) && (s_ipc.pops == 0U)) {
394 return;
395 }

397 " IPC : sends=%u wakes=%u fifo pushes=%u pops=%u\n",

400 s_ipc.pushes,
