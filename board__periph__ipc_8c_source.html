171static bool ipc_decode(uint64_t off, uint32_t* ch, uint32_t* reg)
172{
173 if (off < (uint64_t)k_ipc_ch0_off) {
174 return false;
175 }
176 const uint64_t rel = off - (uint64_t)k_ipc_ch0_off;
177 const uint32_t channel = (uint32_t)(rel / (uint64_t)k_ipc_ch_stride);
178 if (channel >= (uint32_t)k_ipc_ch_count) {
179 return false;
180 }
181 *ch = channel;
182 *reg = (uint32_t)(rel % (uint64_t)k_ipc_ch_stride);
183 return true;
184}

208static uint32_t ipc_sta_value(uint32_t ch)
209{
210 const ipc_fifo_t* f = &s_ipc.fifo[ch];
211 uint32_t sta = s_ipc.sta[ch];
212 if (f->count > 0U) {
213 sta |= (uint32_t)k_ipc_sta_rdy;
214 }
215 if (f->count >= (uint32_t)k_ipc_fifo_depth) {
216 sta |= (uint32_t)k_ipc_sta_full;
217 }
218 if (f->rerr) {
219 sta |= (uint32_t)k_ipc_sta_rerr;
220 }
221 if (f->ferr) {
222 sta |= (uint32_t)k_ipc_sta_ferr;
223 }
224 return sta;
225}

243static uint32_t ipc_fifo_pop(uint32_t ch)
244{
245 ipc_fifo_t* f = &s_ipc.fifo[ch];
246 if (f->count == 0U) {
247 f->rerr = true; /* read-while-empty: latch RERR, data undefined (0). */
248 return 0U;
249 }
250 const uint32_t w = f->word[f->rd];
251 f->rd = (f->rd + 1U) % (uint32_t)k_ipc_fifo_depth;
252 f->count--;
253 s_ipc.pops++;
254 return w;
255}

258static uint64_t ipc_read(uc_engine* uc, uint64_t addr, unsigned size)
259{
260 (void)uc;
261 (void)size;
262 const uint64_t off = addr - (uint64_t)k_ipc_base;
263 uint32_t ch = 0U;
264 uint32_t reg = 0U;
265 if (ipc_decode(off, &ch, &reg)) {
266 if (reg == (uint32_t)k_ipc_reg_sta) {
267 return (uint64_t)ipc_sta_value(ch);
268 }
269 if (reg == (uint32_t)k_ipc_reg_rxd) {
270 return (uint64_t)ipc_fifo_pop(ch);
271 }
272 }
273 /* Semaphores and the NMI window are not modelled: read as 0. */
274 return 0U;
275}

294static void ipc_fifo_push(uint32_t ch, uint32_t value)
295{
296 ipc_fifo_t* f = &s_ipc.fifo[ch];
297 if (f->count >= (uint32_t)k_ipc_fifo_depth) {
298 f->ferr = true; /* write-while-full: latch FERR, drop the word. */
299 return;
300 }
301 f->word[(f->rd + f->count) % (uint32_t)k_ipc_fifo_depth] = value;
302 f->count++;
303 s_ipc.pushes++;
304}

322static void ipc_clr_write(uint32_t ch, uint32_t value)
323{
324 s_ipc.sta[ch] &= ~(value & (uint32_t)k_ipc_sta_irq_mask);
325 ipc_fifo_t* f = &s_ipc.fifo[ch];
326 if ((value & (uint32_t)k_ipc_clr_rst) != 0U) {
327 f->rd = 0U;
328 f->count = 0U; /* FIFO reset also drops RDY + FULL (they are computed). */
329 }
330 if ((value & (uint32_t)k_ipc_clr_rclr) != 0U) {
331 f->rerr = false;
332 }
333 if ((value & (uint32_t)k_ipc_clr_fclr) != 0U) {
334 f->ferr = false;
335 }
336}

339static void ipc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
340{
341 (void)size;
342 const uint64_t off = addr - (uint64_t)k_ipc_base;
343 uint32_t ch = 0U;
344 uint32_t reg = 0U;
345 if (!ipc_decode(off, &ch, &reg)) {
346 return; /* semaphore / NMI window write: benign no-op */
347 }
348 if (reg == (uint32_t)k_ipc_reg_iset) {
349 const uint32_t bits = (uint32_t)value & (uint32_t)k_ipc_sta_irq_mask;
350 if (bits == 0U) {
351 return;
352 }
353 s_ipc.sta[ch] |= bits;
354 s_ipc.sends++;
355 /* Raise the receiving core's ELC event: the core's ICU walks its IELSR
356 * shadow, pends the mapped NVIC line, and the M85 takes it on its next
357 * instruction boundary (cpu1 never drains, so an IPC1 raise is inert). */
358 const uint16_t event =
359 (ch < (uint32_t)k_ipc_unit_split) ? (uint16_t)k_ipc0_irq_event : (uint16_t)k_ipc1_irq_event;
360 board_periph_icu_raise_event(uc, event);
361 s_ipc.wakes++;
362 if (board_periph_trace()) {
363 (void)fprintf(stderr,
364 " IPC : ch%u ISET 0x%02X -> raise 0x%03X\n",
365 ch,
366 bits,
367 (unsigned)event);
368 }
369 return;
370 }
371 if (reg == (uint32_t)k_ipc_reg_txd) {
372 ipc_fifo_push(ch, (uint32_t)value);
373 return;
374 }
375 if (reg == (uint32_t)k_ipc_reg_clr) {
376 ipc_clr_write(ch, (uint32_t)value);
377 }
378}

392static void ipc_report(void)
393{
394 if ((s_ipc.sends == 0U) && (s_ipc.wakes == 0U) && (s_ipc.pushes == 0U) && (s_ipc.pops == 0U)) {
395 return;
396 }
397 (void)fprintf(stderr,
398 " IPC : sends=%u wakes=%u fifo pushes=%u pops=%u\n",
399 s_ipc.sends,
400 s_ipc.wakes,
401 s_ipc.pushes,
402 s_ipc.pops);
