87static int32_t
88internal_quant_round_clamp(float scaled, int32_t zero_point, int32_t qmin, int32_t qmax)
89{
90 float rounded = (scaled >= 0.0F) ? (scaled + s_quant_half) : (scaled - s_quant_half);
91 if (rounded > s_quant_round_cap) {
92 rounded = s_quant_round_cap;
93 }
94 if (rounded < -s_quant_round_cap) {
95 rounded = -s_quant_round_cap;
96 }
97 int32_t q = (int32_t)rounded + zero_point;
98 if (q < qmin) {
99 q = qmin;
100 }
101 if (q > qmax) {
102 q = qmax;
103 }
104 return q;
105}
