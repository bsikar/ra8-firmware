210[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_one_round_trip(void)
211{
212 /* The PSA usage enum is intentionally a bitfield -- combining
213 * encrypt + decrypt yields 0x0C, which is a valid policy mask but
214 * not a declared enumerator. */
215 const ra8_psa_key_attr_t attr = {
216 .type = k_ra8_psa_key_type_aes,
217 .alg = k_ra8_psa_alg_aes_gcm,
218 .usage = (ra8_psa_key_usage_t)k_aes_demo_usage_aead,
219 };
220 ra8_psa_key_t key = nullptr;
221 ra8_err_t err = ra8_psa_key_import(&key, &attr, s_aes_demo_key, (size_t)k_aes_demo_key_bytes);
222 if (err != k_ra8_ok) {
223 return err;
224 }
225
226 uint8_t ct[k_aes_demo_plain_bytes + k_ra8_psa_gcm_tag_len] = {};
227 size_t ct_len = 0U;
228 err = ra8_psa_aead_encrypt(key,
229 k_ra8_psa_alg_aes_gcm,
230 s_aes_demo_nonce,
231 (size_t)k_ra8_psa_gcm_nonce_len,
232 s_aes_demo_aad,
233 (size_t)k_aes_demo_aad_bytes,
234 s_aes_demo_plain,
235 (size_t)k_aes_demo_plain_bytes,
236 ct,
237 sizeof(ct),
238 &ct_len);
239 if (err != k_ra8_ok) {
240 (void)ra8_psa_key_destroy(key);
241 return err;
242 }
243
244 uint8_t recovered[k_aes_demo_plain_bytes] = {};
245 size_t rec_len = 0U;
246 err = ra8_psa_aead_decrypt(key,
247 k_ra8_psa_alg_aes_gcm,
248 s_aes_demo_nonce,
249 (size_t)k_ra8_psa_gcm_nonce_len,
250 s_aes_demo_aad,
251 (size_t)k_aes_demo_aad_bytes,
252 ct,
253 ct_len,
254 recovered,
255 sizeof(recovered),
256 &rec_len);
257 (void)ra8_psa_key_destroy(key);
258 if (err != k_ra8_ok) {
259 return err;
260 }
261 if (rec_len != (size_t)k_aes_demo_plain_bytes) {
262 return k_ra8_err_invalid_size;
263 }
264 if (memcmp(recovered, s_aes_demo_plain, (size_t)k_aes_demo_plain_bytes) != 0) {
265 return k_ra8_err_crc_mismatch;
266 }
267 return k_ra8_ok;
268}

270void main(void)
271{
272 internal_setup_or_halt();
273 ra8_isr_globals_enable();
274
275 while (1) {
276 if (internal_one_round_trip() == k_ra8_ok) {
277 (void)ra8_board_uart_console_write(s_aes_demo_msg_ok,
278 (size_t)(sizeof(s_aes_demo_msg_ok) - 1U));
279 (void)ra8_board_led_toggle(k_ra8_board_led1);
280 } else {
281 (void)ra8_board_uart_console_write(s_aes_demo_msg_fail,
282 (size_t)(sizeof(s_aes_demo_msg_fail) - 1U));
283 (void)ra8_board_led_toggle(k_ra8_board_led2);
284 }
285 ra8_delay_ms(k_aes_demo_period_ms);
286 }
287 internal_panic_halt();
288}
