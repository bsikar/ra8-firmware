221static void prep_mailbox(volatile com33_mailbox_t* mb)
222{
223 mb->magic = 0U;
224 mb->req_magic = 0U;
225 mb->m33_sig = 0U;
226 mb->status = (uint32_t)k_com33_status_running;
227 mb->blob_base = 0U;
228 mb->blob_len = 0U;
229 mb->blob_crc = 0U;
230 mb->chapter_count = 0U;
231 mb->done = 0U;
232 __asm volatile("dsb" ::: "memory");
233 mb->magic = (uint32_t)k_com33_magic;
234
235 /* Stage the source .epub into shared SRAM and POST the job: the M33 compiles
236 * the staged bytes (not a baked-in array), so the same M33 image serves any
237 * book the M85 dispatches -- this is the seam stage c reuses to offload a real
238 * dropped book. req_magic is stamped LAST, behind a dsb, so the M33 sees the
239 * job only once the staged bytes + request fields are settled in memory. */
240 memcpy(com33_epub(), s_m33_parity_epub, (size_t)k_m33_parity_epub_len);
241 mb->epub_base = (uint32_t)(uintptr_t)k_com33_epub_addr;
242 mb->epub_len = (uint32_t)k_m33_parity_epub_len;
243 mb->out_base = (uint32_t)(uintptr_t)k_com33_blob_addr;
244 mb->out_cap = (uint32_t)k_com33_blob_cap;
245 __asm volatile("dsb" ::: "memory");
246 mb->req_magic = (uint32_t)k_com33_req_magic;
247 __asm volatile("dsb" ::: "memory");
248}

267static bool wait_for_m33_sig(volatile com33_mailbox_t* mb)
268{
269 RA8_LOOP_BOUND(k_m85_sig_poll_budget);
270 for (uint32_t i = 0U; i < (uint32_t)k_m85_sig_poll_budget; i++) {
271 if (mb->m33_sig == (uint32_t)k_com33_m33_sig) {
272 return true;
273 }
274 }
275 return false;

302static bool wait_for_done(volatile com33_mailbox_t* mb)
303{
304 RA8_LOOP_BOUND(k_m85_done_poll_budget);
305 for (uint32_t i = 0U; i < (uint32_t)k_m85_done_poll_budget; i++) {
306 if (mb->done == 1U) {
307 return true;
308 }
309 /* IPC-interrupt idle: sleep until the M33 finishes the compile and pokes
310 * IPC0ISET0, which raises the armed IPC0 receive IRQ and wakes the core.
311 * The M33 orders `done` before that poke (a `dsb`), so the next iteration
312 * observes a settled `done` and returns. The bounded count is the NASA
313 * Rule 2 backstop: if the wake is ever missed the loop still terminates and
314 * the caller reports a timeout instead of hanging. WFI here also means a
315 * real import keeps the M85 off the bus while the slow core grinds. */
316 __asm volatile("wfi" ::: "memory");
317 }
318 return false;
319}

344static bool verify_blob(volatile com33_mailbox_t* mb)
345{
346 if (mb == nullptr) {
347 return false;
348 }
349 if (mb->status != (uint32_t)k_com33_status_ok) {
350 return false;
351 }
352 if (mb->blob_base != (uint32_t)k_com33_blob_addr) {
353 return false;
354 }
355 const uint32_t len = mb->blob_len;
356 if (len == 0U) {
357 return false;
358 }
359 if (len > (uint32_t)k_com33_blob_cap) {
360 return false;
361 }
362 const void* base = (const void*)(uintptr_t)mb->blob_base;
363 if (ra8_book_validate(base, (size_t)len) != k_ra8_ok) {
364 return false;
365 }
366 const ra8_book_header_t* hdr = ra8_book_header(base);
367 if (hdr->crc32 != mb->blob_crc) {
368 return false;
369 }
370 if (hdr->chapter_count != mb->chapter_count) {
371 return false;
372 }
373 /* #149 a1: byte-identity on the secondary core. The M33-produced blob must
374 * equal the desktop/M85 golden (s_m33_parity_golden) byte for byte -- the same
375 * acceptance the M85 parity gate proves, now on the slow core. */
376 if (len != (uint32_t)k_m33_parity_golden_len) {
377 return false;
378 }
379 if (memcmp(base, s_m33_parity_golden, (size_t)len) != 0) {
380 return false;
381 }
382 return true;
383}

424int main(void)
425{
426 ra8_log_init();
427 ra8_log_info("M85", "==== RA8D2 compile_on_m33 demo (#149b emitter offload) ====");
428 ra8_log_info("M85", "Cortex-M85 primary core online");
429 ra8_log_info("M85", "shared mailbox + output blob in SRAM at 0x22100000");
430
431 volatile com33_mailbox_t* mb = com33_mailbox();
432 prep_mailbox(mb);
433
434 /* Arm the IPC0 receive IRQ before releasing the M33 so the M85 can WFI-idle
435 * until the M33's compile-done poke wakes it. A setup failure is non-fatal:
436 * wait_for_done still bounds-polls the mailbox `done` flag (its WFI then just
437 * sleeps until the next interrupt, harmless with the flag re-checked). */
438 if (!arm_ipc_wake()) {
439 ra8_log_info("M85", "IPC wake arm failed -- falling back to bounded poll");
440 } else {
441 ra8_log_info("M85", "IPC0 receive IRQ armed -- M85 will WFI-idle for the M33");
442 }
443
444 ra8_log_info("M85", "releasing Cortex-M33 to run the RABOOK1 emitter ...");
445 const ra8_err_t err = ra8_cpu1_release(&g_ra8_ls_cpu1_mram_start, &g_ra8_ls_cpu1_stack_top);
446 ra8_log_info_val("M85", "ra8_cpu1_release rc (0 = ok)", (uint32_t)err);
447 if (err != k_ra8_ok) {
448 ra8_log_info("M85", "release FAILED -- halting");
449 park_forever();
450 }
451
452 if (wait_for_m33_sig(mb)) {
453 ra8_log_info("M85", "M33 emitter is alive");
454 } else {
455 ra8_log_info("M85", "M33 signature not seen -- did it boot?");
456 }
457
458 ra8_log_info("M85", "M85 yielding -- M33 is compiling the book ...");
459 if (!wait_for_done(mb)) {
460 ra8_log_info("M85", "M33 done flag not seen -- timed out");
461 park_forever();
462 }
463
464 /* Consumer-side barrier: order the blob/field loads AFTER observing `done`,
465 * pairing with the M33's pre-`done` `dsb` so the M85 reads a settled blob. */
466 __asm volatile("dmb" ::: "memory");
467 ra8_log_info_val("M85", "M33 reported blob length (bytes)", mb->blob_len);
468
469 if (verify_blob(mb)) {
470 ra8_log_info_val("M85", "ra8_book_validate OK; chapters in blob", mb->chapter_count);
471 ra8_log_info("M85", "compile_on_m33 PASS");
472 park_forever();
473 }
474
475 /* The M33-produced blob did NOT equal the desktop/M85 golden byte for byte (or
476 * failed validation). Trap so the byte-identity is GATED: ra8_emulator's smoke
477 * detects the BKPT and fails the run. A silent park here would read as a clean
478 * boot and hide a compile regression (e.g. a future ra8_emulator long-shift seam
479 * miss on a DEFLATE'd fixture). On real silicon with no debugger the BKPT
480 * escalates to a HardFault -- a loud, visible failure for the demo. */
481 ra8_log_info_val("M85", "compile_on_m33 FAIL -- status", mb->status);
482 __asm volatile("bkpt #0");
483 park_forever();
484}
