444[[nodiscard]] static ra8_err_t blc_print_hex(uint32_t value)
445{
446 uint8_t out[k_blc_hex_chars_u32] = {};
447 for (uint8_t i = 0U; i < (uint8_t)k_blc_hex_chars_u32; i++) {
448 const uint8_t shift = (uint8_t)(((uint8_t)k_blc_hex_chars_u32 - 1U - i) * k_blc_nibble_bits);
449 out[i] = blc_nibble_to_hex((value >> shift) & k_blc_nibble_mask);
450 }
451 return ra8_sci_write_polling((uint8_t)k_blc_sci_channel, out, (uint32_t)k_blc_hex_chars_u32);
452}

481static ra8_dfu_action_t blc_decide(ra8_dfu_slot_t* out_target)
482{
483 if (out_target == nullptr) {
484 return k_ra8_dfu_action_dfu; /* safe fallback: never jump on a bad caller */
485 }
486 const bool trig = (g_dfu_trigger == (uint32_t)k_ra8_dfu_trigger_magic);
487 g_dfu_trigger = 0U; /* one-shot: consume the request */
488
489 const bool a_valid = ra8_dfu_slot_valid(k_ra8_dfu_slot_a);
490 const bool b_valid = ra8_dfu_slot_valid(k_ra8_dfu_slot_b);
491 uint32_t a_seq = 0U;
492 uint32_t b_seq = 0U;
493 /* A failed read leaves seq at 0 (treated as oldest) -- a safe default that
494 * never wins the higher-seq tiebreak, so the discarded return is intentional. */
495 (void)ra8_dfu_slot_seq(k_ra8_dfu_slot_a, &a_seq);
496 (void)ra8_dfu_slot_seq(k_ra8_dfu_slot_b, &b_seq);
497
498 const ra8_dfu_slot_t active = ra8_dfu_select_slot(a_valid, a_seq, b_valid, b_seq);
499 *out_target = (active == k_ra8_dfu_slot_none) ? k_ra8_dfu_slot_a : ra8_dfu_other_slot(active);
500
501 return ra8_dfu_boot_decide(trig, a_valid, a_seq, b_valid, b_seq);
502}

549[[noreturn]] static void blc_system_reset(void)
550{
551 __asm__ volatile("dsb 0xF" ::: "memory");
552 /* HUM Ch 6.1 Table 6.1 p 244 "Software reset (AIRCR.SYSRESETREQ)"; the
553 * Cortex-M85 AIRCR itself is defined in ARMv8-M ARM B3.2.6 at 0xE000ED0C. */
554 *(volatile uint32_t*)(uintptr_t)k_blc_scb_aircr_addr = (uint32_t)k_blc_aircr_sysreset;
555 __asm__ volatile("dsb 0xF" ::: "memory");
556 /* NASA Rule 2 exemption: spin until the reset lands (a few cycles); the
557 * function is [[noreturn]] and never iterates unboundedly in practice. */
558 while (1) {
559 __asm__ volatile("wfi");
560 }
561}

574static VOID blc_device_worker(ULONG arg)
575{
576 (void)arg;
577
578 const ra8_err_t e = ra8_dfu_device_start(k_ra8_usb_speed_fs,
579 s_usbx_pool,
580 (uint32_t)sizeof(s_usbx_pool),
581 s_device_framework,
582 (uint32_t)sizeof(s_device_framework),
583 s_string_framework,
584 (uint32_t)sizeof(s_string_framework),
585 s_language_id_framework,
586 (uint32_t)sizeof(s_language_id_framework));
587 if (e != k_ra8_ok) {
588 s_dbg_dev_err = (uint32_t)e;
589 (void)blc_print("dfu-bootloader: FAIL device bring-up\r\n");
590 return;
591 }
592 s_dbg_dev_step = 4U;
593 (void)blc_print("dfu-bootloader: DFU device up on USB-FS, awaiting DFU_DNLOAD...\r\n");
594
595 /* NASA Rule 2 exemption: deliberate RTOS service loop. It exits only via
596 * blc_system_reset() once an image commits (that call does not return). */
597 while (1) {
598 (void)ra8_dfu_device_worker_step();
599 g_blc_dfu_tick += 1U; /* HIL liveness: proves the DFU service loop iterates */
600 if (ra8_dfu_device_committed() && (ra8_dfu_device_last_error() == k_ra8_ok)) {
601 (void)blc_print("dfu-bootloader: image committed -- resetting into new slot\r\n");
602 blc_system_reset();
603 }
604 tx_thread_sleep((ULONG)k_blc_idle_ticks);
605 }
606}

715{
716 uint32_t cpuclk0_hz = 0U;
717 uint32_t pclka_hz = 0U;
718 if (ra8_cgc_init() != k_ra8_ok) {
719 blc_panic_halt();
720 }
721 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
722 blc_panic_halt();
723 }
724 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
725 blc_panic_halt();
726 }
727 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
728 blc_panic_halt();
729 }
730 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
731 blc_panic_halt();
732 }
733 if (ra8_pfs_route_peripheral(k_blc_pin_sci_tx, k_ra8_psel_sci_async, "blc.txd8") != k_ra8_ok) {
734 blc_panic_halt();
735 }
736 if (ra8_pfs_route_peripheral(k_blc_pin_sci_rx, k_ra8_psel_sci_async, "blc.rxd8") != k_ra8_ok) {
737 blc_panic_halt();
738 }
739 const ra8_sci_cfg_t sci_cfg = {
740 .baud = k_blc_baud,
741 .data_bits = k_ra8_sci_data_8,
742 .parity = k_ra8_sci_parity_none,
743 .stop_bits = k_ra8_sci_stop_1,
744 .pclk_hz = pclka_hz,
745 };
746 if (ra8_sci_init((uint8_t)k_blc_sci_channel, &sci_cfg) != k_ra8_ok) {
747 blc_panic_halt();
748 }
749 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
750 blc_panic_halt();
751 }
752#ifdef RA8_ENABLE_ROOT_OF_TRUST
753 /* Bring up the tf-psa-crypto backend for the root-of-trust launch gate: a
754 * static buffer-alloc arena (no malloc) then psa_crypto_init. A failure here
755 * means the bootloader cannot authenticate any slot, so halt rather than boot
756 * an unverified image. */
757 mbedtls_memory_buffer_alloc_init(s_blc_psa_heap, sizeof(s_blc_psa_heap));
758 if (ra8_psa_crypto_init() != k_ra8_ok) {
759 blc_panic_halt();
760 }
761#endif
762 blc_route_usb_or_halt();
763}

777int32_t main(void)
778{
779 blc_setup_or_halt();
780 ra8_isr_globals_enable();
781
782 (void)blc_print("\r\nra8d2 dfu-bootloader (immutable, 128K @ 0x02000000)\r\n");
783
784 ra8_dfu_slot_t target = k_ra8_dfu_slot_a;
785 const ra8_dfu_action_t action = blc_decide(&target);
786 s_dbg_action = (uint32_t)action;
787 s_dbg_target = (uint32_t)target;
788
789 if (action == k_ra8_dfu_action_jump_a) {
790 (void)blc_print("dfu-bootloader: Slot A -> copy-to-run @0x");
791 (void)blc_print_hex((uint32_t)k_ra8_dfu_run_base);
792 (void)blc_print("\r\n");
793 blc_boot_slot(k_ra8_dfu_slot_a); /* returns only if the run target is invalid */
794 }
795 if (action == k_ra8_dfu_action_jump_b) {
796 (void)blc_print("dfu-bootloader: Slot B -> copy-to-run @0x");
797 (void)blc_print_hex((uint32_t)k_ra8_dfu_run_base);
798 (void)blc_print("\r\n");
799 blc_boot_slot(k_ra8_dfu_slot_b); /* returns only if the run target is invalid */
800 }
801
802 /* k_ra8_dfu_action_dfu (or a slot whose run target failed validation): no
803 * bootable slot -- accept a DFU update into the inactive slot over USB-FS. */
804 (void)blc_print("dfu-bootloader: no bootable slot -- entering DFU\r\n");
805#ifndef RA8_OFF_TARGET
806 ra8_dfu_device_set_target(target);
807 tx_kernel_enter();
808#endif
809
810 blc_panic_halt();
811 return 0;
812}
