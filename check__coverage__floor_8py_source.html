1#!/usr/bin/env python3
2"""check_coverage_floor.py -- per-file line-coverage FLOOR gate (no allowlist).
3
4Per CLAUDE.md "IEC 61508 SIL 3 / DO-178C Level B" and the project's
5"no grandfather" coverage policy: EVERY first-party translation unit must
6carry at least `FLOOR_PCT` line coverage on its own. The aggregate gate
7(`scripts/checks/coverage.sh --gate`, gcovr --fail-under-line 90) can be satisfied
8while an individual file rots far below the bar, because well-covered files
9average it up. This gate closes that hole: it fails CI if ANY first-party
10file drops below the floor, and it deliberately has NO allowlist / no
11per-file exemption table -- if a file cannot reach the floor it must be
12fixed at the root (a real test, or a documented GCOVR_EXCL on a line no host
13input can reach), never grandfathered.
14
15Input: the gcovr JSON `build/coverage/coverage/coverage.json` produced by
16`scripts/checks/coverage.sh` (its `[4/4] Running gcovr` step writes `--json`). Run
17that first, then this. Scope mirrors the aggregate gate: files under `libs/`
18or `src/`, excluding vendored SOUP (`libs/third_party/`) and generated font
19tables (`libs/ra8_fonts/`), which the coding-standards scope explicitly exempts.
20
21Exit 0 if every in-scope file is >= FLOOR_PCT, else exit 1 with the offenders.
22
23Copyright (c) 2026 Brighton Sikarskie
24SPDX-License-Identifier: MIT
25"""
26
27from __future__ import annotations

29import json
30import sys
31from pathlib import Path

33FLOOR_PCT = 90.0
34"""Per-file line-coverage floor, in percent. Matches the aggregate line gate
35(gcovr --fail-under-line 90) but applied to EACH first-party file, not the
36mean. No allowlist: a file below this is a hard CI failure."""
37
38REPO_ROOT = Path(__file__).resolve().parents[2]
39COVERAGE_JSON = REPO_ROOT / "build" / "coverage" / "coverage" / "coverage.json"
40
41IN_SCOPE_PREFIXES = ("libs/", "src/")
42"""First-party source roots held to the floor."""
43
44OUT_OF_SCOPE_PREFIXES = ("libs/third_party/", "libs/ra8_fonts/")
45"""Vendored SOUP and generated font tables -- exempt from first-party rules
46per the CLAUDE.md coding-standards scope, so exempt from the floor too."""

49def normalize(path: str) -> str:
50 """Normalise a gcovr ``file`` field to a repo-root-relative POSIX path.
51
52 The field may arrive absolute or already relative, so both are handled.
53
54 The absolute-path split marker is derived from the checkout directory
55 basename (`REPO_ROOT.name`, itself resolved from this file's location)
56 rather than a hardcoded project name, so the gate strips the prefix
57 correctly from any clone regardless of what the repo directory is named.
58 """
59 p = path.replace("\\", "/")
60 marker = "/" + REPO_ROOT.name + "/"
61 if marker in p:
62 p = p.split(marker, 1)[1]
63 return p.lstrip("./")
64
65
66def in_scope(rel: str) -> bool:
67 """True if `rel` is a first-party file subject to the coverage floor."""
68 if not rel.startswith(IN_SCOPE_PREFIXES):
69 return False
70 return not rel.startswith(OUT_OF_SCOPE_PREFIXES)
71
72
73def file_line_pct(entry: dict) -> tuple[int, int]:
74 """Return (covered, total) *countable* lines for one gcovr file entry.
75
76 Lines carrying `gcovr/excluded: true` (a `GCOVR_EXCL` marker on a line no
77 host input can reach) are dropped from BOTH the numerator and denominator,
78 exactly as gcovr's own `line_total` does -- so an honest exclusion neither
79 helps nor hurts, it simply is not counted. This makes the floor agree with
80 the aggregate gate's per-file line-rate rather than a naive covered/len.
81 """
82 lines = [ln for ln in entry.get("lines", []) if not ln.get("gcovr/excluded", False)]
83 total = len(lines)
84 covered = sum(1 for ln in lines if ln.get("count", 0) > 0)
85 return covered, total

88def main() -> int:
89 """Load the gcovr JSON and fail if any in-scope file is below the floor."""
90 if not COVERAGE_JSON.is_file():
91 print(
92 f"check_coverage_floor.py: ERROR -- {COVERAGE_JSON} not found; "
93 f"run `bash scripts/checks/coverage.sh` first."
94 )
95 return 1
96
97 try:
98 data = json.loads(COVERAGE_JSON.read_text(encoding="utf-8"))
99 except (OSError, json.JSONDecodeError) as exc:
100 print(f"check_coverage_floor.py: ERROR -- cannot read coverage JSON: {exc}")
101 return 1
102
103 files = data.get("files", [])
104 if not files:
105 print("check_coverage_floor.py: ERROR -- coverage JSON has no files.")
106 return 1
107
108 offenders: list[tuple[float, str, int, int]] = []
109 checked = 0
110 for entry in files:
111 rel = normalize(entry.get("file", ""))
112 if not in_scope(rel):
113 continue
114 covered, total = file_line_pct(entry)
115 if total == 0:
116 continue
117 checked += 1
118 pct = 100.0 * covered / total
119 if pct < FLOOR_PCT:
120 offenders.append((pct, rel, covered, total))
121
122 if checked == 0:
123 print(
124 "check_coverage_floor.py: ERROR -- no in-scope files matched; "
125 "check the JSON path / scope."
126 )
127 return 1
128
129 if offenders:
130 offenders.sort()
131 print(
132 f"check_coverage_floor.py: {len(offenders)} first-party file(s) "
133 f"below the {FLOOR_PCT:.0f}% line-coverage floor (NO allowlist):"
134 )
135 print(" cover covered/total file")
136 for pct, rel, covered, total in offenders:
137 print(f" {pct:5.1f}% {covered:5d}/{total:<5d} {rel}")
138 print(
139 "Fix each at the root -- a real test, or a documented GCOVR_EXCL on "
140 "a line no host input can reach. Do NOT add an allowlist."
141 )
142 return 1
143
144 print(
145 f"check_coverage_floor.py: PASS -- all {checked} first-party file(s) "
146 f">= {FLOOR_PCT:.0f}% line coverage."
147 )
148 return 0
