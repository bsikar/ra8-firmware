255static uint16_t internal_pack_wdtcr(const ra8_wdt_cfg_t* cfg)
256{
257 /* HUM Ch 27.2.2 "WDTCR : WDT Control Register", p 1258 */
258 const uint16_t tops = (uint16_t)((uint16_t)cfg->timeout & k_ra8_wdt_mask_tops);
259 const uint16_t cks = (uint16_t)((uint16_t)cfg->clock_div & k_ra8_wdt_mask_cks);
260 const uint16_t rpes = (uint16_t)((uint16_t)cfg->window_end & k_ra8_wdt_mask_rpes);
261 const uint16_t rpss = (uint16_t)((uint16_t)cfg->window_start & k_ra8_wdt_mask_rpss);
262
263 uint16_t word = 0U;
264 word |= (uint16_t)(tops << k_ra8_wdt_shift_tops);
265 word |= (uint16_t)(cks << k_ra8_wdt_shift_cks);
266 word |= (uint16_t)(rpes << k_ra8_wdt_shift_rpes);
267 word |= (uint16_t)(rpss << k_ra8_wdt_shift_rpss);
268 return word;
269}

295[[nodiscard]] ra8_err_t ra8_wdt_init(const ra8_wdt_cfg_t* cfg)
296{
297 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
298 if (!internal_clock_div_is_valid(cfg->clock_div)) {
299 return k_ra8_err_invalid_arg;
300 }
301 if (!internal_timeout_sel_is_valid(cfg->timeout)) {
302 return k_ra8_err_invalid_arg;
303 }
304
305 volatile r_wdt_regs_t* reg = ra8_wdt();
306
307 /* HUM Ch 27.2.2 "WDTCR : WDT Control Register", p 1258 -- one
308 * 16-bit transaction commits the timeout / divider / window. */
309 reg->WDTCR = internal_pack_wdtcr(cfg);
310
311 /* HUM Ch 27.2.4 "WDTRCR : WDT Reset Control Register", p 1262 --
312 * RSTIRQS selects internal reset (1) vs NMI / IRQ (0) on
313 * underflow or refresh-error. */
314 reg->WDTRCR =
315 (uint8_t)((cfg->on_expiry == k_ra8_wdt_on_expiry_reset) ? k_ra8_wdt_rcr_rstirqs : 0U);
316
317 /* HUM Ch 27.2.5 "WDTCSTPR : WDT Count Stop Control Register",
318 * p 1262 -- SLCSTP halts the counter on Sleep / Deep Sleep. */
319 reg->WDTCSTPR =
320 (uint8_t)((cfg->stop_in_sleep == k_ra8_wdt_sleep_stop_count) ? k_ra8_wdt_cstpr_slcstp : 0U);
321
322 /* HUM Ch 27.2.1 "WDTRR : WDT Refresh Register", p 1257 -- the
323 * 0x00 / 0xFF unlock sequence is what arms the down-counter in
324 * register-start mode, and is harmless in auto-start mode. */
325 ra8_wdt_refresh();
326
327 ra8_log_info(s_tag, "wdt_init armed");
328 return k_ra8_ok;
329}
330
331[[nodiscard]] ra8_err_t ra8_wdt_deinit(void)
332{
333 volatile r_wdt_regs_t* reg = ra8_wdt();
334 /* HUM Ch 27.2.5 "WDTCSTPR : WDT Count Stop Control Register",
335 * p 1262 -- the strongest "off" the driver can request is to halt
336 * the counter while the CPU is asleep. The peripheral itself
337 * cannot be disarmed once started. */
338 reg->WDTCSTPR = k_ra8_wdt_cstpr_slcstp;
339 internal_subs_clear_all();
340 ra8_log_info(s_tag, "wdt_deinit (sleep-stop set)");
341 return k_ra8_ok;
342}

378[[nodiscard]] ra8_err_t ra8_wdt_clear_status(void)
379{
380 volatile r_wdt_regs_t* reg = ra8_wdt();
381 /* HUM Ch 27.2.3 "WDTSR : WDT Status Register", p 1260 -- UNDFF /
382 * REFEF are write-0-to-clear; writing 1 has no effect. We mask
383 * out the flag bits and write the result back. */
384 reg->WDTSR = (uint16_t)(reg->WDTSR & (uint16_t)~k_ra8_wdt_status_all);
385 return k_ra8_ok;
386}

388[[nodiscard]] ra8_err_t ra8_wdt_clear_status_blocking(uint16_t mask)
389{
390 if ((mask & ~k_ra8_wdt_status_all) != 0U) {
391 return k_ra8_err_invalid_arg;
392 }
393 if (mask == k_ra8_wdt_status_none) {
394 return k_ra8_ok;
395 }
396
397 volatile r_wdt_regs_t* reg = ra8_wdt();
398
399 /* HUM Ch 27.2.3 "WDTSR : WDT Status Register", p 1261 -- W0C
400 * sequence: write the inverse of the bits we want to clear back
401 * into WDTSR, then poll until the targeted bits read 0. The
402 * (N + 1) PCLKB cycle latency is bounded by the divider; we
403 * cap our polls at k_ra8_wdt_clear_max_polls (16384). */
404 for (uint32_t poll = 0U; poll < k_ra8_wdt_clear_max_polls; ++poll) { /* GCOVR_EXCL_BR_LINE */
405 /* Build a write value in which only the bits NOT in `mask`
406 * remain set; those are the bits we want to leave untouched.
407 * Bits inside `mask` are written 0 to clear. */
408 const uint16_t write_val = (uint16_t)(reg->WDTSR & (uint16_t)~mask);
409 reg->WDTSR = write_val;
410
411 if (((uint16_t)reg->WDTSR & mask) == 0U) { /* GCOVR_EXCL_BR_LINE */
412 return k_ra8_ok;
413 }
414 }
415
416 ra8_log_error(s_tag, "wdt_clear_status_blocking timed out");
417 return k_ra8_err_hw_timeout;
418}

521[[nodiscard]] ra8_err_t ra8_wdt_subscribe(ra8_wdt_event_fn_t fn, void* ctx, uint8_t* out_slot)
522{
523 RA8_CHECK_NULL_PTR(fn, s_tag, "fn must not be nullptr");
524
525 /* Walk the table starting at the first non-legacy slot; slot 0 is
526 * reserved for ``ra8_wdt_attach_handler``. */
527 for (uint8_t i = 1U; i < k_ra8_wdt_max_subs; ++i) {
528 if (s_wdt_subs[i].fn == nullptr) {
529 s_wdt_subs[i].fn = fn;
530 s_wdt_subs[i].ctx = ctx;
531 if (out_slot != nullptr) {
532 *out_slot = i;
533 }
534 return k_ra8_ok;
535 }
536 }
537
538 ra8_log_error(s_tag, "wdt_subscribe table full");
539 return k_ra8_err_no_mem;
540}

566RA8_ISR_SAFE
567void ra8_wdt_dispatch(void)
568{
569 volatile r_wdt_regs_t* reg = ra8_wdt();
570 /* HUM Ch 27.2.3 "WDTSR : WDT Status Register", p 1260 -- snapshot
571 * the flags before clearing, then hand the latched mask to every
572 * registered subscriber. */
573 const uint16_t mask = (uint16_t)(reg->WDTSR & k_ra8_wdt_status_all);
574 reg->WDTSR = (uint16_t)(reg->WDTSR & (uint16_t)~mask);
575
576 for (uint8_t i = 0U; i < k_ra8_wdt_max_subs; ++i) {
577 const ra8_wdt_event_fn_t fn = s_wdt_subs[i].fn;
578 void* const ctx = s_wdt_subs[i].ctx;
579 if (fn != nullptr) {
580 fn(ctx, mask);
581 }
582 }
583}

743static ra8_err_t internal_read_wdt1_word(uint32_t* out_word)
744{
745 /* Single-exit (MISRA C 2012 Rule 15.5): the three reads chain on ``e``
746 * rather than returning early, so a failure at any word falls straight
747 * through to the one return without touching ``*out_word``. */
748 ra8_err_t e = k_ra8_err_null_ptr;
749 if (out_word != nullptr) {
750 uint32_t sel = 0U;
751 uint32_t sec = 0U;
752 uint32_t nonsec = 0U;
753
754 e = s_ofs_reader(k_ra8_ofs3_sel_addr, &sel);
755 if (e == k_ra8_ok) {
756 e = s_ofs_reader(k_ra8_ofs3_sec_addr, &sec);
757 }
758 if (e == k_ra8_ok) {
759 e = s_ofs_reader(k_ra8_ofs3_addr, &nonsec);
760 }
761 if (e == k_ra8_ok) {
762 if (internal_ofs3_sel_is_legal(sel)) {
763 const uint32_t mux = sel & (uint32_t)k_ra8_wdt_ofs_field_mask;
764 *out_word = (nonsec & mux) | (sec & ~mux);
765 } else {
766 ra8_log_error(s_tag, "OFS3_SEL holds a prohibited mixed encoding");
767 e = k_ra8_err_invalid_state;
768 }
769 }
770 }
771 return e;
772}

794static void internal_decode_ofs_word(uint32_t ofsm, ra8_wdt_ofs_decoded_t* out)
795{
796 const uint8_t strt = (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_strt) & k_ra8_wdt_ofs_mask_strt);
797 const uint8_t tops = (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_tops) & k_ra8_wdt_ofs_mask_tops);
798 const uint8_t cks = (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_cks) & k_ra8_wdt_ofs_mask_cks);
799 const uint8_t rpes = (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_rpes) & k_ra8_wdt_ofs_mask_rpes);
800 const uint8_t rpss = (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_rpss) & k_ra8_wdt_ofs_mask_rpss);
801 const uint8_t rstirqs =
802 (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_rstirqs) & k_ra8_wdt_ofs_mask_rstirqs);
803 const uint8_t stpctl =
804 (uint8_t)((ofsm >> k_ra8_wdt_ofs_shift_stpctl) & k_ra8_wdt_ofs_mask_stpctl);
805
806 out->cfg.timeout = (ra8_wdt_timeout_sel_t)tops;
807 out->cfg.clock_div = (ra8_wdt_clock_div_t)cks;
808 out->cfg.window_start = (ra8_wdt_window_start_t)rpss;
809 out->cfg.window_end = (ra8_wdt_window_end_t)rpes;
810 out->cfg.on_expiry = (ra8_wdt_reset_ctrl_t)rstirqs;
811 out->cfg.stop_in_sleep = (ra8_wdt_stop_ctrl_t)stpctl;
812 out->start_mode = (ra8_wdt_ofs_strt_t)strt;
813 out->auto_start = (out->start_mode == k_ra8_wdt_ofs_strt_auto);
814}

816[[nodiscard]] ra8_err_t ra8_wdt_ofs_get(ra8_wdt_instance_t which, ra8_wdt_ofs_decoded_t* out)
817{
818 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
819 if ((uint8_t)which >= k_ra8_wdt_instance_count) {
820 return k_ra8_err_invalid_arg;
821 }
822
823 /* HUM Ch 27.3.8 Table 27.5 "Association between Option Function Select
824 * Register m (OFSm) and the WDT registers" p 1269 -- in auto-start mode the
825 * seven WDT fields come from OFSm rather than from WDTCR/WDTRCR/WDTCSTPR.
826 *
827 * WDT0 takes them from OFS0 (HUM Ch 7.2.1 p 280), a single secure-region
828 * word. WDT1 takes them from OFS3/OFS3_SEC selected field-by-field by
829 * OFS3_SEL (HUM Ch 7.2.6 p 287, Ch 7.2.7 p 289), so it needs three reads.
830 *
831 * The WDT driver never writes OFSm; only ``ra8_ofs.c`` emits the
832 * option-setting sections. Reads go through the hook so tests and
833 * non-secure callers can supply their own transport. */
834 uint32_t ofsm = 0U;
835 const ra8_err_t e =
836 (which == k_ra8_wdt0) ? s_ofs_reader(k_ra8_ofs0_addr, &ofsm) : internal_read_wdt1_word(&ofsm);
837 if (e != k_ra8_ok) {
838 return e;
839 }
840
841 internal_decode_ofs_word(ofsm, out);
842 return k_ra8_ok;
843}
