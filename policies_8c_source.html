128static int cb_lru_init(cb_cache_t* c)
129{
130 cb_lru_t* l = (cb_lru_t*)calloc(1U, sizeof(cb_lru_t));
131 /* cppcheck-suppress memleak ; false positive: cppcheck 2.13 does not model
132 * the C23 nullptr keyword, so it cannot see l is NULL on this path. */
133 if (l == nullptr) {
134 return 1;
135 }
136 l->prev = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
137 l->next = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
138 l->head = -1;
139 l->tail = -1;
140 if ((l->prev == nullptr) || (l->next == nullptr)) {
141 /* The replay harness never deinits a policy whose init failed, so a
142 * partial allocation must be released here, not left on policy_data. */
143 free(l->prev);
144 free(l->next);
145 free(l);
146 return 1;
147 }
148 c->policy_data = l;
149 return 0;
150}
151static void cb_lru_deinit(cb_cache_t* c)
152{
153 cb_lru_t* l = (cb_lru_t*)c->policy_data;
154 if (l != nullptr) {
155 free(l->prev);
156 free(l->next);
157 free(l);
158 }
159}

161static void cb_lru_unlink(cb_lru_t* l, int32_t f)
162{
163 if (l->prev[f] != -1) {
164 l->next[l->prev[f]] = l->next[f];
165 } else if (l->head == f) {
166 l->head = l->next[f];
167 }
168 if (l->next[f] != -1) {
169 l->prev[l->next[f]] = l->prev[f];
170 } else if (l->tail == f) {
171 l->tail = l->prev[f];
172 }
173}

175static void cb_lru_to_head(cb_lru_t* l, int32_t f)
176{
177 l->prev[f] = -1;
178 l->next[f] = l->head;
179 if (l->head != -1) {
180 l->prev[l->head] = f;
181 }
182 l->head = f;
183 if (l->tail == -1) {
184 l->tail = f;
185 }
186}
187static void cb_lru_touch(cb_cache_t* c, uint32_t frame)
188{
189 cb_lru_t* l = (cb_lru_t*)c->policy_data;
190 cb_lru_unlink(l, (int32_t)frame);
191 cb_lru_to_head(l, (int32_t)frame);
192}
193static void cb_lru_insert(cb_cache_t* c, uint32_t frame)
194{
195 cb_lru_t* l = (cb_lru_t*)c->policy_data;
196 cb_lru_to_head(l, (int32_t)frame);
197}

233static uint32_t cb_clock_victim(cb_cache_t* c, uint32_t* scanned)
234{
235 uint32_t* hand = (uint32_t*)c->policy_data;
236 uint32_t seen = 0U;
237 for (;;) {
238 const uint32_t f = *hand;
239 *hand = (*hand + 1U) % c->capacity;
240 seen++;
241 if (c->frames[f].meta[0] != 0U) {
242 c->frames[f].meta[0] = 0U; /* second chance */
243 } else {
244 *scanned = seen;
245 return f;
246 }
247 }
