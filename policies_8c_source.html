265static int cb_lru_init(cb_cache_t* c)
266{
267 cb_lru_t* l = (cb_lru_t*)calloc(1U, sizeof(cb_lru_t));
268 /* cppcheck-suppress memleak ; false positive: cppcheck 2.13 does not model
269 * the C23 nullptr keyword, so it cannot see l is NULL on this path. */
270 if (l == nullptr) {
271 return 1;
272 }
273 l->prev = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
274 l->next = (int32_t*)malloc((size_t)c->capacity * sizeof(int32_t));
275 l->head = -1;
276 l->tail = -1;
277 if ((l->prev == nullptr) || (l->next == nullptr)) {
278 /* The replay harness never deinits a policy whose init failed, so a
279 * partial allocation must be released here, not left on policy_data. */
280 free(l->prev);
281 free(l->next);
282 free(l);
283 return 1;
284 }
285 c->policy_data = l;
286 return 0;
287}

331static void cb_lru_unlink(cb_lru_t* l, int32_t f)
332{
333 if (l->prev[f] != -1) {
334 l->next[l->prev[f]] = l->next[f];
335 } else if (l->head == f) {
336 l->head = l->next[f];
337 }
338 if (l->next[f] != -1) {
339 l->prev[l->next[f]] = l->prev[f];
340 } else if (l->tail == f) {
341 l->tail = l->prev[f];
342 }
343}

362static void cb_lru_to_head(cb_lru_t* l, int32_t f)
363{
364 l->prev[f] = -1;
365 l->next[f] = l->head;
366 if (l->head != -1) {
367 l->prev[l->head] = f;
368 }
369 l->head = f;
370 if (l->tail == -1) {
371 l->tail = f;
372 }
373}

553static uint32_t cb_clock_victim(cb_cache_t* c, uint32_t* scanned)
554{
555 uint32_t* hand = (uint32_t*)c->policy_data;
556 uint32_t seen = 0U;
557 for (;;) {
558 const uint32_t f = *hand;
559 *hand = (*hand + 1U) % c->capacity;
560 seen++;
561 if (c->frames[f].meta[0] != 0U) {
562 c->frames[f].meta[0] = 0U; /* second chance */
563 } else {
564 *scanned = seen;
565 return f;
566 }
