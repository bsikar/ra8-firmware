244RA8_INTERNAL
245static void internal_apply_class_setup(const ra8_usb_setup_t* setup)
246{
247 switch (setup->b_request) {
248 case k_ra8_phid_req_set_idle: {
249 const uint8_t duration =
250 (uint8_t)((setup->w_value >> k_ra8_phid_shift_byte1) & k_ra8_phid_mask_byte);
251 s_state.idle_rate = duration;
252 break;
253 }
254 case k_ra8_phid_req_set_protocol: {
255 if (setup->w_value == (uint16_t)k_ra8_phid_proto_boot) {
256 s_state.protocol = k_ra8_phid_proto_boot;
257 } else if (setup->w_value == (uint16_t)k_ra8_phid_proto_report) {
258 s_state.protocol = k_ra8_phid_proto_report;
259 } else {
260 /* Unknown selector -- leave shadow untouched. */
261 }
262 break;
263 }
264 default: {
265 /* GET_REPORT / GET_IDLE / GET_PROTOCOL / SET_REPORT have payloads
266 * the application owns; defer to the registered callback. */
267 break;
268 }
269 }

277ra8_err_t ra8_usb_phid_init(ra8_usb_speed_t speed)
278{
279 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
280 return k_ra8_err_invalid_arg;
281 }
282 const ra8_err_t usb_err = ra8_usb_device_init(speed);
283 if (usb_err != k_ra8_ok) {
284 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
285 return k_ra8_err_hw_init_failed;
286 }
287
288 internal_reset_shadow(speed);
289
290 const ra8_err_t pipes_err = internal_configure_pipes(speed);
291 if (pipes_err != k_ra8_ok) {
292 (void)ra8_usb_device_deinit(speed);
293 return pipes_err;
294 }
295 s_state.initialized = true;
296 ra8_log_info_val(s_tag, "device-HID ready", (uint32_t)speed);
297 return k_ra8_ok;
298}

345ra8_err_t ra8_usb_phid_send_report(uint8_t report_id, const uint8_t* payload, uint16_t len)
346{
347 if (!s_state.initialized) {
348 return k_ra8_err_invalid_state;
349 }
350 if ((payload == nullptr) && (len != 0U)) {
351 return k_ra8_err_null_ptr;
352 }
353 if ((report_id == 0U) && (len == 0U)) {
354 return k_ra8_err_invalid_arg;
355 }
356 /* Per USB HID 1.11 sec 8 "Report Protocol", multi-report devices
357 * prepend the report ID. The combined frame must still fit the
358 * configured pipe max-packet. */
359 const uint16_t framed_len =
360 (report_id != 0U) ? (uint16_t)(len + (uint16_t)k_ra8_phid_report_id_prepend_len) : len;
361 if (framed_len > s_state.intr_max_packet) {
362 return k_ra8_err_invalid_arg;
363 }
364
365 if (report_id != 0U) {
366 /* Prepend the report ID byte: queue it on its own first, then the
367 * payload. The controller's FIFO write coalesces both into a
368 * single interrupt-IN packet on the next IN token. */
369 const uint8_t rid_byte = report_id;
370 const ra8_err_t rid_err =
371 ra8_usb_queue_in(s_state.speed, k_ra8_phid_pipe_intr_in, &rid_byte, 1U);
372 RA8_RETURN_ON_ERROR(rid_err, s_tag, "send_report: rid byte"); /* GCOVR_EXCL_BR_LINE */
373 }
374 return ra8_usb_queue_in(s_state.speed, k_ra8_phid_pipe_intr_in, payload, len);
375}

377ra8_err_t
378ra8_usb_phid_recv_report(uint8_t report_id, uint8_t* buf, uint16_t max_len, uint16_t* got_len)
379{
380 RA8_CHECK_NULL_PTR(buf, s_tag, "recv_report: buf");
381 RA8_CHECK_NULL_PTR(got_len, s_tag, "recv_report: got_len");
382 if (!s_state.initialized) {
383 return k_ra8_err_invalid_state;
384 }
385 if (max_len == 0U) {
386 return k_ra8_err_invalid_arg;
387 }
388 /* `report_id` is informational here -- the HID Report descriptor
389 * declares which IDs the device recognises; the host transmits the
390 * corresponding payload on PIPE7 directly. */
391 (void)report_id;
392
393 uint16_t inout_len = max_len;
394 const ra8_err_t err =
395 ra8_usb_queue_out(s_state.speed, k_ra8_phid_pipe_intr_out, buf, &inout_len, true);
396 if (err == k_ra8_ok) {
397 *got_len = inout_len;
398 } else {
399 *got_len = 0U;
400 }
401 return err;
402}

424ra8_err_t ra8_usb_phid_handle_setup(const ra8_usb_setup_t* setup)
425{
426 RA8_CHECK_NULL_PTR(setup, s_tag, "handle_setup: setup");
427 if (!s_state.initialized) {
428 return k_ra8_err_invalid_state;
429 }
430 if ((setup->bm_request_type != k_ra8_phid_bm_class_iface_in) &&
431 (setup->bm_request_type != k_ra8_phid_bm_class_iface_out)) {
432 return k_ra8_err_not_supported;
433 }
434 if (!internal_is_known_class_request(setup->b_request)) {
435 return k_ra8_err_not_supported;
436 }
437 internal_apply_class_setup(setup);
438
439 /* Forward to the application handler if one is registered. The
440 * application decides whether to ACK or stall via its return value. */
441 if (s_state.setup_cb != nullptr) {
442 const ra8_err_t cb_err = s_state.setup_cb(s_state.setup_ctx, setup);
443 if (cb_err != k_ra8_ok) {
444 return ra8_usb_control_response(s_state.speed, false);
445 }
446 }
447 return ra8_usb_control_response(s_state.speed, true);
448}
