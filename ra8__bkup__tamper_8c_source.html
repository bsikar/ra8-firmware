77static ra8_err_t internal_validate_chan(const ra8_bkup_tamper_chan_cfg_t* ch)

80 ((uint8_t)ch->edge != k_ra8_bkup_edge_rising)) {

323[[nodiscard]] ra8_err_t ra8_bkup_tamper_init(const ra8_bkup_tamper_config_t* cfg)
324{
325 RA8_CHECK_NULL_PTR(cfg, s_bkup_tag, "tamper cfg must not be nullptr");
326 if ((uint16_t)cfg->nc_width > k_ra8_bkup_max_nc_width) {
327 return k_ra8_err_invalid_arg;
328 }
329 const ra8_err_t v_err = internal_validate_tamper_channels(cfg);
330 RA8_RETURN_ON_ERROR(v_err,
331 s_bkup_tag,
332 "tamper_init: channel cfg out of range"); /* GCOVR_EXCL_BR_LINE */
333
334 /* The whole tamper register file is PRC1 (HUM Ch 13.1 Table 13.1 p 521). */
335 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_lpm)
336 {
337 /* HUM Ch 12.3.5 p 516 + Ch 12.3.7.4 step 0: disable VCHnNCE / VBTADCRn
338 * before changing VINCW. */
339 /* HUM Ch 12.2.9 "VBTICTLR2 : VBATT Input Control Register 2", p 506 */
340 *ra8_bkup_vbtictlr2() = 0U;
341 /* HUM Ch 12.2.15 "VBTADCR1 : VBATT Tamper Detection Control Register 1", p 510 */
342 *ra8_bkup_vbtadcr1() = 0U;
343 /* HUM Ch 12.2.16 "VBTADCR2 : VBATT Tamper Detection Control Register 2", p 511 */
344 *ra8_bkup_vbtadcr2() = 0U;
345 /* HUM Ch 12.2.17 "VBTADCR3 : VBATT Tamper Detection Control Register 3", p 511 */
346 *ra8_bkup_vbtadcr3() = 0U;
347
348 /* Tamper-init step 1 (12.3.7.4 p 518): VCHnINEN. */
349 /* HUM Ch 12.2.8 "VBTICTLR : VBATT Input Control Register", p 505 */
350 *ra8_bkup_vbtictlr() = internal_compose_vbtictlr(cfg);
351
352 /* Tamper-init step 3 (12.3.7.4 p 518): VINCW. */
353 /* HUM Ch 12.2.18 "VBTNCWCR : VBATT Noise Canceler Width Control Register", p 511 */
354 *ra8_bkup_vbtncwcr() = (uint8_t)((uint8_t)cfg->nc_width & k_ra8_bkup_vbtncwcr_mask_vincw);
355
356 /* Tamper-init step 4 (12.3.7.4 p 518): VCHnNCE + VCHnEG. */
357 /* HUM Ch 12.2.9 "VBTICTLR2 : VBATT Input Control Register 2", p 506 */
358 *ra8_bkup_vbtictlr2() = internal_compose_vbtictlr2(cfg);
359
360 /* HUM Ch 12.3.7.4 step 7: dummy-read + W0C VBTADFn (pseudo-edge from
361 * the edge-control register write may have set them). */
362 /* HUM Ch 12.2.14 "VBTADSR : VBATT Tamper Detection Status Register", p 509 */
363 (void)*ra8_bkup_vbtadsr();
364 *ra8_bkup_vbtadsr() = 0U;
365
366 /* Tamper-init step 8 (12.3.7.4 p 518): VBTADCR1/2/3. */
367 /* HUM Ch 12.2.15 "VBTADCR1 : VBATT Tamper Detection Control Register 1", p 510 */
368 *ra8_bkup_vbtadcr1() = internal_compose_vbtadcr1(cfg);
369 /* HUM Ch 12.2.16 "VBTADCR2 : VBATT Tamper Detection Control Register 2", p 511 */
370 *ra8_bkup_vbtadcr2() = internal_compose_vbtadcr2(cfg);
371 /* HUM Ch 12.2.17 "VBTADCR3 : VBATT Tamper Detection Control Register 3", p 511 */
372 *ra8_bkup_vbtadcr3() = internal_compose_vbtadcr3(cfg);
373 }
374
375 s_bkup_initialized = true;
376 ra8_log_info(s_bkup_tag, "bkup_tamper_init");
377 return k_ra8_ok;
378}
