338 TEST_ASSERT_NOT_NULL(f);
339 (void)fseek(f, 0, SEEK_END);
340 const long sz = ftell(f);
341 (void)fseek(f, 0, SEEK_SET);
342 TEST_ASSERT(sz > (long)k_ra8_jof_hdr_bytes);
343 uint8_t* buf = (uint8_t*)malloc((size_t)sz);
344 TEST_ASSERT_NOT_NULL(buf);

351 /* webtoon-native: a single full-width tile column (tile_w == width). */
352 TEST_ASSERT_EQ(rd_u16(buf + k_ra8_jof_ofs_width), rd_u16(buf + k_ra8_jof_ofs_tile_w));
353 free(buf);
354
355 (void)unlink(jpg);
356 (void)unlink(jof);
357 (void)rmdir(dir);
358 TEST_END("export jof round-trip");
