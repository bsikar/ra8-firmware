111RA8_INTERNAL
112static ra8_err_t s_src_take(gz_src_t* s, uint8_t* dst, size_t n)
113{
114 if ((uint64_t)n > (s->size - s->pos)) {
115 return k_ra8_err_validation_failed; /* runs past the member end */
116 }
117 const size_t got = s->read(s->ctx, s->pos, dst, n);
118 if (got != n) {
119 return k_ra8_err_validation_failed; /* backing truncated */
120 }
121 if (s->track_crc) {
122 s->crc = (uint32_t)mz_crc32(s->crc, dst, n);
123 }
124 s->pos += (uint64_t)n;
125 return k_ra8_ok;
126}

176RA8_INTERNAL
177static ra8_err_t s_skip_fextra(gz_src_t* s)
178{
179 uint8_t xl[k_gz_xlen_bytes] = {};
180 const ra8_err_t xerr = s_src_take(s, xl, sizeof(xl));
181 if (xerr != k_ra8_ok) {
182 return xerr;
183 }
184 uint32_t xlen = (uint32_t)xl[0] | ((uint32_t)xl[1] << (uint32_t)k_gz_shift_byte);
185 uint8_t skip[k_gz_in_chunk];
186 while (xlen > 0U) { /* bound: xlen <= 65535, consumed in chunks */
187 const size_t n = (xlen > (uint32_t)sizeof(skip)) ? sizeof(skip) : (size_t)xlen;
188 const ra8_err_t serr = s_src_take(s, skip, n);
189 if (serr != k_ra8_ok) {
190 return serr;
191 }
192 xlen -= (uint32_t)n;
193 }
194 return k_ra8_ok;
195}

217RA8_INTERNAL
218static ra8_err_t s_parse_header(gz_src_t* s)
219{
220 uint8_t fixed[k_gz_hdr_fixed] = {};
221 const ra8_err_t ferr = s_src_take(s, fixed, sizeof(fixed));
222 if (ferr != k_ra8_ok) {
223 return ferr;
224 }
225 if (fixed[k_gz_idx_id1] != (uint8_t)k_gz_id1) {
226 return k_ra8_err_not_supported;
227 }
228 if (fixed[k_gz_idx_id2] != (uint8_t)k_gz_id2) {
229 return k_ra8_err_not_supported;
230 }
231 if (fixed[k_gz_idx_cm] != (uint8_t)k_gz_cm_deflate) {
232 return k_ra8_err_not_supported; /* only DEFLATE is defined */
233 }
234 const uint8_t flg = fixed[k_gz_idx_flg];
235 if ((flg & (uint8_t)k_gz_flg_reserved) != 0U) {
236 return k_ra8_err_not_supported; /* reserved bits: unknown format */
237 }
238 if ((flg & (uint8_t)k_gz_flg_fextra) != 0U) {
239 const ra8_err_t xerr = s_skip_fextra(s);
240 if (xerr != k_ra8_ok) {
241 return xerr;
242 }
243 }
244 if ((flg & (uint8_t)k_gz_flg_fname) != 0U) {
245 const ra8_err_t nerr = s_skip_string(s);
246 if (nerr != k_ra8_ok) {
247 return nerr;
248 }
249 }
250 if ((flg & (uint8_t)k_gz_flg_fcomment) != 0U) {
251 const ra8_err_t cerr = s_skip_string(s);
252 if (cerr != k_ra8_ok) {
253 return cerr;
254 }
255 }
256 if ((flg & (uint8_t)k_gz_flg_fhcrc) != 0U) {
257 const uint32_t want = s->crc & (uint32_t)k_gz_crc16_mask;
258 uint8_t hc[k_gz_fhcrc_bytes] = {};
259 const ra8_err_t herr = s_src_take(s, hc, sizeof(hc));
260 if (herr != k_ra8_ok) {
261 return herr;
262 }
263 const uint32_t stored = (uint32_t)hc[0] | ((uint32_t)hc[1] << (uint32_t)k_gz_shift_byte);
264 if (stored != want) {
265 return k_ra8_err_checksum_mismatch;
266 }
267 }
268 s->track_crc = false; /* header ends here; the payload CRC is separate */
269 return k_ra8_ok;
270}

316static ra8_err_t s_inflate_pass(gz_inflate_t* st)
317{
318 const ra8_err_t ierr = ra8_decomp_budget_charge_iter(&st->budget);
319 if (ierr != k_ra8_ok) {
320 return ierr;
321 }
322 if (st->win_ofs == st->win_len) {
323 const uint64_t remain = st->src->size - st->src->pos;
324 const size_t want = (remain > (uint64_t)sizeof(st->win)) ? sizeof(st->win) : (size_t)remain;
325 if (want == 0U) {
326 return k_ra8_err_validation_failed; /* input exhausted mid-stream */
327 }
328 const ra8_err_t rerr = s_src_take(st->src, st->win, want);
329 if (rerr != k_ra8_ok) {
330 return rerr;
331 }
332 st->win_len = want;
333 st->win_ofs = 0U;
334 }
335 const bool more_input = (st->src->pos < st->src->size);
336 size_t in_bytes = st->win_len - st->win_ofs;
337 const size_t space = st->out_cap - st->total;
338 size_t out_bytes = (space > (size_t)k_gz_out_window) ? (size_t)k_gz_out_window : space;
339 const mz_uint32 flags = (mz_uint32)TINFL_FLAG_USING_NON_WRAPPING_OUTPUT_BUF |
340 (more_input ? (mz_uint32)TINFL_FLAG_HAS_MORE_INPUT : 0U);
341 const tinfl_status ts = tinfl_decompress(&s_gz_inflator,
342 &st->win[st->win_ofs],
343 &in_bytes,
344 st->out,
345 &st->out[st->total],
346 &out_bytes,
347 flags);
348 st->win_ofs += in_bytes;
349 if (out_bytes > 0U) {
350 st->crc = (uint32_t)mz_crc32(st->crc, &st->out[st->total], out_bytes);
351 }
352 st->total += out_bytes;
353 const ra8_err_t oerr =
354 ra8_decomp_budget_charge_output(&st->budget, st->src->pos, (uint64_t)out_bytes);
355 if (oerr != k_ra8_ok) {
356 return oerr;
357 }
358 if (ts == TINFL_STATUS_DONE) {
359 st->done = true;
360 return k_ra8_ok;
361 }
362 if (ts == TINFL_STATUS_NEEDS_MORE_INPUT) {
363 return k_ra8_ok; /* the next pass refills */
364 }
365 if (ts == TINFL_STATUS_HAS_MORE_OUTPUT) {
366 if (st->total == st->out_cap) {
367 return k_ra8_err_no_mem; /* arena full with the stream still open */
368 }
369 return k_ra8_ok; /* output window filled; the next pass continues */
370 }
371 return k_ra8_err_validation_failed; /* corrupt DEFLATE stream */
372}

392RA8_INTERNAL
393static ra8_err_t s_verify_trailer(const gz_inflate_t* st)
394{
395 const uint64_t leftover = (uint64_t)(st->win_len - st->win_ofs);
396 const uint64_t trailer_pos = st->src->pos - leftover;
397 const uint64_t remain = st->src->size - trailer_pos;
398 if (remain < (uint64_t)k_gz_trailer_len) {
399 return k_ra8_err_validation_failed; /* truncated trailer */
400 }
401 if (remain > (uint64_t)k_gz_trailer_len) {
402 return k_ra8_err_validation_failed; /* trailing bytes: no concatenation */
403 }
404 uint8_t tr[k_gz_trailer_len] = {};
405 const size_t got = st->src->read(st->src->ctx, trailer_pos, tr, sizeof(tr));
406 if (got != sizeof(tr)) {
407 return k_ra8_err_validation_failed; /* backing truncated */
408 }
409 uint32_t crc_stored = 0U;
410 uint32_t isize_stored = 0U;
411 for (uint32_t i = 0U; i < 4U; ++i) { /* bound: fixed field width */
412 crc_stored |= (uint32_t)tr[i] << ((uint32_t)k_gz_shift_byte * i);
413 isize_stored |= (uint32_t)tr[i + 4U] << ((uint32_t)k_gz_shift_byte * i);
414 }
415 if (crc_stored != st->crc) {
416 return k_ra8_err_checksum_mismatch;
417 }
418 if (isize_stored != ((uint32_t)st->total & (uint32_t)k_gz_isize_mask)) {
419 return k_ra8_err_checksum_mismatch;
420 }
421 return k_ra8_ok;
422}

438ra8_err_t ra8_unarch_gzip_unwrap(ra8_unarch_read_fn read,
439 void* ctx,
440 uint64_t size,
441 uint8_t* out,
442 size_t out_cap,
443 const ra8_decomp_limits_t* limits,
444 size_t* out_len)
445{
446 RA8_CHECK_NULL_PTR((const void*)read, s_tag_gz, "unwrap: null read");
447 RA8_CHECK_NULL_PTR(out, s_tag_gz, "unwrap: null out");
448 RA8_CHECK_NULL_PTR(out_len, s_tag_gz, "unwrap: null out_len");
449 *out_len = 0U;
450 if (size == 0U) {
451 return k_ra8_err_invalid_size;
452 }
453 if (out_cap == 0U) {
454 return k_ra8_err_invalid_size;
455 }
456 gz_src_t src = {
457 .read = read,
458 .ctx = ctx,
459 .size = size,
460 .pos = 0U,
461 .crc = (uint32_t)MZ_CRC32_INIT,
462 .track_crc = true,
463 };
464 const ra8_err_t herr = s_parse_header(&src);
465 if (herr != k_ra8_ok) {
466 return herr;
467 }
468 gz_inflate_t st = {};
469 st.src = &src;
470 st.out = out;
471 st.out_cap = out_cap;
472 st.crc = (uint32_t)MZ_CRC32_INIT;
473 const ra8_err_t berr = ra8_decomp_budget_init(&st.budget, limits);
474 if (berr != k_ra8_ok) {
475 return berr;
476 }
477 tinfl_init(&s_gz_inflator);
478 while (!st.done) { /* bound: every pass charges the policy iteration budget */
479 const ra8_err_t perr = s_inflate_pass(&st);
480 if (perr != k_ra8_ok) {
481 return perr;
482 }
483 }
484 const ra8_err_t terr = s_verify_trailer(&st);
485 if (terr != k_ra8_ok) {
486 return terr;
487 }
488 *out_len = st.total;
489 return k_ra8_ok;
490}
