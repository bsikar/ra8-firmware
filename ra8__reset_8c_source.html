279static ra8_reset_cause_t internal_decode(const ra8_reset_raw_t* raw)
280{
281 ra8_reset_cause_t cause = internal_decode_rstsr0(raw->rstsr0);
282 if (cause != k_ra8_reset_cause_unknown) {
283 return cause;
284 }
285
286 cause = internal_decode_rstsr3(raw->rstsr3);
287 if (cause != k_ra8_reset_cause_unknown) {
288 return cause;
289 }
290
291 cause = internal_decode_rstsr1(raw->rstsr1);
292 if (cause != k_ra8_reset_cause_unknown) {
293 return cause;
294 }
295
296 /* HUM Ch 6.2.4 "RSTSR2 : Reset Status Register 2" p 261 */
297 if ((raw->rstsr2 & k_ra8_reset_rstsr2_cwsf_msk) != 0U) {
298 return k_ra8_reset_cause_warm_start;
299 }
300
301 return k_ra8_reset_cause_unknown;

349ra8_err_t ra8_reset_clear_cause(uint32_t mask)
350{
351 /* HUM Ch 6.2.2 "RSTSR0 : Reset Status Register 0" p 258 */
352 /* Note from HUM: only 0 can be written; bit must be read as 1 first. */
353 const uint8_t rstsr0_clear = (uint8_t)(mask & k_ra8_reset_mask_rstsr0_msk);
354 if (rstsr0_clear != 0U) {
355 /* HUM Ch 6.2.2 "RSTSR0", p 257 */
356 const uint8_t before = *ra8_reset_rstsr0();
357 *ra8_reset_rstsr0() = (uint8_t)(before & (uint8_t)~rstsr0_clear);
358 }
359
360 /* HUM Ch 6.2.3 "RSTSR1 : Reset Status Register 1" p 258 */
361 /* Same write-0-to-clear semantics as RSTSR0. */
362 const uint32_t rstsr1_clear =
363 (mask & k_ra8_reset_mask_rstsr1_window) >> k_ra8_reset_mask_rstsr1_shift;
364 if (rstsr1_clear != 0U) {
365 /* HUM Ch 6.2.3 "RSTSR1", p 258 */
366 const uint32_t before = *ra8_reset_rstsr1();
367 *ra8_reset_rstsr1() = before & ~rstsr1_clear;
368 }
369
370 if ((mask & k_ra8_reset_mask_rstsr2_cwsf) != 0U) {
371 /* HUM Ch 6.2.4 "RSTSR2 : Reset Status Register 2" p 261 */
372 /* CWSF is set by writing 1 (Note 2). */
373 *ra8_reset_rstsr2() = k_ra8_reset_rstsr2_cwsf_msk;
374 }
375
376 /* Refresh the cached cause if init was already called -- the next
377 * reader should see the post-clear state, not the boot snapshot. */
378 if (s_state.initialized) {
379 internal_read_raw(&s_state.raw);
380 s_state.cause = internal_decode(&s_state.raw);
381 }
382 return k_ra8_ok;
383}

393void ra8_reset_software_reset(void)
394{
395 /* HUM Ch 6.1 Table 6.1 p 244 -- "Software reset / Register setting
396 * (use the software reset bit AIRCR.SYSRESETREQ)". The Cortex-M85
397 * AIRCR is documented in the ARMv8-M Architecture Reference Manual;
398 * we reproduce the constants in ra8_reset_regs.h so the driver has
399 * a single include. */
400 ra8_log_info(s_tag, "software reset");
401
402 const uint32_t aircr_value = (k_ra8_reset_aircr_vectkey_value << k_ra8_reset_aircr_vectkey_pos) |
403 k_ra8_reset_aircr_sysresetreq_msk;
404
405 /* HUM Ch 6.1 "Overview" p 244 */
406 /* Table 6.1 names AIRCR.SYSRESETREQ as the software-reset trigger. */
407 *ra8_reset_aircr() = aircr_value;
408
409 /* On real hardware the AIRCR write triggers a system reset within a few
410 * cycles; ::ra8_hw_wait_for_reset issues a DSB so the write retires, then
411 * spins forever so control never falls through into the caller's tail. On
412 * the host the seam returns at once so a unit test can inspect the AIRCR
413 * write it just staged. */
414 ra8_hw_wait_for_reset();
415}

503ra8_err_t ra8_reset_set_source_mask(ra8_reset_source_t source, bool disable)
504{
505 volatile uint8_t* reg = nullptr;
506 uint8_t mask = 0U;
507 if (!internal_source_loc(source, &reg, &mask)) {
508 ra8_log_error(s_tag, "set_source_mask: invalid reset source");
509 return k_ra8_err_invalid_arg;
510 }
511
512 volatile uint16_t* prcr = ra8_reset_prcr();
513 /* Set PRCR.PRC5 to 1 (write enabled) before rewriting the reset-control
514 * group; preserve the other PRC bits. */
515 /* HUM Ch 6.2.6 "SYRSTMSK0" p 263 */
516 const uint16_t prcr_cur = *prcr;
517 *prcr = (uint16_t)((uint16_t)k_ra8_reset_prcr_key |
518 (uint16_t)(prcr_cur & (uint16_t)k_ra8_reset_prcr_pr_bits_msk) |
519 (uint16_t)k_ra8_reset_prcr_prc5_msk);
520
521 /* 1 disables the corresponding reset, 0 enables it. */
522 /* HUM Ch 6.2.6 "System Reset Mask Control" p 262 */
523 const uint8_t before = *reg;
524 *reg = disable ? (uint8_t)(before | mask) : (uint8_t)(before & (uint8_t)~mask);
525
526 /* Relock PRCR.PRC5. */
527 /* HUM Ch 6.2.6 "SYRSTMSK0" p 263 */
528 const uint16_t prcr_now = *prcr;
529 *prcr = (uint16_t)((uint16_t)k_ra8_reset_prcr_key |
530 (uint16_t)((prcr_now & (uint16_t)k_ra8_reset_prcr_pr_bits_msk) &
531 (uint16_t)~(uint16_t)k_ra8_reset_prcr_prc5_msk));
532 return k_ra8_ok;
533}

537 RA8_CHECK_NULL_PTR(disabled, s_tag, "disabled must not be nullptr");
