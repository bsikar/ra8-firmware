233static void wc_print_hex(uint32_t value)
234{
235 uint8_t buf[k_wc_hex_nibbles];
236 for (uint32_t i = 0U; i < (uint32_t)k_wc_hex_nibbles; i++) {
237 const uint32_t shift = ((uint32_t)k_wc_hex_nibbles - 1U - i) * (uint32_t)k_wc_nibble_bits;
238 const uint32_t nib = (value >> shift) & (uint32_t)k_wc_nibble_mask;
239 buf[i] = (uint8_t)((nib < (uint32_t)k_wc_dec_ten) ? ('0' + nib) : ('A' + (nib - k_wc_dec_ten)));
240 }
241 wc_print(buf, (uint32_t)k_wc_hex_nibbles);
242}

245static void wc_print_uint(uint32_t value)
246{
247 uint8_t buf[k_wc_dec_ten];
248 uint32_t n = 0U;
249 if (value == 0U) {
250 buf[n] = '0';
251 n++;
252 }
253 while ((value > 0U) && (n < (uint32_t)k_wc_dec_ten)) {
254 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_wc_dec_ten));
255 n++;
256 value /= (uint32_t)k_wc_dec_ten;
257 }
258 for (uint32_t i = 0U; i < n; i++) {
259 wc_print(&buf[n - 1U - i], 1U);
260 }
261}

264static uint32_t wc_framebuffer_hash(void)
265{
266 const uint8_t* p = (const uint8_t*)s_framebuffer;
267 const size_t n = sizeof(s_framebuffer);
268 uint32_t hsh = (uint32_t)k_wc_fnv_offset;
269 for (size_t i = 0U; i < n; i++) {
270 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_wc_fnv_prime;
271 }
272 return hsh;
273}

281static void wc_fmt_frame(char* buf, uint32_t frame)
282{
283 static const char prefix[] = "frame=";
284 uint32_t pos = 0U;
285 for (uint32_t i = 0U; i < (uint32_t)(sizeof(prefix) - 1U); i++) {
286 buf[pos] = prefix[i];
287 pos++;
288 }
289 char digits[k_wc_dec_ten];
290 uint32_t n = 0U;
291 if (frame == 0U) {
292 digits[n] = '0';
293 n++;
294 }
295 while ((frame > 0U) && (n < (uint32_t)k_wc_dec_ten)) {
296 digits[n] = (char)('0' + (frame % (uint32_t)k_wc_dec_ten));
297 n++;
298 frame /= (uint32_t)k_wc_dec_ten;
299 }
300 for (uint32_t i = 0U; i < n; i++) {
301 buf[pos] = digits[n - 1U - i];
302 pos++;
303 }
304 buf[pos] = '\0';

498static bool wc_check_full(uint32_t* out_crc, uint16_t* out_dirty)
499{
500 s_left.renders = 0U;
501 s_right.renders = 0U;
502 wc_invalidate_all(k_ra8_widget_refresh_quality);
503 ra8_ui_rect_t dmg = {};
504 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
505 uint16_t nd = 0U;
506 if (wc_compose(&dmg, &hint, &nd) != k_ra8_ok) {
507 return false;
508 }
509 if (nd != (uint16_t)k_wc_root_n) {
510 return false;
511 }
512 if ((dmg.w != (int32_t)k_wc_fb_w) || (dmg.h != (int32_t)k_wc_fb_h)) {
513 return false;
514 }
515 if (hint != k_ra8_widget_refresh_quality) {
516 return false;
517 }
518 if ((s_left.renders != 1U) || (s_right.renders != 1U)) {
519 return false; /* nested body panel must have composited both tiles */
520 }
521 *out_crc = wc_framebuffer_hash();
522 *out_dirty = nd;
523 return true;
524}

527static bool wc_check_partial(uint32_t* out_crc, uint16_t* out_dirty)
528{
529 s_left.renders = 0U;
530 s_right.renders = 0U;
531 s_status.frame++;
532 (void)ra8_widget_invalidate(&s_root_kids[k_wc_w_status], k_ra8_widget_refresh_fast);
533 ra8_ui_rect_t dmg = {};
534 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
535 uint16_t nd = 0U;
536 if (wc_compose(&dmg, &hint, &nd) != k_ra8_ok) {
537 return false;
538 }
539 if (nd != 1U) {
540 return false;
541 }
542 if ((dmg.x != 0) || (dmg.y != 0) || (dmg.w != (int32_t)k_wc_fb_w)) {
543 return false;
544 }
545 if (dmg.h != (int32_t)k_wc_status_h) {
546 return false;
547 }
548 if (hint != k_ra8_widget_refresh_fast) {
549 return false;
550 }
551 if ((s_left.renders != 0U) || (s_right.renders != 0U)) {
552 return false; /* partial flush must NOT re-render the body */
553 }
554 *out_crc = wc_framebuffer_hash();
555 *out_dirty = nd;
556 return true;
557}

560static bool wc_selfcheck(uint32_t* crc0, uint16_t* d0, uint32_t* crc1, uint16_t* d1)
561{
562 if (!wc_check_full(crc0, d0)) {
563 return false;
564 }
565 if (!wc_check_partial(crc1, d1)) {
566 return false;
567 }
568 return (*crc0 != *crc1); /* the advanced frame counter changes the composite */
569}

622#pragma GCC diagnostic push
623#pragma GCC diagnostic ignored "-Wmain"
636int32_t main(void)
637{
638 wc_setup_or_halt();
639 ra8_isr_globals_enable();
640 wc_print(k_wc_msg_boot, (uint32_t)sizeof(k_wc_msg_boot) - 1U);
641
642 if (!wc_panel_up()) {
643 wc_fail_halt(k_wc_msg_fpanl, (uint32_t)sizeof(k_wc_msg_fpanl) - 1U);
644 }
645 if (!wc_build_tree()) {
646 wc_fail_halt(k_wc_msg_ftree, (uint32_t)sizeof(k_wc_msg_ftree) - 1U);
647 }
648
649 uint32_t crc0 = 0U;
650 uint32_t crc1 = 0U;
651 uint16_t d0 = 0U;
652 uint16_t d1 = 0U;
653 if (!wc_selfcheck(&crc0, &d0, &crc1, &d1)) {
654 wc_fail_halt(k_wc_msg_fchk, (uint32_t)sizeof(k_wc_msg_fchk) - 1U);
655 }
656 wc_print_banner(d0, crc0, d1, crc1);
657
658 /* Land on a full composite and show it live on the panel. */
659 (void)ra8_gfx_clear((uint32_t)k_wc_col_clear);
660 wc_invalidate_all(k_ra8_widget_refresh_quality);
661 ra8_ui_rect_t dmg = {};
662 ra8_widget_refresh_t hint = k_ra8_widget_refresh_none;
663 uint16_t nd = 0U;
664 if (wc_compose(&dmg, &hint, &nd) == k_ra8_ok) {
665 wc_flush_rect(&dmg, hint);
666 }
667
668 while (1) {
669 wc_tick_live();
