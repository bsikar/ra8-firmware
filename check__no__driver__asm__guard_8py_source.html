4"""Gate: a HAL driver shall not guard bare CPU asm on RA8_OFF_TARGET.
5
6Issue #293 (following #238) migrated every host-compatibility CPU primitive out
7of the HAL peripheral drivers and onto ONE shared seam:
8
9 - ``libs/ra8_hal/inc/ra8_hw_intrinsics.h`` (target inline asm / host decls)
10 - ``tests/mocks/ra8_host_asm_stub.c`` (the host-safe definitions)
11
12A driver that needs ``wfi`` / ``dsb`` / ``isb`` / ``nop`` / the ``cpsie i`` /
13``cpsid i`` gate / the post-reset spin now calls ``ra8_hw_wfi()`` and friends;
14it carries NO ``#ifdef RA8_OFF_TARGET`` of its own -- the seam owns the
15host/target divergence. That keeps coverage and MC/DC landing on the real
16shipping path instead of a compiled-out detour, exactly as #238 intended.
17
18This gate keeps a NEW guard class from creeping back in. It FAILS if any
19translation unit under ``libs/ra8_hal/src/`` contains an inline-asm statement
20(``__asm`` / ``__asm__``) inside a preprocessor conditional whose controlling
21expression references ``RA8_OFF_TARGET`` -- in EITHER branch. Comment text
22is stripped first, so prose that merely mentions ``__asm__`` never trips it.
23
24Scope note: this is deliberately limited to ``libs/ra8_hal/src/`` -- the HAL
25peripheral drivers. Boot code (``system_init.c``, ``vector_table.c``,
26``trustzone_init.c``), the core runtime (``ra8_core``), and the TrustZone /
27secure-boot flow legitimately host target-only asm (reset vectors, fault
28handlers, SAU bring-up, ``msr msp_ns``) that has no host code path to drive and
29is not a peripheral-driver short-circuit; those are out of scope by design.
30
31There is no allowlist: route the primitive through ``ra8_hw_intrinsics.h`` (add
32a new one there and to the host stub if it is genuinely missing), never behind a
33fresh in-driver ``#ifdef RA8_OFF_TARGET``.
34
35Run::
36
37 check_no_driver_asm_guard.py
38
39Exit status: 0 if every driver is clean, 1 otherwise.
40"""
41
42import re
43import sys
44from pathlib import Path

46# Repo root: this file is scripts/checks/check_no_driver_asm_guard.py .
47REPO_ROOT = Path(__file__).resolve().parents[2]
48
49# The HAL peripheral drivers -- the only scope this gate governs.
50DRIVER_DIR = REPO_ROOT / "libs" / "ra8_hal" / "src"
51
52_RE_IF = re.compile(r"^\s*#\s*(if|ifdef|ifndef)\b(.*)$")
53_RE_ELIF = re.compile(r"^\s*#\s*elif\b(.*)$")
54_RE_ENDIF = re.compile(r"^\s*#\s*endif\b")
55_RE_ASM = re.compile(r"(?<![A-Za-z0-9_])__asm(__)?(?![A-Za-z0-9_])")
56_OFF_TARGET = "RA8_OFF_TARGET"
57

59def strip_comments(text: str) -> list[str]:
60 """Blank out C ``/* */`` and ``//`` comments, preserving line count.
61
62 Preprocessor directives never live inside comments, so a line-preserving
63 strip lets the conditional walk and the asm scan share one clean view
64 without a false positive from a doc block that mentions ``__asm__``.
65 """
66 out: list[str] = []
67 in_block = False
68 for line in text.splitlines():
69 buf: list[str] = []
70 i = 0
71 n = len(line)
72 while i < n:
73 two = line[i : i + 2]
74 if in_block:
75 if two == "*/":
76 in_block = False
77 i += 2
78 else:
79 i += 1
80 continue
81 if two == "/*":
82 in_block = True
83 i += 2
84 continue
85 if two == "//":
86 break
87 buf.append(line[i])
88 i += 1
89 out.append("".join(buf))
90 return out

93def check_file(path: Path) -> list[str]:
94 """Return violation strings for one driver TU (empty when clean)."""
95 rel = path.relative_to(REPO_ROOT).as_posix()
96 lines = strip_comments(path.read_text(encoding="utf-8"))
97
98 # Stack of booleans: does this open conditional's region reference the
99 # fake flag (in its #if / any #elif)? A True anywhere on the stack
100 # means the current line compiles under a off-target-conditioned region.
101 stack: list[bool] = []
102 problems: list[str] = []
103
104 for idx, line in enumerate(lines, start=1):
105 m_if = _RE_IF.match(line)
106 if m_if is not None:
107 stack.append(_OFF_TARGET in m_if.group(2))
108 continue
109 m_elif = _RE_ELIF.match(line)
110 if m_elif is not None:
111 if stack:
112 stack[-1] = stack[-1] or (_OFF_TARGET in m_elif.group(1))
113 continue
114 if _RE_ENDIF.match(line):
115 if stack:
116 stack.pop()
117 continue
118 # #else keeps the frame's off-target-reference flag: both branches of a
119 # `#ifdef RA8_OFF_TARGET` are off-target-conditioned regions.
120 if _RE_ASM.search(line) and any(stack):
121 problems.append(
122 f"{rel}:{idx}: inline asm '{line.strip()}' sits inside a "
123 f"{_OFF_TARGET} conditional -- route it through "
124 f"libs/ra8_hal/inc/ra8_hw_intrinsics.h instead"
125 )
126 return problems

129def main() -> int:
130 """Fail any HAL driver that guards bare CPU asm on RA8_OFF_TARGET.
131
132 A missing driver directory exits 1 rather than 0. That is deliberate: this
133 gate has a single hardcoded scan root, so the directory vanishing means
134 the tree moved under it, and the one thing it must not do is report a
135 clean sweep of somewhere that does not exist.
136
137 Returns 0 when every driver routes its CPU primitives through
138 ra8_hw_intrinsics.h, 1 on a violation or a missing driver directory.
139 """
140 if not DRIVER_DIR.is_dir():
141 print(f"check_no_driver_asm_guard.py: driver dir not found: {DRIVER_DIR}")
142 return 1
143
144 drivers = sorted(DRIVER_DIR.glob("*.c"))
145 all_problems: list[str] = []
146 for path in drivers:
147 all_problems.extend(check_file(path))
148
149 if all_problems:
150 print("check_no_driver_asm_guard.py: a HAL driver guards bare asm on RA8_OFF_TARGET:")
151 for p in all_problems:
152 print(f" {p}")
153 print("Fix at the root -- call the ra8_hw_* primitive from")
154 print(" libs/ra8_hal/inc/ra8_hw_intrinsics.h")
155 print("(add a new one there plus its host body in")
156 print(" tests/mocks/ra8_host_asm_stub.c if it does not exist yet).")
157 return 1
158
159 print(
160 f"check_no_driver_asm_guard.py: PASS -- {len(drivers)} HAL driver TU(s) "
161 f"carry no RA8_OFF_TARGET-guarded asm."
162 )
163 return 0
