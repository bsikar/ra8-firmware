65{

278RA8_INTERNAL static ra8_err_t ra8_wifi_c6link_op_service(void* ctx, ra8_wifi_link_t* out_link)
279{
280 ra8_wifi_c6link_t* self = (ra8_wifi_c6link_t*)ctx;
281 RA8_CHECK_NULL_PTR(self, RA8_WIFI_C6_TAG, "ctx");
282 RA8_CHECK_NULL_PTR(out_link, RA8_WIFI_C6_TAG, "out_link");
283
284 ra8_c6link_stats_t stats = {};
285 const ra8_err_t err =
286 ra8_c6link_poll(self->link, (uint16_t)k_ra8_c6link_announce_transfers, &stats);
287 if (err != k_ra8_ok) {
288 return err;
289 }
290
291 /* Down unless a connect was seen, and a later disconnect wins outright. Two
292 * single-condition tests applied in that order, so the precedence lives in
293 * the sequence rather than in a compound decision that would then owe MC/DC
294 * vectors. */
295 *out_link = k_ra8_wifi_link_down;
296 if (self->connected) {
297 *out_link = k_ra8_wifi_link_up;
298 }
299 if (self->disconnected) {
300 *out_link = k_ra8_wifi_link_down;
301 }
302 return k_ra8_ok;
303}
