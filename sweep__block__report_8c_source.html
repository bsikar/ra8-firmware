203static bool cbs_find_knee(const cbs_row_t* rows,
204 uint32_t nrows,
205 const uint32_t* blocks,
206 uint32_t nblocks,
207 cbs_knee_t* out)
208{
209 if ((out == nullptr) || (blocks == nullptr)) {
210 return false;
211 }
212 *out = (cbs_knee_t){};
213 for (uint32_t i = 0U; i < nblocks; ++i) {
214 const cbs_row_t* s = cbs_find_row(rows, nrows, "rbkc-z9", "seq", blocks[i]);
215 const double m = cbs_row_mibs(s);
216 if (m > out->peak_mibs) {
217 out->peak_mibs = m;
218 out->peak_block = blocks[i];
219 }
220 }
221 if (out->peak_mibs <= 0.0) {
222 return false;
223 }
224 const double bar = out->peak_mibs * ((double)k_cbs_knee_pct / s_cbs_pct_f);
225 out->knee_block = out->peak_block;
226 out->knee_mibs = out->peak_mibs;
227 for (uint32_t i = 0U; i < nblocks; ++i) {
228 const cbs_row_t* s = cbs_find_row(rows, nrows, "rbkc-z9", "seq", blocks[i]);
229 const double m = cbs_row_mibs(s);
230 if (m >= bar) {
231 out->knee_block = blocks[i];
232 out->knee_mibs = m;
233 break;
234 }
235 }
236 return true;
237}

240void cbs_priv_print_crossover(const cbs_row_t* rows,
241 uint32_t nrows,
242 const uint32_t* blocks,
243 uint32_t nblocks)
244{
245 cbs_knee_t k = {};
246 if (!cbs_find_knee(rows, nrows, blocks, nblocks, &k)) {
247 (void)printf("\n(no rbkc-z9 sequential rows; crossover not computed)\n");
248 return;
249 }
250 const double knee_pct = s_cbs_pct_f * k.knee_mibs / k.peak_mibs;
251 char peak_l[16] = {};
252 char knee_l[16] = {};
253 char def_l[16] = {};
254 cbs_block_label(k.peak_block, peak_l, sizeof(peak_l));
255 cbs_block_label(k.knee_block, knee_l, sizeof(knee_l));
256 cbs_block_label((uint32_t)k_cbs_default_chunk, def_l, sizeof(def_l));
257 (void)printf("\n### Measured crossover (rbkc-z9, sequential)\n\n");
258 (void)printf("Peak %.0f MiB/s at %s; knee (first size within %u%% of peak) at %s "
259 "(%.0f MiB/s, %.1f%% of peak).\n",
260 k.peak_mibs,
261 peak_l,
262 (unsigned)k_cbs_knee_pct,
263 knee_l,
264 k.knee_mibs,
265 knee_pct);
266 if (k.knee_block == (uint32_t)k_cbs_default_chunk) {
267 (void)printf("The measured knee lands on the current %s `.rabook` chunk default (#204): "
268 "keep it.\n",
269 def_l);
270 } else if (k.knee_block < (uint32_t)k_cbs_default_chunk) {
271 (void)printf("The knee sits BELOW the current %s default: %s already reaches %.1f%% of "
272 "peak on this backend, at a smaller per-miss inflate latency and less RAM "
273 "per frame.\n",
274 def_l,
275 knee_l,
276 knee_pct);
277 } else {
278 (void)printf("The knee sits ABOVE the current %s default: sequential throughput is still "
279 "climbing past it; consider a larger chunk if per-miss latency allows.\n",
280 def_l);
281 }
282 (void)printf("Caveat: these are host numbers -- per-request cost here is only the chunk "
283 "lookup + zlib stream setup. SD-over-SPI adds real per-command overhead "
284 "(CMD17 loops, #202), which pushes the knee toward larger blocks; the "
285 "hardware leg of #208 must re-run this sweep on the bench before shrinking "
286 "the chunk size below the default.\n");
