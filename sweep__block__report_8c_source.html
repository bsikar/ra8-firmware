172 (double)s->wall_ns / (double)s->reads,

246static bool cbs_find_knee(const cbs_row_t* rows,
247 uint32_t nrows,
248 const uint32_t* blocks,
249 uint32_t nblocks,
250 cbs_knee_t* out)
251{
252 if ((out == nullptr) || (blocks == nullptr)) {
253 return false;
254 }
255 *out = (cbs_knee_t){};
256 for (uint32_t i = 0U; i < nblocks; ++i) {
257 const cbs_row_t* s = cbs_find_row(rows, nrows, "rbkc-z9", "seq", blocks[i]);
258 const double m = cbs_row_mibs(s);
259 if (m > out->peak_mibs) {
260 out->peak_mibs = m;
261 out->peak_block = blocks[i];
262 }
263 }
264 if (out->peak_mibs <= 0.0) {
265 return false;
266 }
267 const double bar = out->peak_mibs * ((double)k_cbs_knee_pct / s_cbs_pct_f);
268 out->knee_block = out->peak_block;
269 out->knee_mibs = out->peak_mibs;
270 for (uint32_t i = 0U; i < nblocks; ++i) {
271 const cbs_row_t* s = cbs_find_row(rows, nrows, "rbkc-z9", "seq", blocks[i]);
272 const double m = cbs_row_mibs(s);
273 if (m >= bar) {
274 out->knee_block = blocks[i];
275 out->knee_mibs = m;
276 break;
277 }
278 }
279 return true;
280}

283void cbs_priv_print_crossover(const cbs_row_t* rows,
284 uint32_t nrows,
285 const uint32_t* blocks,
286 uint32_t nblocks)
287{
288 cbs_knee_t k = {};

291 return;
292 }
293 const double knee_pct = s_cbs_pct_f * k.knee_mibs / k.peak_mibs;
294 char peak_l[16] = {};
295 char knee_l[16] = {};
296 char def_l[16] = {};
297 cbs_block_label(k.peak_block, peak_l, sizeof(peak_l));
298 cbs_block_label(k.knee_block, knee_l, sizeof(knee_l));
299 cbs_block_label((uint32_t)k_cbs_default_chunk, def_l, sizeof(def_l));
300 (void)printf("\n### Measured crossover (rbkc-z9, sequential)\n\n");
301 (void)printf("Peak %.0f MiB/s at %s; knee (first size within %u%% of peak) at %s "
302 "(%.0f MiB/s, %.1f%% of peak).\n",
303 k.peak_mibs,
304 peak_l,
305 (unsigned)k_cbs_knee_pct,
306 knee_l,
307 k.knee_mibs,
308 knee_pct);
309 if (k.knee_block == (uint32_t)k_cbs_default_chunk) {
310 (void)printf("The measured knee lands on the current %s `.rabook` chunk default (#204): "
311 "keep it.\n",
312 def_l);
313 } else if (k.knee_block < (uint32_t)k_cbs_default_chunk) {
314 (void)printf("The knee sits BELOW the current %s default: %s already reaches %.1f%% of "
315 "peak on this backend, at a smaller per-miss inflate latency and less RAM "
316 "per frame.\n",
317 def_l,
318 knee_l,
319 knee_pct);
320 } else {
321 (void)printf("The knee sits ABOVE the current %s default: sequential throughput is still "
322 "climbing past it; consider a larger chunk if per-miss latency allows.\n",
323 def_l);
324 }
325 (void)printf("Caveat: these are host numbers -- per-request cost here is only the chunk "
326 "lookup + zlib stream setup. SD-over-SPI adds real per-command overhead "
327 "(CMD17 loops, #202), which pushes the knee toward larger blocks; the "
