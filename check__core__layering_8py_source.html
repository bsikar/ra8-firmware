4"""Gate: ``libs/ra8_core`` is the foundation layer and depends on no other lib.
5
6``ra8_core`` (error codes, checks, logging, assert, attributes, ...) sits at the
7bottom of the library stack: every other ``libs/*`` module is allowed to build
8on it, but it must build on nothing above itself. An ``#include`` from a
9``ra8_core`` file that resolves to a header owned by another first-party lib is
10an *upward* dependency -- it inverts the layering and creates a cycle risk.
11
12#243 was one such edge (``ra8_register_protection.h`` pulled ``ra8_hal``'s
13system-register header into ``ra8_core``); it was fixed by moving the header
14into ``ra8_hal``. This gate makes the invariant permanent: it fails if any file
15under ``libs/ra8_core`` includes a header owned by a different ``libs/*``
16module. System headers (``<...>``), vendored third-party headers, and
17``ra8_core``'s own headers are all fine.
18
19Run::
20
21 check_core_layering.py # scan libs/ra8_core
22 check_core_layering.py path/to/file.c # scan listed files
23
24Exit 0 if ``ra8_core`` has no upward lib dependency, exit 1 (with the
25offending edges) otherwise, exit 2 when the whole-tree sweep or the header
26ownership oracle collapses below its floor (FILE_FLOOR / OWNER_FLOOR).
27"""
28

38from lint_targets import is_build_output_path

40REPO_ROOT = Path(__file__).resolve().parents[2]
41LIBS_ROOT = REPO_ROOT / "libs"
42
43FOUNDATION_LIB = "ra8_core"
44SOURCE_SUFFIXES = (".c", ".h", ".cpp", ".hpp")
45INCLUDE_RE = re.compile(r'#\s*include\s*[<"]([^">]+)[">]')
46EXCLUDE_FRAGMENTS = ("/third_party/", "/ra8_fonts/")
47
48# A lib header path under libs/ is <module>/<inc|src>/...: at least the module
49# component and the inc/src component must be present to attribute an owner.
50_MIN_LIB_PATH_PARTS = 2
51
52# The foundation lib cannot legitimately collapse to a handful of files; if the
53# whole-tree sweep returns less than this, something broke (an unreachable repo
54# root, a renamed FOUNDATION_LIB) and reporting "no upward lib dependency"
55# would be a lie. Measured 2026-07-28: 37 sources under libs/ra8_core.
56FILE_FLOOR = 28
57
58# Floor on the header-ownership ORACLE, which is the more dangerous vacuity of
59# the two: every finding here is `owners.get(name)` returning a non-empty set,
60# so an oracle that collapses reports a perfectly layered foundation no matter
61# what ra8_core includes -- with the full file list still scanned, and the
62# usual "N file(s) scanned" line still printed. Measured 2026-07-28: 425
63# distinct first-party lib header basenames.
64OWNER_FLOOR = 340
65

67def _header_owners() -> dict[str, set[str]]:
68 """Map each first-party lib header basename to the lib module(s) owning it.
69
70 A lib module is the first path component under ``libs/`` (e.g. ``ra8_hal``).
71 """
72 owners: dict[str, set[str]] = {}
73 for header in LIBS_ROOT.rglob("*.h"):
74 text = str(header)
75 if is_build_output_path(text) or any(frag in text for frag in EXCLUDE_FRAGMENTS):
76 continue
77 rel = header.relative_to(LIBS_ROOT).parts
78 # rel[0] is the lib module; only count headers under a lib's inc/ or src/
79 if len(rel) < _MIN_LIB_PATH_PARTS or rel[1] not in ("inc", "src"):
80 continue
81 owners.setdefault(header.name, set()).add(rel[0])
82 return owners

85def _is_source(path: Path) -> bool:
86 return path.suffix in SOURCE_SUFFIXES

89def _rel(path: Path) -> str:
90 if path.is_relative_to(REPO_ROOT):

95def _enumerate_targets(arg_paths: Iterable[str]) -> list[Path]:
96 args = list(arg_paths)
97 if args:
98 out: list[Path] = []
99 for raw in args:
100 path = Path(raw)
101 if not path.is_absolute():
102 path = REPO_ROOT / path
103 if path.is_dir():
104 for suffix in SOURCE_SUFFIXES:
105 out.extend(path.rglob("*" + suffix))
106 elif _is_source(path):
107 out.append(path)
108 # only files that actually live under the foundation lib matter
109 base = REPO_ROOT / "libs" / FOUNDATION_LIB
110 return [p for p in out if p.is_relative_to(base)]
111
112 root = REPO_ROOT / "libs" / FOUNDATION_LIB
113 out = []
114 for suffix in SOURCE_SUFFIXES:
115 out.extend(root.rglob("*" + suffix))
116 return out

119def main(argv: list[str]) -> int:
120 """Fail if anything in the foundation lib includes a header another lib owns.
121
122 ra8_core sits at the bottom of the stack, so an include pointing UP at a
123 higher lib is a dependency inversion that would make the foundation
124 unbuildable on its own. A header owned by several libs INCLUDING ra8_core
125 is fine -- that is a shared name, not an upward edge -- which is why
126 ownership is compared as a set difference rather than by first match.
127
128 Paths given on argv are filtered down to those under libs/ra8_core before
129 scanning, so pointing this at the whole staged file list is safe and cheap
130 -- and legitimately filters to nothing, which is why FILE_FLOOR guards the
131 whole-tree sweep only.
132
133 OWNER_FLOOR has no such exemption: the ownership map is the oracle every
134 finding is decided against, so it is checked on every path. A collapsed
135 oracle cannot report an edge no matter which files are scanned, and would
136 do it while printing the usual "N file(s) scanned, no upward lib
137 dependency".
138
139 Returns 1 with the offending include sites listed, 0 when no upward edge
140 exists or when the argv filter left nothing under the foundation lib, 2
141 when either floor is not met.
142 """
143 paths = argv[1:]
144 targets = _enumerate_targets(paths)
145 if not paths and len(targets) < FILE_FLOOR:
146 print(
147 f"check_core_layering.py: FATAL -- only {len(targets)} {FOUNDATION_LIB} "
148 f"file(s) in scope, floor is {FILE_FLOOR}. A collapsed sweep reports a "
149 "clean layering because it scanned nothing.",
150 file=sys.stderr,
151 )
152 return 2
153 if not targets:
154 print("check_core_layering.py: no ra8_core files to scan", file=sys.stderr)
155 return 0
156
157 owners = _header_owners()
158 if len(owners) < OWNER_FLOOR:
159 print(
160 f"check_core_layering.py: FATAL -- header ownership map holds only "
161 f"{len(owners)} header(s), floor is {OWNER_FLOOR}. A collapsed oracle "
162 "reports a clean layering because it can attribute nothing.",
163 file=sys.stderr,
164 )
165 return 2
166 edges = []
167 for path in targets:
168 try:
169 text = path.read_text(encoding="utf-8", errors="replace")
170 except OSError:
171 continue
172 for lineno, line in enumerate(text.splitlines(), 1):
173 match = INCLUDE_RE.search(line)
174 if not match:
175 continue
176 name = Path(match.group(1)).name
177 mods = owners.get(name, set())
178 # owned by some other lib, and NOT (also) owned by ra8_core
179 if mods and FOUNDATION_LIB not in mods and mods - {FOUNDATION_LIB}:
180 edges.append((_rel(path), lineno, name, sorted(mods)))
181
182 if not edges:
183 print(
184 f"check_core_layering.py: {len(targets)} ra8_core file(s) scanned, "
185 "no upward lib dependency."
186 )
187 return 0
188
189 print(
190 f"check_core_layering.py: {len(edges)} upward dependency(ies) out of {FOUNDATION_LIB}:\n",
191 file=sys.stderr,
192 )
193 for path, lineno, name, mods in edges:
194 print(f" {path}:{lineno} includes {name} (owned by {', '.join(mods)})", file=sys.stderr)
195 print(
196 f"\n{FOUNDATION_LIB} is the foundation layer: every other lib may build on\n"
197 "it, but it must depend on nothing above itself. Move the shared\n"
198 "declaration down into ra8_core, or keep the consumer in the higher lib.",
199 file=sys.stderr,
200 )
201 return 1

205 sys.exit(main(sys.argv))
