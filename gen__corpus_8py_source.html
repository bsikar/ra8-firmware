4"""Generate the ra8_viewer malformed-input security corpus.
5
6The viewer opens attacker-supplied archives (a .cbz pulled off a random manga
7host, a .jof produced by the downloader) and sizes scratch buffers from length
8fields inside them. This script emits a small corpus that exercises every
9untrusted-allocation guard the viewer added for issue #298, in BOTH directions:
10
11 * malicious fixtures that MUST be refused with a clean ra8_err_t (process exit
12 1), never an OOM, an abort, or a hang; and
13 * legitimate fixtures that MUST still decode (exit 0, a P6 PPM written).
14
15Everything here is pure Python standard library so the corpus regenerates on any
16CI runner with no third-party dependency (unlike make_fixture.py, which needs
17Pillow). Integers on disk are little-endian unless a format dictates otherwise.
18"""

30# --- shared sizing knobs (kept well clear of / above the viewer policy) ------
31# The viewer enforces a 64 MiB per-unit output cap and a 1024:1 ratio bound
32# (ra8_decomp_limits_default). These constants push a crafted fixture safely
33# past one of those bounds so the guard, not luck, decides the outcome.
34MIB = 1024 * 1024
35OVER_CAP_BYTES = 128 * MIB # > 64 MiB output cap -> k_ra8_err_decomp_output_cap
36BOMB_UNCOMP_BYTES = 50 * MIB # < cap but huge ratio -> k_ra8_err_decomp_ratio
37UNWRAP_BOMB_BYTES = 160 * MIB # > the 128 MiB gzip/xz unwrap arena
38PAGE_NAME = "page01.jpg" # ra8_comic treats this extension as a page image
39FILL_BYTE = 0x80 # decoded-pixel fill for generated atlases
40EXPECTED_ARGC = 2 # argv is: script, out_dir
41
42# --- JOF (RTA1 atlas) on-disk layout -----------------------------------------
43JOF_MAGIC_HDR = b"JOF1"
44JOF_MAGIC_FTR = b"JOFE"
45JOF_HDR_BYTES = 32
46JOF_FOOTER_BYTES = 16
47
48# --- ustar (tar) header field offsets ----------------------------------------
49TAR_SIZE_OFF = 124
50TAR_CHKSUM_OFF = 148
51TAR_MAGIC_OFF = 257
52TAR_BLOCK = 512

65def _ceil_div(a: int, b: int) -> int:
66 """Return ceil(a / b) for positive integers (the JOF grid rule)."""
67 return (a + b - 1) // b

70def build_jof(geom: JofGeom) -> bytes:
71 """Build a structurally-valid raw (codec 0) JOF atlas.
72
73 The header carries the DECLARED tile_w/tile_h, but each tile stream holds
74 only its edge-clamped payload (min(tile_w, width - x*tile_w) etc.), exactly
75 as ra8_jof_produce writes it. That split is the point of the giant-tiles
76 fixture: a 16x16 image can declare 65535x65535 tiles, so the file stays tiny
77 while the viewer's band_bytes = tile_w*tile_h*bpp balloons to ~17 GiB.
78
79 Args:
80 geom: Atlas geometry. tile_w/tile_h may exceed the image size (the header
81 does not clamp them); bpp is 1, 3 or 4.
82
83 Returns:
84 The complete atlas bytes (header + tiles + index + footer).
85 """
86 cols = _ceil_div(geom.width, geom.tile_w)
87 rows = _ceil_div(geom.height, geom.tile_h)
88 tile_count = cols * rows
89
90 body = io.BytesIO()
91 index: list[tuple[int, int]] = []
92 offset = JOF_HDR_BYTES
93 for ty in range(rows):
94 for tx in range(cols):
95 clamp_w = min(geom.tile_w, geom.width - tx * geom.tile_w)
96 clamp_h = min(geom.tile_h, geom.height - ty * geom.tile_h)
97 payload = bytes([FILL_BYTE]) * (clamp_w * clamp_h * geom.bpp)
98 body.write(payload)
99 index.append((offset, len(payload)))
100 offset += len(payload)
101
102 index_off = offset
103 header = struct.pack(
104 "<4sHHHHBBHI12x",
105 JOF_MAGIC_HDR,
106 geom.width,
107 geom.height,
108 geom.tile_w,
109 geom.tile_h,
110 geom.bpp,
111 0, # codec 0 = raw
112 0, # reserved u16
113 tile_count,
114 )
115 index_bytes = b"".join(struct.pack("<II", off, length) for off, length in index)
116 total_size = index_off + len(index_bytes) + JOF_FOOTER_BYTES
117 footer = struct.pack("<III4s", index_off, tile_count, total_size, JOF_MAGIC_FTR)
118 return header + body.getvalue() + index_bytes + footer

121def build_cbz(entry_name: str, data: bytes, forced_uncomp: int | None = None) -> bytes:
122 """Build a one-entry ZIP, optionally forging its declared uncompressed size.
123
124 zipfile writes the true sizes; when forced_uncomp is set, the central
125 directory's uncompressed-size field is patched afterwards. miniz's
126 mz_zip_reader_file_stat reads that field, so the viewer's comic backend sees
127 the lie at open and ra8_decomp_check_declared refuses it before any inflate.
128
129 Args:
130 entry_name: Archive member name (must look like a page image).
131 data: The real member bytes (kept tiny for a bomb).
132 forced_uncomp: Value to write into the central-directory uncompressed
133 size, or None to leave the honest size.
134
135 Returns:
136 The ZIP bytes.
137 """
138 buf = io.BytesIO()
139 with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
140 zf.writestr(entry_name, data)
141 raw = bytearray(buf.getvalue())
142 if forced_uncomp is not None:
143 sig = raw.find(b"PK\x01\x02") # central directory file header
144 if sig < 0:
145 msg = "no central directory in generated zip"
146 raise RuntimeError(msg)
147 # CDH layout: sig(4) verMade(2) verNeed(2) flags(2) method(2) time(2)
148 # date(2) crc(4) compSize(4) uncompSize(4) -> uncompressed at sig+24.
149 struct.pack_into("<I", raw, sig + 24, forced_uncomp & 0xFFFFFFFF)
150 return bytes(raw)

153def build_cbt(entry_name: str, data: bytes, forced_size: int | None = None) -> bytes:
154 """Build a one-member ustar tar, optionally forging the member size field.
155
156 The CBT (tar) walker has no open-time declared-size guard, so a forged size
157 is the viewer's own line of defence: viewer_read_page_bytes validates the
158 member's declared size against the archive length before reserving a page
159 buffer. The 512-byte header checksum is recomputed so the forged header is
160 otherwise well-formed.
161
162 Args:
163 entry_name: Member name (a page image name).
164 data: The real (tiny) member bytes.
165 forced_size: Octal size to force into the header, or None for the honest
166 size.
167
168 Returns:
169 The tar bytes (header + data + zero padding + two zero end blocks).
170 """
171 name = entry_name.encode("ascii")
172 header = bytearray(TAR_BLOCK)
173 header[0 : len(name)] = name
174 header[100:108] = b"0000644\x00" # mode
175 header[108:116] = b"0000000\x00" # uid
176 header[116:124] = b"0000000\x00" # gid
177 real_size = len(data)
178 size_field = forced_size if forced_size is not None else real_size
179 header[TAR_SIZE_OFF : TAR_SIZE_OFF + 12] = f"{size_field:011o}\x00".encode("ascii")
180 header[136:148] = b"00000000000\x00" # mtime
181 header[156:157] = b"0" # typeflag: regular file
182 header[TAR_MAGIC_OFF : TAR_MAGIC_OFF + 6] = b"ustar\x00"
183 header[263:265] = b"00"
184 header[TAR_CHKSUM_OFF : TAR_CHKSUM_OFF + 8] = b" " * 8 # spaces while summing
185 chksum = sum(header) & 0o777777
186 header[TAR_CHKSUM_OFF : TAR_CHKSUM_OFF + 8] = f"{chksum:06o}\x00 ".encode("ascii")
187
188 pad = (-real_size) % TAR_BLOCK
189 return bytes(header) + data + (b"\x00" * pad) + (b"\x00" * (2 * TAR_BLOCK))

192def _write(out_dir: Path, name: str, blob: bytes) -> None:
193 """Write blob to out_dir/name and report its size on stderr."""

198def main() -> int:
199 """Emit the whole corpus into the directory named by argv[1]."""
200 if len(sys.argv) != EXPECTED_ARGC:
201 sys.stderr.write("usage: gen_corpus.py <out_dir>\n")
202 return 2
203 out = Path(sys.argv[1])
204 out.mkdir(parents=True, exist_ok=True)
205
206 tiny = b"not a real image, refused before decode"
207
208 # --- malicious: comic declared-size / ratio bombs ------------------------
209 _write(out, "giant_decl.cbz", build_cbz(PAGE_NAME, tiny, forced_uncomp=OVER_CAP_BYTES))
210 _write(out, "zip_bomb.cbz", build_cbz(PAGE_NAME, tiny, forced_uncomp=BOMB_UNCOMP_BYTES))
211 _write(out, "giant_decl.cbt", build_cbt(PAGE_NAME, tiny, forced_size=OVER_CAP_BYTES))
212
213 # --- malicious: truncated containers -------------------------------------
214 full_cbz = build_cbz(PAGE_NAME, tiny)
215 _write(out, "truncated.cbz", full_cbz[: len(full_cbz) // 3])
216 full_cbt = build_cbt(PAGE_NAME, tiny)
217 _write(out, "truncated.cbt", full_cbt[:200])
218 _write(out, "garbage.cbr", b"Rar!\x1a\x07\x00" + b"\x00" * 64) # not a decodable RAR
219
220 # --- malicious: JOF absurd geometry / truncation -------------------------
221 _write(out, "giant_tiles.jof", build_jof(JofGeom(16, 16, 65535, 65535, 4)))
222 _write(out, "truncated.jof", build_jof(JofGeom(32, 32, 32, 32, 1))[:JOF_HDR_BYTES])
223
224 # --- malicious: gzip-wrapped comic that overflows the unwrap arena -------
225 _write(out, "unwrap_bomb.cbt.gz", gzip.compress(b"\x00" * UNWRAP_BOMB_BYTES, compresslevel=9))
226
227 # --- legitimate: must still decode ---------------------------------------
228 _write(out, "legit.jof", build_jof(JofGeom(32, 32, 32, 32, 1)))
229 return 0
