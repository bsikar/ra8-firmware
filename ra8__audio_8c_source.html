47ra8_err_t ra8_audio_frame_validate(const ra8_audio_frame_t* frame)
48{
49 if (frame == nullptr) {
50 return k_ra8_err_null_ptr;
51 }
52 if (frame->data == nullptr) {
53 return k_ra8_err_null_ptr;
54 }
55 const uint32_t container_bytes = internal_audio_container_bytes(frame->format);
56 if (container_bytes == 0U) {
57 return k_ra8_err_invalid_arg;
58 }
59 if (frame->sample_count == 0U) {
60 return k_ra8_err_invalid_arg;
61 }
62 if (frame->sample_rate_hz == 0U) {
63 return k_ra8_err_invalid_arg;
64 }
65 if (frame->channels == 0U) {
66 return k_ra8_err_invalid_arg;
67 }
68 if (frame->valid_bits == 0U) {
69 return k_ra8_err_invalid_arg;
70 }
71 if (frame->valid_bits > (container_bytes * 8U)) {
72 return k_ra8_err_invalid_arg;
73 }
74 const uint64_t required =
75 (uint64_t)frame->sample_count * (uint64_t)frame->channels * (uint64_t)container_bytes;
76 if (required > (uint64_t)UINT32_MAX) {
77 return k_ra8_err_invalid_size;
78 }
79 if (frame->bytes != (uint32_t)required) {
80 return k_ra8_err_invalid_size;
81 }
82 return k_ra8_ok;
83}

119RA8_INTERNAL static ra8_err_t internal_audio_frame_matches_info(const ra8_audio_frame_t* frame,
120 const ra8_audio_source_info_t* info)
121{
122 if (frame->bytes != info->frame_bytes) {
123 return k_ra8_err_invalid_state;
124 }
125 if (frame->sample_count != info->samples_per_frame) {
126 return k_ra8_err_invalid_state;
127 }
128 if (frame->sample_rate_hz != info->sample_rate_hz) {
129 return k_ra8_err_invalid_state;
130 }
131 if (frame->channels != info->channels) {
132 return k_ra8_err_invalid_state;
133 }
134 if (frame->valid_bits != info->valid_bits) {
135 return k_ra8_err_invalid_state;
136 }
137 return (frame->format == info->format) ? k_ra8_ok : k_ra8_err_invalid_state;
138}

190ra8_err_t ra8_audio_source_capture(ra8_audio_source_t* source,
191 const ra8_audio_buffer_t* buffer,
192 ra8_audio_frame_t* out_frame)
193{
194 const ra8_err_t args_valid = internal_audio_capture_args_valid(source, buffer, out_frame);
195 if (args_valid != k_ra8_ok) {
196 return args_valid;
197 }
198 ra8_audio_source_info_t info = {};
199 ra8_err_t err = source->iface->get_info(source->ctx, &info);
200 if (err != k_ra8_ok) {
201 return err;
202 }
203 if (buffer->capacity < info.frame_bytes) {
204 return k_ra8_err_invalid_size;
205 }
206 err = source->iface->capture(source->ctx, buffer, out_frame);
207 if (err != k_ra8_ok) {
208 *out_frame = (ra8_audio_frame_t){};
209 return err;
210 }
211 if (out_frame->data != buffer->data) {
212 *out_frame = (ra8_audio_frame_t){};
213 return k_ra8_err_invalid_state;
214 }
215 err = ra8_audio_frame_validate(out_frame);
216 if (err != k_ra8_ok) {
217 *out_frame = (ra8_audio_frame_t){};
218 return err;
219 }
220 err = internal_audio_frame_matches_info(out_frame, &info);
221 if (err != k_ra8_ok) {
222 *out_frame = (ra8_audio_frame_t){};
223 return err;
224 }
225 return k_ra8_ok;
226}
