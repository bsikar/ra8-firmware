169[[nodiscard]] RA8_INTERNAL static ra8_err_t
170internal_rtc_demo_arm_alarm(const ra8_rtc_datetime_t* now)
171{
172 ra8_rtc_datetime_t a = *now;
173 uint16_t s = (uint16_t)a.second + (uint16_t)k_rtc_demo_alarm_offset_s;
174 a.second = (uint8_t)(s % (uint16_t)k_rtc_demo_secs_per_min);
175 uint16_t add_min = (uint16_t)(s / (uint16_t)k_rtc_demo_secs_per_min);
176 uint16_t m = (uint16_t)a.minute + add_min;
177 a.minute = (uint8_t)(m % (uint16_t)k_rtc_demo_mins_per_hour);
178 uint16_t add_hr = (uint16_t)(m / (uint16_t)k_rtc_demo_mins_per_hour);
179 uint16_t h = (uint16_t)a.hour + add_hr;
180 a.hour = (uint8_t)(h % (uint16_t)k_rtc_demo_hours_per_day);
181 ra8_err_t err = ra8_rtc_set_alarm(&a);
182 if (err != k_ra8_ok) {
183 return err;
184 }
185 /* Enable BOTH alarm and periodic IRQ flags so the demo gates on
186 * whichever fires first. The periodic interval is selected by
187 * RCR1.PES inside ra8_rtc_init's default config; the bench cares
188 * only that ra8_rtc_get_status reports a periodic-or-alarm event. */
189 return ra8_rtc_set_irq_enable(
190 (uint8_t)((uint8_t)k_ra8_rtc_irq_alarm | (uint8_t)k_ra8_rtc_irq_periodic));
191}

193void main(void)
194{
195 internal_rtc_demo_setup_or_halt();
196 ra8_isr_globals_enable();
197
198 /* Boot banner -- emit before the RTC poll loop so the HIL host can
199 * confirm the firmware booted even when the sub-clock crystal is not
200 * running and the alarm-fired path never reaches its own write. */
201 (void)ra8_board_uart_console_write(s_rtc_demo_boot_msg,
202 (size_t)(sizeof(s_rtc_demo_boot_msg) - 1U));
203
204 while (1) {
205 ra8_rtc_datetime_t now = {};
206 if (ra8_rtc_get(&now) != k_ra8_ok) {
207 break;
208 }
209 if (internal_rtc_demo_arm_alarm(&now) != k_ra8_ok) {
210 break;
211 }
212
213 /* Poll for the periodic flag (or alarm as a fallback). On real
214 * silicon RCR1.PIE drives an NVIC line, but for this demo we
215 * just spin against the status bit. */
216 uint8_t status = 0U;
217 const uint8_t any_mask = (uint8_t)k_ra8_rtc_irq_alarm | (uint8_t)k_ra8_rtc_irq_periodic;
218 do {
219 ra8_delay_ms(k_rtc_demo_poll_ms);
220 if (ra8_rtc_get_status(&status) != k_ra8_ok) {
221 internal_rtc_demo_panic_halt();
222 }
223 } while ((status & any_mask) == 0U);
224
225 (void)ra8_board_uart_console_write(s_rtc_demo_log_msg,
226 (size_t)(sizeof(s_rtc_demo_log_msg) - 1U));
227 if (ra8_rtc_clear_status(
228 (uint8_t)((uint8_t)k_ra8_rtc_irq_alarm | (uint8_t)k_ra8_rtc_irq_periodic)) != k_ra8_ok) {
229 break;
230 }
231 ra8_delay_ms((uint32_t)k_rtc_demo_advance_secs * (uint32_t)k_rtc_demo_ms_per_sec);
232 }
233 internal_rtc_demo_panic_halt();
234}
