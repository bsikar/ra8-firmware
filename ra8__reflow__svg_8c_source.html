253int32_t ra8_svgp_num(const uint8_t* s, size_t len, size_t* i)
254{
255 while ((*i < len) && (ra8_svgp_ws((char)s[*i]) || (s[*i] == ','))) {
256 ++(*i);
257 }
258 int32_t sign = 1;
259 if ((*i < len) && ((s[*i] == '-') || (s[*i] == '+'))) {
260 sign = (s[*i] == '-') ? -1 : 1;
261 ++(*i);
262 }
263 int32_t v = 0;
264 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9')) {
265 v = (v * (int32_t)k_svg_dec) + (int32_t)(s[*i] - '0');
266 ++(*i);
267 }
268 /* Skip a fractional part (truncated). */
269 if ((*i < len) && (s[*i] == '.')) {
270 ++(*i);
271 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9')) {
272 ++(*i);
273 }
274 }
275 return v * sign;
276}

314static bool priv_attr_at(const uint8_t* s, size_t len, size_t at, const char* name)
315{
316 const size_t n = strlen(name);
317 if (!ra8_svgp_starts_ci(s, len, at, name)) {
318 return false;
319 }
320 if ((at > 0U) && !ra8_svgp_ws((char)s[at - 1U])) {
321 return false; /* must be preceded by whitespace */
322 }
323 const size_t after = at + n;
324 if (after >= len) {
325 return false;
326 }
327 return (s[after] == '=') || ra8_svgp_ws((char)s[after]);

362bool ra8_svgp_attr(const uint8_t* s, size_t len, const char* name, size_t* voff, size_t* vlen)
363{
364 /* Bounded: i advances by 1 each step, capped by len. */
365 for (size_t i = 0U; i < len; ++i) {
366 if (!priv_attr_at(s, len, i, name)) {
367 continue;
368 }
369 size_t j = i + strlen(name);
370 while ((j < len) && (s[j] != '=')) {
371 ++j;
372 }
373 ++j; /* past '=' */
374 while ((j < len) && ra8_svgp_ws((char)s[j])) {
375 ++j;
376 }
377 if ((j >= len) || ((s[j] != '"') && (s[j] != '\''))) {
378 return false;
379 }
380 const char q = (char)s[j];
381 ++j;
382 const size_t start = j;
383 while ((j < len) && (s[j] != (uint8_t)q)) {
384 ++j;
385 }
386 *voff = start;
387 *vlen = j - start;
388 return true;
389 }
390 return false;
391}

503static uint32_t priv_hex_color(const uint8_t* s, size_t len)
504{
505 uint32_t rgb = 0U;
506 if (len == (size_t)k_svg_hex6) {
507 for (size_t i = 0U; i < len; ++i) {
508 const uint8_t v = priv_hex((char)s[i]);
509 if (v >= (uint8_t)k_svg_hex_base) {
510 return (uint32_t)k_svg_no_paint;
511 }
512 rgb = (rgb << (uint32_t)k_svg_hex_nib) | (uint32_t)v;
513 }
514 return rgb;
515 }
516 if (len == (size_t)k_svg_hex3) {
517 for (size_t i = 0U; i < len; ++i) {
518 const uint8_t v = priv_hex((char)s[i]);
519 if (v >= (uint8_t)k_svg_hex_base) {
520 return (uint32_t)k_svg_no_paint;
521 }
522 rgb =
523 (rgb << (uint32_t)k_svg_hex_chan) | ((uint32_t)v << (uint32_t)k_svg_hex_nib) | (uint32_t)v;
524 }
525 return rgb;
526 }
527 return (uint32_t)k_svg_no_paint;
528}

663int32_t priv_match_grad(const svg_grads_t* grads, const uint8_t* val, size_t vlen)
664{
665 if ((grads == nullptr) || !ra8_svgp_starts_ci(val, vlen, 0U, "url(#")) {
666 return -1;
667 }
668 const size_t idoff = strlen("url(#");
669 size_t idend = idoff;
670 while ((idend < vlen) && (val[idend] != ')')) {
671 ++idend;
672 }
673 const size_t idlen = idend - idoff;
674 /* Bounded: <= grads->n (<= k_svg_grad_max) gradients. */
675 for (int32_t i = 0; i < grads->n; ++i) {
676 const char* gid = grads->g[i].id;
677 if ((strlen(gid) == idlen) && (memcmp(gid, &val[idoff], idlen) == 0)) {
678 return i;
679 }
680 }
681 return -1;
682}
