1#!/usr/bin/env python3
2# SPDX-License-Identifier: MIT
3# Copyright (c) 2026 Brighton Sikarskie
4"""Emit ONE ``compile_commands.json`` covering every cross-compiled first-party TU.
5
6WHY THIS EXISTS
7---------------
8``clang_tidy.sh`` parses against the HOST compile database produced by the unit
9test build. That database describes ``libs/``, ``src/``, ``tests/`` and
10``tools/`` and nothing else: no ``-mcpu=cortex-m85``, no per-app include
11directories, no vendored RTOS paths. Pointing clang-tidy at the firmware anyway
12was measured, not guessed -- 135 findings across 96 files, every one a
13``clang-diagnostic-error`` and not one an actionable style finding (#369).
14
15The fix is not a wider glob, it is a database that actually describes how those
16translation units compile. CMake already knows: it emits
17``compile_commands.json`` from any configure. This script performs the
18configures and merges their output into a single database keyed by real,
19absolute source paths.
20
21WHAT IT COVERS AND HOW IT FINDS IT
22----------------------------------
231. The unified RA8D2 cross-configure with every ``RA8_USE_*`` middleware option
24 forced ON, so apps that ``USES threadx`` / ``usbx`` / ``mbedtls`` are
25 configured instead of skipped. This alone accounts for the overwhelming
26 majority of firmware TUs.
272. Whatever the unified configure still leaves out, discovered by DIFFING the
28 result against ``git ls-files`` -- never from a hardcoded app list. For each
29 app directory still missing, the app's OWN ``Makefile`` is asked for its
30 configure line (``make -n``), which is then re-run into a scratch build
31 directory. That is how the RA8P1 tier (a different chip, a different
32 toolchain file) and ``ra8_cache_store_demo`` (LevelX standalone mode, which
33 is mutually exclusive with the LevelX the unified build selects) are picked
34 up without this file knowing either fact.
353. A last-resort derivation for TUs that NO configure compiles, described
36 below.
37
38Step 2 is the load-bearing one. A hardcoded residual list is the exact defect
39that #296 / #332 / #358 / #359 / #360 were each an instance of: a scan list
40that silently stops matching the tree. Deriving the residual means a new app,
41at any depth, in any tier, needing any middleware, is picked up the day it
42lands -- or fails this script loudly if it cannot be configured at all.
43
44TUs THAT NOTHING BUILDS
45-----------------------
46One first-party TU is compiled by no configure at all, for a reason that is a
47fact about the tree rather than a defect in the enumeration above:
48
49 * ``examples/.../dfu_copy_to_run/payload.c`` -- a freestanding image linked at
50 a fixed SRAM base by its own ``build_payload.sh``, never by CMake.
51
52Being unbuilt is not a licence to go unlinted: it is first-party firmware and
53CLAUDE.md holds it to the same bar. So a command is DERIVED from the nearest
54already-covered sibling TU -- same directory subtree, therefore the same
55middleware, the same board layer and the same include set -- and then VERIFIED
56by actually running the cross-compiler over the file with it.
57
58The verification is what separates this from guessing. A derived command that
59does not compile the file is a hard error here, so the fallback can never
60quietly hand clang-tidy a wrong command and let a bogus parse read as a clean
61lint. Nothing is exempted and no path is allowlisted; a file either gets a
62command that provably works or this script fails.
63
64THE COVERAGE ASSERTION IS THE POINT
65-----------------------------------
66``--check`` fails when any first-party cross-compiled ``.c`` is absent from the
67merged database. Without it this script would degrade exactly the way the thing
68it replaces did: quietly emitting a smaller database, over which clang-tidy
69would report a clean run. A database that covers less must be an error, not a
70faster pass.
71
72USAGE
73-----
74 build_cross_compile_db.py --selftest # assert the merge + gap logic fires
75 build_cross_compile_db.py -o DIR # write DIR/compile_commands.json
76 build_cross_compile_db.py -o DIR --check # ... and fail on any uncovered TU
77"""

102# Middleware switches the unified configure forces ON so that apps declaring
103# `USES <m>` are configured rather than skipped. Derived from the option()
104# declarations in cmake/, checked below -- a new cmake/<m>.cmake that this list
105# does not know about is a hard error, not a silently smaller database.
106#
107# RA8_USE_LEVELX_STANDALONE is deliberately absent: cmake/levelx_standalone.cmake
108# FATAL_ERRORs when RA8_USE_LEVELX is also set, because both compile the same
109# lx_nor_*.c. The app that wants it is picked up by the per-app residual pass.
110UNIFIED_MIDDLEWARE_OFF = ("RA8_USE_LEVELX_STANDALONE",)
111
112# Roots whose C is cross-compiled firmware, i.e. exactly the code the host
113# compile database cannot describe.
114FIRMWARE_ROOTS = ("examples/", "port/")
115
116# Vendored SOUP under a firmware root -- CLAUDE.md exempts it from first-party
117# standards, so it is not part of what this database must cover. None exists
118# today; kept as the extension point if a firmware root ever vendors SOUP.
119FIRMWARE_EXEMPT: tuple[str, ...] = ()
120
121# A firmware tree this size cannot legitimately collapse to a handful of TUs.
122# Fewer than this means a configure silently failed and the "merged" database
123# describes almost nothing -- over which clang-tidy would report a clean run.
124TU_FLOOR = 250
125
126# How many uncovered paths to print before truncating.
127MAX_SHOWN = 40
128
129# The absorb() selftest fixture plants exactly this many distinct source paths.
130EXPECTED_FIXTURE_ENTRIES = 2
131
132# How many donor compile commands to try for a TU that no configure builds.
133# Each probe is a real compilation, so this bounds the fallback's cost; the
134# ordering puts the plausible donors first, so a correct one is found in the
135# first few or the TU genuinely has no working command in the tree.
136MAX_DONOR_PROBES = 60

142def run(argv: list[str], cwd: Path | None = None) -> subprocess.CompletedProcess[str]:
143 """Run `argv`, capturing output. Never raises; callers inspect returncode.
144
145 A missing executable is reported as a failed run rather than an exception:
146 donor probing walks compile commands from several toolchains (Arm and
147 RISC-V), and only some of them are installed on any given machine. An
148 absent compiler must disqualify that donor, not abort the whole build.
149 """
150 try:
151 return subprocess.run( # noqa: S603 -- argv built from repo-local paths
152 argv,
153 cwd=str(cwd) if cwd else None,
154 capture_output=True,
155 text=True,
156 check=False,
157 )
158 except (FileNotFoundError, NotADirectoryError, PermissionError) as exc:
159 return subprocess.CompletedProcess(argv, returncode=127, stdout="", stderr=str(exc))

182def join_continuations(text: str) -> list[str]:
183 """Fold backslash-newline continuations so one shell command is one string.
184
185 ``make -n`` prints a recipe line exactly as written, continuations and all,
186 so the cmake configure invocation arrives split across several output lines.
187 """
188 return [line.strip() for line in text.replace("\\\n", " ").splitlines() if line.strip()]

191# ---------------------------------------------------------------------------
192# Compile database entries
193# ---------------------------------------------------------------------------
194@dataclass
195class Database:
196 """Accumulates compile-command entries keyed by absolute source path."""
197
198 entries: dict[str, dict] = field(default_factory=dict)
199
200 def absorb(self, build_dir: Path) -> int:
201 """Merge ``build_dir/compile_commands.json``. Returns entries added."""
202 path = build_dir / "compile_commands.json"
203 if not path.is_file():
204 return 0
205 added = 0
206 for entry in json.loads(path.read_text(encoding="utf-8")):
207 key = str(Path(entry["directory"], entry["file"]).resolve())
208 if key in self.entries:
209 continue
210 self.entries[key] = entry
211 added += 1
212 return added
213
214 def add(self, directory: Path, source: Path, argv: list[str]) -> None:
215 """Add one hand-assembled compile-database entry (used by the derived-command pass)."""
216 key = os.path.realpath(str(source))
217 self.entries.setdefault(
218 key,
219 {
220 "directory": str(directory),
221 "file": str(source),
222 "arguments": argv,
223 },
224 )
225
226 def covered(self) -> set[str]:
227 """Repo-relative paths of every covered source inside the repo."""
228 root = str(REPO_ROOT)
229 return {os.path.relpath(key, root) for key in self.entries if key.startswith(root + os.sep)}
230
231 def write(self, out_dir: Path) -> Path:
232 """Serialise the accumulated entries to compile_commands.json.
233
234 Writes the values of the entry map, not a list built alongside it, so
235 a translation unit compiled twice (two configurations of one source)
236 contributes exactly one entry -- clangd and clang-tidy both take the
237 first match and a duplicate would make which flags apply arbitrary.
238
239 Returns the path written.
240 """
241 out_dir.mkdir(parents=True, exist_ok=True)
242 path = out_dir / "compile_commands.json"
243 path.write_text(
244 json.dumps(list(self.entries.values()), indent=1) + "\n",
245 encoding="utf-8",
246 )
247 return path

253def known_middleware_options() -> list[str]:
254 """Every ``RA8_USE_*`` option cmake/ declares, read from cmake/ itself.
255
256 Read rather than listed so a new cmake/<middleware>.cmake is picked up
257 automatically; a hardcoded list here would be the same silently-stale scan
258 list this whole script exists to stop reintroducing.
259 """
260 names: set[str] = set()
261 pattern = re.compile(r"option\‍(\s*(RA8_USE_[A-Z0-9_]+)")
262 for cmake_file in sorted((REPO_ROOT / "cmake").glob("*.cmake")):
263 names.update(pattern.findall(cmake_file.read_text(encoding="utf-8")))
264 for match in pattern.findall((REPO_ROOT / "CMakeLists.txt").read_text(encoding="utf-8")):
265 names.add(match)
266 return sorted(names - set(UNIFIED_MIDDLEWARE_OFF))

269def configure_unified(build_dir: Path, verbose: bool) -> None:
270 """Configure the whole-tree RA8D2 cross build with all middleware ON."""
271 argv = [
272 "cmake",
273 "-B",
274 str(build_dir),
275 "-S",
276 str(REPO_ROOT),
277 f"-DCMAKE_TOOLCHAIN_FILE={REPO_ROOT / 'cmake' / 'toolchain-ra8d2.cmake'}",
278 "-DCMAKE_EXPORT_COMPILE_COMMANDS=ON",
279 "-Wno-dev",
280 ]
281 argv += [f"-D{name}=ON" for name in known_middleware_options()]
282 result = run(argv)
283 if result.returncode != 0 or not (build_dir / "compile_commands.json").is_file():
284 sys.stderr.write(result.stdout[-4000:])
285 sys.stderr.write(result.stderr[-4000:])
286 sys.stderr.write("ERROR: the unified RA8D2 cross-configure failed (see above).\n")
287 sys.exit(1)
288 if verbose:
289 print(f"[db] unified RA8D2 configure -> {build_dir}")

295def app_dir_for(rel: str) -> Path | None:
296 """Nearest ancestor directory of `rel` that is a standalone app.
297
298 An app directory is one holding BOTH a CMakeLists.txt and a Makefile: that
299 pair is what makes it configurable on its own terms.
300 """
301 current = (REPO_ROOT / rel).parent
302 while current != REPO_ROOT and REPO_ROOT in current.parents:
303 if (current / "CMakeLists.txt").is_file() and (current / "Makefile").is_file():
304 return current
305 current = current.parent
306 return None

309def configure_line_from_makefile(app_dir: Path) -> list[str] | None:
310 """The app's own cmake configure invocation, read out of its Makefile.
311
312 Asking ``make -n`` means the toolchain file and every extra ``-D`` come
313 from the app's Makefile rather than being restated here -- an app that
314 switches chip or flips a middleware switch needs no edit to this script.
315 """
316 # -B (--always-make) is load-bearing. Without it `make -n build` prints
317 # "Nothing to be done" whenever the app's ELF is already up to date -- which
318 # it is in any tree where the apps have been built, e.g. straight after the
319 # build-cross gate. The configure line then goes missing, the residual pass
320 # silently configures nothing, and the database comes back smaller. The
321 # coverage assertion catches that, but a checker whose answer depends on
322 # whether someone happened to build first is wrong regardless of whether
323 # something downstream notices.
324 dry = run(["make", "-n", "-B", "build"], cwd=app_dir)
325 if dry.returncode != 0:
326 return None
327 for line in join_continuations(dry.stdout):
328 if not line.startswith("cmake ") or " -B" not in line:
329 continue
330 try:
331 argv = shlex.split(line)
332 except ValueError:
333 return None
334 if "--build" in argv:
335 continue
336 return argv
337 return None

340def rewrite_build_dir(argv: list[str], build_dir: Path) -> list[str]:
341 """Point a configure argv at `build_dir` and make it export its database."""
342 out: list[str] = []
343 skip_next = False
344 for arg in argv:
345 if skip_next:
346 skip_next = False
347 continue
348 if arg == "-B":
349 skip_next = True
350 continue
351 if arg.startswith("-B") and len(arg) > len("-B"):
352 continue
353 if arg.startswith("-DCMAKE_EXPORT_COMPILE_COMMANDS"):
354 continue
355 out.append(arg)
356 out += ["-B", str(build_dir), "-DCMAKE_EXPORT_COMPILE_COMMANDS=ON"]
357 return out

360def configure_residual_apps(db: Database, scratch: Path, verbose: bool) -> list[str]:
361 """Configure each app still missing from `db`. Returns paths still uncovered."""
362 missing = sorted(firmware_sources() - db.covered())
363 seen: set[Path] = set()
364 for rel in missing:
365 app_dir = app_dir_for(rel)
366 if app_dir is None or app_dir in seen:
367 continue
368 seen.add(app_dir)
369 argv = configure_line_from_makefile(app_dir)
370 if argv is None:
371 continue
372 build_dir = scratch / f"app-{app_dir.name}"
373 result = run(rewrite_build_dir(argv, build_dir))
374 if result.returncode != 0:
375 if verbose:
376 print(f"[db] per-app configure FAILED for {app_dir.relative_to(REPO_ROOT)}")
377 continue
378 added = db.absorb(build_dir)
379 if verbose:
380 print(f"[db] per-app {app_dir.relative_to(REPO_ROOT)} -> +{added}")
381 return sorted(firmware_sources() - db.covered())

395def own_include_args(source: Path) -> list[str]:
396 """``-I`` flags for the include directories `source`'s own library declares.
397
398 Every library in this tree is laid out ``<lib>/inc`` + ``<lib>/src`` -- the
399 convention check_header_file_placement.py enforces -- and a port library
400 publishes ``<lib>/inc`` through target_include_directories(). A donor from
401 another library supplies the middleware and HAL context but cannot supply
402 that, so walk the TU's own ancestors and offer theirs.
403
404 This is reading the tree's structure, not inventing flags: a directory is
405 only added when it exists, and the result still has to compile the file.
406 """
407 args: list[str] = []
408 current = source.parent
409 while current == REPO_ROOT or REPO_ROOT in current.parents:
410 candidate = current / "inc"
411 if candidate.is_dir():
412 args.append(f"-I{candidate}")
413 if current == REPO_ROOT:
414 break
415 current = current.parent
416 return args

419def retarget(argv: list[str], source: Path, obj: Path) -> list[str]:
420 """Rewrite a sibling's compile command to compile `source` into `obj`."""
421 out: list[str] = []
422 skip_next = False
423 for arg in argv:
424 if skip_next:
425 skip_next = False
426 continue
427 if arg in ("-c", "-o"):
428 skip_next = arg == "-o"
429 continue
430 if arg.endswith((".c", ".o", ".obj")):
431 continue
432 out.append(arg)
433 return out + own_include_args(source) + ["-c", str(source), "-o", str(obj)]

436def shared_prefix_len(a: str, b: str) -> int:
437 """How many leading path components `a` and `b` have in common."""
438 left, right = a.split("/"), b.split("/")
439 count = 0
440 for one, other in zip(left, right):
441 if one != other:
442 break
443 count += 1
444 return count

447def candidate_donors(db: Database, rel: str) -> list[dict]:
448 """Covered entries ordered by how close they sit to `rel` in the tree.
449
450 Closeness is shared path depth, so the same app or the same port library
451 comes first, then the same tier, then the rest. Order is a PREFERENCE, not
452 an answer -- the caller proves a donor correct by compiling with it.
453 """
454 root = str(REPO_ROOT)
455 seen_dirs: set[Path] = set()
456 scored: list[tuple[int, str, dict]] = []
457 for key, entry in db.entries.items():
458 if not key.startswith(root + os.sep) or not key.endswith(".c"):
459 continue
460 parent = Path(key).parent
461 if parent in seen_dirs:
462 continue
463 seen_dirs.add(parent)
464 candidate_rel = os.path.relpath(key, root)
465 scored.append((-shared_prefix_len(rel, candidate_rel), candidate_rel, entry))
466 scored.sort()
467 return [entry for _, _, entry in scored[:MAX_DONOR_PROBES]]

470def derive_unbuilt(db: Database, scratch: Path, verbose: bool) -> list[str]:
471 """Give every still-uncovered TU a sibling-derived, compile-verified command.
472
473 A donor is accepted only once the cross-compiler has compiled the file with
474 its flags. Candidates are tried nearest-first; the first that compiles wins.
475 Guessing is therefore not possible -- either a provably working command is
476 found or the TU is reported uncovered and the run fails.
477 """
478 still: list[str] = []
479 for rel in sorted(firmware_sources() - db.covered()):
480 source = REPO_ROOT / rel
481 obj = scratch / (rel.replace("/", "_") + ".o")
482 accepted = False
483 last_error = "no covered TU was available to derive a command from"
484 for donor in candidate_donors(db, rel):
485 argv = retarget(entry_argv(donor), source, obj)
486 probe = run(argv, cwd=Path(donor["directory"]))
487 if probe.returncode != 0:
488 last_error = probe.stderr.strip()[-1200:]
489 continue
490 db.add(Path(donor["directory"]), source, argv)
491 accepted = True
492 if verbose:
493 donor_rel = os.path.relpath(donor["file"], REPO_ROOT)
494 print(f"[db] derived {rel} from {donor_rel} (compiles clean)")
495 break
496 if not accepted:
497 print(
498 f"ERROR: no compile command could be derived for {rel}.\n"
499 f" last attempt failed with:\n {last_error}",
500 file=sys.stderr,
501 )
502 still.append(rel)
503 return still

509def report_uncovered(uncovered: list[str], total: int) -> int:
510 """Print the coverage verdict. Returns the process exit code."""
511 covered = total - len(uncovered)
512 print(f"cross-compile database: {covered}/{total} first-party firmware TUs covered")
513 if not uncovered:
514 print("OK: every cross-compiled first-party .c has a compile command.")
515 return 0
516 print(file=sys.stderr)
517 print(f"ERROR: {len(uncovered)} firmware TU(s) have no compile command:", file=sys.stderr)
518 for rel in uncovered[:MAX_SHOWN]:
519 print(f" {rel}", file=sys.stderr)
520 if len(uncovered) > MAX_SHOWN:
521 print(f" ... and {len(uncovered) - MAX_SHOWN} more", file=sys.stderr)
522 print(file=sys.stderr)
523 print(
524 "Each needs a configure that reaches it. An app is picked up\n"
525 "automatically when its directory holds BOTH a CMakeLists.txt and a\n"
526 "Makefile whose `build` target configures it -- check that pair first.",
527 file=sys.stderr,
528 )
529 return 1

535def _selftest_database() -> list[str]:
536 """Assertions about Database merging and coverage accounting."""
537 failures: list[str] = []
538
539 # 1. absorb() merges, de-duplicates by real path, and reports what it added.
540 with tempfile.TemporaryDirectory() as tmp:
541 one = Path(tmp) / "one"
542 one.mkdir()
543 (one / "compile_commands.json").write_text(
544 json.dumps(
545 [
546 {"directory": str(REPO_ROOT), "file": "a.c", "arguments": ["cc", "a.c"]},
547 {"directory": str(REPO_ROOT), "file": "./a.c", "arguments": ["cc", "a.c"]},
548 {"directory": str(REPO_ROOT), "file": "b.c", "arguments": ["cc", "b.c"]},
549 ]
550 ),
551 encoding="utf-8",
552 )
553 db = Database()
554 added = db.absorb(one)
555 if added != EXPECTED_FIXTURE_ENTRIES:
556 failures.append(f"absorb() merged {added} entries, expected 2 after de-duplication")
557 if db.covered() != {"a.c", "b.c"}:
558 failures.append(f"covered() returned {sorted(db.covered())}, expected ['a.c', 'b.c']")
559
560 return failures

579def _selftest_command_building() -> list[str]:
580 """Assertions about option discovery and compile-command construction."""
581 failures: list[str] = []
582
583 # 3. The middleware sweep must actually find the vendored options, and must
584 # exclude the mutually-exclusive one. An empty sweep would silently
585 # configure away every `USES` app.
586 options = known_middleware_options()
587 if "RA8_USE_THREADX" not in options or "RA8_USE_USBX" not in options:
588 failures.append(f"known_middleware_options() missed a vendored switch: {options}")
589 if "RA8_USE_LEVELX_STANDALONE" in options:
590 failures.append("known_middleware_options() included the mutually-exclusive LevelX mode")
591
592 # 4. Continuation folding: a `make -n` recipe split over lines is ONE command.
593 # Asserted through shlex.split rather than on the raw string, because the
594 # exact run of intervening whitespace is not what callers depend on.
595 folded = join_continuations("cmake -S . \\\n -B build \\\n -DX=1\nnext\n")
596 expected = [["cmake", "-S", ".", "-B", "build", "-DX=1"], ["next"]]
597 if [shlex.split(line) for line in folded] != expected:
598 failures.append(f"join_continuations() produced {folded!r}")
599
600 # 5. rewrite_build_dir must repoint -B in both spellings and force the export.
601 rewritten = rewrite_build_dir(["cmake", "-S", ".", "-B", "/orig", "-DA=1"], Path("/new"))
602 if "/orig" in rewritten or "-B" not in rewritten or "/new" not in rewritten:
603 failures.append(f"rewrite_build_dir() left the original -B: {rewritten}")
604 if "-DCMAKE_EXPORT_COMPILE_COMMANDS=ON" not in rewritten:
605 failures.append("rewrite_build_dir() did not force the compile-command export")
606
607 # 6. retarget() must strip the donor's own source/object and substitute ours,
608 # otherwise the derived command would compile the DONOR and report a
609 # clean parse for a file it never opened.
610 donor_argv = ["arm-none-eabi-gcc", "-Iinc", "-c", "donor.c", "-o", "donor.o"]

615 failures.append(f"retarget() did not compile the requested file: {retargeted}")

617 failures.append("retarget() dropped the donor's include flags")

620 # "command" would otherwise derive an EMPTY compile line.

622 failures.append("entry_argv() mishandled the 'arguments' form")
623 if entry_argv({"command": "cc x.c"}) != ["cc", "x.c"]:
624 failures.append("entry_argv() mishandled the 'command' form")

652def main() -> int:

656 is what CI runs: it fails when the checked-in database has drifted from
657 the build system rather than silently rewriting it under the runner.
658

664 parser.add_argument("--check", action="store_true", help="fail on any uncovered firmware TU")
665 parser.add_argument("--selftest", action="store_true", help="assert this script still fires")
666 parser.add_argument("-v", "--verbose", action="store_true")
667 args = parser.parse_args()
668
669 if args.selftest:
670 return selftest()
671 if not args.out:
672 parser.error("--out is required unless --selftest is given")

685 "make clang-tidy report a clean run over almost nothing.",

687 )
688 return 1

690 # The CMake build trees live INSIDE the output directory and are kept, not
691 # thrown away: every entry's "directory" field names the tree it came from
692 # and clang-tidy chdir()s there before parsing. Pointing those at a
693 # temporary directory produced a hard LLVM abort once the run cleaned up.
694 # Keeping them also makes a re-run an incremental CMake reconfigure.

710 out_path = db.write(out_dir)
