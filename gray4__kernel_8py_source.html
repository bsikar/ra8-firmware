1"""Deterministic integer gray4 downscale kernel -- host mirror of the firmware.
2
3This is a pure-Python, integer-exact port of the on-device 4-bpp transcode in
4``libs/ra8_rabook_compile/src/ra8_rabook_gray4.c``:
5
6 * :func:`gray4_output_dims` mirrors ``ra8_rabook_gray4_output_dims``.
7 * :func:`gray4_downscale` mirrors ``ra8_rabook_gray4_downscale`` (the Q16.16
8 fixed-point bilinear resample).
9 * :func:`gray4_encode` mirrors ``ra8_rabook_gray4_encode`` (round-to-nearest
10 16-level quantise + two-nibbles-per-byte pack).
11
12Why this file exists (issue #213): the .rabook image pipeline downscales rasters
13with the firmware's integer bilinear kernel, but the desktop compiler used to
14resample with PIL LANCZOS. The two kernels produce different pixels, so a
15downscaled image was NOT byte-identical host-vs-device. Downscale is opt-in
16(default off, issue #210), yet whenever ``--max-edge`` is requested the divergence
17was reachable. ``epub_compile.py`` now calls this module for the resample step so
18the desktop tool and the device emit the same bytes; "green means correct" needs
19one deterministic kernel, not a documented exception.
20
21Every integer operation here matches the C exactly, including the truncating
22(floor) final shift and the edge-repeat clamp -- there is no floating point. The
23loops are O(dst_w * dst_h); this is fine for the opt-in offline compile and the
24tiny parity fixture, and clarity (a 1:1 mirror of the C) is worth more than speed.
25

32# Q16.16 fixed-point constants (mirror ra8_gray4_fp_t in ra8_rabook_gray4.c).
33_FP_SHIFT = 16
34_FP_UNIT = 1 << _FP_SHIFT # 1.0 in Q16.16 (65536).
35_FP_MASK = _FP_UNIT - 1 # Fractional-bits mask (low 16 bits).
36_FP_NORM_SHIFT = 2 * _FP_SHIFT # Right-shift to normalise the 64-bit product.
37_BYTE_MASK = 0xFF # (uint8_t) truncation of the normalised sample.
38
39# 16-level even-palette quantise constants (mirror ra8_rabook_gray4_consts_t).
40_QUANT_DIV = 17 # Palette step: 255 / 15 == 17.
41_ROUND_HALF = 8 # Added before the divide to round-to-nearest.
42_NIB_MAX = 15 # Maximum nibble value (4 bits unsigned).
43_NIB_SHIFT = 4 # Even pixel occupies the high nibble.
44_NIB_PER_BYTE = 2 # Pixels packed per output byte.
45
46# RGB -> gray8 luma coefficients (mirror stbi__compute_y in
47# libs/third_party/stb/stb_image.h). The DEVICE decodes every raster with
48# stb_image at req_comp=1, so the host MUST fold colour the same way or the same
49# source compiles to different .rabook bytes host-vs-device (issue #337). These
50# are stb's exact integer weights and its truncating >>8 -- NOT PIL's ITU-R
51# 601-2 convert("L"), which rounds and weights green/blue differently and so
52# disagrees with the device on ~48% of the RGB cube.
53_LUMA_R = 77 # Red weight (stbi__compute_y).
54_LUMA_G = 150 # Green weight (stbi__compute_y).
55_LUMA_B = 29 # Blue weight (stbi__compute_y).
56_LUMA_SHIFT = 8 # Normalising right-shift (truncating, matches stb).

59def stb_compute_y(r: int, g: int, b: int) -> int:
60 """Fold one RGB triple to an 8-bit gray value, byte-identical to stb_image.
61
62 Exact mirror of ``stbi__compute_y`` in ``libs/third_party/stb/stb_image.h``:
63 ``(r*77 + g*150 + b*29) >> 8`` with a truncating shift. This is the single
64 host luma the device shares; using PIL's ``convert("L")`` here instead would
65 diverge from the on-device stb decode on most colour inputs (issue #337).
66
67 Args:
68 r: Red channel, 0-255.
69 g: Green channel, 0-255.
70 b: Blue channel, 0-255.
71
72 Returns:
73 The stb gray8 value, 0-255.
74 """
75 return ((r * _LUMA_R) + (g * _LUMA_G) + (b * _LUMA_B)) >> _LUMA_SHIFT

78def gray4_output_dims(src_w: int, src_h: int, max_edge: int) -> tuple[int, int]:
79 """Scaled output dims keeping the longer edge within ``max_edge``.
80
81 Exact mirror of ``ra8_rabook_gray4_output_dims``: returns the source dims
82 unchanged when they already fit, otherwise scales both by
83 ``max_edge / longer_edge`` with round-half-up integer division (matching the C
84 ``(dim * max_edge + longer / 2) / longer``), clamping each result to >= 1.
85
86 :param src_w: Source width in pixels.
87 :param src_h: Source height in pixels.
88 :param max_edge: Maximum allowed length of the longer edge (0 -> (0, 0)).
89 :returns: ``(out_w, out_h)``; ``(0, 0)`` when any input is 0.
90 """
91 if src_w == 0 or src_h == 0 or max_edge == 0:
92 return (0, 0)
93 longer = max(src_w, src_h)
94 if longer <= max_edge:
95 return (src_w, src_h)
96 half_longer = longer // 2
97 out_w = ((src_w * max_edge) + half_longer) // longer
98 out_h = ((src_h * max_edge) + half_longer) // longer
99 return (max(1, out_w), max(1, out_h))

102def _bilinear_sample(src: bytes, src_w: int, src_h: int, sx_fp: int, sy_fp: int) -> int:
103 """One bilinear-interpolated sample at Q16.16 source point (sx_fp, sy_fp).
104
105 Exact mirror of ``s_bilinear_sample``: integer bits pick the top-left corner,
106 the fractional bits weight the four neighbours, the right/bottom neighbours are
107 edge-clamped, and the accumulated 64-bit product is normalised by a truncating
108 (floor) right-shift.
109 """
110 sx0 = sx_fp >> _FP_SHIFT
111 sy0 = sy_fp >> _FP_SHIFT
112 sx1 = sx0 + 1 if sx0 + 1 < src_w else src_w - 1
113 sy1 = sy0 + 1 if sy0 + 1 < src_h else src_h - 1
114 fx = sx_fp & _FP_MASK
115 fy = sy_fp & _FP_MASK
116 ifx = _FP_UNIT - fx
117 ify = _FP_UNIT - fy
118 row0 = sy0 * src_w
119 row1 = sy1 * src_w
120 p00 = src[row0 + sx0]
121 p10 = src[row0 + sx1]
122 p01 = src[row1 + sx0]
123 p11 = src[row1 + sx1]
124 val = (p00 * ifx * ify) + (p10 * fx * ify) + (p01 * ifx * fy) + (p11 * fx * fy)
125 return (val >> _FP_NORM_SHIFT) & _BYTE_MASK

128def gray4_downscale(src: bytes, src_w: int, src_h: int, dst_w: int, dst_h: int) -> bytes:
129 """Bilinear-resample an 8-bit gray image from (src_w x src_h) to (dst_w x dst_h).
130
131 Exact mirror of ``ra8_rabook_gray4_downscale``: a left-aligned sample grid
132 (``sx_fp = dx * ((src_w << 16) // dst_w)``) fed through :func:`_bilinear_sample`.
133 An all-zero source yields an all-zero destination, matching the C.
134
135 :param src: Source gray pixels, ``src_w * src_h`` bytes, row-major.
136 :param src_w: Source width (>= 0).
137 :param src_h: Source height (>= 0).
138 :param dst_w: Destination width (> 0).
139 :param dst_h: Destination height (> 0).
140 :returns: ``dst_w * dst_h`` gray bytes, row-major.
141 :raises ValueError: if ``dst_w`` or ``dst_h`` is 0.
142 """
143 if dst_w == 0 or dst_h == 0:
144 msg = "dst_w and dst_h must be > 0"
145 raise ValueError(msg)
146 if src_w == 0 or src_h == 0:
147 return bytes(dst_w * dst_h)
148 x_step = (src_w << _FP_SHIFT) // dst_w
149 y_step = (src_h << _FP_SHIFT) // dst_h
150 out = bytearray(dst_w * dst_h)
151 for dy in range(dst_h):
152 sy_fp = dy * y_step
153 base = dy * dst_w
154 for dx in range(dst_w):
155 out[base + dx] = _bilinear_sample(src, src_w, src_h, dx * x_step, sy_fp)
156 return bytes(out)

159def gray4_nibble(value: int) -> int:
160 """Quantise one 0-255 gray value to a 0-15 nibble (round-to-nearest, clamped).
161
162 Exact mirror of the per-pixel rule in ``s_pack_nibbles``:
163 ``n = min((v + 8) // 17, 15)``.
164 """
165 nib = (value + _ROUND_HALF) // _QUANT_DIV
166 return min(nib, _NIB_MAX)

169def gray4_encode(gray: bytes, width: int, height: int) -> bytes:
170 """Quantise + nibble-pack a gray buffer to 4-bpp (two pixels per byte).
171
172 Exact mirror of ``ra8_rabook_gray4_encode`` / ``s_pack_nibbles``: even pixels
173 land in the high nibble, odd pixels in the low nibble; an odd final pixel keeps
174 a zero low half. Output size is ``(width * height + 1) // 2`` bytes.
175 """
176 n_pixels = width * height
177 out = bytearray((n_pixels + 1) // _NIB_PER_BYTE)
178 for i in range(n_pixels):
179 nib = gray4_nibble(gray[i])
180 byte_idx = i // _NIB_PER_BYTE
181 if (i & 1) == 0:
182 out[byte_idx] = nib << _NIB_SHIFT
183 else:
184 out[byte_idx] |= nib
185 return bytes(out)

188def gray4_transcode(src: bytes, src_w: int, src_h: int, max_edge: int) -> tuple[int, int, bytes]:
189 """Full opt-in transcode: output-dims -> downscale -> quantise/pack.
190
191 Convenience wrapper used by the parity generator to reproduce, in one call, the
192 exact bytes the device emits for a downscaled raster. ``max_edge == 0`` (or a
193 source already within it) means no resample -- the source pixels are packed as
194 is, so the default no-downscale path is covered too.
195
196 :param src: Source gray pixels, ``src_w * src_h`` bytes, row-major.
197 :param src_w: Source width in pixels (> 0).
198 :param src_h: Source height in pixels (> 0).
199 :param max_edge: Opt-in long-edge clamp (0 disables downscaling).
200 :returns: ``(out_w, out_h, packed_4bpp_bytes)``.
201 """
202 out_w, out_h = (src_w, src_h)
203 gray = src
204 if max_edge != 0:
205 out_w, out_h = gray4_output_dims(src_w, src_h, max_edge)
206 if (out_w, out_h) != (src_w, src_h):
207 gray = gray4_downscale(src, src_w, src_h, out_w, out_h)
208 return (out_w, out_h, gray4_encode(gray, out_w, out_h))
