237static void internal_enable_fault_handlers(void)
238{
239 /* ARMv8-M SCB->SHCSR: MEMFAULTENA[16], BUSFAULTENA[17], USGFAULTENA[18],
240 * SECUREFAULTENA[19] take MemManage/BusFault/UsageFault/SecureFault to
241 * their own handlers instead of escalating to HardFault, so
242 * ra8_exception_report sees the real class. SECUREFAULTENA is a
243 * Secure-bank bit (RES0 from NS); this boot executes in the Secure state. */
244 enum : uint32_t {
245 k_ra8_shcsr_memfaultena = 1UL << 16,
246 k_ra8_shcsr_busfaultena = 1UL << 17,
247 k_ra8_shcsr_usgfaultena = 1UL << 18,
248 k_ra8_shcsr_securefaultena = 1UL << 19,
249 };
250 uint32_t shcsr = ra8_boot_read32(k_ra8_scb_shcsr_addr);
251 shcsr |= k_ra8_shcsr_memfaultena | k_ra8_shcsr_busfaultena | k_ra8_shcsr_usgfaultena |
252 k_ra8_shcsr_securefaultena;
253 ra8_boot_write32(k_ra8_scb_shcsr_addr, shcsr);
254 ra8_boot_dsb();
255 ra8_boot_isb();
256}

309/* =============================================================================
310 * Public entry point
311 * =============================================================================
312 *
313 * Called from `Reset_Handler` before the .data copy and .bss zero.
314 * Must therefore not touch any global variables -- everything here
315 * runs on the stack and writes to CPU / SCB memory only. The MPU HAL call
316 * reads a `const` .rodata table only, so it is safe in this window.
317 */

319/* NOLINTNEXTLINE(readability-identifier-naming) -- CMSIS-mandated name. */
320void SystemInit(void)
321{
322 /* Mask global IRQs so the application gets a deterministic init
323 * window. main() must call ``ra8_isr_globals_enable`` once every
324 * driver / handler is wired up before any IRQ source can fire. */
325 ra8_boot_disable_irq();
326
327 internal_set_vtor();
328 internal_enable_fpu();
329 internal_enable_fpu_lazy_stack();
330 /* Always-on fault diagnostics: take MemManage/BusFault/UsageFault/
331 * SecureFault to their own decoded handlers and trap divide-by-zero
332 * as a UsageFault (independent of the cache / MPU build flag). */
333 internal_enable_fault_handlers();
334 internal_enable_div0_trap();
335#ifdef RA8_BOOT_ENABLE_CACHE_MPU
336 /* #576 demo: install the MPU memory-attribute map through the ra8_mpu HAL
337 * (ra8_mpu_apply_boot_map) instead of the raw MAIR/RBAR/RLAR/CTRL pokes the
338 * shared board system_init.c uses -- the two boot paths are otherwise
339 * identical and diff cleanly on HIL. The map goes down FIRST (so the D-cache
340 * sees Device-nGnRE MMIO and the non-cacheable shared SRAM, not Normal
341 * cacheable), then the caches. Enable the caches only if the map installed. */
342 if (ra8_mpu_apply_boot_map() == k_ra8_ok) {
343 internal_enable_icache();
344 internal_enable_dcache();
345 internal_enable_branch_predictor();
346 }
347#else
348 /* Default OFF: caches + MPU stay disabled -- no behaviour change. */
349 (void)internal_enable_icache;
350 (void)internal_enable_dcache;
351 (void)internal_enable_branch_predictor;
352#endif
353 (void)ra8_trustzone_init; /* TrustZone has its own gate (RA8_TRUSTZONE_ENABLE). */
354 internal_set_priority_grouping();
355}
