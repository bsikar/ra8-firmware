96RA8_INTERNAL
97static ra8_ui_rect_t internal_rect_union(ra8_ui_rect_t acc, ra8_ui_rect_t r)
98{
99 if (internal_rect_empty(&acc)) {
100 return r;
101 }
102 if (internal_rect_empty(&r)) {
103 return acc;
104 }
105 const int32_t x0 = internal_min_i32(acc.x, r.x);
106 const int32_t y0 = internal_min_i32(acc.y, r.y);
107 const int32_t x1 = internal_max_i32(acc.x + acc.w, r.x + r.w);
108 const int32_t y1 = internal_max_i32(acc.y + acc.h, r.y + r.h);
109 const ra8_ui_rect_t out = {.x = x0, .y = y0, .w = x1 - x0, .h = y1 - y0};
110 return out;
111}

168static ra8_err_t internal_build_stack_tree(const ra8_widget_t* widgets,
169 uint16_t count,
170 ra8_widget_axis_t axis,
171 int16_t gap,
172 int16_t pad,
173 ra8_box_t* scratch,
174 uint16_t cap,
175 ra8_box_tree_t* out_tree,
176 int16_t* out_root)
177{
178 const uint16_t vis = internal_visible_count(widgets, count);
179 if ((uint32_t)cap < ((uint32_t)vis + 1U)) {
180 return k_ra8_err_invalid_arg;
181 }
182 const ra8_err_t ierr = ra8_box_tree_init(out_tree, scratch, cap);
183 if (ierr != k_ra8_ok) {
184 return ierr;
185 }
186 ra8_box_t container = {};
187 container.kind =
188 (axis == k_ra8_widget_axis_row) ? (uint8_t)k_ra8_box_stack_h : (uint8_t)k_ra8_box_stack_v;
189 container.pad = pad;
190 container.gap = gap;
191 container.flex = 1U;
192 container.tag = (int16_t)k_ra8_box_none;
193 *out_root = ra8_box_add(out_tree, (int16_t)k_ra8_box_none, &container);
194 if (*out_root == (int16_t)k_ra8_box_none) {
195 return k_ra8_err_invalid_arg;
196 }
197 for (uint16_t i = 0U; i < count; ++i) {
198 if (!widgets[i].visible) {
199 continue;
200 }
201 ra8_box_t leaf = {};
202 leaf.kind = (uint8_t)k_ra8_box_leaf;
203 leaf.fixed = widgets[i].fixed;
204 leaf.flex = widgets[i].flex;
205 leaf.tag = (int16_t)widgets[i].action_id;
206 if (ra8_box_add(out_tree, *out_root, &leaf) == (int16_t)k_ra8_box_none) {
207 return k_ra8_err_invalid_arg;
208 }
209 }
210 return k_ra8_ok;
211}

213[[nodiscard]] ra8_err_t ra8_widget_layout_stack(ra8_widget_t* widgets,
214 uint16_t count,
215 const ra8_ui_rect_t* frame,
216 ra8_widget_axis_t axis,
217 int16_t gap,
218 int16_t pad,
219 ra8_box_t* box_scratch,
220 uint16_t box_cap)
221{
222 RA8_CHECK_NULL_PTR(widgets, s_tag, "widgets must not be nullptr");
223 RA8_CHECK_NULL_PTR(frame, s_tag, "frame must not be nullptr");
224 RA8_CHECK_NULL_PTR(box_scratch, s_tag, "box_scratch must not be nullptr");
225
226 ra8_box_tree_t tree = {};
227 int16_t root = (int16_t)k_ra8_box_none;
228 const ra8_err_t berr =
229 internal_build_stack_tree(widgets, count, axis, gap, pad, box_scratch, box_cap, &tree, &root);
230 if (berr != k_ra8_ok) {
231 return berr;
232 }
233 const ra8_err_t lerr = ra8_box_layout(&tree, root, frame);
234 if (lerr != k_ra8_ok) {
235 return lerr;
236 }
237
238 /* Box nodes 1..vis are the visible children in add order. */
239 uint16_t box_idx = 1U;
240 for (uint16_t i = 0U; i < count; ++i) {
241 if (!widgets[i].visible) {
242 continue;
243 }
244 widgets[i].rect = box_scratch[box_idx].rect;
245 ++box_idx;
246 }
247 return k_ra8_ok;
248}

250[[nodiscard]] ra8_err_t ra8_widget_dispatch(ra8_widget_t* widgets,
251 uint16_t count,
252 const ra8_widget_event_t* ev,
253 bool* out_handled)
254{
255 RA8_CHECK_NULL_PTR(ev, s_tag, "ev must not be nullptr");
256 RA8_CHECK_NULL_PTR(out_handled, s_tag, "out_handled must not be nullptr");
257 if ((count > 0U) && (widgets == nullptr)) {
258 return k_ra8_err_null_ptr;
259 }
260 *out_handled = false;
261
262 for (uint16_t i = 0U; i < count; ++i) {
263 ra8_widget_t* w = &widgets[i];
264 if (!w->visible || (w->vt == nullptr) || (w->vt->on_input == nullptr)) {
265 continue;
266 }
267 if (ev->kind == k_ra8_widget_ev_touch) {
268 /* Touch: only the widget under the point is offered the event. */
269 if (!ra8_ui_rect_contains(&w->rect, ev->x, ev->y)) {
270 continue;
271 }
272 *out_handled = w->vt->on_input(w, ev);
273 return k_ra8_ok;
274 }
275 /* Button: offer to each visible widget until one consumes it. */
276 if (w->vt->on_input(w, ev)) {
277 *out_handled = true;
278 return k_ra8_ok;
279 }
280 }
281 return k_ra8_ok;
282}

297[[nodiscard]] ra8_err_t ra8_widget_damage(const ra8_widget_t* widgets,
298 uint16_t count,
299 ra8_ui_rect_t* out_rect,
300 ra8_widget_refresh_t* out_hint,
301 uint16_t* out_count)
302{
303 RA8_CHECK_NULL_PTR(out_rect, s_tag, "out_rect must not be nullptr");
304 RA8_CHECK_NULL_PTR(out_hint, s_tag, "out_hint must not be nullptr");
305 RA8_CHECK_NULL_PTR(out_count, s_tag, "out_count must not be nullptr");
306 if ((count > 0U) && (widgets == nullptr)) {
307 return k_ra8_err_null_ptr;
308 }
309
310 ra8_ui_rect_t acc = {.x = 0, .y = 0, .w = 0, .h = 0};
311 uint8_t hint = (uint8_t)k_ra8_widget_refresh_none;
312 uint16_t dirty = 0U;
313 for (uint16_t i = 0U; i < count; ++i) {
314 if (!widgets[i].visible || !widgets[i].dirty) {
315 continue;
316 }
317 acc = internal_rect_union(acc, widgets[i].rect);
318 if (widgets[i].refresh > hint) {
319 hint = widgets[i].refresh;
320 }
321 ++dirty;
322 }
323 *out_rect = acc;
324 *out_hint = (ra8_widget_refresh_t)hint;
325 *out_count = dirty;
326 return k_ra8_ok;
327}

329[[nodiscard]] ra8_err_t ra8_widget_render_dirty(ra8_widget_t* widgets, uint16_t count)
330{
331 if ((count > 0U) && (widgets == nullptr)) {
332 return k_ra8_err_null_ptr;
333 }
334 for (uint16_t i = 0U; i < count; ++i) {
335 ra8_widget_t* w = &widgets[i];
336 if (!w->visible || !w->dirty) {
337 continue;
338 }

344 }
