195[[nodiscard]] static ra8_err_t canfd_demo_one_round_trip(uint8_t seq)
196{
197 ra8_canfd_frame_t tx = {
198 .id = (uint32_t)k_canfd_demo_id,
199 .dlc = (uint8_t)k_canfd_demo_dlc,
200 .is_extended = 0U,
201 .is_fd = 0U,
202 .is_brs = 0U,
203 .data = {seq,
204 (uint8_t)k_canfd_demo_payload_b1,
205 (uint8_t)k_canfd_demo_payload_b2,
206 (uint8_t)k_canfd_demo_payload_b3,
207 (uint8_t)k_canfd_demo_payload_b4,
208 (uint8_t)k_canfd_demo_payload_b5,
209 (uint8_t)k_canfd_demo_payload_b6,
210 (uint8_t)k_canfd_demo_payload_b7},
211 };
212 if (ra8_canfd_transmit((uint8_t)k_canfd_demo_channel, &tx) != k_ra8_ok) {
213 return k_ra8_err_hw_error;
214 }
215 /* Poll for the loopback frame: 500 kbit/s + 8 data bytes ~ 240 us
216 * round-trip, so wait up to ~10 ms before declaring no_data. */
217 ra8_canfd_frame_t rx = {};
218 for (uint32_t i = 0U; i < (uint32_t)k_canfd_demo_rx_spin; i++) {
219 if (ra8_canfd_receive((uint8_t)k_canfd_demo_channel, &rx) == k_ra8_ok) {
220 return k_ra8_ok;
221 }
222 }
223 return k_ra8_err_no_data;
224}

228int32_t main(void)
229{
230 canfd_demo_setup_or_halt();
231 ra8_isr_globals_enable();
232
233 uint8_t seq = 0U;
234 while (1) {
235 if (canfd_demo_one_round_trip(seq) == k_ra8_ok) {
236 (void)ra8_board_led_toggle(k_ra8_board_led1);
237 g_canfd_match += 1U;
238 } else {
239 (void)ra8_board_led_toggle(k_ra8_board_led2);
240 g_canfd_mismatch += 1U;
241 }
242 seq++;
243 ra8_delay_ms(k_canfd_demo_period_ms);
244 }
245 canfd_demo_panic_halt();
246 return 0;
247}
