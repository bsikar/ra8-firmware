117RA8_INTERNAL
118static uint32_t internal_ra8_ipc_event_to_clr(uint32_t event_mask)
119{
120 uint32_t clr = 0U;
121 /* HUM Ch 3.2.14 "IPC0CLR0" p 216 -- CLR7..CLR0 share bit positions
122 * with IRQ7..IRQ0 in STA. */
123 clr |= (event_mask & k_ra8_ipc_clr_mask_irq_all);
124 if ((event_mask & k_ra8_ipc_event_msg_ready) != 0U) {
125 /* HUM Ch 3.2.14 "RST bit (FIFO Reset)" p 217 -- only way to
126 * drop STA.RDY without consuming a word. */
127 clr |= k_ra8_ipc_clr_mask_rst;
128 }
129 if ((event_mask & k_ra8_ipc_event_err_empty) != 0U) {
130 /* HUM Ch 3.2.14 "RCLR bit" p 217 */
131 clr |= k_ra8_ipc_clr_mask_rclr;
132 }
133 if ((event_mask & k_ra8_ipc_event_err_full) != 0U) {
134 /* HUM Ch 3.2.14 "FCLR bit" p 217 */
135 clr |= k_ra8_ipc_clr_mask_fclr;
136 }
137 return clr;
138}

143RA8_INTERNAL
144static volatile r_ipc_channel_regs_t* internal_ra8_ipc_get_regs(uint8_t channel)
145{
146 if ((uint16_t)channel >= (uint16_t)k_ra8_ipc_channel_count) {
147 return nullptr;
148 }
149#ifdef RA8_BUILD_FOR_CPU1
150 /* CPU1 (M33) has SECEXT disabled and runs as a permanent NS bus */ /* LEGACY-OK: ARMv8-M */
151 /* controller. The IPC channels CPU1 owns (ch0 + ch2) are
152 * NS-attributed via IPCSAR=0x00050000 set by CPU0's secure-boot.
153 * The chip routes NS IPC accesses through the bit-28-set NS alias
154 * (0x50020000), not the Secure alias the M85 HAL defaults to.
155 * Without this offset the first ``reg->CLR`` write inside
156 * ra8_ipc_init BusFaults and CPU1 wedges in cpu1_fault_handler
157 * (HUM Ch 3.2 p 205 + bench probe pinning PC at 0x020C0020). */
158 const uintptr_t ns_offset = k_ipc_ns_alias_offset;
159 return (volatile r_ipc_channel_regs_t*)((uintptr_t)ra8_ipc_channel(channel) + ns_offset);
160#else
161 return ra8_ipc_channel(channel);
162#endif
163}

209RA8_INTERNAL
210static void internal_ra8_ipc_dispatch_irq_lines(uint8_t channel, uint32_t mask)
211{
212 const uint32_t irq_bits = mask & k_ra8_ipc_event_irq_all;
213 if (irq_bits == 0U) {
214 return;
215 }
216 ra8_ipc_channel_state_t* const state = &s_ipc_channels[channel];
217 for (uint8_t i = 0U; i < k_ra8_ipc_irq_event_count; ++i) {
218 const uint32_t bit = (uint32_t)1U << (uint32_t)i;
219 if ((irq_bits & bit) == 0U) {
220 continue;
221 }
222 const ra8_ipc_irq_fn_t fn = state->per_event[i].fn;
223 if (fn != nullptr) {
224 fn(state->per_event[i].ctx, channel, (ra8_ipc_irq_event_id_t)i);
225 }
226 }
227}

234[[nodiscard]] ra8_err_t ra8_ipc_init(const ra8_ipc_config_t* cfg)
235{
236 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
237 if ((uint16_t)cfg->channel >= (uint16_t)k_ra8_ipc_channel_count) {
238 return k_ra8_err_invalid_arg;
239 }
240
241 volatile r_ipc_channel_regs_t* reg = internal_ra8_ipc_get_regs(cfg->channel);
242 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
243
244 if (cfg->reset_fifo) {
245 /* HUM Ch 3.2.14 "IPC0CLR0" p 216 -- CLR.RST (bit 16) drops the
246 * FIFO and clears STA.RDY/STA.FULL. */
247 reg->CLR = k_ra8_ipc_clr_mask_rst;
248 }
249 if (cfg->clear_status) {
250 /* HUM Ch 3.2.14 "IPC0CLR0" p 216 -- ack every IRQn and both
251 * RERR/FERR error flags so init returns with a clean slate. */
252 reg->CLR = k_ra8_ipc_clr_mask_irq_all | k_ra8_ipc_clr_mask_rclr | k_ra8_ipc_clr_mask_fclr;
253 }
254
255 ra8_ipc_channel_state_t* const state = &s_ipc_channels[cfg->channel];
256 state->event_mask = cfg->event_mask;
257 state->active = true;
258 for (uint8_t i = 0U; i < k_ra8_ipc_irq_event_count; ++i) {
259 state->per_event[i].fn = nullptr;
260 state->per_event[i].ctx = nullptr;
261 }
262
263 ra8_log_info_val(s_tag, "ipc_init channel", (uint32_t)cfg->channel);
264 return k_ra8_ok;
265}

361[[nodiscard]] ra8_err_t ra8_ipc_send_event(uint8_t channel, ra8_ipc_irq_event_id_t event_id)
362{
363 if ((uint16_t)channel >= (uint16_t)k_ra8_ipc_channel_count) {
364 return k_ra8_err_invalid_arg;
365 }
366 if ((uint16_t)event_id >= (uint16_t)k_ra8_ipc_irq_event_count) {
367 return k_ra8_err_invalid_arg;
368 }
369 volatile r_ipc_channel_regs_t* reg = internal_ra8_ipc_get_regs(channel);
370 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
371
372 /* Publish barrier: order any shared-SRAM payload the caller staged
373 * before this IRQ notification, so the peer cannot wake on a payload
374 * that is not yet visible (the dual-core ring protocol relies on this). */
375 ra8_ipc_barrier();
376 /* HUM Ch 3.2.11 "IPC0ISET0" p 215 -- write 1 to SETn to assert
377 * IRQn on the receiving core and raise the IPCnIRQm interrupt. */
378 reg->ISET = (uint32_t)1U << (uint32_t)event_id;
379 return k_ra8_ok;

399[[nodiscard]] ra8_err_t ra8_ipc_send_message(uint8_t channel, uint32_t message)
400{
401 if ((uint16_t)channel >= (uint16_t)k_ra8_ipc_channel_count) {
402 return k_ra8_err_invalid_arg;
403 }
404 volatile r_ipc_channel_regs_t* reg = internal_ra8_ipc_get_regs(channel);
405 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
406
407 /* HUM Ch 3.2.10 "IPC0STA0 FULL bit" p 214 -- if the FIFO is full
408 * the write to TXD is silently dropped and FERR is set. */
409 if ((reg->STA & k_ra8_ipc_sta_mask_full) != 0U) {
410 return k_ra8_err_busy;
411 }
412
413 /* Publish barrier: make any shared-SRAM payload the caller staged
414 * visible to the peer before this FIFO word, so a consumer that wakes
415 * on the message cannot read a stale payload. */
416 ra8_ipc_barrier();
417 /* HUM Ch 3.2.12 "IPC0TXD0" p 215 -- only 32-bit writes are valid;
418 * narrower accesses are ignored by the peripheral. */
419 reg->TXD = message;
420 return k_ra8_ok;
421}

423[[nodiscard]] ra8_err_t
424ra8_ipc_send_message_retry(uint8_t channel, uint32_t message, uint16_t max_retries)
425{
426 if ((uint16_t)channel >= (uint16_t)k_ra8_ipc_channel_count) {
427 return k_ra8_err_invalid_arg;
428 }
429 if (max_retries > k_ra8_ipc_retry_max) {
430 max_retries = k_ra8_ipc_retry_max;
431 }
432 volatile r_ipc_channel_regs_t* reg = internal_ra8_ipc_get_regs(channel);
433 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
434
435 /* Publish barrier (once before the attempts): order the caller's
436 * shared-SRAM payload before the first FIFO word reaches the peer. */
437 ra8_ipc_barrier();
438 /* NASA Rule 2: bounded loop with the +1 covering the immediate-try
439 * case where retries==0 still gets a single attempt. */
440 for (uint16_t i = 0U; i <= max_retries; ++i) {
441 /* HUM Ch 3.2.10 "FULL bit" p 214 */
442 if ((reg->STA & k_ra8_ipc_sta_mask_full) == 0U) {
443 /* HUM Ch 3.2.12 "IPC0TXD0" p 215 */
444 reg->TXD = message;
445 return k_ra8_ok;
446 }
447 /* HUM Ch 3.2.14 "FCLR bit" p 217 -- knock down a possible FERR
448 * left over from a previous overflow so the dispatcher does not
449 * spin forever on a stuck error flag. */
450 reg->CLR = k_ra8_ipc_clr_mask_fclr;
451 }
452 return k_ra8_err_hw_timeout;
453}

455[[nodiscard]] ra8_err_t
456ra8_ipc_send_burst(uint8_t channel, const uint32_t* data, uint32_t count, uint32_t* out_written)
457{
458 RA8_CHECK_NULL_PTR(data, s_tag, "data must not be nullptr");
459 RA8_CHECK_NULL_PTR(out_written, s_tag, "out_written must not be nullptr");
460 if ((uint16_t)channel >= (uint16_t)k_ra8_ipc_channel_count) {
461 return k_ra8_err_invalid_arg;
462 }
463 if (count == 0U) {
464 return k_ra8_err_invalid_arg;
465 }
466 volatile r_ipc_channel_regs_t* reg = internal_ra8_ipc_get_regs(channel);
467 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
468
469 /* Publish barrier (once before the burst): order any shared-SRAM
470 * payload the caller staged before the first FIFO word reaches the peer. */
471 ra8_ipc_barrier();
472 uint32_t written = 0U;
473 /* NASA Rule 2: bound is the user-supplied ``count`` and the inner
474 * STA.FULL check breaks early when the 4-stage FIFO fills (HUM
475 * Ch 3.1 p 204). */
476 for (uint32_t i = 0U; i < count; ++i) {
477 /* HUM Ch 3.2.10 "FULL bit" p 214 */
478 if ((reg->STA & k_ra8_ipc_sta_mask_full) != 0U) {
479 break;
480 }
481 /* HUM Ch 3.2.12 "IPC0TXD0" p 215 */
482 reg->TXD = data[i];
483 ++written;
484 }
485 *out_written = written;
486 return k_ra8_ok;
487}

602[[nodiscard]] ra8_err_t ra8_ipc_clear_errors(uint8_t channel)
603{
604 volatile r_ipc_channel_regs_t* reg = internal_ra8_ipc_get_regs(channel);
605 if (reg == nullptr) {
606 return k_ra8_err_invalid_arg;
607 }
608 /* HUM Ch 3.2.14 "RCLR bit / FCLR bit" p 217 -- ack both error

739void ra8_ipc_dispatch(uint8_t channel)
740{
741 if ((uint16_t)channel >= (uint16_t)k_ra8_ipc_channel_count) {
742 return;
743 }
744 volatile r_ipc_channel_regs_t* reg = ra8_ipc_channel(channel);
745 if (reg == nullptr) {
746 return; /* GCOVR_EXCL_LINE -- channel already bounded */
747 }
748
749 /* HUM Ch 3.2.10 "IPC0STA0" p 214 -- snapshot the status register
750 * once so the IRQ/RDY/RERR/FERR bits do not race against the peer. */
751 const uint32_t sta = reg->STA;
752 const uint32_t configured = s_ipc_channels[channel].event_mask;
753 const uint32_t fired = sta & configured & k_ra8_ipc_internal_event_full_mask;
754
755 uint32_t message = 0U;
756 if ((fired & k_ra8_ipc_event_msg_ready) != 0U) {
757 /* HUM Ch 3.2.13 "IPC0RXD0" p 216 */
758 message = reg->RXD;
759 }
760
761 /* Decoded per-IRQ-line callbacks. */
762 internal_ra8_ipc_dispatch_irq_lines(channel, fired);
763
764 /* Single shared callback. */
765 const ra8_ipc_event_fn_t fn = s_ipc_callback;
766 void* const ctx = s_ipc_context;
767 if ((fn != nullptr) && (fired != 0U)) {
768 fn(ctx, channel, fired, message);
769 }
770
771 if (fired != 0U) {
772 const uint32_t clr = internal_ra8_ipc_event_to_clr(fired);
773 if (clr != 0U) {
774 /* HUM Ch 3.2.14 "IPC0CLR0" p 216 */
775 reg->CLR = clr;
776 }
777 }
778}

798{

800 if ((uint16_t)unit >= (uint16_t)k_ra8_ipc_nmi_unit_count) {

810[[nodiscard]] ra8_err_t ra8_ipc_install_isr(uint8_t unit, uint8_t priority)

813 return k_ra8_err_invalid_arg;

816 return k_ra8_err_exists;
817 }

819 /* The IPC IRQ event values (0x05B / 0x05C, HUM Ch 18) are not yet
820 * enumerated in ``ra8_elc_event_t``; cast through ``uint16_t`` is
821 * value-preserving since the enum's underlying type is ``uint16_t``.

829 * the NVIC line. */

834 }

836 return k_ra8_ok;
837}
