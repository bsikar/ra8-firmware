99[[nodiscard]] static ra8_err_t iic_peripheral_service(uint8_t* last_byte)
100{
101 uint8_t mask = 0U;
102 if (ra8_i3c_peripheral_status((uint8_t)k_iic_peripheral_channel, &mask) != k_ra8_ok) {
103 return k_ra8_err_hw_error;
104 }
105 bool did_anything = false;
106 if ((mask & (uint8_t)k_ra8_i3c_peripheral_status_rx_full) != 0U) {
107 uint8_t b = 0U;
108 if (ra8_i3c_peripheral_receive((uint8_t)k_iic_peripheral_channel, &b, 1U) != k_ra8_ok) {
109 return k_ra8_err_hw_error;
110 }
111 *last_byte = b;
112 did_anything = true;
113 }
114 if ((mask & (uint8_t)k_ra8_i3c_peripheral_status_tx_empty) != 0U) {
115 if (ra8_i3c_peripheral_send((uint8_t)k_iic_peripheral_channel, last_byte, 1U) != k_ra8_ok) {
116 return k_ra8_err_hw_error;
117 }
118 did_anything = true;
119 }
120 if (did_anything) {
121 (void)ra8_board_led_toggle(k_ra8_board_led1);
122 }
123 return k_ra8_ok;
124}
