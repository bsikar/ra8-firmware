196ra8_err_t ra8_lvd_enable_elc_event(ra8_lvd_channel_t channel)
197{
198 uint8_t idx = 0U;
199 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
200 RA8_RETURN_ON_ERROR(map_err, s_tag, "lvd_enable_elc_event: bad channel"); /* GCOVR_EXCL_BR_LINE */
201
202 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
203 if (!map.has_irq) {
204 return k_ra8_err_not_supported;
205 }
206
207 /* HUM Ch 8.7 "Event Link Controller (ELC) Output" p 315 -- the event
208 * line tracks the comparator output enable: clear DET first, then
209 * raise CMPE. */
210 /* HUM Ch 8.2.7 "PVDmSR : Voltage Monitor m Circuit Status Register" p 307 */
211 const uint8_t sr = *ra8_lvd_reg8(map.sr);
212 *ra8_lvd_reg8(map.sr) = (uint8_t)(sr & (uint8_t)~k_ra8_lvd_sr_mask_det);
213 /* HUM Ch 8.2.4 "PVDmCR0" p 305 */
214 ra8_lvd_internal_cr0_rmw(&map, 0U, k_ra8_lvd_cr0_mask_cmpe);
215 return k_ra8_ok;
216}

278ra8_err_t ra8_lvd_configure_for_standby(ra8_lvd_channel_t channel)
279{
280 uint8_t idx = 0U;
281 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
282 RA8_RETURN_ON_ERROR(map_err,
283 s_tag,
284 "lvd_configure_for_standby: bad channel"); /* GCOVR_EXCL_BR_LINE */
285
286 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
287
288 /* HUM Ch 8.5(1) "Setting in Software Standby mode" p 311 +
289 * HUM Ch 8.5(2) "Settings in Deep Software Standby mode" p 312 --
290 * disable the digital filter and (for m channels) clear RI + RN. */
291 uint8_t set_bits = k_ra8_lvd_cr0_mask_dfdis;
292 uint8_t clr_bits = 0U;
293 if (map.has_irq) {
294 clr_bits |= k_ra8_lvd_cr0_mask_ri;
295 clr_bits |= k_ra8_lvd_cr0_mask_rn;
296 }
297 /* HUM Ch 8.2.4 "PVDmCR0" p 305 */
298 ra8_lvd_internal_cr0_rmw(&map, clr_bits, set_bits);
299 return k_ra8_ok;
300}

357uint32_t ra8_lvd_filter_delay_us(ra8_lvd_loco_div_t div, uint32_t loco_hz)
358{
359 /* HUM Table 8.4 step 8 (p 312) / Table 8.6 step 8 (p 315): wait for
360 * "2s + 3 cycles of the LOCO" where s = 2^(div+1). */
361 uint8_t safe_div = (uint8_t)div;
362 if (safe_div > k_ra8_lvd_loco_div_max) {
363 safe_div = k_ra8_lvd_loco_div_max;
364 }
365 const uint32_t s_factor = (uint32_t)1U << (safe_div + 1U);
366 const uint32_t loco_cycles = (k_ra8_lvd_filter_factor * s_factor) + k_ra8_lvd_filter_extra;
367 const uint32_t hz = (loco_hz != 0U) ? loco_hz : k_ra8_lvd_loco_hz_default;
368 /* +1 us round-up matches FSP's r_lvd_filter_delay computation. */
369 return ((loco_cycles * k_ra8_lvd_us_per_sec) / hz) + 1U;
370}

467void ra8_lvd_dispatch(ra8_lvd_channel_t channel)
468{
469 uint8_t idx = 0U;
470 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
471 if (map_err != k_ra8_ok) {
472 return;
473 }
474 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
475
476 /* Only m channels have a status register / IRQ path. */
477 if (!map.has_irq) {
478 return;
479 }
480
481 /* HUM Ch 8.2.7 "PVDmSR : Voltage Monitor m Circuit Status Register" p 307
482 * -- only fire the callback if a crossing was actually latched. */
483 const uint8_t sr = *ra8_lvd_reg8(map.sr);
484 if ((sr & k_ra8_lvd_sr_mask_det) == 0U) {
485 return;
486 }
487
488 /* Per-channel callback wins; otherwise fall back to the shared one. */
489 const ra8_lvd_event_fn_t chan_fn = s_lvd_chan_fn[idx];
490 void* const chan_ctx = s_lvd_chan_ctx[idx];
491 const ra8_lvd_event_fn_t fn = (chan_fn != nullptr) ? chan_fn : s_lvd_fn;
492 void* const ctx = (chan_fn != nullptr) ? chan_ctx : s_lvd_ctx;
493 if (fn != nullptr) {
494 fn(ctx, channel);
495 }
496
497 /* Write 0 to DET so the next crossing can latch (HUM 8.2.7 Note 1 p 307). */
498 const uint8_t cleared = (uint8_t)(*ra8_lvd_reg8(map.sr) & (uint8_t)~k_ra8_lvd_sr_mask_det);
499 *ra8_lvd_reg8(map.sr) = cleared;
500}
