168static bool rtt_cb_valid(uc_engine* uc, uint64_t cb_addr)
169{
170 const uint32_t max_up = rtt_rd32(uc, cb_addr + (uint64_t)k_rtt_off_max_up);
171 if ((max_up == 0U) || (max_up > (uint32_t)k_rtt_max_up_sane)) {
172 return false;
173 }
174 const uint64_t up0 = cb_addr + (uint64_t)k_rtt_off_up0;
175 const uint32_t buf = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_buf);
176 const uint32_t size = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_size);
177 const uint32_t wr = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_wr);
178 const uint32_t rd = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_rd);
179 if ((buf == 0U) || (size == 0U) || (size > (uint32_t)k_rtt_size_sane)) {
180 return false;
181 }
182 return (wr < size) && (rd < size);
183}

205static uint64_t rtt_match_stage(uc_engine* uc, uint64_t stage_addr, uint64_t len)
206{
207 const uint8_t* p = s_rtt_stage;
208 uint64_t left = len;
209 while (left >= (uint64_t)k_rtt_id_len) {
210 const uint8_t* hit =
211 memchr(p, (int)k_rtt_id[0], (size_t)(left - ((uint64_t)k_rtt_id_len - 1U)));
212 if (hit == nullptr) {
213 return 0U;
214 }
215 if (memcmp(hit, k_rtt_id, (size_t)k_rtt_id_len) == 0) {
216 const uint64_t cb = stage_addr + (uint64_t)(hit - s_rtt_stage);
217 if (rtt_cb_valid(uc, cb)) {
218 return cb;
219 }
220 }
221 left -= (uint64_t)(hit - p) + 1U;
222 p = hit + 1U;
223 }
224 return 0U;
225}

246static void rtt_scan(uc_engine* uc)
247{
248 for (uint64_t off = 0U; off < (uint64_t)k_rtt_scan_span; off += (uint64_t)k_rtt_scan_step) {
249 /* Stage one step plus the ID-length overlap into the NEXT step (clamped at
250 * the window end) so a block straddling the boundary is still found. */
251 uint64_t want = (uint64_t)k_rtt_scan_step + (uint64_t)k_rtt_id_len;
252 if ((off + want) > (uint64_t)k_rtt_scan_span) {
253 want = (uint64_t)k_rtt_scan_span - off;
254 }
255 if (uc_mem_read(uc, (uint64_t)k_rtt_scan_base + off, s_rtt_stage, (size_t)want) != UC_ERR_OK) {
256 return; /* SRAM window unreadable: give up this scan pass */
257 }
258 if (want < (uint64_t)k_rtt_id_len) {
259 return; /* tail shorter than the ID: nothing left to match */
260 }
261 const uint64_t cb = rtt_match_stage(uc, (uint64_t)k_rtt_scan_base + off, want);
262 if (cb != 0U) {
263 s_rtt.cb_addr = cb;
264 return;
265 }
266 }
267}

286static void rtt_line_feed(uint8_t byte)
287{
288 const char c = (char)byte;
289 if (c == '\r') {
290 return;
291 }
292 if ((c == '\n') || (s_rtt.line_len >= (uint32_t)k_rtt_line_max)) {
293 s_rtt.line[s_rtt.line_len] = '\0';
294 (void)fprintf(stdout, "[rtt] %s\n", s_rtt.line);
295 (void)fflush(stdout);
296 board_console_push(k_board_console_ch_rtt, s_rtt.line);
297 s_rtt.lines++;
298 s_rtt.line_len = 0U;
299 if (c == '\n') {
300 return;
301 }
302 }
303 s_rtt.line[s_rtt.line_len] = c;
304 s_rtt.line_len++;
305}

328static void rtt_drain(uc_engine* uc)
329{
330 uint8_t id_now[k_rtt_id_len];
331 if ((uc_mem_read(uc, s_rtt.cb_addr, id_now, sizeof(id_now)) != UC_ERR_OK) ||
332 (memcmp(id_now, k_rtt_id, sizeof(id_now)) != 0)) {
333 s_rtt.cb_addr = 0U; /* block gone (reboot / clobber): re-discover by scan */
334 return;
335 }
336 const uint64_t up0 = s_rtt.cb_addr + (uint64_t)k_rtt_off_up0;
337 const uint32_t buf = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_buf);
338 const uint32_t size = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_size);
339 const uint32_t wr = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_wr);
340 uint32_t rd = rtt_rd32(uc, up0 + (uint64_t)k_rtt_up_off_rd);
341 if ((buf == 0U) || (size == 0U) || (size > (uint32_t)k_rtt_size_sane) || (wr >= size) ||
342 (rd >= size) || (wr == rd)) {
343 return; /* invalid mid-update descriptor, or simply nothing pending */
344 }
345 const uint32_t avail = (wr > rd) ? (wr - rd) : ((size - rd) + wr);
346 const uint32_t n = (avail > (uint32_t)k_rtt_drain_max) ? (uint32_t)k_rtt_drain_max : avail;
347 uint32_t got = 0U;
348 /* At most two contiguous segments: rd..min(rd+n, size), then a wrap to 0. */
349 const uint32_t first = ((size - rd) < n) ? (size - rd) : n;
350 if (uc_mem_read(uc, (uint64_t)buf + (uint64_t)rd, s_rtt_seg, (size_t)first) == UC_ERR_OK) {
351 got = first;
352 if ((n > first) &&
353 (uc_mem_read(uc, (uint64_t)buf, &s_rtt_seg[first], (size_t)(n - first)) == UC_ERR_OK)) {
354 got = n;
355 }
356 }
357 if (got == 0U) {
358 return; /* ring storage unreadable: leave the offsets untouched */
359 }
360 for (uint32_t i = 0U; i < got; i++) {
361 rtt_line_feed(s_rtt_seg[i]);
362 }
363 rd = (rd + got) % size;
364 /* Store the advanced read offset back so the firmware sees the drain. */
365 (void)uc_mem_write(uc, up0 + (uint64_t)k_rtt_up_off_rd, &rd, sizeof(rd));
366 s_rtt.drained += got;
367}

388static void rtt_tick(uc_engine* uc)
389{
390 s_rtt.ticks++;
391 if (s_rtt.cb_addr == 0U) {
392 if (s_rtt.ticks < s_rtt.next_scan) {
393 return;
394 }
395 rtt_scan(uc);
396 if (s_rtt.cb_addr == 0U) {
397 /* Exponential backoff, capped: a non-RTT app pays a bounded scan cost. */
398 if (s_rtt.scan_gap < (uint32_t)k_rtt_scan_max_gap) {
399 s_rtt.scan_gap *= 2U;
400 }
401 s_rtt.next_scan = s_rtt.ticks + s_rtt.scan_gap;
402 return;
403 }
404 }
405 rtt_drain(uc);
406}
