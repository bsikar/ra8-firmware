4"""Gate: shellcheck + shfmt for first-party shell scripts.
5
6ShellCheck (correctness, at ``--severity=style`` plus the opt-in checks listed
7in :data:`SHELLCHECK_ENABLE`) and shfmt (formatting, 2-space case-indented) are
8the shell equivalents of ruff. This wrapper fails on any finding -- no
9grandfathering. Both tools must be on PATH (or named via ``SHELLCHECK`` /
10``SHFMT``); without them the gate skips locally unless ``--require`` is passed,
11which CI uses to fail on a missing tool.
12
13``style`` is the tightest severity ShellCheck offers, so nothing is filtered by
14level. The opt-in checks are the ones that are both *fixable in place* and map
15to a defect class this tree has actually shipped -- unquoted expansions, values
16that are read but never assigned, ``which`` instead of ``command -v``. Two
17opt-in checks are deliberately NOT enabled; see ``docs/STYLE_GUIDE.md`` and the
18comment on :data:`SHELLCHECK_DISABLED_OPTIONAL` for the measurements behind
19that call.
20
21Run::
22
23 check_shell.py # gate (fail on any finding)
24 check_shell.py --require # fail (not skip) if a tool is absent
25 check_shell.py --selftest # prove the gate fires and stays quiet
26
27Exit 0 if clean, exit 1 on findings, exit 2 on a tool error or a scope that
28collapsed below SCRIPT_FLOOR.
29"""
30

67# Deliberately NOT enabled. Re-measured for #363 on the first-party shell files
68# at ShellCheck 0.11.0, on top of the severity + opt-in set above:
69#
70# check-set-e-suppressed (SC2310/SC2311) -- 90 findings / 24 files.
71# Unsatisfiable by construction, and verified form by form: it fires on
72# `fn || rc=$?`, on the rewrite its own help text recommends
73# (`if fn; then rc=0; else rc=$?; fi`), on a bare one-line predicate, on
74# `! fn` and on `fn && ...`. The only two forms it ACCEPTS are worse than
75# the ones it rejects -- `set +e; fn; rc=$?; set -e` passes while a
76# brace-bodied callee still runs past a mid-body failure, and
77# `( fn ); rc=$?` passes while aborting the parent outright. Enabling it
78# would mean ~90 inline disables and would push authors toward the form it
79# cannot see. The signal is real, so it is covered instead by
80# scripts/checks/check_errexit_masking.py, which fires only where a
81# first-party function with two or more failable commands is invoked with
82# its status masked. The runtime regression for the specific gate-suite
83# failure remains asserted by suite_errexit_selftest in scripts/ci.sh.
84# check-extra-masked-returns (SC2312) -- 163 findings / 34 files.
85# Overwhelmingly command substitutions on commands that cannot meaningfully
86# fail (uname, date -Iseconds, basename, id -un) or inside `< <(...)`
87# process substitutions with no single-statement rewrite. Fixing them means
88# hoisting each into a preceding assignment: some of that is worth doing and
89# some is pure motion, and a blanket enable cannot tell the two apart. The
90# subset that actually matters -- masking a FUNCTION's status -- is exactly
91# what check_errexit_masking.py now covers.
92# require-variable-braces (SC2250) -- 2901 findings / 60 files.
93# Presentation. `$var` and `${var}` are identical outside the
94# disambiguation cases, which ShellCheck already flags separately at the
95# level of correctness.
96# require-double-brackets (SC2292) -- 176 findings / 21 files.
97# Presentation, and actively wrong here: `[` is correct in the POSIX-sh
98# scripts in this tree, so this would push them toward bash-only for no
99# behavioural gain.
100# add-default-case (SC2249) -- 56 findings / 15 files.
101# A default case is right for a dispatch on external input, which this tree
102# already writes; it is noise on an exhaustive match over a fixed internal
103# enum, and the check cannot distinguish them.
104SHELLCHECK_DISABLED_OPTIONAL = (
105 "check-set-e-suppressed",
106 "check-extra-masked-returns",
107 "require-variable-braces",
108 "require-double-brackets",
109 "add-default-case",
110)

126def _shellcheck_args() -> list[str]:
127 """The severity + opt-in flags every ShellCheck invocation here shares."""
128 return [
129 # -x follows `# shellcheck source=<path>` directives instead of only
130 # guessing the target from the sourcing script's own directory. Without
131 # it, a helper sourced across directories (`. "$SCRIPT_DIR/../builders/
132 # select_host_compiler.sh"`) is unresolvable, and every variable that
133 # helper exports is then reported as referenced-but-never-assigned --
134 # findings about the analysis, not about the code. Measured: -x adds
135 # zero new findings over this gate's file set and removes those.
136 "-x",
137 f"--severity={SHELLCHECK_SEVERITY}",
138 "--enable=" + ",".join(SHELLCHECK_ENABLE),
139 ]

149def _git_ls(*pathspec: str) -> list[str]:
150 """Tracked plus untracked-but-not-ignored paths matching `pathspec`.
151
152 Enumerates via ``git ls-files`` instead of a filesystem walk so locally
153 present, git-excluded trees (``.git/info/exclude`` entries such as
154 ``recon/`` vendor drops) never enter the gate -- a raw ``rglob`` scanned
155 them and failed commits on third-party findings CI can never see.
156 """
157 git_tool = shutil.which("git") or "git"
158 proc = subprocess.run( # noqa: S603 -- fixed argv, trusted tool path
159 [git_tool, "ls-files", "--cached", "--others", "--exclude-standard", "--", *pathspec],
160 cwd=REPO_ROOT,
161 capture_output=True,
162 text=True,
163 check=False,
164 )
165 if proc.returncode != 0:
166 sys.stderr.write(proc.stderr)
167 sys.stderr.write(f"git ls-files failed (exit {proc.returncode})\n")
168 sys.exit(2)
169 return [ln.strip() for ln in proc.stdout.splitlines() if ln.strip()]

172def _has_shell_shebang(rel: str) -> bool:
173 """True when `rel` opens with a ``#!`` line naming sh, bash or zsh."""
174 try:
175 with (REPO_ROOT / rel).open("rb") as handle:
176 first = handle.readline(200)
177 except OSError:
178 return False
179 if not first.startswith(b"#!"):
180 return False
181 line = first.decode("utf-8", errors="replace")
182 return any(tok in line for tok in ("bash", "zsh", "/sh", "env sh"))

185def first_party_scripts() -> list[str]:
186 """First-party shell scripts: by ``*.sh`` suffix OR by shebang.
187
188 The shebang sweep is not hypothetical. Every git hook in ``scripts/git/``
189 -- pre-commit, pre-push, commit-msg, post-merge, post-commit,
190 post-checkout -- is an extensionless bash script, so a suffix-only scope
191 left the hooks that enforce this entire tree as the only shell in it that
192 nothing shellchecked or shfmt'd. That is the #296/#332/#358/#359/#360
193 defect class exactly: a scope narrower than the thing it claims to cover,
194 reporting clean.
195 """
196 by_suffix = _git_ls("*.sh")
197 known = set(by_suffix)
198 by_shebang = [rel for rel in _git_ls() if rel not in known and _has_shell_shebang(rel)]
199 out = [
200 rel
201 for rel in known.union(by_shebang)
202 if not is_build_output_path(rel)
203 and not any(frag in f"/{rel}" for frag in EXCLUDE_FRAGMENTS)
204 ]
205 return sorted(out)

208def _run_shellcheck(tool: str, files: list[str], cwd: Path | None = None) -> Findings:
209 proc = subprocess.run( # noqa: S603 -- fixed argv, trusted tool path
210 [tool, *_shellcheck_args(), "-f", "json", *files],
211 cwd=cwd or REPO_ROOT,
212 capture_output=True,
213 text=True,
214 check=False,
215 )
216 if proc.returncode not in (0, 1):
217 sys.stderr.write(proc.stderr)
218 sys.stderr.write(f"shellcheck failed (exit {proc.returncode})\n")
219 sys.exit(2)
220 findings: Findings = {}
221 for item in json.loads(proc.stdout or "[]"):
222 rel = item["file"]
223 code = f"SC{item['code']}"
224 findings.setdefault(rel, {})
225 findings[rel][code] = findings[rel].get(code, 0) + 1
226 return findings

341def selftest(tmp: Path) -> int:
342 """Assert the gate fires on every enforced class and stays quiet otherwise."""
343 sc_tool = _find("SHELLCHECK", "shellcheck")
344 fmt_tool = _find("SHFMT", "shfmt")
345 if not sc_tool or not fmt_tool:
346 missing = " and ".join(
347 n for n, t in (("shellcheck", sc_tool), ("shfmt", fmt_tool)) if not t
348 )
349 sys.stderr.write(f"check_shell.py --selftest: {missing} not found\n")
350 return 2
351
352 failures: list[str] = []
353
354 for label, fname, body, must_fire in SELFTEST_CASES:
355 path = tmp / fname
356 path.write_text(body)
357 fired = bool(_run_shellcheck(sc_tool, [fname], cwd=tmp))
358 if fired != must_fire:
359 verb = "did not fire" if must_fire else "fired"
360 codes = _run_shellcheck(sc_tool, [fname], cwd=tmp).get(fname, {})
361 failures.append(f" shellcheck {verb} (unexpected): {label} {codes or ''}")
362
363 fmt_name, fmt_body = SELFTEST_SHFMT_BAD
364 (tmp / fmt_name).write_text(fmt_body)
365 if not _run_shfmt(fmt_tool, [fmt_name], cwd=tmp):
366 failures.append(f" shfmt did not flag misindented file: {fmt_name}")
367 quiet_name = "quiet_shfmt.sh"
368 (tmp / quiet_name).write_text('#!/usr/bin/env bash\nif true; then\n echo "ok"\nfi\n')
369 if _run_shfmt(fmt_tool, [quiet_name], cwd=tmp):
370 failures.append(f" shfmt flagged a correctly formatted file: {quiet_name}")
371
372 if failures:
373 sys.stderr.write("check_shell.py --selftest: FAILED\n\n")
374 sys.stderr.write("\n".join(failures) + "\n")
375 return 1
376
377 fires = sum(1 for c in SELFTEST_CASES if c[3]) + 1
378 quiets = sum(1 for c in SELFTEST_CASES if not c[3]) + 1
379 print(
380 f"check_shell.py --selftest: PASS "
381 f"({fires + quiets} cases: {fires} must fire, {quiets} must stay quiet)"
382 )
383 return 0

386def _report(checks: Findings, fmt: list[str]) -> None:
387 if checks:
388 sys.stderr.write("check_shell.py: shellcheck finding(s):\n")
389 for relfile in sorted(checks):
390 for code, count in sorted(checks[relfile].items()):
391 sys.stderr.write(f" {relfile}: {code} x{count}\n")
392 if fmt:
393 joined = " ".join(SHFMT_ARGS)
394 sys.stderr.write(f"check_shell.py: file(s) need `shfmt -w {joined}`:\n")
395 for relfile in fmt:
396 sys.stderr.write(f" {relfile}\n")
397 sys.stderr.write(
398 "\nFix the finding or reformat. A `# shellcheck disable=SCxxxx` needs an\n"
399 "inline reason on the same line saying why the finding does not apply.\n"
400 )

403def main(argv: list[str]) -> int:
404 """Run shellcheck and shfmt over every tracked shell script.
405
406 Both tools are REQUIRED, not optional: a missing one fails the gate rather
407 than reducing its scope, because a checker that quietly stops checking is
408 indistinguishable from a clean tree.
409
410 ``--list-files`` reports the covered scope for check_lint_coverage.py and
411 is deliberately independent of whether either tool is installed -- the
412 question is what this gate covers, and a missing binary must not shrink
413 the answer to nothing.
414
415 SCRIPT_FLOOR replaces the old ``no shell scripts to scan`` branch, which
416 exited 0 on an empty enumeration -- a result indistinguishable from a clean
417 tree and produced by having read nothing.
418
419 Returns 0 when clean, 1 on findings or a formatting difference, 2 when the
420 scope collapsed below SCRIPT_FLOOR, and 1 on a missing tool under
421 ``--require``.
422 """
423 if "--selftest" in argv[1:]:
424 with tempfile.TemporaryDirectory() as td:
425 return selftest(Path(td))
426
427 # Scope introspection for check_lint_coverage.py -- see the note in
428 # check_ruff.py's main(). Deliberately independent of whether shellcheck
429 # and shfmt are installed: the question is what this gate COVERS, and a
430 # missing tool must not silently shrink the answer to nothing.
431 if "--list-files" in argv[1:]:
432 print("\n".join(first_party_scripts()))
433 return 0
434
435 sc_tool = _find("SHELLCHECK", "shellcheck")
436 fmt_tool = _find("SHFMT", "shfmt")
437 if not sc_tool or not fmt_tool:
438 missing = " and ".join(
439 n for n, t in (("shellcheck", sc_tool), ("shfmt", fmt_tool)) if not t
440 )
441 msg = f"check_shell.py: {missing} not found"
442 if "--require" in argv[1:]:
443 sys.stderr.write(msg + " (--require set)\n")
444 sys.exit(1)
445 print(msg + " -- skipping (install to enforce locally).")
446 sys.exit(0)
447
448 files = first_party_scripts()
449 if len(files) < SCRIPT_FLOOR:
450 sys.stderr.write(
451 f"check_shell.py: FATAL -- only {len(files)} shell script(s) in scope, "
452 f"floor is {SCRIPT_FLOOR}.\n"
453 " A collapsed scope reports a clean tree because it checked nothing.\n"
454 )
455 return 2
456 checks = _run_shellcheck(sc_tool, files)
457 fmt = _run_shfmt(fmt_tool, files)
458 if not checks and not fmt:
459 print(
460 f"check_shell.py: clean ({len(files)} file(s), "
461 f"severity={SHELLCHECK_SEVERITY} + {len(SHELLCHECK_ENABLE)} opt-in check(s))."
462 )
463 return 0
464 _report(checks, fmt)
465 return 1
