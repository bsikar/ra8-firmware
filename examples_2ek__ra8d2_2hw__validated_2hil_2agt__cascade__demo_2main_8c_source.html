178static void agt_cas_format_tick(uint32_t tick, uint8_t* digits)
179{
180 for (uint32_t i = 0U; i < (uint32_t)k_agt_cas_hex_digits; ++i) {
181 const uint32_t shift =
182 ((uint32_t)k_agt_cas_hex_digits - 1U - i) * (uint32_t)k_agt_cas_hex_shift;
183 const uint32_t nib = (tick >> shift) & (uint32_t)k_agt_cas_hex_mask;
184 digits[i] = (uint8_t)k_agt_cas_hex_chars[nib];
185 }
186}

208[[nodiscard]] static ra8_err_t agt_cas_emit_tick(uint32_t tick)
209{
210 uint8_t digits[k_agt_cas_hex_digits] = {};
211 agt_cas_format_tick(tick, digits);
212 ra8_err_t err =
213 ra8_board_uart_console_write(k_agt_cas_msg_head, (size_t)(sizeof(k_agt_cas_msg_head) - 1U));
214 if (err != k_ra8_ok) {
215 return err;
216 }
217 err = ra8_board_uart_console_write(digits, (size_t)sizeof(digits));
218 if (err != k_ra8_ok) {
219 return err;
220 }
221 return ra8_board_uart_console_write(k_agt_cas_msg_tail,
222 (size_t)(sizeof(k_agt_cas_msg_tail) - 1U));
223}

227int32_t main(void)
228{
229 agt_cas_setup_or_halt();
230 ra8_isr_globals_enable();
231
232 if (agt_cas_arm() != k_ra8_ok) {
233 agt_cas_panic_halt();
234 }
235
236 uint32_t tick = 0U;
237 while (1) {
238 uint8_t status = 0U;
239 if (ra8_agt_get_status((uint8_t)k_agt_cas_hi_channel, &status) != k_ra8_ok) {
240 break;
241 }
242 if ((status & (uint8_t)k_ra8_agt_agtcr_tundf_msk) != 0U) {
243 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
244 break;
245 }
246 ++tick;
247 if (agt_cas_emit_tick(tick) != k_ra8_ok) {
248 break;
249 }
250 /* Stop both halves + re-arm to clear AGT1.AGTCR.TUNDF. */
251 if (ra8_agt_stop(0U) != k_ra8_ok) {
252 break;
253 }
254 if (ra8_agt_stop((uint8_t)k_agt_cas_hi_channel) != k_ra8_ok) {
255 break;
256 }
257 if (agt_cas_arm() != k_ra8_ok) {
258 break;
259 }
260 }
261 ra8_delay_ms((uint32_t)k_agt_cas_poll_ms);
262 }
263 agt_cas_panic_halt();
264 return 0;
265}
