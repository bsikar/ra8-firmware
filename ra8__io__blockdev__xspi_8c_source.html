120xspi_read_chunked(uint8_t instance, uint32_t flash_addr, uint8_t* buf, uint32_t len)
121{
122 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
123 uint32_t done = k_xspi_zero_blocks;
124 while (done < len) {
125 const uint32_t remaining = len - done;
126 const uint32_t chunk =
127 (remaining > (uint32_t)k_ra8_xspi_max_xfer) ? (uint32_t)k_ra8_xspi_max_xfer : remaining;
128 const ra8_err_t e = ra8_xspi_flash_read(instance, flash_addr + done, &buf[done], chunk);
129 if (e != k_ra8_ok) {
130 return e;
131 }
132 done += chunk;
133 }
134 return k_ra8_ok;
135}

214RA8_INTERNAL
215static ra8_err_t write_one_sector(const ra8_io_blockdev_xspi_state_t* st,
216 uint32_t sector_lba,
217 uint32_t first_blk,
218 uint32_t n_blocks,
219 const uint8_t* src)
220{
221 RA8_CHECK_NULL_PTR(st, s_tag, "st must not be nullptr");
222 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
223 const uint32_t sec_addr = st->base_off + (sector_lba * (uint32_t)k_ra8_io_block_size_bytes);
224 const ra8_err_t r =
225 xspi_read_chunked(st->instance, sec_addr, s_sector, (uint32_t)k_xspi_sector_bytes);
226 if (r != k_ra8_ok) {
227 return r;
228 }
229 const size_t dst_off = (size_t)first_blk * (size_t)k_ra8_io_block_size_bytes;
230 const size_t n_bytes = (size_t)n_blocks * (size_t)k_ra8_io_block_size_bytes;
231 (void)memcpy(&s_sector[dst_off], src, n_bytes);
232 const ra8_err_t e = ra8_xspi_flash_erase_sector(st->instance, sec_addr);
233 if (e != k_ra8_ok) {
234 return e;
235 }
236 return xspi_program_chunked(st->instance, sec_addr, s_sector, (uint32_t)k_xspi_sector_bytes);
237}

311RA8_INTERNAL
312static ra8_err_t xspi_write(void* ctx, uint32_t lba, uint32_t count, const uint8_t* buf)
313{
314 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
315 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
316 const ra8_io_blockdev_xspi_state_t* st = (const ra8_io_blockdev_xspi_state_t*)ctx;
317 if (st->read_only) {
318 return k_ra8_err_not_supported;
319 }
320 const ra8_err_t b = xspi_bounds(st, lba, count);
321 if (b != k_ra8_ok) {
322 return b;
323 }
324 uint32_t done = k_xspi_zero_blocks;
325 while (done < count) {
326 const uint32_t blk = lba + done;
327 const uint32_t first_blk = blk % (uint32_t)k_xspi_blocks_per_sector;
328 const uint32_t sector_lba = blk - first_blk;
329 const uint32_t room = (uint32_t)k_xspi_blocks_per_sector - first_blk;
330 const uint32_t left = count - done;
331 const uint32_t n_blocks = (left < room) ? left : room;
332 const uint8_t* src = &buf[(size_t)done * (size_t)k_ra8_io_block_size_bytes];
333 const ra8_err_t e = write_one_sector(st, sector_lba, first_blk, n_blocks, src);
334 if (e != k_ra8_ok) {
335 return e;
336 }
337 done += n_blocks;
338 }
339 return k_ra8_ok;
340}

371RA8_INTERNAL
372static ra8_err_t xspi_erase(void* ctx, uint32_t lba, uint32_t count)
373{
374 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
375 const ra8_io_blockdev_xspi_state_t* st = (const ra8_io_blockdev_xspi_state_t*)ctx;
376 if (st->read_only) {
377 return k_ra8_err_not_supported;
378 }
379 if ((lba % (uint32_t)k_xspi_blocks_per_sector) != k_xspi_zero_blocks) {
380 return k_ra8_err_invalid_arg;
381 }
382 if ((count % (uint32_t)k_xspi_blocks_per_sector) != k_xspi_zero_blocks) {
383 return k_ra8_err_invalid_arg;
384 }
385 const ra8_err_t b = xspi_bounds(st, lba, count);
386 if (b != k_ra8_ok) {
387 return b;
388 }
389 uint32_t blk = lba;
390 while (blk < lba + count) {
391 const uint32_t addr = st->base_off + (blk * (uint32_t)k_ra8_io_block_size_bytes);
392 const ra8_err_t e = ra8_xspi_flash_erase_sector(st->instance, addr);
393 if (e != k_ra8_ok) {
394 return e;
395 }
396 blk += (uint32_t)k_xspi_blocks_per_sector;
397 }
398 return k_ra8_ok;
399}
