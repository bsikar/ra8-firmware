204static void dtc_decode_ti(uc_engine* uc, uint32_t ti_addr, dtc_ti_t* out)
205{
206 const uint32_t mr = dtc_mem_read(uc, (uint64_t)ti_addr + (uint64_t)k_ti_off_mr, 4U);
207 out->sar = dtc_mem_read(uc, (uint64_t)ti_addr + (uint64_t)k_ti_off_sar, 4U);
208 out->dar = dtc_mem_read(uc, (uint64_t)ti_addr + (uint64_t)k_ti_off_dar, 4U);
209 out->crb = dtc_mem_read(uc, (uint64_t)ti_addr + (uint64_t)k_ti_off_crb, 2U);
210 out->cra = dtc_mem_read(uc, (uint64_t)ti_addr + (uint64_t)k_ti_off_cra, 2U);
211
212 const uint32_t mra = (mr >> (uint32_t)k_mr_mra_shift) & (uint32_t)k_mr_byte_mask;
213 const uint32_t mrb = (mr >> (uint32_t)k_mr_mrb_shift) & (uint32_t)k_mr_byte_mask;
214 out->md = (mra >> (uint32_t)k_mra_md_pos) & (uint32_t)k_mr_2bit_mask;
215 out->sz = (mra >> (uint32_t)k_mra_sz_pos) & (uint32_t)k_mr_2bit_mask;
216 out->sm = (mra >> (uint32_t)k_mra_sm_pos) & (uint32_t)k_mr_2bit_mask;
217 out->dm = (mrb >> (uint32_t)k_mrb_dm_pos) & (uint32_t)k_mr_2bit_mask;
218}

238static uint32_t dtc_unit_count(const dtc_ti_t* ti)
239{
240 uint32_t per_block = ti->cra;
241 uint32_t blocks = 1U;
242 if (ti->md == (uint32_t)k_mra_md_block) {
243 if (ti->cra == 0U) {
244 per_block = (uint32_t)k_dtc_block_cra_zero;
245 }
246 blocks = (ti->crb == 0U) ? 1U : ti->crb;
247 }
248 const uint32_t units = per_block * blocks;
249 return (units > (uint32_t)k_dtc_max_units) ? 0U : units;
250}

270static void dtc_copy_units(uc_engine* uc, const dtc_ti_t* ti, uint32_t units, uint32_t unit)
271{
272 uint64_t s_addr = (uint64_t)ti->sar;
273 uint64_t d_addr = (uint64_t)ti->dar;
274 const uint64_t s_step = (ti->sm == (uint32_t)k_mr_addr_inc) ? (uint64_t)unit : 0U;
275 const uint64_t d_step = (ti->dm == (uint32_t)k_mr_addr_inc) ? (uint64_t)unit : 0U;
276 for (uint32_t i = 0U; i < units; ++i) {
277 uint8_t tmp[4] = {};
278 (void)uc_mem_read(uc, s_addr, tmp, (size_t)unit);
279 (void)uc_mem_write(uc, d_addr, tmp, (size_t)unit);
280 s_addr += s_step;
281 d_addr += d_step;
282 }
283}

286static void dtc_run_transfer(uc_engine* uc, uint32_t ti_addr)
287{
288 dtc_ti_t ti = {};
289 dtc_decode_ti(uc, ti_addr, &ti);
290
291 const uint32_t units = dtc_unit_count(&ti);
292 if (units == 0U) {
293 return;
294 }
295 const uint32_t unit = dtc_unit_bytes(ti.sz);
296 dtc_copy_units(uc, &ti, units, unit);
297
298 s_dtc.activations++;
299 s_dtc.bytes = units * unit;
300 /* Console DMA tab: one line per DTC activation (the DTC shares the DMA tab
301 * with the DMAC). Reports the unit count and unit size. */
302 char ln[k_dtc_console_line_cap];
303 (void)snprintf(ln, sizeof(ln), "DTC %u units x %uB", (unsigned)units, (unsigned)unit);
304 board_console_push(k_board_console_ch_dma, ln);
305}

308static void dtc_activate_swevt0(uc_engine* uc)
309{
310 if (s_dtc.dtcvbr == 0U) {
311 return; /* Vector base not programmed: nothing to activate. */
312 }
313 const uint32_t slot = board_periph_icu_dtc_slot((uint16_t)k_elc_swevt0_event);
314 if (slot >= (uint32_t)k_dtc_ielsr_slots) {
315 return; /* No IELSR slot links the event with DTCE set. */
316 }
317 const uint64_t vt_entry =
318 (uint64_t)s_dtc.dtcvbr + ((uint64_t)slot * (uint64_t)k_dtc_vt_entry_bytes);
319 const uint32_t ti_addr = dtc_mem_read(uc, vt_entry, 4U) & (uint32_t)k_dtc_ti_addr_mask;
320 if (ti_addr == 0U) {
321 return; /* No TI registered for the slot. */
322 }
323 dtc_run_transfer(uc, ti_addr);
324 /* Real silicon clears the slot's DTCE and asserts its CPU interrupt on the
325 * last transfer (HUM Figure 18.5 p 801). The demo gates on the destination
326 * settling (a direct poll), not the ISR, so the model performs the copy and
327 * leaves interrupt delivery to the firmware's own polling path -- pending the
328 * NVIC line here is unnecessary for the transfer contract and depends on the
329 * app's vector wiring. The transfer (the modelled behaviour) is what flips the
330 * demo to match=Y. */
331}

381static uint64_t elc_read(uc_engine* uc, uint64_t addr, unsigned size)
382{
383 (void)uc;
384 const uint64_t off = addr - (uint64_t)k_elc_base;
385 uint64_t v = 0U;
386 for (unsigned i = 0U; (i < size) && ((off + (uint64_t)i) < (uint64_t)k_elc_span); ++i) {
387 v |= (uint64_t)s_elc[off + (uint64_t)i] << (8U * i);
388 }
389 return v;
390}

393static void elc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
394{
395 const uint64_t off = addr - (uint64_t)k_elc_base;
396 for (unsigned i = 0U; (i < size) && ((off + (uint64_t)i) < (uint64_t)k_elc_span); ++i) {
397 s_elc[off + (uint64_t)i] = (uint8_t)(value >> (8U * i));
398 }
399 /* The third write of the unlock/arm/trigger sequence (SEG=1) on ELSEGR0 fires
400 * software event 0, which activates the DTC if a DTCE-enabled slot links it. */
401 if ((off == (uint64_t)k_elc_off_elsegr0) &&
402 (((uint8_t)value & (uint8_t)k_elc_elsegr_trigger) == (uint8_t)k_elc_elsegr_trigger)) {
403 dtc_activate_swevt0(uc);
404 }
405}
