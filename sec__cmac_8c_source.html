209/* =============================================================================

211 * =============================================================================
212 *
213 * The AES round transform and CMAC subkey/GF(2^128) doubling below are direct
214 * transcriptions of FIPS 197 and NIST SP 800-38B. The S-box and the field
215 * constants are standard values, not tunable parameters; the loops all have
216 * compile-time-fixed bounds. The host suite pins the output to the published
217 * SP 800-38B AES-128 and AES-256 known-answer vectors.

309RA8_INTERNAL static void
310internal_aes_key_expand(const uint8_t* key, uint8_t nk, uint8_t nr, uint8_t* rk)
311{
312 const uint16_t total_words = (uint16_t)((uint16_t)k_aes_state_cols * (uint16_t)(nr + 1U));
313 for (uint16_t i = 0U; i < ((uint16_t)nk * (uint16_t)k_aes_word_bytes); ++i) {
314 rk[i] = key[i];
315 }
316 uint8_t rcon = (uint8_t)k_aes_rcon_seed;
317 for (uint16_t i = nk; i < total_words; ++i) {
318 uint8_t t[k_aes_word_bytes];
319 for (uint8_t j = 0U; j < (uint8_t)k_aes_word_bytes; ++j) {
320 t[j] = rk[((uint16_t)(i - 1U) * (uint16_t)k_aes_word_bytes) + j];
321 }
322 if ((i % nk) == 0U) {
323 const uint8_t tmp = t[0];
324 t[0] = (uint8_t)(s_aes_sbox[t[1]] ^ rcon);
325 t[1] = s_aes_sbox[t[2]];
326 t[2] = s_aes_sbox[t[3]];
327 t[3] = s_aes_sbox[tmp];
328 rcon = internal_aes_xtime(rcon);
329 } else if ((nk > 6U) && ((i % nk) == (uint8_t)k_aes_ext_sub_i)) {
330 for (uint8_t j = 0U; j < (uint8_t)k_aes_word_bytes; ++j) {
331 t[j] = s_aes_sbox[t[j]];
332 }
333 } else {
334 /* No transform: straight copy from t[] below. */
335 }
336 for (uint8_t j = 0U; j < (uint8_t)k_aes_word_bytes; ++j) {
337 rk[((uint16_t)i * (uint16_t)k_aes_word_bytes) + j] =
338 (uint8_t)(rk[((uint16_t)(i - nk) * (uint16_t)k_aes_word_bytes) + j] ^ t[j]);
339 }
340 }

361{
362 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
363 s[i] = s_aes_sbox[s[i]];

394RA8_INTERNAL static void internal_aes_mix_columns(uint8_t* s)
395{
396 for (uint8_t c = 0U; c < (uint8_t)k_aes_state_cols; ++c) {
397 uint8_t* col = &s[(uint8_t)(c * (uint8_t)k_aes_state_cols)];
398 const uint8_t a0 = col[0];
399 const uint8_t a1 = col[1];
400 const uint8_t a2 = col[2];
401 const uint8_t a3 = col[3];
402 col[0] = (uint8_t)(internal_aes_xtime(a0) ^ internal_aes_xtime(a1) ^ a1 ^ a2 ^ a3);
403 col[1] = (uint8_t)(a0 ^ internal_aes_xtime(a1) ^ internal_aes_xtime(a2) ^ a2 ^ a3);
404 col[2] = (uint8_t)(a0 ^ a1 ^ internal_aes_xtime(a2) ^ internal_aes_xtime(a3) ^ a3);
405 col[3] = (uint8_t)(internal_aes_xtime(a0) ^ a0 ^ a1 ^ a2 ^ internal_aes_xtime(a3));

431RA8_INTERNAL static void
432internal_aes_encrypt(const uint8_t* rk, uint8_t nr, const uint8_t* in, uint8_t* out)
433{
434 uint8_t s[k_aes_block_bytes];
435 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
436 s[i] = (uint8_t)(in[i] ^ rk[i]);
437 }
438 for (uint8_t round = 1U; round <= nr; ++round) {
439 internal_aes_sub_shift(s);
440 if (round != nr) {
441 internal_aes_mix_columns(s);
442 }
443 const uint8_t* round_key = &rk[(size_t)round * (size_t)k_aes_block_bytes];
444 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
445 s[i] = (uint8_t)(s[i] ^ round_key[i]);
446 }
447 }
448 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
449 out[i] = s[i];
450 }
451}

471RA8_INTERNAL static void internal_cmac_double(const uint8_t* in, uint8_t* out)
472{
473 const uint8_t msb = (uint8_t)(in[0] >> k_aes_msb_shift);
474 for (uint8_t i = 0U; i < (uint8_t)k_aes_last_byte; ++i) {
475 out[i] = (uint8_t)((uint8_t)(in[i] << 1U) | (uint8_t)(in[i + 1U] >> k_aes_msb_shift));
476 }
477 out[k_aes_last_byte] = (uint8_t)(in[k_aes_last_byte] << 1U);
478 if (msb != 0U) {
479 out[k_aes_last_byte] = (uint8_t)(out[k_aes_last_byte] ^ (uint8_t)k_cmac_rb);
480 }
481}

537RA8_INTERNAL static void internal_cmac_build_last(const uint8_t* msg,
538 uint32_t msg_len,
539 uint32_t last_off,
540 bool complete,
541 const uint8_t* k1,
542 const uint8_t* k2,
543 uint8_t* last)
544{
545 if (complete) {
546 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
547 last[i] = (uint8_t)(msg[last_off + i] ^ k1[i]);
548 }
549 return;
550 }
551 const uint32_t rem = msg_len - last_off;
552 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
553 uint8_t byte = 0U;
554 if ((uint32_t)i < rem) {
555 byte = msg[last_off + (uint32_t)i];
556 } else if ((uint32_t)i == rem) {
557 byte = (uint8_t)k_cmac_pad_marker;
558 } else {
559 byte = 0U;
560 }
561 last[i] = (uint8_t)(byte ^ k2[i]);
562 }
563}

588RA8_INTERNAL static void internal_cmac_tag(const uint8_t* key,
589 uint16_t key_len,
590 const uint8_t* msg,
591 uint32_t msg_len,
592 uint8_t* out_mac)
593{
594 const uint8_t nk = (uint8_t)(key_len / (uint16_t)k_aes_word_bytes);
595 const uint8_t nr =
596 (key_len == (uint16_t)k_ra8_sec_cmac_key_128) ? (uint8_t)k_aes_nr_128 : (uint8_t)k_aes_nr_256;
597 uint8_t rk[k_aes_max_rk_bytes];
598 internal_aes_key_expand(key, nk, nr, rk);
599
600 uint8_t k1[k_aes_block_bytes];
601 uint8_t k2[k_aes_block_bytes];
602 internal_cmac_subkeys(rk, nr, k1, k2);
603
604 /* Block count and whether the last block is a whole 16 bytes. */
605 const uint32_t n_blocks =
606 (msg_len == 0U) ? 1U : ((msg_len + (uint32_t)k_aes_last_byte) / (uint32_t)k_aes_block_bytes);
607 const bool complete = (msg_len != 0U) && ((msg_len % (uint32_t)k_aes_block_bytes) == 0U);
608 const uint32_t last_off = (n_blocks - 1U) * (uint32_t)k_aes_block_bytes;
609
610 uint8_t last[k_aes_block_bytes];
611 internal_cmac_build_last(msg, msg_len, last_off, complete, k1, k2, last);
612
613 uint8_t x[k_aes_block_bytes] = {};
614 uint8_t y[k_aes_block_bytes];
615 for (uint32_t b = 0U; (b + 1U) < n_blocks; ++b) {
616 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
617 y[i] = (uint8_t)(x[i] ^ msg[((uint32_t)b * (uint32_t)k_aes_block_bytes) + (uint32_t)i]);
618 }
619 internal_aes_encrypt(rk, nr, y, x);
620 }
621 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
622 y[i] = (uint8_t)(x[i] ^ last[i]);
623 }
624 internal_aes_encrypt(rk, nr, y, out_mac);
625
626 ra8_secure_memzero(rk, sizeof(rk));
627 ra8_secure_memzero(k1, sizeof(k1));
628 ra8_secure_memzero(k2, sizeof(k2));
629}

646ra8_err_t ra8_sec_cmac_verify(const uint8_t* key,
647 uint16_t key_len,
648 const uint8_t* msg,
649 uint32_t msg_len,
650 const uint8_t* mac,
651 uint16_t mac_len)
652{
653 RA8_CHECK_NULL_PTR(mac, s_tag, "verify: mac");
654 const ra8_err_t ck = internal_cmac_check_args(key, key_len, msg, msg_len);
655 if (ck != k_ra8_ok) {
656 return ck;
657 }
658 uint8_t computed[k_ra8_sec_cmac_tag_bytes];
659 internal_cmac_tag(key, key_len, msg, msg_len, computed);
660 /* Security verdict: reject a wrong-length tag OR a tag that does not match
661 * in constant time. ra8_ct_equal has no data-dependent early-out, so the
662 * timing does not leak how many leading bytes matched (ra8_secure.h). */
663 const bool len_bad = (mac_len != (uint16_t)k_ra8_sec_cmac_tag_bytes);
664 const bool tag_bad = !ra8_ct_equal(computed, mac, (size_t)k_ra8_sec_cmac_tag_bytes);
665 ra8_secure_memzero(computed, sizeof(computed));
666 if (len_bad || tag_bad) {
667 return k_ra8_err_invalid_arg;
668 }
669 return k_ra8_ok;
