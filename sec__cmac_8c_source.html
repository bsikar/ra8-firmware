202/* =============================================================================
203 * Reference backend -- self-contained AES-128/256 + CMAC (FIPS 197 / SP 800-38B)
204 * =============================================================================
205 *
206 * The AES round transform and CMAC subkey/GF(2^128) doubling below are direct
207 * transcriptions of FIPS 197 and NIST SP 800-38B. The S-box and the field
208 * constants are standard values, not tunable parameters; the loops all have
209 * compile-time-fixed bounds. The host suite pins the output to the published
210 * SP 800-38B AES-128 and AES-256 known-answer vectors.

302RA8_INTERNAL static void
303internal_aes_key_expand(const uint8_t* key, uint8_t nk, uint8_t nr, uint8_t* rk)
304{
305 const uint16_t total_words = (uint16_t)((uint16_t)k_aes_state_cols * (uint16_t)(nr + 1U));
306 for (uint16_t i = 0U; i < ((uint16_t)nk * (uint16_t)k_aes_word_bytes); ++i) {
307 rk[i] = key[i];
308 }
309 uint8_t rcon = (uint8_t)k_aes_rcon_seed;
310 for (uint16_t i = nk; i < total_words; ++i) {
311 uint8_t t[k_aes_word_bytes];
312 for (uint8_t j = 0U; j < (uint8_t)k_aes_word_bytes; ++j) {
313 t[j] = rk[((uint16_t)(i - 1U) * (uint16_t)k_aes_word_bytes) + j];
314 }
315 if ((i % nk) == 0U) {
316 const uint8_t tmp = t[0];
317 t[0] = (uint8_t)(s_aes_sbox[t[1]] ^ rcon);
318 t[1] = s_aes_sbox[t[2]];
319 t[2] = s_aes_sbox[t[3]];
320 t[3] = s_aes_sbox[tmp];
321 rcon = internal_aes_xtime(rcon);
322 } else if ((nk > 6U) && ((i % nk) == (uint8_t)k_aes_ext_sub_i)) {
323 for (uint8_t j = 0U; j < (uint8_t)k_aes_word_bytes; ++j) {
324 t[j] = s_aes_sbox[t[j]];
325 }
326 } else {
327 /* No transform: straight copy from t[] below. */
328 }
329 for (uint8_t j = 0U; j < (uint8_t)k_aes_word_bytes; ++j) {
330 rk[((uint16_t)i * (uint16_t)k_aes_word_bytes) + j] =
331 (uint8_t)(rk[((uint16_t)(i - nk) * (uint16_t)k_aes_word_bytes) + j] ^ t[j]);
332 }
333 }
334}

354{
355 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
356 s[i] = s_aes_sbox[s[i]];

363 }

387RA8_INTERNAL static void internal_aes_mix_columns(uint8_t* s)
388{
389 for (uint8_t c = 0U; c < (uint8_t)k_aes_state_cols; ++c) {
390 uint8_t* col = &s[(uint8_t)(c * (uint8_t)k_aes_state_cols)];
391 const uint8_t a0 = col[0];
392 const uint8_t a1 = col[1];
393 const uint8_t a2 = col[2];

395 col[0] = (uint8_t)(internal_aes_xtime(a0) ^ internal_aes_xtime(a1) ^ a1 ^ a2 ^ a3);
396 col[1] = (uint8_t)(a0 ^ internal_aes_xtime(a1) ^ internal_aes_xtime(a2) ^ a2 ^ a3);
397 col[2] = (uint8_t)(a0 ^ a1 ^ internal_aes_xtime(a2) ^ internal_aes_xtime(a3) ^ a3);

424RA8_INTERNAL static void
425internal_aes_encrypt(const uint8_t* rk, uint8_t nr, const uint8_t* in, uint8_t* out)
426{
427 uint8_t s[k_aes_block_bytes];
428 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
429 s[i] = (uint8_t)(in[i] ^ rk[i]);
430 }
431 for (uint8_t round = 1U; round <= nr; ++round) {
432 internal_aes_sub_shift(s);
433 if (round != nr) {
434 internal_aes_mix_columns(s);
435 }
436 const uint8_t* round_key = &rk[(size_t)round * (size_t)k_aes_block_bytes];
437 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
438 s[i] = (uint8_t)(s[i] ^ round_key[i]);
439 }
440 }
441 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
442 out[i] = s[i];
443 }
444}

464RA8_INTERNAL static void internal_cmac_double(const uint8_t* in, uint8_t* out)
465{
466 const uint8_t msb = (uint8_t)(in[0] >> k_aes_msb_shift);
467 for (uint8_t i = 0U; i < (uint8_t)k_aes_last_byte; ++i) {
468 out[i] = (uint8_t)((uint8_t)(in[i] << 1U) | (uint8_t)(in[i + 1U] >> k_aes_msb_shift));
469 }
470 out[k_aes_last_byte] = (uint8_t)(in[k_aes_last_byte] << 1U);
471 if (msb != 0U) {
472 out[k_aes_last_byte] = (uint8_t)(out[k_aes_last_byte] ^ (uint8_t)k_cmac_rb);
473 }
474}

530RA8_INTERNAL static void internal_cmac_build_last(const uint8_t* msg,
531 uint32_t msg_len,
532 uint32_t last_off,
533 bool complete,
534 const uint8_t* k1,
535 const uint8_t* k2,
536 uint8_t* last)
537{
538 if (complete) {
539 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
540 last[i] = (uint8_t)(msg[last_off + i] ^ k1[i]);
541 }
542 return;
543 }
544 const uint32_t rem = msg_len - last_off;
545 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
546 uint8_t byte = 0U;
547 if ((uint32_t)i < rem) {
548 byte = msg[last_off + (uint32_t)i];
549 } else if ((uint32_t)i == rem) {
550 byte = (uint8_t)k_cmac_pad_marker;
551 } else {
552 byte = 0U;
553 }
554 last[i] = (uint8_t)(byte ^ k2[i]);
555 }
556}

581RA8_INTERNAL static void internal_cmac_tag(const uint8_t* key,
582 uint16_t key_len,
583 const uint8_t* msg,
584 uint32_t msg_len,
585 uint8_t* out_mac)
586{
587 const uint8_t nk = (uint8_t)(key_len / (uint16_t)k_aes_word_bytes);
588 const uint8_t nr =
589 (key_len == (uint16_t)k_ra8_sec_cmac_key_128) ? (uint8_t)k_aes_nr_128 : (uint8_t)k_aes_nr_256;
590 uint8_t rk[k_aes_max_rk_bytes];
591 internal_aes_key_expand(key, nk, nr, rk);
592
593 uint8_t k1[k_aes_block_bytes];
594 uint8_t k2[k_aes_block_bytes];
595 internal_cmac_subkeys(rk, nr, k1, k2);
596
597 /* Block count and whether the last block is a whole 16 bytes. */
598 const uint32_t n_blocks =
599 (msg_len == 0U) ? 1U : ((msg_len + (uint32_t)k_aes_last_byte) / (uint32_t)k_aes_block_bytes);
600 const bool complete = (msg_len != 0U) && ((msg_len % (uint32_t)k_aes_block_bytes) == 0U);
601 const uint32_t last_off = (n_blocks - 1U) * (uint32_t)k_aes_block_bytes;
602
603 uint8_t last[k_aes_block_bytes];
604 internal_cmac_build_last(msg, msg_len, last_off, complete, k1, k2, last);
605
606 uint8_t x[k_aes_block_bytes] = {};
607 uint8_t y[k_aes_block_bytes];
608 for (uint32_t b = 0U; (b + 1U) < n_blocks; ++b) {
609 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
610 y[i] = (uint8_t)(x[i] ^ msg[((uint32_t)b * (uint32_t)k_aes_block_bytes) + (uint32_t)i]);
611 }
612 internal_aes_encrypt(rk, nr, y, x);
613 }
614 for (uint8_t i = 0U; i < (uint8_t)k_aes_block_bytes; ++i) {
615 y[i] = (uint8_t)(x[i] ^ last[i]);
616 }
617 internal_aes_encrypt(rk, nr, y, out_mac);
618
619 ra8_secure_memzero(rk, sizeof(rk));
620 ra8_secure_memzero(k1, sizeof(k1));
621 ra8_secure_memzero(k2, sizeof(k2));
622}

639ra8_err_t ra8_sec_cmac_verify(const uint8_t* key,
640 uint16_t key_len,
641 const uint8_t* msg,
642 uint32_t msg_len,
643 const uint8_t* mac,
644 uint16_t mac_len)
645{
646 RA8_CHECK_NULL_PTR(mac, s_tag, "verify: mac");
647 const ra8_err_t ck = internal_cmac_check_args(key, key_len, msg, msg_len);
648 if (ck != k_ra8_ok) {
649 return ck;
650 }
651 uint8_t computed[k_ra8_sec_cmac_tag_bytes];
652 internal_cmac_tag(key, key_len, msg, msg_len, computed);
653 /* Security verdict: reject a wrong-length tag OR a tag that does not match
654 * in constant time. ra8_ct_equal has no data-dependent early-out, so the
655 * timing does not leak how many leading bytes matched (ra8_secure.h). */
656 const bool len_bad = (mac_len != (uint16_t)k_ra8_sec_cmac_tag_bytes);
657 const bool tag_bad = !ra8_ct_equal(computed, mac, (size_t)k_ra8_sec_cmac_tag_bytes);
658 ra8_secure_memzero(computed, sizeof(computed));
659 if (len_bad || tag_bad) {
660 return k_ra8_err_invalid_arg;
661 }
662 return k_ra8_ok;
663}
