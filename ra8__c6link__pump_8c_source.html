53RA8_INTERNAL static bool ra8_c6link_pump_handshake(ra8_c6link_t* link)
54{
55 for (uint16_t waited = 0U; waited < (uint16_t)k_ra8_c6link_hs_wait_ms;
56 waited = (uint16_t)(waited + (uint16_t)k_ra8_c6link_hs_poll_ms)) {
57 if (link->transport.handshake_active(link->transport.ctx)) {
58 return true;
59 }

81RA8_INTERNAL static bool ra8_c6link_pump_receive(ra8_c6link_t* link, ra8_c6link_stats_t* stats)
82{
83 ra8_c6link_rx_view_t view = {};
84 const ra8_c6link_frame_class_t cls = ra8_c6link_priv_frame_classify(link->rx, &view);
85
86 switch (cls) {
87 case k_ra8_c6link_frame_idle:
88 stats->idle++;
89 return false;
90 case k_ra8_c6link_frame_malformed:
91 stats->malformed++;
92 return false;
93 case k_ra8_c6link_frame_bad_checksum:
94 stats->bad_checksum++;
95 return false;
96 case k_ra8_c6link_frame_data:

98 break;
99 }
100
101 stats->data++;
102 return ra8_c6link_priv_dispatch(link, &view);

105RA8_PRIV ra8_err_t ra8_c6link_priv_pump(ra8_c6link_t* link,
106 uint16_t max_transactions,
107 ra8_c6link_stats_t* stats)
108{
109 if ((link == nullptr) || (stats == nullptr)) {
110 return k_ra8_err_null_ptr;
111 }
112 link->stats = stats;
113 uint16_t misses = 0U;
114 ra8_err_t verdict = k_ra8_ok;
115
116 for (uint16_t i = 0U; i < max_transactions; i++) {
117 if (!ra8_c6link_pump_handshake(link)) {
118 stats->hs_timeouts++;
119 misses++;
120 if (misses >= (uint16_t)k_ra8_c6link_hs_giveup) {
121 break;
122 }
123 continue;
124 }
125 misses = 0U;
126
127 if (link->tx_len != 0U) {
128 ra8_c6link_priv_frame_seal(link->tx, link->tx_if, 0U, link->tx_len);
129 link->tx_len = 0U;
130 } else {
131 ra8_c6link_priv_frame_filler(link->tx);
132 }
133
134 const ra8_err_t moved = link->transport.transfer(link->transport.ctx,
135 link->tx,
136 link->rx,
137 (uint16_t)k_ra8_c6link_frame_bytes);
138 if (moved != k_ra8_ok) {
139 verdict = k_ra8_err_spi_error;
140 break;
141 }
142 stats->transfers++;
143
144 if (ra8_c6link_pump_receive(link, stats)) {
145 break;
146 }
147 link->transport.delay_ms(link->transport.ctx, (uint16_t)k_ra8_c6link_gap_ms);
148 }
149
150 link->stats = nullptr;
151 if (verdict != k_ra8_ok) {
152 return verdict;
153 }
154 return (stats->transfers == 0U) ? k_ra8_err_hw_timeout : k_ra8_ok;
155}
