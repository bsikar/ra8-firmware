97bool board_usb_loop_setup(uc_engine* uc, uint16_t req, uint16_t val, uint16_t indx, uint16_t leng)
98{
99 const uint8_t bm = (uint8_t)(req & (uint16_t)k_loop_req_bm_mask);
100 const uint8_t breq = (uint8_t)(req >> k_usb_byte_bits);
101 /* A new SETUP supersedes whatever control transfer preceded it: drop stale
102 * DCP staging and consume a stale CCPL (the fw host is strictly serial, so
103 * any completion still latched here belongs to the finished transfer). */
104 loop_reset_dcp();
105 (void)host_take_ccpl();
106 s_loop_pending_cfg = false;
107 s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_usbreq)] = req;
108 s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_usbval)] = val;
109 s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_usbindx)] = indx;
110 s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_usbleng)] = leng;
111 s_loop_setups++;
112 char line[k_usb_log_width];
113 (void)snprintf(line,
114 sizeof(line),
115 "loop: SETUP bm=0x%02X bReq=0x%02X wVal=0x%04X wLen=%u",
116 (unsigned)bm,
117 (unsigned)breq,
118 (unsigned)val,
119 (unsigned)leng);
120 usb_log_line(line);
121 if ((bm == (uint8_t)k_usb_dir_host_to_device) && (breq == (uint8_t)k_usb_req_set_address)) {
122 /* The device SIE latches USBADDR and runs the whole status stage itself
123 * (mirrors ::host_apply_no_data): the host needs no device CCPL. */
124 s_usb.reg[usb_word((uint64_t)k_ra8_usb_off_usbaddr)] =
125 (uint16_t)(val & (uint16_t)k_ra8_usbaddr_addr_mask);
126 s_usb.dvsq = (uint16_t)k_ra8_dvsq_address;
127 usb_intsts0_set((uint8_t)k_ra8_int0_bit_dvst);
128 usb_raise_irq(uc);
129 return true;
130 }
131 if ((bm == (uint8_t)k_usb_dir_host_to_device) && (breq == (uint8_t)k_usb_req_set_config)) {
132 s_loop_pending_cfg = true; /* configured-state advance rides the CCPL. */
133 }
134 /* For an OUT-data request the payload arrives via board_usb_loop_ctrl_out. */
135 s_usb.setup_valid = true;
136 if ((bm & (uint8_t)k_usb_dir_device_to_host) != 0U) {
137 s_usb.ctsq = (uint16_t)k_ra8_ctsq_rdds;
138 } else if (leng != 0U) {
139 s_usb.ctsq = (uint16_t)k_ra8_ctsq_wrds;
140 } else {
141 s_usb.ctsq = (uint16_t)k_ra8_ctsq_wrnd;
142 }
143 usb_intsts0_set((uint8_t)k_ra8_int0_bit_ctrt);
144 usb_raise_irq(uc);
145 return false;
146}

168uint16_t board_usb_loop_ctrl_in_read(uint8_t* dst, uint16_t cap)
169{
170 const uint16_t avail = board_usb_loop_ctrl_in_avail();
171 const uint16_t n = (avail < cap) ? avail : cap;
172 for (uint16_t i = 0U; i < n; i++) {
173 dst[i] = s_usb.dcp_in.data[s_loop_dcp_rd + i];
174 }
175 s_loop_dcp_rd = (uint16_t)(s_loop_dcp_rd + n);
176 if ((n > 0U) && (s_loop_dcp_rd >= s_usb.dcp_in.len)) {
177 usb_detect_class(s_usb.dcp_in.data, s_usb.dcp_in.len);
178 s_usb.dcp_in.len = 0U;
179 s_usb.dcp_in.valid = false;
180 s_loop_dcp_rd = 0U;
181 }
182 return n;
183}

192void board_usb_loop_ctrl_out(uc_engine* uc, const uint8_t* data, uint16_t len)
193{
194 uint16_t n = len;
195 if (n > (uint16_t)sizeof(s_usb.dcp_out.data)) {
196 n = (uint16_t)sizeof(s_usb.dcp_out.data);
197 }
198 (void)memcpy(s_usb.dcp_out.data, data, n);
199 s_usb.dcp_out.len = n;

204 usb_intsts0_set((uint8_t)k_ra8_int0_bit_brdy);

216void board_usb_loop_bulk_out(uc_engine* uc, uint8_t ep, const uint8_t* data, uint16_t len)
217{
218 usb_out_buf_t* b = &s_usb.pipe_out[ep % k_usb_pipe_count];
219 uint16_t n = len;
220 if (n > (uint16_t)sizeof(b->data)) {
221 n = (uint16_t)sizeof(b->data);
222 }
223 (void)memcpy(b->data, data, n);
224 b->len = n;
225 b->rd = 0U;
226 b->ready = true;
227 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_brdysts);
228 s_usb.reg[w] = (uint16_t)(s_usb.reg[w] | (uint16_t)(1U << (ep % k_usb_pipe_count)));
229 usb_intsts0_set((uint8_t)k_ra8_int0_bit_brdy);
230 s_loop_bulk_out_pkts++;
231 usb_raise_irq(uc);

243uint16_t board_usb_loop_bulk_in_read(uc_engine* uc, uint8_t ep, uint8_t* dst, uint16_t cap)
244{
245 const uint8_t pipe = (uint8_t)(ep % k_usb_pipe_count);
246 usb_in_buf_t* b = &s_usb.pipe_in[pipe];
247 const uint16_t avail = board_usb_loop_bulk_in_avail(ep);
248 const uint16_t n = (avail < cap) ? avail : cap;
249 for (uint16_t i = 0U; i < n; i++) {
250 dst[i] = b->data[s_loop_pipe_rd[pipe] + i];
251 }
252 s_loop_pipe_rd[pipe] = (uint16_t)(s_loop_pipe_rd[pipe] + n);
253 if ((n > 0U) && (s_loop_pipe_rd[pipe] >= b->len)) {
254 b->len = 0U;
255 b->valid = false;
256 s_loop_pipe_rd[pipe] = 0U;
257 s_loop_bulk_in_pkts++;
258 /* Transmit buffer drained: raise the device's BEMP for this pipe so its
259 * firmware can queue the next packet (mirrors ::host_echo_read_in). */
260 const uint32_t w = usb_word((uint64_t)k_ra8_usb_off_bempsts);
261 s_usb.reg[w] = (uint16_t)(s_usb.reg[w] | (uint16_t)(1U << pipe));
262 usb_intsts0_set((uint8_t)k_ra8_int0_bit_bemp);
263 usb_raise_irq(uc);
264 }
265 return n;
266}
