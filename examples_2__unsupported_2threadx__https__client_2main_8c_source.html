493static int demo_bio_send(void* ctx, const unsigned char* buf, size_t len)
494{
495 (void)ctx;
496 if (buf == nullptr || len == 0U) {
497 return 0;
498 }
499 NX_PACKET* pkt = NX_NULL;
500 UINT s = nx_packet_allocate(&s_packet_pool, &pkt, NX_TCP_PACKET, NX_WAIT_FOREVER);
501 if (s != NX_SUCCESS || pkt == NX_NULL) {
502 return MBEDTLS_ERR_SSL_INTERNAL_ERROR;
503 }
504 s = nx_packet_data_append(pkt, (VOID*)buf, (ULONG)len, &s_packet_pool, NX_WAIT_FOREVER);
505 if (s != NX_SUCCESS) {
506 (void)nx_packet_release(pkt);
507 return MBEDTLS_ERR_SSL_INTERNAL_ERROR;
508 }
509 s = nx_tcp_socket_send(&s_tls_socket, pkt, (ULONG)k_demo_recv_timeout);
510 if (s != NX_SUCCESS) {
511 (void)nx_packet_release(pkt);
512 if (s == NX_NO_PACKET || s == NX_WINDOW_OVERFLOW) {
513 return MBEDTLS_ERR_SSL_WANT_WRITE;
514 }
515 return MBEDTLS_ERR_NET_SEND_FAILED;
516 }
517 return (int)len;

535{

546 return MBEDTLS_ERR_NET_RECV_FAILED;

552 /* Should not happen with a 1568-byte pool packet, but cap. */
553 copied = (ULONG)len;
554 }

648static int demo_verify_cert_pin(void)
649{

680static int demo_http_get(void)
681{
682 int n = snprintf(
683 (char*)s_request_buf,
684 sizeof(s_request_buf),
685 "GET / HTTP/1.1\r\nHost: %s\r\nConnection: close\r\nUser-Agent: ra8d2-https-client/0.1\r\n\r\n",
686 k_demo_host_name);
687 if (n <= 0 || (size_t)n >= sizeof(s_request_buf)) {
688 return -1;
689 }
690 size_t written = 0U;
691 while (written < (size_t)n) {
692 int rc = mbedtls_ssl_write(&s_ssl, s_request_buf + written, (size_t)n - written);
693 if (rc == MBEDTLS_ERR_SSL_WANT_READ || rc == MBEDTLS_ERR_SSL_WANT_WRITE) {
694 continue;
695 }
696 if (rc < 0) {
697 return rc;
698 }
699 written += (size_t)rc;
700 }
701
702 /* Read until we've dumped k_demo_dump_bytes of body or the peer
703 * closes. */
704 uint32_t body_dumped = 0U;
705 uint8_t saw_header_end = 0U;
706 uint8_t match = 0U; /* tracks "\r\n\r\n" run. */
707 while (body_dumped < (uint32_t)k_demo_dump_bytes) {
708 int rc = mbedtls_ssl_read(&s_ssl, s_response_buf, sizeof(s_response_buf));
709 if (rc == MBEDTLS_ERR_SSL_WANT_READ || rc == MBEDTLS_ERR_SSL_WANT_WRITE) {
710 continue;
711 }
712 if (rc == MBEDTLS_ERR_SSL_PEER_CLOSE_NOTIFY || rc == 0) {
713 break;
714 }
715 if (rc < 0) {
716 return rc;
717 }
718 int start = 0;
719 if (saw_header_end == 0U) {
720 /* Walk byte-by-byte looking for CRLFCRLF. */
721 for (int i = 0; i < rc; i++) {
722 uint8_t c = s_response_buf[i];
723 if ((match == 0U && c == k_ascii_cr) || (match == 2U && c == k_ascii_cr)) {
724 match++;
725 } else if ((match == 1U && c == k_ascii_lf) || (match == 3U && c == k_ascii_lf)) {
726 match++;
727 } else {
728 match = 0U;
729 }
730 if (match == 4U) {
731 saw_header_end = 1U;
732 start = i + 1;
733 break;
734 }
735 }
736 if (saw_header_end == 0U) {
737 continue;
738 }
739 }
740 int avail = rc - start;
741 if (avail <= 0) {
742 continue;
743 }
744 uint32_t take = (uint32_t)avail;
745 if (body_dumped + take > (uint32_t)k_demo_dump_bytes) {
746 take = (uint32_t)k_demo_dump_bytes - body_dumped;
747 }
748 demo_write_bytes(s_response_buf + start, take);
749 body_dumped += take;
750 }
751 return 0;
752}

764static int demo_tls_session(void)
765{
766 mbedtls_ssl_init(&s_ssl);
767 mbedtls_ssl_config_init(&s_ssl_cfg);
768
769 /* Mbed TLS 4.x removed MBEDTLS_ENTROPY_C and MBEDTLS_CTR_DRBG_C from
770 * the standard build path; the SSL layer pulls random bytes from PSA
771 * crypto (psa_generate_random), which the firmware wires to the RSIP
772 * TRNG via the mbedtls_psa_external_get_random() implementation
773 * above. The legacy mbedtls_entropy_* / mbedtls_ctr_drbg_* setup
774 * dance is therefore no longer needed. We still need PSA itself
775 * online; demo_thread_entry calls psa_crypto_init() before this. */
776 (void)demo_entropy_source; /* Retained for reference; unused on 4.x. */
777 (void)s_drbg;
778 (void)s_entropy;
779
780 if (mbedtls_ssl_config_defaults(&s_ssl_cfg,
781 MBEDTLS_SSL_IS_CLIENT,
782 MBEDTLS_SSL_TRANSPORT_STREAM,
783 MBEDTLS_SSL_PRESET_DEFAULT) != 0) {
784 return -1;
785 }
786 /* No CA bundle is provisioned in this demo -- we authenticate the
787 * peer by SHA-256 cert-pin instead. ``OPTIONAL`` lets the
788 * handshake complete even with an empty trust store; the pin
789 * check below is what gates the GET. */
790 mbedtls_ssl_conf_authmode(&s_ssl_cfg, MBEDTLS_SSL_VERIFY_OPTIONAL);
791 /* Mbed TLS 4.x removed mbedtls_ssl_conf_rng() -- the SSL layer
792 * pulls random bytes from PSA crypto's psa_generate_random() now,
793 * which is wired up by demo_psa_init() below via the
794 * MBEDTLS_PSA_CRYPTO_EXTERNAL_RNG hook
795 * (mbedtls_psa_external_get_random). */
796
797 if (mbedtls_ssl_setup(&s_ssl, &s_ssl_cfg) != 0) {
798 return -1;
799 }
800 if (mbedtls_ssl_set_hostname(&s_ssl, k_demo_host_name) != 0) {
801 return -1;
802 }
803 mbedtls_ssl_set_bio(&s_ssl, nullptr, demo_bio_send, demo_bio_recv, nullptr);
804
805 /* Drive the handshake to completion. */
806 int rc = 0;
807 while ((rc = mbedtls_ssl_handshake(&s_ssl)) != 0) {
808 if (rc != MBEDTLS_ERR_SSL_WANT_READ && rc != MBEDTLS_ERR_SSL_WANT_WRITE) {
809 return rc;
810 }
811 }
812 if (demo_verify_cert_pin() != 0) {
813 demo_print("[https] cert pin MISMATCH -- aborting\r\n");
814 return -1;
815 }
816 demo_print("[https] handshake OK, cert pin matches, dumping body\r\n");
817 return demo_http_get();
818}

830static void demo_thread_entry(ULONG thread_input)
831{
832 (void)thread_input;
833 /* Bring PSA crypto online before any TLS / X.509 work. The library
834 * lazy-initialises inside the SSL layer, but doing it explicitly
835 * here surfaces failures earlier and binds the RSIP-TRNG external
836 * RNG hook before the handshake's first random call. */
837 if (psa_crypto_init() != PSA_SUCCESS) {
838 demo_print("[https] psa_crypto_init failed\r\n");

843 demo_print("[https] NetX bring-up failed\r\n");

850 }
851 if (demo_tls_session() != 0) {
852 demo_print("[https] TLS session failed\r\n");

854 demo_print("\r\n[https] done\r\n");
