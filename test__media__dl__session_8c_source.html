72static ra8_err_t robo_get_buf(void* ctx,
73 const char* url,
74 const mdl_net_req_t* req,
75 char* buf,
76 size_t cap,
77 size_t* out_len,
78 mdl_net_resp_t* resp)
79{
80 robo_net_t* f = (robo_net_t*)ctx;
81 f->calls += 1U;
82 (void)snprintf(f->last_url, sizeof(f->last_url), "%s", url);
83 (void)snprintf(f->last_ua,
84 sizeof(f->last_ua),
85 "%s",
86 (req->user_agent != nullptr) ? req->user_agent : "");
87 if (resp != nullptr) {
88 resp->status = f->status; /* the session reads the status class through resp */
89 }
90 size_t got = 0U;
91 if ((f->rc == k_ra8_ok) && (f->body != nullptr)) {
92 const int w = snprintf(buf, cap, "%s", f->body);
93 got = (w < 0) ? 0U : (size_t)w;
94 if (got >= cap) {
95 got = cap - 1U;
96 }
97 }
98 if (out_len != nullptr) {
99 *out_len = got;
100 }
101 return f->rc;
102}

188static void test_session_ua_build_honest(void)
189{
190 TEST_BEGIN("session ua build honest");
191 char ua[k_ua_cap];
192 /* A/V1 + B control: a real contact yields the full honest identity. */
193 TEST_ASSERT(mdl_session_build_ua("ops@example.org", ua, sizeof(ua)));
194 TEST_ASSERT(strstr(ua, "media_dl/") == ua);
195 TEST_ASSERT(strstr(ua, "github.com/bsikar/ra8-firmware") != nullptr);
196 TEST_ASSERT(strstr(ua, "ops@example.org") != nullptr);
197 /* No browser-impersonation token survives anywhere in the string. */
198 TEST_ASSERT(strstr(ua, "Mozilla") == nullptr);
199 TEST_ASSERT(strstr(ua, "Chrome") == nullptr);
200 TEST_ASSERT(strstr(ua, "AppleWebKit") == nullptr);
201 TEST_ASSERT(strstr(ua, "Safari") == nullptr);
202 /* A/V2: NULL contact -> no clause, false; still honest, still no impersonation. */
203 TEST_ASSERT(!mdl_session_build_ua(nullptr, ua, sizeof(ua)));
204 TEST_ASSERT(strstr(ua, "media_dl/") == ua);
205 TEST_ASSERT(strstr(ua, "Mozilla") == nullptr);
206 /* A/V3: empty contact -> treated as no contact. */
207 TEST_ASSERT(!mdl_session_build_ua("", ua, sizeof(ua)));
208 /* B: NULL out and zero cap are refused. */
209 TEST_ASSERT(!mdl_session_build_ua("ops@example.org", nullptr, sizeof(ua)));
210 TEST_ASSERT(!mdl_session_build_ua("ops@example.org", ua, 0U));
211 TEST_END("session ua build honest");
212}

312static void test_session_crawl_delay_respected(void)
313{
314 TEST_BEGIN("session crawl-delay respected");
315 robo_net_t f = {.body = "User-agent: *\nCrawl-delay: 7\n",
316 .rc = k_ra8_ok,
317 .status = (long)k_http_ok};
318 mdl_net_iface_t net = robo_iface(&f);
319 sess_reset(&net, "media_dl/0.1.0", true);
320 /* V1: the host's Crawl-delay is reported in milliseconds. */
321 uint32_t crawl = 0U;
322 TEST_ASSERT(mdl_session_url_allowed(&s_sess, "https://slow/p", &crawl));
323 TEST_ASSERT_EQ((uint32_t)k_crawl_ms, crawl);
324 /* V2: a NULL out pointer is accepted (no crash, nothing written). */
325 TEST_ASSERT(mdl_session_url_allowed(&s_sess, "https://slow/q", nullptr));
326 /* V3: a host with no Crawl-delay reports zero. */
327 robo_net_t f2 = {.body = "User-agent: *\nDisallow: /no\n",
328 .rc = k_ra8_ok,
329 .status = (long)k_http_ok};
330 mdl_net_iface_t net2 = robo_iface(&f2);
331 sess_reset(&net2, "media_dl/0.1.0", true);
332 uint32_t none = k_nonzero_marker;
333 TEST_ASSERT(mdl_session_url_allowed(&s_sess, "https://fast/p", &none));
334 TEST_ASSERT_EQ((uint32_t)0, none);
335 /* Respected: the crawl floor RAISES a lower base delay ... */
336 rec_clock_t clk = {};
337 mdl_politeness_t pol;
338 mdl_politeness_init_clock(&pol, k_pol_seed, rec_clock_sleep, &clk);
339 (void)mdl_politeness_wait(&pol,
340 max_u32(k_base_lo_min, k_crawl_ms),
341 max_u32(k_base_lo_max, k_crawl_ms));
342 TEST_ASSERT_EQ((uint32_t)k_crawl_ms, clk.last_ms);
343 /* ... and never LOWERS a stricter configured base delay. */
344 (void)mdl_politeness_wait(&pol,
345 max_u32(k_base_hi_min, k_crawl_ms),
346 max_u32(k_base_hi_max, k_crawl_ms));
347 TEST_ASSERT(clk.last_ms >= (uint32_t)k_base_hi_min);
348 TEST_END("session crawl-delay respected");
349}
