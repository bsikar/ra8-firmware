31void eth_hook_return(uc_engine* uc, uint32_t r0)
32{
33 uint32_t lr = 0U;
34 (void)uc_reg_read(uc, UC_ARM_REG_LR, &lr);
35 (void)uc_reg_write(uc, UC_ARM_REG_R0, &r0);
36 uint32_t pc = lr & ~1U; /* drop the Thumb bit; the M-class core stays Thumb. */
37 (void)uc_reg_write(uc, UC_ARM_REG_PC, &pc);
38 (void)uc_emu_stop(uc); /* relaunch from the returned PC (chunk contract). */
39}

42void eth_seam_hook(uc_engine* uc, const uint8_t* elf, long len, const char* name, void* cb)
43{
44 const uint32_t addr = elf_sym_addr(elf, len, name, nullptr);
45 if (addr == 0U) {
46 return;
47 }
48 enum : uint8_t {
49 k_seam_hook_slots = 24U,
50 };
51 static uc_hook s_handles[k_seam_hook_slots];
52 static uint32_t s_n;
53 if (s_n < (uint32_t)(sizeof(s_handles) / sizeof(s_handles[0]))) {
54 (void)uc_hook_add(uc, &s_handles[s_n], UC_HOOK_CODE, cb, nullptr, addr, addr);
55 s_n++;
56 }

108void sym_trace_install(uc_engine* uc,
109 const uint8_t* elf,
110 long len,
111 const char* const* names,
112 uint32_t count)
113{
114 static uc_hook s_th[k_trace_sym_max];
115 for (uint32_t i = 0U; (i < count) && (i < (uint32_t)k_trace_sym_max); i++) {
116 const uint32_t addr = elf_sym_addr(elf, len, names[i], nullptr);
117 if (addr == 0U) {
118 (void)fprintf(stderr, " [symtrace] %s: symbol not found -- skipped\n", names[i]);
119 continue;
120 }
121 (void)uc_hook_add(uc, &s_th[i], UC_HOOK_CODE, (void*)on_sym_trace, (void*)names[i], addr, addr);
122 (void)fprintf(stderr, " [symtrace] %s armed @ 0x%08X\n", names[i], (unsigned)addr);
123 }
