495RA8_INTERNAL static bool internal_cli_ends_ci(const char* text, const char* suffix)
496{
497 const size_t tl = strlen(text);
498 const size_t sl = strlen(suffix);
499 if (sl > tl) {
500 return false;
501 }
502 for (size_t i = 0U; i < sl; ++i) {
503 char a = text[tl - sl + i];
504 char b = suffix[i];
505 if ((a >= 'A') && (a <= 'Z')) {
506 a = (char)(a + ('a' - 'A'));
507 }
508 if ((b >= 'A') && (b <= 'Z')) {
509 b = (char)(b + ('a' - 'A'));
510 }
511 if (a != b) {
512 return false;
513 }
514 }
515 return true;
516}

772RA8_INTERNAL static ra8_err_t internal_validate_mode_fields(const mdl_args_t* a,
773 ra8_io_stream_t* diagnostic,
774 mdl_cli_mode_t selected)
775{
776 const bool config_mode =
777 (selected == k_mdl_cli_mode_series) || (selected == k_mdl_cli_mode_search) ||
778 (selected == k_mdl_cli_mode_browse) || (selected == k_mdl_cli_mode_update_all);
779 if (config_mode && ((a->cfg == nullptr) || (a->cfg[0] == '\0'))) {
780 return internal_cli_invalid(diagnostic, "this mode requires --config SITE.conf");
781 }
782 if ((selected == k_mdl_cli_mode_search) && (a->search[0] == '\0')) {
783 return internal_cli_invalid(diagnostic, "--search requires a non-empty term");
784 }
785 if ((selected == k_mdl_cli_mode_pack) && (a->format == nullptr)) {
786 return internal_cli_invalid(diagnostic, "--pack requires an explicit --format");
787 }
788 if ((selected == k_mdl_cli_mode_verify) && (a->verify_dir != nullptr) && (a->out != nullptr)) {
789 return internal_cli_invalid(
790 diagnostic,
791 "--verify DIR and --out DIR are alternate directory spellings; use one");
792 }
793 if ((selected == k_mdl_cli_mode_artifact) &&
794 (strncmp(a->page_url, "https://", strlen("https://")) != 0)) {
795 return internal_cli_invalid(diagnostic, "direct artifact downloads require an https:// URL");
796 }
797 return k_ra8_ok;

832ra8_err_t mdl_cli_validate(const mdl_args_t* a, ra8_io_stream_t* diagnostic, mdl_cli_mode_t* mode)
833{
834 if ((a == nullptr) || (diagnostic == nullptr) || (mode == nullptr)) {
835 return k_ra8_err_null_ptr;
836 }
837 *mode = k_mdl_cli_mode_invalid;
838 if (a->bad) {
839 return internal_cli_invalid(diagnostic, "unknown, duplicate, or missing-value option");
840 }
841 size_t primary_count = 0U;
842 const mdl_cli_mode_t selected = internal_resolve_mode(a, &primary_count);
843 if (primary_count == 0U) {
844 return internal_cli_invalid(diagnostic, "exactly one command mode is required");
845 }
846 if (primary_count != 1U) {
847 return internal_cli_invalid(diagnostic, "command modes are mutually exclusive");
848 }
849 ra8_err_t err = internal_validate_allowed_args(a, diagnostic, selected);
850 if (err == k_ra8_ok) {
851 err = internal_validate_mode_fields(a, diagnostic, selected);
852 }
853 if (err == k_ra8_ok) {
854 err = internal_validate_network_args(a, diagnostic);
855 }
856 if (err != k_ra8_ok) {
857 return err;
858 }
859 *mode = selected;
860 return k_ra8_ok;

862mdl_run_opts_t mdl_cli_run_opts(const mdl_args_t* a)
863{
864 const uint64_t max_bytes = (a->max_bytes == nullptr)
