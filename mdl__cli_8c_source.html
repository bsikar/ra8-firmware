75RA8_INTERNAL static bool take_opt(char** argv, int argc, int* i, const char* flag, const char** dst)
76{
77 if ((argv[*i] == nullptr) || (strcmp(argv[*i], flag) != 0)) {
78 return false;
79 }
80 if ((*i + 1) < argc) {
81 *i += 1;
82 *dst = argv[*i];
83 }
84 return true;
85}

88RA8_INTERNAL static bool take_flag(const char* arg, const char* flag, bool* dst)
89{
90 if ((arg != nullptr) && (strcmp(arg, flag) == 0)) {
91 *dst = true;
92 return true;

94 return false;

109void mdl_cli_parse(int argc, char** argv, mdl_args_t* a)
110{
111 /* Table-driven long options: each entry binds a flag to the field it fills. */
112 const struct {
113 const char* flag;
114 const char** dst;
115 } opts[] = {
116 {"--config", &a->cfg},
117 {"--series", &a->series},
118 {"--out", &a->out},
119 {"--attr", &a->attr},
120 {"--chapters", &a->chapters},
121 {"--from", &a->from},
122 {"--max", &a->max},
123 {"--seed", &a->seed},
124 {"--timeout", &a->timeout},
125 {"--format", &a->format},
126 {"--pack", &a->pack},
127 {"--contact", &a->contact},
128 {"--max-bytes", &a->max_bytes},
129 {"--remove", &a->remove_series},
130 {"--search", &a->search},
131 {"--pick", &a->pick},
132 };
133 for (int i = 1; i < argc; ++i) {
134 if (parse_bool_flags(argv[i], a)) {
135 continue;
136 }
137 bool matched = false;
138 for (size_t k = 0U; (k < (sizeof(opts) / sizeof(opts[0]))) && !matched; ++k) {
139 matched = take_opt(argv, argc, &i, opts[k].flag, opts[k].dst);
140 }
141 if (matched) {
142 continue;
143 }
144 if ((argv[i] != nullptr) && (argv[i][0] != '-')) {
145 a->page_url = argv[i];
146 continue;
147 }
148 a->bad = true;
149 }
150}

169RA8_INTERNAL static bool parse_ul(const char* s, unsigned long* out)
170{
171 if (s == nullptr) {
172 return false;
173 }
174 if (s[0] < '0') {
175 return false; /* rejects "", sign, whitespace, and other leading non-digits */
176 }
177 if (s[0] > '9') {
178 return false;
179 }
180 errno = 0;
181 char* end = nullptr;
182 const unsigned long v = strtoul(s, &end, k_cli_dec_base);
183 if (errno != 0) {
184 return false;
185 }
186 if (*end != '\0') {
187 return false; /* trailing garbage ("12abc") is not a number */
188 }
189 *out = v;
190 return true;
191}

219RA8_INTERNAL static bool parse_l(const char* s, long* out)
220{
221 if (s == nullptr) {
222 return false;
223 }
224 if (s[0] == '\0') {
225 return false;
226 }
227 errno = 0;
228 char* end = nullptr;
229 const long v = strtol(s, &end, k_cli_dec_base);
230 if (errno != 0) {
231 return false;
232 }
233 if (end == s) {
234 return false;
235 }
236 if (*end != '\0') {
237 return false;
238 }
239 *out = v;
240 return true;

272bool mdl_cli_parse_nums(const mdl_args_t* a, mdl_nums_t* n)
273{
274 unsigned long timeout_ul = 0UL;
275 unsigned long chapters_ul = 0UL;
276 unsigned long max_ul = 0UL;
277 if (!opt_ul("timeout", a->timeout, (unsigned long)k_req_timeout_def, &timeout_ul)) {
278 return false;
279 }
280 if (!opt_ul("chapters", a->chapters, 1UL, &chapters_ul)) {
281 return false;
282 }
283 if (!opt_ul("max", a->max, 0UL, &max_ul)) {
284 return false;
285 }
286 if (!opt_ull("seed", a->seed, 1U, &n->seed)) {
287 return false;
288 }
289 uint64_t max_bytes = 0U;
290 if (!opt_ull("max-bytes", a->max_bytes, 0U, &max_bytes)) {
291 return false; /* validate presence-garbage; the default is applied in run_opts */
292 }
293 n->from_present = (a->from != nullptr);
294 n->from_num = 0L;
295 if (a->from != nullptr) {
296 if (!parse_l(a->from, &n->from_num)) {
297 (void)fprintf(stderr, "media_dl: --from expects an integer, got '%s'\n", a->from);
298 return false;
299 }
300 }
301 unsigned long pick_ul = 0UL;
302 if (!opt_ul("pick", a->pick, 0UL, &pick_ul)) {
303 return false;
304 }
305 n->timeout = (uint32_t)timeout_ul;
306 n->chapters = (chapters_ul == 0UL) ? 1U : (size_t)chapters_ul;
307 n->max_imgs = (uint32_t)max_ul;
308 n->pick = (size_t)pick_ul;
309 return true;
310}
