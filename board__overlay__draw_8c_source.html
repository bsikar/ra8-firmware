161static void draw_glyph(uint16_t* out,
162 uint16_t w,
163 uint16_t h,
164 int32_t x,
165 int32_t y,
166 char ch,
167 uint16_t color,
168 int32_t sc)
169{
170 uint8_t c = (uint8_t)ch;
171 if ((c < (uint8_t)k_ovl_glyph_first) || (c > (uint8_t)k_ovl_glyph_last)) {
172 c = (uint8_t)k_ovl_glyph_first;
173 }
174 const uint8_t* g = k_font5x7[c - (uint8_t)k_ovl_glyph_first];
175 for (uint32_t col = 0U; col < (uint32_t)k_ovl_glyph_w; col++) {
176 for (uint32_t row = 0U; row < (uint32_t)k_ovl_glyph_h; row++) {
177 if (((g[col] >> row) & 1U) != 0U) {
178 fill_rect(out, w, h, x + ((int32_t)col * sc), y + ((int32_t)row * sc), sc, sc, color);
179 }
180 }
181 }

185int32_t draw_text(uint16_t* out,
186 uint16_t w,
187 uint16_t h,
188 int32_t x,
189 int32_t y,
190 const char* s,
191 uint16_t color,
192 int32_t sc)
193{
194 const int32_t adv = ((int32_t)k_ovl_glyph_w + 1) * sc;
195 int32_t cx = x;
196 for (uint32_t i = 0U; (s != nullptr) && (s[i] != '\0'); i++) {
197 draw_glyph(out, w, h, cx, y, s[i], color, sc);
198 cx += adv;
199 }
200 return cx;
201}

205{

219int32_t kv_row(uint16_t* out,
220 uint16_t w,
221 uint16_t h,
222 int32_t x,
223 int32_t y,
224 const char* label,
225 const char* value,
226 uint16_t val_color)
227{
228 const int32_t adv = ((int32_t)k_ovl_glyph_w + 1);
229 const int32_t value_dx = (int32_t)k_kv_label_cols * adv; /* fixed label column so values align */
230 draw_text(out, w, h, x, y, label, (uint16_t)k_ovl_dim, 1);
231 draw_text(out, w, h, x + value_dx, y, value, val_color, 1);
232 return y + (int32_t)k_row_step;
233}

236void blit_panel(uint16_t* out,
237 uint16_t w,
238 uint16_t h,
239 const uint16_t* panel,
240 uint16_t panel_w,
241 uint16_t panel_h)
242{
243 for (uint16_t y = 0U; y < h; y++) {
244 for (uint16_t x = 0U; x < panel_w; x++) {
245 const bool in_panel = (panel != nullptr) && (y < panel_h);
246 const uint16_t c =
247 in_panel ? panel[((size_t)y * (size_t)panel_w) + (size_t)x] : (uint16_t)k_ovl_panel_bg;
248 out[((size_t)y * (size_t)w) + (size_t)x] = c;
249 }
250 }
