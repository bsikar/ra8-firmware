434/* `priv_exfat_upcase_unit()`: see header for the documented contract. */
435uint16_t priv_exfat_upcase_unit(uint16_t unit)
436{
437 const uint32_t want = (uint32_t)unit;
438 if (want <= (uint32_t)k_utf_ascii_max) {
439 /* Not a different rule -- the same answer, reached without the walk. The
440 * table's first 128 entries are the identity apart from a-z -> A-Z, which
441 * is what `priv_ascii_upper` does, and what
442 * `test_upcase_ascii_fast_path_is_the_table` re-derives for all 128. */
443 return (uint16_t)(uint32_t)(unsigned char)priv_ascii_upper((char)(unsigned char)want);
444 }
445 uint32_t idx = 0U;
446 uint32_t w = 0U;
447 while (w < (uint32_t)k_exfat_upc_words) {
448 const uint32_t v = (uint32_t)priv_rd16(&s_exfat_upcase[(size_t)w * 2U]);
449 w++;
450 if (v != (uint32_t)k_exfat_upc_run_tag) {
451 if (idx == want) {
452 return (uint16_t)v;
453 }
454 idx++;
455 continue;
456 }
457 if (w >= (uint32_t)k_exfat_upc_words) {
458 break; /* the trailing tag: U+FFFF, whose up-case is itself */
459 }
460 const uint32_t run = (uint32_t)priv_rd16(&s_exfat_upcase[(size_t)w * 2U]);
461 w++;
462 if (want < (idx + run)) {
463 return unit; /* inside an identity run */
464 }
465 idx += run;
466 }
467 return unit;
468}

477ra8_err_t priv_exfat_write_upcase(const ra8_fs_backend_t* backend,
478 uint64_t abs_lba,
479 uint32_t bps,
480 uint32_t* out_csum)
481{
482 const uint32_t upc_secs = ((uint32_t)k_exfat_fmt_upc_std_bytes + bps - 1U) / bps;
483 uint32_t cs = 0U;
484 uint32_t off = 0U;
485 for (uint32_t s = 0U; s < upc_secs; s++) {
486 for (uint32_t i = 0U; i < bps; i++) {
487 s_scratch[i] = 0U;
488 }
489 uint32_t n = (uint32_t)k_exfat_fmt_upc_std_bytes - off;
490 if (n > bps) {
491 n = bps;
492 }
493 priv_byte_copy(s_scratch, &s_exfat_upcase[off], n);
494 cs = priv_exfat_csum32(cs, s_scratch, n);
495 const ra8_err_t err = backend->write_block(backend->ctx, abs_lba + s, 1U, s_scratch);
496 if (err != k_ra8_ok) {
497 return err;
498 }
499 off += n;
500 }
501 *out_csum = cs;
502 return k_ra8_ok;
