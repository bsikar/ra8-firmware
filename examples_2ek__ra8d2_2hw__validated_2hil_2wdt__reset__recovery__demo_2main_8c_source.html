121static void wdt_rr_arm_and_wait_for_reset(void)
122{
123 /* on_expiry = reset (RSTIRQS=1) so the WWDT underflow drops us
124 * straight into a hardware reset instead of NMI. */
125 const ra8_wdt_cfg_t cfg = {
126 .timeout = k_ra8_wdt_timeout_1024,
127 .clock_div = k_ra8_wdt_clkdiv_4,
128 .window_start = k_ra8_wdt_window_start_100,
129 .window_end = k_ra8_wdt_window_end_0,
130 .on_expiry = k_ra8_wdt_on_expiry_reset,
131 .stop_in_sleep = k_ra8_wdt_sleep_stop_count,
132 };
133 if (ra8_wdt_init(&cfg) != k_ra8_ok) {
134 wdt_rr_panic_halt();
135 }
136 (void)ra8_board_uart_console_write(k_wdt_rr_msg_arming,
137 (size_t)(sizeof(k_wdt_rr_msg_arming) - 1U));
138
139 for (uint32_t i = 0U; i < (uint32_t)k_wdt_rr_arm_iters; ++i) {
140 (void)ra8_wdt_refresh_for(k_ra8_wdt0);
141 (void)ra8_board_led_toggle(k_ra8_board_led1);
142 ra8_delay_ms((uint32_t)k_wdt_rr_refresh_ms);
143 }
144 /* Stop refreshing -- the WWDT counter will underflow and the chip
145 * will reset. The HIL probe scrapes the reset banner on the next
146 * boot. */
147 while (1) {
148 __asm__ volatile("wfi");
149 }
150}

152#pragma GCC diagnostic push
153#pragma GCC diagnostic ignored "-Wmain"
154int32_t main(void)
155{
156 wdt_rr_setup_or_halt();
157 ra8_isr_globals_enable();
158
159 ra8_reset_cause_t cause = k_ra8_reset_cause_unknown;
160 (void)ra8_reset_get_cause(&cause);
161 /* Bench-truth: RSTSRn flags are sticky across resets (HUM Ch 6.2 -
162 * the registers note "Only 0 can be written. To clear ..."). With
163 * IWDTRF + WDTRF + SWRF all latched from prior bench cycles the
164 * decoder's IWDTRF-first priority order would mask out the real
165 * cause. Clear every flag so the next reset surfaces cleanly. */
166 (void)ra8_reset_clear_cause(k_reset_cause_all_mask);
167 uint32_t msg_len = 0U;
168 const uint8_t* msg = wdt_rr_banner_for(cause, &msg_len);
169 (void)ra8_board_uart_console_write(msg, (size_t)msg_len);
170
171 if (cause == k_ra8_reset_cause_wdt0) {
172 /* Stage B: just recovered from a WWDT reset. Sit here so the
173 * HIL probe scrapes the recovery banner. The chip will not
174 * reset again until power-cycled or the demo is re-flashed. */
175 while (1) {
176 __asm__ volatile("wfi");
177 }
178 }
179
180 /* Stage A: power-on (or any non-WWDT cause). Arm WWDT, refresh
181 * briefly, then stop refreshing to trigger the reset. */
182 wdt_rr_arm_and_wait_for_reset();
183 return 0;
184}
