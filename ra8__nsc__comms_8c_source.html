395static bool internal_spi_byte_span(uint8_t bytes_per_unit, uint32_t len, uint32_t* out_span)
396{
397 const uint64_t span = (uint64_t)bytes_per_unit * (uint64_t)len;
398 if (span > (uint64_t)UINT32_MAX) {
399 return false;
400 }
401 *out_span = (uint32_t)span;
402 return true;
403}

434RA8_NSC_VENEER ra8_err_t ra8_nsc_spi_write(uint8_t channel,
435 const void* tx,
436 uint32_t len,
437 ra8_spi_bit_width_t bit_width)
438{
439 if (len > 0U) {
440 RA8_CHECK_NULL_PTR(tx, s_tag, "spi_write: tx");
441 const uint8_t bytes_per_unit = internal_spi_unit_bytes(bit_width);
442 if (bytes_per_unit == 0U) {
443 return k_ra8_err_invalid_arg;
444 }
445 uint32_t span = 0U;
446 if (!internal_spi_byte_span(bytes_per_unit, len, &span)) {
447 return k_ra8_err_invalid_arg;
448 }
449 RA8_NSC_CHECK_NS_RANGE_R(tx, span);
450 }
451 return ra8_spi_write(channel, tx, len, bit_width);
452}

502RA8_NSC_VENEER ra8_err_t ra8_nsc_spi_write_read(uint32_t ch_bw,
503 const void* tx,
504 void* rx,
505 uint32_t len)
506{
507 const uint8_t channel = (uint8_t)(ch_bw & (uint32_t)k_ra8_nsc_spi_byte_msk);
508 const ra8_spi_bit_width_t bit_width =
509 (ra8_spi_bit_width_t)((ch_bw >> (uint32_t)k_ra8_nsc_spi_bw_shift) &
510 (uint32_t)k_ra8_nsc_spi_byte_msk);
511
512 if (len > 0U) {
513 RA8_CHECK_NULL_PTR(tx, s_tag, "spi_write_read: tx");
514 RA8_CHECK_NULL_PTR(rx, s_tag, "spi_write_read: rx");
515 const uint8_t bytes_per_unit = internal_spi_unit_bytes(bit_width);
516 if (bytes_per_unit == 0U) {
517 return k_ra8_err_invalid_arg;
518 }
519 uint32_t span = 0U;
520 if (!internal_spi_byte_span(bytes_per_unit, len, &span)) {
521 return k_ra8_err_invalid_arg;
522 }
523 RA8_NSC_CHECK_NS_RANGE_R(tx, span);
524 RA8_NSC_CHECK_NS_RANGE_RW(rx, span);
525 }
526 return ra8_spi_write_read(channel, tx, rx, len, bit_width);
527}
