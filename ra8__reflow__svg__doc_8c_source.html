68RA8_INTERNAL
69static bool priv_elem_at(const uint8_t* s, size_t len, size_t at, const char* name)
70{
71 if (!ra8_svgp_starts_ci(s, len, at, name)) {
72 return false;
73 }
74 const size_t after = at + strlen(name);
75 if (after >= len) {
76 return true;
77 }
78 const char c = (char)s[after];
79 return ra8_svgp_ws(c) || (c == '>') || (c == '/');
80}

113static bool priv_tag_span(const uint8_t* s,
114 size_t len,
115 size_t from,
116 const char* name,
117 size_t* open,
118 size_t* close)
119{
120 const size_t at = ra8_svgp_find_ci(s, len, from, name);
121 if (at >= len) {
122 return false;
123 }
124 size_t j = at;
125 while ((j < len) && (s[j] != '>')) {
126 ++j;
127 }
128 *open = at;
129 *close = j;
130 return true;
131}

158static void priv_read_viewbox(const uint8_t* s, size_t len, svg_xform_t* t)
159{
160 t->vx = 0;
161 t->vy = 0;
162 t->vw = t->bw;
163 t->vh = t->bh;
164 size_t open = 0U;
165 size_t close = 0U;
166 if (!priv_tag_span(s, len, 0U, "<svg", &open, &close)) {
167 return;
168 }
169 const uint8_t* tag = &s[open];
170 const size_t tlen = close - open;
171 size_t voff = 0U;
172 size_t vlen = 0U;
173 if (ra8_svgp_attr(tag, tlen, "viewBox", &voff, &vlen)) {
174 int32_t nums[k_svg_viewbox_n] = {};
175 size_t k = 0U;
176 for (size_t n = 0U; n < (size_t)k_svg_viewbox_n; ++n) {
177 nums[n] = ra8_svgp_num(&tag[voff], vlen, &k);
178 }
179 if ((nums[2] > 0) && (nums[3] > 0)) {
180 t->vx = nums[0];
181 t->vy = nums[1];
182 t->vw = nums[2];
183 t->vh = nums[3];
184 }
185 } else {
186 const int32_t wv = ra8_svgp_attr_num(tag, tlen, "width", t->bw);
187 const int32_t hv = ra8_svgp_attr_num(tag, tlen, "height", t->bh);
188 if ((wv > 0) && (hv > 0)) {
189 t->vw = wv;
190 t->vh = hv;
191 }
192 }
193}

221static void
222priv_dispatch_shape(const uint8_t* s, size_t len, size_t at, size_t close, const svg_xform_t* t)
223{
224 const uint8_t* tag = &s[at];
225 const size_t tlen = (close > at) ? (close - at) : 0U;
226 /* Compose this element's own `transform=` onto the inherited group transform. */
227 svg_xform_t lt = *t;
228 ra8_svgp_apply_xform(&lt, tag, tlen);
229 if (priv_elem_at(s, len, at, "<rect")) {
230 ra8_svgp_draw_rect(tag, tlen, &lt);
231 } else if (priv_elem_at(s, len, at, "<circle")) {
232 ra8_svgp_draw_circle(tag, tlen, &lt);
233 } else if (priv_elem_at(s, len, at, "<polygon")) {
234 ra8_svgp_draw_polygon(tag, tlen, &lt);
235 } else if (priv_elem_at(s, len, at, "<polyline")) {
236 ra8_svgp_draw_polyline(tag, tlen, &lt);
237 } else if (priv_elem_at(s, len, at, "<path")) {
238 ra8_svgp_draw_path(tag, tlen, &lt);
239 } else if (priv_elem_at(s, len, at, "<line")) {
240 ra8_svgp_draw_line(tag, tlen, &lt);
241 } else {
242 /* Unsupported element -> skip. */
243 }
244}

313RA8_INTERNAL
314static int32_t priv_group_open(const uint8_t* s,
315 size_t i,
316 size_t close,
317 const svg_xform_t* t,
318 svg_utf_t* gstk,
319 int32_t gsp)
320{
321 svg_xform_t gt = {};
322 priv_xform_with_group(t, gstk[gsp], &gt);
323 ra8_svgp_apply_xform(&gt, &s[i], (close > i) ? (close - i) : 0U);
324 /*
325 * priv_group_open is dispatched only for a well-formed element tag whose name
326 * occupies at least one byte between '<' (scan position `i`) and '>'
327 * (position `close`), so close > i is structurally invariant on every
328 * reachable call. The close <= i arm defends against an empty "<>" tag that
329 * the upstream element scanner never yields, so the first condition cannot be
330 * flipped independently.
331 */
332 /* mcdc-deactivated: close>i is invariant for a named tag; the empty-"<>" (close<=i) arm is unreachable via public API. */
333 const bool self_close = (close > i) && (s[close - 1U] == '/');
334 if (self_close || (gsp >= (int32_t)k_svg_g_depth_max)) {
335 return gsp;
336 }
337 gstk[gsp + 1] =
338 (svg_utf_t){.a = gt.ua, .b = gt.ub, .c = gt.uc, .d = gt.ud, .e = gt.ue, .f = gt.uf};
339 return gsp + 1;
340}

367RA8_INTERNAL
368static void priv_draw_shapes(const uint8_t* s, size_t len, const svg_xform_t* t)
369{
370 svg_utf_t gstk[k_svg_g_depth_max + 1U];
371 int32_t gsp = 0;
372 gstk[0] = (svg_utf_t){.a = t->ua, .b = t->ub, .c = t->uc, .d = t->ud, .e = t->ue, .f = t->uf};
373 size_t i = 0U;
374 /* Bounded: each iteration advances past one '<...>' element or breaks at EOF. */
375 while (i < len) {
376 while ((i < len) && (s[i] != '<')) {
377 ++i;
378 }
379 if (i >= len) {
380 break;
381 }
382 size_t close = i;
383 while ((close < len) && (s[close] != '>')) {
384 ++close;
385 }
386 if (priv_elem_at(s, len, i, "</g")) {
387 gsp = (gsp > 0) ? (gsp - 1) : 0;
388 } else if (priv_elem_at(s, len, i, "<g")) {
389 gsp = priv_group_open(s, i, close, t, gstk, gsp);
390 } else {
391 svg_xform_t ct = {};
392 priv_xform_with_group(t, gstk[gsp], &ct);
393 priv_dispatch_shape(s, len, i, close, &ct);
394 }
395 i = (close < len) ? (close + 1U) : len;
396 }
397}

628static void priv_parse_stops(const uint8_t* s, size_t from, size_t end, svg_grad_t* g)
629{
630 size_t at = from;
631 /* Bounded: <= k_svg_grad_stops stops; `at` advances past each `<stop>`. */
632 while (g->nstops < (uint8_t)k_svg_grad_stops) {
633 const size_t sp = ra8_svgp_find_ci(s, end, at, "<stop");
634 if (sp >= end) {
635 break;
636 }
637 size_t close = sp;
638 while ((close < end) && (s[close] != '>')) {
639 ++close;
640 }
641 const uint8_t* tag = &s[sp];
642 const size_t tlen = (close > sp) ? (close - sp) : 0U;
643 g->stops[g->nstops].off = priv_clamp01(priv_stop_offset(tag, tlen));
644 g->stops[g->nstops].col = priv_stop_color(tag, tlen);
645 g->nstops += 1U;
646 at = (close < end) ? (close + 1U) : end;
647 }
648}

675RA8_INTERNAL
676static void priv_scan_grads(const uint8_t* s, size_t len, svg_grads_t* gs)
677{
678 gs->n = 0;
679 size_t from = 0U;
680 /* Bounded: <= k_svg_grad_max gradients; `from` strictly advances each pass. */
681 while (gs->n < (int32_t)k_svg_grad_max) {
682 const size_t lin = ra8_svgp_find_ci(s, len, from, "<linearGradient");
683 const size_t rad = ra8_svgp_find_ci(s, len, from, "<radialGradient");
684 const bool is_rad = (rad < lin);
685 const size_t at = is_rad ? rad : lin;
686 if (at >= len) {
687 break;
688 }
689 size_t close = at;
690 while ((close < len) && (s[close] != '>')) {
691 ++close;
692 }
693 const uint8_t* tag = &s[at];
694 const size_t tlen = (close > at) ? (close - at) : 0U;
695 svg_grad_t* g = &gs->g[gs->n];
696 *g = (svg_grad_t){};
697 g->kind = is_rad ? (uint8_t)k_svg_grad_radial : (uint8_t)k_svg_grad_linear;
698 priv_copy_attr_id(tag, tlen, g->id);
699 if (is_rad) {
700 g->x1 = priv_attr_numf(tag, tlen, "cx", s_svg_half);
701 g->y1 = priv_attr_numf(tag, tlen, "cy", s_svg_half);
702 g->x2 = priv_attr_numf(tag, tlen, "r", s_svg_half);
703 } else {
704 g->x1 = priv_attr_numf(tag, tlen, "x1", 0.0F);
705 g->y1 = priv_attr_numf(tag, tlen, "y1", 0.0F);
706 g->x2 = priv_attr_numf(tag, tlen, "x2", s_svg_unit);
707 g->y2 = priv_attr_numf(tag, tlen, "y2", 0.0F);
708 }
709 const char* endtag = is_rad ? "</radialGradient" : "</linearGradient";
710 const size_t endpos = ra8_svgp_find_ci(s, len, close, endtag);
711 const size_t stopend = (endpos < len) ? endpos : len;
712 priv_parse_stops(s, close, stopend, g);
713 if (g->nstops > 0U) {
714 gs->n += 1;
715 }
716 from = (endpos < len) ? (endpos + 1U) : len;
717 }
718}

725bool ra8_svg_is_svg(const uint8_t* bytes, size_t len)
726{
727 if ((bytes == nullptr) || (len == 0U)) {
728 return false;
729 }
730 size_t i = 0U;
731 /* Skip a UTF-8 BOM if present. */
732 if ((len >= (size_t)k_svg_bom_len) && (bytes[0] == (uint8_t)k_svg_bom0) &&
733 (bytes[1] == (uint8_t)k_svg_bom1) && (bytes[2] == (uint8_t)k_svg_bom2)) {
734 i = (size_t)k_svg_bom_len;
735 }
736 while ((i < len) && ra8_svgp_ws((char)bytes[i])) {
737 ++i;
738 }
739 return ra8_svgp_starts_ci(bytes, len, i, "<?xml") || ra8_svgp_starts_ci(bytes, len, i, "<svg");
740}

765ra8_err_t ra8_svg_size(const uint8_t* svg, size_t len, int32_t* out_w, int32_t* out_h)
766{
767 if ((svg == nullptr) || (out_w == nullptr) || (out_h == nullptr)) {
768 return k_ra8_err_null_ptr;
769 }
770 size_t open = 0U;
771 size_t close = 0U;
772 if (!priv_tag_span(svg, len, 0U, "<svg", &open, &close)) {
773 return k_ra8_err_not_found;
774 }
775 const uint8_t* tag = &svg[open];
776 const size_t tlen = close - open;
777 int32_t w = ra8_svgp_attr_num(tag, tlen, "width", 0);
778 int32_t h = ra8_svgp_attr_num(tag, tlen, "height", 0);
779 if ((w <= 0) || (h <= 0)) {
780 size_t voff = 0U;
781 size_t vlen = 0U;
782 if (ra8_svgp_attr(tag, tlen, "viewBox", &voff, &vlen)) {
783 int32_t nums[k_svg_viewbox_n] = {};
784 size_t k = 0U;
785 for (size_t n = 0U; n < (size_t)k_svg_viewbox_n; ++n) {
786 nums[n] = ra8_svgp_num(&tag[voff], vlen, &k);
787 }
788 w = nums[2];
789 h = nums[3];
790 }
791 }
792 if ((w <= 0) || (h <= 0)) {
793 return k_ra8_err_not_found;
794 }
795 *out_w = w;
796 *out_h = h;
797 return k_ra8_ok;
798}
