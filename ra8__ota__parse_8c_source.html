263RA8_INTERNAL
264static ra8_err_t priv_json_str(const char* json, const char* key, char* dst, uint32_t cap)
265{
266 const char* p = strstr(json, key);
267 if (p == nullptr) {
268 return k_ra8_err_invalid_arg;
269 }
270 p = strchr(p + strlen(key), '"');
271 if (p == nullptr) {
272 return k_ra8_err_invalid_arg;
273 }
274 ++p;
275 const char* q = strchr(p, '"');
276 if (q == nullptr) {
277 return k_ra8_err_invalid_arg;
278 }
279 const uint32_t n = (uint32_t)(q - p);
280 if (n + 1U > cap) {
281 return k_ra8_err_invalid_size;
282 }
283 (void)memcpy(dst, p, n);
284 dst[n] = '\0';
285 return k_ra8_ok;
286}

313ra8_err_t ra8_ota_internal_json_u32(const char* json, const char* key, uint32_t* out_v)
314{
315 const char* p = strstr(json, key);
316 if (p == nullptr) {
317 return k_ra8_err_invalid_arg;
318 }
319 p += strlen(key);
320 /* Skip past quote/colon/whitespace. */
321 for (uint32_t guard = 0U; guard < (uint32_t)k_ra8_ota_json_skip_max; ++guard) {
322 if (*p == ':' || *p == ' ' || *p == '"') {
323 ++p;
324 } else {
325 break;
326 }
327 }
328 uint32_t v = 0U;
329 uint32_t i = 0U;
330 for (; i < (uint32_t)k_ra8_ota_u32_decimal_digits; ++i) {
331 const char c = p[i];
332 if ((c < '0') || (c > '9')) {
333 break;
334 }
335 v = (v * (uint32_t)k_ra8_ota_u32_decimal_base) + (uint32_t)(c - '0');
336 }
337 if (i == 0U) {
338 return k_ra8_err_invalid_arg;
339 }
340 *out_v = v;
341 return k_ra8_ok;
342}

365RA8_INTERNAL
366static uint8_t priv_hex_nibble(char c)
367{
368 if ((c >= '0') && (c <= '9')) {
369 return (uint8_t)(c - '0');
370 }
371 if ((c >= 'a') && (c <= 'f')) {
372 return (uint8_t)((uint8_t)k_ra8_ota_hex_alpha_offset + (c - 'a'));
373 }
374 if (ra8_ota_internal_char_in_range(c, 'A', 'F')) {
375 return (uint8_t)((uint8_t)k_ra8_ota_hex_alpha_offset + (c - 'A'));
376 }
377 return (uint8_t)k_ra8_ota_hex_invalid_nibble;
378}

403RA8_INTERNAL
404static uint32_t priv_hex_decode(const char* in, uint8_t* out, uint32_t out_cap)
405{
406 const uint32_t in_len = (uint32_t)strlen(in);
407 if ((in_len % (uint32_t)k_ra8_ota_hex_chars_per_byte) != 0U) {
408 return 0U;
409 }
410 const uint32_t bytes = in_len / (uint32_t)k_ra8_ota_hex_chars_per_byte;
411 if (bytes > out_cap) {
412 return 0U;
413 }
414 for (uint32_t i = 0U; i < bytes; ++i) {
415 const size_t base_idx = (size_t)i * (size_t)k_ra8_ota_hex_chars_per_byte;
416 const uint8_t hi = priv_hex_nibble(in[base_idx]);
417 const uint8_t lo = priv_hex_nibble(in[base_idx + 1U]);
418 if ((hi == (uint8_t)k_ra8_ota_hex_invalid_nibble) ||
419 (lo == (uint8_t)k_ra8_ota_hex_invalid_nibble)) {
420 return 0U;
421 }
422 out[i] = (uint8_t)((hi << (uint8_t)k_ra8_ota_hex_nibble_shift) | lo);
423 }
424 return bytes;
425}

453RA8_INTERNAL
454static ra8_err_t priv_manifest_decode_crypto(const char* json, ra8_ota_manifest_t* out)
455{
456 char hex[k_ra8_ota_hex_buf_bytes];
457 ra8_err_t e = priv_json_str(json, "\"sha256\"", hex, sizeof hex);
458 if (e != k_ra8_ok) {
459 return e;
460 }
461 const uint32_t n_d = priv_hex_decode(hex, out->image_sha256, k_ra8_ota_sha256_bytes);
462 if (n_d != k_ra8_ota_sha256_bytes) {
463 return k_ra8_err_invalid_arg;
464 }
465
466 e = priv_json_str(json, "\"signature\"", hex, sizeof hex);
467 if (e != k_ra8_ok) {
468 return e;
469 }
470 const uint32_t n_s = priv_hex_decode(hex, out->signature, k_ra8_ota_signature_max_bytes);
471 if (n_s == 0U) {
472 return k_ra8_err_invalid_arg;
473 }
474 out->signature_len = (uint16_t)n_s;
475 return k_ra8_ok;
476}
