4r"""check_no_wave_references.py -- ban session-bookkeeping "Wave N" references.
5
6Rationale: comments and commit messages that cite "Wave 70 fixed FRDY" or
7"see Wave 43b" leak internal session bookkeeping into the source tree.
8Future readers do not care WHEN a fix was found, only WHY. Reference the
9function or symbol or HUM section instead.
10
11What this gate flags (case-insensitive):
12 - "Wave 70", "wave 3", "WAVE-12", "wave_43b" -- a "wave" token immediately
13 followed (after at most one [\\s_-]) by a digit. Optional trailing
14 letter(s) for sub-numbering ("wave 43b").
15 - "(Wave 12)" / "[wave-7]" -- same pattern wrapped in punctuation.
16
17What this gate does NOT flag (legitimate domain usage):
18 - "waveform", "wave shape", "saw wave", "sine wave" (no digit follows).
19 - Identifiers like "k_ra8_pdg_wave_saw", "wave_select", "wave_table".
20 - Renesas / MIPI / vendor symbols that happen to contain "wave".
21
22Per-line opt-out: append "WAVE-OK: <reason>" on the offending line. Reserve
23for unavoidable upstream-symbol citations (e.g. a Renesas register name
24that literally encodes the wording).
25
26Scope is DERIVED from ``git ls-files`` via ``lint_targets.first_party_paths``
27rather than a hardcoded root list, so a new top-level directory (``tools/``,
28``coprocessor/``, ``infra/`` and ``mk/`` were the ones the old list silently
29omitted, #549) is covered the day it lands. ``--selftest`` proves the detector
30fires and stays quiet, and that the derived scope clears its floor.
31
32Exit code:
33 0 -- gate clean
34 1 -- violations exist
35 2 -- the scope collapsed below FILE_FLOOR (a scan of almost nothing)
36
37@copyright Copyright (c) 2026 Brighton Sikarskie
38SPDX-License-Identifier: MIT
39"""

70SCAN_BASENAMES: frozenset[str] = frozenset(
71 {"Makefile", "Dockerfile", "CMakeLists.txt", "GNUmakefile"}
72)
73
74# Vendored / generated docs content that is not ours to police: committed
75# datasheets and register maps under docs/reference/, and any generated Doxygen
76# output (docs/**/doxygen, docs/**/html) that is tracked. first_party_paths
77# already drops third_party/ and build output; these three are the docs-side
78# equivalents the old SKIP_DIR_NAMES carried. Matched by path component, the
79# same way the previous walk skipped them.
80DOCS_VENDOR_DIRS: frozenset[str] = frozenset({"reference", "doxygen", "html"})
81
82# A tree this size cannot legitimately collapse to a handful of files. A scan
83# that enumerates almost nothing reports a clean tree because it read almost
84# nothing -- the exact failure the gate-honesty epic (#190) exists to prevent.
85# Measured 2026-08-02: 3416 first-party files in the derived scope. Same
86# trip-wire as check_ruff.py.
87FILE_FLOOR = 2500
88
89# Maximum number of violations to print before truncating output.
90MAX_FINDINGS_SHOWN = 50
91# Maximum line length (chars) for a snippet printed in the report.
92SNIPPET_MAX_LEN = 120
93# Truncated snippet suffix consumes 3 chars ("..."), so trim to this length.
94SNIPPET_TRIM_LEN = SNIPPET_MAX_LEN - 3
95
96# "Wave 70", "wave-3", "WAVE_43b", "(Wave 12)" -- token + optional sep +
97# digit(s) + optional letter(s).
98WAVE_RE: re.Pattern[str] = re.compile(r"\b[Ww]ave[\s_\-]?\d+[A-Za-z]?\b")
99
100OPTOUT_RE: re.Pattern[str] = re.compile(r"WAVE-OK\s*:")
101
102SELF_EXEMPT_FILES: frozenset[str] = frozenset(
103 {
104 "scripts/checks/check_no_wave_references.py",
105 "scripts/fix/fix_wave_references.py",
106 "docs/STYLE_GUIDE.md",
107 "CLAUDE.md",
108 }
109)

112def iter_source_files(root: Path) -> list[Path]:
113 """Every in-scope first-party file, derived from git rather than a root list.
114
115 Enumeration goes through ``lint_targets.first_party_paths`` -- the shared
116 derived-scope primitive -- so ``tools/``, ``coprocessor/``, ``infra/`` and
117 ``mk/`` (the roots a hardcoded list silently dropped, #549) are covered. The
118 only subtractions on top of what that primitive already exempts are the
119 docs-side vendored/generated directories in ``DOCS_VENDOR_DIRS``.
120 """
121 rels = set(first_party_paths(tuple(SCAN_EXTS)))
122 for name in SCAN_BASENAMES:
123 rels |= {rel for rel in first_party_paths((name,)) if Path(rel).name == name}
124 out: list[Path] = []
125 for rel in sorted(rels):
126 if set(Path(rel).parts) & DOCS_VENDOR_DIRS:
127 continue
128 out.append(root / rel)
129 return out

132def scan_file(path: Path, root: Path) -> list[tuple[Path, int, str]]:
133 """Report every "Wave N" session reference in one file.
134
135 Self-exempt files are skipped whole: this checker and the policy doc must
136 spell the banned pattern to describe it, and tagging every such line
137 individually would bury them.
138 """
139 rel = path.relative_to(root)
140 if str(rel) in SELF_EXEMPT_FILES:
141 return []
142 try:
143 text = path.read_text(encoding="utf-8")
144 except (OSError, UnicodeDecodeError):
145 return []
146 out: list[tuple[Path, int, str]] = []
147 for lineno, line in enumerate(text.splitlines(), start=1):
148 if OPTOUT_RE.search(line):
149 continue
150 if WAVE_RE.search(line):
151 out.append((rel, lineno, line.rstrip()))
152 return out

155def selftest() -> int:
156 """Prove the detector fires on a "Wave N" tag, stays quiet otherwise, and scans real files.
157
158 Both directions plus a scope probe: the numbered-session pattern must FIRE,
159 the legitimate domain uses (``waveform``, ``sine wave``, ``wave_table``)
160 must stay QUIET, the derived scope must clear ``FILE_FLOOR``, and it must
161 reach the roots a hardcoded list had dropped (``infra/``, ``mk/``) -- a
162 clean run over a scope that never sees those roots proves nothing.
163
164 Returns:
165 0 when every assertion held in both directions, 1 otherwise.
166 """
167 failures: list[str] = []
168 for text, must_fire, label in (
169 ("fixed in Wave 70", True, 'MUST FIRE: "Wave 70"'),
170 ("see wave-43b for context", True, 'MUST FIRE: "wave-43b"'),
171 ("the sine wave is smooth", False, 'MUST NOT FIRE: "sine wave"'),
172 ("k_ra8_pdg_wave_saw selects the waveform", False, "MUST NOT FIRE: wave_saw / waveform"),
173 ("wave_table[0] holds the sample", False, "MUST NOT FIRE: wave_table identifier"),
174 ):
175 fired = bool(WAVE_RE.search(text))
176 expect(fired == must_fire, label, failures)
177
178 root = Path(__file__).resolve().parents[2]
179 files = iter_source_files(root)
180 rels = {str(p.relative_to(root)) for p in files if p.is_relative_to(root)}
181 expect(
182 len(files) >= FILE_FLOOR,
183 f"derived scope sees {len(files)} file(s) (floor {FILE_FLOOR})",
184 failures,
185 )
186 for root_name in ("infra", "mk"):
187 expect(
188 any(rel.startswith(root_name + "/") for rel in rels),
189 f"the derived scope reaches {root_name}/ (previously omitted)",
190 failures,
191 )
192 return report(failures)

195def main() -> int:
196 """Ban session-bookkeeping "Wave N" references from the tree.
197
198 These leak an internal working chronology into source that outlives it: a
199 future reader cannot resolve "see Wave 43b" to anything, whereas the
200 function, symbol or HUM section the fix touched stays findable.
201
202 Returns 1 listing each reference, 0 when the tree is clean, 2 when the
203 derived scope collapsed below FILE_FLOOR.
204 """
205 if "--selftest" in sys.argv[1:]:
206 return selftest()
207 root = Path(__file__).resolve().parents[2]
208 files = iter_source_files(root)
209 if len(files) < FILE_FLOOR:
210 sys.stderr.write(
211 f"check_no_wave_references.py: FATAL -- only {len(files)} file(s) in scope, "
212 f"floor is {FILE_FLOOR}. A collapsed scope reports a clean tree because it "
213 "scanned nothing.\n"
214 )
215 return 2
216 findings: list[tuple[Path, int, str]] = []
217 for f in files:
218 findings.extend(scan_file(f, root))
219
220 if not findings:
221 print("no-wave-refs: 0 violations -- gate clean.")
222 return 0
223
224 print(f"no-wave-refs: {len(findings)} violations found.")
225 for rel, lineno, line in findings[:MAX_FINDINGS_SHOWN]:
226 snippet = line if len(line) <= SNIPPET_MAX_LEN else line[:SNIPPET_TRIM_LEN] + "..."
227 print(f" {rel}:{lineno} {snippet}")
228 if len(findings) > MAX_FINDINGS_SHOWN:
229 print(f" ... {len(findings) - MAX_FINDINGS_SHOWN} more (truncated)")
230 print()
231 print('Per-line opt-out: append "WAVE-OK: <reason>" on the offending line.')
232 print("Auto-fix helper: scripts/fix/fix_wave_references.py --apply")
233 return 1
