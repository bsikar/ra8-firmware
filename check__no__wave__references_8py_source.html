1#!/usr/bin/env python3
2r"""check_no_wave_references.py -- ban session-bookkeeping "Wave N" references.
3
4Rationale: comments and commit messages that cite "Wave 70 fixed FRDY" or
5"see Wave 43b" leak internal session bookkeeping into the source tree.
6Future readers do not care WHEN a fix was found, only WHY. Reference the
7function or symbol or HUM section instead.
8
9What this gate flags (case-insensitive):
10 - "Wave 70", "wave 3", "WAVE-12", "wave_43b" -- a "wave" token immediately
11 followed (after at most one [\\s_-]) by a digit. Optional trailing
12 letter(s) for sub-numbering ("wave 43b").
13 - "(Wave 12)" / "[wave-7]" -- same pattern wrapped in punctuation.
14
15What this gate does NOT flag (legitimate domain usage):
16 - "waveform", "wave shape", "saw wave", "sine wave" (no digit follows).
17 - Identifiers like "k_ra8_pdg_wave_saw", "wave_select", "wave_table".
18 - Renesas / MIPI / vendor symbols that happen to contain "wave".
19
20Per-line opt-out: append "WAVE-OK: <reason>" on the offending line. Reserve
21for unavoidable upstream-symbol citations (e.g. a Renesas register name
22that literally encodes the wording).
23
24Exit code:
25 0 -- gate clean
26 1 -- violations exist
27

96def _is_skip_dir(name: str) -> bool:
97 """Whether a directory is build output and therefore skipped.
98
99 Matches the ``build-*`` family by prefix so CMake variant directories are
100 excluded without being enumerated.
101 """
102 if name in SKIP_DIR_NAMES:
103 return True
104 return bool(name.startswith("build-") or name == "build")

107def should_scan(path: Path) -> bool:
108 """Whether a file's basename or suffix puts it in scope."""
109 if path.name in SCAN_BASENAMES:
110 return True
111 return path.suffix in SCAN_EXTS

114def iter_source_files(root: Path) -> list[Path]:
115 """Every in-scope file beneath the configured scan roots."""
116 out: list[Path] = []
117 for scan_root in SCAN_ROOTS:
118 base = root / scan_root
119 if not base.exists():
120 continue
121 for dirpath, dirnames, filenames in os.walk(base):
122 dirnames[:] = [d for d in dirnames if not _is_skip_dir(d)]
123 for fn in filenames:
124 p = Path(dirpath) / fn
125 if should_scan(p):
126 out.append(p)
127 for top in ("Makefile", "CMakeLists.txt", "README.md"):
128 p = root / top
129 if p.exists():
130 out.append(p)
131 return out

134def scan_file(path: Path, root: Path) -> list[tuple[Path, int, str]]:
135 """Report every "Wave N" session reference in one file.
136
137 Self-exempt files are skipped whole: this checker and the policy doc must
138 spell the banned pattern to describe it, and tagging every such line
139 individually would bury them.
140 """
141 rel = path.relative_to(root)
142 if str(rel) in SELF_EXEMPT_FILES:
143 return []
144 try:
145 text = path.read_text(encoding="utf-8")
146 except (OSError, UnicodeDecodeError):
147 return []
148 out: list[tuple[Path, int, str]] = []
149 for lineno, line in enumerate(text.splitlines(), start=1):
150 if OPTOUT_RE.search(line):
151 continue
152 if WAVE_RE.search(line):
153 out.append((rel, lineno, line.rstrip()))
154 return out

157def main() -> int:
158 """Ban session-bookkeeping "Wave N" references from the tree.
159
160 These leak an internal working chronology into source that outlives it: a
161 future reader cannot resolve "see Wave 43b" to anything, whereas the
162 function, symbol or HUM section the fix touched stays findable.
163
164 Returns 1 listing each reference, 0 when the tree is clean.
165 """
166 root = Path(__file__).resolve().parents[2]
167 files = iter_source_files(root)
168 findings: list[tuple[Path, int, str]] = []
169 for f in files:
170 findings.extend(scan_file(f, root))
171
172 if not findings:
173 print("no-wave-refs: 0 violations -- gate clean.")
174 return 0
175
176 print(f"no-wave-refs: {len(findings)} violations found.")
177 for rel, lineno, line in findings[:MAX_FINDINGS_SHOWN]:
178 snippet = line if len(line) <= SNIPPET_MAX_LEN else line[:SNIPPET_TRIM_LEN] + "..."
179 print(f" {rel}:{lineno} {snippet}")
180 if len(findings) > MAX_FINDINGS_SHOWN:
181 print(f" ... {len(findings) - MAX_FINDINGS_SHOWN} more (truncated)")
182 print()
183 print('Per-line opt-out: append "WAVE-OK: <reason>" on the offending line.')
184 print("Auto-fix helper: scripts/fix/fix_wave_references.py --apply")
185 return 1
