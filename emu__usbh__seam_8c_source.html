32/* ============================================================================
33 * Virtual USB host-mode device: a HID boot keyboard behind the ra8_usb_host_*
34 * seam.
35 *
36 * board_usb.c models the USBFS controller in DEVICE mode (a virtual host drives
37 * the firmware's device stack). The inverse case -- the firmware acting as USB
38 * HOST -- drives the USBHS controller (0x40351000, unmodelled) through the
39 * first-party `ra8_usb_host_*` primitives, and with no peer the control transfer
40 * wedges (SUREQ never acked -> k_ra8_err_busy). Rather than model a second
41 * controller register-by-register, seam those primitives the same way ra8_eth_*
42 * is seamed to a virtual net peer (the register model "cannot satisfy" that
43 * sequence either): present a virtual boot keyboard that answers chapter-9
44 * enumeration and streams interrupt-IN reports. This lets a host example
45 * (usb_host_keyboard) enumerate + read reports end to end with no hardware --
46 * validating the host stack's control/data logic, not silicon timing.
47 * ==========================================================================*/

154static void on_usbh_control_xfer(uc_engine* uc, uint64_t address, uint32_t size, void* user)
155{
156 (void)address;
157 (void)size;
158 (void)user;
159 uint32_t setup_ptr = 0U;
160 uint32_t data_ptr = 0U;
161 uint32_t data_len = 0U;
162 (void)uc_reg_read(uc, UC_ARM_REG_R1, &setup_ptr);
163 (void)uc_reg_read(uc, UC_ARM_REG_R2, &data_ptr);
164 (void)uc_reg_read(uc, UC_ARM_REG_R3, &data_len);
165 const uint32_t out_ptr = usbh_arg5(uc);
166
167 uint8_t s[8] = {};
168 (void)uc_mem_read(uc, (uint64_t)setup_ptr, s, sizeof(s));
169 const uint8_t b_request = s[1];
170 const uint8_t desc_type = s[3]; /* wValue high byte = descriptor type. */
171 const uint8_t* src = nullptr;
172 uint16_t src_len = 0U;
173 if (b_request == (uint8_t)k_vkbd_breq_get_descriptor) {
174 if (desc_type == (uint8_t)k_vkbd_dt_device) {
175 src = k_vkbd_device_desc;
176 src_len = (uint16_t)k_vkbd_dev_desc_len;
177 } else if (desc_type == (uint8_t)k_vkbd_dt_config) {
178 src = k_vkbd_config_desc;
179 src_len = (uint16_t)k_vkbd_cfg_desc_len;
180 } else if (desc_type == (uint8_t)k_vkbd_dt_hid_report) {
181 src = k_vkbd_report_desc;
182 src_len = (uint16_t)sizeof(k_vkbd_report_desc);
183 }
184 }
185 uint16_t n = 0U;
186 if ((src != nullptr) && (data_ptr != 0U)) {
187 n = (src_len < (uint16_t)data_len) ? src_len : (uint16_t)data_len;
188 (void)uc_mem_write(uc, (uint64_t)data_ptr, src, n);
189 }
190 /* No-data control writes (SET_ADDRESS / SET_CONFIGURATION / SET_IDLE /
191 * SET_PROTOCOL) just leave n = 0; the host treats that as a successful ack. */
192 if (out_ptr != 0U) {
193 (void)uc_mem_write(uc, (uint64_t)out_ptr, &n, sizeof(n));
194 }
195 s_vkbd_ctrl_serviced++;
196 eth_hook_return(uc, 0U); /* k_ra8_ok */
197}

202{
203 (void)address;
204 (void)size;

219 if ((uint32_t)n > max_len) {
220 n = (uint16_t)max_len;
221 }
222 if (buf != 0U) {
223 (void)uc_mem_write(uc, (uint64_t)buf, rep, n);
224 }
225 if (out_ptr != 0U) {
226 (void)uc_mem_write(uc, (uint64_t)out_ptr, &n, sizeof(n));

410static void vmsc_fill_fat(uint32_t fat_sector, uint8_t* out)
411{
412 const uint32_t first = fat_sector * (uint32_t)k_vmsc_entries_per_fs;
413 for (uint32_t j = 0U; j < (uint32_t)k_vmsc_entries_per_fs; j++) {
414 const uint32_t entry = first + j;
415 uint16_t value = 0U;
416 if (entry == 0U) {
417 value = (uint16_t)k_vmsc_fat_entry0;
418 } else if ((entry == 1U) || (entry == (uint32_t)k_vmsc_last_mram_clus)) {
419 /* Entry 1 is the reserved media/EOC marker and the last cluster ends the
420 * chain -- both hold the same EOC word. */
421 value = (uint16_t)k_vmsc_fat_eoc;
422 } else if (entry < (uint32_t)k_vmsc_last_mram_clus) {
423 value = (uint16_t)(entry + 1U);
424 }
425 vmsc_put16(&out[(size_t)j * 2U], value);
426 }
427}

559/* cppcheck-suppress constParameterCallback ; UC_HOOK_CODE callback ABI is void*. */
560static void on_hmsc_write10(uc_engine* uc, uint64_t address, uint32_t size, void* user)
561{
562 (void)address;
563 (void)size;
564 (void)user;
565 if (!s_vmsc_writable) {
566 /* Read-only disk (usb_host_msc_browse): reject. This is that app's last step
567 * before PASS, so flag it for the host early-stop. */
568 s_vmsc_write_seen = true;
569 eth_hook_return(uc, (uint32_t)k_ra8_err_inval_st); /* write protected */
570 return;
571 }
572 /* Writable disk (usb_host_file_ops): stash the written sectors in the overlay
573 * so the host's read-back sees them, and accept. The write is mid-ladder here,
574 * so do NOT trip the write-seen early-stop -- that app stops on its banner. */
575 uint32_t lba = 0U;
576 uint32_t count = 0U;
577 uint32_t buf = 0U;
578 (void)uc_reg_read(uc, UC_ARM_REG_R1, &lba);
579 (void)uc_reg_read(uc, UC_ARM_REG_R2, &count);
580 (void)uc_reg_read(uc, UC_ARM_REG_R3, &buf);
581 count &= (uint32_t)k_lo16_mask;
582 for (uint32_t i = 0U; (i < count) && (buf != 0U); i++) {
583 uint8_t sec[k_vmsc_block_size];
584 (void)uc_mem_read(uc,
585 (uint64_t)buf + ((uint64_t)i * (uint64_t)k_vmsc_block_size),
586 sec,
587 sizeof(sec));
588 vmsc_overlay_put(lba + i, sec);
589 }
590 eth_hook_return(uc, 0U); /* k_ra8_ok -- write accepted */
591}

617bool usbh_seam_install(uc_engine* uc, const uint8_t* elf, long len)
618{
619 const uint32_t msc = elf_sym_addr(elf, len, "ra8_usb_hmsc_read10", nullptr);
620 if (msc != 0U) {
621 /* usb_host_file_ops creates a file (it links fileops_backend_write) -> the
622 * virtual disk is writable; usb_host_msc_browse tests a read-only LUN. */

624 eth_seam_hook(uc, elf, len, "ra8_usb_hmsc_init", (void*)on_hmsc_ok);
625 eth_seam_hook(uc, elf, len, "ra8_usb_hmsc_enumerate", (void*)on_hmsc_enumerate);

629 eth_seam_hook(uc, elf, len, "ra8_usb_hmsc_close", (void*)on_hmsc_ok);
630 (void)fprintf(stderr,
631 " usb-host seam : hmsc=0x%08X (virtual MSC FAT16 disk, file MRAM.BIN, %s)\n",
632 msc,
633 s_vmsc_writable ? "read-write" : "read-only");
634 return true;
635 }
636 const uint32_t cx = elf_sym_addr(elf, len, "ra8_usb_host_control_xfer", nullptr);
637 if (cx == 0U) {
638 return false; /* not a USB-host-capable firmware -- nothing to seam. */
639 }
640 eth_seam_hook(uc, elf, len, "ra8_usb_host_line_state", (void*)on_usbh_line_state);
641 eth_seam_hook(uc, elf, len, "ra8_usb_host_control_xfer", (void*)on_usbh_control_xfer);

644 eth_seam_hook(uc, elf, len, "ra8_usb_host_set_uact", (void*)on_usbh_ok);
645 eth_seam_hook(uc, elf, len, "ra8_usb_host_set_target", (void*)on_usbh_ok);
646 eth_seam_hook(uc, elf, len, "ra8_usb_host_pipe_setup", (void*)on_usbh_ok);
647 (void)fprintf(stderr,
648 " usb-host seam : control=0x%08X (virtual HID boot keyboard \"RA8D2\")\n",
649 cx);
650 return true;
651}
