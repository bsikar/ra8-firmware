352static void dtc_demo_program_ti(void)
353{
354 const uint8_t mra = (uint8_t)(((uint8_t)k_dtc_md_block << k_dtc_mra_md_pos) |
355 ((uint8_t)k_dtc_sz_word << k_dtc_mra_sz_pos) |
356 ((uint8_t)k_dtc_sm_inc << k_dtc_mra_sm_pos));
357 const uint8_t mrb = (uint8_t)((uint8_t)k_dtc_dm_inc << k_dtc_mrb_dm_pos);
358 /* MR[31:24]=MRA, MR[23:16]=MRB, MR[15:8]=MRC(0); HUM Ch 18.2.2 p 786 /
359 * 18.2.3 p 787 / 18.2.4 p 789, Figure 18.4 p 799. SRAM (not MMIO). */
360 s_dtc_ti.MR = ((uint32_t)mra << k_dtc_mra_byte_pos) | ((uint32_t)mrb << k_dtc_mrb_byte_pos);
361 s_dtc_ti.SAR = (uint32_t)(uintptr_t)s_src;
362 s_dtc_ti.DAR = (uint32_t)(uintptr_t)s_dst;
363 s_dtc_ti.CRB = (uint16_t)k_dtc_crb_one_block;
364 s_dtc_ti.CRA = (uint16_t)k_dtc_cra_block_256;
365}

382[[nodiscard]] static ra8_err_t dtc_demo_arm_slot(void)
383{
384 volatile uint32_t* slot_reg = ra8_icu_ielsr(s_dtc_slot);
385 if (slot_reg == nullptr) {
386 return k_ra8_err_hw_error;
387 }
388 /* HUM Ch 14.2.10 "IELSRn : ICU Event Link Setting Register n" p 524:
389 * DTCE[24] enables DTC activation; IR[16] is the RW1C status flag. */
390 const uint32_t cur = *slot_reg;
391 *slot_reg = cur | (uint32_t)k_ra8_ielsr_dtce_mask;
392 return k_ra8_ok;
393}

408static void dtc_demo_bringup_or_halt(void)
409{
410 if (ra8_dtc_init(s_dtc_vt) != k_ra8_ok) {
411 dtc_demo_panic_halt();
412 }
413 if (ra8_dtc_attach_handler(dtc_demo_complete_cb, nullptr) != k_ra8_ok) {
414 dtc_demo_panic_halt();
415 }
416 if (ra8_isr_register((ra8_elc_event_t)k_dtc_demo_event_swevt0,
417 dtc_demo_swevt_isr,
418 nullptr,
419 (uint8_t)k_dtc_demo_isr_prio,
420 &s_dtc_slot) != k_ra8_ok) {
421 dtc_demo_panic_halt();
422 }
423 if (s_dtc_slot >= (uint16_t)k_dtc_vt_entries) {
424 dtc_demo_panic_halt();
425 }
426 /* DTCVBR + slot*4 holds the 16-byte-aligned TI start address; bit0 is
427 * the privilege attribution (0 = privileged). HUM Ch 18.3.1 p 796 +
428 * Figure 18.3 p 798. SRAM vector-table write (not MMIO). */
429 s_dtc_vt[s_dtc_slot] = (uint32_t)(uintptr_t)&s_dtc_ti;
430 if (ra8_dtc_enable() != k_ra8_ok) {
431 dtc_demo_panic_halt();
432 }
433}

478[[nodiscard]] static ra8_err_t dtc_demo_run_once(uint8_t* out_ok)
479{
480 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
481
482 dtc_demo_program_ti();
483 const ra8_err_t arm = dtc_demo_arm_slot();
484 if (arm != k_ra8_ok) {
485 return arm;
486 }
487 const ra8_err_t trig = ra8_elc_software_trigger((uint8_t)k_dtc_demo_swevt_index);
488 if (trig != k_ra8_ok) {
489 return trig;
490 }
491 ++g_dtc_activations;
492
493 for (uint32_t i = 0U; i < (uint32_t)k_dtc_demo_poll_limit; ++i) {
494 if (s_dst[k_dtc_demo_buf_words - 1U] == s_src[k_dtc_demo_buf_words - 1U]) {
495 break;
496 }
497 }
498
499 *out_ok = dtc_demo_verify();
500 return k_ra8_ok;
501}

505int32_t main(void)
506{
507 dtc_demo_setup_or_halt();
508 dtc_demo_bringup_or_halt();
509 /* Completion is detected by polling ``s_dst`` (step 4 below), so the global
510 * interrupt is intentionally left masked. On RA8 silicon, enabling the
511 * DTC-complete CPU interrupt lets its ISR (which writes ``DTCSTS``) race the
512 * in-flight DTC transfer and corrupt the copy; the headless emulator, which
513 * runs the transfer synchronously on the ELSEGR write, does not expose this.
514 * The IELSR slot + ``DTCE`` armed in bring-up still activate the DTC -- we
515 * simply do not take the completion IRQ. */
516
517 while (1) {
518 dtc_demo_fill_buffers();
519 uint8_t ok = 0U;
520 const ra8_err_t err = dtc_demo_run_once(&ok);
521 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
522 g_dtc_match = (uint32_t)good;
523 g_dtc_bytes = (good != 0U) ? (uint32_t)k_dtc_demo_buf_bytes : 0U;
524 if (good != 0U) {
525 (void)ra8_board_uart_console_write(k_dtc_demo_ok_msg,
526 (size_t)(sizeof(k_dtc_demo_ok_msg) - 1U));
527 (void)ra8_board_led_toggle(k_ra8_board_led1);
528 } else {
529 (void)ra8_board_uart_console_write(k_dtc_demo_bad_msg,
530 (size_t)(sizeof(k_dtc_demo_bad_msg) - 1U));
531 (void)ra8_board_led_toggle(k_ra8_board_led2);
532 }
533 ++g_dtc_heartbeat;
534 ra8_delay_ms(k_dtc_demo_period_ms);
535 }
536 dtc_demo_panic_halt();
537 return 0;
538}
