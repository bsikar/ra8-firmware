351static bool priv_glyph_render_cached(ra8_glyph_atlas_t* atlas,
352 uint8_t face_id,
353 const stbtt_fontinfo* font,
354 float scale,

408RA8_INTERNAL
409static void priv_blit_glyph(ra8_glyph_atlas_t* atlas,
410 uint8_t face_id,
411 const stbtt_fontinfo* font,
412 const ra8_reflow_glyph_t* g,
413 int32_t ox,
414 int32_t oy)
415{
416 const float scale = stbtt_ScaleForPixelHeight(font, (float)g->font_px);
417 int x0 = 0;
418 int y0 = 0;
419 int x1 = 0;
420 int y1 = 0;
421 stbtt_GetCodepointBitmapBox(font, g->cp, scale, scale, &x0, &y0, &x1, &y1);
422 const int w = x1 - x0;
423 const int h = y1 - y0;
424 /*
425 * w and h are the glyph bitmap-box extents returned by
426 * stbtt_GetCodepointBitmapBox(). A codepoint is either inked -- a non-empty
427 * box with w > 0 AND h > 0 -- or empty -- a zero box with w == 0 AND h == 0
428 * (e.g. U+0020 space). The mixed states (w <= 0, h > 0) and (w > 0, h <= 0)
429 * never occur for a real font glyph, so the vectors that would give w and h
430 * independent MC/DC influence are structurally unreachable.
431 */
432 /* mcdc-deactivated: glyph bbox w,h co-dependent (inked or empty); mixed vectors unreachable. */
433 if ((w > 0) && (h > 0)) {
434 bool drawn = false;
435 if (atlas != nullptr) {
436 drawn = priv_glyph_render_cached(atlas, face_id, font, scale, g, w, h, x0, y0, ox, oy);
437 }
438 if (!drawn) {
439 priv_glyph_render_direct(font, scale, g, w, h, x0, y0, ox, oy);
440 }
441 }
442
443 if ((g->style & k_ra8_reflow_style_underline) != 0U) {
444 priv_draw_underline(font, g, scale, ox, oy);
445 }
446}

700RA8_INTERNAL
701static ra8_err_t
702priv_render_page(const ra8_reflow_t* engine, uint32_t page_idx, int32_t ox, int32_t oy)
703{
704 if (engine == nullptr) {
705 return k_ra8_err_null_ptr;
706 }
707 if (engine->in_use == 0U) {
708 return k_ra8_err_not_initialized;
709 }
710 if (page_idx >= engine->page_count) {
711 return k_ra8_err_out_of_range;
712 }
713
714 /* One fontinfo per face (static -- render is single-threaded, like s_glyph_mask
715 * below; keeps the ~1.4 KB off the stack). Default at 0, embedded at 1..N. */
716 static stbtt_fontinfo s_faces[1U + (uint32_t)k_ra8_reflow_max_faces];
717 uint8_t nfaces = 0U;
718 ra8_err_t err = priv_init_faces(engine, s_faces, &nfaces);
719 if (err != k_ra8_ok) {
720 return err;
721 }
722
723 const ra8_reflow_page_t* page = &engine->pages[page_idx];
724 for (uint32_t i = 0U; i < page->glyph_count; ++i) {
725 const ra8_reflow_glyph_t* g = &engine->glyphs[page->glyph_first + i];
726 uint8_t fi = (uint8_t)(((uint32_t)g->style >> (uint32_t)k_ra8_reflow_face_shift) &
727 (uint32_t)k_ra8_reflow_face_mask);
728 if (fi >= nfaces) {
729 fi = 0U; /* defensive: out-of-range face index -> default */
730 }
731 priv_blit_glyph(engine->glyph_atlas, fi, &s_faces[fi], g, ox, oy);
732 }
733 priv_render_images(engine, page_idx, ox, oy);
734 return k_ra8_ok;
735}
