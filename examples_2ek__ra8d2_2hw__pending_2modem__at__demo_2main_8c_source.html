281static bool modem_parse_first_uint(const char* line, uint32_t* out)
282{
283 if ((line == nullptr) || (out == nullptr)) {
284 return false;
285 }
286 uint32_t i = 0U;
287 while ((i < (uint32_t)k_modem_scan_max) && (line[i] != '\0') && (line[i] != ':')) {
288 i++;
289 }
290 if ((i >= (uint32_t)k_modem_scan_max) || (line[i] != ':')) {
291 return false;
292 }
293 i++;
294 while ((i < (uint32_t)k_modem_scan_max) && (line[i] == ' ')) {
295 i++;
296 }
297 if ((i >= (uint32_t)k_modem_scan_max) || (line[i] < '0') || (line[i] > '9')) {
298 return false;
299 }
300 uint32_t v = 0U;
301 while ((i < (uint32_t)k_modem_scan_max) && (line[i] >= '0') && (line[i] <= '9')) {
302 v = (v * (uint32_t)k_modem_dec_base) + (uint32_t)(line[i] - '0');
303 i++;
304 }
305 *out = v;
306 return true;
307}

330static bool modem_parse_last_uint(const char* line, uint32_t* out)
331{
332 if ((line == nullptr) || (out == nullptr)) {
333 return false;
334 }
335 bool found = false;
336 uint32_t v = 0U;
337 uint32_t i = 0U;
338 while ((i < (uint32_t)k_modem_scan_max) && (line[i] != '\0')) {
339 if ((line[i] >= '0') && (line[i] <= '9')) {
340 v = 0U;
341 while ((i < (uint32_t)k_modem_scan_max) && (line[i] >= '0') && (line[i] <= '9')) {
342 v = (v * (uint32_t)k_modem_dec_base) + (uint32_t)(line[i] - '0');
343 i++;
344 }
345 found = true;
346 } else {
347 i++;
348 }
349 }
350 if (found) {
351 *out = v;
352 }
353 return found;
354}

372static bool modem_line_contains(const char* line, const char* needle)
373{
374 if ((line == nullptr) || (needle == nullptr)) {
375 return false;
376 }
377 for (uint32_t i = 0U; (i < (uint32_t)k_modem_scan_max) && (line[i] != '\0'); i++) {
378 uint32_t j = 0U;
379 while ((j < (uint32_t)k_modem_scan_max) && (needle[j] != '\0') && (line[i + j] == needle[j])) {
380 j++;
381 }
382 if (needle[j] == '\0') {
383 return true;
384 }
385 }
386 return false;
387}

409static uint32_t modem_u32_to_dec(uint32_t value, uint8_t* out)
410{
411 uint8_t tmp[k_modem_dec_cap] = {};
412 uint32_t n = 0U;
413 if (value == 0U) {
414 tmp[n] = (uint8_t)'0';
415 n++;
416 } else {
417 while ((value > 0U) && (n < (uint32_t)k_modem_dec_cap)) {
418 tmp[n] = (uint8_t)('0' + (uint8_t)(value % (uint32_t)k_modem_dec_base));
419 n++;
420 value = value / (uint32_t)k_modem_dec_base;
421 }
422 }
423 for (uint32_t i = 0U; i < n; i++) {
424 out[i] = tmp[n - 1U - i];
425 }
426 return n;
427}

903int32_t main(void)
904{
905 modem_setup_or_halt();
906 ra8_isr_globals_enable();
907
908 static uint8_t s_line[k_modem_capture_cap * 2U];
909 if (modem_bind(s_line, (uint16_t)sizeof s_line) != k_ra8_ok) {
910 static const uint8_t k_step[] = "init";
911 modem_fail(k_step, (uint32_t)(sizeof(k_step) - 1U));
912 }
913
914 modem_phase_sync();
915 const bool sim_ready = modem_phase_sim();
916 uint32_t rssi = (uint32_t)k_modem_csq_unknown;
917 const bool signal_ok = modem_phase_signal(&rssi);
918 const bool registered = modem_phase_registration();
919 const bool attached = modem_phase_attach();
920 const bool cme_ok = modem_phase_error();
921
922 const bool ok = modem_run_ok(sim_ready, signal_ok, registered, attached, cme_ok);
923 /* "ok"/"no" are both 2 bytes and " PASS\r\n"/" FAIL\r\n" both 7, so one length
924 * expression per pair keeps -Wduplicated-branches happy. */
925 modem_tx(k_modem_str_cme, (uint32_t)(sizeof(k_modem_str_cme) - 1U));
926 modem_tx(cme_ok ? k_modem_str_ok : k_modem_str_no, (uint32_t)(sizeof(k_modem_str_ok) - 1U));
927 modem_tx(ok ? k_modem_str_pass : k_modem_str_fail, (uint32_t)(sizeof(k_modem_str_pass) - 1U));
928
929 while (1) {
930 (void)ra8_board_led_toggle(k_ra8_board_led1);
931 ra8_delay_ms((uint32_t)k_modem_heartbeat_ms);
932 }
933
934 modem_panic_halt();
935 return 0;
936}
