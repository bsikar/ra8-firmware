165static char sdcard_nibble_to_hex(uint8_t nibble)
166{
167 const uint8_t n = (uint8_t)(nibble & (uint8_t)k_sdcard_hex_nibble_mask);
168 if (n < (uint8_t)k_sdcard_hex_alpha_threshold) {
169 return (char)('0' + (char)n);
170 }
171 return (char)('A' + (char)(n - (uint8_t)k_sdcard_hex_alpha_threshold));
172}

184static void sdcard_hex_dump(const uint8_t* in, uint8_t len, char* out)
185{
186 for (uint8_t i = 0U; i < len; i++) {
187 const uint8_t b = in[i];
188 out[i * k_sdcard_hex_chars_per_byte + 0U] =
189 sdcard_nibble_to_hex((uint8_t)(b >> k_sdcard_hex_nibble_shift));
190 out[i * k_sdcard_hex_chars_per_byte + 1U] =
191 sdcard_nibble_to_hex((uint8_t)(b & (uint8_t)k_sdcard_hex_nibble_mask));
192 out[i * k_sdcard_hex_chars_per_byte + 2U] = ' ';
193 }
194}

226static void sdcard_one_pass(void)
227{
228 const ra8_err_t err = ra8_sdcard_read_blocks(0U, s_sdcard_block, 1U);
229 if (err != k_ra8_ok) {
230 sdcard_log("sdcard: read block 0 failed\r\n");
231 return;
232 }
233 char dump[k_sdcard_dump_bytes * k_sdcard_hex_chars_per_byte + 1U] = {};
234 sdcard_hex_dump(s_sdcard_block, (uint8_t)k_sdcard_dump_bytes, dump);
235 dump[k_sdcard_dump_bytes * k_sdcard_hex_chars_per_byte] = '\0';
236 sdcard_log("sdcard: blk0[0..15] = ");
237 sdcard_log(dump);
238 sdcard_log("\r\n");
239}

249static void sdcard_thread_entry(ULONG thread_input)
250{
251 (void)thread_input;
252
253 const ra8_sdcard_cfg_t cfg = {.instance = k_sdcard_sdhi_instance};
254 if (ra8_sdcard_init(&cfg) != k_ra8_ok) {
255 sdcard_log("sdcard: init failed (no card / bad pinmux?)\r\n");
256 while (1) {
257 (void)tx_thread_sleep((ULONG)k_sdcard_loop_ticks);
258 }
259 }
260 sdcard_log("sdcard: card initialized\r\n");
261
262 while (1) {
263 sdcard_one_pass();
264 (void)ra8_board_led_toggle(k_ra8_board_led1);
265 (void)tx_thread_sleep((ULONG)k_sdcard_loop_ticks);
266 }
267}

283/* NOLINTNEXTLINE(misc-use-internal-linkage) -- exported symbol expected by ThreadX. */
284void tx_application_define(void* first_unused_memory)
285{
286 (void)first_unused_memory;
287
288 /* Retune SysTick to the live CPUCLK0 (raised to 1 GHz by
289 * ra8_cgc_init() in main()). tx_initialize_low_level.S already
290 * programmed the reload from the same compile-time 1 GHz assumption,
291 * but this makes the 1 ms kernel tick correct-by-construction for
292 * whatever clock the app actually brought up (issue #287). Runs after
293 * _tx_initialize_low_level and before the first scheduling decision. */
294 if (ra8_threadx_systick_retune() != k_ra8_ok) {
295 while (1) {
296 __asm__ volatile("wfi");
297 }
298 }
299
300 const UINT err = tx_thread_create(&s_sdcard_thread,
301 "sdcard",
302 sdcard_thread_entry,
303 0U,
304 s_sdcard_stack,
305 (ULONG)k_sdcard_thread_stack_bytes,
306 (UINT)k_sdcard_thread_priority,
307 (UINT)k_sdcard_thread_priority,
308 TX_NO_TIME_SLICE,
309 TX_AUTO_START);
310 if (err != TX_SUCCESS) {
311 while (1) {
312 __asm__ volatile("wfi");
313 }
314 }
315}
316#endif /* !RA8_OFF_TARGET */

335#pragma GCC diagnostic push
336#pragma GCC diagnostic ignored "-Wmain"
337int32_t main(void)
338{
339 /* CGC bring-up FIRST. tx_initialize_low_level.S programs SysTick from
340 * the post-CGC CPUCLK0 target (1 GHz); entering the kernel on the
341 * boot-default MOCO (~8 MHz) instead would run the "1 ms" tick ~119x
342 * too slow, so tx_thread_sleep(5000) would stretch to ~10 minutes.
343 * Raise the PLL before tx_kernel_enter (and tx_application_define
344 * additionally retunes SysTick from the live clock -- issue #287). */
345 if (ra8_cgc_init() != k_ra8_ok) {
346 while (1) {
347 __asm__ volatile("wfi");
348 }
349 }
350 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
351 while (1) {
352 __asm__ volatile("wfi");
353 }
354 }
355 if (ra8_board_uart_console_init((uint32_t)k_sdcard_console_baud) != k_ra8_ok) {
356 while (1) {
357 __asm__ volatile("wfi");
358 }
359 }
360#ifndef RA8_OFF_TARGET
361 if (ra8_board_sdhi_pins_init() != k_ra8_ok) {
362 while (1) {
363 __asm__ volatile("wfi");
364 }
365 }
366 ra8_isr_globals_enable();
367 tx_kernel_enter();
368#endif
369
370 while (1) {
371 __asm__ volatile("wfi");
372 }
373}
