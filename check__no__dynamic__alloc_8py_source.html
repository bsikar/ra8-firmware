4"""Enforce NASA Power-of-10 Rule 3 -- no dynamic memory after init.
5
6Applies to RA8D2 firmware code.
7
8Flags two classes of violation:
9
10 * **Direct allocator calls** -- malloc, calloc, realloc, reallocarray,
11 aligned_alloc, posix_memalign, valloc, memalign, free, strdup,
12 strndup, asprintf, vasprintf.
13
14 * **Vendored helpers that allocate transitively** under the hood, so a
15 plain grep for malloc/free won't catch them:
16
17 - stb_truetype: stbtt_GetCodepointBitmap[Subpixel],
18 stbtt_GetGlyphBitmap[Subpixel],
19 stbtt_GetCodepointSDF, stbtt_GetGlyphSDF,
20 stbtt_GetCodepointShape, stbtt_GetGlyphShape,
21 stbtt_FreeBitmap, stbtt_FreeSDF, stbtt_FreeShape.
22 The no-alloc paths are the Make*/Box variants.
23
24 - stb_image: stbi_load, stbi_load_from_memory,
25 stbi_load_from_callbacks, stbi_loadf,
26 stbi_loadf_from_memory, stbi_image_free.
27
28 - miniz: mz_zip_reader_extract_to_heap,
29 mz_zip_reader_extract_file_to_heap.
30 Use the *_to_mem variants with a caller buffer.
31
32Scope:
33
34 Only firmware code under libs/ra8_*/, src/, and examples/<app>/ where
35 <app> has main.c + CMakeLists.txt. Vendored libs (libs/third_party/,
36 libs/threadx, libs/filex, ...), build outputs, and host-side tests/
37 are exempt -- those are allowed to allocate.
38
39Inline suppression:
40
41 Append `alloc-allow: <reason>` on the same line. The reason is
42 mandatory; bare `alloc-allow` with no reason is itself rejected.
43
44Example:
45 void* p = malloc(64); /* alloc-allow: bringup scratch, removed in v0.2 */
46
47Usage:
48
49 # explicit file list (used by pre-commit):
50 python3 scripts/checks/check_no_dynamic_alloc.py FILE [FILE ...]
51
52 # full-repo sweep (CI):
53 python3 scripts/checks/check_no_dynamic_alloc.py --all
54
55Exit code:
56 0 no violations
57 1 violations found
58 2 CLI usage error
59"""

110# Match "<name>(" at a word boundary so e.g. mz_free(...) does not match
111# free, and stbtt_GetCodepointBitmapBox does not match
112# stbtt_GetCodepointBitmap.
113SYM_RE = re.compile(r"\b(" + "|".join(re.escape(n) for n in ALL_NAMES) + r")\b\s*\‍(")
114
115# Inline exemption marker. The reason after the colon is mandatory.
116ALLOW_RE = re.compile(r"alloc-allow\s*:\s*\S")
117BARE_ALLOW_RE = re.compile(r"alloc-allow\b")
118
119BLOCK_COMMENT_RE = re.compile(r"/\*.*?\*/", re.DOTALL)
120LINE_COMMENT_RE = re.compile(r"//[^\n]*")

126def _scan_dirs() -> list[pathlib.Path]:
127 """The firmware directories this rule governs.
128
129 Deliberately narrower than the whole tree: host tools and tests may
130 allocate freely, and Rule 3 is a statement about what runs on the target.
131 """
132 out: list[pathlib.Path] = []
133 libs = REPO_ROOT / "libs"
134 if libs.is_dir():
135 out.extend(
136 entry
137 for entry in sorted(libs.iterdir())
138 if entry.is_dir() and entry.name.startswith("ra8_")
139 )
140 src = REPO_ROOT / "src"
141 if src.is_dir():
142 out.append(src)
143 examples = REPO_ROOT / "examples"
144 if examples.is_dir():
145 for tier in sorted(examples.iterdir()):
146 if not tier.is_dir():
147 continue
148 for entry in sorted(tier.iterdir()):
149 if not entry.is_dir():
150 continue
151 if (entry / "main.c").is_file() and (entry / "CMakeLists.txt").is_file():
152 out.append(entry)
153 return out

160def _is_in_scope(path: pathlib.Path) -> bool:
161 """True if `path` is firmware code we should lint."""
162 if path.suffix not in SOURCE_SUFFIXES:
163 return False
164 try:
165 rel = path.resolve().relative_to(REPO_ROOT)
166 except ValueError:
167 return False
168 excluded_parts = {"third_party", "build", "_deps"}
169 if any(part in excluded_parts for part in rel.parts):
170 return False
171 rel_str = str(rel)
172 return any(rel_str == str(d) or rel_str.startswith(str(d) + "/") for d in SCAN_DIR_RELS)

175def _strip_comments(text: str) -> str:
176 """Strip /* */ and // comments while preserving line numbers."""
177 no_block = BLOCK_COMMENT_RE.sub(lambda m: "\n" * m.group(0).count("\n"), text)
178 return LINE_COMMENT_RE.sub("", no_block)

181def check(path: pathlib.Path) -> list[str]:
182 """Report every dynamic-allocation call in one firmware file.
183
184 An unreadable file yields an empty list rather than raising, so one bad
185 file cannot abort the sweep.
186 """
187 try:
188 text = path.read_text(encoding="utf-8")
189 except (OSError, UnicodeDecodeError):
190 return []
191
192 original_lines = text.splitlines()
193 stripped_lines = _strip_comments(text).splitlines()
194 problems: list[str] = []
195
196 for idx, stripped in enumerate(stripped_lines):
197 line_no = idx + 1
198 original = original_lines[idx] if idx < len(original_lines) else ""
199 for m in SYM_RE.finditer(stripped):
200 sym = m.group(1)
201 if ALLOW_RE.search(original):
202 continue
203 kind = "direct" if sym in DIRECT_ALLOCATORS else "transitive"
204 problems.append(
205 f"{path}:{line_no}: {kind} dynamic allocation: {sym}() -- {original.strip()}"
206 )
207
208 for idx, original in enumerate(original_lines):
209 line_no = idx + 1
210 if BARE_ALLOW_RE.search(original) and not ALLOW_RE.search(original):
211 problems.append(
212 f"{path}:{line_no}: alloc-allow without reason "
213 f"-- write `alloc-allow: <reason>` instead"
214 )
215
216 return problems

219def collect_repo_paths() -> list[pathlib.Path]:
220 """Every in-scope firmware source file, for the whole-tree sweep."""
221 return [p for d in SCAN_DIRS for p in d.rglob("*") if _is_in_scope(p)]

224def main() -> int:
225 """Fail when firmware calls malloc, free or a relative.
226
227 The rule targets the target image only -- host tools and tests are out of
228 scope -- because the hazard is heap exhaustion and fragmentation on a
229 device with no allocation-failure path, not allocation as such.
230
231 Returns 1 listing each call site, 0 when the firmware is clean.
232 """
233 if len(sys.argv) >= MIN_ARGC_WITH_ARG and sys.argv[1] == "--all":
234 paths = collect_repo_paths()
235 elif len(sys.argv) >= MIN_ARGC_WITH_ARG:
236 paths = [pathlib.Path(p).resolve() for p in sys.argv[1:]]
237 else:
238 print(
239 "usage: check_no_dynamic_alloc.py FILE [FILE ...] | --all",
240 file=sys.stderr,
241 )
242 return 2
243
244 failures: list[str] = []
245 for path in paths:
246 if not path.is_file():
247 continue
248 if not _is_in_scope(path):
249 continue
250 failures.extend(check(path))
251
252 if failures:
253 print(
254 "check_no_dynamic_alloc.py: NASA Rule 3 -- no dynamic allocation in firmware code.",
255 file=sys.stderr,
256 )
257 for line in failures:
258 print(line, file=sys.stderr)
259 print(f"\n{len(failures)} violation(s) found.", file=sys.stderr)
260 print(
261 "If a call site is genuinely OK (host-only test path, "
262 "vendored glue you cannot move), append "
263 "`alloc-allow: <reason>` on the same line.",
264 file=sys.stderr,
265 )
266 return 1
267 return 0
