185RA8_INTERNAL static ra8_err_t internal_demo_read_once(const uint8_t* expect)
186{
187 RA8_CHECK_NULL_PTR(expect, s_tag, "expect");
188 ra8_fs_file_t* f = nullptr;
189 RA8_RETURN_ON_ERROR(ra8_io_vfs_open("ram:/HELLO.TXT", k_ra8_fs_mode_read, &f), s_tag, "open");
190 uint8_t got[(size_t)k_demo_payload] = {};
191 uint32_t got_len = 0;
192 RA8_RETURN_ON_ERROR(ra8_fs_read(f, got, (uint32_t)k_demo_payload, &got_len), s_tag, "read");
193 RA8_RETURN_ON_ERROR(ra8_fs_close(f), s_tag, "close");
194 if (got_len != (uint32_t)k_demo_payload) {
195 return k_ra8_err_invalid_size;
196 }
197 if (memcmp(got, expect, (size_t)k_demo_payload) != 0) {
198 return k_ra8_err_checksum_mismatch;
199 }
200 return k_ra8_ok;
201}

221RA8_INTERNAL static ra8_err_t internal_demo_run(uint32_t* out_hits, uint32_t* out_misses)
222{
223 RA8_CHECK_NULL_PTR(out_hits, s_tag, "out_hits");
224 RA8_CHECK_NULL_PTR(out_misses, s_tag, "out_misses");
225 ra8_fs_mount_t* mnt = nullptr;
226 RA8_RETURN_ON_ERROR(internal_demo_mount(&mnt), s_tag, "mount stage");
227
228 uint8_t data[(size_t)k_demo_payload];
229 for (uint32_t i = 0; i < (uint32_t)k_demo_payload; ++i) {
230 data[i] = (uint8_t)(i * (uint32_t)k_demo_seed_mul + 1U);
231 }
232 RA8_RETURN_ON_ERROR(ra8_fs_write_file(mnt, "HELLO.TXT", data, (uint32_t)k_demo_payload),
233 s_tag,
234 "write");
235
236 for (uint32_t pass = 0; pass < (uint32_t)k_demo_reads; ++pass) {
237 RA8_RETURN_ON_ERROR(internal_demo_read_once(data), s_tag, "reread");
238 }
239
240 RA8_RETURN_ON_ERROR(ra8_io_blockdev_cache_stats(&s_cstate, out_hits, out_misses), s_tag, "stats");
241 if (*out_hits == 0U) {
242 return k_ra8_err_checksum_mismatch;
243 }
244 return k_ra8_ok;
245}
