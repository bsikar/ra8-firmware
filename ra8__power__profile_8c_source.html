192ra8_err_t ra8_power_profile_mark_enter(ra8_power_profile_region_id_t region_id)
193{
194 RA8_VALIDATE_INIT(s_initialized, RA8_POWER_PROFILE_TAG, "init not called");
195
196 const ra8_err_t range_err = internal_validate_region(region_id);
197 if (range_err != k_ra8_ok) {
198 return range_err;
199 }
200
201 ra8_power_profile_region_stats_t* slot = &s_stats.regions[(uint8_t)region_id];
202 slot->entries += 1U;
203 slot->last_enter_us = internal_now_us();
204 slot->is_open = true;
205
206 internal_fire_pulse(region_id, true);
207 return k_ra8_ok;
208}

210ra8_err_t ra8_power_profile_mark_exit(ra8_power_profile_region_id_t region_id)
211{
212 RA8_VALIDATE_INIT(s_initialized, RA8_POWER_PROFILE_TAG, "init not called");
213
214 const ra8_err_t range_err = internal_validate_region(region_id);
215 if (range_err != k_ra8_ok) {
216 return range_err;
217 }
218
219 ra8_power_profile_region_stats_t* slot = &s_stats.regions[(uint8_t)region_id];
220 slot->exits += 1U;
221
222 ra8_err_t ret = k_ra8_ok;
223 if (slot->is_open) {
224 const uint64_t now = internal_now_us();
225 if (now >= slot->last_enter_us) {
226 slot->total_time_us += (now - slot->last_enter_us);
227 }
228 slot->is_open = false;
229 } else {
230 ra8_log_error(RA8_POWER_PROFILE_TAG, "exit without matching enter");
231 ret = k_ra8_err_invalid_state;
232 }
233
234 internal_fire_pulse(region_id, false);
235 return ret;
236}
