123static void mpu_install_ro_hooks(uc_engine* uc)

190static void on_mpu_rlar_write(uc_engine* uc,
191 uc_mem_type type,
192 uint64_t addr,
193 int size,
194 int64_t value,
195 void* user)
196{
197 (void)type;
198 (void)addr;
199 (void)size;
200 (void)user;
201 const uint32_t rnr = rd32(uc, (uint64_t)k_mpu_rnr) & (uint32_t)k_mpu_rnr_mask;
202 const uint32_t rbar = rd32(uc, (uint64_t)k_mpu_rbar);
203 const uint32_t rlar = (uint32_t)value;
204 s_mpu_region[rnr].base = rbar & (uint32_t)k_mpu_addr_mask;
205 s_mpu_region[rnr].limit = (rlar & (uint32_t)k_mpu_addr_mask) | (uint32_t)k_mpu_region_low_mask;
206 s_mpu_region[rnr].ro = (rbar & (uint32_t)k_mpu_rbar_ap_ro_bit) != 0U;
207 s_mpu_region[rnr].en = (rlar & (uint32_t)k_mpu_rlar_en_bit) != 0U;
208}

291void emu_mpu_install(uc_engine* uc)
292{
293 /* Armv8-M MPU enforcement: capture each region at its RLAR write, then arm /
294 * disarm read-only-region write traps on MPU_CTRL.ENABLE edges. Inert unless
295 * the firmware programs and enables the MPU (mpu_partition_simple); a trapped
296 * store synthesises MemManage in the run loop (see on_mpu_ro_write). */
297 static uc_hook s_h_mpu_rlar;
298 static uc_hook s_h_mpu_ctrl;
299 (void)uc_hook_add(uc,
300 &s_h_mpu_rlar,
301 UC_HOOK_MEM_WRITE,
302 (void*)on_mpu_rlar_write,
303 nullptr,
304 (uint64_t)k_mpu_rlar,
305 (uint64_t)k_mpu_rlar + 3U);
306 (void)uc_hook_add(uc,
307 &s_h_mpu_ctrl,
308 UC_HOOK_MEM_WRITE,
309 (void*)on_mpu_ctrl_write,
310 nullptr,
311 (uint64_t)k_mpu_ctrl,
312 (uint64_t)k_mpu_ctrl + 3U);
313}
