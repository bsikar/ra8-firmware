203[[nodiscard]] static ra8_err_t drw_demo_fill_and_check(uint8_t* out_ok)
204{
205 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
206
207 for (uint32_t i = 0U; i < (uint32_t)k_drw_demo_fb_dim * (uint32_t)k_drw_demo_fb_dim; ++i) {
208 g_drw_fb[i] = 0U;
209 }
210
211 const ra8_drw_rect_t rect = {
212 .x = (int16_t)k_drw_demo_rect_xy,
213 .y = (int16_t)k_drw_demo_rect_xy,
214 .width_px = (uint16_t)k_drw_demo_rect_wh,
215 .height_px = (uint16_t)k_drw_demo_rect_wh,
216 .color_argb8888 = (uint32_t)k_drw_demo_fill_argb,
217 };
218 const ra8_err_t ferr = ra8_drw_fill_rect(&rect);
219 if (ferr != k_ra8_ok) {
220 g_drw_fill_err = (uint32_t)ferr;
221 return ferr;
222 }
223 const ra8_err_t werr = ra8_drw_wait_idle((uint32_t)k_drw_demo_idle_budget);
224 g_drw_fill_err = (werr == k_ra8_ok) ? 0U : (uint32_t)werr;
225
226 const uint32_t centre = g_drw_fb[drw_demo_px(k_drw_demo_center, k_drw_demo_center)];
227 const uint32_t tl = g_drw_fb[drw_demo_px(0U, 0U)];
228 const uint32_t br =
229 g_drw_fb[drw_demo_px((uint16_t)(k_drw_demo_fb_dim - 1U), (uint16_t)(k_drw_demo_fb_dim - 1U))];
230 *out_ok = (centre == (uint32_t)k_drw_demo_fill_argb && tl == 0U && br == 0U) ? 1U : 0U;
231 return k_ra8_ok;
232}

236int32_t main(void)
237{
238 drw_demo_setup_or_halt();
239 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
240 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No NVIC sources
241 * are armed by this demo. */
242 ra8_isr_globals_enable();
243
244 if (drw_demo_configure() != k_ra8_ok) {
245 drw_demo_panic_halt();
246 }
247
248 while (1) {
249 uint8_t ok = 0U;
250 const ra8_err_t err = drw_demo_fill_and_check(&ok);
251 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
252 g_drw_match = (uint32_t)good;
253 if (good != 0U) {
254 (void)ra8_board_uart_console_write(k_drw_demo_ok_msg,
255 (size_t)(sizeof(k_drw_demo_ok_msg) - 1U));
256 (void)ra8_board_led_toggle(k_ra8_board_led1);
257 } else {
258 (void)ra8_board_uart_console_write(k_drw_demo_bad_msg,
259 (size_t)(sizeof(k_drw_demo_bad_msg) - 1U));
260 (void)ra8_board_led_toggle(k_ra8_board_led2);
261 }
262 ++g_drw_heartbeat;
263 ra8_delay_ms(k_drw_demo_period_ms);
264 }
265 drw_demo_panic_halt();
266 return 0;
267}
