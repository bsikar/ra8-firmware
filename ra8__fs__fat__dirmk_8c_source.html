54static void priv_pack_dot_entry(uint8_t* ent, uint32_t dots, uint32_t cluster)
55{
56 for (uint32_t i = 0; i < (uint32_t)k_ra8_fs_dir_entry_bytes; i++) {
57 ent[i] = 0;
58 }
59 for (uint32_t i = 0; i < (uint32_t)k_dir_name_field_len; i++) {
60 ent[i] = ' ';
61 }
62 for (uint32_t i = 0; i < dots; i++) {
63 ent[i] = '.';
64 }
65 ent[k_dir_off_attr] = k_ra8_fs_attr_directory;
66 priv_entry_set_cluster_size(ent, cluster, 0U);
67 /* "." and ".." are ordinary directory entries to every host, and every mkfs
68 * stamps them. Leaving them at zero would put an illegal date inside a
69 * directory this firmware had just created (#601). */
70 priv_fat_entry_stamp_create(ent);
71}

98static ra8_err_t
99priv_dir_cluster_init(const ra8_fs_mount_t* m, uint32_t new_cluster, uint32_t parent_cluster)
100{
101 uint8_t* const buf = priv_sec_walk();
102 priv_byte_fill(buf, 0U, priv_bps(m));
103 priv_pack_dot_entry(&buf[0], 1U, new_cluster);
104 priv_pack_dot_entry(&buf[k_ra8_fs_dir_entry_bytes], 2U, parent_cluster);
105 const uint64_t base = priv_cluster_to_lba(m, new_cluster);
106 ra8_err_t err = priv_write_sector(m, base, buf);
107 if (err != k_ra8_ok) {
108 return err;
109 }
110 for (uint32_t s = 1; s < m->sectors_per_cluster; s++) {
111 err = priv_write_sector(m, base + s, k_zero_sector);
112 if (err != k_ra8_ok) {
113 return err;
114 }
115 }
116 return k_ra8_ok;
117}

147RA8_INTERNAL
148static ra8_err_t priv_fat_mkdir(ra8_fs_mount_t* handle, const char* path)
149{
150 dir_loc_t parent = {};
151 const char* leaf = nullptr;
152 const ra8_err_t rerr = priv_resolve_parent(handle, path, &parent, &leaf);
153 if (rerr != k_ra8_ok) {
154 return rerr;
155 }
156 uint64_t lba = 0;
157 uint32_t off = 0;
158 uint8_t entry[k_ra8_fs_dir_entry_bytes] = {};
159 /* By long name as well as by 8.3: a second `mkdir("/Reading List")` has to
160 * be refused, and the alias probe behind `priv_dir_reserve()` would happily
161 * file a second copy as `READIN~2` if it were not. */
162 if (priv_dir_lookup_any(handle, &parent, leaf, &lba, &off, entry) == k_ra8_ok) {
163 return k_ra8_err_exists;
164 }
165 dir_insert_t plan = {};
166 ra8_err_t err = priv_dir_reserve(handle, &parent, leaf, &plan);
167 if (err != k_ra8_ok) {
168 return err;
169 }
170 uint32_t new_cluster = 0;
171 err = priv_alloc_eoc_cluster(handle, &new_cluster);
172 if (err != k_ra8_ok) {
173 return err;
174 }
175 const uint32_t parent_cluster = (parent.is_root != 0U) ? 0U : parent.cluster;
176 err = priv_dir_cluster_init(handle, new_cluster, parent_cluster);
177 if (err != k_ra8_ok) {
178 (void)priv_free_chain(handle, new_cluster);
179 return err;
180 }
181 uint8_t tmpl[k_ra8_fs_dir_entry_bytes] = {};
182 tmpl[k_dir_off_attr] = (uint8_t)k_ra8_fs_attr_directory;
183 priv_entry_set_cluster_size(tmpl, new_cluster, 0U);
184 /* Same reasoning as the file case: a zero FAT date is illegal, and the
185 * directory's own "." and ".." were stamped when the cluster was built, so
186 * the entry in the parent must agree with them (#601). */
187 priv_fat_entry_stamp_create(tmpl);
188 err = priv_dir_commit(handle, &plan, tmpl, &lba, &off);
189 if (err != k_ra8_ok) {
190 (void)priv_free_chain(handle, new_cluster);
191 return err;
192 }
193 return k_ra8_ok;
194}

288static dir_scan_t priv_rmdir_scan_sector(const ra8_fs_mount_t* m, const uint8_t* buf)
289{
290 for (uint32_t e = 0; e < priv_dir_eps(m); e++) {
291 const uint8_t* ent = &buf[(size_t)e * (size_t)k_ra8_fs_dir_entry_bytes];
292 if (ent[k_dir_off_name] == k_dir_marker_free_perm) {
293 return k_dir_scan_empty;
294 }
295 if (ent[k_dir_off_name] == k_dir_marker_free_used) {
296 continue;
297 }
298 if (ent[k_dir_off_attr] == k_ra8_fs_attr_lfn) {
299 continue;
300 }
301 if (ent[k_dir_off_name] == k_dir_marker_dot) {
302 continue;
303 }
304 return k_dir_scan_used;
305 }
306 return k_dir_scan_more;
307}

336RA8_INTERNAL
337static ra8_err_t priv_dir_is_empty(const ra8_fs_mount_t* m, uint32_t cluster, uint8_t* out_empty)
338{
339 const dir_loc_t loc = {.is_root = 0U, .cluster = cluster};
340 dir_walk_t w = {};
341 priv_dir_walk_init_loc(m, &loc, &w);
342 uint8_t eod = 0;
343 uint8_t* const buf = priv_sec_walk();
344 while (eod == 0U) {
345 ra8_err_t err = priv_read_sector(m, w.cur_lba, buf);
346 if (err != k_ra8_ok) {
347 return err;
348 }
349 const dir_scan_t verdict = priv_rmdir_scan_sector(m, buf);
350 if (verdict == k_dir_scan_used) {
351 *out_empty = 0U;
352 return k_ra8_ok;
353 }
354 if (verdict == k_dir_scan_empty) {
355 *out_empty = 1U;
356 return k_ra8_ok;
357 }
358 err = priv_dir_walk_next_sector(m, &w, &eod);
359 if (err != k_ra8_ok) {
360 return err;
361 }
362 }
363 *out_empty = 1U; /* chain exhausted with no occupant and no 0x00 marker */
364 return k_ra8_ok;
365}

398static ra8_err_t priv_rmdir_locate(const ra8_fs_mount_t* handle,
399 const char* path,
400 dir_target_t* out,
401 uint32_t* out_cluster)
402{
403 const char* leaf = nullptr;
404 const ra8_err_t rerr = priv_resolve_parent(handle, path, &out->parent, &leaf);
405 if (rerr != k_ra8_ok) {
406 return rerr;
407 }
408 if (leaf[0] == '\0') {
409 return k_ra8_err_invalid_arg; /* "/" or a trailing slash: the root itself */
410 }
411 const ra8_err_t err =
412 priv_dir_lookup_any(handle, &out->parent, leaf, &out->lba, &out->off, out->entry);
413 if (err != k_ra8_ok) {
414 return err;
415 }
416 if ((out->entry[k_dir_off_attr] & (uint8_t)k_ra8_fs_attr_directory) == 0U) {
417 return k_ra8_err_invalid_arg; /* a file: ra8_fs_unlink() is the verb for it */
418 }
419 *out_cluster = priv_entry_first_cluster(out->entry);
420 if (*out_cluster < (uint32_t)k_cluster_first_data) {
421 return k_ra8_err_protocol_error; /* a directory always owns a cluster */
422 }
423 return k_ra8_ok;
424}

458static ra8_err_t priv_fat_rmdir(const ra8_fs_mount_t* handle, const char* path)
459{
460 dir_target_t t = {};
461 uint32_t cluster = 0;
462 ra8_err_t err = priv_rmdir_locate(handle, path, &t, &cluster);
463 if (err != k_ra8_ok) {
464 return err;
465 }
466 uint8_t empty = 0;
467 err = priv_dir_is_empty(handle, cluster, &empty);
468 if (err != k_ra8_ok) {
469 return err;
470 }
471 if (empty == 0U) {
472 return k_ra8_err_not_empty;
473 }
474 err = priv_free_chain(handle, cluster);
475 if (err != k_ra8_ok) {
476 return err;
477 }
478 return priv_dir_erase_chain(handle, &t.parent, t.lba, t.off, &t.entry[k_dir_off_name]);
479}
