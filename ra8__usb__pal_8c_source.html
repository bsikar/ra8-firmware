306ra8_err_t ra8_usb_pal_init(ra8_usb_speed_t speed)
307{
308 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
309 return k_ra8_err_invalid_arg;
310 }
311 const ra8_err_t usb_err = ra8_usb_device_init(speed);
312 if (usb_err != k_ra8_ok) {
313 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
314 return k_ra8_err_hw_init_failed;
315 }
316
317 s_state.speed = speed;
318 s_state.state = k_ra8_usb_pal_state_detached;
319 s_state.event_fn = nullptr;
320 s_state.event_ctx = nullptr;
321 s_state.initialized = true;
322 internal_reset_eps();
323
324 const ra8_err_t att_err = ra8_usb_attach_handler(speed, internal_usb_event, nullptr);
325 if (att_err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
326 /* GCOVR_EXCL_START */
327 ra8_log_error_val(s_tag, "ra8_usb_attach_handler failed", (uint32_t)att_err);
328 s_state.initialized = false;
329 (void)ra8_usb_device_deinit(speed);
330 return k_ra8_err_hw_init_failed;
331 /* GCOVR_EXCL_STOP */
332 }
333
334 ra8_log_info(s_tag, "PAL ready");
335 return k_ra8_ok;
336}

452ra8_err_t ra8_usb_pal_ep_open(uint8_t ep_addr,
453 ra8_usb_pal_ep_dir_t dir,
454 ra8_usb_pal_ep_type_t type,
455 uint16_t max_packet)
456{
457 /* USB descriptor bEndpointAddress encodes direction in bit 7
458 * (0x80 = IN). Strip it so callers can pass either the descriptor
459 * form (e.g. 0x83) or the bare endpoint number (e.g. 3). */
460 ep_addr = (uint8_t)(ep_addr & (uint8_t)k_ra8_usb_pal_ep_addr_mask);
461 if (!s_state.initialized) {
462 return k_ra8_err_invalid_state;
463 }
464 if (ra8_usb_pal_internal_ep_out_of_range(ep_addr, (uint8_t)k_ra8_usb_pal_ep_max)) {
465 return k_ra8_err_invalid_arg;
466 }
467 if ((dir != k_ra8_usb_pal_ep_dir_out) && (dir != k_ra8_usb_pal_ep_dir_in)) {
468 return k_ra8_err_invalid_arg;
469 }
470 if ((type > k_ra8_usb_pal_ep_type_intr) || (max_packet == 0U) ||
471 (max_packet > k_ra8_usb_pal_xfer_max)) {
472 return k_ra8_err_invalid_arg;
473 }
474 ra8_usb_pal_ep_slot_t* slot = &s_state.eps[ep_addr];
475 slot->opened = true;
476 slot->dir = dir;
477 slot->type = type;
478 slot->max_packet = max_packet;
479 slot->head = 0U;
480 slot->tail = 0U;
481 slot->count = 0U;
482 for (uint16_t j = 0U; j < k_ra8_usb_pal_ring_slots; ++j) {
483 slot->ring[j].len = 0U;
484 }
485 return k_ra8_ok;
486}

516ra8_err_t ra8_usb_pal_ep_send(uint8_t ep_addr, const uint8_t* data, uint16_t len)
517{
518 /* Accept descriptor-shaped (0x80 | ep) and bare-number ep_addr. */
519 ep_addr = (uint8_t)(ep_addr & (uint8_t)k_ra8_usb_pal_ep_addr_mask);
520 if (!s_state.initialized) {
521 return k_ra8_err_invalid_state;
522 }
523 if (ra8_usb_pal_internal_ep_out_of_range(ep_addr, (uint8_t)k_ra8_usb_pal_ep_max)) {
524 return k_ra8_err_invalid_arg;
525 }
526 if ((len > k_ra8_usb_pal_xfer_max) || ((data == nullptr) && (len != 0U))) {
527 return (data == nullptr) ? k_ra8_err_null_ptr : k_ra8_err_invalid_arg;
528 }
529 ra8_usb_pal_ep_slot_t* slot = &s_state.eps[ep_addr];
530 if (!slot->opened) {
531 return k_ra8_err_invalid_state;
532 }
533 if (len > slot->max_packet) {
534 return k_ra8_err_invalid_arg;
535 }
536 if (slot->count >= k_ra8_usb_pal_ring_slots) {
537 return k_ra8_err_no_mem;
538 }
539 ra8_usb_pal_packet_t* pkt = &slot->ring[slot->tail];
540 if (len > 0U) {
541 internal_copy_bytes(pkt->data, data, len);
542 }
543 pkt->len = len;
544 slot->tail = (uint16_t)((slot->tail + 1U) % k_ra8_usb_pal_ring_slots);
545 ++slot->count;
546 if (s_state.event_fn != nullptr) {
547 s_state.event_fn(s_state.event_ctx, s_state.speed, k_ra8_usb_pal_event_ep_in);
548 }
549 return k_ra8_ok;
550}

582{
583 RA8_CHECK_NULL_PTR(out_buf, s_tag, "ep_recv: out_buf");
584 RA8_CHECK_NULL_PTR(inout_len, s_tag, "ep_recv: inout_len");
585 /* Accept descriptor-shaped (0x80 | ep) and bare-number ep_addr. */
586 ep_addr = (uint8_t)(ep_addr & (uint8_t)k_ra8_usb_pal_ep_addr_mask);
587 if (!s_state.initialized) {
588 return k_ra8_err_invalid_state;
589 }
590 if (ra8_usb_pal_internal_ep_out_of_range(ep_addr, (uint8_t)k_ra8_usb_pal_ep_max)) {
591 return k_ra8_err_invalid_arg;
592 }
593 if (*inout_len == 0U) {
594 return k_ra8_err_invalid_arg;
595 }
596 ra8_usb_pal_ep_slot_t* slot = &s_state.eps[ep_addr];
597 if (!slot->opened) {
598 return k_ra8_err_invalid_state;
599 }
600 if (slot->count == 0U) {
601 return k_ra8_err_no_data;
602 }
603 ra8_usb_pal_packet_t* pkt = &slot->ring[slot->head];
604 const uint16_t n = (pkt->len < *inout_len) ? pkt->len : *inout_len;
605 if (n > 0U) {
606 internal_copy_bytes(out_buf, pkt->data, n);
607 }
608 *inout_len = n;
609 pkt->len = 0U;
610 slot->head = (uint16_t)((slot->head + 1U) % k_ra8_usb_pal_ring_slots);
611 --slot->count;
612 return k_ra8_ok;
613}
