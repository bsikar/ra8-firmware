53static uint8_t
54priv_listdir_visit_sector(const uint8_t* buf, lfn_state_t* lfn, ra8_fs_listdir_cb_t cb, void* ctx)
55{
56 for (uint32_t e = 0; e < (uint32_t)k_dir_entries_per_sector; e++) {
57 const uint8_t* ent = &buf[(size_t)e * (size_t)k_ra8_fs_dir_entry_bytes];
58 if (ent[k_dir_off_name] == k_dir_marker_free_perm) {
59 return 1U;
60 }
61 if (ent[k_dir_off_name] == k_dir_marker_free_used) {
62 priv_lfn_reset(lfn); /* a deleted slot breaks any in-progress chain */
63 continue;
64 }
65 if (ent[k_dir_off_attr] == k_ra8_fs_attr_lfn) {
66 priv_lfn_add(lfn, ent);
67 continue;
68 }
69 if (ent[k_dir_off_name] == '.') {
70 priv_lfn_reset(lfn); /* synthetic "." / ".." -- not reported to the caller */
71 continue;
72 }
73 /* +1 for the NUL: a maximal 8.3 name "12345678.123" is 12 chars, and
74 * priv_83_to_str writes the terminator at index 12 -- without the slack
75 * that store overflows the buffer and corrupts the adjacent `ent` pointer,
76 * mismatching the LFN checksum so the long name is dropped (matches the
77 * `+ 1U` sizing already used in ra8_fs_fat_file.c). */
78 char short_name[(uint32_t)k_ra8_fs_short_name_len + 1U] = {};
79 priv_83_to_str(&ent[k_dir_off_name], short_name);
80 /* Report the VFAT long name when the preceding chain matches; else the 8.3. */
81 const char* lname = priv_lfn_name_for(lfn, &ent[k_dir_off_name]);
82 const uint32_t size = priv_rd32(&ent[k_dir_off_file_size]);
83 cb((lname != nullptr) ? lname : short_name, ent[k_dir_off_attr], size, ctx);
84 priv_lfn_reset(lfn);
85 }
86 return 0U;
87}

119ra8_err_t
120ra8_fs_listdir(ra8_fs_mount_t* handle, const char* path, ra8_fs_listdir_cb_t cb, void* ctx)
121{
122 if (handle == nullptr || cb == nullptr || path == nullptr) {
123 return k_ra8_err_null_ptr;
124 }
125 if (handle->in_use == 0U) {
126 return k_ra8_err_invalid_state;
127 }
128 if (handle->type == k_ra8_fs_type_exfat) {
129 /* exFAT directory listing is root-only for now. */
130 if (path[0] != '/') {
131 return k_ra8_err_not_supported;
132 }
133 if (path[1] != '\0') {
134 return k_ra8_err_not_supported;
135 }
136 return priv_exfat_listdir(handle, cb, ctx);
137 }
138 dir_loc_t loc = {};
139 const ra8_err_t rerr = priv_resolve_dir(handle, path, &loc);
140 if (rerr != k_ra8_ok) {
141 return rerr;
142 }
143 dir_walk_t w = {};
144 priv_dir_walk_init_loc(handle, &loc, &w);
145 lfn_state_t lfn = {}; /* persists across sectors -- LFN chains can straddle them */
146 priv_lfn_reset(&lfn);
147 uint8_t eod = 0;
148 uint8_t buf[k_ra8_fs_bytes_per_sector] = {};
149 while (eod == 0U) {
150 ra8_err_t err = priv_read_sector(handle, w.cur_lba, buf);
151 if (err != k_ra8_ok) {
152 return err;
153 }
154 if (priv_listdir_visit_sector(buf, &lfn, cb, ctx) != 0U) {
155 return k_ra8_ok;
156 }
157 err = priv_dir_walk_next_sector(handle, &w, &eod);
158 if (err != k_ra8_ok) {
159 return err;
160 }
161 }
162 return k_ra8_ok;
163}
