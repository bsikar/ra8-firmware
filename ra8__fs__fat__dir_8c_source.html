56static uint8_t priv_listdir_visit_sector(const ra8_fs_mount_t* m,
57 const uint8_t* buf,
58 lfn_state_t* lfn,
59 ra8_fs_listdir_cb_t cb,
60 void* ctx)
61{
62 for (uint32_t e = 0; e < priv_dir_eps(m); e++) {
63 const uint8_t* ent = &buf[(size_t)e * (size_t)k_ra8_fs_dir_entry_bytes];
64 if (ent[k_dir_off_name] == k_dir_marker_free_perm) {
65 return 1U;
66 }
67 if (ent[k_dir_off_name] == k_dir_marker_free_used) {
68 priv_lfn_reset(lfn); /* a deleted slot breaks any in-progress chain */
69 continue;
70 }
71 if (ent[k_dir_off_attr] == k_ra8_fs_attr_lfn) {
72 priv_lfn_add(lfn, ent);
73 continue;
74 }
75 if (ent[k_dir_off_name] == k_dir_marker_dot) {
76 priv_lfn_reset(lfn); /* synthetic "." / ".." -- not reported to the caller */
77 continue;
78 }
79 /* +1 for the NUL: a maximal 8.3 name "12345678.123" is 12 chars, and
80 * priv_83_to_str writes the terminator at index 12 -- without the slack
81 * that store overflows the buffer and corrupts the adjacent `ent` pointer,
82 * mismatching the LFN checksum so the long name is dropped (matches the
83 * `+ 1U` sizing already used in ra8_fs_fat_file.c). */
84 char short_name[(uint32_t)k_ra8_fs_short_name_len + 1U] = {};
85 /* The NTRes case flags come along: an entry this driver wrote for
86 * `data.log` stores `DATA LOG` with both flags set, and reporting it as
87 * `DATA.LOG` would lose the caller's own name for no reason (#600). */
88 priv_83_to_str(&ent[k_dir_off_name], ent[k_dir_off_ntres], short_name);
89 /* Report the VFAT long name when the preceding chain matches; else the 8.3.
90 * The chain is UTF-16 on disk and UTF-8 at this boundary, and a chain that
91 * does not convert -- an unpaired surrogate, which nothing conforming
92 * writes -- falls back to the alias rather than being reported under a name
93 * that would not re-open it (#606). */
94 char lname[k_lfn_utf8_cap] = {};
95 uint32_t lnunits = 0U;
96 const uint16_t* lunits = priv_lfn_units_for(lfn, &ent[k_dir_off_name], &lnunits);
97 uint8_t have_long = 0U;
98 if (lunits != nullptr) {
99 have_long = (priv_utf16_to_utf8(lunits, lnunits, lname, (uint32_t)k_lfn_utf8_cap) == k_ra8_ok)
100 ? 1U
101 : 0U;
102 }
103 const uint32_t size = priv_rd32(&ent[k_dir_off_file_size]);
104 cb((have_long != 0U) ? lname : short_name, ent[k_dir_off_attr], size, ctx);
105 priv_lfn_reset(lfn);
106 }
107 return 0U;
108}

143RA8_EXPECTS_LOCK("ra8_fs_lock")
144static ra8_err_t
145priv_listdir_locked(ra8_fs_mount_t* handle, const char* path, ra8_fs_listdir_cb_t cb, void* ctx)
146{
147 if (handle == nullptr || cb == nullptr || path == nullptr) {
148 return k_ra8_err_null_ptr;
149 }
150 if (handle->in_use == 0U) {
151 return k_ra8_err_invalid_state;
152 }
153 if (handle->type == k_ra8_fs_type_exfat) {
154 /* Any exFAT directory, not just the root (#605). A path that names nothing
155 * reports not_found and one that names a FILE reports invalid_arg, which is
156 * what the FAT side has always answered -- the two filesystems no longer
157 * disagree about what "list this path" means. */
158 exfat_dir_t dir = {};
159 const ra8_err_t xerr = priv_exfat_resolve_dir(handle, path, &dir);
160 if (xerr != k_ra8_ok) {
161 return xerr;
162 }
163 return priv_exfat_listdir(handle, &dir, cb, ctx);
164 }
165 dir_loc_t loc = {};
166 const ra8_err_t rerr = priv_resolve_dir(handle, path, &loc);
167 if (rerr != k_ra8_ok) {
168 return rerr;
169 }
170 dir_walk_t w = {};
171 priv_dir_walk_init_loc(handle, &loc, &w);
172 lfn_state_t lfn = {}; /* persists across sectors -- LFN chains can straddle them */
173 priv_lfn_reset(&lfn);
174 uint8_t eod = 0;
175 uint8_t* const buf = priv_sec_walk();
176 while (eod == 0U) {
177 ra8_err_t err = priv_read_sector(handle, w.cur_lba, buf);
178 if (err != k_ra8_ok) {
179 return err;
180 }
181 if (priv_listdir_visit_sector(handle, buf, &lfn, cb, ctx) != 0U) {
182 return k_ra8_ok;
183 }
184 err = priv_dir_walk_next_sector(handle, &w, &eod);
185 if (err != k_ra8_ok) {
186 return err;
187 }
188 }
189 return k_ra8_ok;
190}

222RA8_INTERNAL
223static ra8_err_t priv_unlink_locate(const ra8_fs_mount_t* handle,
224 const char* path,
225 dir_target_t* out,
226 uint32_t* out_cluster)
227{
228 const char* leaf = nullptr;
229 const ra8_err_t rerr = priv_resolve_parent(handle, path, &out->parent, &leaf);
230 if (rerr != k_ra8_ok) {
231 return rerr;
232 }
233 const ra8_err_t err =
234 priv_dir_lookup_any(handle, &out->parent, leaf, &out->lba, &out->off, out->entry);
235 if (err != k_ra8_ok) {
236 return err;
237 }
238 /* The lookup matches on the name alone, so a directory matches like any
239 * other entry. Freeing its chain orphans every file inside it -- their
240 * clusters stay allocated in the FAT with nothing referencing them, which
241 * fsck.fat reports as lost clusters (#604). */
242 if ((out->entry[k_dir_off_attr] & (uint8_t)k_ra8_fs_attr_directory) != 0U) {
243 return k_ra8_err_invalid_arg; /* a directory: ra8_fs_rmdir() is the verb */
244 }
245 /* Honor the read-only attribute (#681): a file a host marked read-only is not
246 * deletable through the ordinary path, matching DOS/Windows `del`. Refused
247 * before priv_free_chain() touches the FAT, so a denied unlink changes nothing. */
248 if ((out->entry[k_dir_off_attr] & (uint8_t)k_ra8_fs_attr_read_only) != 0U) {
249 return k_ra8_err_access_denied;
250 }
251 *out_cluster = priv_entry_first_cluster(out->entry);
252 return k_ra8_ok;
253}

286RA8_EXPECTS_LOCK("ra8_fs_lock")
287static ra8_err_t priv_unlink_locked(ra8_fs_mount_t* handle, const char* path)
288{
289 if (handle == nullptr || path == nullptr) {
290 return k_ra8_err_null_ptr;
291 }
292 if (handle->in_use == 0U) {
293 return k_ra8_err_invalid_state;
294 }
295 if (handle->type == k_ra8_fs_type_exfat) {
296 return priv_exfat_unlink(handle, path);
297 }
298 dir_target_t t = {};
299 uint32_t cluster = 0;
300 const ra8_err_t lerr = priv_unlink_locate(handle, path, &t, &cluster);
301 if (lerr != k_ra8_ok) {
302 return lerr;
303 }
304 if (cluster >= (uint32_t)k_cluster_first_data) {
305 const ra8_err_t ferr = priv_free_chain(handle, cluster);
306 if (ferr != k_ra8_ok) {
307 return ferr;
308 }
309 }
310 /* The whole chain, not only the 8.3 entry: clearing one slot left the file's
311 * long-name entries on disk with a checksum nothing matched, which fsck.fat
312 * and chkdsk both report as orphans (#600). */
313 return priv_dir_erase_chain(handle, &t.parent, t.lba, t.off, &t.entry[k_dir_off_name]);
314}

419RA8_INTERNAL
420static ra8_err_t
421priv_fat_rename(const ra8_fs_mount_t* handle, const char* old_path, const char* new_path)
422{
423 dir_target_t t = {};
424 const char* ol = nullptr;
425 const char* nl = nullptr;
426 ra8_err_t perr = priv_rename_prepare(handle, old_path, new_path, &t.parent, &ol, &nl);
427 if (perr != k_ra8_ok) {
428 return perr;
429 }
430 uint64_t dup_lba = 0U;
431 uint32_t dup_off = 0U;
432 uint8_t dup[k_ra8_fs_dir_entry_bytes] = {};
433 if (priv_dir_lookup_any(handle, &t.parent, nl, &dup_lba, &dup_off, dup) == k_ra8_ok) {
434 return k_ra8_err_exists;
435 }
436 ra8_err_t err = priv_dir_lookup_any(handle, &t.parent, ol, &t.lba, &t.off, t.entry);
437 if (err != k_ra8_ok) {
438 return err;
439 }
440 /* Honor the read-only attribute (#681): a read-only file is refused before any
441 * new entry is filed or the old chain is erased, so the directory is untouched. */
442 if ((t.entry[k_dir_off_attr] & (uint8_t)k_ra8_fs_attr_read_only) != 0U) {
443 return k_ra8_err_access_denied;
444 }
445 dir_insert_t plan = {};
446 err = priv_dir_reserve(handle, &t.parent, nl, &plan);
447 if (err != k_ra8_ok) {
448 return err;
449 }
450 /* Access date only, deliberately. A rename changes the name, not the bytes,
451 * so advancing DIR_WrtDate would tell every rsync, backup and "newest image"
452 * OTA heuristic that the contents changed -- the same class of false signal
453 * #601 exists to remove, only inverted. FAT has no metadata-change field, so
454 * the access date is the honest record that the entry was touched. It goes on
455 * the template BEFORE the commit, because since #600 a rename writes a new
456 * entry rather than editing the old one in place. */
457 priv_fat_entry_stamp_access(t.entry);
458 uint64_t new_lba = 0U;
459 uint32_t new_off = 0U;
460 err = priv_dir_commit(handle, &plan, t.entry, &new_lba, &new_off);
461 if (err != k_ra8_ok) {
462 return err;
463 }
464 return priv_dir_erase_chain(handle, &t.parent, t.lba, t.off, &t.entry[k_dir_off_name]);
465}
