255RA8_INTERNAL
256static void internal_put_pixel(uint8_t* dst,
257 size_t stride,
258 ra8_gfx_format_t format,
259 size_t x,
260 size_t y,
261 uint32_t color)
262{
263 uint8_t* p = dst + (y * stride) + (x * (size_t)internal_bpp(format));
264 switch (format) {
265 case k_ra8_gfx_format_rgb565: {
266 const uint16_t v = s_gfx_text_pack_565(color);
267 p[k_idx_r] = (uint8_t)(v & k_mask_byte);
268 p[k_idx_g] = (uint8_t)((v >> k_glyph_bits_per_byte) & k_mask_byte);
269 break;
270 }
271 case k_ra8_gfx_format_rgb888: {
272 p[k_idx_r] = internal_color_r(color);
273 p[k_idx_g] = internal_color_g(color);
274 p[k_idx_b] = internal_color_b(color);
275 break;
276 }
277 case k_ra8_gfx_format_argb8888: {
278 p[k_idx_argb_b] = internal_color_b(color);
279 p[k_idx_argb_g] = internal_color_g(color);
280 p[k_idx_argb_r] = internal_color_r(color);
281 p[k_idx_argb_a] = internal_color_a(color);
282 break;
283 }
284 }

305RA8_INTERNAL
306static uint32_t
307internal_get_pixel(const uint8_t* src, size_t stride, ra8_gfx_format_t format, size_t x, size_t y)
308{
309 const uint8_t* p = src + (y * stride) + (x * (size_t)internal_bpp(format));
310 switch (format) {
311 case k_ra8_gfx_format_rgb565: {
312 const uint16_t v = (uint16_t)(p[k_idx_r] | ((uint16_t)p[k_idx_g] << k_glyph_bits_per_byte));
313 const uint32_t r = ((uint32_t)(v >> k_565_r_shift_out) & k_565_r_mask) << k_565_r_shift_in;
314 const uint32_t g = ((uint32_t)(v >> k_565_g_shift_out) & k_565_g_mask) << k_565_g_shift_in;
315 const uint32_t b = ((uint32_t)v & k_565_b_mask) << k_565_b_shift_in;
316 return (r << k_shift_red) | (g << k_shift_green) | (b << k_shift_blue);
317 }
318 case k_ra8_gfx_format_rgb888: {
319 const uint32_t r = p[k_idx_r];
320 const uint32_t g = p[k_idx_g];
321 const uint32_t b = p[k_idx_b];
322 return (r << k_shift_red) | (g << k_shift_green) | (b << k_shift_blue);
323 }
324 case k_ra8_gfx_format_argb8888: {
325 const uint32_t bb = p[k_idx_argb_b];
326 const uint32_t gg = p[k_idx_argb_g];
327 const uint32_t rr = p[k_idx_argb_r];
328 const uint32_t aa = p[k_idx_argb_a];
329 return (aa << k_shift_alpha) | (rr << k_shift_red) | (gg << k_shift_green) | bb;
330 }
331 }
332 return 0U;
333}

376RA8_INTERNAL
377static inline void internal_fill_565(uint8_t* p, size_t count, uint8_t lo, uint8_t hi)
378{
379 if (lo == hi) {
380 /* Symmetric packed colour (e.g. white paper 0xFFFF, black 0x0000): every
381 * byte of the run is the same value, so the optimised libc memset replaces
382 * the byte loop. Byte-identical -- both write `lo` to every byte. This is the
383 * common full-screen clear, so it is the hot path. */
384 (void)memset(p, (int)lo, count * (size_t)k_rgb565_bpp);
385 return;
386 }
387 for (size_t i = 0; i < count; i++) {
388 *p++ = lo;
389 *p++ = hi;
390 }
391}

415RA8_INTERNAL
416static void internal_fill_rect_565(int32_t x, int32_t y, int32_t w, int32_t h, uint32_t color)
417{
418 /* Clip against the active clip rectangle (which is itself within the
419 * framebuffer), so this is both the on-screen clip and the dirty-region clip.
420 * With the default full-framebuffer clip it reduces to min/max against the
421 * framebuffer, i.e. byte-identical to the unclipped fill. */
422 const int32_t cl = s_gfx_text_state.clip_x0;
423 const int32_t ct = s_gfx_text_state.clip_y0;
424 const int32_t cr = s_gfx_text_state.clip_x1;
425 const int32_t cb = s_gfx_text_state.clip_y1;
426 const int32_t x0 = (x >= cl) ? x : cl;
427 const int32_t y0 = (y >= ct) ? y : ct;
428 /* Right/bottom edges in 64-bit so x+w / y+h cannot overflow int32 for a
429 * pathological caller; then clamped to the clip. */
430 const int64_t xr = (int64_t)x + (int64_t)w;
431 const int64_t yr = (int64_t)y + (int64_t)h;
432 const int32_t x1 = (xr < (int64_t)cr) ? (int32_t)xr : cr;
433 const int32_t y1 = (yr < (int64_t)cb) ? (int32_t)yr : cb;
434 if ((x1 <= x0) || (y1 <= y0)) {
435 return; /* fully clipped -- nothing to fill. */
436 }
437 const uint16_t v = s_gfx_text_pack_565(color);
438 const uint8_t lo = (uint8_t)(v & (uint16_t)k_mask_byte);
439 const uint8_t hi = (uint8_t)((v >> k_glyph_bits_per_byte) & (uint16_t)k_mask_byte);
440 const size_t bpp = (size_t)s_gfx_text_state.bpp;
441 const size_t stride = (size_t)s_gfx_text_state.width * bpp;
442 const size_t count = (size_t)(x1 - x0);
443 for (int32_t row = y0; row < y1; row++) {
444 internal_fill_565(s_gfx_text_state.fb + ((size_t)row * stride) + ((size_t)x0 * bpp),
445 count,
446 lo,
447 hi);
448 }
449}

524{

550ra8_err_t ra8_gfx_clear(uint32_t color)
551{
552 if (!s_gfx_text_state.initialized) {
553 return k_ra8_err_not_initialized;
554 }
555 /* "Clear" fills the active clip region (the whole framebuffer by default).
556 * For RGB565 this is the span-fill path -- pack once, memset each row when the
557 * bytes are symmetric -- restricted to the clip, so an incremental repaint
558 * only touches the damaged area. With the default clip the rect is the whole
559 * framebuffer and the bytes written are identical to the old contiguous fill. */
560 if (s_gfx_text_state.format == k_ra8_gfx_format_rgb565) {
561 internal_fill_rect_565(0,
562 0,
563 (int32_t)s_gfx_text_state.width,
564 (int32_t)s_gfx_text_state.height,
565 color);
566 return k_ra8_ok;
567 }
568 for (int32_t y = s_gfx_text_state.clip_y0; y < s_gfx_text_state.clip_y1; y++) {
569 for (int32_t x = s_gfx_text_state.clip_x0; x < s_gfx_text_state.clip_x1; x++) {
570 internal_put_pixel(s_gfx_text_state.fb,
571 (size_t)s_gfx_text_state.width * (size_t)s_gfx_text_state.bpp,
572 s_gfx_text_state.format,
573 (size_t)x,
574 (size_t)y,
575 color);
576 }
577 }
578 return k_ra8_ok;
579}

581ra8_err_t ra8_gfx_set_clip(int32_t x, int32_t y, int32_t w, int32_t h)
582{
583 if (!s_gfx_text_state.initialized) {
584 return k_ra8_err_not_initialized;
585 }
586 const int32_t fbw = (int32_t)s_gfx_text_state.width;
587 const int32_t fbh = (int32_t)s_gfx_text_state.height;
588 /* Clamp the top-left into [0, fb]; the bottom-right into [top-left, fb]. The
589 * edges are computed in 64-bit so x+w / y+h cannot overflow int32, and a
590 * non-positive size or fully off-screen rect collapses to an empty clip
591 * (x1==x0 or y1==y0), which makes every subsequent draw a no-op. */
592 int32_t x0 = (x > 0) ? x : 0;
593 int32_t y0 = (y > 0) ? y : 0;
594 if (x0 > fbw) {
595 x0 = fbw;
596 }
597 if (y0 > fbh) {
598 y0 = fbh;
599 }
600 const int64_t xr = (int64_t)x + (int64_t)w;
601 const int64_t yr = (int64_t)y + (int64_t)h;
602 int32_t x1 = (xr < (int64_t)fbw) ? (int32_t)xr : fbw;
603 int32_t y1 = (yr < (int64_t)fbh) ? (int32_t)yr : fbh;
604 if (x1 < x0) {
605 x1 = x0;
606 }
607 if (y1 < y0) {
608 y1 = y0;
609 }
610 s_gfx_text_state.clip_x0 = x0;
611 s_gfx_text_state.clip_y0 = y0;
612 s_gfx_text_state.clip_x1 = x1;
613 s_gfx_text_state.clip_y1 = y1;
614 return k_ra8_ok;
615}

701 int32_t w,

759ra8_err_t ra8_gfx_blit_gray8(const uint8_t* src, int32_t w, int32_t h, int32_t dst_x, int32_t dst_y)
760{
761 if (!s_gfx_text_state.initialized) {
762 return k_ra8_err_not_initialized;
763 }
764 if ((src == nullptr) || (w <= 0) || (h <= 0)) {
765 return k_ra8_err_invalid_arg;
766 }
767 /* Non-panel formats keep the exact per-pixel path; the RGB565 panel resolves
768 * the clip once and runs a tight row loop. */
769 if (s_gfx_text_state.format != k_ra8_gfx_format_rgb565) {
770 internal_blit_gray8_slow(src, w, h, dst_x, dst_y);
771 return k_ra8_ok;
772 }
773 /* Clip the block to the clip rectangle ONCE (always within the framebuffer). */
774 const int32_t x0 = (dst_x > s_gfx_text_state.clip_x0) ? dst_x : s_gfx_text_state.clip_x0;
775 const int32_t y0 = (dst_y > s_gfx_text_state.clip_y0) ? dst_y : s_gfx_text_state.clip_y0;
776 const int32_t x1 =
777 ((dst_x + w) < s_gfx_text_state.clip_x1) ? (dst_x + w) : s_gfx_text_state.clip_x1;
778 const int32_t y1 =
779 ((dst_y + h) < s_gfx_text_state.clip_y1) ? (dst_y + h) : s_gfx_text_state.clip_y1;
780 if ((x0 < x1) && (y0 < y1)) {
781 internal_blit_gray8_565(src, w, dst_x, dst_y, x0, y0, x1, y1);
782 }
783 return k_ra8_ok;
784}

786ra8_err_t ra8_gfx_line(int32_t x0, int32_t y0, int32_t x1, int32_t y1, uint32_t color)
787{
788 if (!s_gfx_text_state.initialized) {
789 return k_ra8_err_not_initialized;
790 }
791 /* Bresenham. */
792 int32_t dx = (x1 >= x0) ? (x1 - x0) : (x0 - x1);
793 int32_t dy = (y1 >= y0) ? -(y1 - y0) : -(y0 - y1);
794 const int32_t sx = (x0 < x1) ? 1 : -1;
795 const int32_t sy = (y0 < y1) ? 1 : -1;
796 int32_t err = dx + dy;
797 int32_t x = x0;
798 int32_t y = y0;
799 /* Bound the iteration count statically (NASA Rule 2). */
800 const int32_t max_iters = (int32_t)(k_ra8_gfx_max_dim * 2);
801 for (int32_t i = 0; i < max_iters; i++) {
802 s_gfx_text_plot(x, y, color);
803 if ((x == x1) && (y == y1)) {
804 break;
805 }
806 const int32_t e2 = err * 2;
807 if (e2 >= dy) {
808 err += dy;
809 x += sx;
810 }
811 if (e2 <= dx) {
812 err += dx;
813 y += sy;
814 }
815 }
816 return k_ra8_ok;
817}

895ra8_err_t ra8_gfx_circle(int32_t cx, int32_t cy, int32_t r, uint32_t color, bool filled)
896{
897 if (!s_gfx_text_state.initialized) {
898 return k_ra8_err_not_initialized;
899 }
900 if (r < 0) {
901 return k_ra8_err_invalid_arg;
902 }
903 /* Midpoint circle algorithm. */
904 int32_t x = r;
905 int32_t y = 0;
906 int32_t err = 1 - r;
907 /* Static loop bound. */
908 const int32_t max_iters = (int32_t)k_ra8_gfx_max_dim;
909 for (int32_t i = 0; i < max_iters; i++) {
910 if (filled) {
911 internal_circle_filled_step(cx, cy, x, y, color);
912 } else {
913 internal_circle_outline_step(cx, cy, x, y, color);
914 }
915 if (x <= y) {
916 break;
917 }
918 y++;
919 if (err < 0) {
920 err += (2 * y) + 1;
921 } else {
922 x--;
923 err += (2 * (y - x)) + 1;
924 }
925 }
926 return k_ra8_ok;
927}

929ra8_err_t ra8_gfx_blit(const void* src_buf,
930 uint16_t src_w,
931 uint16_t src_h,
932 ra8_gfx_format_t src_format,
933 int32_t dst_x,
934 int32_t dst_y)
935{
936 if (src_buf == nullptr) {
937 return k_ra8_err_null_ptr;
938 }
939 if (!s_gfx_text_state.initialized) {
940 return k_ra8_err_not_initialized;
941 }
942 if ((src_w == 0) || (src_h == 0) || !internal_format_ok(src_format)) {
943 return k_ra8_err_invalid_arg;
944 }
945 const uint8_t* src = (const uint8_t*)src_buf;
946 const size_t src_stride = (size_t)src_w * (size_t)internal_bpp(src_format);
947 for (uint32_t row = 0; row < src_h; row++) {
948 for (uint32_t col = 0; col < src_w; col++) {
949 const uint32_t color = internal_get_pixel(src, src_stride, src_format, col, row);
950 s_gfx_text_plot(dst_x + (int32_t)col, dst_y + (int32_t)row, color);
951 }
952 }
953 return k_ra8_ok;
954}
