149/* Issue ACMD41 with the host capacity-support bit set and loop -- see implementation for details. */
150RA8_INTERNAL
151static ra8_err_t internal_run_acmd41(uint8_t instance, uint32_t* out_ocr)
152{
153 RA8_CHECK_NULL_PTR(out_ocr, s_tag, "ocr");
154
155 uint32_t rsp[4] = {0U, 0U, 0U, 0U};
156 uint32_t ocr = 0U;
157 uint8_t done = 0U;
158 for (uint32_t i = 0U; i < (uint32_t)k_ra8_sdcard_init_retry_max; ++i) {
159 /* HUM Ch 47.2.1 "SD_CMD : Command Type Register" p 3123 */
160 /* CMD55 prefix is required before every ACMDxx. */
161 const ra8_err_t e55 =
162 ra8_sdhi_send_command(instance, (uint32_t)k_ra8_sdcard_cmd55_app_cmd, 0U, rsp);
163 if (e55 != k_ra8_ok) {
164 return e55;
165 }
166 /* HUM Ch 47.2.2 "SD_ARG : SD Command Argument" p 3124 */
167 /* HCS=1 (host supports SDHC/SDXC) | voltage window 2.7-3.6 V. */
168 const uint32_t arg =
169 (uint32_t)k_ra8_sdcard_acmd41_hcs | (uint32_t)k_ra8_sdcard_acmd41_voltage_win;
170 const ra8_err_t e41 =
171 ra8_sdhi_send_command(instance, (uint32_t)k_ra8_sdcard_acmd41_send_op_cond, arg, rsp);
172 if (e41 != k_ra8_ok) {
173 return e41;
174 }
175 ocr = rsp[0];
176 if ((ocr & (uint32_t)k_ra8_sdcard_ocr_busy_done_mask) != 0U) {
177 done = 1U;
178 break;
179 }
180 }
181 if (done == 0U) {
182 return k_ra8_err_hw_init_failed;
183 }
184 *out_ocr = ocr;
185 return k_ra8_ok;
186}

211RA8_INTERNAL
212static ra8_err_t internal_decode_csd(const uint32_t* rsp, uint32_t* out_blocks)
213{
214 RA8_CHECK_NULL_PTR(rsp, s_tag, "rsp");
215 RA8_CHECK_NULL_PTR(out_blocks, s_tag, "blocks");
216
217 /* CSD_STRUCTURE is rsp[3][31:30]. SDHI shifts the response left by
218 * 8 (CRC drop), so the architectural CSD[127:126] lands at
219 * rsp[3][31:30] -- the same place FSP r_sdhi.c reads it from. */
220 const uint8_t csd_structure = (uint8_t)((rsp[3] >> k_sd_csd_struct_shift) & 0x3U);
221 if (csd_structure == 1U) {
222 /* CSD v2 (SDHC/SDXC): C_SIZE is bits [69:48], i.e. rsp[1][29:8]
223 * once you account for the CRC shift. The 22-bit field needs the
224 * low 6 bits of rsp[2] OR'd with the high 16 bits of rsp[1]. */
225 const uint32_t c_size =
226 (((rsp[2] & k_sd_csize_v2_lo_mask) << 16U) | ((rsp[1] >> 16U) & k_sd_csize_v2_mid_mask)) &
227 k_sd_csize_v2_mask;
228 *out_blocks = (c_size + 1U) * (uint32_t)k_ra8_sdcard_csd_v2_blocks_per_unit;
229 return k_ra8_ok;
230 }
231 if (csd_structure == 0U) {
232 /* CSD v1 (SDSC): READ_BL_LEN [83:80], C_SIZE [73:62],
233 * C_SIZE_MULT [49:47]. Compute capacity in bytes then divide by
234 * 512 to land on the block count. */
235 const uint8_t read_bl_len = (uint8_t)((rsp[2] >> 16U) & k_sd_read_bl_mask);
236 const uint16_t c_size =
237 (uint16_t)((((rsp[2] & 0x00000003UL) << k_sd_csize_v1_shift) |
238 ((rsp[1] >> k_sd_csize_v1_lo_shift) & k_sd_csize_v1_lo_mask)) &
239 k_sd_csize_v1_mask);
240 const uint8_t c_size_mult = (uint8_t)((rsp[1] >> k_sd_csize_mult_shift) & k_sd_csize_mult_mask);
241 const uint32_t mult = 1U << ((uint32_t)c_size_mult + 2U);
242 const uint32_t block_len = 1U << (uint32_t)read_bl_len;
243 const uint32_t blocknr = ((uint32_t)c_size + 1U) * mult;
244 /* Convert to 512-byte sectors. */
245 *out_blocks = (blocknr * block_len) / k_sd_block_size_bytes;
246 return k_ra8_ok;
247 }
248 return k_ra8_err_hw_init_failed;
249}

274RA8_INTERNAL
275static ra8_err_t internal_sdcard_identify(uint8_t instance)
276{
277 uint32_t rsp[4] = {0U, 0U, 0U, 0U};
278
279 /* HUM Ch 47.2.1 "SD_CMD : Command Type Register" p 3123 */
280 /* CMD0 GO_IDLE_STATE -- no response, but send_command still polls
281 * RSPEND so the host stays in lockstep with the bus state machine. */
282 const ra8_err_t e0 =
283 ra8_sdhi_send_command(instance, (uint32_t)k_ra8_sdcard_cmd0_go_idle, 0U, rsp);
284 RA8_RETURN_ON_ERROR(e0, s_tag, "cmd0"); /* GCOVR_EXCL_BR_LINE */
285
286 /* HUM Ch 47.2.5 "SD_RSP10 : Response Register 10" p 3132 */
287 /* CMD8 SEND_IF_COND -- echoes the low 12 bits of the argument back
288 * in R7. Mismatch means a v1 (no-CMD8) card or no card at all. */
289 const ra8_err_t e8 = ra8_sdhi_send_command(instance,
290 (uint32_t)k_ra8_sdcard_cmd8_send_if_cond,
291 (uint32_t)k_ra8_sdcard_cmd8_pattern,
292 rsp);
293 RA8_RETURN_ON_ERROR(e8, s_tag, "cmd8"); /* GCOVR_EXCL_BR_LINE */
294 if ((rsp[0] & (uint32_t)k_ra8_sdcard_cmd8_pattern_mask) !=
295 ((uint32_t)k_ra8_sdcard_cmd8_pattern & (uint32_t)k_ra8_sdcard_cmd8_pattern_mask)) {
296 return k_ra8_err_hw_init_failed;
297 }
298 return k_ra8_ok;
299}

369static ra8_err_t
370internal_sdcard_publish_and_select(uint8_t instance, uint16_t* out_rca, uint32_t* out_blocks)
371{
372 uint16_t rca = 0U;
373 const ra8_err_t rca_err = internal_sdcard_publish_rca(instance, &rca);
374 if (rca_err != k_ra8_ok) {
375 return rca_err;
376 }
377 const uint32_t rca_arg = ((uint32_t)rca) << 16U;
378 uint32_t rsp[4] = {0U, 0U, 0U, 0U};
379
380 /* CMD9 SEND_CSD -- arg = RCA<<16; R2 holds the 128-bit CSD. */
381 const ra8_err_t e9 =
382 ra8_sdhi_send_command(instance, (uint32_t)k_ra8_sdcard_cmd9_send_csd, rca_arg, rsp);
383 RA8_RETURN_ON_ERROR(e9, s_tag, "cmd9"); /* GCOVR_EXCL_BR_LINE */
384
385 const ra8_err_t dec_err = internal_decode_csd(rsp, out_blocks);
386 if (dec_err != k_ra8_ok) {
387 return dec_err;
388 }
389
390 /* CMD7 SELECT_CARD -- arg = RCA<<16 to put this card into TRAN. */
391 const ra8_err_t e7 =
392 ra8_sdhi_send_command(instance, (uint32_t)k_ra8_sdcard_cmd7_select_card, rca_arg, rsp);
393 RA8_RETURN_ON_ERROR(e7, s_tag, "cmd7"); /* GCOVR_EXCL_BR_LINE */
394
395 *out_rca = rca;
396 return k_ra8_ok;

501RA8_INTERNAL
502static ra8_err_t internal_sdcard_card_online(uint8_t instance,
503 uint16_t* out_rca,
504 uint32_t* out_blocks,
505 uint8_t* out_high_capacity)
506{
507 /* HUM Ch 47.1 "SDHI Overview" p 3122 */ /* module bring-up first. */
508 const ra8_err_t hw_err = ra8_sdhi_init(instance);
509 RA8_RETURN_ON_ERROR(hw_err, s_tag, "sdhi_init"); /* GCOVR_EXCL_BR_LINE */
510
511 const ra8_err_t id_err = internal_sdcard_identify(instance);
512 if (id_err != k_ra8_ok) {
513 (void)ra8_sdhi_deinit(instance);
514 return id_err;
515 }
516
517 /* ACMD41 op-cond loop -- card moves from idle to ready. */
518 uint32_t ocr = 0U;
519 const ra8_err_t e41 = internal_run_acmd41(instance, &ocr);
520 if (e41 != k_ra8_ok) {
521 (void)ra8_sdhi_deinit(instance);
522 return e41;
523 }
524 *out_high_capacity = ((ocr & (uint32_t)k_ra8_sdcard_ocr_ccs_mask) != 0U) ? 1U : 0U;
525
526 const ra8_err_t pub_err = internal_sdcard_publish_and_select(instance, out_rca, out_blocks);
527 if (pub_err != k_ra8_ok) {
528 (void)ra8_sdhi_deinit(instance);
529 return pub_err;
530 }
531 return k_ra8_ok;
532}

534ra8_err_t ra8_sdcard_init(const ra8_sdcard_cfg_t* cfg)
535{
536 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg");
537 if (s_sdcard.initialized != 0U) {
538 return k_ra8_err_invalid_state;
539 }
540
541 uint16_t rca = 0U;
542 uint32_t blocks = 0U;
543 uint8_t high_capacity = 0U;
544 const ra8_err_t on_err =
545 internal_sdcard_card_online(cfg->instance, &rca, &blocks, &high_capacity);
546 if (on_err != k_ra8_ok) {
547 return on_err;
548 }
549
550 /* Optional bus-width widening (best-effort, never fatal) while still
551 * at the identification clock, before the default-speed clock bump. */
552 internal_sdcard_negotiate_width(cfg->instance, cfg->bus_width, rca);
553
554 /* HUM Ch 47.2.18 "SD_CLK_CTRL : SD Clock Control Register" p 3138 */
555 /* Bump the bus from the 400 kHz identification rate to default
556 * speed. The SDR50 / DDR50 negotiation lives in the future
557 * ::ra8_sdcard_set_speed extension. */
558 const ra8_err_t eclk = ra8_sdhi_set_clock(cfg->instance, (uint32_t)k_ra8_sdcard_default_clk_div);
559 RA8_RETURN_ON_ERROR(eclk, s_tag, "set_clock"); /* GCOVR_EXCL_BR_LINE */
560
561 s_sdcard.instance = cfg->instance;
562 s_sdcard.rca = rca;
563 s_sdcard.capacity_blocks = blocks;
564 s_sdcard.type = internal_classify(high_capacity, blocks);
565 s_sdcard.initialized = 1U;
566
567 ra8_log_info_val(s_tag, "sdcard init blocks", blocks);
568 return k_ra8_ok;
569}
