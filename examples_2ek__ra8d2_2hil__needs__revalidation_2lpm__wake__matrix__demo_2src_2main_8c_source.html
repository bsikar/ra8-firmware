358void main(void)
359{
360 internal_lpm_wake_setup_or_halt();
361 ra8_isr_globals_enable();
362
363 /* Boot banner -- HIL gate. */
364 (void)ra8_board_uart_console_write(s_lpm_wake_boot_msg,
365 (size_t)(sizeof(s_lpm_wake_boot_msg) - 1U));
366
367 if (internal_lpm_wake_walk_wupen0() != k_ra8_ok) {
368 internal_lpm_wake_panic_halt();
369 }
370 if (internal_lpm_wake_walk_wupen1() != k_ra8_ok) {
371 internal_lpm_wake_panic_halt();
372 }
373 if (internal_lpm_wake_disarm_all() != k_ra8_ok) {
374 internal_lpm_wake_panic_halt();
375 }
376
377 /* Final-state banner -- emit so the operator sees the walk
378 * completed even when no debugger is attached. */
379 (void)ra8_board_uart_console_write(s_lpm_wake_done_msg,
380 (size_t)(sizeof(s_lpm_wake_done_msg) - 1U));
381
382 (void)ra8_board_led_on(k_ra8_board_led1);
383
384 while (1) {
385 __asm__ volatile("wfi");
386 }
