199static void flash_journal_pack(uint32_t counter, uint8_t* rec)
200{
201 for (uint8_t i = 0U; i < (uint8_t)k_journal_counter_bytes; i++) {
202 rec[i] =
203 (uint8_t)((counter >> ((uint32_t)k_journal_byte_shift * i)) & (uint32_t)k_journal_byte_mask);
204 }
205 for (uint8_t i = (uint8_t)k_journal_counter_bytes; i < (uint8_t)k_journal_record_bytes; i++) {
206 rec[i] = i;
207 }
208}

216static uint32_t flash_journal_unpack(const uint8_t* rec)
217{
218 uint32_t v = 0U;
219 for (uint8_t i = 0U; i < (uint8_t)k_journal_counter_bytes; i++) {
220 v |= ((uint32_t)rec[i]) << ((uint32_t)k_journal_byte_shift * i);
221 }
222 return v;

241[[nodiscard]] static ra8_err_t flash_journal_round_trip(uint32_t counter, uint32_t* echoed)
242{
243 uint8_t rec[k_journal_record_bytes];
244 uint8_t back[k_journal_record_bytes];
245 g_fj_last_counter = counter;
246 flash_journal_pack(counter, rec);
247 if (ra8_xspi_flash_erase_sector((uint8_t)k_journal_xspi_instance,
248 (uint32_t)k_journal_record_addr) != k_ra8_ok) {
249 g_fj_last_step = k_fj_step_erase_failed;
250 return k_ra8_err_hw_error;
251 }
252 g_fj_last_step = k_fj_step_erase_ok;
253 if (ra8_xspi_flash_program((uint8_t)k_journal_xspi_instance,
254 (uint32_t)k_journal_record_addr,
255 rec,
256 (uint32_t)k_journal_record_bytes) != k_ra8_ok) {
257 g_fj_last_step = k_fj_step_program_failed;
258 return k_ra8_err_hw_error;
259 }
260 g_fj_last_step = k_fj_step_program_ok;
261 if (ra8_xspi_flash_read((uint8_t)k_journal_xspi_instance,
262 (uint32_t)k_journal_record_addr,
263 back,
264 (uint32_t)k_journal_record_bytes) != k_ra8_ok) {
265 g_fj_last_step = k_fj_step_read_failed;
266 return k_ra8_err_hw_error;
267 }
268 g_fj_last_step = k_fj_step_read_ok;
269 *echoed = flash_journal_unpack(back);
270 g_fj_last_echoed = *echoed;
271 return k_ra8_ok;
272}

288static void flash_journal_setup_or_halt(void)
289{
290 uint32_t cpuclk0_hz = 0U;
291 if (ra8_cgc_init() != k_ra8_ok) {
292 flash_journal_panic_halt();
293 }
294 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
295 flash_journal_panic_halt();
296 }
297 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
298 flash_journal_panic_halt();
299 }
300 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
301 flash_journal_panic_halt();
302 }
303 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
304 flash_journal_panic_halt();
305 }
306 /* Best-effort U15 I/O-expander courtesy write. The on-board Octo-SPI
307 * flash (U3, ISSI IS25LX512M; UM section 6.3 p 35) is connected through
308 * the SW4-3 analog mux, which is hardware-only. A firmware sweep of the
309 * full U15 output space (issue #44; see docs/HARDWARE_BRINGUP.md) proved
310 * the expander's GPIOs do NOT gate the OSPI bus, so this call cannot
311 * affect flash reachability -- it is kept only as a no-op courtesy and
312 * its return code is stamped to g_fj_expander_err for the memprobe. If
313 * the JEDEC read below returns 0x00FFFFFF the bus is sitting at the board
314 * pull-ups (flash not passed through SW4-3), which is a physical check. */
315 const ra8_err_t exp_err = ra8_board_io_expander_set_octospi_active();
316 g_fj_expander_err = (uint32_t)exp_err;
317 /* OCTA pins come out of reset as GPIO; ra8_board_xspi_pins_init
318 * routes them to PSEL=0x1C and pulses RESET_L on P106 so the
319 * IS25LX512M lands in standard-SPI mode with deterministic state.
320 * HUM Ch 20.6 "Multiplexed Pin Function Selector" p 871. */
321 if (ra8_board_xspi_pins_init() != k_ra8_ok) {
322 flash_journal_panic_halt();
323 }
324 if (ra8_xspi_init((uint8_t)k_journal_xspi_instance, k_ra8_xspi_lio_1s1s1s) != k_ra8_ok) {
325 flash_journal_panic_halt();
326 }
327 uint32_t jedec_id = 0U;
328 (void)ra8_xspi_flash_read_id((uint8_t)k_journal_xspi_instance, &jedec_id);
329 g_fj_jedec_id = jedec_id;
330}

334int32_t main(void)
335{
336 flash_journal_setup_or_halt();
337 ra8_isr_globals_enable();
338
339 uint32_t counter = 0U;
340 while (1) {
341 uint32_t echoed = 0U;
342 const ra8_err_t err = flash_journal_round_trip(counter, &echoed);
343 if (err == k_ra8_ok && echoed == counter) {
344 (void)ra8_board_led_toggle(k_ra8_board_led1);
345 g_fj_match += 1U;
346 g_fj_last_step = k_fj_step_compare_ok;
347 } else {
348 (void)ra8_board_led_toggle(k_ra8_board_led2);
349 g_fj_mismatch += 1U;
350 if (err == k_ra8_ok) {
351 g_fj_last_step = k_fj_step_compare_mismatch;
352 }
353 }
354 counter++;
355 ra8_delay_ms(k_journal_period_ms);
356 }
357 flash_journal_panic_halt();
358 return 0;
359}
