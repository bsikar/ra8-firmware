63 * feature overview (p 3302-3307) with no register map -- and the hand-written

120ra8_err_t internal_wait_bit(ra8_rsip_off_t offset, uint32_t mask)
121{
122 volatile uint32_t* reg = ra8_rsip_reg32(offset);
123 for (uint32_t i = 0U; i < k_ra8_rsip_poll_budget; ++i) {
124#if defined(RA8_OFF_TARGET) && defined(UNIT_TEST)
125 /* Host unit-test MMIO wait seam (tests/mocks/ra8_fake_mmio.c): an unarmed
126 * register satisfies the wait on its first poll; a test arms fail_wait /
127 * satisfy_after to drive the timeout / continuation legs of this loop. */
128 if (ra8_fake_mmio_wait_eval(reg, i, ((*reg & mask) == mask))) {
129 return k_ra8_ok;
130 }
131#else
132 if ((*reg & mask) == mask) {
133 return k_ra8_ok;
134 }
135#endif
136 }
137 return k_ra8_err_hw_timeout;
138}

235ra8_err_t ra8_rsip_init(const ra8_rsip_config_t* cfg)
236{
237 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
238
239 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C" p 446 */
240 /* HUM Ch 52.3.2 "Module-Stop Function Setting" p 3307 */
241 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_rsip);
242 RA8_RETURN_ON_ERROR(mst_err, s_tag, "rsip_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
243
244 /* HUM Ch 52.1 "Overview" p 3302 */
245 /* Engine reset + enable mailbox. */
246 volatile uint32_t* ctrl = ra8_rsip_reg32(k_ra8_rsip_off_ctrl);
247 *ctrl = k_ra8_rsip_mask_ctrl_reset;
248 *ctrl = k_ra8_rsip_mask_ctrl_enable;
249
250 if (cfg->run_bist) {
251 const ra8_err_t bist_err = internal_run_bist();
252 if (bist_err != k_ra8_ok) {
253 (void)ra8_mstp_disable(k_ra8_mstp_rsip);
254 return bist_err;
255 }
256 }
257
258 /* HUM Ch 52.1 "Overview" p 3302 */
259 /* Ack any pending IRQ bits. */
260 *ra8_rsip_reg32(k_ra8_rsip_off_isr) = k_ra8_rsip_mask_isr_all;
261
262 ra8_log_info(s_tag, "rsip_init");
263 return k_ra8_ok;
264}

314void ra8_rsip_dispatch(void)
315{
316 /* HUM Ch 52.1 "Overview" p 3302 */
317 /* Snapshot then ack the ISR. */
318 volatile uint32_t* isr = ra8_rsip_reg32(k_ra8_rsip_off_isr);
319 const uint32_t snapshot = *isr;
320 if (snapshot == 0U) {
321 return;
322 }
323 const ra8_rsip_event_fn_t fn = s_rsip_fn;
324 void* const ctx = s_rsip_ctx;
325 /* W1C ack: writing the snapshot back clears each pending bit on real
326 * hardware. The host-test RAM backing has no W1C semantics, so a test
327 * that needs the post-ack state stages the ISR word itself. */
328 *isr = snapshot;
329 if (fn != nullptr) {
330 fn(ctx, snapshot);
331 }

334// `buf` is an output the RA8_RSIP_TRNG_HARDWARE path fills; the fail-closed build
335// writes nothing, so clang-tidy cannot see a write and wrongly wants it const.
336// NOLINTNEXTLINE(readability-non-const-parameter) -- buf is filled only by the RA8_RSIP_TRNG_HARDWARE path; the fail-closed build writes nothing.
337ra8_err_t ra8_rsip_trng_read(uint8_t* buf, uint32_t len)
338{
339 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
340 if (len == 0U) {
341 return k_ra8_err_invalid_arg;
342 }
343 if ((len & ((uint32_t)k_ra8_rsip_trng_word_bytes - 1U)) != 0U) {
344 return k_ra8_err_invalid_arg;
345 }
346
347#ifdef RA8_RSIP_TRNG_HARDWARE
348 volatile uint32_t* status = ra8_rsip_reg32(k_ra8_rsip_off_rnd_status);
349 volatile uint32_t* data = ra8_rsip_reg32(k_ra8_rsip_off_rnd_data);
350
351 /* HUM Ch 52.1 "Overview" p 3302 */
352 /* Arm TRNG control word. */
353 *ra8_rsip_reg32(k_ra8_rsip_off_rnd_ctrl) = k_ra8_rsip_mask_ctrl_enable;
354
355 const uint32_t words = len >> k_ra8_rsip_word_shift;
356 for (uint32_t w = 0U; w < words; ++w) {
357 /* On hardware the engine asserts READY when a fresh word is
358 * available. The host fake has no producer thread, so we
359 * pre-assert and re-assert each iteration to keep the spin
360 * deterministic. */
361 *status |= k_ra8_rsip_mask_status_ready;
362
363 const ra8_err_t wait_err =
364 internal_wait_bit(k_ra8_rsip_off_rnd_status, k_ra8_rsip_mask_status_ready);
365 if (wait_err != k_ra8_ok) {
366 return wait_err;
367 }
368
369 const uint32_t word = *data;
370 /* HUM Ch 52.1 "Overview" p 3302 */
371 /* TRNG output is little-endian. */
372 buf[(w << k_ra8_rsip_word_shift) + 0U] = (uint8_t)(word & k_ra8_rsip_byte_mask);
373 buf[(w << k_ra8_rsip_word_shift) + 1U] =
374 (uint8_t)((word >> k_ra8_rsip_byte_bits) & k_ra8_rsip_byte_mask);
375 buf[(w << k_ra8_rsip_word_shift) + 2U] =
376 (uint8_t)((word >> k_ra8_rsip_byte_shift_2) & k_ra8_rsip_byte_mask);
377 buf[(w << k_ra8_rsip_word_shift) + 3U] =
378 (uint8_t)((word >> k_ra8_rsip_byte_shift_3) & k_ra8_rsip_byte_mask);
379
380 /* Clear READY so the next iteration genuinely waits. */
381 *status &= ~k_ra8_rsip_mask_status_ready;
382 }
383 return k_ra8_ok;
384#else
385 /* The RSIP-E50D TRNG has no documented register interface (HUM Ch 52 is a
386 * 6-page feature overview with no register map); the `RND_*` offsets above
387 * are invented and do not work on silicon -- the READY bit never asserts.
388 * Fail closed with a clear status rather than spin to a timeout or, worse,
389 * hand back an all-zero or deterministic value: predictable "entropy" is far
390 * more dangerous than an honest error. A real TRNG needs an FSP-derived
391 * RSIP primitive sequence; a software PRNG is NOT a substitute here. */
392 return k_ra8_err_not_supported;
393#endif
394}

396ra8_err_t ra8_rsip_sha256(const uint8_t* msg, uint32_t msg_len, uint8_t* digest)
397{
398 RA8_CHECK_NULL_PTR(msg, s_tag, "msg must not be nullptr");
399 RA8_CHECK_NULL_PTR(digest, s_tag, "digest must not be nullptr");
400
401#ifdef RA8_RSIP_HASH_HARDWARE
402 /* HUM Ch 52.2.3 "Hash Generator" p 3306 */
403 /* Real silicon command-issue sequence (NON-FUNCTIONAL on this silicon; never
404 * compiled -- see the backend note near the top of the file):
405 * 1. poll HASH_STATUS.READY (engine quiescent);
406 * 2. write algorithm selector to HASH_CTRL;
407 * 3. stream message words through HASH_DATA_IN;
408 * 4. spin on HASH_STATUS.DONE;
409 * 5. drain 8 digest words from HASH_DIGEST. */
410 *ra8_rsip_reg32(k_ra8_rsip_off_hash_status) |= k_ra8_rsip_mask_status_ready;
411 const ra8_err_t ready_err =
412 internal_wait_bit(k_ra8_rsip_off_hash_status, k_ra8_rsip_mask_status_ready);
413 RA8_RETURN_ON_ERROR(ready_err, s_tag, "rsip_sha256: hash ready");
414
415 /* HASH algorithm select. */
416 *ra8_rsip_reg32(k_ra8_rsip_off_hash_ctrl) = k_ra8_rsip_hash_sha256;
417
418 internal_sha256_push_msg(msg, msg_len);
419
420 const ra8_err_t wait_err = internal_hash_wait_done();
421 RA8_RETURN_ON_ERROR(wait_err, s_tag, "rsip_sha256: hash done");
422
423 internal_sha256_pull_digest(digest);
424 return k_ra8_ok;
425#else
426 /* The RSIP HASH hardware has no usable register interface, so compute the
427 * digest in software. This is the path used on silicon and on the host; it is
428 * what makes ra8_rot's on-silicon image digest work. */
429 internal_sw_sha256(msg, msg_len, digest);
430 return k_ra8_ok;
431#endif
432}

577RA8_INTERNAL
578static void internal_sw_sha256_rounds(uint32_t s[k_ra8_rsip_sw_sha256_state_w],
579 const uint32_t w[k_ra8_rsip_sw_sha256_round_cnt])
580{
581 /* FIPS PUB 180-4 Section 6.2.2: a..h working-state lanes are spec-named
582 * indices 0..7 of the 8-word hash state, not arbitrary literals. */
583 uint32_t a = s[k_sha256_lane_a];
584 uint32_t b = s[k_sha256_lane_b];
585 uint32_t c = s[k_sha256_lane_c];
586 uint32_t d = s[k_sha256_lane_d];
587 uint32_t e = s[k_sha256_lane_e];
588 uint32_t f = s[k_sha256_lane_f];
589 uint32_t g = s[k_sha256_lane_g];
590 uint32_t h = s[k_sha256_lane_h];
591 for (uint32_t i = 0U; i < k_ra8_rsip_sw_sha256_round_cnt; ++i) {
592 const uint32_t s1 = internal_sw_rotr(e, k_ra8_rsip_sw_rotr_6) ^
593 internal_sw_rotr(e, k_ra8_rsip_sw_rotr_11) ^
594 internal_sw_rotr(e, k_ra8_rsip_sw_rotr_25);
595 const uint32_t ch = (e & f) ^ ((~e) & g);
596 const uint32_t temp1 = h + s1 + ch + s_sw_sha256_k[i] + w[i];

639/* Build the SHA-256 padding tail (0x80 + zeros + 64-bit length) -- see surrounding code and HUM citations. */
640RA8_INTERNAL
641static void internal_sw_sha256_pad(uint32_t state[k_ra8_rsip_sw_sha256_state_w],
642 const uint8_t* msg,
643 uint32_t msg_len,
644 uint32_t consumed)
645{
646 uint8_t block[k_ra8_rsip_sha256_block];
647 const uint32_t rem = msg_len - consumed;
648 const uint64_t bit_len = (uint64_t)msg_len * (uint64_t)k_ra8_rsip_byte_bits;
649
650 for (uint32_t b = 0U; b < rem; ++b) {
651 block[b] = msg[consumed + b];
652 }
653 block[rem] = (uint8_t)k_ra8_rsip_sw_sha256_pad_byte;
654 uint32_t pad_idx = rem + 1U;
655 while (pad_idx < k_ra8_rsip_sha256_block) {
656 block[pad_idx] = 0x00U;
657 ++pad_idx;
658 }
659 /* If we cannot fit the 8-byte length, push this block and start a fresh one. */
660 if (rem >= ((uint32_t)k_ra8_rsip_sha256_block - (uint32_t)k_ra8_rsip_sw_sha256_pad_min + 1U)) {
661 internal_sw_sha256_compress(state, block);
662 for (uint32_t b = 0U; b < k_ra8_rsip_sha256_block; ++b) {
663 block[b] = 0x00U;
664 }
665 }
666 /* Encode bit length big-endian into the last 8 bytes. */
667 for (uint32_t b = 0U; b < k_ra8_rsip_sw_sha256_len_bytes; ++b) {
668 const uint32_t shift = (k_ra8_rsip_sw_sha256_len_bytes - 1U - b) * k_ra8_rsip_byte_bits;
669 block[(uint32_t)k_ra8_rsip_sha256_block - (uint32_t)k_ra8_rsip_sw_sha256_len_bytes + b] =
670 (uint8_t)((bit_len >> shift) & (uint64_t)k_ra8_rsip_byte_mask);
671 }
672 internal_sw_sha256_compress(state, block);
673}

675/* Emit a big-endian 32-byte SHA-256 digest from working state -- see surrounding code and HUM citations. */
676RA8_INTERNAL
677static void internal_sw_sha256_emit(const uint32_t state[k_ra8_rsip_sw_sha256_state_w],
678 uint8_t* digest)
679{
680 for (uint32_t w = 0U; w < k_ra8_rsip_sw_sha256_state_w; ++w) {
681 const size_t base = (size_t)w * (size_t)k_ra8_rsip_trng_word_bytes;
682 digest[base + 0U] = (uint8_t)((state[w] >> k_ra8_rsip_byte_shift_3) & k_ra8_rsip_byte_mask);
683 digest[base + 1U] = (uint8_t)((state[w] >> k_ra8_rsip_byte_shift_2) & k_ra8_rsip_byte_mask);
684 digest[base + 2U] = (uint8_t)((state[w] >> k_ra8_rsip_byte_bits) & k_ra8_rsip_byte_mask);
685 digest[base + 3U] = (uint8_t)(state[w] & k_ra8_rsip_byte_mask);
686 }
