317[[nodiscard]] static ra8_err_t microsd_sd_spi_open(void)
318{
319 ra8_err_t err =
320 ra8_pfs_route_peripheral(k_microsd_pin_sd_sck, k_ra8_psel_sci_async, "microsd.sck");
321 if (err != k_ra8_ok) {
322 return err;
323 }
324 err = ra8_pfs_route_peripheral(k_microsd_pin_sd_cipo, k_ra8_psel_sci_async, "microsd.cipo");
325 if (err != k_ra8_ok) {
326 return err;
327 }
328 err = ra8_pfs_route_peripheral(k_microsd_pin_sd_copi, k_ra8_psel_sci_async, "microsd.copi");
329 if (err != k_ra8_ok) {
330 return err;
331 }
332 err = ra8_gpio_output_init(k_microsd_pin_sd_cs, k_ra8_level_high);
333 if (err != k_ra8_ok) {
334 return err;
335 }
336 const ra8_sci_spi_cfg_t spi_cfg = {
337 .baud_hz = (uint32_t)k_ra8_sdmmc_spi_clock_init_hz,
338 .pclk_hz = s_pclka_hz,
339 .mode = k_ra8_spi_mode_0,
340 .lsb_first = false,
341 };
342 return ra8_sci_spi_init((uint8_t)k_microsd_sd_spi_channel, &spi_cfg);
343}

367[[nodiscard]] static ra8_err_t microsd_sd_snapshot(void)
368{
369 ra8_err_t err = microsd_sd_spi_open();
370 if (err != k_ra8_ok) {
371 s_dbg_sd_err = (uint32_t)err;
372 return err;
373 }
374 const ra8_sdmmc_spi_transport_t transport = {
375 .set_clock = microsd_spi_set_clock,
376 .cs = microsd_spi_cs,
377 .xfer = microsd_spi_xfer,
378 .ctx = &s_pclka_hz,
379 };
380 err = ra8_sdmmc_spi_init(&transport);
381 if (err != k_ra8_ok) {
382 s_dbg_sd_err = (uint32_t)err;
383 return err;
384 }
385 uint32_t blocks = 0U;
386 (void)ra8_sdmmc_spi_get_capacity(&blocks);
387 s_dbg_sd_blocks = blocks;
388 for (uint32_t lba = 0U; lba < (uint32_t)k_microsd_sectors; lba++) {
389 err = ra8_sdmmc_spi_read_block(
390 lba,
391 &s_usb_selftest_microsd_sd_snapshot[lba * (uint32_t)k_microsd_block_size]);
392 if (err != k_ra8_ok) {
393 s_dbg_sd_err = (uint32_t)err;
394 return err;
395 }
396 }
397 const bool sig =
398 (s_usb_selftest_microsd_sd_snapshot[k_microsd_bootsig_off] == (UCHAR)k_microsd_bootsig_lo) &&
399 (s_usb_selftest_microsd_sd_snapshot[k_microsd_bootsig_off + 1U] == (UCHAR)k_microsd_bootsig_hi);
400 s_usb_selftest_microsd_dbg_bootsig = sig ? 1U : 0U;
401 s_dbg_sd_err = 0U;
402 return k_ra8_ok;
403}

535static void microsd_setup_or_halt(void)
536{
537 uint32_t cpuclk0_hz = 0U;
538 uint32_t pclka_hz = 0U;
539 if (ra8_cgc_init() != k_ra8_ok) {
540 microsd_panic_halt();
541 }
542 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
543 microsd_panic_halt();
544 }
545 if (ra8_cgc_usbhs_pll_enable() != k_ra8_ok) {
546 microsd_panic_halt();
547 }
548 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
549 microsd_panic_halt();
550 }
551 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
552 microsd_panic_halt();
553 }
554 s_pclka_hz = pclka_hz;
555 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
556 microsd_panic_halt();
557 }
558 if (ra8_board_uart_console_init((uint32_t)k_microsd_baud) != k_ra8_ok) {
559 microsd_panic_halt();
560 }
561 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
562 microsd_panic_halt();
563 }
564 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
565 microsd_panic_halt();
566 }
567 /* Snapshot SD LBA 0..63 into RAM (read-only) before the kernel starts.
568 * On failure s_dbg_sd_err records it and the host verifier reports the
569 * missing boot signature -- the board does not hang. */
570 (void)microsd_sd_snapshot();
571 microsd_route_usb_or_halt();
572}
