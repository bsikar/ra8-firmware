96static ra8_err_t mock_get_buf(void* ctx,
97 const char* url,
98 const mdl_net_req_t* req,
99 char* buf,
100 size_t cap,
101 size_t* out_len,
102 mdl_net_resp_t* resp)
103{
104 (void)req;
105 (void)resp;
106 mock_net_t* f = (mock_net_t*)ctx;
107 f->get_buf_calls += 1U;
108 for (size_t i = 0U; i < f->map_n; ++i) {
109 if (strcmp(f->map[i].url, url) == 0) {
110 const int w = snprintf(buf, cap, "%s", f->map[i].html);
111 if (out_len != nullptr) {
112 *out_len = (w < 0) ? 0U : (size_t)w;
113 }
114 return k_ra8_ok;
115 }
116 }
117 return k_ra8_fail;
118}

121static ra8_err_t mock_get_file(void* ctx,
122 const char* url,
123 const mdl_net_req_t* req,
124 const char* out_path,
125 size_t* out_len,
126 mdl_net_resp_t* resp)
127{
128 (void)req;
129 mock_net_t* f = (mock_net_t*)ctx;
130 f->get_file_calls += 1U;
131 if ((f->busy_on_file_call != 0U) && (f->get_file_calls == f->busy_on_file_call)) {
132 if (resp != nullptr) { /* scripted throttle: 503 + Retry-After, nothing written */
133 resp->status = (long)k_http_unavailable;
134 (void)snprintf(resp->retry_after,
135 sizeof(resp->retry_after),
136 "%s",
137 (f->busy_retry_after != nullptr) ? f->busy_retry_after : "");
138 }
139 return k_ra8_err_busy;
140 }
141 if ((f->fail_on_file_call != 0U) && (f->get_file_calls == f->fail_on_file_call)) {
142 return k_ra8_fail; /* a single transient failure: the retry recovers it */
143 }
144 if ((f->fail_url != nullptr) && (strcmp(f->fail_url, url) == 0)) {
145 return k_ra8_fail; /* a permanently-unreachable page: every attempt fails */
146 }
147 FILE* fp = fopen(out_path, "wb");
148 if (fp == nullptr) {
149 return k_ra8_fail;
150 }
151 (void)fwrite(url, 1U, strlen(url), fp);
152 (void)fclose(fp);
153 if (out_len != nullptr) {
154 *out_len = strlen(url);
155 }
156 return k_ra8_ok;
157}

327
328 /* Reload the state from disk (simulating a fresh process). */
329 TEST_ASSERT_EQ((int64_t)k_ra8_ok, mdl_state_load(state_path, &g_state));
330
331 /* --update with chapter 3 newly present: only chapter 3 is fetched. */
332 const char* c123[] = {"http://s/chapter-1", "http://s/chapter-2", "http://s/chapter-3"};
333 set_chapters(c123, 3U);
334 mdl_fetch_stats_t s2;
335 TEST_ASSERT_EQ((int64_t)k_ra8_ok,
336 run_fetch(abs_dir, state_path, k_mdl_layout_separate, nullptr, true, 0U, &s2));
337 TEST_ASSERT_EQ((uint16_t)2, (uint16_t)s2.chapters_skipped);
338 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)s2.chapters_completed);
339 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)s2.pages_fetched);
340 /* Only chapter 3's HTML was requested; the complete chapters were untouched. */
341 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)g_mock.get_buf_calls);
342 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)g_mock.get_file_calls);
343 TEST_ASSERT(page_exists(abs_dir, "chapter-3/page_0001.jpg"));
344 TEST_END("first run + update only new");
345}

378static void test_resume_equals_uninterrupted(void)
379{
380 TEST_BEGIN("resume equals uninterrupted");
381 setup_site();
382 g_mock.map = s_map_combined;
383 g_mock.map_n = sizeof(s_map_combined) / sizeof(s_map_combined[0]);
384 const char* c12[] = {"http://s/chapter-1", "http://s/chapter-2"};
385
386 /* (a) Uninterrupted combined run into dir A: pages 1..4, continuous. */
387 char dir_a[PATH_MAX];
388 char state_a[PATH_MAX];
389 make_series_dir(dir_a, sizeof(dir_a), state_a, sizeof(state_a));
390 mdl_state_init(&g_state);
391 set_chapters(c12, 2U);
392 mdl_fetch_stats_t sa;
393 TEST_ASSERT_EQ((int64_t)k_ra8_ok,
394 run_fetch(dir_a, state_a, k_mdl_layout_combined, "foo-1-2", false, 0U, &sa));
395 TEST_ASSERT_EQ((uint16_t)4, (uint16_t)sa.pages_fetched);
396 TEST_ASSERT(page_exists(dir_a, "foo-1-2/page_0004.jpg"));
397
398 /* (b) Interrupted combined run into dir B: page 4 (d.jpg) is unreachable on
399 * every attempt, so its retries all fail and the chapter is left partial. */
400 char dir_b[PATH_MAX];
401 char state_b[PATH_MAX];
402 make_series_dir(dir_b, sizeof(dir_b), state_b, sizeof(state_b));
403 mdl_state_init(&g_state);
404 set_chapters(c12, 2U);
405 g_mock.fail_url = "http://cdn/d.jpg";
406 mdl_fetch_stats_t sb;
407 TEST_ASSERT_EQ((int64_t)k_ra8_fail,
408 run_fetch(dir_b, state_b, k_mdl_layout_combined, "foo-1-2", false, 0U, &sb));
409 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)sb.chapters_failed);
410 TEST_ASSERT(!page_exists(dir_b, "foo-1-2/page_0004.jpg"));
411
412 /* (c) Resume from dir B's on-disk state, page 4 now reachable: only it is
413 * re-fetched. */
414 g_mock.fail_url = nullptr;
415 TEST_ASSERT_EQ((int64_t)k_ra8_ok, mdl_state_load(state_b, &g_state));
416 set_chapters(c12, 2U);
417 mdl_fetch_stats_t sc;
418 TEST_ASSERT_EQ((int64_t)k_ra8_ok,
419 run_fetch(dir_b, state_b, k_mdl_layout_combined, "foo-1-2", false, 0U, &sc));
420 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)sc.pages_fetched); /* only the missing page */
421 TEST_ASSERT_EQ((uint16_t)3, (uint16_t)sc.pages_reused); /* pages 1..3 reused */
422 TEST_ASSERT(page_exists(dir_b, "foo-1-2/page_0004.jpg"));
423
424 /* The resumed output equals the uninterrupted one, page for page. */
425 assert_combined_pages_equal(dir_a, dir_b);
426 TEST_END("resume equals uninterrupted");
427}

528{
529 TEST_BEGIN("fetch loop honours governor backoff");
530 setup_site();
531 g_mock.map = s_map1;
532 g_mock.map_n = sizeof(s_map1) / sizeof(s_map1[0]);

537 make_series_dir(abs_dir, sizeof(abs_dir), state_path, sizeof(state_path));

539 const char* c1[] = {"http://s/chapter-1"};
540 set_chapters(c1, 1U);
541
542 fetch_clock_t clk = {};
543 mdl_gov_cfg_t cfg = mdl_gov_cfg_default();
544 cfg.rate_per_min = 0U; /* isolate: only the backoff/Retry-After gate sleeps */
545 mdl_governor_t gov;
546 mdl_governor_init_clock(&gov, &cfg, 1U, fetch_now, &clk, fetch_sleep, &clk);
547 g_fetch_gov = &gov;
548
549 mdl_fetch_stats_t st;
550 const ra8_err_t rc =
551 run_fetch(abs_dir, state_path, k_mdl_layout_separate, nullptr, false, 0U, &st);
552 g_fetch_gov = nullptr; /* restore the default (governor-less) path */
553 g_mock.busy_on_file_call = 0U;

555 TEST_ASSERT_EQ((int64_t)k_ra8_ok, rc);
556 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)st.pages_fetched); /* succeeded after the retry */
557 TEST_ASSERT_EQ((uint16_t)1, (uint16_t)st.chapters_completed);
558 TEST_ASSERT_EQ((uint16_t)2, (uint16_t)g_mock.get_file_calls); /* initial 503 + one retry */
559 TEST_ASSERT(clk.total_slept >= 1000); /* honoured the 1 s Retry-After */
560 TEST_ASSERT(page_exists(abs_dir, "chapter-1/page_0001.jpg"));
561 uint16_t level = 0U;
562 TEST_ASSERT(mdl_governor_peek(&gov, "cdn", &level, nullptr)); /* host of a.jpg */
563 TEST_ASSERT_EQ((int64_t)1, (int64_t)level); /* one throttle recorded */
564 TEST_END("fetch loop honours governor backoff");
565}
