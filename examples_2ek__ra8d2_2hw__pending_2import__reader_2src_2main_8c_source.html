256static void imp_print_uint(uint32_t value)
257{
258 uint8_t buf[k_imp_dec_base];
259 uint32_t n = 0U;
260 if (value == 0U) {
261 buf[n] = '0';
262 n++;
263 }
264 while ((value > 0U) && (n < (uint32_t)k_imp_dec_base)) {
265 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_imp_dec_base));
266 n++;
267 value /= (uint32_t)k_imp_dec_base;
268 }
269 for (uint32_t i = 0U; i < n; i++) {
270 imp_print(&buf[n - 1U - i], 1U);
271 }
272}

511[[nodiscard]] static ra8_err_t
512imp_read_cache(ra8_fs_mount_t* mount, const char* path, uint32_t* out_len)
513{
514 ra8_fs_file_t* file = nullptr;
515 ra8_err_t err = ra8_fs_open(mount, path, k_ra8_fs_mode_read, &file);
516 if (err != k_ra8_ok) {
517 return err;
518 }
519 uint64_t size = 0U;
520 err = ra8_fs_size(file, &size);
521 if (err != k_ra8_ok) {
522 (void)ra8_fs_close(file);
523 return err;
524 }
525 if (size > (uint64_t)sizeof(s_imp_readback)) {
526 (void)ra8_fs_close(file);
527 return k_ra8_err_no_mem;
528 }
529 uint32_t got = 0U;
530 err = ra8_fs_read(file, s_imp_readback, (uint32_t)sizeof(s_imp_readback), &got);
531 (void)ra8_fs_close(file);
532 if (err != k_ra8_ok) {
533 return err;
534 }
535 *out_len = got;
536 return k_ra8_ok;
537}

553[[nodiscard]] static ra8_err_t imp_walk_book(uint32_t len)
554{
555 const void* base = (const void*)s_imp_readback;
556 ra8_err_t err = book_validate(base, (size_t)len);
557 if (err != k_ra8_ok) {
558 return err;
559 }
560 const book_header_t* hdr = book_header(base);
561 IMP_PUTS(k_msg_title);
562 imp_print_cstr(book_string(base, hdr->title_off));
563 IMP_PUTS(k_msg_chaps);
564 imp_print_uint(hdr->chapter_count);
565 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
566
567 for (uint32_t ci = 0U; ci < hdr->chapter_count; ci++) {
568 size_t tlen = 0U;
569 ra8_err_t cerr = book_chapter_text(base, ci, (char*)s_imp_xhtml, sizeof(s_imp_xhtml), &tlen);
570 if (cerr != k_ra8_ok) {
571 return cerr;
572 }
573 IMP_PUTS(k_msg_ch);
574 imp_print_uint(ci);
575 IMP_PUTS(k_msg_textlen);
576 imp_print_uint((uint32_t)tlen);
577 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
578 }
579 return k_ra8_ok;
580}

598static void imp_run(ra8_fs_mount_t* mount)
599{
600 rabook_import_cfg_t cfg = imp_make_cfg(mount);
601 rabook_import_outcome_t out = k_rabook_import_hit;
602 char path1[k_imp_path_cap] = {};
603 char path2[k_imp_path_cap] = {};
604
605 /* First open: cache miss -> compile + write-through. */
606 if (rabook_import_open(&cfg, k_imp_epub_path, path1, (uint32_t)sizeof(path1), &out) != k_ra8_ok) {
607 imp_panic("import compile");
608 }
609 if (out != k_rabook_import_compiled) {
610 imp_panic("expected miss");
611 }
612 IMP_PUTS(k_msg_miss);
613 imp_print_cstr(path1);
614 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
615
616 /* The .rabook now exists on the card. */
617 uint32_t cache_len = 0U;
618 if (imp_read_cache(mount, path1, &cache_len) != k_ra8_ok) {
619 imp_panic("cache read");
620 }
621 IMP_PUTS(k_msg_cbytes);
622 imp_print_uint(cache_len);
623 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
624
625 /* Second open: fresh cache -> hit, the compiler seam is NOT invoked. */
626 if (rabook_import_open(&cfg, k_imp_epub_path, path2, (uint32_t)sizeof(path2), &out) != k_ra8_ok) {
627 imp_panic("import reopen");
628 }
629 if (out != k_rabook_import_hit) {
630 imp_panic("expected hit");
631 }
632 IMP_PUTS(k_msg_hit);
633 imp_print_cstr(path2);
634 imp_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
635
636 /* Read the cached book back and walk it. */
637 if (imp_read_cache(mount, path2, &cache_len) != k_ra8_ok) {
638 imp_panic("cache reread");
639 }
640 if (imp_walk_book(cache_len) != k_ra8_ok) {
641 imp_panic("book walk");
642 }
643}

680 }
