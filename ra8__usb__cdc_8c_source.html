192/* GCOVR_EXCL_START -- reachable only once the SIE has actually landed a
193 * control-OUT packet in the DCP bank, which the host register window cannot
194 * produce. ra8_usb_dcp_out_arm() clears the DCP BRDY latch as its first act
195 * (BRDYSTS is write-0-to-clear on silicon; the host model is plain memory, so
196 * the write simply stores the cleared bit), and only real hardware receiving a
197 * host OUT token sets it again. ra8_usb_dcp_out_read() therefore reports
198 * k_ra8_err_no_data on every one of the bounded poll's iterations, and this
199 * helper is never entered from any host-side input.
200 *
201 * This is NOT the blanket exclusion the former stub carried: that one covered
202 * the first 236 lines of the file and was justified by the helper's caller
203 * being hard-wired to fail. The data stage is implemented now, and the
204 * exclusion is scoped to the one helper whose entry condition is a physical
205 * bus event. The decode itself is covered on silicon by the usb_cdc_echo HIL
206 * app, which reports the host's requested baud back over the same link. */
228RA8_INTERNAL
229static void internal_apply_line_coding(const uint8_t* data, uint16_t len)
230{
231 // mcdc-deactivated: TU-local helper internal_apply_line_coding; the USB stack delivers SET_LINE_CODING control transfers with a 7-byte payload buffer (USB CDC PSTN spec 6.3.10), so data is always non-NULL and len is always exactly k_ra8_cdc_line_coding_len -- both short-circuit conditions cannot independently flip on any reachable path.
232 if ((data == nullptr) || (len < k_ra8_cdc_line_coding_len)) {
233 return;
234 }
235 uint32_t baud = (uint32_t)data[k_ra8_cdc_idx_baud_b0];
236 baud |= (uint32_t)((uint32_t)data[k_ra8_cdc_idx_baud_b1] << k_ra8_cdc_shift_byte1);
237 baud |= (uint32_t)((uint32_t)data[k_ra8_cdc_idx_baud_b2] << k_ra8_cdc_shift_byte2);
238 baud |= (uint32_t)((uint32_t)data[k_ra8_cdc_idx_baud_b3] << k_ra8_cdc_shift_byte3);
239 s_state.coding.dte_rate = baud;
240 s_state.coding.char_format = data[k_ra8_cdc_idx_char_format];
241 s_state.coding.parity_type = data[k_ra8_cdc_idx_parity_type];
242 s_state.coding.data_bits = data[k_ra8_cdc_idx_data_bits];
243}

251ra8_err_t ra8_usb_cdc_init(ra8_usb_speed_t speed)
252{
253 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
254 return k_ra8_err_invalid_arg;
255 }
256 const ra8_err_t usb_err = ra8_usb_device_init(speed);
257 if (usb_err != k_ra8_ok) {
258 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
259 return k_ra8_err_hw_init_failed;
260 }
261
262 s_state.speed = speed;
263 s_state.dtr = false;
264 s_state.rts = false;
265 internal_default_coding(&s_state.coding);
266
267 const ra8_err_t pipes_err = internal_configure_pipes(speed);
268 if (pipes_err != k_ra8_ok) {
269 (void)ra8_usb_device_deinit(speed);
270 return pipes_err;
271 }
272 s_state.initialized = true;
273 ra8_log_info(s_tag, "CDC ready");
274 return k_ra8_ok;
275}

374RA8_BOUNDED_LOOP(k_ra8_cdc_data_stage_polls)
375static ra8_err_t internal_pull_data_stage(uint8_t* buf, uint16_t cap, uint16_t* out_len)
376{
377 RA8_CHECK_NULL_PTR(buf, s_tag, "pull_data_stage: buf");
378 RA8_CHECK_NULL_PTR(out_len, s_tag, "pull_data_stage: out_len");
379 if (cap == 0U) {
380 return k_ra8_err_invalid_arg;
381 }
382 *out_len = 0U;
383 const ra8_err_t armed = ra8_usb_dcp_out_arm(s_state.speed);
384 if (armed != k_ra8_ok) {
385 return armed;
386 }
387 ra8_err_t rc = k_ra8_err_no_data;
388 for (uint16_t i = 0U; i < k_ra8_cdc_data_stage_polls; ++i) {
389 rc = ra8_usb_dcp_out_read(s_state.speed, buf, cap, out_len);
390 if (rc != k_ra8_err_no_data) {
391 return rc; /* GCOVR_EXCL_LINE -- needs a landed DCP packet; see the
392 * exclusion note on internal_apply_line_coding. */
393 }
394 }
395 return rc;
396}

414RA8_INTERNAL
415static ra8_err_t internal_dispatch_class_setup(const ra8_usb_setup_t* setup)
416{
417 switch (setup->b_request) {
418 case k_ra8_cdc_req_set_line_coding: {
419 uint8_t buf[k_ra8_cdc_line_coding_len + 1U] = {};
420 uint16_t plen = 0U;
421 const ra8_err_t pull = internal_pull_data_stage(buf, (uint16_t)sizeof(buf), &plen);
422 if (pull == k_ra8_ok) {
423 internal_apply_line_coding(buf, plen); /* GCOVR_EXCL_LINE -- same. */
424 }
425 /* The status stage is ACKed either way: a host that abandoned the data
426 * stage still gets a well-formed control-write completion, and the cached
427 * coding simply keeps its previous value. */
428 return ra8_usb_control_response(s_state.speed, true);
429 }
430 case k_ra8_cdc_req_get_line_coding: {
431 return ra8_usb_control_response(s_state.speed, true);
432 }
433 case k_ra8_cdc_req_set_control_line_state: {
434 const uint16_t mask = setup->w_value;
435 s_state.dtr = ((mask & k_ra8_cdc_line_state_dtr) != 0U);
436 s_state.rts = ((mask & k_ra8_cdc_line_state_rts) != 0U);
437 return ra8_usb_control_response(s_state.speed, true);
438 }
439 default: {
440 return k_ra8_err_not_supported;
441 }
442 }
443}
