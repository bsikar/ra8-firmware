562static void cdc_echo_iter(UCHAR* buf, ULONG cap)
563{
564 if (_ux_system_slave != UX_NULL) {
565 _ux_system_slave->ux_system_slave_device.ux_slave_device_state =
566 (unsigned long)UX_DEVICE_CONFIGURED;
567 }
568 ULONG n = 0UL;
569 if (_ux_device_class_cdc_acm_read(s_cdc_acm, buf, cap, &n) != UX_SUCCESS) {
570 tx_thread_sleep(1U);
571 return;
572 }
573 if (n == 0UL) {
574 return;
575 }
576 if (_ux_device_class_cdc_acm_write(s_cdc_acm, buf, n, &n) != UX_SUCCESS) {
577 return;
578 }
579 s_dbg_dev_echo_calls++;
580 s_dbg_dev_last_len = (uint32_t)n;
581 (void)ra8_board_led_toggle(k_ra8_board_led1);
582}

601static VOID cdc_device_worker(ULONG arg)
602{
603 (void)arg;
604
605 UINT ux = cdc_usbx_stack_up();
606 if (ux != UX_SUCCESS) {
607 s_dbg_dev_err = (uint32_t)ux;
608 return;
609 }
610 s_dbg_dev_step = (uint32_t)k_cdc_dev_step_stack;
611 ux = cdc_class_register();
612 if (ux != UX_SUCCESS) {
613 s_dbg_dev_err = (uint32_t)ux;
614 return;
615 }
616 s_dbg_dev_step = (uint32_t)k_cdc_dev_step_class;
617 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
618 if (e != k_ra8_ok) {
619 s_dbg_dev_err = (uint32_t)e;
620 return;
621 }
622 s_dbg_dev_step = (uint32_t)k_cdc_dev_step_dcd;
623 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
624 if (e != k_ra8_ok) {
625 s_dbg_dev_err = (uint32_t)e;
626 return;
627 }
628 s_dbg_dev_step = (uint32_t)k_cdc_dev_step_attach;
629
630 static UCHAR s_echo_buf[k_cdc_echo_buf];
631 while (1) {
632 if (s_cdc_acm == UX_NULL) {
633 (void)tx_semaphore_get(&s_cdc_active_sem, TX_WAIT_FOREVER);
634 continue;
635 }
636 s_dbg_dev_step = (uint32_t)k_cdc_dev_step_echo;
637 cdc_echo_iter(s_echo_buf, (ULONG)k_cdc_echo_buf);
638 }
639}
