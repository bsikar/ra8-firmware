51static ra8_ui_rect_t internal_inset(ra8_ui_rect_t outer, int32_t pad)

93static bool internal_iter_live(int32_t link, uint16_t guard, uint16_t count)
94{
95 /*
96 * The (guard < count) clause is a NASA Power-of-10 Rule 2 defensive iteration
97 * bound. Every sibling/child chain built through the public ra8_box API is
98 * acyclic and terminates when `link` becomes k_ra8_box_none, before `guard`
99 * can ever reach `count`. guard >= count is reachable only on a corrupted
100 * (cyclic) tree that no public-API sequence produces, so the second condition
101 * cannot be given independent MC/DC influence on any reachable path.
102 */
103 /* mcdc-deactivated: guard<count is an acyclic-tree cycle bound; guard>=count is unreachable via the public API. */
104 return (link != (int32_t)k_ra8_box_none) && (guard < count);
105}

130static void internal_tally(const ra8_box_tree_t* tree,
131 int16_t parent,
132 uint16_t* out_count,
133 int32_t* out_fixed,
134 uint32_t* out_flex)
135{
136 uint16_t n = 0U;
137 int32_t fixed = 0;
138 uint32_t flex = 0U;
139 int32_t c = tree->nodes[parent].first_child;
140 for (uint16_t guard = 0U; internal_iter_live(c, guard, tree->count); ++guard) {
141 const ra8_box_t* child = &tree->nodes[c];
142 if (child->fixed > 0) {
143 fixed += (int32_t)child->fixed;
144 } else {
145 flex += (uint32_t)child->flex;
146 }
147 n++;
148 c = child->next;
149 }
150 *out_count = n;
151 *out_fixed = fixed;
152 *out_flex = flex;
153}

177static void internal_layout_stack(ra8_box_tree_t* tree, int16_t parent, bool horizontal)
178{
179 const ra8_box_t* p = &tree->nodes[parent];
180 const ra8_ui_rect_t cont = internal_inset(p->rect, (int32_t)p->pad);
181 const int32_t gap = (int32_t)p->gap;
182
183 uint16_t n = 0U;
184 int32_t fixed = 0;
185 uint32_t flex = 0U;
186 internal_tally(tree, parent, &n, &fixed, &flex);
187
188 const int32_t main_extent = horizontal ? cont.w : cont.h;
189 const int32_t gaps = (n > 0U) ? ((int32_t)(n - 1U) * gap) : 0;
190 int32_t flex_space = main_extent - fixed - gaps;
191 if (flex_space < 0) {
192 flex_space = 0;
193 }
194
195 int32_t cursor = horizontal ? cont.x : cont.y;
196 int32_t c = p->first_child;
197 for (uint16_t guard = 0U; internal_iter_live(c, guard, tree->count); ++guard) {
198 ra8_box_t* child = &tree->nodes[c];
199 int32_t main_sz;
200 if (child->fixed > 0) {
201 main_sz = (int32_t)child->fixed;
202 } else if (flex > 0U) {
203 main_sz = (flex_space * (int32_t)child->flex) / (int32_t)flex;
204 } else {
205 main_sz = 0;
206 }
207 if (horizontal) {
208 child->rect.x = cursor;
209 child->rect.y = cont.y;
210 child->rect.w = main_sz;
211 child->rect.h = cont.h;
212 } else {
213 child->rect.x = cont.x;
214 child->rect.y = cursor;
215 child->rect.w = cont.w;
216 child->rect.h = main_sz;
217 }
218 cursor += main_sz + gap;
219 c = child->next;
220 }
221}

244static void internal_layout_grid(ra8_box_tree_t* tree, int16_t parent)
245{
246 const ra8_box_t* p = &tree->nodes[parent];
247 const ra8_ui_rect_t cont = internal_inset(p->rect, (int32_t)p->pad);
248 const int32_t gap = (int32_t)p->gap;
249 const int32_t cols = (p->grid_cols >= 1U) ? (int32_t)p->grid_cols : 1;
250
251 uint16_t n = 0U;
252 int32_t fixed = 0;
253 uint32_t flex = 0U;
254 internal_tally(tree, parent, &n, &fixed, &flex);
255
256 const int32_t rows = ((int32_t)n + cols - 1) / cols;
257 const int32_t row_cnt = (rows > 0) ? rows : 1;
258 int32_t cell_w = (cont.w - ((cols - 1) * gap)) / cols;
259 if (cell_w < 0) {
260 cell_w = 0;
261 }
262 int32_t cell_h = (cont.h - ((row_cnt - 1) * gap)) / row_cnt;
263 if (cell_h < 0) {
264 cell_h = 0;
265 }
266
267 int32_t idx = 0;
268 int32_t c = p->first_child;
269 for (uint16_t guard = 0U; internal_iter_live(c, guard, tree->count); ++guard) {
270 ra8_box_t* child = &tree->nodes[c];
271 const int32_t col = idx % cols;
272 const int32_t row = idx / cols;
273 child->rect.x = cont.x + (col * (cell_w + gap));
274 child->rect.y = cont.y + (row * (cell_h + gap));
275 child->rect.w = cell_w;
276 child->rect.h = (child->fixed > 0) ? (int32_t)child->fixed : cell_h;
277 idx++;
278 c = child->next;
279 }
280}

295int16_t ra8_box_add(ra8_box_tree_t* tree, int16_t parent, const ra8_box_t* node)
296{
297 if ((tree == nullptr) || (node == nullptr)) {
298 return (int16_t)k_ra8_box_none;
299 }
300 if (tree->count >= tree->cap) {
301 return (int16_t)k_ra8_box_none;
302 }
303 if ((parent != (int16_t)k_ra8_box_none) && ((parent < 0) || (parent >= (int16_t)tree->count))) {
304 return (int16_t)k_ra8_box_none;
305 }
306
307 const int16_t idx = (int16_t)tree->count;
308 tree->nodes[idx] = *node;
309 tree->nodes[idx].first_child = (int32_t)k_ra8_box_none;
310 tree->nodes[idx].next = (int32_t)k_ra8_box_none;
311 tree->count++;
312
313 if (parent != (int16_t)k_ra8_box_none) {
314 ra8_box_t* par = &tree->nodes[parent];
315 if (par->first_child == (int32_t)k_ra8_box_none) {
316 par->first_child = (int32_t)idx;
317 } else {
318 int32_t sib = par->first_child;
319 for (uint16_t guard = 0U; internal_iter_live(tree->nodes[sib].next, guard, tree->count);
320 ++guard) {
321 sib = tree->nodes[sib].next;
322 }
323 tree->nodes[sib].next = (int32_t)idx;
324 }
325 }
326 return idx;
327}

329ra8_err_t ra8_box_layout(ra8_box_tree_t* tree, int16_t root, const ra8_ui_rect_t* frame)
330{
331 RA8_CHECK_NULL_PTR(tree, s_tag, "tree must not be nullptr");
332 RA8_CHECK_NULL_PTR(frame, s_tag, "frame must not be nullptr");
333 if ((tree->count == 0U) || (root < 0) || (root >= (int16_t)tree->count)) {
334 return k_ra8_err_invalid_arg;
335 }
336
337 tree->nodes[root].rect = *frame;
338 for (uint16_t i = 0U; i < tree->count; ++i) {
339 switch ((ra8_box_kind_t)tree->nodes[i].kind) {
340 case k_ra8_box_stack_v:
341 internal_layout_stack(tree, (int16_t)i, false);
342 break;
343 case k_ra8_box_stack_h:
344 internal_layout_stack(tree, (int16_t)i, true);
345 break;
346 case k_ra8_box_grid:
347 internal_layout_grid(tree, (int16_t)i);
348 break;
349 case k_ra8_box_leaf:
350 default:
351 break;
352 }
353 }
354 return k_ra8_ok;
355}
