259RA8_INTERNAL static void internal_usb_audio_u32_to_ascii(uint32_t value, char* buf, uint32_t cap)
260{
261 enum : uint32_t {
262 k_dec_radix = 10U,
263 k_max_digits = 10U,
264 };
265 if (buf == nullptr || cap == 0U) {
266 return;
267 }
268 char tmp[k_max_digits + 1U];
269 uint32_t n = 0U;
270 if (value == 0U) {
271 tmp[n] = '0';
272 n++;
273 } else {
274 while (value > 0U && n < k_max_digits) {
275 const uint32_t digit = value % k_dec_radix;
276 tmp[n] = (char)('0' + (char)digit);
277 value /= k_dec_radix;
278 n++;
279 }
280 }
281 uint32_t out = 0U;
282 while (n > 0U && out < (cap - 1U)) {
283 n--;
284 buf[out] = tmp[n];
285 out++;
286 }
287 buf[out] = '\0';
288}

464void main(void)
465{
466 internal_usb_audio_setup_or_halt();
467 ra8_isr_globals_enable();
468 internal_usb_audio_log("ra8d2: USB Audio device ready (UAC1 48 kHz / 16-bit / stereo)\r\n");
469
470 uint32_t frames = 0U;
471 while (1) {
472 if (internal_usb_audio_send_one_frame() != k_ra8_ok) {
473 break;
474 }
475 frames++;
476 if ((frames % k_usb_audio_log_period) == 0U) {
477 internal_usb_audio_log_frames(frames);
478 }
479 ra8_delay_ms(k_usb_audio_idle_step_ms);
480 }
481
482 internal_usb_audio_panic_halt();
483}
