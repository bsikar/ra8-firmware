161static uint8_t priv_bps_valid(uint32_t bs)
162{
163 if ((bs < (uint32_t)k_ra8_fs_sector_min) || (bs > (uint32_t)k_ra8_fs_sector_max) ||
164 ((bs & (bs - 1U)) != 0U)) {
165 return 0U;
166 }

170/* `priv_parse_bpb_into_mount()`: see header for the documented contract. */
171ra8_err_t priv_parse_bpb_into_mount(ra8_fs_mount_t* m)
172{
173 if (s_scratch[k_bpb_off_signature_lo] != k_bpb_sig_lo ||
174 s_scratch[k_bpb_off_signature_hi] != k_bpb_sig_hi) {
175 return k_ra8_err_validation_failed;
176 }
177 const uint32_t bpb_bps = priv_rd16(&s_scratch[k_bpb_off_bytes_per_sec]);
178 m->sectors_per_cluster = (uint32_t)s_scratch[k_bpb_off_sec_per_clus];
179 m->reserved_sectors = priv_rd16(&s_scratch[k_bpb_off_rsvd_sec_cnt]);
180 m->num_fats = (uint32_t)s_scratch[k_bpb_off_num_fats];
181 m->root_entries = priv_rd16(&s_scratch[k_bpb_off_root_ent_cnt]);
182 /* The BPB must agree with the DEVICE: `m->bytes_per_sector` was seeded from
183 * the backend's reported block size before this parse ran, and a volume
184 * formatted for a different sector size than the medium presents (a 512e
185 * image on a 4Kn device, or vice versa) is unmountable, not reinterpretable
186 * (#683). Every supported size passes ::priv_bps_valid at mount, so the
187 * comparison also enforces 512..4096.
188
189 MC/DC: the three-condition guard's vectors live with the tests that drive
190 it, cited as `libs/ra8_fs/src/ra8_fs_fat_mount.c@priv_parse_bpb_into_mount`. */
191 if (bpb_bps != m->bytes_per_sector || m->sectors_per_cluster == 0U || m->num_fats == 0U) {
192 return k_ra8_err_validation_failed;
193 }
194 const uint32_t fat_sz_16 = priv_rd16(&s_scratch[k_bpb_off_fat_sz_16]);
195 const uint32_t fat_sz_32 = priv_rd32(&s_scratch[k_bpb_off_fat_sz_32]);
196 const uint32_t tot_sec_16 = priv_rd16(&s_scratch[k_bpb_off_tot_sec_16]);
197 const uint32_t tot_sec_32 = priv_rd32(&s_scratch[k_bpb_off_tot_sec_32]);
198 m->fat_size_sectors = (fat_sz_16 != 0U) ? fat_sz_16 : fat_sz_32;
199 /* Chosen in 32 bits first: assigning the composite pick straight into the
200 * 64-bit field would be a MISRA 10.6 composite-widening. */
201 const uint32_t tot_sec = (tot_sec_16 != 0U) ? tot_sec_16 : tot_sec_32;
202 m->total_sectors = tot_sec;
203 m->root_cluster = priv_rd32(&s_scratch[k_bpb_off_root_clus]);
204 return k_ra8_ok;
205}

228RA8_INTERNAL
229static ra8_err_t priv_compute_geometry(ra8_fs_mount_t* m)
230{
231 m->first_fat_lba = m->reserved_sectors;
232 const uint32_t root_dir_sectors =
233 ((m->root_entries * k_ra8_fs_dir_entry_bytes) + (priv_bps(m) - 1U)) / priv_bps(m);
234 m->first_root_lba = m->first_fat_lba + ((uint64_t)m->num_fats * m->fat_size_sectors);
235 m->first_data_lba = m->first_root_lba + root_dir_sectors;
236 if (m->total_sectors < m->first_data_lba) {
237 return k_ra8_err_validation_failed;
238 }
239 const uint64_t data_sectors = m->total_sectors - m->first_data_lba;
240 m->count_of_clusters = (uint32_t)(data_sectors / m->sectors_per_cluster);
241 if (m->count_of_clusters < k_cluster_count_fat12_max) {
242 m->type = k_ra8_fs_type_fat12;
243 } else if (m->count_of_clusters < k_cluster_count_fat16_max) {
244 m->type = k_ra8_fs_type_fat16;
245 } else {
246 m->type = k_ra8_fs_type_fat32;
247 }
248 return k_ra8_ok;
249}

310static ra8_err_t priv_mbr_select_entry(const uint8_t* buf, uint8_t index, uint64_t* out_base)
311{
312 if ((uint32_t)index >= (uint32_t)k_mbr_part_entry_count) {
313 return k_ra8_err_out_of_range;
314 }
315 const uint32_t slot = (uint32_t)index * (uint32_t)k_mbr_part_entry_stride;
316 const uint32_t type_off = (uint32_t)k_mbr_off_part0_type + slot;
317 const uint32_t lba_off = (uint32_t)k_mbr_off_part0_lba + slot;
318 if (buf[type_off] == 0U) {
319 return k_ra8_err_not_found;
320 }
321 const uint32_t base = priv_rd32(&buf[lba_off]);
322 if (base == 0U) {
323 return k_ra8_err_validation_failed;
324 }
325 *out_base = base;
326 return k_ra8_ok;
327}

400RA8_EXPECTS_LOCK("ra8_fs_lock")
401static ra8_err_t priv_format_locked(const ra8_fs_backend_t* backend,
402 const ra8_fs_format_opts_t* opts)
403{
404 if (backend == nullptr || opts == nullptr) {
405 return k_ra8_err_null_ptr;
406 }
407 if (backend->write_block == nullptr || backend->get_capacity == nullptr) {
408 return k_ra8_err_invalid_arg;
409 }
410 if (opts->type != k_ra8_fs_type_fat12 && opts->type != k_ra8_fs_type_fat16 &&
411 opts->type != k_ra8_fs_type_fat32 && opts->type != k_ra8_fs_type_exfat) {
412 return k_ra8_err_not_supported;
413 }
414 if (!priv_fmt_spc_valid(opts->sectors_per_cluster)) {
415 return k_ra8_err_invalid_arg;
416 }
417 uint64_t block_count = 0U;
418 uint32_t block_size = 0U;
419 ra8_err_t err = backend->get_capacity(backend->ctx, &block_count, &block_size);
420 if (err != k_ra8_ok) {
421 return err;
422 }
423 if ((priv_bps_valid(block_size) == 0U) || (block_count == 0U)) {
424 return k_ra8_err_invalid_arg;
425 }
426 if (opts->type == k_ra8_fs_type_exfat) {
427 return priv_exfat_format(backend, block_count, block_size, opts->label);
428 }
429 ra8_fs_fmt_geom_t geom = {};
430 geom.type = opts->type;
431 geom.total_sectors = block_count;
432 geom.bytes_per_sector = block_size;
433 geom.reserved_sectors = priv_fmt_reserved_for(opts->type);
434 geom.root_entries = (opts->type == k_ra8_fs_type_fat32) ? 0U : (uint32_t)k_fmt_root_ents_f16;
435 geom.root_sectors =
436 ((geom.root_entries * (uint32_t)k_ra8_fs_dir_entry_bytes) + (block_size - 1U)) / block_size;
437 err = priv_fmt_choose_geometry(&geom, opts->sectors_per_cluster);
438 if (err != k_ra8_ok) {
439 return err;
440 }
441 return priv_fmt_emit_volume(backend, &geom, opts->label);
442}

471{

580RA8_INTERNAL
581RA8_EXPECTS_LOCK("ra8_fs_lock")
582static ra8_err_t
583priv_mount_locked(const ra8_fs_backend_t* backend, uint8_t index, ra8_fs_mount_t** out_handle)
584{
585 if (backend == nullptr || out_handle == nullptr) {
586 return k_ra8_err_null_ptr;
587 }
588 if (backend->read_block == nullptr || backend->write_block == nullptr ||
589 backend->get_capacity == nullptr) {
590 return k_ra8_err_invalid_arg;
591 }
592 ra8_fs_mount_t* m = priv_alloc_mount_slot();
593 if (m == nullptr) {
594 return k_ra8_err_no_mem;
595 }
596 /* Zero the whole slot on claim (#684). The mount table is a fixed array, so
597 * this ra8_fs_mount_t address has very likely served another volume already,
598 * and the FAT and exFAT parse paths each populate a DIFFERENT subset of the
599 * struct. Clearing every field here makes "repopulate everything" a structural
600 * invariant instead of something each future field author must remember: no
601 * value can survive a card-type change into a field the new parse never
602 * touches. `in_use` is set last, on success. */
603 *m = (ra8_fs_mount_t){};
604 m->backend = *backend;
605 m->partition_base_lba = 0U;
606 const ra8_err_t cerr = priv_mount_probe_bps(backend, m);
607 if (cerr != k_ra8_ok) {
608 return cerr;
609 }
610 /* The mount table is a fixed array, so this is very likely the slot some
611 * earlier volume used. Both fields below are reset for the same reason the
612 * allocator's own state is (see ra8_fs_fat_alloc_internal.h): a slot handed
613 * out again carrying the last volume's answers hands them to this one.
614 * `exfat_upcase_ok` says "this build's up-case table is the one this VOLUME
615 * carries", and a fresh mount vouches for nothing until it has looked (#606). */
616 m->exfat_upcase_ok = 0U;
617 ra8_err_t err = priv_read_boot_sector(m, index);
618 if (err != k_ra8_ok) {
619 return err;
620 }
621 /* exFAT geometry is parsed directly in priv_exfat_parse; the FAT-BPB
622 * geometry computation applies only to FAT12/16/32. */
623 if (m->type != k_ra8_fs_type_exfat) {
624 err = priv_compute_geometry(m);
625 if (err != k_ra8_ok) {
626 return err;
627 }
628 }
629 /* Claim a fresh allocator slot BEFORE seeding it: the mount table is a fixed
630 * array, so this ra8_fs_mount_t address has very likely served another volume
631 * already, and its cached FAT sector and next-free hint describe that disk. */
632 priv_alloc_state_bind(m);
633 err = priv_fsinfo_seed(m);
634 if (err != k_ra8_ok) {
635 priv_alloc_state_release(m);
636 return err;
637 }
638 m->in_use = 1;
639 *out_handle = m;
640 return k_ra8_ok;
641}

670RA8_INTERNAL
671RA8_EXPECTS_LOCK("ra8_fs_lock")
672static ra8_err_t priv_unmount_locked(ra8_fs_mount_t* handle)
673{
674 if (handle == nullptr) {
675 return k_ra8_err_null_ptr;
676 }
677 if (handle->in_use == 0U) {
678 return k_ra8_err_invalid_state;
679 }
680 /* Last chance to make the on-disk free count true (#607). Reported, not
681 * swallowed: a card whose FSInfo could not be updated is a card a desktop
682 * will report the wrong free space for, and the caller is the only one who
683 * can decide whether that matters. The slot is released either way, so a
684 * failed flush cannot strand the mount. */
685 const ra8_err_t ferr = priv_fsinfo_flush(handle);
686 priv_alloc_state_release(handle);
687 handle->in_use = 0;
688 handle->type = k_ra8_fs_type_unknown;
689 return ferr;
690}
