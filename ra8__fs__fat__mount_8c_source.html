91ra8_err_t priv_parse_bpb_into_mount(ra8_fs_mount_t* m)
92{
93 if (s_scratch[k_bpb_off_signature_lo] != k_bpb_sig_lo ||

95 return k_ra8_err_validation_failed;

101 m->root_entries = priv_rd16(&s_scratch[k_bpb_off_root_ent_cnt]);
102 if (m->bytes_per_sector != k_ra8_fs_bytes_per_sector || m->sectors_per_cluster == 0U ||
103 m->num_fats == 0U) {
104 return k_ra8_err_validation_failed;
105 }
106 const uint32_t fat_sz_16 = priv_rd16(&s_scratch[k_bpb_off_fat_sz_16]);

111 m->total_sectors = (tot_sec_16 != 0U) ? tot_sec_16 : tot_sec_32;

137RA8_INTERNAL
138static ra8_err_t priv_compute_geometry(ra8_fs_mount_t* m)
139{
140 m->first_fat_lba = m->reserved_sectors;
141 const uint32_t root_dir_sectors =
142 ((m->root_entries * k_ra8_fs_dir_entry_bytes) + (k_ra8_fs_bytes_per_sector - 1U)) /
143 k_ra8_fs_bytes_per_sector;
144 m->first_root_lba = m->first_fat_lba + (m->num_fats * m->fat_size_sectors);
145 m->first_data_lba = m->first_root_lba + root_dir_sectors;
146 if (m->total_sectors < m->first_data_lba) {
147 return k_ra8_err_validation_failed;
148 }
149 const uint32_t data_sectors = m->total_sectors - m->first_data_lba;
150 m->count_of_clusters = data_sectors / m->sectors_per_cluster;
151 if (m->count_of_clusters < k_cluster_count_fat12_max) {
152 m->type = k_ra8_fs_type_fat12;
153 } else if (m->count_of_clusters < k_cluster_count_fat16_max) {
154 m->type = k_ra8_fs_type_fat16;
155 } else {
156 m->type = k_ra8_fs_type_fat32;
157 }
158 return k_ra8_ok;
159}

350priv_gpt_scan_entries(ra8_fs_mount_t* m, uint32_t entry_lba, uint32_t count, uint32_t* out_base)
351{
352 uint32_t basic_lba = 0U;
353 uint32_t any_lba = 0U;
354 for (uint32_t i = 0U; i < count; i++) {
355 const uint32_t sector = i / (uint32_t)k_gpt_entries_per_sector;
356 const uint32_t offset = (i % (uint32_t)k_gpt_entries_per_sector) * (uint32_t)k_gpt_entry_bytes;
357 if (offset == 0U) {
358 const ra8_err_t err = priv_read_sector(m, entry_lba + sector, s_scratch);
359 if (err != k_ra8_ok) {
360 return err;
361 }
362 }
363 priv_gpt_note_entry(&s_scratch[offset], &basic_lba, &any_lba);
364 }
365 if (basic_lba != 0U) {
366 *out_base = basic_lba;
367 return k_ra8_ok;
368 }
369 if (any_lba != 0U) {
370 *out_base = any_lba;
371 return k_ra8_ok;
372 }
373 return k_ra8_err_not_found;
374}

399{

427ra8_err_t ra8_fs_format(const ra8_fs_backend_t* backend, const ra8_fs_format_opts_t* opts)
428{
429 if (backend == nullptr || opts == nullptr) {
430 return k_ra8_err_null_ptr;
431 }
432 if (backend->write_block == nullptr || backend->get_capacity == nullptr) {
433 return k_ra8_err_invalid_arg;
434 }
435 if (opts->type != k_ra8_fs_type_fat12 && opts->type != k_ra8_fs_type_fat16 &&
436 opts->type != k_ra8_fs_type_fat32 && opts->type != k_ra8_fs_type_exfat) {
437 return k_ra8_err_not_supported;
438 }
439 if (!priv_fmt_spc_valid(opts->sectors_per_cluster)) {
440 return k_ra8_err_invalid_arg;
441 }
442 uint32_t block_count = 0U;
443 uint32_t block_size = 0U;
444 ra8_err_t err = backend->get_capacity(backend->ctx, &block_count, &block_size);
445 if (err != k_ra8_ok) {
446 return err;
447 }
448 if (block_size != (uint32_t)k_ra8_fs_bytes_per_sector || block_count == 0U) {
449 return k_ra8_err_invalid_arg;
450 }
451 if (opts->type == k_ra8_fs_type_exfat) {
452 return priv_exfat_format(backend, block_count, opts->label);
453 }
454 ra8_fs_fmt_geom_t geom = {};
455 geom.type = opts->type;
456 geom.total_sectors = block_count;
457 geom.reserved_sectors = priv_fmt_reserved_for(opts->type);
458 geom.root_entries = (opts->type == k_ra8_fs_type_fat32) ? 0U : (uint32_t)k_fmt_root_ents_f16;
459 geom.root_sectors = ((geom.root_entries * (uint32_t)k_ra8_fs_dir_entry_bytes) +
460 ((uint32_t)k_ra8_fs_bytes_per_sector - 1U)) /
461 (uint32_t)k_ra8_fs_bytes_per_sector;
462 err = priv_fmt_choose_geometry(&geom, opts->sectors_per_cluster);
463 if (err != k_ra8_ok) {
464 return err;
465 }
466 return priv_fmt_emit_volume(backend, &geom, opts->label);
467}

491static ra8_err_t priv_read_boot_sector(ra8_fs_mount_t* m)
492{
493 ra8_err_t err = priv_read_sector(m, 0, s_scratch);
494 if (err != k_ra8_ok) {
495 return err;
496 }
497 err = priv_parse_volume(m);
498 if (err == k_ra8_ok) {
499 return k_ra8_ok;
500 }
501 uint32_t base = priv_mbr_part0_lba(s_scratch);
502 if (base == 0U) {
503 return err;
504 }
505 if (s_scratch[k_mbr_off_part0_type] == (uint8_t)k_gpt_part_type_protective) {
506 const ra8_err_t gpt_err = priv_gpt_locate_volume(m, &base);
507 if (gpt_err != k_ra8_ok) {
508 return gpt_err;
509 }
510 }
511 m->partition_base_lba = base;
512 err = priv_read_sector(m, 0, s_scratch);
513 if (err != k_ra8_ok) {
514 return err;
515 }
516 return priv_parse_volume(m);
517}

545ra8_err_t ra8_fs_mount(const ra8_fs_backend_t* backend, ra8_fs_mount_t** out_handle)
546{
547 if (backend == nullptr || out_handle == nullptr) {
548 return k_ra8_err_null_ptr;
549 }
550 if (backend->read_block == nullptr || backend->write_block == nullptr ||
551 backend->get_capacity == nullptr) {
552 return k_ra8_err_invalid_arg;
553 }
554 ra8_fs_mount_t* m = priv_alloc_mount_slot();
555 if (m == nullptr) {
556 return k_ra8_err_no_mem;
557 }
558 m->backend = *backend;
559 m->partition_base_lba = 0U;
560 ra8_err_t err = priv_read_boot_sector(m);
561 if (err != k_ra8_ok) {
562 return err;
563 }
564 /* exFAT geometry is parsed directly in priv_exfat_parse; the FAT-BPB
565 * geometry computation applies only to FAT12/16/32. */
566 if (m->type != k_ra8_fs_type_exfat) {
567 err = priv_compute_geometry(m);
568 if (err != k_ra8_ok) {
569 return err;
570 }
571 }
572 m->in_use = 1;
573 *out_handle = m;
574 return k_ra8_ok;
575}
