178RA8_INTERNAL static void write_kv(FILE* fp, char type, const char* val)
179{
180 (void)fputc(type, fp);
181 (void)fputc('\t', fp);
182 for (const char* c = val; *c != '\0'; ++c) {
183 const char ch = (char)(((*c == '\t') || (*c == '\n') || (*c == '\r')) ? ' ' : *c);
184 (void)fputc(ch, fp);
185 }
186 (void)fputc('\n', fp);

190RA8_INTERNAL static void write_records(FILE* fp, const mdl_state_t* st)
191{
192 for (uint16_t i = 0U; i < st->chapter_count; ++i) {
193 const mdl_chapter_rec_t* c = &st->chapters[i];
194 (void)fprintf(fp,
195 "C\t%s\t%ld\t%d\t%u\t%u\t%lld\t%s\n",
196 c->chapter_id,
197 c->number,
198 c->complete ? 1 : 0,
199 (unsigned)c->page_count,
200 (unsigned)c->pages_done,
201 (long long)c->fetched_at,
202 c->source_url);
203 }
204 for (uint32_t i = 0U; i < st->page_rec_count; ++i) {
205 const mdl_page_rec_t* p = &st->pages[i];
206 (void)fprintf(fp,
207 "P\t%016llx\t%016llx\t%s\n",
208 (unsigned long long)p->url_hash,
209 (unsigned long long)p->content_hash,
210 p->rel_path);
211 }

214ra8_err_t mdl_state_save(const char* path, const mdl_state_t* st)
215{
216 if ((path == nullptr) || (st == nullptr)) {
217 return k_ra8_err_invalid_arg;
218 }
219 char tmp[k_state_line_max];
220 const int tn = snprintf(tmp, sizeof(tmp), "%s.tmp", path);
221 if ((tn < 0) || ((size_t)tn >= sizeof(tmp))) {
222 return k_ra8_fail;
223 }
224 FILE* fp = fopen(tmp, "w");
225 if (fp == nullptr) {
226 return k_ra8_fail;
227 }
228 (void)fprintf(fp, "# media_dl library state v%u\n", (unsigned)k_mdl_state_version);
229 (void)fprintf(fp, "V\t%u\n", (unsigned)k_mdl_state_version);
230 write_kv(fp, 'S', st->series_url);
231 write_kv(fp, 'T', st->series_title);
232 write_kv(fp, 'N', st->site_name);
233 write_kv(fp, 'H', st->site_host);
234 write_kv(fp, 'G', st->config_path);
235 write_records(fp, st);
236 const bool io_ok = (ferror(fp) == 0);
237 if ((fclose(fp) != 0) || !io_ok) {
238 (void)remove(tmp);
239 return k_ra8_fail;
240 }
241 if (rename(tmp, path) != 0) {
242 (void)remove(tmp);
243 return k_ra8_fail;
244 }
245 return k_ra8_ok;
246}

251RA8_INTERNAL static size_t split_tabs(char* line, char* fld[], size_t max)
252{
253 size_t n = 0U;
254 fld[n] = line;
255 ++n;
256 for (char* c = line; (*c != '\0') && (n < max); ++c) {
257 if (*c == '\t') {
258 *c = '\0';
259 fld[n] = c + 1;
260 ++n;
261 }
262 }
263 return n;
264}

268{
269 size_t n = strlen(s);
270 while ((n > 0U) && ((s[n - 1U] == '\n') || (s[n - 1U] == '\r'))) {

272 --n;

277RA8_INTERNAL static bool apply_chapter(mdl_state_t* st, char* fld[], size_t nf)
278{
279 if (nf < (size_t)k_c_fields) {
280 return false;
281 }
282 const long num = strtol(fld[k_c_number], nullptr, (int)k_state_dec_base);
283 mdl_chapter_rec_t* rec = mdl_state_add_chapter(st, fld[k_c_id], fld[k_c_url], num);
284 if (rec == nullptr) {
285 return false;
286 }
287 rec->complete = (strtol(fld[k_c_done], nullptr, (int)k_state_dec_base) != 0);
288 rec->page_count = (uint16_t)strtoul(fld[k_c_pages], nullptr, (int)k_state_dec_base);
289 rec->pages_done = (uint16_t)strtoul(fld[k_c_ready], nullptr, (int)k_state_dec_base);
290 rec->fetched_at = (int64_t)strtoll(fld[k_c_epoch], nullptr, (int)k_state_dec_base);
291 return true;
292}

306RA8_INTERNAL static bool apply_line(mdl_state_t* st, char* fld[], size_t nf, bool* seen_v)
307{
308 const char type = fld[0][0];
309 if ((fld[0][1] != '\0')) {
310 return false; /* a record type is exactly one character */
311 }
312 if (!*seen_v) {
313 /* The first record MUST be a version line this build understands. */
314 if ((type != 'V') || (nf < 2U) ||
315 (strtoul(fld[1], nullptr, (int)k_state_dec_base) != (unsigned long)k_mdl_state_version)) {
316 return false;
317 }
318 *seen_v = true;
319 return true;
320 }
321 switch (type) {
322 case 'V':
323 return true; /* a duplicate version line is harmless */
324 case 'S':
325 set_opt(st->series_url, sizeof(st->series_url), (nf > 1U) ? fld[1] : "");
326 return true;
327 case 'T':
328 set_opt(st->series_title, sizeof(st->series_title), (nf > 1U) ? fld[1] : "");
329 return true;
330 case 'N':
331 set_opt(st->site_name, sizeof(st->site_name), (nf > 1U) ? fld[1] : "");
332 return true;
333 case 'H':
334 set_opt(st->site_host, sizeof(st->site_host), (nf > 1U) ? fld[1] : "");
335 return true;
336 case 'G':
337 set_opt(st->config_path, sizeof(st->config_path), (nf > 1U) ? fld[1] : "");
338 return true;
339 case 'C':
340 return apply_chapter(st, fld, nf);
341 case 'P':
342 return apply_page(st, fld, nf);
343 default:
344 return false;
345 }
346}

349RA8_INTERNAL static bool parse_stream(FILE* fp, mdl_state_t* st)
350{
351 char line[k_state_line_max];
352 bool seen_v = false;
353 while (fgets(line, (int)sizeof(line), fp) != nullptr) {
354 chomp(line);
355 if ((line[0] == '\0') || (line[0] == '#')) {
356 continue; /* blank or comment */
357 }
358 char* fld[k_state_max_flds];
359 const size_t nf = split_tabs(line, fld, (size_t)k_state_max_flds);
360 if (!apply_line(st, fld, nf, &seen_v)) {
361 return false;
362 }
363 }
364 return seen_v; /* a file with no version record is corrupt */
365}

367ra8_err_t mdl_state_load(const char* path, mdl_state_t* st)
368{
369 if ((path == nullptr) || (st == nullptr)) {
370 return k_ra8_err_invalid_arg;
371 }
372 mdl_state_init(st);
373 FILE* fp = fopen(path, "r");
374 if (fp == nullptr) {
375 if (errno == ENOENT) {
376 return k_ra8_ok; /* first run of this series: start empty */
377 }
378 (void)fprintf(stderr, "media_dl: cannot read state '%s'\n", path);
379 return k_ra8_err_invalid_state;
380 }
381 const bool ok = parse_stream(fp, st);
382 (void)fclose(fp);
383 if (!ok) {
384 mdl_state_init(st);
385 (void)fprintf(stderr,
386 "media_dl: state '%s' is corrupt or unsupported; rebuilding "
387 "from scratch (already-downloaded pages are reused by content)\n",
388 path);
389 return k_ra8_err_invalid_state;
390 }
391 return k_ra8_ok;
392}

397RA8_INTERNAL static bool complete_span(const mdl_state_t* st, long* lo, long* hi, size_t* n)
398{
399 bool any = false;
400 *n = 0U;
401 for (uint16_t i = 0U; i < st->chapter_count; ++i) {
402 if (!st->chapters[i].complete) {
403 continue;
404 }
405 const long num = st->chapters[i].number;
406 if (!any) {
407 *lo = num;
408 *hi = num;
409 any = true;
410 } else {
411 *lo = (num < *lo) ? num : *lo;
412 *hi = (num > *hi) ? num : *hi;
413 }
414 *n += 1U;
415 }
416 return any;
417}

431RA8_INTERNAL static void append_gaps(const mdl_state_t* st, long lo, long hi, char* buf, size_t cap)
432{
433 size_t len = strlen(buf);
434 size_t listed = 0U;
435 bool wrote = false;
436 for (long num = lo; (num <= hi) && (listed < (size_t)k_state_gap_cap); ++num) {
437 if (has_complete_number(st, num)) {
438 continue;
439 }
440 const int w = snprintf(&buf[len],
441 (len < cap) ? (cap - len) : 0U,
442 "%s%ld",
443 wrote ? ", " : ", missing ",
444 num);
445 if ((w < 0) || ((size_t)w >= ((len < cap) ? (cap - len) : 0U))) {
446 return; /* out of room: stop rather than truncate mid-number */
447 }
448 len += (size_t)w;
449 wrote = true;
450 ++listed;
451 }
452}

454void mdl_state_coverage(const mdl_state_t* st, char* buf, size_t cap)
455{
456 if ((st == nullptr) || (buf == nullptr) || (cap == 0U)) {
457 return;
458 }
459 long lo = 0;
460 long hi = 0;
461 size_t n = 0U;
462 if (!complete_span(st, &lo, &hi, &n)) {
463 (void)snprintf(buf, cap, "no chapters complete");
464 return;
465 }
466 (void)snprintf(buf, cap, "%zu chapter(s) complete, %ld..%ld", n, lo, hi);
467 append_gaps(st, lo, hi, buf, cap);
