212ra8_err_t mdl_state_save(const char* path, const mdl_state_t* st)
213{
214 if ((path == nullptr) || (st == nullptr)) {
215 return k_ra8_err_invalid_arg;
216 }
217 char tmp[k_state_line_max];
218 const int tn = snprintf(tmp, sizeof(tmp), "%s.tmp", path);
219 if ((tn < 0) || ((size_t)tn >= sizeof(tmp))) {
220 return k_ra8_fail;
221 }
222 FILE* fp = fopen(tmp, "w");
223 if (fp == nullptr) {
224 return k_ra8_fail;
225 }
226 (void)fprintf(fp, "# media_dl library state v%u\n", (unsigned)k_mdl_state_version);
227 (void)fprintf(fp, "V\t%u\n", (unsigned)k_mdl_state_version);
228 write_kv(fp, 'S', st->series_url);
229 write_kv(fp, 'T', st->series_title);
230 write_kv(fp, 'N', st->site_name);
231 write_kv(fp, 'H', st->site_host);
232 write_kv(fp, 'G', st->config_path);
233 write_records(fp, st);
234 const bool io_ok = (ferror(fp) == 0);
235 if ((fclose(fp) != 0) || !io_ok) {
236 (void)remove(tmp);
237 return k_ra8_fail;
238 }
239 if (rename(tmp, path) != 0) {
240 (void)remove(tmp);
241 return k_ra8_fail;
242 }
243 return k_ra8_ok;
244}

304RA8_INTERNAL static bool apply_line(mdl_state_t* st, char* fld[], size_t nf, bool* seen_v)
305{
306 const char type = fld[0][0];
307 if ((fld[0][1] != '\0')) {
308 return false; /* a record type is exactly one character */
309 }
310 if (!*seen_v) {
311 /* The first record MUST be a version line this build understands. */
312 if ((type != 'V') || (nf < 2U) ||
313 (strtoul(fld[1], nullptr, (int)k_state_dec_base) != (unsigned long)k_mdl_state_version)) {
314 return false;
315 }
316 *seen_v = true;
317 return true;
318 }
319 switch (type) {
320 case 'V':
321 return true; /* a duplicate version line is harmless */
322 case 'S':
323 set_opt(st->series_url, sizeof(st->series_url), (nf > 1U) ? fld[1] : "");
324 return true;
325 case 'T':
326 set_opt(st->series_title, sizeof(st->series_title), (nf > 1U) ? fld[1] : "");
327 return true;
328 case 'N':
329 set_opt(st->site_name, sizeof(st->site_name), (nf > 1U) ? fld[1] : "");
330 return true;
331 case 'H':
332 set_opt(st->site_host, sizeof(st->site_host), (nf > 1U) ? fld[1] : "");
333 return true;
334 case 'G':
335 set_opt(st->config_path, sizeof(st->config_path), (nf > 1U) ? fld[1] : "");
336 return true;
337 case 'C':
338 return apply_chapter(st, fld, nf);
339 case 'P':
340 return apply_page(st, fld, nf);
341 default:
342 return false;
343 }
344}

347RA8_INTERNAL static bool parse_stream(FILE* fp, mdl_state_t* st)
348{
349 char line[k_state_line_max];
350 bool seen_v = false;
351 while (fgets(line, (int)sizeof(line), fp) != nullptr) {
352 chomp(line);
353 if ((line[0] == '\0') || (line[0] == '#')) {
354 continue; /* blank or comment */
355 }
356 char* fld[k_state_max_flds];
357 const size_t nf = split_tabs(line, fld, (size_t)k_state_max_flds);
358 if (!apply_line(st, fld, nf, &seen_v)) {
359 return false;
360 }
361 }
362 return seen_v; /* a file with no version record is corrupt */
363}

365ra8_err_t mdl_state_load(const char* path, mdl_state_t* st)
366{
367 if ((path == nullptr) || (st == nullptr)) {
368 return k_ra8_err_invalid_arg;
369 }
370 mdl_state_init(st);
371 FILE* fp = fopen(path, "r");
372 if (fp == nullptr) {
373 if (errno == ENOENT) {
374 return k_ra8_ok; /* first run of this series: start empty */
375 }
376 (void)fprintf(stderr, "media_dl: cannot read state '%s'\n", path);
377 return k_ra8_err_invalid_state;
378 }
379 const bool ok = parse_stream(fp, st);
380 (void)fclose(fp);
381 if (!ok) {
382 mdl_state_init(st);
383 (void)fprintf(stderr,
384 "media_dl: state '%s' is corrupt or unsupported; rebuilding "
385 "from scratch (already-downloaded pages are reused by content)\n",
386 path);
387 return k_ra8_err_invalid_state;
388 }
389 return k_ra8_ok;
390}

395RA8_INTERNAL static bool complete_span(const mdl_state_t* st, long* lo, long* hi, size_t* n)
396{
397 bool any = false;
398 *n = 0U;
399 for (uint16_t i = 0U; i < st->chapter_count; ++i) {
400 if (!st->chapters[i].complete) {
401 continue;
402 }
403 const long num = st->chapters[i].number;
404 if (!any) {
405 *lo = num;
406 *hi = num;
407 any = true;
408 } else {
409 *lo = (num < *lo) ? num : *lo;
410 *hi = (num > *hi) ? num : *hi;
411 }
412 *n += 1U;
413 }
414 return any;
415}

429RA8_INTERNAL static void append_gaps(const mdl_state_t* st, long lo, long hi, char* buf, size_t cap)
430{
431 size_t len = strlen(buf);
432 size_t listed = 0U;
433 bool wrote = false;
434 for (long num = lo; (num <= hi) && (listed < (size_t)k_state_gap_cap); ++num) {
435 if (has_complete_number(st, num)) {
436 continue;
437 }
438 const int w = snprintf(&buf[len],
439 (len < cap) ? (cap - len) : 0U,
440 "%s%ld",
441 wrote ? ", " : ", missing ",
442 num);
443 if ((w < 0) || ((size_t)w >= ((len < cap) ? (cap - len) : 0U))) {
444 return; /* out of room: stop rather than truncate mid-number */
445 }
446 len += (size_t)w;
447 wrote = true;
448 ++listed;
449 }
450}
