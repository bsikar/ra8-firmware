139static ra8_err_t internal_target_wait_spsr(volatile r_spi_regs_t* reg, uint32_t flag_mask)
140{
141 /* Bounded busy-poll of SPSR. On the host test build the ra8_hw_err MMIO fault
142 * seam (ra8_fake_mmio_*) drives this real loop to succeed-after-N or to time out,
143 * so both the success and timeout legs are exercised on host (T1-01) rather
144 * than compiled out behind an RA8_OFF_TARGET short-circuit. On target it is
145 * a plain register spin with a fixed iteration bound. */
146 /* HUM Ch 43.2.9 "SPSR : SPI Status Register" p 2898 */
147 return ra8_hw_wait_flag_set32(&reg->SPSR, flag_mask, (uint32_t)k_spi_b_target_poll_limit);
148}

173static uint32_t internal_target_spcr(void)
174{
175 uint32_t v = 0U;
176 /* SPCR.MSTR (bit 30) is NOT set -- Peripheral (Target) mode. */
177 /* SPCR.SCKASE (bit 12) is NOT set -- valid in controller mode only. */
178 v |= (uint32_t)k_ra8_spcr_mask_spe; /* SPI Function Enable. */
179 v |= (uint32_t)k_ra8_spcr_mask_modfen; /* Mode Fault Error Detection. */
180 return v;
181}

213static uint32_t internal_spcmd_target(const ra8_spi_cfg_t* cfg)
214{
215 uint32_t v = 0U;
216 switch (cfg->mode) {
217 case k_ra8_spi_mode_1:
218 v |= (uint32_t)k_ra8_spcmd_mask_cpha;
219 break;
220 case k_ra8_spi_mode_2:
221 v |= (uint32_t)k_ra8_spcmd_mask_cpol;
222 break;
223 case k_ra8_spi_mode_3:
224 v |= (uint32_t)k_ra8_spcmd_mask_cpha;
225 v |= (uint32_t)k_ra8_spcmd_mask_cpol;
226 break;
227 case k_ra8_spi_mode_0:
228 default:
229 break;
230 }
231 if (cfg->lsb_first) {
232 v |= (uint32_t)k_ra8_spcmd_mask_lsbf;
233 }
234 /* 8-bit frame. */
235 v |= ((uint32_t)k_ra8_spcmd_spb_8bit << k_ra8_spcmd_bit_spb_lo) & k_ra8_spcmd_mask_spb;
236 return v;
237}

262static void internal_target_program_regs(volatile r_spi_regs_t* reg, const ra8_spi_cfg_t* cfg)
263{
264 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
265 reg->SPSRC = (uint32_t)k_ra8_spsrc_mask_all;
266
267 /* SPBR=0 in target mode: clock is external; field ignored by HW. */
268 /* HUM Ch 43.2.6 "SPCR3 : SPI Control Register 3" p 2891 */
269 reg->SPCR3 = 0U;
270
271 /* HUM Ch 43.2.3 "SPDECR : SPI Delay Control Register" p 2883 */
272 reg->SPDECR = 0U;
273
274 /* No loopback in target mode. SPCR2 must be written while SPE=0. */
275 /* HUM Ch 43.2.4 "SPCR2 : SPI Control Register 2" p 2889 */
276 reg->SPCR2 = 0U;
277
278 /* HUM Ch 43.2.7 "SPCMDm : SPI Command Register" p 2893 */
279 reg->SPCMD[0] = internal_spcmd_target(cfg);
280
281 /* HUM Ch 43.2.10 "SPDCR : SPI Data Control Register" p 2896 */
282 reg->SPDCR = 0U;
283 reg->SPDCR2 = 0U;
284
285 /* HUM Ch 43.2.14 "SPFCR : SPI FIFO Clear Register" p 2906 */
286 reg->SPFCR = (uint32_t)k_ra8_spfcr_mask_spfrst;
287
288 /* Re-clear status after FIFO reset in case SPRF was set by flush. */
289 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
290 reg->SPSRC = (uint32_t)k_ra8_spsrc_mask_all;
291}

298ra8_err_t ra8_spi_b_target_init(uint8_t channel, const ra8_spi_cfg_t* cfg)
299{
300 RA8_CHECK_NULL_PTR(cfg, s_tag, "target_init: cfg");
301 if (channel >= (uint8_t)k_ra8_spi_b_channel_count) {
302 return k_ra8_err_invalid_arg;
303 }
304 volatile r_spi_regs_t* reg = ra8_spi(channel);
305 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
306 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
307 }
308 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B" p 444 */
309 const ra8_err_t mst_err = ra8_mstp_enable(s_spi_mstp_table[channel]);
310 RA8_RETURN_ON_ERROR(mst_err, s_tag, "target_init: mstp"); /* GCOVR_EXCL_BR_LINE */
311
312 /* Disable (SPE=0) before reprogramming. */
313 /* HUM Ch 43.2.4 "SPCR : SPI Control Register" p 2884 */
314 reg->SPCR = 0U;
315
316 internal_target_program_regs(reg, cfg);
317
318 /* Enable SPI in target/peripheral mode: MSTR=0, MODFEN=1, SPE=1. */
319 /* HUM Ch 43.2.4 "SPCR : SPI Control Register" p 2884 */
320 /* HUM Ch 43.3.14 "SPI peripheral-mode operation" p 2912 */
321 reg->SPCR = internal_target_spcr();
322
323 ra8_log_info_val(s_tag, "target_init channel", (uint32_t)channel);
324 return k_ra8_ok;
325}

327ra8_err_t ra8_spi_b_target_xfer(uint8_t channel, uint8_t tx, uint8_t* rx)
328{
329 volatile r_spi_regs_t* reg = ra8_spi(channel);
330 RA8_CHECK_NULL_PTR(reg, s_tag, "target_xfer: channel out of range");
331
332 /* Wait until the TX buffer is empty (safe to pre-load response). */
333 /* HUM Ch 43.2.9 "SPSR : SPI Status Register" p 2898 */
334 ra8_err_t err = internal_target_wait_spsr(reg, (uint32_t)k_ra8_spsr_mask_sptef);
335 if (err != k_ra8_ok) {
336 return err;
337 }
338
339 /* Pre-load the response byte -- shifted out on CIPO when the external */
340 /* controller asserts CS and provides RSPCK. */
341 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
342 reg->SPDR = (uint32_t)tx;
343
344 /* Clear TX-empty flag (write-1-to-clear). */
345 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
346 reg->SPSRC = (uint32_t)k_ra8_spsrc_mask_sptefc;
347
348 /* Wait for the external controller to complete a full frame (SPRF). */
349 /* HUM Ch 43.2.9 "SPSR : SPI Status Register" p 2898 */
350 err = internal_target_wait_spsr(reg, (uint32_t)k_ra8_spsr_mask_sprf);
351 if (err != k_ra8_ok) {
352 return err;
353 }
354
355 /* Drain the byte received on COPI from the RX FIFO front-end. */
356 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
357 const uint8_t received = (uint8_t)reg->SPDR;
358
359 /* Clear RX-full flag. */
360 /* HUM Ch 43.2.13 "SPSRC : SPI Status Clear Register" p 2905 */
361 reg->SPSRC = (uint32_t)k_ra8_spsrc_mask_sprfc;
362
363 if (rx != nullptr) {
364 *rx = received;
365 }
366 return k_ra8_ok;
367}
