4"""Gate: examples shall not hand-encode board connector pins.
5
6The EK-RA8D2 pinout is a fixed board fact, and the board layer
7(``libs/ra8_board_ek_ra8d2``) is its single source of truth: it exposes the
8USB, I2C/I3C, SD, Ethernet, audio, console, LED and switch pins as
9``k_ra8_board_*`` symbols (and accessors like ``ra8_board_sw_pin``). When an
10application instead re-encodes a pin as ``(port << 8) | pin`` it duplicates
11that fact -- and #251 showed the cost: the four USB-FS pins were copy-pasted,
12byte-identical, across 29 apps under a dozen different local names, so a pin
13correction in one silently skipped the rest.
14
15This gate forbids the ``((uint16_t)k_ra8_port_N << 8) | (uint16_t)k_ra8_pin_M``
16board-pin encoding idiom anywhere under ``examples/``. The fix is always to
17reference the board symbol, or -- if the pin is a real board connector the
18board layer does not expose yet -- to add it there first, then reference it.
19
20Scope note: this targets the specific *encoding* idiom that was duplicated,
21not every pin literal. A genuinely app-specific pin (e.g. where an external
22motor driver is wired, which is not an EK-RA8D2 board fact and has no board
23home) is out of scope by construction -- it does not use this idiom.
24
25Run::
26
27 check_example_board_pins.py # scan examples/
28 check_example_board_pins.py path/to/main.c ... # scan listed files
29
30Exit 0 if no example hand-encodes a board pin, exit 1 (with the offenders)
31otherwise, exit 2 when the whole-tree sweep collapses below FILE_FLOOR.
32"""
33
34from __future__ import annotations
35
36import re
37import sys
38from collections.abc import Iterable
39from pathlib import Path

41REPO_ROOT = Path(__file__).resolve().parents[2]
42
43SOURCE_SUFFIXES = (".c", ".h", ".cpp", ".hpp")
44SCAN_ROOT = "examples"
45
46# ((uint16_t)k_ra8_port_N << 8) | (uint16_t)k_ra8_pin_M, tolerant of spacing.
47ENCODING_RE = re.compile(r"k_ra8_port_\d+\s*<<\s*8\s*\‍)\s*\|\s*\‍(\s*uint16_t\s*\‍)\s*k_ra8_pin_\d+")
48
49# An examples/ tree this size cannot legitimately collapse to a handful of
50# files. If the whole-tree sweep returns less than this, something broke (an
51# unreachable repo root, a renamed SCAN_ROOT) and reporting "none hand-encode a
52# board pin" would be a lie: the idiom cannot be found in a file nobody read.
53# Measured 2026-07-28: 408 example sources. Same trip-wire as check_ruff.py.
54FILE_FLOOR = 320
55
56
57def _is_source(path: Path) -> bool:
58 return path.suffix in SOURCE_SUFFIXES
59
60
61def _rel(path: Path) -> str:
62 if path.is_relative_to(REPO_ROOT):
63 return str(path.relative_to(REPO_ROOT))
64 return str(path)
65
66
67def _enumerate_targets(arg_paths: Iterable[str]) -> list[Path]:
68 args = list(arg_paths)
69 if args:
70 out: list[Path] = []
71 for raw in args:
72 path = Path(raw)
73 if not path.is_absolute():
74 path = REPO_ROOT / path
75 if path.is_dir():
76 for suffix in SOURCE_SUFFIXES:
77 out.extend(path.rglob("*" + suffix))
78 elif _is_source(path):
79 out.append(path)
80 return out
81
82 out = []
83 for suffix in SOURCE_SUFFIXES:
84 out.extend((REPO_ROOT / SCAN_ROOT).rglob("*" + suffix))
85 return out

88def main(argv: list[str]) -> int:
89 """Fail any example that spells a board pin as an inline ``(port << 8) | pin``.
90
91 The match is textual, on the encoding SHAPE rather than on known pin
92 values, because the defect #251 found was the same four USB-FS pins
93 copy-pasted across 29 apps under a dozen different local names -- a value
94 allowlist would have had to know all twelve names, while the shift-or
95 pattern catches the next one regardless of what it is called.
96
97 Undecodable bytes are replaced rather than raising, so one bad file cannot
98 abort the sweep; encoding is check-encoding's gate, not this one's.
99
100 FILE_FLOOR applies to the whole-tree sweep ONLY, and exits 2 below it. An
101 argv file list comes from the pre-commit hook and legitimately filters to
102 nothing when a commit touches no example source, so an empty list there is
103 a real answer; an empty SWEEP is a broken enumeration reporting a clean
104 tree because it read nothing.
105
106 Returns 1 listing each site, 0 when clean or when argv filtered to nothing,
107 2 when the whole-tree sweep enumerated too few files to trust.
108 """
109 paths = argv[1:]
110 targets = _enumerate_targets(paths)
111 if not paths and len(targets) < FILE_FLOOR:
112 print(
113 f"check_example_board_pins.py: FATAL -- only {len(targets)} example file(s) "
114 f"in scope, floor is {FILE_FLOOR}. A collapsed sweep reports a clean tree "
115 "because it scanned nothing.",
116 file=sys.stderr,
117 )
118 return 2
119 if not targets:
120 print("check_example_board_pins.py: no files to scan", file=sys.stderr)
121 return 0
122
123 hits = []
124 for path in targets:
125 try:
126 text = path.read_text(encoding="utf-8", errors="replace")
127 except OSError:
128 continue
129 for lineno, line in enumerate(text.splitlines(), 1):
130 if ENCODING_RE.search(line):
131 hits.append((_rel(path), lineno, line.strip()))
132
133 if not hits:
134 print(
135 f"check_example_board_pins.py: {len(targets)} example file(s) "
136 "scanned, none hand-encode a board pin."
137 )
138 return 0
139
140 print(
141 f"check_example_board_pins.py: {len(hits)} hand-encoded board pin(s) in examples:\n",
142 file=sys.stderr,
143 )
144 for path, lineno, snippet in hits:
145 print(f" {path}:{lineno} {snippet}", file=sys.stderr)
146 print(
147 "\nThe EK-RA8D2 pinout belongs to the board layer, not to each app.\n"
148 "Reference the board symbol (k_ra8_board_*_pin_*, or an accessor like\n"
149 "ra8_board_sw_pin) instead of re-encoding (port << 8 | pin). If the pin\n"
150 "is a real board connector the board layer does not expose yet, add it\n"
151 "to libs/ra8_board_ek_ra8d2 first, then reference it here.",
152 file=sys.stderr,
153 )
154 return 1

157if __name__ == "__main__":
158 sys.exit(main(sys.argv))
