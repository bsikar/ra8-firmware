236{
237 *out = (ra8_fs_datetime_t){};

240 out->day = (uint8_t)k_fs_time_day_min;

242 if (s_clock_bound) {

313RA8_INTERNAL
314static uint8_t priv_utc_byte(int16_t minutes)
315{
316 if (minutes < (int16_t)k_fs_utc_span_min) {
317 return (uint8_t)k_fs_utc_unknown;
318 }
319 if (minutes > (int16_t)k_fs_utc_span_max) {
320 return (uint8_t)k_fs_utc_unknown;
321 }
322 if ((int32_t)minutes % (int32_t)k_fs_utc_step_min != 0) {
323 return (uint8_t)k_fs_utc_unknown;
324 }
325 const int32_t steps = (int32_t)minutes / (int32_t)k_fs_utc_step_min;
326 return (uint8_t)((uint32_t)k_fs_utc_valid_bit |
327 ((uint32_t)steps & (uint32_t)k_fs_utc_field_mask));
328}

353RA8_INTERNAL
354static void priv_fat_pack(const ra8_fs_datetime_t* t, fat_stamp_t* out)
355{
356 const uint32_t years = (uint32_t)t->year - (uint32_t)k_fs_time_epoch_year;
357 out->date =
358 (uint16_t)((years << (uint32_t)k_fs_date_shift_year) |
359 ((uint32_t)t->month << (uint32_t)k_fs_date_shift_month) | (uint32_t)t->day);
360 out->time = (uint16_t)(((uint32_t)t->hour << (uint32_t)k_fs_time_shift_hour) |
361 ((uint32_t)t->minute << (uint32_t)k_fs_time_shift_minute) |
362 ((uint32_t)t->second / (uint32_t)k_fs_time_second_div));
363 out->tenth = priv_tenths_of(t);
364}

420RA8_INTERNAL
421static void priv_exfat_pack(const ra8_fs_datetime_t* t, bool real, exfat_stamp_t* out)
422{
423 const uint32_t years = (uint32_t)t->year - (uint32_t)k_fs_time_epoch_year;
424 const uint32_t date = (years << (uint32_t)k_fs_xf_shift_year) |
425 ((uint32_t)t->month << (uint32_t)k_fs_xf_shift_month) | (uint32_t)t->day;
426 const uint32_t time = ((uint32_t)t->hour << (uint32_t)k_fs_time_shift_hour) |
427 ((uint32_t)t->minute << (uint32_t)k_fs_time_shift_minute) |
428 ((uint32_t)t->second / (uint32_t)k_fs_time_second_div);
429 out->stamp = (date << (uint32_t)k_fs_xf_shift_date) | time;
430 out->inc10 = priv_tenths_of(t);
431 /* An offset is a claim about a real instant. The epoch placeholder is not
432 * one, so saying "this is UTC" about it would be a fabrication -- exFAT's
433 * OffsetValid-clear encoding exists for exactly that case. A caller-supplied
434 * `utime` reading IS a real instant, so its offset is honoured. */
435 out->utc = real ? priv_utc_byte(t->utc_offset_min) : (uint8_t)k_fs_utc_unknown;
436}
