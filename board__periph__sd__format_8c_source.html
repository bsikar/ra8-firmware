41static void sd_put16(uint8_t* p, uint16_t v)
42{
43 p[k_le_byte0] = (uint8_t)(v & (uint16_t)k_le_byte_mask);

143void sd_label_field(uint8_t* dst, const char* label)
144{
145 bool past_end = (label == nullptr);
146 for (uint32_t i = 0U; i < (uint32_t)k_fmt_label_len; i++) {
147 if (!past_end && (label[i] == '\0')) {
148 past_end = true; /* stop reading at the NUL; pad the rest with spaces. */
149 }
150 dst[i] = past_end ? (uint8_t)' ' : (uint8_t)label[i];
151 }

154/* @brief Format an empty FAT16 volume; returns the chosen sectors-per-cluster. */
177static uint32_t fat16_solve_geometry(uint32_t total_sectors, uint32_t root_sectors, uint32_t* fatsz)
178{
179 uint32_t spc = 1U;
180 for (;;) {
181 const uint32_t tmp1 = total_sectors - ((uint32_t)k_fmt_resv_f16 + root_sectors);
182 const uint32_t tmp2 = ((uint32_t)k_bpb_fat16_div * spc) +
183 (uint32_t)k_bpb_num_fats_val; /* 256 = 512 / 2-byte ent. */
184 *fatsz = (tmp1 + tmp2 - 1U) / tmp2;
185 const uint32_t data = total_sectors - (uint32_t)k_fmt_resv_f16 - (2U * *fatsz) - root_sectors;
186 const uint32_t clus = data / spc;
187 if (clus <= (uint32_t)k_fmt_fat16_max) {
188 break;
189 }
190 spc *= 2U;
191 if (spc >= (uint32_t)k_fmt_spc_max) {
192 spc = (uint32_t)k_fmt_spc_max;
193 break;
194 }
195 }
196 return spc;
197}

215static void fat16_write_bpb(uint8_t* img,
216 uint32_t total_sectors,
217 uint32_t spc,
218 uint32_t fatsz,
219 const char* label)
220{
221 sd_boot_frame(img, "MSDOS5.0");
222 sd_put16(&img[k_bpb_bytes_per_sec], (uint16_t)k_fmt_sec_bytes);
223 img[k_bpb_sec_per_clus] = (uint8_t)spc;
224 sd_put16(&img[k_bpb_resv_sec], (uint16_t)k_fmt_resv_f16);
225 img[k_bpb_num_fats] = (uint8_t)k_bpb_num_fats_val;
226 sd_put16(&img[k_bpb_root_ents], (uint16_t)k_fmt_root_ents);
227 if (total_sectors < (uint32_t)k_fmt_totsec16_max) {
228 sd_put16(&img[k_bpb_totsec16], (uint16_t)total_sectors);
229 } else {
230 sd_put32(&img[k_bpb_totsec32], total_sectors);
231 }
232 img[k_bpb_media] = (uint8_t)k_bpb_media_fixed;
233 sd_put16(&img[k_bpb_fatsz16], (uint16_t)fatsz);
234 sd_put16(&img[k_bpb_sec_per_trk], (uint16_t)k_bpb_sec_per_trk_val);
235 sd_put16(&img[k_bpb_num_heads], (uint16_t)k_bpb_num_heads_val);
236 img[k_bpb16_drive_num] = (uint8_t)k_bpb_drive_num_val;
237 img[k_bpb16_boot_sig] = (uint8_t)k_bpb_ext_boot_sig;
238 sd_put32(&img[k_bpb16_volid], (uint32_t)k_fmt_volid | total_sectors);
239 sd_label_field(&img[k_bpb16_vol_label], label);
240 (void)memcpy(&img[k_bpb16_fstype], "FAT16 ", (size_t)k_bpb_fstype_len);
241}

243uint32_t sd_format_fat16(uint8_t* img, uint32_t total_sectors, const char* label)
244{
245 const uint32_t root_sectors = (((uint32_t)k_fmt_root_ents * (uint32_t)k_bpb_dir_ent_size) +
246 ((uint32_t)k_fmt_sec_bytes - 1U)) /
247 (uint32_t)k_fmt_sec_bytes;
248 uint32_t fatsz = 1U;
249 const uint32_t spc = fat16_solve_geometry(total_sectors, root_sectors, &fatsz);
250
251 fat16_write_bpb(img, total_sectors, spc, fatsz, label);
252
253 /* FAT[0] media + FAT[1] EOC, in both FAT copies. */
254 for (uint32_t f = 0U; f < (uint32_t)k_bpb_num_fats_val; f++) {
255 uint8_t* fat =
256 &img[(size_t)((uint32_t)k_fmt_resv_f16 + (f * fatsz)) * (uint32_t)k_fmt_sec_bytes];
257 sd_put16(&fat[k_le_byte0], (uint16_t)k_fat16_eoc_clus0);
258 sd_put16(&fat[k_fat_ent1_off], (uint16_t)k_fat16_eoc_clus1);
259 }
260 return spc;
261}

287static uint32_t
288sd_fat32_geometry(uint32_t total_sectors, uint32_t* out_fatsz, uint32_t* out_clusters)
289{
290 uint32_t spc = 1U;
291 uint32_t fatsz = 1U;
292 uint32_t clusters = 0U;
293 for (;;) {
294 const uint32_t tmp1 = total_sectors - (uint32_t)k_fmt_resv_f32;
295 /* Microsoft FAT32 FAT-size divisor: ((256*spc) + NumFATs) / 2 = 128*spc + 1.
296 * (Using +2 can undersize the FAT by a sector on large volumes.) */
297 const uint32_t tmp2 = ((uint32_t)k_bpb_fat32_div * spc) + 1U;
298 fatsz = (tmp1 + tmp2 - 1U) / tmp2;
299 const uint32_t data = total_sectors - (uint32_t)k_fmt_resv_f32 - (2U * fatsz);
300 clusters = data / spc;
301 /* Grow the cluster size for big cards so the FAT stays bounded and the
302 * cluster size is realistic (8 KiB at ~2 GiB, 32 KiB at ~8 GiB+), capped at
303 * the FAT32 cluster ceiling. k_fmt_fat32_pref keeps small cards at 1 spc. */
304 if (clusters <= (uint32_t)k_fmt_fat32_pref) {
305 break;
306 }
307 spc *= 2U;
308 if (spc >= (uint32_t)k_fmt_spc_max) {
309 spc = (uint32_t)k_fmt_spc_max;
310 break;
311 }
312 }
313 *out_fatsz = fatsz;
314 *out_clusters = clusters;
315 return spc;
316}

319uint32_t sd_format_fat32(uint8_t* img, uint32_t total_sectors, const char* label)
320{
321 uint32_t fatsz = 0U;
322 uint32_t clusters = 0U;
323 const uint32_t spc = sd_fat32_geometry(total_sectors, &fatsz, &clusters);
324 sd_boot_frame(img, "MSDOS5.0");
325 sd_put16(&img[k_bpb_bytes_per_sec], (uint16_t)k_fmt_sec_bytes);
326 img[k_bpb_sec_per_clus] = (uint8_t)spc;
327 sd_put16(&img[k_bpb_resv_sec], (uint16_t)k_fmt_resv_f32);
328 img[k_bpb_num_fats] = (uint8_t)k_bpb_num_fats_val;
329 sd_put16(&img[k_bpb_root_ents], 0U); /* root entries: 0 for FAT32 */
330 sd_put16(&img[k_bpb_totsec16], 0U); /* totsec16: 0, use totsec32 */
331 img[k_bpb_media] = (uint8_t)k_bpb_media_fixed;
332 sd_put16(&img[k_bpb_fatsz16], 0U); /* fatsz16: 0 for FAT32 */
333 sd_put16(&img[k_bpb_sec_per_trk], (uint16_t)k_bpb_sec_per_trk_val);
334 sd_put16(&img[k_bpb_num_heads], (uint16_t)k_bpb_num_heads_val);
335 sd_put32(&img[k_bpb_totsec32], total_sectors);
336 sd_put32(&img[k_bpb_fatsz32], fatsz); /* BPB_FATSz32 */
337 sd_put32(&img[k_bpb32_root_clus], (uint32_t)k_fat32_root_clus);
338 sd_put16(&img[k_bpb32_fsinfo_sec], (uint16_t)k_fat32_fsinfo_sec);
339 sd_put16(&img[k_bpb32_bkboot_sec], (uint16_t)k_fat32_bkboot_sec);
340 img[k_bpb32_drive_num] = (uint8_t)k_bpb_drive_num_val;
341 img[k_bpb32_boot_sig] = (uint8_t)k_bpb_ext_boot_sig;
342 sd_put32(&img[k_bpb32_volid], (uint32_t)k_fmt_volid | total_sectors);
343 sd_label_field(&img[k_bpb32_vol_label], label);
344 (void)memcpy(&img[k_bpb32_fstype], "FAT32 ", (size_t)k_bpb_fstype_len);
345 /* FSInfo sector (lead + struct + trail signatures). */
346 uint8_t* fsi = &img[(size_t)(uint32_t)k_fat32_fsinfo_sec * (uint32_t)k_fmt_sec_bytes];
347 sd_put32(&fsi[k_fsi_off_lead], (uint32_t)k_fsi_lead_sig);
348 sd_put32(&fsi[k_fsi_off_struc], (uint32_t)k_fsi_struc_sig);
349 sd_put32(&fsi[k_fsi_off_free],
350 (clusters > 0U) ? (clusters - 1U) : (uint32_t)k_fsi_free_unknown); /* free clusters */
351 sd_put32(&fsi[k_fsi_off_nxtfree], (uint32_t)k_fsi_nxt_free); /* next free cluster */
352 sd_put32(&fsi[k_fsi_off_trail], (uint32_t)k_fsi_trail_sig);
353 /* Backup boot copy at sector 6. */
354 (void)memcpy(&img[(size_t)(uint32_t)k_fat32_bkboot_sec * (uint32_t)k_fmt_sec_bytes],
355 img,
356 (size_t)k_fmt_sec_bytes);
357 /* FAT[0..2]: media + EOC + root-cluster EOC, in both FAT copies. */
358 for (uint32_t f = 0U; f < (uint32_t)k_bpb_num_fats_val; f++) {
359 uint8_t* fat =
360 &img[(size_t)((uint32_t)k_fmt_resv_f32 + (f * fatsz)) * (uint32_t)k_fmt_sec_bytes];
361 sd_put32(&fat[k_le_byte0], (uint32_t)k_fat32_eoc_media);
362 sd_put32(&fat[k_fat32_ent1_off], (uint32_t)k_fat32_eoc);
363 sd_put32(&fat[k_fat32_ent2_off], (uint32_t)k_fat32_eoc);
364 }
365 return spc;
