104static uint16_t rgb888_to_565(uint32_t rgb)
105{
106 const uint32_t r = (rgb >> (uint32_t)k_rgb888_r_shift) & (uint32_t)k_byte_mask;
107 const uint32_t g = (rgb >> (uint32_t)k_rgb888_g_shift) & (uint32_t)k_byte_mask;
108 const uint32_t b = rgb & (uint32_t)k_byte_mask;
109 return (uint16_t)(((r & (uint32_t)k_rgb565_r5_keep) << (uint32_t)k_rgb565_r_pos) |
110 ((g & (uint32_t)k_rgb565_g6_keep) << (uint32_t)k_rgb565_g_pos) |
111 (b >> (uint32_t)k_rgb565_b_drop));
112}

138static void build_frame(uc_engine* uc, uint16_t* fb, uint16_t width_px, uint16_t height_px)
139{
140 /* Read GLCDC registers from the stable shadow (mmio_peek), never the guest-
141 * facing toggling read -- otherwise a firmware that never programs the GLCDC
142 * makes the panel strobe black<->white (see mmio_peek). */
143 const uint16_t bg = rgb888_to_565(mmio_peek((uint64_t)k_glcdc_bg_bgc) & (uint32_t)k_rgb888_mask);
144 const size_t n = (size_t)width_px * (size_t)height_px;
145 for (size_t i = 0U; i < n; i++) {
146 fb[i] = bg;
147 }
148
149 const uint32_t saddr = mmio_peek((uint64_t)k_glcdc_gr1_saddr);
150 const uint32_t fmt = (mmio_peek((uint64_t)k_glcdc_gr1_fmt) >> (uint32_t)k_glcdc_fmt_shift) &
151 (uint32_t)k_glcdc_fmt_mask;
152 if (!addr_is_ram(saddr) || (fmt != (uint32_t)k_glcdc_fmt_rgb565)) {
153 return; /* no graphics layer -- background-only frame */
154 }
155 const uint32_t stride = (mmio_peek((uint64_t)k_glcdc_gr1_flm3) >> (uint32_t)k_glcdc_high_shift) &
156 (uint32_t)k_glcdc_stride_mask;
157 const uint32_t lnnum = (mmio_peek((uint64_t)k_glcdc_gr1_flm5) >> (uint32_t)k_glcdc_high_shift) &
158 (uint32_t)k_glcdc_lnnum_mask;
159 if (stride < 2U) {
160 return;
161 }
162 const uint32_t fb_w = stride / 2U; /* RGB565: 2 bytes per pixel */
163 const uint32_t fb_h = lnnum + 1U;
164 const uint32_t cw = (fb_w < (uint32_t)width_px) ? fb_w : (uint32_t)width_px;
165 const uint32_t ch = (fb_h < (uint32_t)height_px) ? fb_h : (uint32_t)height_px;
166 for (uint32_t y = 0U; y < ch; y++) {
167 (void)uc_mem_read(uc,
168 (uint64_t)saddr + ((uint64_t)y * (uint64_t)stride),
169 &fb[(size_t)y * (size_t)width_px],
170 (size_t)cw * sizeof(uint16_t));
171 }
172}

175int write_ppm(const char* path, const uint16_t* fb, uint16_t width_px, uint16_t height_px)
176{
177 FILE* f = fopen(path, "wb"); /* alloc-allow: host dev tool, not firmware */
178 if (f == nullptr) {
179 return -1;
180 }
181 (void)fprintf(f, "P6\n%u %u\n255\n", (unsigned)width_px, (unsigned)height_px);
182 const size_t n = (size_t)width_px * (size_t)height_px;
183 for (size_t i = 0U; i < n; i++) {
184 const uint16_t p = fb[i];
185 const uint32_t r5 = (uint32_t)((p >> (uint32_t)k_rgb565_r_shift) & (uint32_t)k_rgb565_5bit);
186 const uint32_t g6 = (uint32_t)((p >> (uint32_t)k_rgb565_g_shift) & (uint32_t)k_rgb565_6bit);
187 const uint32_t b5 = (uint32_t)(p & (uint32_t)k_rgb565_5bit);
188 const uint8_t rgb[3] = {(uint8_t)((r5 << 3) | (r5 >> 2)),
189 (uint8_t)((g6 << 2) | (g6 >> 4)),
190 (uint8_t)((b5 << 3) | (b5 >> 2))};
191 (void)fwrite(rgb, 1U, 3U, f);
192 }
193 (void)fclose(f);
194 return 0;
195}

276static uint32_t console_advance_scroll(uint32_t avail, uint32_t total)
277{
278 if (s_view_autoscroll) {
279 s_view_scroll = 0U; /* follow the live tail */
280 } else if (total > s_view_log_seen) {
281 const uint32_t delta = total - s_view_log_seen;
282 s_view_scroll += delta;
283 if (s_view_scroll > (uint32_t)k_uart_log_max) {
284 s_view_scroll = (uint32_t)k_uart_log_max;
285 }
286 }
287 s_view_log_seen = total;
288 const uint32_t max_scroll =
289 (avail > (uint32_t)k_overlay_console_rows) ? (avail - (uint32_t)k_overlay_console_rows) : 0U;
290 if (s_view_scroll > max_scroll) {
291 s_view_scroll = max_scroll; /* re-clamp if history shrank or is shallow */
292 }
293 return s_view_scroll;
294}

296static void fill_status_console(board_status_t* st)
297{
298 fill_console_tabs(st);
299 /* Console window: copy up to k_overlay_console_rows lines from the active
300 * channel's scrollback ring, starting `scroll` lines back from the newest,
301 * so the mouse-wheel can page through history. console[0] is the newest
302 * visible line (= ring line `scroll`). */
303 const uint32_t avail = board_console_count(s_view_console_ch);
304 const uint32_t total = board_console_total(s_view_console_ch);
305 const uint32_t scroll = console_advance_scroll(avail, total);
306 uint32_t rows = 0U;
307 for (uint32_t i = 0U; i < (uint32_t)k_overlay_console_rows; i++) {
308 const char* line = board_console_line(s_view_console_ch, scroll + i);
309 if (line == nullptr) {
310 break;
311 }
312 (void)snprintf(st->console[i], (size_t)k_overlay_line_cap, "%s", line);
313 rows++;
314 }
315 st->console_count = rows;
316 st->console_scroll = scroll;
317 st->console_total = avail;
318 st->console_autoscroll = s_view_autoscroll;
319 /* Run-state telemetry (set by the run loop before each present). */
320 st->pc = s_view_pc;
321 st->chunks = s_view_chunks;
322 st->running = s_view_running;
323 st->mmio_reads = emu_mmio_reads();
324 st->mmio_writes = emu_mmio_writes();
325 st->uart_tx_total = board_periph_uart_tx_total();
326}

357static void rotate_panel(const uint16_t* src, uint16_t sw, uint16_t sh, uint16_t* dst, uint32_t deg)
358{
359 for (uint16_t y = 0U; y < sh; y++) {
360 for (uint16_t x = 0U; x < sw; x++) {
361 const uint16_t p = src[((size_t)y * (size_t)sw) + (size_t)x];
362 uint32_t dx = x;
363 uint32_t dy = y;
364 uint32_t dw = sw;
365 if (deg == (uint32_t)k_rotate_90) {
366 dx = (uint32_t)(sh - 1U - y);
367 dy = x;
368 dw = sh;
369 } else if (deg == (uint32_t)k_rotate_180) {
370 dx = (uint32_t)(sw - 1U - x);
371 dy = (uint32_t)(sh - 1U - y);
372 } else if (deg == (uint32_t)k_rotate_270) {
373 dx = y;
374 dy = (uint32_t)(sw - 1U - x);
375 dw = sh;
376 }
377 dst[((size_t)dy * (size_t)dw) + (size_t)dx] = p;
378 }
379 }
380}

407void set_switch(board_overlay_btn_t btn, bool pressed)
408{
409 const uint8_t pin =

411 /* SW1/SW2 are momentary push-buttons wired active-low (internal pull-up): held

467board_overlay_btn_t route_click(uint16_t cx,
468 uint16_t cy,
469 uint16_t panel_w,
470 uint16_t panel_h,
471 uint16_t disp_w,
472 uint32_t rotate_deg)
473{
474 /* A click on the console tab bar switches the active channel (and takes
475 * precedence over the console-body autoscroll toggle below, since the tab row
476 * sits inside the console region). Switching resets the scrollback to the live
477 * tail of the newly selected channel. */
478 uint32_t tab_idx = 0U;
479 if (board_overlay_hit_console_tab(cx, cy, disp_w, (uint32_t)k_board_console_ch_count, &tab_idx)) {
480 s_view_console_ch = (board_console_ch_t)tab_idx;
481 s_view_scroll = 0U;
482 s_view_autoscroll = true;
483 s_view_log_seen = board_console_total(s_view_console_ch);

485 }
486 const board_overlay_btn_t btn = board_overlay_hit_button(cx, cy, disp_w);
487 if (btn == k_board_overlay_btn_console) {
488 /* Toggle the console autoscroll, Arduino-Serial-Monitor style: pause it
489 * while reading, or click again to jump back to the live tail. */
490 s_view_autoscroll = !s_view_autoscroll;
491 if (s_view_autoscroll) {
492 s_view_scroll = 0U; /* resume following the newest line */
493 }
494 return btn;
495 }
496 if ((btn == k_board_overlay_btn_battery) || (btn == k_board_overlay_btn_batt_chg) ||
497 (btn == k_board_overlay_btn_lowpower)) {
498 apply_battery_click(btn, cx, disp_w);
499 return btn;
500 }
501 if (btn != k_board_overlay_btn_none) {
502 set_switch(btn, true); /* mouse-down -> press; mouse-up releases it (run loop). */
503 return btn;
504 }
505 uint16_t nx = cx;
506 uint16_t ny = cy;
507 unrotate_click(cx, cy, panel_w, panel_h, rotate_deg, &nx, &ny);
508 board_periph_touch_inject(nx, ny);
509 return k_board_overlay_btn_none;
510}

569static bool panel_split_kv(char* line, char** key, char** val)
570{
571 char* p = line;
572 while ((*p == ' ') || (*p == '\t')) {
573 p++;
574 }
575 if ((*p == '#') || (*p == '\0') || (*p == '\n') || (*p == '\r')) {
576 return false;
577 }
578 char* eq = strchr(p, '=');
579 if (eq == nullptr) {
580 return false;
581 }
582 *eq = '\0';
583 char* v = eq + 1;
584 while ((*v == ' ') || (*v == '\t')) {
585 v++;
586 }
587 panel_rstrip(p);
588 panel_rstrip(v);
589 *key = p;
590 *val = v;
591 return true;
592}

663bool load_panel(const char* path, board_panel_t* out)
664{
665 (void)memset(out, 0, sizeof(*out));
666 FILE* f = fopen(path, "r"); /* alloc-allow: host dev tool, not firmware */
667 if (f == nullptr) {
668 (void)fprintf(stderr, "ra8_emulator: cannot open panel config %s\n", path);
669 return false;
670 }
671 char line[k_panel_line_max];
672 while (fgets(line, (int)sizeof(line), f) != nullptr) {
673 char* key = nullptr;
674 char* val = nullptr;
675 if (panel_split_kv(line, &key, &val)) {
676 panel_apply_kv(out, key, val);
677 }
678 }
679 (void)fclose(f);
680 if ((out->width == 0U) || (out->width > (uint16_t)k_panel_dim_max) || (out->height == 0U) ||
681 (out->height > (uint16_t)k_panel_dim_max)) {
682 (void)fprintf(stderr, "ra8_emulator: panel %s: width/height missing or out of range\n", path);
683 return false;
684 }
685 return true;
686}

703void emu_view_wheel(int32_t notches)
704{
705 if (notches > 0) {
706 s_view_autoscroll = false; /* user scrolled up -> pause the live follow */
