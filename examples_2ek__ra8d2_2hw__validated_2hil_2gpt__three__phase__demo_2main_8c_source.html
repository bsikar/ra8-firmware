139 ra8_isr_globals_enable();

143 }
144
145 /* Cycle the three-phase duties via ra8_gpt_three_phase_set_duty
146 * once per iteration. The call exercises GTCCR writes on all
147 * three channels and proves the driver's open state is intact.
148 * On a scope the U/V/W outputs would sweep their duty cycles in
149 * lockstep; on the HIL bench we just count successful set_duty
150 * calls. */
151 uint32_t duty_offset = 0U;
152 while (1) {
153 const uint32_t period = (uint32_t)k_gpt_3p_demo_period;
154 const uint32_t u_duty = ((uint32_t)k_gpt_3p_demo_duty_u + duty_offset) & period;
155 const uint32_t v_duty = ((uint32_t)k_gpt_3p_demo_duty_v + duty_offset) & period;
156 const uint32_t w_duty = ((uint32_t)k_gpt_3p_demo_duty_w + duty_offset) & period;
157 if (ra8_gpt_three_phase_set_duty(u_duty, v_duty, w_duty) == k_ra8_ok) {
158 g_gpt_three_phase_match += 1U;
159 (void)ra8_board_led_toggle(k_ra8_board_led1);
160 } else {
161 g_gpt_three_phase_mismatch += 1U;
162 }
163 duty_offset = (duty_offset + k_gpt_duty_advance) & period;
164 ra8_delay_ms((uint32_t)k_gpt_3p_demo_step_ms);
165 }
166 gpt_3p_demo_panic_halt();
167 return 0;
168}
