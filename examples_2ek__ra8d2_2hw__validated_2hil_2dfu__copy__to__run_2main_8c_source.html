138 * path loudly (and freeze the heartbeat) if a future edit breaks either pin. */
139 if ((uintptr_t)&g_dcr_run_sentinel != (uintptr_t)k_dcr_probe_sentinel_addr) {
140 dcr_panic_halt();
141 }
142 if ((uintptr_t)&g_dcr_run_heartbeat != (uintptr_t)k_dcr_probe_counter_addr) {
143 dcr_panic_halt();
144 }
145
146 uint32_t initial_sp = 0U;
147 uint32_t reset_entry = 0U;
148 memcpy(&initial_sp, &g_payload_image[0], sizeof(initial_sp));
149 memcpy(&reset_entry, &g_payload_image[sizeof(initial_sp)], sizeof(reset_entry));
150 /* Preconditions (NASA Rule 5): the embedded image must carry a plausible SRAM
151 * stack pointer and a Thumb reset vector before we hand off to it. */
152 if ((initial_sp & (uint32_t)k_dcr_sram_mask) != (uint32_t)k_dcr_sram_base) {
153 dcr_panic_halt();
154 }
155 if ((reset_entry & (uint32_t)k_dcr_thumb_bit) == 0U) {
156 dcr_panic_halt();
157 }
158
159 /* Copy-to-run: ra8_dfu_launch copies the image to k_ra8_dfu_run_base and branches
160 * there. It returns only if the run-target check fails (it should not). */
161 ra8_dfu_launch((uintptr_t)g_payload_image,
162 (uint32_t)sizeof(g_payload_image),
