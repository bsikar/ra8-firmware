309int32_t main(void)
310{
311 lpm_wake_setup_or_halt();
312 ra8_isr_globals_enable();
313
314 /* Boot banner -- HIL gate. */
315 (void)ra8_board_uart_console_write(k_lpm_wake_boot_msg,
316 (size_t)(sizeof(k_lpm_wake_boot_msg) - 1U));
317
318 if (lpm_wake_walk_wupen0() != k_ra8_ok) {
319 lpm_wake_panic_halt();
320 }
321 if (lpm_wake_walk_wupen1() != k_ra8_ok) {
322 lpm_wake_panic_halt();
323 }
324 if (lpm_wake_disarm_all() != k_ra8_ok) {
325 lpm_wake_panic_halt();
326 }
327
328 /* Final-state banner -- emit so the operator sees the walk
329 * completed even when no debugger is attached. */
330 (void)ra8_board_uart_console_write(k_lpm_wake_done_msg,
331 (size_t)(sizeof(k_lpm_wake_done_msg) - 1U));
332
333 (void)ra8_board_led_on(k_ra8_board_led1);
334
335 while (1) {
336 __asm__ volatile("wfi");
337 }
338 return 0;
339}
