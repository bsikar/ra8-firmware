166 /* Pre 1: this function is called from SystemInit, BEFORE the .data
167 * copy / .bss zero. We must not read or write any non-volatile
168 * globals -- the diagnostic stamps below are volatile. */
169 /* Pre 2: the boot ROM left SAU disabled and IDAU in reset state. */
170
171 const uint32_t* ns_vt = (const uint32_t*)(uintptr_t)k_ns_vector_table_addr;
172
173 const ra8_err_t err = ra8_tz_secure_boot_sau_init();
174 if (err != k_ra8_ok) {
175 ra8_log_error_val("CPU1IPC", "sau_init failed", (uint32_t)err);
176 const ra8_tz_secure_boot_step_t step_fail = ra8_tz_secure_boot_get_step();
177 g_cpu1_pingpong_ipc_tz_step = (uint8_t)step_fail;
178 return;
179 }
180
181 const ra8_err_t sec_err =
182 ra8_tz_secure_boot_security_init((uint32_t)k_ipcsar_value, (uint32_t)k_ipcpar_value);
183 if (sec_err != k_ra8_ok) {
184 /* Do NOT release CPU1 or BLXNS when the IPC channel S/NS attribution
185 * failed -- the NS core would face Secure-locked IPC channels. */
186 ra8_log_error_val("CPU1IPC", "security_init failed", (uint32_t)sec_err);
187 const ra8_tz_secure_boot_step_t step_sec = ra8_tz_secure_boot_get_step();
188 g_cpu1_pingpong_ipc_tz_step = (uint8_t)step_sec;
189 return;
190 }
191 /* HUM Ch 3.2.1 "IPCSAR" p 205 -- read-back so a J-Link memprobe can
192 * confirm the write landed even if BLXNS later wedges. */
193 g_cpu1_pingpong_ipc_ipcsar_post = *(volatile uint32_t*)0x40008610UL;
194 const ra8_tz_secure_boot_step_t step_ok = ra8_tz_secure_boot_get_step();
195 g_cpu1_pingpong_ipc_tz_step = (uint8_t)step_ok;
196
197 internal_release_cpu1();
198
199 /* On hardware ra8_tz_secure_boot_jump_ns does not return. On host
200 * the BLXNS is captured into the secure-boot library's state for
201 * the unit tests to inspect. */
202 const ra8_err_t jump_err = ra8_tz_secure_boot_jump_ns(ns_vt);
203 if (jump_err != k_ra8_ok) {
204 /* Reached on hardware ONLY when the NS image was DENIED (vector-table
205 * validation failed): stamp the step for the J-Link memprobe and fall
206 * back to the S-side main() -- never treat the NS world as live. */
207 ra8_log_error_val("CPU1IPC", "jump_ns denied", (uint32_t)jump_err);
208 const ra8_tz_secure_boot_step_t step_denied = ra8_tz_secure_boot_get_step();
209 g_cpu1_pingpong_ipc_tz_step = (uint8_t)step_denied;
210 return;
211 }
