204static ra8_err_t demo_roundtrip(void)
205{
206 uint8_t data[(size_t)k_demo_payload];
207 for (uint32_t i = 0; i < (uint32_t)k_demo_payload; ++i) {
208 data[i] = (uint8_t)((i * (uint32_t)k_demo_seed_mul) + (uint32_t)k_demo_seed_add);
209 }
210 RA8_RETURN_ON_ERROR(
211 ra8_io_blockdev_erase(&s_bd, (uint32_t)k_demo_test_lba, (uint32_t)k_demo_block_count),
212 s_tag,
213 "erase");
214 RA8_RETURN_ON_ERROR(
215 ra8_io_blockdev_write(&s_bd, (uint32_t)k_demo_test_lba, (uint32_t)k_demo_block_count, data),
216 s_tag,
217 "write");
218 uint8_t got[(size_t)k_demo_payload] = {};
219 RA8_RETURN_ON_ERROR(
220 ra8_io_blockdev_read(&s_bd, (uint32_t)k_demo_test_lba, (uint32_t)k_demo_block_count, got),
221 s_tag,
222 "read");
223 if (memcmp(got, data, sizeof(data)) != 0) {
224 return k_ra8_err_checksum_mismatch;
225 }
226 return k_ra8_ok;
227}

246int main(void)
247{

262 }

264 while (true) {
265 }
266}
