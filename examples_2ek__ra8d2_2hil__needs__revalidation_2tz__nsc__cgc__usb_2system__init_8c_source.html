131static void internal_set_vtor(void)

289[[maybe_unused]] static void internal_mpu_init(void)
290{
291 enum : uint32_t {
292 k_ra8_mpu_ctrl_enable = 1UL << 0,
293 k_ra8_mpu_ctrl_privdefena = 1UL << 2,
294 k_ra8_mair0_default = 0x000404FFUL,
295 };
296
297 /* MAIR0: idx 0 = WB/WA, idx 1 = non-cacheable, idx 2 = device-nGnRE. */
298 internal_write32(k_ra8_mpu_mair0_addr, k_ra8_mair0_default);
299 internal_write32(k_ra8_mpu_mair1_addr, 0U);
300
301 internal_mpu_set_region(0U,
302 (k_ra8_mpu_mram_base | k_ra8_rbar_attr_ro_x),
303 (k_ra8_mpu_mram_limit | k_ra8_mpu_rlar_enable));
304 internal_mpu_set_region(1U,
305 (k_ra8_mpu_sram_base | k_ra8_rbar_attr_rw_xn),
306 (k_ra8_mpu_sram_limit | k_ra8_mpu_rlar_enable));
307 internal_mpu_set_region(2U,
308 (k_ra8_mpu_sdram_base | k_ra8_rbar_attr_rw_xn),
309 (k_ra8_mpu_sdram_limit | k_ra8_mpu_rlar_enable));
310 internal_mpu_set_region(3U,
311 (k_ra8_mpu_peri_base | k_ra8_rbar_attr_device_rw),
312 (k_ra8_mpu_peri_limit | k_ra8_mpu_rlar_enable));

314 /* Enable MPU with PRIVDEFENA so unclassified privileged accesses

350/* =============================================================================
351 * Public entry point
352 * =============================================================================
353 *
354 * Called from `Reset_Handler` before the .data copy and .bss zero.
355 * Must therefore not touch any global variables -- everything here
356 * runs on the stack and writes to CPU / SCB memory only.
357 */
358
359/* NOLINTNEXTLINE(readability-identifier-naming) -- CMSIS-mandated name. */
360void SystemInit(void)
361{
362 /* Mask global IRQs so the application gets a deterministic init
363 * window. main() must call ``ra8_isr_globals_enable`` once every
364 * driver / handler is wired up before any IRQ source can fire. */
365 internal_disable_irq();
366
367 internal_set_vtor();
368 internal_enable_fpu();
369 internal_enable_fpu_lazy_stack();
370 /* Cache enable, MPU init, and TrustZone bring-up are temporarily
371 * disabled -- they HardFault on first reset because the CCSIDR-driven
372 * invalidate-by-set/way loop is not yet implemented and the MPU
373 * regions are unconfigured. Re-enable once those code paths land. */
374 (void)internal_enable_icache;
375 (void)internal_enable_dcache;
376 (void)internal_enable_branch_predictor;
377 (void)internal_mpu_init;
378 internal_set_priority_grouping();
379
380 /* Bring up the Secure clock tree (XTAL -> PLL1 -> CPUCLK0 = 1 GHz) BEFORE the
381 * SAU + BLXNS below. ra8_trustzone_init() does NOT return on hardware (BLXNS
382 * leaves Secure thread mode), so anything sequenced after it never runs on the
383 * Secure side. The NS-side NSC CGC veneers (ra8_nsc_cgc_pll2_enable etc.) trap
384 * back into this Secure CGC driver, which needs PLL1 already locked -- without
385 * it the first PLL2 enable fails because its PLL1 source is missing. */
386 if (ra8_cgc_init() != k_ra8_ok) {
387 /* NASA Rule 7: a failed Secure clock bring-up must not BLXNS into NS code
388 * on a dead clock tree -- park at a named symbol instead of switching. */
389 internal_halt_secure_clock_fault();
390 }
391
392#ifdef RA8_TRUSTZONE_ENABLE
393 /* Programme the SAU, then BLXNS into the NS image. The NSC veneers depend on
394 * the SAU being live so the cmse_nonsecure_entry traps land on the Secure
395 * side. Does not return on hardware. */
396 ra8_trustzone_init();
397#else
398 (void)ra8_trustzone_init;
399#endif
400}
