4"""Gate: no magic numbers -- every integer literal belongs in a named enum.
5
6Floating-point constants (which C enums cannot hold) use a
7``const``. Macros are NOT an acceptable home for an integer constant
8(CLAUDE.md "Constants and Macros": enums always, macros only for code
9de-duplication, conditional compilation, or build-configuration flags).
10
11The project's `.clang-tidy` already configures
12``readability-magic-numbers``, but clang-tidy only checks files that are
13present in the build's ``compile_commands.json``. ``clang_tidy.sh`` runs
14against the host unit-test build, which drops every ARM-cross-compiled
15translation unit and -- crucially -- contains **no** example ``main.c``
16at all. The result was that every ``examples/<tier>/.../<app>/main.c``
17was invisible to the magic-number check (a bare ``ra8_delay_ms(500U)``
18sailed straight through both the pre-commit hook and CI).
19
20This checker is a backstop that walks the source text directly so the
21rule is enforced for **every** ``.c`` file under ``libs/``, ``src/``,
22``port/``, and ``examples/`` regardless of which compile database it
23ended up in. It is the magic-number analogue of
24``check_function_size.py``.
25
26Detection rules (kept deliberately aligned with the ``.clang-tidy``
27``readability-magic-numbers`` configuration):
28
29* Numeric literals (decimal, ``0x`` hex, ``0b`` binary, octal, and
30 floating-point, with optional ``u``/``l``/``f`` suffixes) are flagged.
31* The ignored-value set matches ``.clang-tidy``:
32 integers ``0;1;2;3;4;6;8;16;32`` and floats ``0.0;1.0``.
33* Literals inside an ``enum`` body are the *allowed* home for a constant
34 and are never flagged.
35* Preprocessor directives (``#define``, ``#if`` ...) are skipped here --
36 but note that an object-like ``#define FOO 500`` is itself a policy
37 violation (integer constants must be enums); the pre-commit hook's
38 "bare numeric #define" gate is what flags that pattern.
39* Numbers embedded in identifiers (``uint8_t``, ``crc32``, ``s_buf2``)
40 are not literals and are never flagged.
41* Comments and string / character literals are stripped before scanning.
42* clang-tidy ``NOLINT`` suppressions are honoured: a
43 ``NOLINT`` / ``NOLINTNEXTLINE`` / ``NOLINTBEGIN`` .. ``NOLINTEND`` that
44 is bare or lists ``readability-magic-numbers`` (or the
45 ``cppcoreguidelines-avoid-magic-numbers`` alias) suppresses the same
46 lines a clang-tidy run would. A backstop for clang-tidy must respect
47 clang-tidy's own waiver mechanism -- the codebase uses it heavily to
48 exempt the crypto, JPEG-codec, and BLE-host translation units.
49
50Per-line opt-out: append ``MAGIC-OK: <reason>`` to a line to suppress
51it (mirrors the ``LEGACY-OK`` / ``WAVE-OK`` / ``AI-OK`` opt-outs used by
52the sibling gates). Use it sparingly and only with a written reason;
53prefer a scoped ``NOLINT`` where a whole block is genuinely exempt.
54
55Run::
56
57 check_magic_numbers.py # scan the whole tree
58 check_magic_numbers.py path/to/file.c ... # scan listed files
59
60Exit 0 if no magic numbers remain, exit 1 (with a diagnostic table) if
61any are found.
62"""

135def _nolint_applies(arg: str | None) -> bool:
136 """Whether a NOLINT marker suppresses the magic-number check specifically.
137
138 A bare ``NOLINT`` with no argument list suppresses EVERY check, so it
139 counts here too -- reading it as "suppresses some other check" would let
140 a blanket suppression silently cover this one.
141 """
142 if arg is None or arg.strip() == "":
143 return True
144 return any(name in arg for name in MAGIC_CHECK_NAMES)

147# A numeric literal: hex, binary, octal, float, or decimal, each with an
148# optional integer/float suffix. Order matters -- hex/binary/float must
149# be tried before the bare-decimal alternative.
150_NUM_RE = re.compile(
151 r"""
152 (?P<hex>0[xX][0-9a-fA-F]+(?:[uUlL]*))
153 | (?P<bin>0[bB][01]+(?:[uUlL]*))
154 | (?P<flt>(?:\d+\.\d*|\.\d+|\d+)(?:[eE][+-]?\d+)?[fF])
155 | (?P<dec_flt>(?:\d+\.\d*|\.\d+)(?:[eE][+-]?\d+)?[lL]?)
156 | (?P<dec>\d+(?:[uUlL]*))
157 """,
158 re.VERBOSE,
159)
160
161# Characters that, immediately before a match, mean the digits are part
162# of an identifier (uint8_t, crc32) or a larger numeric token rather
163# than a standalone literal.
164_IDENT_BEFORE = re.compile(r"[0-9A-Za-z_.]")
165
166# A "data row": a line (after comment/string stripping) that holds only
167# numeric literals, commas, braces, signs and whitespace -- i.e. a row
168# of a const lookup table (font glyphs, JPEG quant tables, crypto
169# S-boxes, MIPI PHY register sequences). These are data, not logic; the
170# magic-number rule is meaningless for them, exactly as clang-tidy never
171# saw them (the data-table TUs are absent from the host compile-db). A
172# real magic number always rides on an identifier or operator
173# (`ra8_delay_ms(500U)`, `buf[0] = 500`, `x << 7`), which breaks the
174# pattern below and is still flagged.
175_DATA_ROW_RE = re.compile(r"^[\s{}\‍[\‍],0-9a-fA-FxXuUlL.+\-]*$")
176
177# A declaration line carries a storage class, qualifier, or type token.
178# clang-tidy ignores an array-*dimension* literal in a declaration
179# (``uint8_t z[64]``) but still flags an array *subscript* (``b[5]``);
180# the dimension only counts as a magic number on a declaration line.
181_DECL_HINT = re.compile(
182 r"\b(static|const|extern|volatile|register|struct|union|enum|"
183 r"unsigned|signed|void|char|short|int|long|float|double|bool|"
184 r"[A-Za-z_][A-Za-z0-9_]*_t)\b" # uint8_t, int32_t, size_t, my_type_t, ...
185)
186
187# A `const` float/double definition is the sanctioned home for a
188# floating-point constant (C enums cannot hold floats -- see CLAUDE.md
189# "Constants and Macros"). A float literal initialising such a
190# definition is the named value itself, analogous to an enum member, and
191# is not a magic number -- exactly as `1.0`/`0.0` are already ignored.
192_CONST_FLOAT_DEF = re.compile(r"\bconst\b")
193_FLOAT_TYPE = re.compile(r"\b(float|double)\b")

196def _strip_comments_and_strings(text: str) -> str:
197 """Blank comment and literal bytes to spaces, preserving every position.
198
199 Newlines survive, so line AND column numbers reported against the blanked
200 view remain valid for the original source -- this gate reports columns,
201 not just lines, so length preservation is required and not merely tidy.
202 """
203 out: list[str] = []
204 i = 0
205 n = len(text)
206 while i < n:
207 c = text[i]
208 nxt = text[i + 1] if i + 1 < n else ""
209 if c == "/" and nxt == "/":
210 while i < n and text[i] != "\n":
211 i += 1
212 continue
213 if c == "/" and nxt == "*":
214 out.append(" ")
215 i += 2
216 while i < n and not (text[i] == "*" and i + 1 < n and text[i + 1] == "/"):
217 out.append("\n" if text[i] == "\n" else " ")
218 i += 1
219 if i < n:
220 out.append(" ")
221 i += 2
222 continue
223 if c in {'"', "'"}:
224 quote = c
225 out.append(" ")
226 i += 1
227 while i < n and text[i] != quote:
228 if text[i] == "\\" and i + 1 < n:
229 out.append(" ")
230 i += 2
231 continue
232 out.append("\n" if text[i] == "\n" else " ")
233 i += 1
234 if i < n:
235 out.append(" ")
236 i += 1
237 continue
238 out.append(c)
239 i += 1
240 return "".join(out)

243def _literal_value(match: re.Match) -> tuple[bool, float]:
244 """Decode a numeric-literal match to ``(is_float, value)``.
245
246 Returns ``(False, NaN)`` for anything un-parseable, and the caller treats
247 NaN as "do not flag" -- an unrecognised literal form is not evidence of a
248 magic number.
249
250 Suffix stripping is type-aware: ``f`` is never stripped from a hex
251 literal, where it is a digit (0xAF) rather than a float suffix.
252 """
253 raw = match.group(0)
254 group = match.lastgroup
255 # Strip only true literal suffixes -- and never strip `f` from a hex
256 # literal, where it is a hex digit (0xAF), not a float suffix.
257 if group == "hex":
258 body = raw.rstrip("uUlL")
259 return False, float(int(body, 16))
260 if group == "bin":
261 body = raw.rstrip("uUlL")
262 return False, float(int(body, 2))
263 if group in ("flt", "dec_flt"):
264 body = raw.rstrip("fFlL")
265 try:
266 return True, float(body)
267 except ValueError:
268 return False, float("nan")
269 body = raw.rstrip("uUlL")
270 # Leading-zero decimals are octal in C (0700 == 448); fall back to
271 # base 8, then base 10, before giving up.
272 for base in (0, 8, 10):
273 try:
274 return False, float(int(body, base))
275 except ValueError: # noqa: PERF203 # per-base fallback; loop is only 3 iterations
276 continue
277 return False, float("nan")

280def _is_array_dimension(code_line: str, start: int, end: int) -> bool:
281 """Whether a literal is an array dimension rather than a magic number.
282
283 True only when it is the SOLE content of a ``[ ... ]`` on a declaration
284 line. Matching clang-tidy's treatment here is deliberate: two gates
285 disagreeing about the same literal is worse than either rule alone.
286 """
287 i = start - 1
288 while i >= 0 and code_line[i].isspace():
289 i -= 1
290 if i < 0 or code_line[i] != "[":
291 return False
292 j = end
293 while j < len(code_line) and code_line[j].isspace():
294 j += 1
295 if j >= len(code_line) or code_line[j] != "]":
296 return False
297 return bool(_DECL_HINT.search(code_line))

300def _is_ignored(is_float: bool, value: float) -> bool:
301 if math.isnan(value): # could not parse, do not flag
302 return True
303 if is_float:
304 return value in IGNORED_FLOAT
305 return int(value) in IGNORED_INT

308class _SuppressionState:
309 """Tracks clang-tidy NOLINT scope across the lines of one file.
310
311 Honours the project's own readability-magic-numbers waivers, so a literal
312 this gate would flag but clang-tidy is already told to ignore is not
313 reported twice under two different names.
314 """
315
316 def __init__(self) -> None:
317 self.in_region = False # inside a magic-relevant NOLINTBEGIN/END block
318 self.skip_next = False # previous line was a magic-relevant NOLINTNEXTLINE
319
320 def suppresses(self, raw: str) -> bool:
321 """True when ``raw`` is inside, or is itself, a NOLINT suppression."""
322 m_beg = _NOLINT_BEGIN_RE.search(raw)
323 if m_beg and _nolint_applies(m_beg.group(1)):
324 self.in_region = True
325 m_end = _NOLINT_END_RE.search(raw)
326 if m_end and _nolint_applies(m_end.group(1)):
327 self.in_region = False
328 if self.in_region or m_beg or m_end:
329 return True
330 if self.skip_next:
331 self.skip_next = False
332 return True
333 m_nl = _NOLINT_NEXT_RE.search(raw)
334 if m_nl and _nolint_applies(m_nl.group(1)):
335 self.skip_next = True
336 return True
337 m_inl = _NOLINT_INLINE_RE.search(raw)
338 return bool(m_inl and _nolint_applies(m_inl.group(1)))

341class _EnumState:
342 """Tracks brace depth inside an enum body.
343
344 Literals inside an enum ARE the named constants this gate exists to
345 require, so the body is skipped rather than reported.
346 """
347
348 def __init__(self) -> None:
349 self.depth = 0
350 self.pending = False # saw `enum`, awaiting its `{`
351
352 def inside(self, code_line: str) -> bool:
353 """Advance the tracker over ``code_line``; True if it is enum body."""
354 if self.depth == 0 and re.search(r"\benum\b", code_line):
355 self.pending = True
356 if not (self.pending or self.depth > 0):
357 return False
358 opens = code_line.count("{")
359 closes = code_line.count("}")
360 if self.pending and opens > 0:
361 self.pending = False
362 self.depth += opens - closes
363 return True # the `... enum ... {` line itself is a definition
364 self.depth = max(self.depth + opens - closes, 0)
365 return self.depth > 0

368def _line_literals(code_line: str) -> list[tuple[int, str]]:
369 """Return (column, literal) for every magic number on one line of code."""
370 out: list[tuple[int, str]] = []
371 for m in _NUM_RE.finditer(code_line):
372 start = m.start()
373 if start > 0 and _IDENT_BEFORE.match(code_line[start - 1]):
374 continue
375 if _is_array_dimension(code_line, start, m.end()):
376 continue
377 is_float, value = _literal_value(m)
378 if _is_ignored(is_float, value):
379 continue
380 if is_float and _CONST_FLOAT_DEF.search(code_line) and _FLOAT_TYPE.search(code_line):
381 continue # named const float/double definition
382 out.append((start + 1, m.group(0)))
383 return out

386def _skip_line(code_line: str, orig_line: str, enums: _EnumState) -> bool:
387 """True when this line cannot hold a reportable literal."""
388 if enums.inside(code_line):
389 return True
390 stripped = code_line.lstrip()
391 # Preprocessor directives are governed by other gates.
392 if stripped.startswith("#"):
393 return True
394 # Pure data rows of a const lookup table are not logic.
395 if stripped and _DATA_ROW_RE.match(code_line):
396 return True
397 # Per-line opt-out (checked against the original source line).
398 return OPT_OUT in orig_line

401def _scan_file(path: Path) -> list[tuple[int, int, str]]:
402 """Return a list of (line, column, literal) for every magic number in `path`."""
403 try:
404 text = path.read_text()
405 except (OSError, UnicodeDecodeError):
406 return []
407
408 code_lines = _strip_comments_and_strings(text).splitlines()
409 orig_lines = text.splitlines()
410
411 violations: list[tuple[int, int, str]] = []
412 nolint = _SuppressionState()
413 enums = _EnumState()
414
415 for idx, code_line in enumerate(code_lines):
416 raw = orig_lines[idx] if idx < len(orig_lines) else ""
417 if nolint.suppresses(raw):
418 continue
419 if _skip_line(code_line, raw, enums):
420 continue
421 violations.extend((idx + 1, col, lit) for col, lit in _line_literals(code_line))
422
423 return violations

431def _enumerate_targets(arg_paths: Iterable[str]) -> list[Path]:
432 """Resolve the list of files to scan from CLI arguments."""
433 args = [a for a in arg_paths if a not in ("--check", "--all")]
434 if args:
435 out: list[Path] = []
436 for raw in args:
437 p = Path(raw)
438 if not p.is_absolute():
439 p = REPO_ROOT / p
440 if p.is_dir():
441 out.extend(p.rglob("*.c"))
442 elif p.suffix == ".c":
443 out.append(p)
444 return [p for p in out if not _is_excluded(p)]

446 # No-argument mode scans every GIT-VISIBLE .c so nothing first-party is
447 # silently skipped (new top-level dirs, tools/, port/, etc. are covered
448 # automatically -- there is no allowlist to forget to update).
449 # Enumerating via ``git ls-files`` (tracked plus untracked-but-not-
450 # ignored) instead of a filesystem walk keeps gitignored artifacts out:
451 # a raw rglob scanned stray CMake compiler-probe files under app
452 # build-*/ dirs and the .claude/worktrees checkouts, failing commits on
453 # files CI can never see. The per-app boot boilerplate (vector_table.c,
454 # system_init.c, ...) is copied verbatim into every app and is full of
455 # vector/IRQ-slot indices, so it is dropped BY FILENAME via
456 # BOOT_BOILERPLATE. Vendor trees are dropped via EXCLUDE_FRAGMENTS.
457 #
458 # There is deliberately no longer an "examples/ scans only main.c" rule.
459 # BOOT_BOILERPLATE already names the copied boot files precisely, so that
460 # extra filename test protected nothing -- it silently dropped every OTHER
461 # example TU: each app's src/*.c, plus cpu1_main.c, ns_main.c and ns_usb.c.
462 # 47 real magic numbers were sitting in those files, among them a whole
463 # block of hand-addressed Armv8-M SAU registers, while this gate reported
464 # the tree clean. Scope by what a file IS, never by what it is named
465 # (the #296 / #332 / #358 / #369 defect family).
466 git_tool = shutil.which("git") or "git"
467 proc = subprocess.run( # noqa: S603 -- fixed argv, trusted tool path
468 [git_tool, "ls-files", "--cached", "--others", "--exclude-standard", "--", "*.c"],
469 cwd=REPO_ROOT,
470 capture_output=True,
471 text=True,
472 check=False,
473 )
474 if proc.returncode != 0:
475 sys.stderr.write(proc.stderr)
476 sys.stderr.write(f"git ls-files failed (exit {proc.returncode})\n")
477 sys.exit(2)
478 out = []
479 for line in proc.stdout.splitlines():
480 rel = line.strip()
481 if not rel:
482 continue
483 c = REPO_ROOT / rel
484 if c.name in BOOT_BOILERPLATE:
485 continue
486 out.append(c)
487 return [p for p in out if not _is_excluded(p)]

490def main(argv: list[str]) -> int:
491 """Fail on integer literals that should be named typed enums.
492
493 Reports column as well as line, which is why the whole scan runs over a
494 length-preserving blanked view of the source rather than a stripped one.
495
496 Returns 1 listing each literal, 0 when clean or when argv filtered to
497 nothing.
498 """
499 targets = _enumerate_targets(argv[1:])
500 if not targets:
501 print("check_magic_numbers.py: no files to scan", file=sys.stderr)
502 return 0
503
504 findings: list[tuple[str, int, int, str]] = []
505 for path in targets:
506 for line_no, col, literal in _scan_file(path):
507 rel = path.relative_to(REPO_ROOT) if path.is_relative_to(REPO_ROOT) else path
508 findings.append((str(rel), line_no, col, literal))
509
510 if not findings:
511 print(f"check_magic_numbers.py: {len(targets)} file(s) scanned, no magic numbers found.")
512 return 0
513
514 findings.sort()
515 print(
516 f"check_magic_numbers.py: {len(findings)} magic number(s) found "
517 "-- every integer literal must be a named typed enum (use a const "
518 "only for floating-point; macros are not allowed for constants):\n",
519 file=sys.stderr,
520 )
521 print(" file:line:col literal", file=sys.stderr)
522 for path, line_no, col, literal in findings:
523 print(f" {path}:{line_no}:{col} {literal}", file=sys.stderr)
524 print(
525 "\nReplace each literal with a named value. Genuine exceptions may "
526 f"carry a trailing `{OPT_OUT}: <reason>` comment on the line.",
527 file=sys.stderr,
528 )
529 return 1
