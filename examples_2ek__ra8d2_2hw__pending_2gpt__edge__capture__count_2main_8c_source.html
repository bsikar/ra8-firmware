294static void gpt_ecc_timers_or_halt(void)
295{
296 const ra8_gpt_cfg_t cfg = {
297 .mode = k_ra8_gpt_mode_saw_pwm,
298 .prescaler = k_ra8_gpt_ps_div_1,
299 .period = (uint32_t)k_gpt_ecc_period,
300 .duty_a = 0U,
301 .duty_b = 0U,
302 .auto_start = true,
303 };
304 if (ra8_gpt_init((uint8_t)k_gpt_ecc_capture_channel, &cfg) != k_ra8_ok) {
305 gpt_ecc_panic_halt();
306 }
307 /* #185 input capture: latch GTCNT into GTCCRA on each GTIOC0A rising edge. */
308 if (ra8_gpt_capture_configure((uint8_t)k_gpt_ecc_capture_channel,
309 k_ra8_gpt_ccr_a,
310 (uint32_t)k_ra8_gpt_cap_src_ioca_rising) != k_ra8_ok) {

314 gpt_ecc_panic_halt();

317 * Quadrature variant: pass k_ra8_gpt_cnt_src_ioca_falling as the down
318 * source instead of k_ra8_gpt_cnt_src_none for an encoder up/down count. */
319 if (ra8_gpt_event_count_configure((uint8_t)k_gpt_ecc_count_channel,
320 (uint32_t)k_ra8_gpt_cnt_src_ioca_rising,
321 (uint32_t)k_ra8_gpt_cnt_src_none) != k_ra8_ok) {

344static bool gpt_ecc_service_capture(uint32_t* prev_latch)
345{
346 uint32_t status = 0U;
347 if (ra8_gpt_get_status((uint8_t)k_gpt_ecc_capture_channel, &status) != k_ra8_ok) {
348 return false;
349 }
350 if ((status & (uint32_t)k_ra8_gpt_status_ccra) == 0U) {
351 return false;
352 }
353 uint32_t latch = 0U;
354 if (ra8_gpt_capture_read((uint8_t)k_gpt_ecc_capture_channel, k_ra8_gpt_ccr_a, &latch) ==
355 k_ra8_ok) {
356 const uint32_t delta = latch - *prev_latch;
357 *prev_latch = latch;
358 if (gpt_ecc_period_valid(delta)) {
359 g_gpt_ecc_last_period = delta;
360 g_gpt_ecc_capture_events = g_gpt_ecc_capture_events + 1U;
361 (void)ra8_board_led_toggle(k_ra8_board_led1);
362 }
363 }
364 (void)ra8_gpt_clear_status((uint8_t)k_gpt_ecc_capture_channel, (uint32_t)k_ra8_gpt_status_ccra);
365 return true;
366}

370int32_t main(void)
371{
372 gpt_ecc_clocks_or_halt();
373 gpt_ecc_route_pins_or_halt();
374 gpt_ecc_timers_or_halt();
375 ra8_isr_globals_enable();
376
377 uint32_t prev_latch = 0U;
378 uint32_t prev_free = 0U;
379 bool have_prev = false;
380
381 while (1) {
382 /* #185 input capture: hardware edge timestamp -> period. */
383 (void)gpt_ecc_service_capture(&prev_latch);
384
385 /* #186 external event counting: accumulated GTIOC1A edge count. */
386 uint32_t pulses = 0U;
387 if (ra8_gpt_read((uint8_t)k_gpt_ecc_count_channel, &pulses) == k_ra8_ok) {
388 g_gpt_ecc_pulse_count = pulses;
389 }
390
391 /* Pin-independent liveness: capture-channel free-run counter alive. */
392 uint32_t now_free = 0U;
393 if (ra8_gpt_read((uint8_t)k_gpt_ecc_capture_channel, &now_free) == k_ra8_ok) {
394 if (have_prev) {
395 if (now_free != prev_free) {
396 g_gpt_ecc_tick = g_gpt_ecc_tick + 1U;
397 }
398 }
399 prev_free = now_free;
400 have_prev = true;
401 }
402 ra8_delay_ms((uint32_t)k_gpt_ecc_poll_period_ms);
403 }
404 gpt_ecc_panic_halt();
405 return 0;
406}
