156static void thread_tx_entry(ULONG thread_input)
157{
158 (void)thread_input;
159 uint8_t seq = 0U;
160 while (1) {
161 ra8_canfd_frame_t tx = {
162 .id = (uint32_t)k_canfd_heartbeat_id,
163 .dlc = (uint8_t)k_canfd_heartbeat_dlc,
164 .is_extended = 0U,
165 .is_fd = 0U,
166 .is_brs = 0U,
167 .data = {seq,
168 k_canfd_marker_0,
169 k_canfd_marker_1,
170 k_canfd_marker_2,
171 k_canfd_marker_3,
172 k_canfd_marker_4,
173 k_canfd_marker_5,
174 k_canfd_marker_6},
175 };
176 if (ra8_canfd_transmit((uint8_t)k_canfd_demo_channel, &tx) == k_ra8_ok) {
177 (void)ra8_board_led_toggle(k_ra8_board_led1);
178 g_threadx_canfd_match += 1U;
179 } else {
180 g_threadx_canfd_mismatch += 1U;
181 }
182 seq++;
183 (void)tx_thread_sleep((ULONG)k_canfd_heartbeat_ticks);
184 }

286int32_t main(void)
287{
288 /* CGC bring-up FIRST so the ThreadX SysTick reload (programmed in
289 * tx_initialize_low_level.S assuming RA8_BOOT_CLOCK_HZ = 1 GHz)
290 * ticks at the intended 1 ms wallclock cadence. On MOCO (~8.4 MHz)
291 * the tick rate would be 1/119 the expected, making
292 * tx_thread_sleep(...) ~119x slower than the ms value declared in
293 * source. */
294 if (ra8_cgc_init() != k_ra8_ok) {
295 while (1) {
296 __asm__ volatile("wfi");
297 }
298 }
299
300 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
301 while (1) {
302 __asm__ volatile("wfi");
303 }
304 }
305 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
306 while (1) {
307 __asm__ volatile("wfi");
308 }
309 }
310
311 if (ra8_canfd_init((uint8_t)k_canfd_demo_channel) != k_ra8_ok) {
312 while (1) {
313 __asm__ volatile("wfi");
314 }
315 }
316 if (ra8_canfd_set_bitrate((uint8_t)k_canfd_demo_channel,
317 (uint32_t)k_canfd_demo_bitrate,
318 (uint32_t)k_canfd_demo_bitrate) != k_ra8_ok) {
319 while (1) {
320 __asm__ volatile("wfi");
321 }
322 }
323 /* Internal-loopback (HUM Ch 41 "CFDCnCTR" p 2710): TX frames echo
324 * into RX so this single-board demo can validate the round trip
325 * without a second CAN node on the bus. */
326 if (ra8_canfd_set_test_mode((uint8_t)k_canfd_demo_channel, k_ra8_ctms_self_test_1) != k_ra8_ok) {
327 while (1) {
328 __asm__ volatile("wfi");
329 }
330 }
331
332 ra8_isr_globals_enable();
333
334#ifndef RA8_OFF_TARGET
335 tx_kernel_enter();
336#endif
337
338 while (1) {
339 __asm__ volatile("wfi");
340 }
341}
