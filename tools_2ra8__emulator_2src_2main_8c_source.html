125static int main_open_engine(uc_engine** uc_out)
126{
127 uc_engine* uc = nullptr;
128 if (uc_open(UC_ARCH_ARM, (uc_mode)(UC_MODE_THUMB | UC_MODE_MCLASS), &uc) != UC_ERR_OK) {
129 (void)fprintf(stderr, "uc_open failed\n");
130 return 1;
131 }
132 (void)uc_ctl_set_cpu_model(uc, UC_CPU_ARM_CORTEX_M33); /* closest to the M85. */
133 if (!emu_memmap_init(uc)) {
134 return 1;
135 }
136 /* Seed hardwired MPU_TYPE.DREGION (=8) so ra8_mpu_configure validates; the PPB
137 * is plain RAM here, so it would otherwise read 0 and panic mpu_partition_simple. */
138 wr32(uc, (uint64_t)k_mpu_type, (uint32_t)k_mpu_type_seed);
139 *uc_out = uc;
140 return 0;
141}

160static void main_arm_touch_seq(const char* touch_seq_str)
161{
162 if (touch_seq_str == nullptr) {
163 return;
164 }
165 enum : uint32_t {
166 k_touch_seq_parse_max = 32U,
167 };
168 board_periph_touch_seq_reset();
169 const char* p = touch_seq_str;
170 uint32_t pushed = 0U;
171 for (uint32_t n = 0U; (n < (uint32_t)k_touch_seq_parse_max) && (*p != '\0'); n++) {
172 char* end = nullptr;
173 const long x = strtol(p, &end, (int)k_strtol_base10);
174 if ((end == p) || (*end != ':')) {
175 break;
176 }
177 p = end + 1;
178 const long y = strtol(p, &end, (int)k_strtol_base10);
179 if (end == p) {
180 break;
181 }
182 p = end;
183 if ((x >= 0L) && (y >= 0L) && board_periph_touch_seq_push((uint16_t)x, (uint16_t)y)) {
184 pushed++;
185 }
186 if (*p == ',') {
187 p++;
188 }
189 }
190 (void)fprintf(stderr, "ra8_emulator: --touch-seq armed %u raw point(s)\n", (unsigned)pushed);
191}

329{
330 long elf_len = 0;

348 free(elf);
349 return 1;

379static int main_load_ns(uc_engine* uc,
380 const char* ns_elf_path,
381 uint8_t* elf,
382 uint8_t** ns_out,
383 long* ns_len_out)
384{
385 long ns_len = 0;
386 uint8_t* const ns_elf = (ns_elf_path != nullptr) ? read_file(ns_elf_path, &ns_len) : nullptr;
387 if (ns_elf_path != nullptr) {
388 if (ns_elf == nullptr) {
389 (void)fprintf(stderr, "cannot read --ns %s\n", ns_elf_path);
390 free(elf);
391 return 1;
392 }
393 (void)fprintf(stderr, "ra8_emulator: loading NS image %s (%ld bytes)\n", ns_elf_path, ns_len);
394 if (load_elf(uc, ns_elf, ns_len) != 0) {
395 free(ns_elf);
396 free(elf);
397 return 1;
398 }
399 /* Track the NS image's vector base (0x32100000 RAM-resident, 0x90000000 XIP
400 * from OSPI) so the BLXNS world switch reads MSP/reset from the right place. */
401 const uint32_t ns_vbase = elf_vector_base(ns_elf, ns_len);
402 if (ns_vbase != 0U) {
403 emu_tz_set_ns_vector_base(ns_vbase);
404 }
405 (void)fprintf(stderr, "ra8_emulator: NS vector base @ 0x%08X\n", emu_tz_ns_vector_base());
406 }
407 *ns_out = ns_elf;
408 *ns_len_out = ns_len;
409 return 0;

484static uint32_t main_reset_vector(uc_engine* uc)
485{
486 uint32_t sp = 0U;
487 uint32_t pc = 0U;
488 /* MRAM[0] */
489 (void)uc_mem_read(uc, emu_memmap_mram_base() + 0U, &sp, 4);
490 /* MRAM[4] (Thumb: bit0=1) */
491 (void)uc_mem_read(uc, emu_memmap_mram_base() + 4U, &pc, 4);
492 pc |= 1U; /* M-profile is always Thumb (EPSR.T must be 1). */
493 uint32_t xpsr = (uint32_t)k_xpsr_t_bit; /* xPSR.T */
494 (void)uc_reg_write(uc, UC_ARM_REG_SP, &sp);
495 (void)uc_reg_write(uc, UC_ARM_REG_PC, &pc);
496 (void)uc_reg_write(uc, UC_ARM_REG_XPSR, &xpsr);
497 (void)fprintf(
498 stderr,
499 "ra8_emulator: reset SP=0x%08X PC=0x%08X -- running (<= %u x %u insns, %u s wall)\n",
500 sp,
501 pc,
502 (unsigned)k_run_max_chunks,
503 (unsigned)k_run_chunk_insns,
504 (unsigned)k_run_wall_s);
505 return pc;
506}

562static void
563main_install_run_seams(uc_engine* uc, uint8_t* elf, long elf_len, const emu_args_t* args)
564{
565 bool usbh_seamed = false;
566 if (!args->usbhs_loop) {
567 usbh_seamed = usbh_seam_install(uc, elf, elf_len);
568 }
569 /* Register-level USBHS host model: only an UNSEAMED, non-loop firmware engages
570 * it, so the C-level seam / loop-only register block do not both answer. */
571 board_usb_host_set_allowed(!usbh_seamed && !args->usbhs_loop);
572 sym_trace_install(uc, elf, elf_len, args->trace_sym_names, args->trace_sym_n);
573 /* M85-only long-shift (LSLL/LSRL/ASRL) + MVE (Helium) seams: off under
574 * --primary-core m33 (pure Armv8-M), inert but honest. CSEL rides the invalid-
575 * instruction hook and stays armed (also inert for an M33 image). */
576 if (emu_primary_core() == k_core_m85) {
577 long_shift_seam_install(uc, elf, elf_len);
578 }
579 div0_seam_install(elf, elf_len); /* UDIV/SDIV sites, patched only under DIV_0_TRP. */
580 fast_sd_seam_install(uc, elf, elf_len); /* --fast-sd whole-block serve; else inert. */
581 prof_load(elf, elf_len); /* RA8_EMU_PROFILE FUNC symbols + code hook. */
582 emu_prof_install(uc);
583 emu_cpu1_init(elf, elf_len); /* dual-core cpu1 engine (shares SRAM); NULL for single-core. */
584}

653int main(int argc, char** argv)
654{
655 emu_args_t args = {};
656 if (!emu_args_parse(argc, argv, &args)) {
657 return 2;
658 }
659 uint16_t view_w = args.view_w;
660 uint16_t view_h = args.view_h;
661 const char* const win_title = main_apply_panel(&args, &view_w, &view_h);
662
663 uc_engine* uc = nullptr;
664 if (main_open_engine(&uc) != 0) {
665 return 1;
666 }
667 main_bringup_peripherals(&args);
668
669 uint8_t* elf = nullptr;
670 long elf_len = 0;
671 if (main_load_primary(uc, args.elf_path, &elf, &elf_len) != 0) {
672 return 1;
673 }
674 uint8_t* ns_elf = nullptr;
675 long ns_len = 0;
676 if (main_load_ns(uc, args.ns_elf_path, elf, &ns_elf, &ns_len) != 0) {
677 return 1;
678 }
679
680 uint32_t dump_sym_addrs[k_dump_sym_max] = {}; /* resolved while ELF is alive. */
681 uint32_t stop_sym_addr = 0U; /* resolved while ELF is alive. */
682 main_resolve_symbols(elf, elf_len, ns_elf, ns_len, &args, dump_sym_addrs, &stop_sym_addr);
683 free(ns_elf); /* NS-image symbol table no longer needed (bytes are in Unicorn memory). */
684 /* TrustZone NSC pointer validation: patch cmse_check_address_range to BX LR. */
685 emu_tz_patch_cmse(uc, elf, elf_len);
686
687 const uint32_t pc = main_reset_vector(uc);
688 /* MRAM holds the vector table; the run loop passes this as the VTOR fallback. */
689 const uint32_t vtor_base = (uint32_t)emu_memmap_mram_base();
690 main_install_core_seams(uc, elf, elf_len);
691 main_install_run_seams(uc, elf, elf_len, &args);
692
693 const emu_run_cfg_t run_cfg = main_build_run_cfg(&args,
694 uc,
695 elf,
696 elf_len,
697 pc,
698 vtor_base,
699 view_w,
700 view_h,
701 win_title,
702 dump_sym_addrs,
703 stop_sym_addr);
704 return emu_run_and_report(&run_cfg);
705}
