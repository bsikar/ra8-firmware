83RA8_INTERNAL
84static ra8_err_t priv_mem_pull(void* ctx, uint8_t* buf, size_t cap, size_t* got)
85{
86 ra8_comic_tiles_pull_t* p = (ra8_comic_tiles_pull_t*)ctx;
87 const size_t remain = p->n - p->off;
88 const size_t k = (cap < remain) ? cap : remain;
89 if (k > 0U) {
90 (void)memcpy(buf, &p->d[p->off], k);
91 }
92 p->off += k;
93 *got = k;
94 return k_ra8_ok;
95}

144ra8_err_t ra8_comic_tiles_footprint(const uint8_t* enc,
145 size_t len,
146 uint16_t* out_w,
147 uint16_t* out_h,
148 uint64_t* out_decoded_bytes)
149{
150 RA8_CHECK_NULL_PTR(enc, s_tag, "footprint: null enc");
151 RA8_CHECK_NULL_PTR(out_w, s_tag, "footprint: null out_w");
152 RA8_CHECK_NULL_PTR(out_h, s_tag, "footprint: null out_h");
153 RA8_CHECK_NULL_PTR(out_decoded_bytes, s_tag, "footprint: null out_decoded_bytes");
154 if (len == 0U) {
155 return k_ra8_err_invalid_size;
156 }
157 uint16_t w = 0U;
158 uint16_t h = 0U;
159 const ra8_err_t err = ra8_jof_probe_dims(enc, len, &w, &h);
160 if (err != k_ra8_ok) {
161 return err;
162 }
163 *out_w = w;
164 *out_h = h;
165 *out_decoded_bytes = (uint64_t)w * (uint64_t)h * (uint64_t)k_ra8_comic_tiles_decoded_bpp;
166 return k_ra8_ok;
167}

240RA8_INTERNAL
241static ra8_err_t priv_transcode(ra8_comic_tile_reader_t* r,
242 const uint8_t* enc,
243 size_t enc_len,
244 const ra8_comic_tiles_import_cfg_t* cfg)
245{
246 r->store = (ra8_jof_memstore_t){.buf = cfg->atlas, .cap = cfg->atlas_cap, .len = 0U};
247 ra8_comic_tiles_pull_t pull = {.d = enc, .n = enc_len, .off = 0U};
248 const ra8_jof_produce_cfg_t pcfg = {
249 .pull = priv_mem_pull,
250 .pull_ctx = &pull,
251 .sink = ra8_jof_memstore_sink,
252 .sink_ctx = &r->store,
253 .tile_w = cfg->tile_w,
254 .tile_h = cfg->tile_h,
255 .codec = cfg->codec,
256 .max_width = cfg->max_width,
257 .max_height = cfg->max_height,
258 .work = cfg->work,
259 .work_cap = cfg->work_cap,
260 .webp_work = cfg->webp_work,
261 .webp_work_cap = cfg->webp_work_cap,
262 };
263 ra8_jof_info_t produced = {};
264 const ra8_err_t perr = ra8_jof_produce(&pcfg, &produced);
265 if (perr != k_ra8_ok) {
266 return perr;
267 }
268 return ra8_jof_parse(ra8_jof_memstore_pread, &r->store, (uint64_t)r->store.len, &r->info);
269}

271ra8_err_t ra8_comic_tiles_import(ra8_comic_tile_reader_t* r,
272 const uint8_t* enc,
273 size_t enc_len,
274 const ra8_comic_tiles_import_cfg_t* cfg)
275{
276 const ra8_err_t nz = priv_import_args_ok(r, enc, cfg);
277 if (nz != k_ra8_ok) {
278 return nz;
279 }
280 if (enc_len == 0U) {
281 return k_ra8_err_invalid_size;
282 }
283 r->bound = false; /* invalidate first: any error leaves no bound page */
284 const ra8_err_t err = priv_transcode(r, enc, enc_len, cfg);
285 if (err != k_ra8_ok) {
286 return err;
287 }
288 r->epoch += 1U;
289 r->bound = true;
290 return k_ra8_ok;
291}

304ra8_err_t ra8_comic_tiles_tile(ra8_comic_tile_reader_t* r,
305 uint16_t tile_x,
306 uint16_t tile_y,
307 ra8_tile_t* out_tile)
308{
309 RA8_CHECK_NULL_PTR(r, s_tag, "tile: null r");
310 RA8_CHECK_NULL_PTR(out_tile, s_tag, "tile: null out_tile");
311 if (!r->bound) {
312 return k_ra8_err_invalid_state;
313 }
314 if (tile_x >= r->info.tile_cols) {
315 return k_ra8_err_out_of_range;
316 }
317 if (tile_y >= r->info.tile_rows) {
318 return k_ra8_err_out_of_range;
319 }
320 const ra8_tile_key_t key = {.image_id = r->epoch,
321 .tile_x = tile_x,
322 .tile_y = tile_y,
323 .zoom = 0U,
324 .reserved = 0U};
325 return ra8_tile_cache_get(&r->cache, &key, out_tile);
326}
