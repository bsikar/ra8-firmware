4"""Generate synthetic, non-copyright page images for the media_dl harness.
5
6Feeds integration.sh. Every page is a distinct, high-contrast pattern, and both
7properties are what make the downstream checks meaningful:
8
9* Multiple colors per page, so `check_ppm.py`'s "did the viewer render
10 anything" test can distinguish a real frame from a blank fill. A solid page
11 would decode fine and prove nothing.
12* Every page different from its neighbours, so an export that silently drops or
13 duplicates a page cannot pass -- identical pages would compare equal.
14
15No fonts, no network, no committed fixtures: the whole set is reproducible from
16the command-line parameters alone, which is why this can run on any machine
17without checking binaries into the tree.
18"""
19

40def make_page(idx: int, count: int, w: int, h: int) -> Image.Image:
41 """Render one synthetic page whose pixels encode its own index.
42
43 Three overlaid features, each with a job:
44
45 * A two-tone diagonal split, whose color is a function of `idx`. The large
46 flat regions survive JPEG quantization, so the distinct-color count holds
47 up after a lossy round trip.
48 * A horizontal band whose vertical position interpolates `idx` across the
49 page. This is what makes any two pages differ visibly.
50 * A block "bar code" of the 1-based page number along the top edge, so a
51 human looking at a dumped frame can read which page it is.
52
53 Deterministic: the same arguments always produce identical pixels.
54
55 Args:
56 idx: 0-based page index; must be < `count` for the band to stay on the
57 page.
58 count: Total pages, used only to scale the band. A `count` of 1 pins
59 the band to the top rather than dividing by zero.
60 w: Page width in pixels.
61 h: Page height in pixels. Both must be large enough for the derived
62 feature sizes (`h // 20`, `w // 14`) to be non-zero.
63
64 Returns:
65 An RGB Pillow Image.
66 """
67 # Two-tone diagonal split gives a large, JPEG-robust set of distinct colors.
68 img = Image.new("RGB", (w, h), (250, 250, 250))
69 d = ImageDraw.Draw(img)
70 base = (40 + (idx * 37) % 160, 60 + (idx * 53) % 160, 90 + (idx * 71) % 160)
71 d.polygon([(0, 0), (w, 0), (0, h)], fill=base)
72 # A moving horizontal band whose position encodes the page index, so no two
73 # pages decode to the same image (guards against an export that silently
74 # duplicates or drops pages).
75 band_y = int((h - h // 8) * (idx / max(1, count - 1))) if count > 1 else 0
76 d.rectangle([0, band_y, w, band_y + h // 8], fill=(255, 200, 0))
77 # A block "bar code" of the 1-based page number along the top edge.
78 for bit in range(8):
79 if (idx + 1) & (1 << bit):
80 x = 10 + bit * (w // 10)
81 d.rectangle([x, 10, x + w // 14, 10 + h // 20], fill=(0, 0, 0))
82 return img

85def main() -> None:
86 """Write COUNT synthetic pages into OUTDIR, from `OUTDIR COUNT W H` in argv.
87
88 Pages are saved as baseline JPEG named `page_0001.jpg` upward, 1-based and
89 zero-padded so a plain lexical sort matches page order. JPEG rather than PNG
90 is deliberate: real scraped pages are JPEG, so this exercises the actual
91 decode paths (the viewer probe and the jof producer) instead of a format
92 they never see in production.
93
94 OUTDIR must already exist; it is not created. Existing pages are
95 overwritten, and pages from a previous larger run are NOT removed, so a
96 reused directory can end up with stale extra pages.
97
98 Returns:
99 EXIT_OK, or EXIT_USAGE when too few arguments were given.
100
101 Raises:
102 ValueError: COUNT, W or H is not an integer.
103 OSError: OUTDIR does not exist or is not writable.
104 """
105 if len(sys.argv) < ARGV_REQUIRED:
106 sys.stderr.write("usage: gen_pages.py OUTDIR COUNT W H\n")
107 return EXIT_USAGE
108 outdir, count, w, h = sys.argv[1], int(sys.argv[2]), int(sys.argv[3]), int(sys.argv[4])
109 for i in range(count):
110 page = make_page(i, count, w, h)
111 # Baseline JPEG (what real scraped pages are) so the JPEG-decode paths
112 # (viewer probe, jof producer) are exercised, not just PNG.
113 page.save(f"{outdir}/page_{i + 1:04d}.jpg", "JPEG", quality=JPEG_QUALITY)
114 print(f"gen_pages: wrote {count} page(s) ({w}x{h}) to {outdir}")
115 return EXIT_OK

118if __name__ == "__main__":
119 sys.exit(main())
