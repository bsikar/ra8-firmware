312/* Drive the CMD32/CMD33/CMD38 erase sequence over [lba, lba+count) -- see implementation for details. */
313static ra8_err_t internal_erase_range(uint32_t lba, uint32_t count)
314{
315 ra8_err_t err = internal_cmd_require_ready(k_sd_cmd_erase_wr_blk_start, internal_lba_to_arg(lba));
316 if (err != k_ra8_ok) {
317 return err;
318 }
319 err =
320 internal_cmd_require_ready(k_sd_cmd_erase_wr_blk_end, internal_lba_to_arg((lba + count) - 1U));
321 if (err != k_ra8_ok) {
322 return err;
323 }
324 /* CMD38 then a long busy wait -- a bulk erase can hold the card busy for
325 * seconds, far longer than a single-block write. */
326 err = ra8_sdmmc_spi_cs_assert();
327 if (err != k_ra8_ok) {
328 return err;
329 }
330 uint8_t r1 = 0U;
331 err = ra8_sdmmc_spi_send_command(k_sd_cmd_erase, 0U, &r1);
332 if (err != k_ra8_ok) {
333 (void)ra8_sdmmc_spi_cs_release();
334 return err;
335 }
336 if (r1 != 0U) {
337 (void)ra8_sdmmc_spi_cs_release();
338 return k_ra8_err_protocol_error;
339 }
340 err = ra8_sdmmc_spi_wait_not_busy_bounded((uint32_t)k_sd_max_erase_poll_bytes);
341 (void)ra8_sdmmc_spi_cs_release();
342 return err;
343}

357/* Drain the 2-byte CRC16 trailer from the card and verify it -- see implementation for details. */
358static ra8_err_t internal_read_block_crc_check(const uint8_t* buf)
359{
360 uint8_t crc_hi = 0U;
361 uint8_t crc_lo = 0U;
362 (void)ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &crc_hi);
363 (void)ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &crc_lo);
364 const uint16_t expected = ra8_sdmmc_spi_crc16(buf, (uint32_t)k_ra8_sdmmc_spi_block_size);
365 const uint16_t actual = (uint16_t)(((uint16_t)crc_hi << k_sd_bit_byte) | (uint16_t)crc_lo);
366 if (expected != actual) {
367 return k_ra8_err_crc_mismatch;
368 }
369 return k_ra8_ok;
370}

447ra8_err_t ra8_sdmmc_spi_read_blocks(uint32_t lba, uint8_t* buf, uint32_t count)
448{
449 RA8_CHECK_NULL_PTR(buf, s_tag, "buf is null");
450 if (!s_sdmmc_spi_state.initialized) {
451 return k_ra8_err_invalid_state;
452 }
453 if (count == 0U) {
454 return k_ra8_ok;
455 }
456 if ((lba >= s_sdmmc_spi_state.capacity_blocks) ||
457 (count > (s_sdmmc_spi_state.capacity_blocks - lba))) {
458 return k_ra8_err_out_of_range;
459 }
460 if (count == 1U) {
461 return ra8_sdmmc_spi_read_block(lba, buf);
462 }
463 ra8_err_t err = ra8_sdmmc_spi_cs_assert();
464 RA8_RETURN_ON_ERROR(err, s_tag, "cs assert");
465 uint8_t r1 = 0U;
466 err = ra8_sdmmc_spi_send_command(k_sd_cmd_read_multi_block, internal_lba_to_arg(lba), &r1);
467 if ((err != k_ra8_ok) || (r1 != 0U)) {
468 (void)ra8_sdmmc_spi_cs_release();
469 return (err != k_ra8_ok) ? err : k_ra8_err_protocol_error;
470 }
471 err = internal_read_multi_stream(buf, count);
472 /* CMD12 runs on the success AND the abort path: the card keeps streaming
473 * read data until it sees STOP_TRANSMISSION, so it must leave the data
474 * state before CS is released or the next command collides with the
475 * still-open stream. A stream error takes precedence over a stop error. */
476 const ra8_err_t stop_err = internal_read_multi_stop();
477 (void)ra8_sdmmc_spi_cs_release();
478 if (err != k_ra8_ok) {
479 return err;
480 }
481 return stop_err;
482}

484/* Stream one data block out (token + payload + CRC16) and check -- see implementation for details. */
485static ra8_err_t internal_write_data_block(const uint8_t* buf, uint8_t start_token)
486{
487 ra8_err_t err = ra8_sdmmc_spi_send_idle(1U); /* N_WR pad (spec >= 1 byte). */
488 if (err != k_ra8_ok) {
489 return err;
490 }
491 err = ra8_sdmmc_spi_xfer_one(start_token, nullptr);
492 if (err != k_ra8_ok) {
493 return err;
494 }
495 err = s_sdmmc_spi_state.transport.xfer(s_sdmmc_spi_state.transport.ctx,
496 buf,
497 nullptr,
498 (uint32_t)k_ra8_sdmmc_spi_block_size);
499 if (err != k_ra8_ok) {
500 /* Some transports require non-NULL rx -- fall back to per-byte. */
501 for (uint32_t i = 0U; i < (uint32_t)k_ra8_sdmmc_spi_block_size; i++) {
502 uint8_t dummy = 0U;
503 err = ra8_sdmmc_spi_xfer_one(buf[i], &dummy);
504 if (err != k_ra8_ok) {
505 return err;
506 }
507 }
508 }
509 const uint16_t crc = ra8_sdmmc_spi_crc16(buf, (uint32_t)k_ra8_sdmmc_spi_block_size);
510 (void)ra8_sdmmc_spi_xfer_one(
511 (uint8_t)((uint32_t)(crc >> k_sd_bit_byte) & (uint32_t)k_sd_mask_byte),
512 nullptr);
513 (void)ra8_sdmmc_spi_xfer_one((uint8_t)((uint32_t)crc & (uint32_t)k_sd_mask_byte), nullptr);
514 uint8_t response = 0U;
515 err = ra8_sdmmc_spi_xfer_one((uint8_t)k_sd_token_idle, &response);
516 if (err != k_ra8_ok) {
517 return err;
518 }
519 if ((response & (uint8_t)k_sd_data_response_mask) != (uint8_t)k_sd_data_response_accepted) {
520 (void)ra8_sdmmc_spi_wait_not_busy();
521 return k_ra8_err_protocol_error;
522 }
523 return ra8_sdmmc_spi_wait_not_busy();
524}

631ra8_err_t ra8_sdmmc_spi_erase_blocks(uint32_t lba, uint32_t count)
632{
633 if (!s_sdmmc_spi_state.initialized) {
634 return k_ra8_err_invalid_state;
635 }
636 if (count == 0U) {
637 return k_ra8_err_invalid_arg;
638 }
639 if ((lba >= s_sdmmc_spi_state.capacity_blocks) ||
640 (count > (s_sdmmc_spi_state.capacity_blocks - lba))) {
641 return k_ra8_err_out_of_range;
642 }
643 /* Probe: erase only the FIRST block and read it back. The SD post-erase value
644 * is card-dependent (0x00 on some, 0xFF on others), and the SCR
645 * DATA_STAT_AFTER_ERASE bit's polarity is unreliable in practice, so measure
646 * it. A non-zero read-back means this card erases to ones: report "not
647 * supported" so the caller writes zeros -- and skip erasing the rest, which
648 * would be wasted work (a one-block probe instead of the whole region). */
649 ra8_err_t err = internal_erase_range(lba, 1U);
650 if (err != k_ra8_ok) {
651 return err;
652 }
653 uint8_t blk[k_ra8_sdmmc_spi_block_size] = {};
654 err = ra8_sdmmc_spi_read_block(lba, blk);
655 if (err != k_ra8_ok) {
656 return err;
657 }
658 for (uint32_t i = 0U; i < (uint32_t)k_ra8_sdmmc_spi_block_size; i++) {
659 if (blk[i] != 0U) {
660 return k_ra8_err_not_supported;
661 }
662 }
663 /* The card erases to zero: erase the remaining range in one operation. */
664 if (count > 1U) {
665 err = internal_erase_range(lba + 1U, count - 1U);
666 if (err != k_ra8_ok) {
667 return err;
668 }
669 }
670 return k_ra8_ok;
671}
