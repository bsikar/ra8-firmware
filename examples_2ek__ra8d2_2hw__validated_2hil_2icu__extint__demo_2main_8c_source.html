105int32_t main(void)
106{
107 icu_extint_demo_setup_or_halt();
108 ra8_isr_globals_enable();
109
110 if (icu_extint_demo_arm() != k_ra8_ok) {
111 icu_extint_demo_panic_halt();
112 }
113
114 /* HIL boot banner -- scraped by scripts/hil/run_direct.sh to confirm
115 * the CGC + SCI + ICU bring-up reached the main poll loop. */
116 (void)ra8_board_uart_console_write(k_icu_extint_demo_msg_boot,
117 (size_t)(sizeof(k_icu_extint_demo_msg_boot) - 1U));
118
119 while (1) {
120 uint8_t irqcr = 0U;
121 if (ra8_icu_read_irqcr((uint8_t)k_icu_extint_demo_irq_num, &irqcr) != k_ra8_ok) {
122 break;
123 }
124 if ((irqcr & (uint8_t)k_icu_extint_demo_irqf_mask) != 0U) {
125 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
126 break;
127 }
128 if (ra8_board_uart_console_write(k_icu_extint_demo_msg_press,
129 (size_t)(sizeof(k_icu_extint_demo_msg_press) - 1U)) !=
130 k_ra8_ok) {
131 break;
132 }
133 }
134 ra8_delay_ms((uint32_t)k_icu_extint_demo_poll_ms);
135 }
136 icu_extint_demo_panic_halt();
137 return 0;
138}
