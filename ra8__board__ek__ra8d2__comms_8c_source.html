139ra8_err_t ra8_board_mipi_dsi_init(void)
140{
141 /* Step 1: PHY first -- HUM Ch 64.3.1 startup procedure. The HAL warns
142 * "The MIPI PHY (HUM Ch 64) must be brought up first" before the DSI
143 * host can clock its LP/HS lanes. */
144 ra8_err_t err = ra8_mipi_phy_init(&s_mipi_phy_cfg);
145 if (err != k_ra8_ok) {
146 return err;
147 }
148
149 /* Step 2: DSI host link layer (HUM Ch 65). Programmes TXSETR / DSISETR
150 * / guard-band timing / timeouts -- but does NOT start the HS clock
151 * yet, so the application can splice in the panel-side reset pulse on
152 * k_ra8_board_mipi_dsi_reset_n (P606) and backlight enable on
153 * k_ra8_board_mipi_dsi_backlight (P514) between init and clock start. */
154 err = ra8_mipi_dsi_init(&s_mipi_panel_cfg);
155 if (err != k_ra8_ok) {
156 /* ra8_mipi_dsi_init with a valid static cfg always returns k_ra8_ok in RA8_OFF_TARGET. */
157 return err; /* GCOVR_EXCL_LINE */
158 }
159
160 /* Step 3: kick the differential HS clock. After this returns the link
161 * is HS-ready; callers can replay the panel DCS init stream via
162 * ra8_mipi_dsi_send_command() and finally call ra8_mipi_dsi_video_start.
163 *
164 * TODO(panel-datasheet): the per-panel DCS command sequence (sleep-out,
165 * pixel-format set, display-on, etc.) for the Focus E45RA-MW276-C is
166 * not committed here -- the application currently owns it. */
167 return ra8_mipi_dsi_hs_clock_start();
168}

207ra8_err_t ra8_board_uart_console_init(uint32_t baud)
208{
209 if (baud == 0U) {
210 return k_ra8_err_invalid_arg;
211 }
212
213 /* SCI8's baud generator is fed by PCLKA on RA8D2 (chip HUM Ch 38.2
214 * "SCI registers"). Hardcoding a constant here was the bug behind
215 * every garbled UART symptom on apps that retune CGC: the BRR was
216 * being computed for an assumed 60 MHz against a real 125 MHz clock,
217 * so the line rate came out 2x too fast. Read PCLKA from CGC at
218 * runtime instead so the BRR tracks whatever the application's CGC
219 * tree settled on. */
220 uint32_t pclka_hz = 0U;
221 ra8_err_t err = ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz);
222 if (err != k_ra8_ok) {
223 /* ra8_cgc_get_clock_hz with a valid clock-id and non-null out always returns k_ra8_ok. */
224 return err; /* GCOVR_EXCL_LINE */
225 }
226 if (pclka_hz < (uint32_t)k_ra8_board_uart_console_min_pclka_hz) {
227 return k_ra8_err_not_initialized;
228 }
229
230 /* Route PD02 -> TXD8, PD03 -> RXD8 (UM Table 13 p 24). PSEL value
231 * k_ra8_psel_sci_async = 0x04 covers SCI async TXD/RXD per the chip
232 * HUM Multiplexed Pin Function Selector. */
233 /* NOLINTBEGIN(clang-analyzer-optin.core.EnumCastOutOfRange) -- RA8_PIN()-packed board pin is valid ra8_port_pin_t data outside the enumerator list. */
234 err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_ra8_board_uart_console_pin_txd,
235 k_ra8_psel_sci_async,
236 "ra8_board.uart.console.txd");
237 if (err != k_ra8_ok) {
238 return err;
239 }
240 err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_ra8_board_uart_console_pin_rxd,
241 k_ra8_psel_sci_async,
242 "ra8_board.uart.console.rxd");
243 /* NOLINTEND(clang-analyzer-optin.core.EnumCastOutOfRange) */
244 if (err != k_ra8_ok) {
245 return err;
246 }

251 .parity = k_ra8_sci_parity_none,
252 .stop_bits = k_ra8_sci_stop_1,

262}

264ra8_err_t ra8_board_uart_console_write(const uint8_t* data, size_t len)
265{
266 if (len == 0U) {
267 return k_ra8_ok;
268 }
269 if (data == nullptr) {
270 return k_ra8_err_invalid_arg;
271 }

278ra8_err_t ra8_board_uart_console_read(uint8_t* out, size_t cap, size_t* out_len)
279{
280 if (out_len == nullptr) {
281 return k_ra8_err_invalid_arg;
282 }
283 *out_len = 0U;
284 if (cap == 0U) {
285 return k_ra8_ok;
286 }
287 if (out == nullptr) {
288 return k_ra8_err_invalid_arg;
289 }
290 if (!s_uart_console_initialized) {
291 return k_ra8_err_not_initialized;
292 }
293
294 /* Polled non-blocking drain: pull bytes while RDRF stays set, stop
295 * the moment ra8_sci_getc_polling reports nothing available. The cap
296 * bounds the loop (NASA Rule 2). */
297 for (size_t i = 0U; i < cap; ++i) {
298 uint8_t byte = 0U;
299 const ra8_err_t err =
300 ra8_sci_getc_polling((uint8_t)k_ra8_board_uart_console_sci_channel, &byte);
301 if (err != k_ra8_ok) {
302 /* No byte available yet -- treat as a non-blocking stop. */
303 return k_ra8_ok;
304 }
305 out[i] = byte;
306 *out_len += 1U;
307 }
308 return k_ra8_ok;
309}

311ra8_err_t ra8_board_uart_console_flush(void)
312{
313 if (!s_uart_console_initialized) {
314 return k_ra8_err_not_initialized;
315 }
316 /* HUM Ch 38.2.17 "CSR : Common Status Register", p 2225 -- defer to
317 * ra8_sci_flush which spins on CSR.TEND for the SCI8 channel that
318 * backs this console. Lets panic_halt() drain the failure log before
319 * WFI gates the SCI clock. */
320 return ra8_sci_flush((uint8_t)k_ra8_board_uart_console_sci_channel);
321}
