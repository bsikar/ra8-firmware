189[[nodiscard]] static ra8_err_t selftest_print_dec(uint32_t value)
190{
191 uint8_t scratch[k_selftest_dec_chars_u32] = {};
192 uint8_t out[k_selftest_dec_chars_u32] = {};
193 uint8_t count = 0U;
194 uint32_t v = value;
195 if (v == 0U) {
196 out[0] = (uint8_t)'0';
197 return selftest_sci_write(out, 1U);
198 }
199 while (v != 0U) {
200 if (count >= (uint8_t)k_selftest_dec_chars_u32) {
201 break;
202 }
203 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_selftest_dec_radix));
204 v = v / k_selftest_dec_radix;
205 count++;
206 }
207 for (uint8_t i = 0U; i < count; i++) {
208 out[i] = scratch[count - 1U - i];
209 }
210 return selftest_sci_write(out, (uint32_t)count);
211}

232[[nodiscard]] static ra8_err_t selftest_print_hex(uint32_t value, uint8_t digits)
233{
234 uint8_t out[k_selftest_hex_chars_u32] = {};
235 uint8_t width = digits;
236 if (width > (uint8_t)k_selftest_hex_chars_u32) {
237 width = (uint8_t)k_selftest_hex_chars_u32;
238 }
239 for (uint8_t i = 0U; i < width; i++) {
240 const uint8_t shift = (uint8_t)((width - 1U - i) * k_selftest_nibble_bits);
241 out[i] = selftest_nibble_to_hex((value >> shift) & k_selftest_nibble_mask);
242 }
243 return selftest_sci_write(out, (uint32_t)width);
244}

502[[nodiscard]] static ra8_err_t selftest_verify_mram_raw(void)
503{
504 static uint8_t s_burst[k_selftest_burst_bytes] = {};
505
506 s_dbg_verified_bytes = 0U;
507 s_dbg_mismatch_off = (uint32_t)k_selftest_no_mismatch;
508 const uint32_t t0 = ra8_time_ms();
509 for (uint32_t blk = 0U; blk < (uint32_t)k_fat_mram_clusters;
510 blk += (uint32_t)k_selftest_burst_blocks) {
511 const uint32_t lba = (uint32_t)k_fat_data_lba + blk;
512 ra8_err_t err = ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun,
513 lba,
514 (uint16_t)k_selftest_burst_blocks,
515 s_burst);
516 if (err != k_ra8_ok) {
517 (void)selftest_print_fail("READ(10) burst", err);
518 return err;
519 }
520 const uint32_t offset = blk * (uint32_t)k_selftest_block_size;
521 const uint8_t* mram = (const uint8_t*)(uintptr_t)((uint32_t)k_mram_base_addr + offset);
522 if (memcmp(s_burst, mram, (size_t)k_selftest_burst_bytes) != 0) {
523 s_dbg_mismatch_off = offset;
524 (void)selftest_print_fail("content mismatch at 0x", (ra8_err_t)offset);
525 return k_ra8_err_invalid_state;
526 }
527 s_dbg_verified_bytes = offset + (uint32_t)k_selftest_burst_bytes;
528 }
529 s_dbg_verify_ms = ra8_time_ms() - t0;
530 return k_ra8_ok;
531}

717[[nodiscard]] static ra8_err_t selftest_host_pass(void)
718{
719 ra8_usb_hmsc_device_t device = {};
720 ra8_err_t err = selftest_host_enumerate(&device);
721 if (err != k_ra8_ok) {
722 return err;
723 }
724
725 /* Mount proves the host can PARSE the FAT16 volume over the loop;
726 * then unmount and do the heavy integrity check with raw multi-block
727 * READ(10) (ra8_fs's per-cluster metadata re-reads are far too slow
728 * for a 1 MiB sweep). */
729 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
730 ra8_fs_mount_t* mount = nullptr;
731 err = selftest_mount_volume(&mount);
732 if (err != k_ra8_ok) {
733 (void)ra8_usb_hmsc_close();
734 return err;
735 }
736 (void)ra8_fs_unmount(mount);
737
738 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
739 err = selftest_verify_mram_raw();
740 if (err == k_ra8_ok) {
741 err = selftest_print_verify_verdict();
742 }
743 if (err == k_ra8_ok) {
744 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
745 err = selftest_write_protect_probe();
746 }
747 if (err != k_ra8_ok) {
748 (void)ra8_usb_hmsc_close();
749 return err;
750 }
751
752 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
753 s_dbg_pass_count++;
754 err = selftest_print("ra8d2 selftest: USB SELFTEST CONFIG B PASS\r\n");
755 if (err != k_ra8_ok) {
756 return err;
757 }
758 (void)ra8_board_led_on(k_ra8_board_led2);
759 return k_ra8_ok;
760}

781{
