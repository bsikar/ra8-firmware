189[[nodiscard]] static ra8_err_t selftest_print_dec(uint32_t value)
190{
191 uint8_t scratch[k_selftest_dec_chars_u32] = {};
192 uint8_t out[k_selftest_dec_chars_u32] = {};
193 uint8_t count = 0U;
194 uint32_t v = value;
195 if (v == 0U) {
196 out[0] = (uint8_t)'0';
197 return selftest_sci_write(out, 1U);
198 }
199 while (v != 0U) {
200 if (count >= (uint8_t)k_selftest_dec_chars_u32) {
201 break;
202 }
203 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_selftest_dec_radix));
204 v = v / k_selftest_dec_radix;
205 count++;
206 }
207 for (uint8_t i = 0U; i < count; i++) {
208 out[i] = scratch[count - 1U - i];
209 }
210 return selftest_sci_write(out, (uint32_t)count);
211}

232[[nodiscard]] static ra8_err_t selftest_print_hex(uint32_t value, uint8_t digits)
233{
234 uint8_t out[k_selftest_hex_chars_u32] = {};
235 uint8_t width = digits;
236 if (width > (uint8_t)k_selftest_hex_chars_u32) {
237 width = (uint8_t)k_selftest_hex_chars_u32;
238 }
239 for (uint8_t i = 0U; i < width; i++) {
240 const uint8_t shift = (uint8_t)((width - 1U - i) * k_selftest_nibble_bits);
241 out[i] = selftest_nibble_to_hex((value >> shift) & k_selftest_nibble_mask);
242 }
243 return selftest_sci_write(out, (uint32_t)width);
244}

485[[nodiscard]] static ra8_err_t selftest_verify_mram_raw(void)
486{
487 static uint8_t s_burst[k_selftest_burst_bytes] = {};
488
489 s_dbg_verified_bytes = 0U;
490 s_dbg_mismatch_off = (uint32_t)k_selftest_no_mismatch;
491 const uint32_t t0 = ra8_time_ms();
492 for (uint32_t blk = 0U; blk < (uint32_t)k_fat_mram_clusters;
493 blk += (uint32_t)k_selftest_burst_blocks) {
494 const uint32_t lba = (uint32_t)k_fat_data_lba + blk;
495 ra8_err_t err = ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun,
496 lba,
497 (uint16_t)k_selftest_burst_blocks,
498 s_burst);
499 if (err != k_ra8_ok) {
500 (void)selftest_print_fail("READ(10) burst", err);
501 return err;
502 }
503 const uint32_t offset = blk * (uint32_t)k_selftest_block_size;
504 const uint8_t* mram = (const uint8_t*)(uintptr_t)((uint32_t)k_mram_base_addr + offset);
505 if (memcmp(s_burst, mram, (size_t)k_selftest_burst_bytes) != 0) {
506 s_dbg_mismatch_off = offset;
507 (void)selftest_print_fail("content mismatch at 0x", (ra8_err_t)offset);
508 return k_ra8_err_invalid_state;
509 }
510 s_dbg_verified_bytes = offset + (uint32_t)k_selftest_burst_bytes;
511 }
512 s_dbg_verify_ms = ra8_time_ms() - t0;
513 return k_ra8_ok;
514}

700[[nodiscard]] static ra8_err_t selftest_host_pass(void)
701{
702 ra8_usb_hmsc_device_t device = {};
703 ra8_err_t err = selftest_host_enumerate(&device);
704 if (err != k_ra8_ok) {
705 return err;
706 }
707
708 /* Mount proves the host can PARSE the FAT16 volume over the loop;
709 * then unmount and do the heavy integrity check with raw multi-block
710 * READ(10) (ra8_fs's per-cluster metadata re-reads are far too slow
711 * for a 1 MiB sweep). */
712 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
713 ra8_fs_mount_t* mount = nullptr;
714 err = selftest_mount_volume(&mount);
715 if (err != k_ra8_ok) {
716 (void)ra8_usb_hmsc_close();
717 return err;
718 }
719 (void)ra8_fs_unmount(mount);
720
721 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
722 err = selftest_verify_mram_raw();
723 if (err == k_ra8_ok) {
724 err = selftest_print_verify_verdict();
725 }
726 if (err == k_ra8_ok) {
727 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
728 err = selftest_write_protect_probe();
729 }
730 if (err != k_ra8_ok) {
731 (void)ra8_usb_hmsc_close();
732 return err;
733 }
734
735 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
736 s_dbg_pass_count++;
737 err = selftest_print("ra8d2 selftest: USB SELFTEST CONFIG B PASS\r\n");
738 if (err != k_ra8_ok) {
739 return err;
740 }
741 (void)ra8_board_led_on(k_ra8_board_led2);
742 return k_ra8_ok;
743}

763VOID selftest_host_worker(ULONG arg)
764{
765 (void)arg;
766
767 tx_thread_sleep(k_selftest_boot_wait_ticks);
768 for (;;) {
769 const ra8_err_t err = selftest_host_pass();
770 if (err == k_ra8_ok) {
771 break;
772 }
773 tx_thread_sleep(k_selftest_retry_ticks);
774 }
775 while (1) {
776 tx_thread_sleep(k_selftest_idle_ticks);
777 }
778}
