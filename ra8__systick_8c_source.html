90ra8_err_t ra8_systick_reload_for(uint32_t cpu_hz, uint32_t tick_hz, uint32_t* out_reload)
91{
92 RA8_CHECK_NULL_PTR(out_reload, s_tag, "out_reload is NULL");
93
94 /* Reject a zero clock or a zero tick rate: both make the reload arithmetic
95 * meaningless (a 0 Hz core, or a division by zero). */
96 if ((cpu_hz == 0U) || (tick_hz == 0U)) {
97 ra8_log_error(s_tag, "cpu_hz / tick_hz must be non-zero");
98 return k_ra8_err_invalid_arg;
99 }
100
101 const uint32_t ticks = cpu_hz / tick_hz;
102 if (ticks == 0U) {
103 /* Clock slower than one tick period -> reload would underflow. */
104 ra8_log_error(s_tag, "cpu_hz below one tick period");
105 return k_ra8_err_invalid_arg;
106 }
107
108 const uint32_t reload = ticks - 1U;
109 if (reload > (uint32_t)k_ra8_systick_rvr_max) {
110 /* Too high for the 24-bit SYST_RVR field: refuse rather than truncate,
111 * which would silently make the tick far too fast. */
112 ra8_log_error(s_tag, "reload exceeds 24-bit SysTick range");
113 return k_ra8_err_out_of_range;
114 }
115
116 *out_reload = reload;
117 return k_ra8_ok;
118}

120ra8_err_t ra8_systick_configure(uint32_t reload, ra8_systick_clock_source_t src, bool tick_irq)
121{
122 if (reload > (uint32_t)k_ra8_systick_rvr_max) {
123 ra8_log_error(s_tag, "reload exceeds 24-bit SysTick range");
124 return k_ra8_err_out_of_range;
125 }
126
127 /* Arm v8-M ARM: SYST_CSR -- clear ENABLE first so the reload/current writes
128 * land on a stopped counter, then re-enable with the requested config. */
129 *ra8_systick_reg(k_ra8_systick_csr) = 0U;
130 *ra8_systick_reg(k_ra8_systick_rvr) = reload;
131 *ra8_systick_reg(k_ra8_systick_cvr) = 0U;
132
133 uint32_t csr_val = (uint32_t)k_ra8_systick_csr_enable;
134 if (tick_irq) {
135 csr_val |= (uint32_t)k_ra8_systick_csr_tickint;
136 }
137 if (src == k_ra8_systick_clk_cpu) {
138 csr_val |= (uint32_t)k_ra8_systick_csr_clksource;
139 }
140 *ra8_systick_reg(k_ra8_systick_csr) = csr_val;
141 return k_ra8_ok;
142}

144ra8_err_t ra8_systick_set_reload(uint32_t reload)
145{
146 if (reload > (uint32_t)k_ra8_systick_rvr_max) {
147 ra8_log_error(s_tag, "reload exceeds 24-bit SysTick range");
148 return k_ra8_err_out_of_range;
149 }
150
151 /* Arm v8-M ARM: re-arm the reload, then any write to SYST_CVR clears it so
152 * the next count starts from the new reload. The SYST_CSR control bits are
153 * deliberately left as-is. */
154 *ra8_systick_reg(k_ra8_systick_rvr) = reload;
155 *ra8_systick_reg(k_ra8_systick_cvr) = 0U;
156 return k_ra8_ok;
157}

166void ra8_dwt_cyccnt_enable(void)
167{
168 /* Arm v8-M ARM: DEMCR.TRCENA unlocks the DWT unit; DWT_CTRL.CYCCNTENA then
169 * starts DWT_CYCCNT. Read-modify-write preserves any other trace bits. */
170 *ra8_systick_reg(k_ra8_dwt_demcr) |= (uint32_t)k_ra8_dwt_demcr_trcena;
171 *ra8_systick_reg(k_ra8_dwt_ctrl) |= (uint32_t)k_ra8_dwt_ctrl_cyccntena;
172}
