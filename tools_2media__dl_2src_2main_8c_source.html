176{

194RA8_INTERNAL static void sort_by_chapter_num(mdl_url_list_t* l)
195{
196 for (size_t i = 0U; i + 1U < l->count; ++i) {
197 size_t min = i;
198 for (size_t j = i + 1U; j < l->count; ++j) {
199 if (mdl_urlname_chapter_number(l->urls[j]) < mdl_urlname_chapter_number(l->urls[min])) {
200 min = j;
201 }
202 }
203 swap_rows(l, i, min);
204 }

218RA8_INTERNAL static void filter_prefix(mdl_url_list_t* l, const char* prefix)
219{
220 const size_t plen = strlen(prefix);
221 size_t w = 0U;
222 for (size_t r = 0U; r < l->count; ++r) {
223 if (strncmp(l->urls[r], prefix, plen) == 0) {
224 if (w != r) {
225 memcpy(l->urls[w], l->urls[r], k_mdl_url_max);
226 }
227 ++w;
228 }
229 }
230 l->count = w;

246RA8_INTERNAL static ra8_err_t
247prepare_chapters(const mdl_site_t* site, const char* series_url, uint32_t timeout)
248{
249 if (!mdl_session_url_allowed(&s_session, series_url, nullptr)) {
250 return k_ra8_fail; /* robots refused the series page (message printed) */
251 }
252 const mdl_net_req_t req = {.user_agent = s_session.user_agent,
253 .referer = nullptr,
254 .timeout_ms = timeout};
255 size_t len = 0U;
256 ra8_err_t rc =
257 mdl_net_get_buf(s_session.net, series_url, &req, s_page, sizeof(s_page), &len, nullptr);
258 if (rc != k_ra8_ok) {
259 return rc;
260 }
261 (void)mdl_extract_anchors(s_page, len, series_url, site->chapter_url_contains, &s_chapters);
262
263 /* Keep only this series' chapters (drop cross-series relation links). */
264 char prefix[k_dir_path_bytes];
265 (void)snprintf(prefix, sizeof(prefix), "%s", series_url);
266 size_t pl = strlen(prefix);
267 if ((pl > 0U) && (prefix[pl - 1U] != '/') && (pl + 1U < sizeof(prefix))) {
268 prefix[pl] = '/';
269 prefix[pl + 1U] = '\0';
270 }
271 filter_prefix(&s_chapters, prefix);
272 apply_order(&s_chapters, site->chapter_order);
273 return k_ra8_ok;
274}

316RA8_INTERNAL static void list_range(const mdl_url_list_t* l, long* lo, long* hi)
317{
318 *lo = 0;
319 *hi = 0;
320 if (l->count == 0U) {
321 return;
322 }
323 *lo = mdl_urlname_chapter_number(l->urls[0]);
324 *hi = *lo;
325 for (size_t i = 0U; i < l->count; ++i) {
326 const long num = mdl_urlname_chapter_number(l->urls[i]);
327 *lo = (num < *lo) ? num : *lo;
328 *hi = (num > *hi) ? num : *hi;
329 }
330}

333RA8_INTERNAL static void
334select_window(bool from_present, long from_num, size_t count, mdl_url_list_t* out)
335{
336 size_t start = 0U;
337 if (from_present) {
338 while ((start < s_chapters.count) &&
339 (mdl_urlname_chapter_number(s_chapters.urls[start]) < from_num)) {
340 ++start;
341 }
342 }
343 out->count = 0U;
344 for (size_t i = start; (i < s_chapters.count) && (out->count < count); ++i) {
345 memcpy(out->urls[out->count], s_chapters.urls[i], k_mdl_url_max);
346 out->count += 1U;
347 }
348}

476RA8_INTERNAL static int run_prepared(const series_run_t* r,
477 const mdl_site_t* site,
478 const char* abs_dir,
479 const char* slug,
480 const char* state_path)
481{
482 const mdl_url_list_t* sel = &s_chapters;
483 if (!r->update) {
484 select_window(r->from_present, r->from_num, r->chapters, &s_selected);
485 sel = &s_selected;
486 if (sel->count == 0U) {
487 (void)fprintf(stderr, "media_dl: no chapters match the requested range\n");
488 return 1;
489 }
490 }
491 char combined_rel[k_leaf_name_bytes];
492 mdl_fetch_layout_t layout = choose_layout(r, sel, combined_rel, sizeof(combined_rel), slug);
493
494 mdl_governor_t gov;
495 const mdl_gov_cfg_t cfg = mdl_config_gov_cfg(site);
496 mdl_governor_init(&gov, &cfg, r->seed);
497 memset(&s_faillog, 0, sizeof(s_faillog));
498 mdl_fetch_ctx_t ctx = make_ctx(r, site, abs_dir, state_path, &gov);
499 const int64_t run_start = (int64_t)time(nullptr);
500 mdl_fetch_stats_t stats;
501 const ra8_err_t frc = mdl_fetch_run(&ctx,
502 sel,
503 layout,
504 (layout == k_mdl_layout_combined) ? combined_rel : nullptr,
505 &stats);
506 (void)mdl_state_save(state_path, &s_state);
507 report_stats(abs_dir, &stats);
508 mdl_report_failures(&s_faillog);
509 const size_t efail = export_after(r, abs_dir, layout, combined_rel, sel, &stats, run_start);
510 return ((frc == k_ra8_ok) && (efail == 0U)) ? 0 : 1;
511}

514RA8_INTERNAL static int run_series(const series_run_t* r)
515{
516 mdl_site_t site;
517 if (mdl_config_load(r->cfg_path, &site) != k_ra8_ok) {
518 return 1;
519 }
520 if (r->opts->polite) {
521 mdl_config_apply_polite(&site);
522 }
523 (void)printf("site: %s (host %s, kind %s)\n", site.name, site.host, site.kind);
524
525 mdl_net_iface_t* net = mdl_net_curl_create(&r->opts->policy);
526 if (net == nullptr) {
527 (void)fprintf(stderr, "media_dl: network init failed\n");
528 return 1;
529 }
530 char ua[k_mdl_ua_max];
531 start_session(net, r->opts, site.contact, ua, sizeof(ua));
532
533 int rc = 1;
534 if (prepare_chapters(&site, r->series_url, r->timeout) != k_ra8_ok) {
535 (void)fprintf(stderr, "media_dl: series fetch failed\n");
536 } else if (s_chapters.count == 0U) {
537 (void)fprintf(stderr, "media_dl: no chapters (check chapter_url_contains)\n");
538 } else {
539 (void)printf("chapters found: %zu\n", s_chapters.count);
540 char slug[k_slug_bytes];
541 char abs_dir[PATH_MAX];
542 char state_path[PATH_MAX];
543 if (prepare_series_dir(r->out_dir, r->series_url, slug, sizeof(slug), abs_dir) &&
544 state_path_of(abs_dir, state_path, sizeof(state_path))) {
545 (void)mdl_state_load(state_path, &s_state); /* corrupt -> rebuild (message printed) */
546 mdl_state_set_series(&s_state, r->series_url, slug, site.name, site.host, r->cfg_path);
547 rc = run_prepared(r, &site, abs_dir, slug, state_path);
548 }
549 }
550 mdl_net_destroy(net);
551 return rc;
552}

653RA8_INTERNAL static size_t download_page_image(const char* url,
654 const char* out_dir,
655 uint32_t dmin,
656 uint32_t dmax,
657 uint32_t timeout,
658 mdl_politeness_t* pol,
659 size_t idx)
660{
661 uint32_t crawl = 0U;
662 if (!mdl_session_url_allowed(&s_session, s_images.urls[idx], &crawl)) {
663 return 1U;
664 }
665 (void)mdl_politeness_wait(pol, max_u32(dmin, crawl), max_u32(dmax, crawl));
666 char ext[k_ext_bytes];
667 mdl_urlname_ext(s_images.urls[idx], ext, sizeof(ext));
668 char path[k_file_path_bytes];
669 (void)snprintf(path, sizeof(path), "%s/page_%03zu.%s", out_dir, idx + 1U, ext);
670 const mdl_net_req_t ir = {.user_agent = s_session.user_agent,
671 .referer = url,
672 .timeout_ms = timeout};
673 size_t got = 0U;
674 mdl_net_resp_t resp = {};
675 const ra8_err_t rc = mdl_net_get_file(s_session.net, s_images.urls[idx], &ir, path, &got, &resp);
676 if (rc != k_ra8_ok) {
677 char reason[k_mdl_reason_max];
678 mdl_fetch_reason(rc, resp.status, reason, sizeof(reason));
679 (void)fprintf(stderr, " page %zu FAILED %s -- %s\n", idx + 1U, s_images.urls[idx], reason);
680 return 1U;
681 }
682 return 0U;
683}

686RA8_INTERNAL static size_t download_page_images(const char* url,
687 const char* out_dir,
688 uint32_t max_imgs,
689 uint64_t seed,
690 uint32_t timeout,
691 bool polite)
692{
693 mdl_politeness_t pol;
694 mdl_politeness_init(&pol, seed);
695 const uint32_t dmin = polite ? (uint32_t)k_polite_img_min_ms : (uint32_t)k_page_img_delay_min;
696 const uint32_t dmax = polite ? (uint32_t)k_polite_img_max_ms : (uint32_t)k_page_img_delay_max;
697 const size_t limit = (max_imgs == 0U) ? s_images.count : (size_t)max_imgs;
698 size_t fail = 0U;
699 for (size_t i = 0U; (i < s_images.count) && (i < limit); ++i) {
700 fail += download_page_image(url, out_dir, dmin, dmax, timeout, &pol, i);
701 }
702 (void)printf("done: %zu ok, %zu failed, into %s/\n",
703 (s_images.count < limit ? s_images.count : limit) - fail,
704 fail,
705 out_dir);
706 return fail;
707}

711 const char* out_dir,
712 const char* attr,

714 uint64_t seed,
715 uint32_t timeout,
716 const mdl_run_opts_t* opts)
717{

732 .timeout_ms = timeout};

753RA8_INTERNAL static int run_pack(const char* dir, mdl_format_t format)
754{
755 if ((format == k_mdl_fmt_loose) || (format == k_mdl_fmt_invalid)) {
756 (void)fprintf(stderr,
757 "media_dl: --pack needs a --format "
758 "(cbz|cbt|cbt.gz|cbt.xz|cbr|epub|jof|rabook)\n");
759 return 2;
760 }
761 char abs[PATH_MAX];
762 if (realpath(dir, abs) == nullptr) {
763 (void)fprintf(stderr, "media_dl: cannot resolve '%s'\n", dir);
764 return 1;
765 }
766 const char* ext = mdl_format_ext(format);
767 if (mdl_format_is_dir_output(format)) {
768 /* JOF writes per-page `.jof` siblings into the packed directory itself;
769 * report that directory rather than a container file it never creates. */
770 const ra8_err_t drc = mdl_export_chapter(format, abs, abs);
771 if (drc != k_ra8_ok) {
772 (void)
773 fprintf(stderr, "media_dl: pack '%s' as .%s FAILED (err 0x%X)\n", dir, ext, (unsigned)drc);
774 return 1;
775 }
776 (void)printf("packed %s -> %s/*.%s\n", dir, abs, ext);
777 return 0;
778 }
779 char out[PATH_MAX];
780 const int n = snprintf(out, sizeof(out), "%s.%s", abs, ext);
781 if ((n < 0) || ((size_t)n >= sizeof(out))) {
782 (void)fprintf(stderr, "media_dl: output path for '%s' is too long\n", dir);
783 return 1;
784 }
785 const ra8_err_t rc = mdl_export_chapter(format, abs, out);
786 if (rc != k_ra8_ok) {
787 (void)fprintf(stderr, "media_dl: pack '%s' as .%s FAILED (err 0x%X)\n", dir, ext, (unsigned)rc);
788 return 1;
789 }
790 (void)printf("packed %s -> %s\n", dir, out);
791 return 0;
792}

814RA8_INTERNAL static int run_discover(const mdl_args_t* a,
815 const mdl_run_opts_t* opts,
816 const mdl_nums_t* n,
817 const series_run_t* base)
818{
819 if (a->cfg == nullptr) {
820 (void)fprintf(stderr, "media_dl: --search/--browse requires --config SITE.conf\n");
821 return 2;
822 }
823 mdl_site_t site;
824 if (mdl_config_load(a->cfg, &site) != k_ra8_ok) {
825 return 1;
826 }
827 if (opts->polite) {
828 mdl_config_apply_polite(&site);
829 }
830 mdl_net_iface_t* net = mdl_net_curl_create(&opts->policy);
831 if (net == nullptr) {
832 (void)fprintf(stderr, "media_dl: network init failed\n");
833 return 1;
834 }

836 start_session(net, opts, site.contact, ua, sizeof(ua));
837
838 mdl_governor_t gov;
839 const mdl_gov_cfg_t cfg = mdl_config_gov_cfg(&site);
840 mdl_governor_init(&gov, &cfg, n->seed);
841
842 const mdl_discover_req_t req = {
843 .session = &s_session,
844 .gov = &gov,
845 .site = &site,
846 .mode = a->browse ? k_mdl_discover_browse : k_mdl_discover_search,
847 .term = a->search,
848 .timeout_ms = n->timeout,
849 .page_buf = s_page,
850 .page_cap = sizeof(s_page),
851 .hits = &s_results,
852 };

854 const int rc = mdl_discover_run(&req, n->pick, picked, sizeof(picked));
855 mdl_net_destroy(net);

857 return rc; /* listed results, or an error, with nothing to download */
858 }
859 series_run_t run = *base;
860 run.series_url = picked; /* combined search-and-select: feed the pick in */
861 return run_series(&run);
862}

881RA8_INTERNAL static int dispatch_run(const mdl_args_t* a,
882 mdl_format_t format,
883 const mdl_run_opts_t* opts,
884 const mdl_nums_t* nums,
885 const series_run_t* run,
886 const char* prog)
887{
888 if (a->list || (a->remove_series != nullptr) || a->update_all) {
889 return run_library(a, run);
890 }
891 if ((a->search != nullptr) || a->browse) {
892 return run_discover(a, opts, nums, run);
893 }
894 if (a->pack != nullptr) {
895 return run_pack(a->pack, format);
896 }
897 if (a->cfg != nullptr) {
898 if (a->series == nullptr) {
899 (void)fprintf(stderr, "media_dl: --config requires --series URL\n");
900 return 2;
901 }
902 return run_series(run);
903 }
904 if (a->page_url != nullptr) {
905 return run_page(a->page_url, a->out, a->attr, nums->max_imgs, nums->seed, nums->timeout, opts);
906 }
907 mdl_cli_usage(prog);
908 return 2;
909}
