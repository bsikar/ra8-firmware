285float ra8_svgp_numf(const uint8_t* s, size_t len, size_t* i)
286{
287 while ((*i < len) && (ra8_svgp_ws((char)s[*i]) || (s[*i] == ','))) {
288 ++(*i);
289 }
290 float sgn = 1.0F;
291 if ((*i < len) && ((s[*i] == '-') || (s[*i] == '+'))) {
292 sgn = (s[*i] == '-') ? -1.0F : 1.0F;
293 ++(*i);
294 }
295 float v = 0.0F;
296 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9')) {
297 v = (v * (float)k_svg_dec) + (float)(s[*i] - '0');
298 ++(*i);
299 }
300 if ((*i < len) && (s[*i] == '.')) {
301 ++(*i);
302 float frac = 1.0F;
303 while ((*i < len) && (s[*i] >= '0') && (s[*i] <= '9')) {
304 frac /= (float)k_svg_dec;
305 v += frac * (float)(s[*i] - '0');
306 ++(*i);
307 }
308 }
309 return v * sgn;
310}

370static svg_utf_t priv_utf_compose(svg_utf_t a, svg_utf_t b)
371{
372 svg_utf_t r = {.a = (a.a * b.a) + (a.c * b.b),
373 .b = (a.b * b.a) + (a.d * b.b),
374 .c = (a.a * b.c) + (a.c * b.d),
375 .d = (a.b * b.c) + (a.d * b.d),
376 .e = (a.a * b.e) + (a.c * b.f) + a.e,
377 .f = (a.b * b.e) + (a.d * b.f) + a.f};

506static int32_t priv_xform_read(const uint8_t* v, size_t vlen, size_t* j, float* args)
507{
508 int32_t na = 0;
509 /* Bounded: <= k_svg_argc_cube args; each ra8_svgp_numf advances *j. */
510 while (na < (int32_t)k_svg_argc_cube) {
511 while ((*j < vlen) && (ra8_svgp_ws((char)v[*j]) || (v[*j] == ','))) {
512 ++(*j);
513 }
514 if (!priv_is_num_start(v, vlen, *j)) {
515 break;
516 }
517 args[na] = ra8_svgp_numf(v, vlen, j);
518 ++na;
519 }
520 return na;
521}

549RA8_INTERNAL
550static svg_utf_t priv_xform_rotate(const float* args, int32_t na)
551{
552 const float co = cosf(priv_deg2rad(args[0]));
553 const float si = sinf(priv_deg2rad(args[0]));
554 const float cx = (na >= 3) ? args[1] : 0.0F;
555 const float cy = (na >= 3) ? args[2] : 0.0F;
556 return (svg_utf_t){.a = co,
557 .b = si,
558 .c = -si,
559 .d = co,
560 .e = (cx * (1.0F - co)) + (cy * si),
561 .f = (cy * (1.0F - co)) - (cx * si)};
562}

591RA8_INTERNAL
592static svg_utf_t priv_xform_build(svg_xf_kind_t kind, const float* args, int32_t na)
593{
594 switch (kind) {
595 case k_svg_xf_translate:
596 return (svg_utf_t){.a = 1.0F,
597 .b = 0.0F,
598 .c = 0.0F,
599 .d = 1.0F,
600 .e = args[0],
601 .f = (na >= 2) ? args[1] : 0.0F};
602 case k_svg_xf_scale:
603 return (svg_utf_t){.a = args[0],
604 .b = 0.0F,
605 .c = 0.0F,
606 .d = (na >= 2) ? args[1] : args[0],
607 .e = 0.0F,
608 .f = 0.0F};
609 case k_svg_xf_rotate:
610 return priv_xform_rotate(args, na);
611 case k_svg_xf_skewx:
612 return (svg_utf_t){.a = 1.0F,
613 .b = 0.0F,
614 .c = tanf(priv_deg2rad(args[0])),
615 .d = 1.0F,
616 .e = 0.0F,
617 .f = 0.0F};
618 case k_svg_xf_skewy:
619 return (svg_utf_t){.a = 1.0F,
620 .b = tanf(priv_deg2rad(args[0])),
621 .c = 0.0F,
622 .d = 1.0F,
623 .e = 0.0F,
624 .f = 0.0F};
625 case k_svg_xf_matrix:
626 return (svg_utf_t){.a = args[k_svg_xf_arg_a],
627 .b = args[k_svg_xf_arg_b],
628 .c = args[k_svg_xf_arg_c],
629 .d = args[k_svg_xf_arg_d],
630 .e = args[k_svg_xf_arg_e],
631 .f = args[k_svg_xf_arg_f]};
632 case k_svg_xf_none:
633 default:
634 return priv_utf_identity();
635 }
636}

723RA8_INTERNAL
724static svg_utf_t priv_parse_xform(const uint8_t* v, size_t vlen)
725{
726 svg_utf_t acc = priv_utf_identity();
727 size_t i = 0U;
728 /* Bounded: each pass consumes one `name(...)` group or breaks; <= vlen steps. */
729 while (i < vlen) {
730 while ((i < vlen) && (ra8_svgp_ws((char)v[i]) || (v[i] == ','))) {
731 ++i;
732 }
733 const svg_xf_kind_t kind = priv_xform_kind(v, vlen, i);
734 size_t op = i;
735 while ((op < vlen) && (v[op] != '(')) {
736 ++op;
737 }
738 if (op >= vlen) {
739 break;
740 }
741 size_t j = op + 1U;
742 if (kind != k_svg_xf_none) {
743 float args[k_svg_path_args] = {};
744 const int32_t na = priv_xform_read(v, vlen, &j, args);
745 acc = priv_utf_compose(acc, priv_xform_build(kind, args, na));
746 }
747 while ((j < vlen) && (v[j] != ')')) {
748 ++j;
749 }
750 i = (j < vlen) ? (j + 1U) : vlen;
751 }
752 return acc;
753}
