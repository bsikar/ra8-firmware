4"""Offline Ethos-U55 model build step for issue #227.
5
6Two responsibilities, deliberately separated so the golden pipeline runs in CI
7WITHOUT the heavy, optional Vela toolchain:
8
91. `compile` runs the pinned ethos-u-vela on a real quantized .tflite to
10 produce a _vela.tflite, in which the "ethos-u" custom op wraps a Vela
11 command stream. This one needs Vela, and SKIPS with a clear notice and exit
12 0 when it is absent -- so a CI job that never installs Vela still passes.
13 Read that exit 0 as "not run", never as "the model compiled".
14
152. `emit` / `check` turn a committed model DESCRIPTOR
16 (tools/vela/models/*.json) into the lean, linkable ".npub" container defined
17 by libs/ra8_hal/inc/ra8_npu_blob.h, baked as a C header the firmware links.
18 `emit` writes the header; `check` regenerates it in memory and diffs it
19 against the committed golden, failing on drift. Neither needs Vela, so this
20 pair IS the regenerate-and-diff gate the issue asks for.
21
22What the committed descriptor actually is matters for reading a green run: it
23is a SIM model -- the tiny, documented "SE55" command-stream convention in
24libs/ra8_hal/inc/ra8_npu_fake_cmd.h -- and NOT a real Vela program. It is the
25only Ethos-U55 command stream this repo can produce deterministically without
26a Vela install and without inventing NPU opcodes. The ra8_emulator NPU model and
27the ra8_npu driver both decode that same convention, so the whole
28submit -> run -> read-output path is exercised end to end, but a passing gate
29says nothing about real Vela output. Distilling a REAL _vela.tflite into a
30.npub is wired below as a documented follow-up (see `compile`), pending a
31validated Vela output and an RA8P1 board.
32
33Usage:
34 python3 tools/vela/vela_gen.py emit tools/vela/models/npu_addk_fake.json \
35 -o tools/vela/generated/ra8_npu_model_addk_fake.h
36 python3 tools/vela/vela_gen.py check tools/vela/models/npu_addk_fake.json \
37 tools/vela/generated/ra8_npu_model_addk_fake.h
38 python3 tools/vela/vela_gen.py compile model_int8.tflite -o build/vela
39"""

54# Container constants -- MUST match libs/ra8_hal/inc/ra8_npu_blob.h exactly.
55# ---------------------------------------------------------------------------
56BLOB_MAGIC = 0x3155504E # "NPU1" little-endian
57BLOB_VERSION = 1
58HEADER_WORDS = 8
59WORD_BYTES = 4
60HEADER_BYTES = HEADER_WORDS * WORD_BYTES
61REGION_WORDS = 4
62REGION_DESC_BYTES = REGION_WORDS * WORD_BYTES
63RFLAG_BAKED = 0x1
64FNV_OFFSET = 0x811C9DC5
65FNV_PRIME = 0x01000193
66U32_MASK = 0xFFFFFFFF
67BYTE_MASK = 0xFF

69# Fake command-stream convention -- MUST match libs/ra8_hal/inc/ra8_npu_fake_cmd.h.
70FAKE_MAGIC = 0x5E550000
71FAKE_OP = {"copy": 0x0001, "addk": 0x0002}

74 "weights": 0,
75 "scratch": 1,
76 "input": 2,
77 "output": 3,

91def _fnv1a(data: bytes) -> int:
92 """FNV-1a 32-bit digest over `data` (matches ra8_npu_blob.h / the loader)."""
93 digest = FNV_OFFSET
94 for byte in data:
95 digest = ((digest ^ byte) * FNV_PRIME) & U32_MASK
96 return digest

99def _seed_bytes(size: int, mul: int, add: int) -> bytes:
100 """Deterministic byte pattern: out[i] = (i*mul + add) & 0xFF."""
101 return bytes(((i * mul + add) & BYTE_MASK) for i in range(size))

104def _region_payload(region: dict) -> bytes:
105 """Build the baked bytes for one region per its `fill` rule."""
106 size = int(region["size"])
107 fill = region.get("fill", "zero")
108 if fill == "seed":
109 return _seed_bytes(size, int(region.get("mul", 1)), int(region.get("add", 0)))
110 if fill == "const":
111 return bytes([int(region.get("value", 0)) & BYTE_MASK]) * size
112 return bytes(size)

115def _build_command_stream(desc: dict) -> bytes:
116 """Pack the SE55 stand-in command stream (5 little-endian 32-bit words)."""
117 op_name = desc["op"]
118 if op_name not in FAKE_OP:
119 msg = f"unknown op '{op_name}' (expected one of {sorted(FAKE_OP)})"
120 raise ValueError(msg)
121 words = [
122 FAKE_MAGIC | FAKE_OP[op_name],
123 int(desc["src_region"]),
124 int(desc["dst_region"]),
125 int(desc["count"]),
126 int(desc.get("addk", 0)),
127 ]
128 return b"".join(struct.pack("<I", w & U32_MASK) for w in words)

131def build_blob(desc: dict) -> bytes:
132 """Assemble a full .npub blob (header + region table + cmd + baked data)."""
133 regions = desc["regions"]
134 if len(regions) > MAX_REGIONS:
135 msg = f"too many regions: {len(regions)} > {MAX_REGIONS} (k_ra8_npu_region_count)"
136 raise ValueError(msg)
137 cmd = _build_command_stream(desc)
138 cmd_offset = HEADER_BYTES + (len(regions) * REGION_DESC_BYTES)
139
140 # Baked region data blocks follow the command stream, in descriptor order.
141 table = bytearray()
142 baked = bytearray()
143 data_cursor = cmd_offset + len(cmd)
144 for region in regions:
145 role = ROLE[region["role"]]
146 size = int(region["size"])
147 if region.get("mode", "runtime") == "baked":
148 payload = _region_payload(region)
149 if len(payload) != size:
150 msg = "baked payload length must equal region size"
151 raise ValueError(msg)
152 table += struct.pack("<IIII", role, RFLAG_BAKED, size, data_cursor)
153 baked += payload
154 data_cursor += size
155 else:
156 table += struct.pack("<IIII", role, 0, size, 0)
157
158 total = data_cursor
159 payload = bytes(table) + cmd + bytes(baked)
160 checksum = _fnv1a(payload)
161 header = struct.pack(
162 "<IIIIIIII",
163 BLOB_MAGIC,
164 BLOB_VERSION,
165 total,
166 len(regions),
167 cmd_offset,
168 len(cmd),
169 int(desc.get("accel", ACCEL_ETHOS_U55_256)),
170 checksum,
171 )
172 blob = header + payload
173 if len(blob) != total:
174 msg = f"blob length {len(blob)} != declared total {total}"
175 raise ValueError(msg)
176 return blob

179def emit_header(desc: dict, blob: bytes) -> str:
180 """Render `blob` as a self-contained, linkable C byte-array header."""
181 symbol = desc["symbol"]
182 name = desc["name"]
183 # 16 bytes/row keeps each data line at 97 columns (2 indent + 16*"0xXX, "
184 # minus the trailing space), inside the .clang-format ColumnLimit of 100, so
185 # the emitted golden is already clang-format-22 clean and `make vela-check`
186 # (generator output vs committed header) and the format gate never disagree.
187 per_row = 16
188 rows = []
189 for start in range(0, len(blob), per_row):
190 chunk = blob[start : start + per_row]
191 rows.append(" " + "".join(f"0x{b:02X}, " for b in chunk).rstrip())
192 body = "\n".join(rows)
193 lines = [
194 "/*",
195 " * Copyright (c) 2026 Brighton Sikarskie",
196 " * SPDX-License-Identifier: MIT",
197 " *",
198 f" * GENERATED by tools/vela/vela_gen.py from tools/vela/models/{name}.json.",
199 " * DO NOT EDIT BY HAND. Regenerate with `make vela-regen`; the committed copy",
200 " * is diffed against a fresh render by `make vela-check` (needs no Vela).",
201 " *",
202 " * A '.npub' Ethos-U55 model container (see libs/ra8_hal/inc/ra8_npu_blob.h):",
203 " * an Ethos-U55 command stream plus its tensor region layout, baked as a byte",
204 " * array the firmware links and the on-target loader (ra8_npu_loader.h) maps",
205 " * into an ra8_npu_job_t. The command stream here is the documented SE55 stand-in",
206 " * convention (ra8_npu_fake_cmd.h), NOT a real Vela program -- see vela_gen.py.",
207 " */",
208 "",
209 "#pragma once",
210 "",
211 "#include <stdint.h>",
212 "",
213 f"/** @brief Raw bytes of the '{name}' .npub model container. */",
214 f"static const uint8_t s_{symbol}_data[] = {{",
215 body,
216 "};",
217 "",
218 f"/** @brief Base pointer of the '{name}' .npub blob. */",
219 f"static inline const uint8_t* {symbol}_blob(void)",
220 "{",
221 f" return s_{symbol}_data;",
222 "}",
223 "",
224 f"/** @brief Byte length of the '{name}' .npub blob. */",
225 f"static inline uint32_t {symbol}_bytes(void)",
226 "{",
227 f" return (uint32_t)sizeof(s_{symbol}_data);",
228 "}",
229 "",
230 ]
231 return "\n".join(lines)

234def _load_desc(path: Path) -> dict:
235 """Read and minimally validate a model descriptor JSON."""
236 desc = json.loads(path.read_text(encoding="ascii"))
237 for key in ("name", "symbol", "op", "regions", "src_region", "dst_region", "count"):
238 if key not in desc:
239 msg = f"{path}: descriptor missing required key '{key}'"
240 raise ValueError(msg)
241 return desc

244def cmd_emit(args: argparse.Namespace) -> int:
245 """emit: descriptor -> committed C header."""
246 desc = _load_desc(Path(args.descriptor))
247 header = emit_header(desc, build_blob(desc))
248 out = Path(args.output)
249 out.parent.mkdir(parents=True, exist_ok=True)
250 out.write_text(header, encoding="ascii")
251 print(f"vela_gen: wrote {out} ({len(header)} bytes)")
252 return 0

255def cmd_check(args: argparse.Namespace) -> int:
256 """check: regenerate the header and diff it against the committed copy."""
257 desc = _load_desc(Path(args.descriptor))
258 fresh = emit_header(desc, build_blob(desc))
259 golden = Path(args.header)
260 if not golden.is_file():
261 print(f"vela_gen: MISSING golden {golden} (run `make vela-regen`)", file=sys.stderr)
262 return 1
263 current = golden.read_text(encoding="ascii")
264 if current != fresh:
265 print(
266 f"vela_gen: DRIFT -- {golden} is stale vs {args.descriptor}.\n"
267 " Run `make vela-regen` and commit the result.",
268 file=sys.stderr,
269 )
270 return 1
271 print(f"vela_gen: {golden} is up to date")
272 return 0

275def cmd_compile(args: argparse.Namespace) -> int:
276 """compile: run the pinned Vela on a real .tflite (optional; skip if absent)."""
277 vela = shutil.which("vela")
278 if vela is None:
279 print(
280 "vela_gen: ethos-u-vela is not installed -- skipping compile "
281 "(this is fine; the .npub golden pipeline needs no Vela).\n"
282 " Install it with: pip install -r tools/vela/requirements.txt",
283 )
284 return 0
285 tflite = Path(args.tflite)
286 if not tflite.is_file():
287 print(f"vela_gen: no such .tflite: {tflite}", file=sys.stderr)
288 return 1
289 out_dir = Path(args.output)
290 out_dir.mkdir(parents=True, exist_ok=True)
291 argv = [
292 vela,
293 "--accelerator-config",
294 VELA_ACCEL_CONFIG,
295 "--output-dir",
296 str(out_dir),
297 str(tflite),
298 ]
299 print(f"vela_gen: running {' '.join(argv)}")
300 proc = subprocess.run(argv, check=False) # noqa: S603 -- resolved path, fixed argv
301 if proc.returncode != 0:
302 return proc.returncode
303 produced = out_dir / f"{tflite.stem}_vela.tflite"
304 print(f"vela_gen: Vela wrote {produced}")
305 # TODO(#227 follow-up): distilling this real _vela.tflite (the ethos-u custom
306 # op command stream + region layout) into a .npub requires a validated Vela
307 # output and an RA8P1 board to confirm the extracted stream runs on silicon.
308 # Until then the committed golden uses the SE55 stand-in descriptor via `emit`.
309 print(
310 "vela_gen: NOTE -- extracting the command stream from _vela.tflite into a "
311 ".npub is a documented follow-up; use `emit` on a model descriptor for now.",
312 )
313 return 0

316def main(argv: list[str]) -> int:
317 """Parse the subcommand line and dispatch to the matching `cmd_*` handler.
318
319 A subcommand is required, so a bare invocation is an argparse usage error
320 rather than a default action.
321
322 Note the asymmetry in what a 0 means across the three: `emit` and `check`
323 return 0 only when they really ran, but `compile` also returns 0 when Vela
324 is not installed and nothing was compiled. See the module docstring.
325
326 Args:
327 argv: Argument list WITHOUT the program name (callers pass
328 `sys.argv[1:]`).
329
330 Returns:
331 The handler's status, for `sys.exit`.
332 """
333 parser = argparse.ArgumentParser(description="Offline Ethos-U55 model build step (#227).")
334 sub = parser.add_subparsers(dest="command", required=True)
335
336 p_emit = sub.add_parser("emit", help="descriptor -> committed C header")
337 p_emit.add_argument("descriptor", help="model descriptor JSON")
338 p_emit.add_argument("-o", "--output", required=True, help="output C header path")
339 p_emit.set_defaults(func=cmd_emit)
340
341 p_check = sub.add_parser("check", help="regenerate and diff against the golden header")
342 p_check.add_argument("descriptor", help="model descriptor JSON")
343 p_check.add_argument("header", help="committed golden C header to diff")
344 p_check.set_defaults(func=cmd_check)
345
346 p_compile = sub.add_parser("compile", help="run the pinned Vela on a .tflite (optional)")
347 p_compile.add_argument("tflite", help="quantized INT8/INT16 .tflite model")
348 p_compile.add_argument("-o", "--output", default="build/vela", help="Vela output dir")
349 p_compile.set_defaults(func=cmd_compile)
350
351 args = parser.parse_args(argv)
352 return int(args.func(args))
