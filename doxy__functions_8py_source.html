3"""The function gate: every function carries the tags CLAUDE.md requires.
4
5Recognising a function is most of the work, and it is done with a regex over
6comment-stripped source rather than a parse. That is a deliberate trade --
7see the ``doxy_audit`` docstring on why this stayed regex-driven while
8``check_doc_attachment.py`` took the libclang dependency -- and it is why
9:func:`_is_declaration` exists: the regex over-matches call expressions, and
10each rejection below corresponds to a real construct in this tree that was
11once audited as if it were a function.
12
13The waivers in :func:`_waiver_row` are equally load-bearing. Two of them exist
14because this gate and ``check_doc_attachment.py`` would otherwise contradict
15each other -- demanding a block here that the other rejects as a duplicate --
16and no source file could satisfy both at once.
17"""

28# Required tags for every function (per CLAUDE.md)
29REQUIRED_SCALAR = ["@brief", "@details", "@return", "@retval", "@note", "@since"]
30# @pre and @post require a minimum of 2 each (NASA Rule 5)
31REQUIRED_MIN2 = ["@pre", "@post"]

33# NASA Power of 10 Rule 5 mandates at least 2 preconditions and 2 postconditions.

36FUNC_RE = re.compile(
37 # return-type tokens (allow pointers, qualifiers, attributes)
38 r"^[ \t]*"
39 r"(?P<ret>(?:(?:static|inline|extern|const|volatile|register|signed|unsigned|"
40 r"struct|union|enum|__attribute__\s*\‍(\‍([^)]*\‍)\‍)|[A-Za-z_][A-Za-z_0-9]*)\s+|\*\s*)+)"
41 r"(?P<name>[A-Za-z_][A-Za-z_0-9]*)\s*"
42 r"\‍((?P<args>[^;{]*)\‍)\s*"
43 r"(?:__attribute__\s*\‍(\‍([^)]*\‍)\‍)\s*)?"
44 r"(?P<term>[{;])",
45 re.MULTILINE,
46)

108def _split_params(args_text: str) -> list[str]:
109 """Split a parameter list on top-level commas only.
110
111 Depth-aware because a function-pointer parameter or a nested initialiser
112 carries commas of its own -- `void f(int (*cb)(int, int), size_t n)` is
113 two parameters, not three.
114 """
115 s = args_text.strip()
116 if s in {"", "void"}:
117 return []
118 s = re.sub(r"__attribute__\s*\‍(\‍([^)]*\‍)\‍)", "", s)
119 params: list[str] = []
120 depth = 0
121 cur: list[str] = []
122 for ch in s:
123 if ch in {"(", "[", "{"}:
124 depth += 1
125 cur.append(ch)
126 elif ch in {")", "]", "}"}:
127 depth -= 1
128 cur.append(ch)
129 elif ch == "," and depth == 0:
130 params.append("".join(cur).strip())
131 cur = []
132 else:
133 cur.append(ch)
134 if cur:
135 params.append("".join(cur).strip())
136 return params

139def _param_name(param: str, position: int) -> str | None:
140 """Return the documented name of one parameter, or None when it has none."""
141 if not param or param in {"void", "..."}:
142 return None
143 stripped = re.sub(r"\‍[[^\‍]]*\‍]", "", param).strip()
144 # Function pointer: the name sits inside the (*name) group.
145 fp = re.search(r"\‍(\s*\*\s*([A-Za-z_][A-Za-z_0-9]*)\s*\‍)", stripped)
146 if fp:
147 return fp.group(1)
148 toks = re.findall(r"[A-Za-z_][A-Za-z_0-9]*", stripped)
149 if not toks:
150 return None
151 if len(toks) == 1 and toks[0] in _BARE_TYPE_KEYWORDS:
152 return f"arg{position}"
153 return toks[-1]

156def parse_args(args_text: str) -> list[str]:
157 """Return the list of parameter names. ``(void)`` -> ``[]``."""
158 names: list[str] = []
159 for param in _split_params(args_text):
160 name = _param_name(param, len(names))
161 if name is not None:
162 names.append(name)
163 return names

166def is_returning_void(ret: str) -> bool:
167 """Whether a return-type string denotes plain ``void``.
168
169 Answers the question behind the "documents a return value it cannot have"
170 check, so it has to be tolerant of how the type was written: whitespace is
171 collapsed and ``static`` / ``inline`` / ``extern`` / ``__attribute__((...))``
172 are stripped before comparing.
173
174 A pointer return is never void, which is checked BEFORE the name match --
175 otherwise ``void *`` would read as void and every allocator in the tree
176 would be reported for documenting its return.
177
178 The trailing-word test also accepts a qualified spelling such as
179 ``const void``.
180 """
181 r = ret.strip()
182 # collapse whitespace
183 r = re.sub(r"\s+", " ", r)
184 # strip qualifiers
185 r = re.sub(r"\b(static|inline|extern|__attribute__\‍(\‍([^)]*\‍)\‍))\b", "", r).strip()
186 # pointer return is not void
187 if "*" in r:
188 return False
189 # exact "void"
190 return r == "void" or r.endswith(" void")

193def definition_carries_block(raw: str, src_no_comments: str, name: str) -> bool:
194 """True when a *definition* of ``name`` in this same file carries a block.
195
196 Used to tell a file-local forward prototype apart from a genuinely
197 undocumented function. See the call site for why that distinction has to
198 exist.
199 """
200 raw_lines = raw.splitlines(keepends=True)
201 for dm in FUNC_RE.finditer(src_no_comments):
202 if dm.group("name") != name or dm.group("term") != "{":
203 continue
204 line_no = src_no_comments.count("\n", 0, dm.start()) + 1
205 if line_no - 1 >= len(raw_lines):
206 continue
207 offset = sum(len(ln) for ln in raw_lines[: line_no - 1])
208 _, has_block = find_preceding_doxy(raw, offset)
209 if has_block:
210 return True
211 return False

214def _is_declaration(m: re.Match) -> bool:
215 """Reject the call expressions ``FUNC_RE`` also matches.
216
217 Each rejection below is a real shape from this tree, not a hypothetical:
218 the ``*foo(off) = bar;`` inline-accessor register write, a `return f(x);`
219 statement, and a bare dereference of an accessor's return value all match
220 a regex looking for `<tokens> name(args) {` or `;`.
221 """
222 if m.group("name") in NON_FUNC_NAMES:
223 return False
224 ret, args = m.group("ret"), m.group("args")
225 if "return" in ret.split() or re.search(r"\btypedef\b", ret):
226 return False
227 # An assignment target on the LHS: the args group spilled past the real
228 # closing paren and now holds '=' or an extra ')'.
229 if "=" in args or ")" in args:
230 return False
231 # A return type that is *only* a dereference is a call expression.
232 ret_stripped = ret.strip()
233 if ret_stripped in ("*", "&") or re.fullmatch(r"[*&\s]+", ret_stripped):
234 return False
235 # Matched text starting `*name(` / `&name(` -- deref of an accessor.
236 return not m.group(0).lstrip().startswith(("*", "&"))

244 ``stripped`` has them blanked (that is where a definition is found).

252def _is_waived(views: _SourceViews, m: re.Match, block: str, *, has_block: bool) -> bool:
253 """True when this declaration legitimately needs no block of its own.
254
255 Three waivers, each closing a contradiction rather than lowering the bar:
256
257 * A non-static definition in a ``.c``. CLAUDE.md ("Definition-site
258 comments") puts the authoritative block on the *declaration* in the
259 header, which this tool audits when it scans that ``.h``. A restating
260 block at the definition only rots. A ``static`` function has no header
261 to carry the contract, so it still needs a full block.
262 * A file-local forward prototype whose definition below carries the block.
263 The prototype is an ordering device, not a second contract, and
264 demanding a block on both is precisely the duplication
265 ``check_doc_attachment.py`` rejects as DOC004/DOC006 -- without this the
266 two gates contradict each other and no file can satisfy both
267 (``ra8_rsip.c``'s ``internal_sw_sha256`` is the live example). A
268 prototype whose definition is ALSO bare is still reported, at the
269 definition.
270 * A block that defers to the header, or substitutes one with ``@copydoc``.
271 """
272 ret = m.group("ret")
273 if str(views.path).endswith(".c") and not re.search(r"\bstatic\b", ret):
274 return True
275 if (
276 m.group("term") == ";"
277 and not has_block
278 and re.search(r"\bstatic\b", ret)
279 and definition_carries_block(views.raw, views.stripped, m.group("name"))
280 ):
281 return True
282 if not has_block:
283 return False
284 is_header_stub = (
285 block.startswith("/*")
286 and not block.startswith("/**")
287 and (
288 any(marker in block for marker in _HEADER_STUB_MARKERS)
289 or "see implementation for details" in block.lower()
290 )
291 )
292 return is_header_stub or bool(_COPY_RE.search(block))

295def _missing_without_block(args: list[str], ret: str) -> list[str]:
296 """Every required tag, for a function carrying no block at all."""
297 missing = ["@brief", "@details", *(f"@param[{a}]" for a in args)]
298 if not is_returning_void(ret):
299 missing += ["@return", "@retval"]
300 return [*missing, "@pre", "@post", "@note", "@since"]

303def _missing_from_block(block: str, args: list[str], ret: str) -> list[str]:
304 """Every required tag the block does not carry."""
305 missing = []
306 if "@brief" not in block:
307 missing.append("@brief")
308 if "@details" not in block:
309 missing.append("@details")
310 for a in args:
311 # match @param[...] name OR @param name (any direction)
312 pat = re.compile(r"@param(?:\s*\‍[[^\‍]]*\‍])?\s+" + re.escape(a) + r"\b")
313 if not pat.search(block):
314 missing.append(f"@param[{a}]")
315 if not is_returning_void(ret):
316 if "@return" not in block and "@returns" not in block:
317 missing.append("@return")
318 if "@retval" not in block:
319 missing.append("@retval")
320 n_pre = len(re.findall(r"@pre\b", block))
321 n_post = len(re.findall(r"@post\b", block))
322 if n_pre < NASA_RULE5_MIN_PRE_POST:
323 missing.append(f"@pre(<2:{n_pre})")
324 if n_post < NASA_RULE5_MIN_PRE_POST:
325 missing.append(f"@post(<2:{n_post})")
326 if "@note" not in block:
327 missing.append("@note")
328 if "@since" not in block:
329 missing.append("@since")
330 return missing

333def _severity(missing: list[str]) -> str:
334 """Rank a gap: a missing @brief or @param is worse than a missing @since."""
335 if any(t == "@brief" or t.startswith("@param[") for t in missing):
336 return "high"
337 if any(t in ("@return", "@retval") or t.startswith(("@pre", "@post")) for t in missing):
338 return "medium"
339 return "low"

342def audit_file(path: Path) -> list[tuple[str, int, str, str, str]]:
343 """Return one row per function in ``path``: (file, line, name, missing, severity)."""
344 try:
345 raw = path.read_text(encoding="utf-8", errors="replace")
346 except OSError:
347 return []
348 src_no_comments = strip_comments(raw)
349 views = _SourceViews(path, raw, src_no_comments)
350 # strip_comments preserves newlines, so line numbers agree between the
351 # stripped and raw views and one offset conversion serves both.
352 raw_lines = raw.splitlines(keepends=True)
353 rel = str(path.relative_to(repo_root()))
354
355 rows = []
356 for m in FUNC_RE.finditer(src_no_comments):
357 if not _is_declaration(m):
358 continue
359 line_no = src_no_comments.count("\n", 0, m.start()) + 1
360 if line_no - 1 >= len(raw_lines):
361 continue
362 # Read the block from the ORIGINAL source, where comments still exist.
363 offset_in_raw = sum(len(ln) for ln in raw_lines[: line_no - 1])
364 block, has_block = find_preceding_doxy(raw, offset_in_raw)
365
366 name, ret = m.group("name"), m.group("ret")
367 if _is_waived(views, m, block, has_block=has_block):
368 rows.append((rel, line_no, name, [], "ok"))
369 continue
370
371 args = parse_args(m.group("args"))
372 missing = (
373 _missing_from_block(block, args, ret)
374 if has_block
375 else _missing_without_block(args, ret)
376 )
377 if not missing:
378 rows.append((rel, line_no, name, [], "ok"))
379 continue
380 rows.append((rel, line_no, name, missing, _severity(missing)))
381
382 return rows
