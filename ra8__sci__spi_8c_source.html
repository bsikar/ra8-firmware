95static void
96internal_resolve_rate(uint32_t baud_hz, uint32_t pclk_hz, uint32_t* out_cks, uint32_t* out_brr)
97{
98 for (uint32_t cks = 0U; cks <= (uint32_t)k_ra8_sci_spi_cks_max; cks++) {
99 const uint32_t divisor = (uint32_t)k_ra8_sci_spi_div_base << (2U * cks);
100 const uint32_t denom = divisor * baud_hz;
101 const uint32_t q = pclk_hz / denom;
102 if (q == 0U) {
103 /* Requested rate exceeds this CKS ceiling -> fastest (BRR = 0). */
104 *out_cks = cks;
105 *out_brr = 0U;
106 return;
107 }
108 if ((q - 1U) <= (uint32_t)k_ra8_sci_spi_brr_max) {
109 *out_cks = cks;
110 *out_brr = q - 1U;
111 return;
112 }
113 }
114 /* Unreachably slow: clamp to the slowest divider. */
115 *out_cks = (uint32_t)k_ra8_sci_spi_cks_max;
116 *out_brr = (uint32_t)k_ra8_sci_spi_brr_max;
117}

136static void internal_apply_rate(volatile r_sci_regs_t* reg, uint32_t baud_hz, uint32_t pclk_hz)
137{
138 uint32_t cks = 0U;
139 uint32_t brr = 0U;
140 internal_resolve_rate(baud_hz, pclk_hz, &cks, &brr);
141 /* HUM Ch 38.2.7 "CCR2 : Common Control Register 2" p 2189 */
142 uint32_t ccr2 =
143 (brr << (uint32_t)k_ra8_sci_ccr2_shift_brr) & (uint32_t)k_ra8_sci_ccr2_mask_brr_field;
144 ccr2 |= (cks << (uint32_t)k_ra8_sci_ccr2_shift_cks) & (uint32_t)k_ra8_sci_ccr2_mask_cks_field;
145 ccr2 |= ((uint32_t)k_ra8_sci_mddr_default << (uint32_t)k_ra8_sci_ccr2_shift_mddr);
146 reg->CCR2 = ccr2;
147}

166static uint32_t internal_ccr3(const ra8_sci_spi_cfg_t* cfg)
167{
168 uint32_t ccr3 = ((uint32_t)k_ra8_sci_ccr3_mod_simple_spi << (uint32_t)k_ra8_sci_ccr3_shift_mod);
169 ccr3 |= ((uint32_t)k_ra8_sci_ccr3_chr_8bit << (uint32_t)k_ra8_sci_ccr3_shift_chr);
170 /* ra8_spi_mode_t is 0..3 where bit 1 = CPOL and bit 0 = CPHA. */
171 if (((uint32_t)cfg->mode & (uint32_t)k_ra8_spi_mode_2) != 0U) {
172 ccr3 |= (1U << (uint32_t)k_ra8_sci_ccr3_bit_cpol);
173 }
174 if (((uint32_t)cfg->mode & (uint32_t)k_ra8_spi_mode_1) != 0U) {
175 ccr3 |= (1U << (uint32_t)k_ra8_sci_ccr3_bit_cpha);
176 }
177 if (cfg->lsb_first) {
178 ccr3 |= (1U << (uint32_t)k_ra8_sci_ccr3_bit_lsbf);
179 }
180 return ccr3;
181}

183ra8_err_t ra8_sci_spi_init(uint8_t channel, const ra8_sci_spi_cfg_t* cfg)
184{
185 RA8_CHECK_NULL_PTR(cfg, s_tag, "spi_init: cfg");
186 if (channel >= (uint8_t)k_ra8_sci_spi_channel_count) {
187 return k_ra8_err_invalid_arg;
188 }
189 if (cfg->baud_hz == 0U) {
190 return k_ra8_err_invalid_arg;
191 }
192 volatile r_sci_regs_t* reg = ra8_sci(channel);
193 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
194 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
195 }
196 const ra8_err_t mst_err = ra8_mstp_enable(s_mstp_table[channel]);
197 RA8_RETURN_ON_ERROR(mst_err, s_tag, "spi_init: mstp"); /* GCOVR_EXCL_BR_LINE */
198
199 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0" p 2182 */
200 reg->CCR0 = 0U;
201 reg->CCR1 = 0U;
202 reg->CCR3 = internal_ccr3(cfg);
203 internal_apply_rate(reg, cfg->baud_hz, cfg->pclk_hz);
204 reg->CCR4 = 0U;
205 /* HUM Ch 38.2.24 "CFCLR : Common Flag Clear Register" p 2238 */
206 reg->CFCLR = (uint32_t)k_ra8_sci_cfclr_default;
207 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0" p 2182 */
208 reg->CCR0 = (1U << (uint32_t)k_ra8_sci_ccr0_bit_te) | (1U << (uint32_t)k_ra8_sci_ccr0_bit_re);
209 return k_ra8_ok;
210}

226ra8_err_t ra8_sci_spi_set_clock(uint8_t channel, uint32_t baud_hz, uint32_t pclk_hz)

246ra8_err_t ra8_sci_spi_xfer8(uint8_t channel, uint8_t tx, uint8_t* rx)
247{
248 volatile r_sci_regs_t* reg = ra8_sci(channel);
249 RA8_CHECK_NULL_PTR(reg, s_tag, "spi_xfer8: channel out of range");
250
251 /* HUM Ch 38.2.17 "CSR : Common Status Register" p 2225 */
252 const uint32_t tdre = (1U << (uint32_t)k_ra8_sci_csr_bit_tdre);
253 ra8_err_t err = ra8_hw_wait_flag_set32(&reg->CSR, tdre, k_ra8_hw_budget_long);
254 if (err != k_ra8_ok) {
255 return err;
256 }
257 /* HUM Ch 38.2.3 "TDR : Transmit Data Register" p 2181 */
258 reg->TDR = (uint32_t)tx;
259
260 /* HUM Ch 38.2.17 "CSR : Common Status Register" p 2225 */
261 const uint32_t rdrf = (1U << (uint32_t)k_ra8_sci_csr_bit_rdrf);
262 err = ra8_hw_wait_flag_set32(&reg->CSR, rdrf, k_ra8_hw_budget_long);
263 if (err != k_ra8_ok) {
264 return err;
265 }
266 /* HUM Ch 38.2.2 "RDR : Receive Data Register" p 2180 */
267 const uint8_t received = (uint8_t)(reg->RDR & (uint32_t)k_ra8_sci_rdr_mask_data8);
268 /* HUM Ch 38.2.24 "CFCLR : Common Flag Clear Register" p 2238 */
269 reg->CFCLR = (1U << (uint32_t)k_ra8_sci_cfclr_bit_rdrfc);
270
271 if (rx != nullptr) {
272 *rx = received;
273 }
274 return k_ra8_ok;
275}

277ra8_err_t ra8_sci_spi_xfer(uint8_t channel, const uint8_t* tx, uint8_t* rx, uint32_t len)
278{
279 if (channel >= (uint8_t)k_ra8_sci_spi_channel_count) {
280 return k_ra8_err_invalid_arg;
281 }
282 for (uint32_t i = 0U; i < len; i++) {
283 const uint8_t out = (tx != nullptr) ? tx[i] : (uint8_t)k_ra8_sci_spi_idle_byte;
284 uint8_t in = 0U;
285 const ra8_err_t err = ra8_sci_spi_xfer8(channel, out, &in);
286 if (err != k_ra8_ok) {
287 return err;
288 }
289 if (rx != nullptr) {
290 rx[i] = in;
291 }
292 }
293 return k_ra8_ok;
