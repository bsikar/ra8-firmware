322static void internal_usbhs_arm_phy_reset(volatile uint16_t* physet_reg, uint16_t clksel_value)
323{
324 /* HUM Ch 37.2.17 PHYSET, p 2080: assert PHY power-down and PLL
325 * reset, mask out CLKSEL[1:0], then OR in the candidate field. */
326 uint16_t v = *physet_reg;
327 v = (uint16_t)(v | (uint16_t)k_ra8_physet_dirpd | (uint16_t)k_ra8_physet_pllreset);
328 v = (uint16_t)(v & (uint16_t)~(uint16_t)k_ra8_physet_clksel);
329 v = (uint16_t)(v | (uint16_t)(clksel_value & (uint16_t)k_ra8_physet_clksel));
330 *physet_reg = v;

365RA8_INTERNAL
366static ra8_err_t internal_usbhs_try_clksel(volatile uint16_t* physet,
367 volatile uint16_t* lpsts,
368 uint16_t clksel_value)
369{
370 /* Drop SUSPENDM so the PHY clock is gated; subsequent PHYSET writes
371 * are then sequenced through a single SUSPENDM 0->1 transition. */
372 *lpsts = (uint16_t)(*lpsts & (uint16_t)~(uint16_t)k_ra8_lpsts_suspendm);
373 internal_usb_delay_1us();
374
375 internal_usbhs_arm_phy_reset(physet, clksel_value);
376 internal_usb_delay_1us();
377
378 /* Release DIRPD (PHY analog up), wait 1 ms, release PLLRESET. */
379 *physet = (uint16_t)(*physet & (uint16_t)~(uint16_t)k_ra8_physet_dirpd);
380 ra8_delay_ms(1U);
381 *physet = (uint16_t)(*physet & (uint16_t)~(uint16_t)k_ra8_physet_pllreset);
382
383 /* HUM Ch 37.2.43 LPSTS, p 2111: SUSPENDM=1 starts the PHY clock,
384 * which both commits any previously-latched PHYSET writes and
385 * begins the PLL lock countdown. */
386 *lpsts = (uint16_t)(*lpsts | (uint16_t)k_ra8_lpsts_suspendm);
387
388 return internal_usbhs_wait_pll_lock_short();
389}

428ra8_err_t internal_usbhs_phy_bringup(volatile r_usb_regs_t* reg)
429{
430 volatile uint16_t* const physet = ra8_usbhs_physet();
431 volatile uint16_t* const lpsts = ra8_usbhs_lpsts();
432
433 /* Step 1: SYSCFG.HSE = 1 (mirrors FSP `hw_usb_set_hse`, called from
434 * r_usb_basic.c BEFORE hw_usb_pmodule_init). HUM Ch 37 "SYSCFG"
435 * register, p 2060. */
436 reg->SYSCFG = (uint16_t)(reg->SYSCFG | (uint16_t)(1U << k_ra8_syscfg_bit_hse));
437 s_phy_step_probe = (uint8_t)k_ra8_usbhs_phy_step_hse_set;
438
439 /* Step 2: CLKSEL = 24 MHz (PHYSET[5:4] = 11b, value 0x30).
440 *
441 * Per HUM Ch 37.3.3 "Supplying the Clock", Table 37.17 (p 2120) the
442 * USB-PHY internal PLL takes its reference clock DIRECTLY FROM THE
443 * EXTAL PIN -- not from USB60CLK. Allowed EXTAL frequencies are 12,
444 * 20, 24 or 48 MHz; PHYSET.CLKSEL[1:0] selects which one.
445 *
446 * EK-RA8D2 fits a 24 MHz crystal (see ra8_cgc.c::k_ra8_pll2_local_t
447 * comments), so CLKSEL must be 11b (k_ra8_physet_clksel_24). USB60CLK
448 * (60 MHz, PLL2P/4) is still required -- it is the LINK domain clock
449 * for the USBHS module itself -- but it does NOT feed the PHY UTMI
450 * PLL. Earlier revisions of this code mis-modeled CLKSEL as
451 * "USB60CLK / 5 = 12 MHz" and bisected over {12,48,24,20}; the host
452 * never enumerated because the PHY PLL was locking at the wrong
453 * frequency, producing malformed HS chirp K signaling. */
454 s_phy_step_probe = (uint8_t)k_ra8_usbhs_phy_step_clksel_12;
455 ra8_err_t lock_err = internal_usbhs_try_clksel(physet, lpsts, (uint16_t)k_ra8_physet_clksel_24);
456 s_clksel_attempt_pllsta[0] = s_usbhs_pllsta_probe;
457 s_usbhs_init_probe = (uint32_t)*physet;
458
459 if (lock_err != k_ra8_ok) {
460 s_clksel_winner = (uint8_t)k_ra8_usbhs_clksel_no_winner;
461 ra8_log_error(s_tag, "usbhs: PHY PLL lock timeout (CLKSEL=24)");
462 return lock_err;
463 }
464 s_clksel_winner = (uint8_t)k_ra8_physet_clksel_24;
465 s_phy_step_probe = (uint8_t)k_ra8_usbhs_phy_step_pll_locked;
466
467 /* HUM Figure 37.2 p 2121: USBE/DRPD are programmed AFTER PLLLOCK
468 * is observed. Internally splits into a separate helper to keep
469 * each function under the NASA Rule 4 budget. */
470 internal_usbhs_enable_syscfg(reg);
471 s_phy_step_probe = (uint8_t)k_ra8_usbhs_phy_step_usbe_set;
472
473 /* HUM Ch 37.2.2 "BUSWAIT : CPU Bus Wait Register", p 2062 */
474 reg->BUSWAIT = (uint16_t)k_ra8_buswait_default;
475 *physet = (uint16_t)(*physet | (uint16_t)k_ra8_physet_repsel_16);
476 return k_ra8_ok;
477}

495void internal_usb_init_common(volatile r_usb_regs_t* reg)
496{
497 /* HUM Ch 36.2.7 "CFIFOSEL : CFIFO Port Select Register", p 1976 */
498 /* HUM Ch 37.2.8 "CFIFOSEL : CFIFO Port Select Register", p 2071 */
499 reg->CFIFOSEL = k_ra8_fifosel_mbw_16;
500 reg->D0FIFOSEL = k_ra8_fifosel_mbw_16;
501 reg->D1FIFOSEL = k_ra8_fifosel_mbw_16;
502
503 /* HUM Ch 36.2.20 "DCPMAXP : DCP Max Packet Size Register", p 1990 */
504 /* HUM Ch 37.2.30 "DCPMAXP : DCP Max Packet Size Register", p 2092 */
505 reg->DCPCFG = 0U;
506 reg->DCPMAXP = k_ra8_usb_dcp_max_packet;
507 reg->DCPCTR = 0U;
508
509 /* HUM Ch 37.2.18 INTSTS0 Note 3 (p 2082): "To clear the CTRT, DVST,
510 * SOFR, RESM, or VBINT flags, write 0 only to the flags to be cleared.
511 * Write 1 to the other flags. Do not write 0 to the status flags
512 * indicating 0." A wholesale ``INTSTS0 = 0`` write therefore clears
513 * any stale CTRT/DVST/SOFR/RESM/VBINT edges that the PHY bring-up
514 * sequence latched (mirrors FSP IP1 ``hw_usb_pmodule_init``: see
515 * r_usb_preg_access.c::hw_usb_pmodule_init which writes
516 * ``INTSTS0 = 0`` immediately before INTENB0 on the HS branch).
517 * Without this flush a stale RESM/VBINT bit gets ORed into every
518 * polled-dispatch snapshot and clutters the JLink-readable
519 * ``s_intsts0_observed_or`` probe (observed value 0xD0D0 with bits
520 * 14/15 set even though the host issued no real resume). */
521 reg->INTSTS0 = 0U;
522
523 /* HUM Ch 36.2.10 "INTENB0 : Interrupt Enable Register 0", p 1980.
524 * Mirrors FSP r_usb_basic.c::usb_pmodule_init mask: BEMP, BRDY, NRDY,
525 * CTRT, DVST, SOFR, RSME, VBSE. Enabling SOFR + RSME is required for
526 * HS so the IP latches resume / SOF events in INTSTS0; without RSME
527 * a host wake from suspend leaves the device asleep. */
528 /* SOFR and RSME are intentionally NOT enabled here:
529 * - SOFR fires every 125us on HS (= 8 kHz);
530 * - RSME stays asserted on USBHS via the PHY's USBR signal
531 * while the host holds the bus in resume signalling;
532 * Both starve PendSV and the demo worker thread never gets
533 * scheduled. NRDY is still useful (drives the bridge's per-pipe
534 * NAK re-arm) and does not cause a storm on its own. RSME / SOFR
535 * status is still readable from INTSTS0 for poll-style drivers. */
536 reg->INTENB0 = (uint16_t)((1U << k_ra8_int0_bit_bemp) | (1U << k_ra8_int0_bit_brdy) |
537 (1U << k_ra8_int0_bit_nrdy) | (1U << k_ra8_int0_bit_ctrt) |
538 (1U << k_ra8_int0_bit_dvst) | (1U << k_ra8_int0_bit_vbse));
539 reg->INTENB1 = 0U;
540 reg->BRDYENB = 0U;
541 reg->NRDYENB = 0U;
542 reg->BEMPENB = 0U;
543}
