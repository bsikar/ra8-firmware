400static void internal_program_dll(const ra8_pdg_config_t* cfg, ra8_pdg_frange_t frange_use)
401{
402 volatile r_pdg_regs_t* reg = ra8_pdg();
403
404 /* Step 1 of Figure 23.2 (p 1160): DLLEN=0, DLYRST=1, DLYBSn=0,
405 * FRANGE=user-supplied band. Performed in one 16-bit write so the
406 * intermediate state is never visible. */
407 /* HUM Ch 23.2.1 "GTDLYCR: PWM Output Delay Control Register" p 1154 */
408 const uint16_t frange_bits =
409 (uint16_t)((uint16_t)frange_use << (uint16_t)k_ra8_pdg_gtdlycr_shift_frange);
410 reg->GTDLYCR = (uint16_t)(k_ra8_pdg_gtdlycr_mask_dlyrst | frange_bits);
411 /* HUM Ch 23.2.2 "GTDLYCR2: PWM Output Delay Control Register 2" p 1155 */
412 reg->GTDLYCR2 = 0U;
413
414 /* Step 2: enable the DLL. */
415 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 */
416 reg->GTDLYCR =
417 (uint16_t)(k_ra8_pdg_gtdlycr_mask_dlyrst | k_ra8_pdg_gtdlycr_mask_dllen | frange_bits);
418
419 /* Step 3: wait >= 20 us for the DLL to lock. */
420 /* HUM Ch 23.3.1 "Adjustments to the Timing of Rising and Falling Edges" p 1159 */
421 internal_busy_wait_us(k_ra8_pdg_dll_lock_us);
422
423 /* Step 4: release the PDG from reset. */
424 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 */
425 reg->GTDLYCR = (uint16_t)(k_ra8_pdg_gtdlycr_mask_dllen | frange_bits);
426
427 /* Step 5: wait >= 5 GTCLK cycles. */
428 internal_wait_5_gtclk();
429
430 /* Step 6: turn off bypass for the channels actually in use. */
431 /* HUM Ch 23.2.2 "GTDLYCR2" p 1155 */
432 const uint16_t bypass_off =
433 (uint16_t)((uint16_t)cfg->channel_mask << (uint16_t)k_ra8_pdg_gtdlycr2_shift_dlybs);
434 reg->GTDLYCR2 = bypass_off;
435}

459ra8_err_t ra8_pdg_deinit(void)
460{
461 volatile r_pdg_regs_t* reg = ra8_pdg();
462 /* Park the block: DLLEN=0, DLYRST=1, FRANGE=0. */
463 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 */
464 reg->GTDLYCR = k_ra8_pdg_gtdlycr_mask_dlyrst;
465 /* HUM Ch 23.2.2 "GTDLYCR2" p 1155 */
466 reg->GTDLYCR2 = 0U;
467
468 /* Wipe every per-edge delay cell back to the post-reset value of 0. */
469 /* HUM Ch 23.2.3 "GTDLYRnA" p 1156 */
470 for (uint8_t ch = 0U; ch < k_ra8_pdg_channel_count; ++ch) {
471 reg->GTDLYR[ch].A = 0U;
472 reg->GTDLYR[ch].B = 0U;
473 /* HUM Ch 23.2.4 "GTDLYFnA" p 1156 */
474 reg->GTDLYF[ch].A = 0U;
475 reg->GTDLYF[ch].B = 0U;
476 }
477
478 /* PDG bit MSTPD6, see HUM Ch 23.4.1 p 1162. */
479 /* HUM Ch 11.2.9 "MSTPCRD: Module Stop Control Register D" p 449 */
480 (void)ra8_mstp_disable(k_ra8_mstp_pdg);
481 return k_ra8_ok;
482}

520ra8_err_t ra8_pdg_set_delay_batch(const ra8_pdg_delay_entry_t* entries, uint8_t count)
521{
522 RA8_CHECK_NULL_PTR(entries, s_tag, "entries must not be nullptr");
523 if (!internal_pdg_is_initialized()) {
524 return k_ra8_err_not_initialized;
525 }
526 if ((count == 0U) || (count > k_ra8_pdg_slot_count)) {
527 return k_ra8_err_invalid_arg;
528 }
529 /* Pre-validate every entry so a partial batch is impossible. */
530 for (uint8_t i = 0U; i < count; ++i) {
531 const ra8_err_t v =
532 internal_validate_slot(entries[i].channel, entries[i].pin, entries[i].edge, entries[i].code);
533 if (v != k_ra8_ok) {
534 return v;
535 }
536 }
537 /* HUM Ch 23.3.2 "Timing for Transfer of GTDLYRnA, GTDLYRnB, GTDLYFnA, GTDLYFnB
538 * Register Settings" p 1161 -- temp-register transfer happens at next
539 * overflow / underflow / trough; writing all entries back-to-back keeps the
540 * batch coherent. */
541 for (uint8_t i = 0U; i < count; ++i) {
542 volatile uint16_t* cell =
543 internal_dly_cell(entries[i].channel, entries[i].pin, entries[i].edge);
544 *cell = (uint16_t)((uint16_t)entries[i].code & k_ra8_pdg_dly_mask);
545 }
546 return k_ra8_ok;
547}

549ra8_err_t ra8_pdg_delay_ns_to_code(uint32_t delay_ns,
550 uint32_t gptclk_hz,
551 ra8_pdg_frange_t frange,
552 uint8_t* out_code)
553{
554 RA8_CHECK_NULL_PTR(out_code, s_tag, "out_code must not be nullptr");
555 if (gptclk_hz == 0U) {
556 return k_ra8_err_invalid_arg;
557 }
558 if (!internal_frange_ok(frange)) {
559 return k_ra8_err_invalid_arg;
560 }
561 /* HUM Ch 23.2.3 "GTDLYRnA: GTIOCnA Rising Output Delay Register" p 1156
562 * code = round(delay_ns / (period_ns / divider))
563 * = round(delay_ns * divider / period_ns)
564 * = round(delay_ns * divider * gptclk_hz / 1e9) */
565 const uint32_t divider = (frange == k_ra8_pdg_frange_80_160_mhz)
566 ? (uint32_t)k_ra8_pdg_dly_div_low
567 : (uint32_t)k_ra8_pdg_dly_div_high;
568 /* Use 64-bit arithmetic to avoid overflow. */
569 const uint64_t numer = (uint64_t)delay_ns * (uint64_t)divider * (uint64_t)gptclk_hz;
570 const uint64_t half = (uint64_t)k_ra8_pdg_ns_per_sec / 2U;
571 uint64_t code = (numer + half) / (uint64_t)k_ra8_pdg_ns_per_sec;
572 if (code > (uint64_t)k_ra8_pdg_dly_max) {
573 code = (uint64_t)k_ra8_pdg_dly_max;
574 }
575 *out_code = (uint8_t)code;
576 return k_ra8_ok;
577}

755/* =============================================================================
756 * Runtime FRANGE switch + auto-tune calibration
757 * =============================================================================
758 */
759
760ra8_err_t ra8_pdg_pick_frange(uint32_t gptclk_hz, ra8_pdg_frange_t* out)
761{
762 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
763 if (gptclk_hz == 0U) {
764 return k_ra8_err_invalid_arg;
765 }
766 if (gptclk_hz < k_ra8_pdg_freq_low_min_hz) {
767 return k_ra8_err_out_of_range;
768 }
769 if (gptclk_hz > k_ra8_pdg_freq_high_max_hz) {
770 return k_ra8_err_out_of_range;
771 }
772 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 -- prefer the low band in the
773 * 155..160 MHz overlap because it gives 1/128 (finer) resolution. */
774 if (gptclk_hz <= k_ra8_pdg_freq_overlap_top) {
775 *out = k_ra8_pdg_frange_80_160_mhz;
776 } else {
777 *out = k_ra8_pdg_frange_155_300_mhz;
778 }
779 return k_ra8_ok;
780}
781
782ra8_err_t ra8_pdg_set_frange(ra8_pdg_frange_t new_frange)
783{
784 if (!internal_pdg_is_initialized()) {
785 return k_ra8_err_not_initialized;
786 }
787 if (!internal_frange_ok(new_frange)) {
788 return k_ra8_err_invalid_arg;
789 }
790
791 volatile r_pdg_regs_t* reg = ra8_pdg();
792 /* Preserve the per-channel state across the FRANGE switch. */
793 /* HUM Ch 23.2.2 "GTDLYCR2: PWM Output Delay Control Register 2" p 1155 */
794 const uint16_t saved_cr2 = reg->GTDLYCR2;
795
796 /* "Set the FRANGE[1:0] bits only when the DLLEN bit is 0." So drop
797 * DLLEN and assert DLYRST first. */
798 /* HUM Ch 23.2.1 "GTDLYCR: PWM Output Delay Control Register" p 1154 */
799 reg->GTDLYCR = k_ra8_pdg_gtdlycr_mask_dlyrst;
800
801 /* Park GTDLYCR2 (Step 1 of Figure 23.2 also sets DLYBSn = 0). */
802 /* HUM Ch 23.2.2 "GTDLYCR2" p 1155 */
803 reg->GTDLYCR2 = 0U;
804
805 const uint16_t frange_bits =
806 (uint16_t)((uint16_t)new_frange << (uint16_t)k_ra8_pdg_gtdlycr_shift_frange);
807 /* Write FRANGE while DLLEN is still 0. */
808 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 */
809 reg->GTDLYCR = (uint16_t)(k_ra8_pdg_gtdlycr_mask_dlyrst | frange_bits);
810
811 /* Re-run Steps 2..5 of HUM Figure 23.2 (p 1160). Enable DLL: */
812 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 */
813 reg->GTDLYCR =
814 (uint16_t)(k_ra8_pdg_gtdlycr_mask_dlyrst | k_ra8_pdg_gtdlycr_mask_dllen | frange_bits);
815 /* 20 us DLL lock wait per HUM Figure 23.2. */
816 /* HUM Ch 23.3.1 "Adjustments to the Timing of Rising and Falling Edges" p 1159 */
817 internal_busy_wait_us(k_ra8_pdg_dll_lock_us);
818 /* Release reset. */
819 /* HUM Ch 23.2.1 "GTDLYCR" p 1154 */
820 reg->GTDLYCR = (uint16_t)(k_ra8_pdg_gtdlycr_mask_dllen | frange_bits);
821 internal_wait_5_gtclk();
822
823 /* Restore the per-channel state. */
824 /* HUM Ch 23.2.2 "GTDLYCR2" p 1155 */
825 reg->GTDLYCR2 = saved_cr2;
826 return k_ra8_ok;
827}

833
834ra8_err_t ra8_pdg_bind_gpt_channel(uint8_t channel)
835{
836 if (!internal_pdg_is_initialized()) {
837 return k_ra8_err_not_initialized;
838 }
839 if ((uint16_t)channel >= (uint16_t)k_ra8_pdg_channel_count) {
840 return k_ra8_err_invalid_arg;
841 }
842 /* HUM Ch 23.2.3 "GTDLYRnA" p 1156 -- "GTDLYRnA can be written when
843 * register write protection is disabled (GPT32n.GTWP.WP = 0)." */
844 volatile r_gpt_channel_regs_t* gpt = ra8_gpt(channel);
845 if (gpt == nullptr) {
846 return k_ra8_err_hw_unmapped;

848 gpt->GTWP = k_ra8_pdg_gtwp_unlock;

850 * before changing GTDLYCR". The dummy read enforces ordering. */
851 (void)gpt->GTWP;
852 return k_ra8_ok;
853}

862 volatile r_gpt_channel_regs_t* gpt = ra8_gpt(channel);
863 if (gpt == nullptr) {
864 return k_ra8_err_hw_unmapped;
865 }
866 gpt->GTWP = k_ra8_pdg_gtwp_lock;
867 (void)gpt->GTWP;
868 return k_ra8_ok;

871/* =============================================================================
872 * Constraint validation (HUM Ch 23.4.2 / 23.4.3)
873 * =============================================================================
874 */
875
876ra8_err_t ra8_pdg_check_constraints(ra8_pdg_wave_mode_t mode,
877 ra8_pdg_count_dir_t dir,
878 uint32_t compare_match,
879 uint32_t gtpr)
880{
881 if ((mode != k_ra8_pdg_wave_saw) && (mode != k_ra8_pdg_wave_triangle)) {
882 return k_ra8_err_invalid_arg;
883 }
884 if ((dir != k_ra8_pdg_dir_up) && (dir != k_ra8_pdg_dir_down)) {
885 return k_ra8_err_invalid_arg;
886 }
887 /* HUM Ch 23.4.2 "Notes on Delay Settings for PWM Delay Generation Circuit" p 1162 */
888 if (mode == k_ra8_pdg_wave_saw) {
889 if (dir == k_ra8_pdg_dir_up) {
890 /* "Saw-wave Up: GTPR - 2 or above" -- compare_match must be < gtpr - 2. */
891 if (gtpr < k_ra8_pdg_constraint_high_pad) {
892 /* gtpr is too small to even compute the boundary safely. */
893 return k_ra8_err_invalid_state;
894 }
895 const uint32_t boundary = gtpr - k_ra8_pdg_constraint_high_pad;
896 if (compare_match >= boundary) {
897 return k_ra8_err_invalid_state;
898 }
899 } else {
900 /* "Saw-wave Down: <= 2". */
901 if (compare_match <= k_ra8_pdg_constraint_low_thresh) {
902 return k_ra8_err_invalid_state;
903 }
904 }
905 } else {
906 /* "Triangle-wave Down: <= 2". Up direction has no constraint per Table 23.4. */
907 if (dir == k_ra8_pdg_dir_down) {
908 if (compare_match <= k_ra8_pdg_constraint_low_thresh) {
909 return k_ra8_err_invalid_state;
910 }
911 }
912 }
913 return k_ra8_ok;
914}

916ra8_err_t ra8_pdg_required_write_ns(uint32_t pclka_hz, uint32_t gptclk_hz, uint32_t* out_ns)
917{
918 RA8_CHECK_NULL_PTR(out_ns, s_tag, "out_ns must not be nullptr");
919 if ((pclka_hz == 0U) || (gptclk_hz == 0U)) {
920 return k_ra8_err_invalid_arg;
921 }
922 /* HUM Ch 23.4.3 "Register Write Interval" p 1163:
923 * Write_Interval = Period_of_PCLKA x 6 + Period_of_GPTCLK x 4 */
924 const uint32_t per_pclka = internal_freq_to_period_ns(pclka_hz);
925 const uint32_t per_gptclk = internal_freq_to_period_ns(gptclk_hz);
926 *out_ns = (per_pclka * k_ra8_pdg_pclka_mult) + (per_gptclk * k_ra8_pdg_gptclk_mult);
927 return k_ra8_ok;
