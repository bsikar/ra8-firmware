283#pragma GCC diagnostic push
284#pragma GCC diagnostic ignored "-Wmain"
285int32_t main(void)
286{
287 /* CGC bring-up FIRST. tx_initialize_low_level.S programs SysTick
288 * with a reload of (1 GHz / 1000 - 1) = 999999 cycles per tick
289 * (chip HUM Ch 11 CGC, plus our port/threadx/src/cortex_m85/
290 * tx_initialize_low_level.S RA8_BOOT_CLOCK_HZ value). If the chip
291 * is still on MOCO (~8.4 MHz) at scheduler-enter time, that
292 * 999999-cycle reload takes ~119 ms wallclock instead of 1 ms --
293 * a 119x slowdown, which makes tx_thread_sleep(500 ticks) sleep
294 * for ~60 s and the LED toggle look frozen in any reasonable
295 * HIL window. Bring up the PLL before tx_kernel_enter so the
296 * scheduler tick rate matches what tx_user.h declared. */
297 if (ra8_cgc_init() != k_ra8_ok) {
298 while (1) {
299 __asm__ volatile("wfi");
300 }
301 }
302
303 /* GPIO init runs with PRIMASK set; ThreadX will re-enable IRQs once
304 * its scheduler is up. */
305 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
306 while (1) {
307 __asm__ volatile("wfi");
308 }
309 }
310 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
311 while (1) {
312 __asm__ volatile("wfi");
313 }
314 }
315
316 /* Drop into ThreadX. Returns only on internal scheduler error.
317 * The host build has no kernel, so it just falls into the WFI loop. */
318#ifndef RA8_OFF_TARGET
319 tx_kernel_enter();
320#endif
321
322 /* Belt + suspenders. */
323 while (1) {
324 __asm__ volatile("wfi");
325 }
326}
