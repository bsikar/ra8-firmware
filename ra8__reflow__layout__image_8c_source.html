114static void priv_image_fit(int32_t iw,
115 int32_t ih,
116 int32_t col_w,
117 int32_t avail_h,
118 int32_t* out_w,
119 int32_t* out_h)
120{
121 int32_t bw = (iw < col_w) ? iw : col_w;
122 int32_t bh = (int32_t)(((int64_t)ih * (int64_t)bw) / (int64_t)iw);
123 if (bh > avail_h) {
124 bh = avail_h;
125 bw = (int32_t)(((int64_t)iw * (int64_t)bh) / (int64_t)ih);
126 }
127 *out_w = (bw < 1) ? 1 : bw;
128 *out_h = (bh < 1) ? 1 : bh;
129}

151static bool priv_image_resolve_size(ra8_reflow_t* engine,
152 const ra8_reflow_token_t* tok,
153 int32_t* out_w,
154 int32_t* out_h)
155{
156 const char* href = (const char*)&engine->text_pool[tok->text_off];
157 const uint8_t* bytes = nullptr;
158 size_t blen = 0U;
159 if (engine->img_loader(engine->img_loader_ctx, href, tok->text_len, &bytes, &blen) != k_ra8_ok) {
160 return false;
161 }
162 int32_t iw = 0;
163 int32_t ih = 0;
164 /* mcdc-deactivated: ra8_img_probe_size returns k_ra8_ok only via stbi_info (which rejects 0-pixel headers) or ra8_svg_size (which errors on w/h <= 0), so iw and ih are provably > 0 here; the (iw <= 0) / (ih <= 0) defensive guards are unreachable on any public path. */
165 if ((ra8_img_probe_size(bytes, blen, &iw, &ih) != k_ra8_ok) || (iw <= 0) || (ih <= 0)) {
166 return false;
167 }
168 const int32_t col_w = (int32_t)engine->viewport_w - (2 * (int32_t)k_ra8_reflow_margin_px);
169 const int32_t avail_h = (int32_t)engine->viewport_h - (2 * (int32_t)k_ra8_reflow_margin_px);
170 if ((col_w < 1) || (avail_h < 1)) {
171 return false;
172 }
173 priv_image_fit(iw, ih, col_w, avail_h, out_w, out_h);
174 return true;
175}

196static void priv_image_record(ra8_reflow_t* engine,
197 priv_cursor_t* cur,
198 const ra8_reflow_token_t* tok,

201{

204 box->y = cur->y;
205 box->w = bw;
206 box->h = bh;
207 box->src_off = tok->text_off;
208 box->src_len = tok->text_len;
209 box->page_index = engine->page_count;

211 engine->image_box_count++;
212
213 cur->y += bh + (int32_t)k_ra8_reflow_paragraph_gap_px;
214 cur->line_top = cur->y;
215 cur->x = (int32_t)k_ra8_reflow_margin_px + (int32_t)cur->indent_px;
216 cur->line_has_content = 0U;
217}

238static bool
239priv_place_image(ra8_reflow_t* engine, priv_cursor_t* cur, const ra8_reflow_token_t* tok)
240{
241 if (engine->image_box_count >= (uint32_t)k_ra8_reflow_max_images) {
242 return false;
243 }
244 int32_t bw = 0;
245 int32_t bh = 0;
246 if (!priv_image_resolve_size(engine, tok, &bw, &bh)) {
247 return false;
248 }
249 /* Block-level: drop below the current line, then page-break if needed. */
250 if (cur->line_has_content != 0U) {
251 if (!ra8_reflow_layout_newline(engine, cur, false)) {
252 return false;
253 }
254 }
255 const int32_t bottom_limit = (int32_t)engine->viewport_h - (int32_t)k_ra8_reflow_margin_px;
256 if (((cur->y + bh) > bottom_limit) && priv_page_has_content(engine, cur)) {
257 if (!ra8_reflow_layout_finish_page(engine, cur)) {
258 return false;
259 }
260 }
261 priv_image_record(engine, cur, tok, bw, bh);
262 /* mcdc-deactivated: priv_image_record incremented image_box_count immediately above, so priv_page_has_content is invariantly true here; its false arm is unreachable once an image box has been recorded on the current page. */
263 if (((cur->y + (int32_t)cur->line_height_px) > bottom_limit) &&
264 priv_page_has_content(engine, cur)) {
265 if (!ra8_reflow_layout_finish_page(engine, cur)) {
266 return false;
267 }
268 }
269 return true;
270}

272bool ra8_reflow_layout_apply_image(ra8_reflow_t* engine,
273 priv_cursor_t* cur,
274 const ra8_reflow_token_t* tok)
275{
276 if ((engine->img_loader != nullptr) && (engine->img_arena != nullptr) && (tok->text_len > 0U)) {
277 if (priv_place_image(engine, cur, tok)) {
278 return true;
279 }
280 /* Resolution / pool overflow: fall back to the placeholder advance. */
281 }
282 return priv_apply_image_placeholder(engine, cur);
