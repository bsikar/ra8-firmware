286/* =============================================================================
287 * Directory growth: the same allocation, one directory-specific rewrite (#677)
288 * =============================================================================
289 *
290 * A directory's on-disk allocation is structurally a file's: a run of clusters
291 * that carries `NoFatChain` while it is contiguous and becomes a real FAT chain
292 * the moment it is not. So directory growth reuses the machinery above --
293 * ::priv_exfat_pick_cluster to prefer the tail's successor, and
294 * ::priv_exfat_materialize_run for the conversion -- and adds only the two
295 * things a directory needs that a mid-write file does not: the new cluster is
296 * ZEROED before it is linked (a directory cluster reads as its own contents, so
297 * a half-built one must never be reachable), and the directory's OWN Stream
298 * entry in its parent is rewritten so `DataLength` and the `NoFatChain` flag
299 * keep describing the run after the mount that grew it is gone.
300 */

332RA8_INTERNAL
333static ra8_err_t priv_exfat_dir_survey(const ra8_fs_mount_t* m,
334 const exfat_dir_t* dir,
335 uint32_t* out_alloc,
336 uint32_t* out_tail,
337 uint8_t* out_nofat)
338{
339 if (dir->contig_end != 0U) {
340 *out_nofat = 1U;
341 *out_alloc = dir->contig_end - dir->cluster;
342 *out_tail = dir->contig_end - 1U;
343 return k_ra8_ok;
344 }
345 *out_nofat = 0U;
346 uint32_t clus = dir->cluster;
347 uint32_t held = 1U;
348 for (uint32_t guard = 0U; guard < m->count_of_clusters; guard++) {
349 uint32_t next = 0U;
350 const ra8_err_t e = priv_fat_get(m, clus, &next);
351 if (e != k_ra8_ok) {
352 return e;
353 }
354 if (priv_is_eoc(m, next) != 0U) {
355 break;
356 }
357 if (next < (uint32_t)k_cluster_first_data) {
358 return k_ra8_err_protocol_error;
359 }
360 clus = next;
361 held++;
362 }
363 *out_alloc = held;
364 *out_tail = clus;
365 return k_ra8_ok;
366}

460static ra8_err_t priv_exfat_dir_relen(const ra8_fs_mount_t* m,
461 const exfat_dir_t* dir,
462 uint64_t new_bytes,
463 uint8_t nofat)
464{
465 uint8_t set[k_exfat_max_set_bytes] = {};
466 exfat_cursor_t cur = {.cluster = dir->self_cluster,
467 .entry_in_cluster = dir->self_index,
468 .scanned = 0U,
469 .contig_end = 0U};
470 ra8_err_t e = priv_exfat_next_entry(m, &cur, &set[0]);
471 if (e != k_ra8_ok) {
472 return e;
473 }
474 const uint32_t count = 1U + (uint32_t)set[k_exfat_off_file_secnt];
475 if (count > (uint32_t)k_exfat_set_writable) {
476 return k_ra8_err_not_supported;
477 }
478 for (uint32_t k = 1U; k < count; k++) {
479 e = priv_exfat_next_entry(m, &cur, &set[(size_t)k * (size_t)k_exfat_entry_bytes]);
480 if (e != k_ra8_ok) {
481 return e;
482 }
483 }
484 uint8_t* strm = &set[k_exfat_entry_bytes];
485 strm[k_exfat_strm_off_flags] =
486 (nofat != 0U) ? (uint8_t)k_exfat_secflag_alloc : (uint8_t)k_exfat_secflag_poss;
487 priv_wr64(&strm[k_exfat_off_strm_valid], new_bytes);
488 priv_wr64(&strm[k_exfat_strm_off_dlen], new_bytes);
489 priv_wr16(&set[k_exfat_off_file_csum],
490 priv_exfat_set_checksum(set, count * (uint32_t)k_exfat_entry_bytes));
491 return priv_exfat_write_dir_set(m,
492 dir->self_cluster,
493 dir->self_index,
494 set,
495 count * (uint32_t)k_exfat_entry_bytes);
496}

563/* `priv_exfat_grow_dir()`: see header for the documented contract. */
564ra8_err_t priv_exfat_grow_dir(const ra8_fs_mount_t* m, exfat_dir_t* dir)
565{
566 uint32_t alloc = 0U;
567 uint32_t tail = 0U;
568 uint8_t nofat = 0U;
569 ra8_err_t e = priv_exfat_dir_survey(m, dir, &alloc, &tail, &nofat);
570 if (e != k_ra8_ok) {
571 return e;
572 }
573 e = priv_exfat_dir_append(m, dir, alloc, tail, &nofat);
574 if (e != k_ra8_ok) {
575 return e;
576 }
577 const uint32_t new_alloc = alloc + 1U;
578 const uint32_t cbytes = priv_cluster_bytes(m);
579 /* The volume root has no entry set to patch -- its extent is the boot-sector
580 * FAT chain, which ::priv_exfat_dir_link just extended. A subdirectory carries
581 * the location of its own set and must update it, or the next mount reads the
582 * old length and cannot see the cluster this call added (#677). */
583 if (dir->self_cluster >= (uint32_t)k_cluster_first_data) {
584 e = priv_exfat_dir_relen(m, dir, (uint64_t)new_alloc * cbytes, nofat);
585 if (e != k_ra8_ok) {
586 return e;
587 }
588 }
589 dir->contig_end = (nofat != 0U) ? (dir->cluster + new_alloc) : 0U;
590 return k_ra8_ok;
591}

623static ra8_err_t priv_exfat_cluster_at(ra8_fs_file_t* file, uint32_t idx, uint32_t* out)
624{
625 if (file->no_fat_chain != 0U) {
626 *out = file->first_cluster + idx;
627 return k_ra8_ok;
628 }
629 /* Nested rather than joined, because only the inner guard has both answers
630 * available: by the time a chained file is walked its waypoint is always a
631 * real cluster, so a joined decision would carry a condition no input can
632 * make false -- a permanent MC/DC hole. The inner guard varies with every
633 * backward seek. */
634 uint32_t from = file->first_cluster;
635 uint32_t hops = idx;
636 if (file->walk_cache_cluster >= (uint32_t)k_cluster_first_data) {
637 if (idx >= file->walk_cache_idx) {
638 from = file->walk_cache_cluster;
639 hops = idx - file->walk_cache_idx;
640 }
641 }
642 uint32_t cur = from;
643 for (uint32_t i = 0U; i < hops; i++) {
644 uint32_t next = 0U;
645 const ra8_err_t e = priv_fat_get(file->mount, cur, &next);
646 if (e != k_ra8_ok) {
647 return e;
648 }
649 if (priv_is_eoc(file->mount, next) != 0U) {
650 return k_ra8_err_invalid_state;
651 }
652 cur = next;
653 }
654 file->walk_cache_idx = idx;
655 file->walk_cache_cluster = cur;
656 *out = cur;
657 return k_ra8_ok;
658}

689static ra8_err_t priv_exfat_slice_at(ra8_fs_file_t* file,
690 uint64_t pos,
691 uint64_t* out_lba,
692 uint32_t* out_off,
693 uint32_t* out_room)
694{
695 const ra8_fs_mount_t* m = file->mount;
696 const uint32_t cbytes = priv_cluster_bytes(m);
697 const uint32_t idx = (uint32_t)(pos / cbytes);
698 ra8_err_t e = priv_exfat_ensure_clusters(file, idx + 1U);
699 if (e != k_ra8_ok) {
700 return e;
701 }
702 uint32_t cur = 0U;

704 if (e != k_ra8_ok) {
705 return e;
706 }
707 file->cur_cluster = cur;
708 const uint32_t in_cluster = (uint32_t)(pos % cbytes);

710 *out_off = in_cluster % priv_bps(m);
711 *out_room = priv_bps(m) - *out_off;
712 return k_ra8_ok;

743static ra8_err_t priv_exfat_close_gap(ra8_fs_file_t* file)
744{
745 while (file->valid_bytes < file->offset) {
746 uint64_t lba = 0U;
747 uint32_t off = 0U;
748 uint32_t room = 0U;

767/* `priv_exfat_write_stream()`: see header for the documented contract. */
768ra8_err_t priv_exfat_write_stream(ra8_fs_file_t* file, const uint8_t* buf, uint32_t len)
769{
770 ra8_err_t e = priv_exfat_close_gap(file);
771 if (e != k_ra8_ok) {
772 return e;
773 }
774 uint32_t consumed = 0U;
775 while (consumed < len) {
776 uint64_t lba = 0U;
777 uint32_t off = 0U;
778 uint32_t room = 0U;
779 e = priv_exfat_slice_at(file, file->offset, &lba, &off, &room);
780 if (e != k_ra8_ok) {
781 return e;
782 }
783 uint32_t put = len - consumed;
784 if (put > room) {
785 put = room;
786 }
787 e = priv_write_into_sector(file->mount, lba, off, &buf[consumed], put);
788 if (e != k_ra8_ok) {
789 return e;
790 }
791 consumed += put;
792 file->offset += put;
793 if (file->offset > file->valid_bytes) {
794 file->valid_bytes = file->offset;
795 }
796 if (file->offset > file->size_bytes) {
797 file->size_bytes = file->offset;
798 }
799 }
800 return k_ra8_ok;
801}

906/* `priv_exfat_flush_set()`: see header for the documented contract. */
907ra8_err_t priv_exfat_flush_set(ra8_fs_file_t* file)
908{
909 const ra8_fs_mount_t* m = file->mount;
910 uint8_t set[k_exfat_max_set_bytes] = {};
911 exfat_setpos_t at_file = {};
912 exfat_setpos_t at_stream = {};
913 ra8_err_t e = priv_exfat_gather_set(file, set, &at_file, &at_stream);
914 if (e != k_ra8_ok) {
915 return e;
916 }
917 /* Stamp and patch BEFORE the checksum: it covers every byte both touch, so
918 * the other order writes a set a host `fsck` rejects (#601). */
919 priv_exfat_file_stamp_write(set);
920 /* A write also sets the archive attribute (#681): the convention a backup tool
921 * clears and the OS re-sets on modification. The low byte of the 16-bit
922 * FileAttributes field carries it; the other bits (read-only can never be set
923 * on a handle that reached a write) are preserved. */
924 set[k_exfat_off_file_attr] =
925 (uint8_t)(set[k_exfat_off_file_attr] | (uint8_t)k_exfat_attr_archive);
926 priv_exfat_patch_stream(file, &set[k_exfat_entry_bytes]);
927 priv_wr16(&set[k_exfat_off_file_csum],
928 priv_exfat_set_checksum(set, file->entry_set_count * (uint32_t)k_exfat_entry_bytes));
929 e =
930 priv_exfat_write_dir_set(m, at_file.cluster, at_file.index, set, (uint32_t)k_exfat_entry_bytes);
931 if (e != k_ra8_ok) {
932 return e;
933 }
934 return priv_exfat_write_dir_set(m,
935 at_stream.cluster,
936 at_stream.index,
937 &set[k_exfat_entry_bytes],
938 (uint32_t)k_exfat_entry_bytes);
