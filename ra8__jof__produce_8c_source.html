80RA8_PRIV void* ra8_jof_priv_bump_take(ra8_jof_bump_t* bump, size_t len)
81{
82 if ((bump == nullptr) || (len == 0U)) {
83 return nullptr;
84 }
85 const size_t mask = (size_t)k_ra8_jof_bump_align - 1U;
86 const size_t aligned = (bump->off + mask) & ~mask;
87 if ((aligned > bump->cap) || (len > (bump->cap - aligned))) {
88 return nullptr;
89 }
90 bump->off = aligned + len;
91 return &bump->base[aligned];
92}

112static void priv_wr_u16(uint8_t* buf, uint16_t v)
113{
114 buf[0] = (uint8_t)(v & (uint16_t)k_ra8_jof_byte_mask);

130RA8_INTERNAL
131static void priv_wr_u32(uint8_t* buf, uint32_t v)
132{
133 buf[0] = (uint8_t)(v & (uint32_t)k_ra8_jof_byte_mask);
134 buf[1] = (uint8_t)((v >> k_ra8_jof_le_sh8) & (uint32_t)k_ra8_jof_byte_mask);
135 buf[2] = (uint8_t)((v >> k_ra8_jof_le_sh16) & (uint32_t)k_ra8_jof_byte_mask);
136 buf[3] = (uint8_t)((v >> k_ra8_jof_le_sh24) & (uint32_t)k_ra8_jof_byte_mask);

209static ra8_err_t priv_emit_header(ra8_jof_prod_state_t* st)
210{
211 uint8_t hdr[k_ra8_jof_hdr_bytes] = {};
212 hdr[0] = 'J';
213 hdr[1] = 'O';
214 hdr[2] = 'F';
215 hdr[3] = '1';
216 priv_wr_u16(&hdr[k_ra8_jof_ofs_width], st->w);
217 priv_wr_u16(&hdr[k_ra8_jof_ofs_height], st->h);
218 priv_wr_u16(&hdr[k_ra8_jof_ofs_tile_w], st->cfg->tile_w);
219 priv_wr_u16(&hdr[k_ra8_jof_ofs_tile_h], st->cfg->tile_h);
220 hdr[k_ra8_jof_ofs_bpp] = st->bpp;
221 hdr[k_ra8_jof_ofs_codec] = st->cfg->codec;
222 priv_wr_u32(&hdr[k_ra8_jof_ofs_tile_count], st->tile_count);
223 return priv_sink(st, hdr, sizeof(hdr));

241RA8_INTERNAL
242static ra8_err_t priv_carve_pixel_path(ra8_jof_prod_state_t* st)
243{
244 const uint16_t tw = st->cfg->tile_w;
245 const uint16_t th = st->cfg->tile_h;
246 const uint64_t band_bytes = (uint64_t)st->w * (uint64_t)th * (uint64_t)st->bpp;
247 const uint32_t tw_eff = (tw < st->w) ? tw : st->w;
248 const uint32_t th_eff = (th < st->h) ? th : st->h;
249 const uint64_t stage_bytes = (uint64_t)tw_eff * (uint64_t)th_eff * (uint64_t)st->bpp;
250 // mcdc-deactivated: stage_bytes = tw_eff*th_eff*bpp with tw_eff <= w and th_eff <= tile_h, so stage_bytes <= w*tile_h*bpp = band_bytes always; the stage-overflow arm cannot flip independently of the band arm (which tests/test_ra8_jof_produce_guards.c drives true via a 32768-wide RGBA source and a 65535-tall tile).
251 if ((band_bytes > (uint64_t)UINT32_MAX) || (stage_bytes > (uint64_t)UINT32_MAX)) {
252 return k_ra8_err_invalid_size;
253 }
254 st->band = ra8_jof_priv_bump_take(st->bump, (size_t)band_bytes);
255 st->stage = ra8_jof_priv_bump_take(st->bump, (size_t)stage_bytes);
256 st->idx =
257 ra8_jof_priv_bump_take(st->bump, (size_t)st->tile_count * (size_t)k_ra8_jof_index_entry);
258 if ((st->band == nullptr) || (st->stage == nullptr) || (st->idx == nullptr)) {
259 return k_ra8_err_invalid_size;
260 }
261 if (st->cfg->codec == (uint8_t)k_ra8_jof_codec_deflate) {
262 st->cmp_cap = ra8_jof_stored_bound((uint32_t)stage_bytes);
263 st->cmp = ra8_jof_priv_bump_take(st->bump, (size_t)st->cmp_cap);
264 if (st->cmp == nullptr) {
265 return k_ra8_err_invalid_size;
266 }
267 }
268 return k_ra8_ok;
269}

293RA8_PRIV ra8_err_t ra8_jof_priv_on_geom(void* ctx,
294 uint16_t width,
295 uint16_t height,
296 uint8_t channels)
297{
298 ra8_jof_prod_state_t* st = (ra8_jof_prod_state_t*)ctx;
299 if (st->geom_done != 0U) {
300 return k_ra8_err_invalid_state;
301 }
302 if ((width == 0U) || (width > st->cap_w) || (height == 0U) || (height > st->cap_h)) {
303 return k_ra8_err_invalid_size;
304 }
305 const uint32_t cols = ((uint32_t)width + st->cfg->tile_w - 1U) / st->cfg->tile_w;
306 const uint32_t rows = ((uint32_t)height + st->cfg->tile_h - 1U) / st->cfg->tile_h;
307 if ((cols * rows) > (uint32_t)k_ra8_jof_max_tiles) {
308 return k_ra8_err_invalid_size;
309 }
310 st->w = width;
311 st->h = height;
312 st->bpp = channels;
313 st->tile_cols = (uint16_t)cols;
314 st->tile_rows = (uint16_t)rows;
315 st->tile_count = cols * rows;
316 ra8_err_t err = priv_carve_pixel_path(st);
317 if (err != k_ra8_ok) {
318 return err;
319 }
320 err = priv_emit_header(st);
321 if (err != k_ra8_ok) {
322 return err;
323 }
324 st->geom_done = 1U;
325 return k_ra8_ok;
326}

348RA8_INTERNAL
349static ra8_err_t priv_encode_tile(ra8_jof_prod_state_t* st, uint32_t payload)
350{
351 const uint8_t* out_bytes = st->stage;
352 uint32_t out_len = payload;
353 if (st->cfg->codec == (uint8_t)k_ra8_jof_codec_deflate) {
354 uint32_t clen = 0U;
355 if (ra8_io_compress(st->stage, payload, st->cmp, st->cmp_cap, st->dfl, st->dfl_len, &clen) !=
356 k_ra8_ok) {
357 return k_ra8_err_validation_failed;
358 }
359 out_bytes = st->cmp;
360 out_len = clen;
361 }
362 uint8_t* entry = &st->idx[(size_t)st->tiles_done * (size_t)k_ra8_jof_index_entry];
363 priv_wr_u32(&entry[k_ra8_jof_idx_ofs_offset], st->written);
364 priv_wr_u32(&entry[k_ra8_jof_idx_ofs_length], out_len);
365 const ra8_err_t err = priv_sink(st, out_bytes, (size_t)out_len);
366 if (err != k_ra8_ok) {
367 return err;
368 }
369 st->tiles_done++;
370 return k_ra8_ok;
371}

390RA8_INTERNAL
391static ra8_err_t priv_flush_band(ra8_jof_prod_state_t* st, uint32_t th)
392{
393 const uint32_t stride = (uint32_t)st->w * (uint32_t)st->bpp;
394 for (uint32_t tx = 0U; tx < (uint32_t)st->tile_cols; tx++) {
395 const uint32_t x0 = tx * (uint32_t)st->cfg->tile_w;
396 const uint32_t tw = (((uint32_t)st->w - x0) < (uint32_t)st->cfg->tile_w)
397 ? ((uint32_t)st->w - x0)
398 : (uint32_t)st->cfg->tile_w;
399 const uint32_t row_bytes = tw * (uint32_t)st->bpp;
400 for (uint32_t r = 0U; r < th; r++) {
401 (void)memcpy(&st->stage[(size_t)r * (size_t)row_bytes],
402 &st->band[((size_t)r * (size_t)stride) + ((size_t)x0 * (size_t)st->bpp)],
403 (size_t)row_bytes);
404 }
405 const ra8_err_t err = priv_encode_tile(st, row_bytes * th);
406 if (err != k_ra8_ok) {
407 return err;
408 }
409 }
410 return k_ra8_ok;
411}

413RA8_PRIV ra8_err_t ra8_jof_priv_on_rows(void* ctx,
414 const uint8_t* px,
415 uint16_t width,
416 uint16_t y0,
417 uint16_t nrows,
418 uint8_t channels)
419{
420 ra8_jof_prod_state_t* st = (ra8_jof_prod_state_t*)ctx;
421 RA8_CHECK_NULL_PTR(px, s_tag, "px must not be nullptr");
422 // mcdc-deactivated: row-sink contract guard; both in-tree decoders fire the geometry hook before any row (a failed hook aborts the decode) and pass their bound width/channels verbatim into every on_rows call, so no public-API source can flip geom_done/width/channels here independently.
423 if ((st->geom_done == 0U) || (width != st->w) || (channels != st->bpp)) {
424 return k_ra8_err_validation_failed;
425 }
426 // mcdc-deactivated: row-ordering contract guard; the PNG scanline assembler emits exactly one row per call at y0 == rows_done and the JPEG stripe walker emits edge-clamped nrows >= 1 at strictly increasing MCU-row origins, so zero/duplicated/overshooting deliveries are not constructible from a public-API source.
427 if ((nrows == 0U) || ((uint32_t)y0 != st->rows_seen) ||
428 (((uint32_t)y0 + (uint32_t)nrows) > (uint32_t)st->h)) {
429 return k_ra8_err_validation_failed;
430 }
431 const uint32_t stride = (uint32_t)st->w * (uint32_t)st->bpp;
432 const uint8_t* src = px;
433 uint32_t left = nrows;
434 while (left > 0U) {
435 const uint32_t room = (uint32_t)st->cfg->tile_h - st->band_fill;
436 const uint32_t take = (left < room) ? left : room;
437 (void)memcpy(&st->band[(size_t)st->band_fill * (size_t)stride],
438 src,
439 (size_t)take * (size_t)stride);
440 st->band_fill += take;
441 st->rows_seen += take;
442 src += (size_t)take * (size_t)stride;
443 left -= take;
444 if (st->band_fill == (uint32_t)st->cfg->tile_h) {
445 const ra8_err_t err = priv_flush_band(st, st->band_fill);
446 if (err != k_ra8_ok) {
447 return err;
448 }
449 st->band_fill = 0U;
450 }
451 }
452 return k_ra8_ok;
453}

523RA8_INTERNAL
524static ra8_err_t priv_finish(ra8_jof_prod_state_t* st, ra8_jof_info_t* out_info)
525{
526 if (st->tiles_done != st->tile_count) {
527 return k_ra8_err_validation_failed;
528 }
529 const uint32_t index_off = st->written;
530 ra8_err_t err = priv_sink(st, st->idx, (size_t)st->tile_count * (size_t)k_ra8_jof_index_entry);
531 if (err != k_ra8_ok) {
532 return err;
533 }
534 uint8_t ftr[k_ra8_jof_footer_bytes] = {};
535 priv_wr_u32(&ftr[k_ra8_jof_ftr_index_off], index_off);
536 priv_wr_u32(&ftr[k_ra8_jof_ftr_tile_count], st->tile_count);
537 priv_wr_u32(&ftr[k_ra8_jof_ftr_total_size], st->written + (uint32_t)k_ra8_jof_footer_bytes);
538 ftr[k_ra8_jof_ftr_magic] = 'J';
539 ftr[k_ra8_jof_ftr_magic + 1U] = 'O';
540 ftr[k_ra8_jof_ftr_magic + 2U] = 'F';
541 ftr[k_ra8_jof_ftr_magic + 3U] = 'E';
542 err = priv_sink(st, ftr, sizeof(ftr));
543 if (err != k_ra8_ok) {
544 return err;
545 }
546 out_info->width = st->w;
547 out_info->height = st->h;
548 out_info->tile_w = st->cfg->tile_w;
549 out_info->tile_h = st->cfg->tile_h;
550 out_info->tile_cols = st->tile_cols;
551 out_info->tile_rows = st->tile_rows;
552 out_info->bpp = st->bpp;
553 out_info->codec = st->cfg->codec;
554 out_info->tile_count = st->tile_count;
555 out_info->index_off = index_off;
556 out_info->total_size = st->written;
557 return k_ra8_ok;
558}

586uint32_t
587ra8_jof_work_bytes(uint16_t max_width, uint16_t max_height, uint16_t tile_w, uint16_t tile_h)
588{
589 if ((max_width == 0U) || ((uint32_t)max_width > (uint32_t)k_ra8_jof_max_dim) ||
590 (max_height == 0U) || ((uint32_t)max_height > (uint32_t)k_ra8_jof_max_dim)) {
591 return 0U;
592 }
593 if ((tile_w == 0U) || (tile_h == 0U)) {
594 return 0U;
595 }
596 const uint64_t cols = ((uint64_t)max_width + tile_w - 1U) / tile_w;
597 const uint64_t rows = ((uint64_t)max_height + tile_h - 1U) / tile_h;
598 if ((cols * rows) > (uint64_t)k_ra8_jof_max_tiles) {
599 return 0U;
600 }
601 const uint64_t bpp = (uint64_t)k_ra8_jof_bpp_max;
602 const uint64_t jpeg_set =
603 (uint64_t)k_ra8_jpeg_sw_stream_min_window +
604 ((uint64_t)max_width * (uint64_t)k_ra8_jpeg_sw_stream_mcu_rows_max * 3U);
605 const uint64_t png_set = (uint64_t)sizeof(tinfl_decompressor) + (uint64_t)k_ra8_jof_png_ring +
606 (uint64_t)k_ra8_jof_png_inbuf + (3U * ((uint64_t)max_width * bpp)) + 2U;
607 const uint64_t dec_set = (jpeg_set > png_set) ? jpeg_set : png_set;
608 const uint64_t band = (uint64_t)max_width * (uint64_t)tile_h * bpp;
609 const uint64_t tw_eff = ((uint64_t)tile_w < (uint64_t)max_width) ? tile_w : max_width;
610 const uint64_t th_eff = ((uint64_t)tile_h < (uint64_t)max_height) ? tile_h : max_height;
611 const uint64_t stage = tw_eff * th_eff * bpp;
612 const uint64_t cmp = (uint64_t)ra8_jof_stored_bound((uint32_t)stage);
613 const uint64_t index_bytes = cols * rows * (uint64_t)k_ra8_jof_index_entry;
614 const uint64_t dfl = (uint64_t)k_ra8_io_compress_scratch_bytes;
615 const uint64_t total =
616 dec_set + band + stage + cmp + index_bytes + dfl + (uint64_t)k_ra8_jof_carve_slack;
617 if ((stage > (uint64_t)UINT32_MAX) || (total > (uint64_t)UINT32_MAX)) {
618 return 0U;
619 }
620 return (uint32_t)total;
621}

680static ra8_err_t priv_sniff_head(const ra8_jof_produce_cfg_t* cfg, uint8_t* head)
681{
682 size_t head_len = 0U;

684 size_t got = 0U;
685 const ra8_err_t err =
686 cfg->pull(cfg->pull_ctx, &head[head_len], (size_t)k_ra8_jof_sniff_bytes - head_len, &got);

719RA8_INTERNAL
720static ra8_err_t
721priv_dispatch(ra8_jof_prod_state_t* st, const uint8_t* head, ra8_jof_prefix_pull_t* pfx)
722{
723 if ((head[0] == (uint8_t)k_ra8_jof_jpeg_soi_first) &&
724 (head[1] == (uint8_t)k_ra8_jof_jpeg_soi_second)) {
725 uint8_t* window = ra8_jof_priv_bump_take(st->bump, (size_t)k_ra8_jpeg_sw_stream_min_window);
726 if (window == nullptr) {
727 return k_ra8_err_invalid_size;
728 }
729 return ra8_jpeg_sw_decode_stripes(ra8_jof_priv_prefix_pull,
730 pfx,
731 window,
732 (uint32_t)k_ra8_jpeg_sw_stream_min_window,
733 priv_jpeg_geom,
734 ra8_jof_priv_on_rows,
735 st);
736 }
737 if (memcmp(head, s_prod_png_sig, sizeof(s_prod_png_sig)) == 0) {
738 return ra8_jof_priv_png_rows(ra8_jof_priv_prefix_pull,

857RA8_INTERNAL
858static ra8_err_t priv_probe_sniff(const uint8_t* data, size_t len, uint16_t* out_w, uint16_t* out_h)
859{
860 if (len < (size_t)k_ra8_jof_sniff_bytes) {
861 return k_ra8_err_not_supported; /* too short to carry any accepted magic */
862 }
863 uint32_t w = 0U;
864 uint32_t h = 0U;
865 if ((data[0] == (uint8_t)k_ra8_jof_jpeg_soi_first) &&
866 (data[1] == (uint8_t)k_ra8_jof_jpeg_soi_second)) {
867 return ra8_jpeg_sw_get_dimensions(data, (uint32_t)len, out_w, out_h);
868 }
869 if (memcmp(data, s_prod_png_sig, sizeof(s_prod_png_sig)) == 0) {
870 const ra8_err_t rc = priv_png_dims(data, len, &w, &h);
871 if (rc != k_ra8_ok) {
872 return rc;
873 }
874 } else if (priv_is_webp(data)) {
875 const ra8_err_t rc = ra8_webp_get_info(data, len, &w, &h);
876 if (rc != k_ra8_ok) {
877 return rc;
878 }
879 } else {
880 return k_ra8_err_not_supported; /* not a JPEG/PNG/WebP source */
881 }
882 if ((w == 0U) || (h == 0U) || (w > (uint32_t)k_ra8_jof_max_dim) ||
883 (h > (uint32_t)k_ra8_jof_max_dim)) {
884 return k_ra8_err_invalid_size;
885 }
886 *out_w = (uint16_t)w;
887 *out_h = (uint16_t)h;
888 return k_ra8_ok;
889}
