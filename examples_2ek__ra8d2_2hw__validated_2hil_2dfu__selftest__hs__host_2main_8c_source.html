411static VOID dfu_device_worker(ULONG arg)
412{
413 (void)arg;
414
415 /* Config A device half: the FS controller runs the DFU device class wired
416 * to real MRAM (libs/ra8_dfu). DFU_DNLOAD programs the inactive Slot B. */
417 ra8_dfu_device_set_target(k_ra8_dfu_slot_b);
418 const ra8_err_t e = ra8_dfu_device_start(k_ra8_usb_speed_fs,
419 s_usbx_pool,
420 (uint32_t)sizeof(s_usbx_pool),
421 s_device_framework,
422 (uint32_t)sizeof(s_device_framework),
423 s_string_framework,
424 (uint32_t)sizeof(s_string_framework),
425 s_language_id_framework,
426 (uint32_t)sizeof(s_language_id_framework));
427 if (e != k_ra8_ok) {
428 s_dbg_host_err = (uint32_t)e;
429 return;
430 }
431 s_dbg_dev_step = 4U;
432 /* Each DFU_DNLOAD block is programmed synchronously inside the DFU write
433 * callback; this worker only finalizes the image header once the host
434 * signals end-of-download (the self-test uses DFU_ABORT, so that path is
435 * the bootloader's, not this twin's). */
436 while (1) {
437 (void)ra8_dfu_device_worker_step();
438 tx_thread_sleep(k_dfu_idle_ticks);
439 }
440}

538[[nodiscard]] static ra8_err_t dfu_print_hex(uint32_t value, uint8_t digits)
539{
540 uint8_t out[k_dfu_hex_chars_u32] = {};
541 uint8_t width = digits;
542 if (width > (uint8_t)k_dfu_hex_chars_u32) {
543 width = (uint8_t)k_dfu_hex_chars_u32;
544 }
545 for (uint8_t i = 0U; i < width; i++) {
546 const uint8_t shift = (uint8_t)((width - 1U - i) * k_dfu_nibble_bits);
547 out[i] = dfu_nibble_to_hex((value >> shift) & k_dfu_nibble_mask);
548 }
549 return dfu_sci_write(out, (uint32_t)width);
550}

564[[nodiscard]] static ra8_err_t dfu_print_dec(uint32_t value)
565{
566 uint8_t scratch[k_dfu_dec_chars_u32] = {};
567 uint8_t out[k_dfu_dec_chars_u32] = {};
568 uint8_t count = 0U;
569 uint32_t v = value;
570 if (v == 0U) {
571 out[0] = (uint8_t)'0';
572 return dfu_sci_write(out, 1U);
573 }
574 while (v != 0U) {
575 if (count >= (uint8_t)k_dfu_dec_chars_u32) {
576 break;
577 }
578 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_dfu_dec_radix));
579 v = v / k_dfu_dec_radix;
580 count++;
581 }
582 for (uint8_t i = 0U; i < count; i++) {
583 out[i] = scratch[count - 1U - i];
584 }
585 return dfu_sci_write(out, (uint32_t)count);
586}

680[[nodiscard]] static ra8_err_t dfu_host_pass(void)
681{
682 s_dbg_phase = (uint32_t)k_dfu_phase_init;
683 ra8_err_t err = dfu_print("ra8d2 dfu: host up on USB-HS, probing the loop...\r\n");
684 if (err != k_ra8_ok) {
685 return err;
686 }
687
688 /* Build the deterministic test image the host will DFU_DNLOAD; the device
689 * programs it into the inactive MRAM slot and we DFU_UPLOAD + byte-verify. */
690 for (uint32_t b = 0U; b < (uint32_t)k_dfu_blocks; b++) {
691 dfu_pattern_fill(b, &s_dfu_image[b * (uint32_t)k_dfu_xfer_size], (uint32_t)k_dfu_xfer_size);
692 }
693
694 s_dbg_phase = (uint32_t)k_dfu_phase_download;
695 ra8_dfu_host_result_t res = {};
696 err = ra8_dfu_host_run(k_ra8_usb_speed_hs, s_dfu_image, (uint32_t)k_dfu_image_bytes, &res);
697 s_dbg_pid = res.pid;
698 s_dbg_blocks_ok = res.blocks_ok;
699 s_dbg_mismatch = res.mismatch;
700 if (err != k_ra8_ok) {
701 (void)dfu_print_fail("dfu round-trip", err);
702 return err;
703 }
704
705 err = dfu_print("ra8d2 dfu: enumerated pid=0x");
706 if (err == k_ra8_ok) {
707 err = dfu_print_hex(s_dbg_pid, (uint8_t)k_dfu_hex_chars_u16);
708 }
709 if (err == k_ra8_ok) {
710 err = dfu_print("\r\n");
711 }
712 if (err != k_ra8_ok) {
713 return err;
714 }
715
716 s_dbg_phase = (uint32_t)k_dfu_phase_pass;
717 s_dbg_pass_count++;
718 err = dfu_print("ra8d2 dfu: ");
719 if (err == k_ra8_ok) {
720 err = dfu_print_dec((uint32_t)k_dfu_blocks);
721 }
722 if (err == k_ra8_ok) {
723 err =
724 dfu_print(" blocks DFU-programmed to MRAM + verified -- USB SELFTEST DFU CONFIG A PASS\r\n");
725 }
726 if (err != k_ra8_ok) {
727 return err;
728 }
729 (void)ra8_board_led_on(k_ra8_board_led2);
730 return k_ra8_ok;
731}
