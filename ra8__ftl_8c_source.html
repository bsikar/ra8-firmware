81{
82 RA8_CHECK_NULL_PTR(ftl, s_tag, "ftl must not be nullptr");
83 RA8_CHECK_NULL_PTR(ftl->raw, s_tag, "raw must not be nullptr");
84 for (uint32_t p = 0; p < ftl->physical_blocks; ++p) {
85 if (ftl->pblocks[p].state != (uint8_t)k_ra8_ftl_pstate_stale) {

89 if (e != k_ra8_ok) {
90 return e;
91 }
92 ftl->pblocks[p].erase_count += 1U;
93 ftl->pblocks[p].state = (uint8_t)k_ra8_ftl_pstate_free;
94 }
95 return k_ra8_ok;

125static ra8_err_t ftl_pick_free(const ra8_ftl_t* ftl, uint32_t* out)
126{
127 RA8_CHECK_NULL_PTR(ftl, s_tag, "ftl must not be nullptr");
128 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
129 bool found = false;
130 uint32_t best = 0;
131 uint32_t bestc = 0;
132 for (uint32_t p = 0; p < ftl->physical_blocks; ++p) {
133 if (ftl->pblocks[p].state != (uint8_t)k_ra8_ftl_pstate_free) {
134 continue;
135 }
136 if (found) {
137 if (ftl->pblocks[p].erase_count >= bestc) {
138 continue;
139 }
140 }
141 found = true;
142 best = p;
143 bestc = ftl->pblocks[p].erase_count;
144 }
145 if (!found) {
146 return k_ra8_err_no_data;
147 }
148 *out = best;
149 return k_ra8_ok;

181static ra8_err_t ftl_alloc_blank(ra8_ftl_t* ftl, uint32_t* out)
182{
183 RA8_CHECK_NULL_PTR(ftl, s_tag, "ftl must not be nullptr");
184 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
185 uint32_t phys = 0;
186 ra8_err_t pick = ftl_pick_free(ftl, &phys);
187 if (pick == k_ra8_err_no_data) {
188 const ra8_err_t rec = ftl_reclaim_stale(ftl);
189 if (rec != k_ra8_ok) {
190 return rec;
191 }
192 pick = ftl_pick_free(ftl, &phys);
193 }
194 if (pick == k_ra8_err_no_data) {
195 return k_ra8_err_no_mem;
196 }
197 /* ftl_pick_free with non-null args returns only k_ra8_ok or k_ra8_err_no_data. */
198 if (pick != k_ra8_ok) {
199 return pick; /* GCOVR_EXCL_LINE */
200 }
201 const ra8_err_t e = ra8_io_blockdev_erase(ftl->raw, phys, (uint32_t)k_ra8_ftl_one_block);
202 if (e != k_ra8_ok) {
203 return e;
204 }
205 ftl->pblocks[phys].erase_count += 1U;
206 *out = phys;
207 return k_ra8_ok;
