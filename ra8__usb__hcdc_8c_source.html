591ra8_err_t ra8_usb_hcdc_init(ra8_usb_speed_t speed)
592{
593 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
594 return k_ra8_err_invalid_arg;
595 }
596 const ra8_err_t usb_err = ra8_usb_host_init(speed);
597 if (usb_err != k_ra8_ok) {
598 ra8_log_error_val(s_tag, "ra8_usb_host_init failed", (uint32_t)usb_err);
599 return k_ra8_err_hw_init_failed;
600 }
601
602 s_state.speed = speed;
603 s_state.step = k_ra8_hcdc_step_idle;
604 s_state.attached = false;
605 s_state.attach_cb = nullptr;
606 s_state.attach_ctx = nullptr;
607 s_state.device = (ra8_usb_hcdc_device_t){};
608 s_state.initialized = true;
609
610 ra8_log_info_val(s_tag, "host-CDC ready", (uint32_t)speed);
611 return k_ra8_ok;
612}

614ra8_err_t ra8_usb_hcdc_close(void)
615{
616 if (!s_state.initialized) {
617 return k_ra8_err_invalid_state;
618 }
619 /* Bus power down: drop UACT before tearing the controller. */
620 (void)ra8_usb_host_set_uact(s_state.speed, false);
621 const ra8_err_t err = ra8_usb_host_deinit(s_state.speed);
622 s_state.initialized = false;
623 s_state.attached = false;
624 s_state.attach_cb = nullptr;
625 s_state.attach_ctx = nullptr;
626 s_state.step = k_ra8_hcdc_step_idle;
627 return err;
628}

664ra8_err_t ra8_usb_hcdc_recv(uint8_t* out_buf, uint16_t max_len, uint16_t* got_len)
665{
666 RA8_CHECK_NULL_PTR(out_buf, s_tag, "hcdc_recv: out_buf");
667 RA8_CHECK_NULL_PTR(got_len, s_tag, "hcdc_recv: got_len");
668 if (!s_state.initialized) {
669 return k_ra8_err_invalid_state;
670 }
671 if (!s_state.attached) {
672 return k_ra8_err_invalid_state;
673 }
674 if (max_len == 0U) {
675 return k_ra8_err_invalid_arg;
676 }
677 uint16_t inout_len = max_len;
678 const ra8_err_t err =
679 ra8_usb_queue_out(s_state.speed, k_ra8_hcdc_pipe_bulk_in, out_buf, &inout_len, true);
680 if (err == k_ra8_ok) {
681 *got_len = inout_len;
682 } else {
683 *got_len = 0U;
684 }
685 return err;
686}

693ra8_err_t ra8_usb_hcdc_set_line_coding(uint32_t baud,
694 ra8_usb_hcdc_parity_t parity,
695 ra8_usb_hcdc_stop_bits_t stop_bits)
696{
697 if (!s_state.initialized) {
698 return k_ra8_err_invalid_state;
699 }
700 if (!s_state.attached) {
701 return k_ra8_err_invalid_state;
702 }
703 if (baud < k_ra8_hcdc_baud_min) {
704 return k_ra8_err_invalid_arg;
705 }
706 if (parity > k_ra8_hcdc_parity_space) {
707 return k_ra8_err_invalid_arg;
708 }
709 if (stop_bits > k_ra8_hcdc_stop_2) {
710 return k_ra8_err_invalid_arg;
711 }
712
713 /* The 7-byte payload for SET_LINE_CODING is staged in the DCP
714 * data stage; the SETUP packet itself only carries class /
715 * recipient / wLength = 7. The data stage (USBREQ / FIFO write)
716 * is wired in the production CTRT path; the starter issues the
717 * SETUP and lets the controller drive the data stage from the
718 * host-side CFIFO. The serialised line-coding word is held off
719 * the stack so its lifetime crosses the SETUP request. */
720 const ra8_usb_setup_t setup = {
721 .bm_request_type = k_ra8_hcdc_bm_class_iface_out,
722 .b_request = k_ra8_hcdc_req_set_line_coding,
723 .w_value = 0U,
724 .w_index = 0U,
725 .w_length = k_ra8_hcdc_line_coding_len,
726 };
727 /* Touch the parameters so the unit-test build does not warn about
728 * unused locals: the production data-stage path consumes them. */
729 (void)baud;
730 (void)parity;
731 (void)stop_bits;
732 return ra8_usb_host_setup_request(s_state.speed, &setup);
733}
