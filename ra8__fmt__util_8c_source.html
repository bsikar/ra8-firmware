113ra8_err_t ra8_fmt_slurp(const char* path, ra8_fmt_blob_t* out)
114{
115 RA8_CHECK_NULL_PTR(path, s_tag, "path must not be nullptr");
116 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
117 *out = (ra8_fmt_blob_t){};
118 FILE* f = fopen(path, "rb");
119 if (f == nullptr) {
120 return k_ra8_err_not_found;
121 }
122 size_t len = 0U;
123 ra8_err_t rc = priv_stream_len(f, &len);
124 if (rc != k_ra8_ok) {
125 (void)fclose(f);
126 return rc;
127 }
128 uint8_t* buf = (uint8_t*)malloc(len);
129 if (buf == nullptr) {
130 (void)fclose(f);
131 return k_ra8_err_no_mem;
132 }
133 const size_t got = fread(buf, 1U, len, f);
134 (void)fclose(f);
135 if (got != len) {
136 free(buf);
137 return k_ra8_err_not_found;
138 }
139 out->bytes = buf;
140 out->len = len;
141 return k_ra8_ok;
142}

154ra8_err_t ra8_fmt_write_file(const char* path, const uint8_t* buf, size_t len)
155{
156 RA8_CHECK_NULL_PTR(path, s_tag, "path must not be nullptr");
157 if ((buf == nullptr) && (len > 0U)) {
158 ra8_log_error(s_tag, "buf must not be nullptr for a non-empty write");
159 return k_ra8_err_null_ptr;
160 }
161 FILE* f = fopen(path, "wb");
162 if (f == nullptr) {
163 return k_ra8_fail;
164 }
165 const bool wrote_ok = (len == 0U) || (fwrite(buf, 1U, len, f) == len);
166 const bool close_ok = (fclose(f) == 0);
167 if (!wrote_ok || !close_ok) {
168 (void)remove(path);
169 return k_ra8_fail;
170 }
171 return k_ra8_ok;
172}

190static void priv_hex_line(FILE* out, const uint8_t* row, size_t n, uint64_t addr)
191{
192 (void)fprintf(out, " %08llX ", (unsigned long long)addr);
193 for (size_t i = 0U; i < (size_t)k_ra8_fmt_hex_row; ++i) {
194 if (i < n) {
195 (void)fprintf(out, "%02X ", row[i]);
196 } else {
197 (void)fprintf(out, " ");
198 }
199 }
200 (void)fprintf(out, " |");
201 for (size_t i = 0U; i < n; ++i) {
202 const bool printable =
203 (row[i] >= (uint8_t)k_fmt_ascii_low) && (row[i] <= (uint8_t)k_fmt_ascii_high);
204 (void)fputc(printable ? (int)row[i] : (int)k_fmt_ascii_dot, out);
205 }
206 (void)fprintf(out, "|\n");
207}

209void ra8_fmt_hex_dump(FILE* out, const char* label, const uint8_t* buf, size_t len, uint64_t base)
210{
211 if ((out == nullptr) || (label == nullptr)) {
212 return;
213 }
214 (void)fprintf(out, "%s (%zu bytes at +%llu):\n", label, len, (unsigned long long)base);
215 if ((buf == nullptr) || (len == 0U)) {
216 (void)fprintf(out, " <empty>\n");
217 return;
218 }
219 size_t done = 0U;
220 while (done < len) { /* bounded by len */
221 size_t n = len - done;
222 if (n > (size_t)k_ra8_fmt_hex_row) {
223 n = (size_t)k_ra8_fmt_hex_row;
224 }
225 priv_hex_line(out, &buf[done], n, base + (uint64_t)done);
226 done += n;
227 }
