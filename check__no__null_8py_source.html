4"""check_no_null.py -- enforce the C23 nullptr-only rule on first-party code.
5
6Project policy: first-party C must use ``nullptr`` instead of ``NULL`` for
7null pointer constants. This script walks staged or all-tracked source files
8and rejects bare ``NULL`` tokens in code positions. It allows NULL in:
9
10 * comment lines and inline `/* ... */` / `// ...` comments
11 * string literals
12 * Doxygen annotation prose
13 * UX_NULL / similar vendor macros (USBX expects literal NULL)
14
15Scope is DERIVED from git ls-files (#358), so tools/ -- host tooling held to
16the same C23 bar, and silently omitted by the old ROOT_DIRS tuple -- is now in
17scope, along with every future top-level directory. Vendored SOUP
18(libs/third_party/, libs/ra8_fonts/, port/threadx/, ...) is skipped wholesale, and
19one first-party tree is exempt for a stated reason (see EXEMPT_PREFIXES):
20tests/ (NULL is deliberate null-guard stimulus).
21
22Usage:
23 python3 scripts/checks/check_no_null.py FILE [FILE ...]
24 python3 scripts/checks/check_no_null.py --all
25
26Returns 0 on clean, 1 on findings, 2 on usage error.
27"""

65# bare NULL token in a code context. \bNULL\b matches the identifier;
66# we strip comments and strings first so this only fires in code.
67NULL_RE = re.compile(r"\bNULL\b")

70# Strip C/C++ inline comments and string/char literals so we don't
71# match NULL inside them. Naive but adequate for this use case.
72def _strip_noncode(line: str) -> str: # noqa: PLR0912 PLR0915 # char-by-char state machine, splitting hurts readability
73 out = []
74 i = 0
75 in_str = False
76 in_chr = False
77 in_lc = False
78 in_bc = False
79 n = len(line)
80 while i < n:
81 c = line[i]
82 nxt = line[i + 1] if i + 1 < n else ""
83 if in_lc:
84 break
85 if in_bc:
86 if c == "*" and nxt == "/":
87 in_bc = False
88 i += 2
89 continue
90 i += 1
91 continue
92 if in_str:
93 if c == "\\" and i + 1 < n:
94 i += 2
95 continue
96 if c == '"':
97 in_str = False
98 i += 1
99 continue
100 if in_chr:
101 if c == "\\" and i + 1 < n:
102 i += 2
103 continue
104 if c == "'":
105 in_chr = False
106 i += 1
107 continue
108 if c == "/" and nxt == "/":
109 break
110 if c == "/" and nxt == "*":
111 in_bc = True
112 i += 2
113 continue
114 if c == '"':
115 in_str = True
116 i += 1
117 continue
118 if c == "'":
119 in_chr = True
120 i += 1
121 continue
122 out.append(c)
123 i += 1
124 return "".join(out)

127# Identifiers that LOOK like NULL but are actually allowed: vendor macros
128# the project doesn't own (USBX expects literal NULL in its API contract,
129# ThreadX-FileX-NetXDuo similar). We accept a token if its full
130# identifier is in this set or it has one of these suffixes.
131ALLOWED_TOKENS = {"UX_NULL", "TX_NULL", "FX_NULL", "NX_NULL"}

134def find_violations(path: pathlib.Path) -> list[tuple[int, str]]:
135 """Report every use of ``NULL`` in one file.
136
137 C23 spells the null pointer constant ``nullptr``, which is typed; ``NULL``
138 is a macro that expands to an untyped 0 and so silently satisfies an
139 integer parameter. That is the defect this catches, not the spelling.
140
141 Returns ``(line_no, line_text)`` per finding; an unreadable file yields an
142 empty list rather than raising.
143 """
144 violations: list[tuple[int, str]] = []
145 try:
146 text = path.read_text(encoding="utf-8", errors="replace")
147 except OSError:
148 return violations
149 in_block_comment = False
150 for n, raw in enumerate(text.splitlines(), 1):
151 # Multi-line block comment continuation handling.
152 cur = raw
153 if in_block_comment:
154 end = cur.find("*/")
155 if end == -1:
156 continue
157 cur = cur[end + 2 :]
158 in_block_comment = False
159 # Detect a /* on this line that doesn't close.
160 bo = cur.find("/*")
161 if bo != -1 and cur.find("*/", bo + 2) == -1:
162 cur = cur[:bo]
163 in_block_comment = True
164 code = _strip_noncode(cur)
165 if "NULL" not in code:
166 continue
167 # Skip allowed tokens.
168 # Replace allowed tokens with a placeholder so the regex doesn't fire.
169 scrubbed = code
170 for tok in ALLOWED_TOKENS:
171 scrubbed = scrubbed.replace(tok, "_OK_")
172 if NULL_RE.search(scrubbed):
173 violations.append((n, raw.strip()))
174 return violations

177def _in_scope(rel: str) -> bool:
178 """Whether repo-relative ``rel`` is first-party C the nullptr rule governs.
179
180 Pure and total, so the selftest asserts scope on synthetic paths without
181 touching the tree: a re-narrowing that drops tools/ fails the selftest
182 instead of passing green.
183 """
184 if not rel.endswith(EXTENSIONS):
185 return False
186 if rel.startswith(EXEMPT_PREFIXES) or rel.startswith(SOUP_PREFIXES):
187 return False

191def needs_check(path: pathlib.Path) -> bool:
192 """Whether a CLI-supplied path is first-party C subject to the nullptr rule."""
193 if path.suffix.lower() not in EXTENSIONS:
194 return False
195 try:
196 rel = path.resolve().relative_to(REPO_ROOT).as_posix()
197 except ValueError:
198 # Outside the repo (an isolated selftest fixture): the suffix already
199 # matched and no repo-relative prefix rule can apply, so it is in scope.
200 return True
201 return _in_scope(rel)

204def iter_all_files() -> Iterable[pathlib.Path]:
205 """Every first-party C file the rule governs, for the ``--all`` sweep.
206
207 Derived from git ls-files via first_party_paths (which already removes
208 vendored SOUP, generated tables, build output and the vendored port/threadx
209 tree), minus the two documented EXEMPT_PREFIXES. A newly-added top-level
210 directory of first-party C is covered the day it lands -- no allowlist.
211 """
212 for rel in first_party_paths(EXTENSIONS):
213 if not rel.startswith(EXEMPT_PREFIXES):
214 yield REPO_ROOT / rel

217# ---------------------------------------------------------------------------
218# Selftest -- both directions, plus a scope assertion under tools/, the root
219# ROOT_DIRS silently omitted until #358. A scope that quietly re-narrows must
220# fail here, not report a clean tree over files it stopped scanning.
221# ---------------------------------------------------------------------------
222_BAD_FIXTURE = "int f(void) { char *p = NULL; return p == NULL; }\n"
223_GOOD_FIXTURE = (
224 "int f(void) { char *p = nullptr; // NULL in a comment is fine\n"
225 ' const char *s = "NULL literal"; // and in a string literal\n'
226 " return (p == nullptr) && (UX_NULL == p); }\n"
227)

230def selftest() -> int:
231 """Prove bare NULL fires, legal constructs stay quiet, and the scope holds."""
232 print("check_no_null.py --selftest")
233 failures: list[str] = []
234 with tempfile.TemporaryDirectory() as tmp:
235 bad = pathlib.Path(tmp) / "bad.c"
236 bad.write_text(_BAD_FIXTURE, encoding="utf-8")
237 good = pathlib.Path(tmp) / "good.c"
238 good.write_text(_GOOD_FIXTURE, encoding="utf-8")
239 expect(bool(find_violations(bad)), "bare NULL in code fires", failures)
240 expect(
241 not find_violations(good),
242 "nullptr / vendor macro / comment / string stays quiet",
243 failures,
244 )
245 expect(
246 _in_scope("tools/mkbookimg/mkbookimg.c"),
247 "tools/ is in scope (ROOT_DIRS omitted it before #358)",
248 failures,
249 )
250 expect(not _in_scope("tests/test_x.c"), "tests/ exempt (deliberate NULL stimulus)", failures)
251 expect(not _in_scope("libs/third_party/miniz/miniz.c"), "vendored SOUP exempt", failures)
252 return report(failures)

255def main() -> int:
256 """Fail on any use of ``NULL`` where C23 ``nullptr`` is required.
257
258 ``--all`` sweeps the tree; otherwise only the named files are checked,
259 which is how the pre-commit hook stays fast.
260
261 Returns 1 listing each use, 0 when clean.
262 """
263 parser = argparse.ArgumentParser(description=__doc__)
264 parser.add_argument("--all", action="store_true", help="scan all tracked source files")
265 parser.add_argument(
266 "--selftest", action="store_true", help="prove the rule fires and the scope holds"
267 )
268 parser.add_argument(
269 "files", nargs="*", type=pathlib.Path, help="explicit file list (e.g. staged files)"
270 )
271 args = parser.parse_args()
272
273 if args.selftest:
274 return selftest()
275
276 if args.all:
277 candidates = list(iter_all_files())
278 elif args.files:
279 candidates = [p for p in args.files if needs_check(p)]
280 else:
281 parser.print_usage(sys.stderr)
282 return 2
283
284 total_violations = 0
285 for path in candidates:
286 for line, snippet in find_violations(path):
287 print(f"{path}:{line}: bare NULL -- use nullptr (C23): {snippet}", file=sys.stderr)
288 total_violations += 1
289
290 if total_violations:
291 print(
292 f"\n{total_violations} bare NULL token(s) found. Replace with "
293 "`nullptr` (C23 builtin). Allowed: UX_NULL / TX_NULL / FX_NULL "
294 "/ NX_NULL vendor macros, comments, string literals.",
295 file=sys.stderr,
296 )
297 return 1
298 print("check_no_null.py: 0 findings.")
299 return 0
