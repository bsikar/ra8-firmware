64static void pdm_write_mode(volatile r_pdm_ch_regs_t* reg, const ra8_pdm_channel_cfg_t* cfg)
65{
66 uint32_t pdmdsr = (uint32_t)(cfg->edge & (uint8_t)k_ra8_pdm_pdmdsr_inpsel);
67 pdmdsr |= ((uint32_t)cfg->sinc_order & k_pdm_field_3bit) << k_ra8_pdm_pdmdsr_sfmd_pos;
68 pdmdsr |= (uint32_t)cfg->hpf_shift << k_ra8_pdm_pdmdsr_hfis_pos;
69 pdmdsr |= (uint32_t)cfg->cf_shift << k_ra8_pdm_pdmdsr_cfis_pos;
70 pdmdsr |= (uint32_t)cfg->lpf_shift << k_ra8_pdm_pdmdsr_lfis_pos;
71 pdmdsr |= (uint32_t)cfg->data_shift << k_ra8_pdm_pdmdsr_dbis_pos;
72 /* HUM Ch 49.2.19 "PDMDSRCHn : Mode Setting Register" p 3208 */
73 reg->PDMDSR = pdmdsr;
74
75 uint32_t pdsfcr = (uint32_t)cfg->clock_div << k_ra8_pdm_pdsfcr_ckdiv_pos;
76 pdsfcr |= (uint32_t)cfg->sinc_dec << k_ra8_pdm_pdsfcr_sincdec_pos;
77 pdsfcr |= (uint32_t)cfg->sinc_range << k_ra8_pdm_pdsfcr_sincrng_pos;
78 /* HUM Ch 49.2.20 "PDSFCRCHn : Sinc Filter Control Register" p 3210 */
79 reg->PDSFCR = pdsfcr;
80}

146static int32_t pdm_sign_extend20(uint32_t raw)

215ra8_err_t ra8_pdm_read(uint8_t ch, int32_t* out, uint32_t max, uint32_t* out_count)
216{
217 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
218 RA8_CHECK_NULL_PTR(out_count, s_tag, "out_count must not be nullptr");
219 RA8_CHECK_RANGE_TAG(ch, 0, (uint8_t)k_ra8_pdm_ch_count - 1U, k_ra8_err_invalid_arg, s_tag);
220 if (max == 0U) {
221 ra8_log_error(s_tag, "read: max must be > 0");
222 return k_ra8_err_invalid_arg;
223 }
224
225 volatile r_pdm_ch_regs_t* reg = ra8_pdm_ch(ch);
226 /* HUM Ch 49.2.66 "PDDSRCHn : Data Status Register" p 3228 */
227 const uint32_t avail = reg->PDDSR & (uint32_t)k_ra8_pdm_pddsr_num_mask;
228 const uint32_t n = (avail < max) ? avail : max;
229 for (uint32_t i = 0U; i < n; ++i) {
230 /* HUM Ch 49.2.65 "PDDRRCHn : Data Read Register" p 3227 */
231 out[i] = pdm_sign_extend20(reg->PDDRR);
232 }
233 *out_count = n;
234 return k_ra8_ok;
235}

237ra8_err_t ra8_pdm_stop(uint8_t ch)
238{
239 RA8_CHECK_RANGE_TAG(ch, 0, (uint8_t)k_ra8_pdm_ch_count - 1U, k_ra8_err_invalid_arg, s_tag);
240
241 /* HUM Ch 49.2.63 "PDDRCRCHn : Data Read Control Register" p 3227 */
242 ra8_pdm_ch(ch)->PDDRCR = 0U;
243 /* HUM Ch 49.2.3 "PDCSTPTR : Channel Software Stop Trigger Register" p 3197 */
244 ra8_pdm()->PDCSTPTR = (uint32_t)k_ra8_pdm_pdstptr_stp << ch;
245
246 const uint32_t running = (uint32_t)k_ra8_pdm_pdsr_state << ch;
247 for (uint32_t i = 0U; i < (uint32_t)k_pdm_stop_poll_max; ++i) {
248 /* HUM Ch 49.2.6 "PDCSR : Channel Status Register" p 3199 */
249 if ((ra8_pdm()->PDCSR & running) == 0U) {
250 return k_ra8_ok;
251 }
252 }
253 ra8_log_error(s_tag, "stop: channel did not halt");
254 return k_ra8_err_hw_timeout;
