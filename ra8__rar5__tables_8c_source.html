51static uint8_t s_fetch_byte(ra8_rar5_state_t* st)
52{
53 if (st->refill_pos >= st->refill_len) {
54 if (st->fetched >= st->packlen) {
55 st->overrun = true;
56 return 0U;
57 }
58 const uint64_t remain = st->packlen - st->fetched;
59 const size_t want = (remain < (uint64_t)k_ra8_rar5_refill_bytes)
60 ? (size_t)remain
61 : (size_t)k_ra8_rar5_refill_bytes;
62 const size_t r = st->rar->read(st->rar->ctx, st->base + st->fetched, st->refill, want);
63 if (r == 0U) {
64 st->overrun = true;
65 return 0U;
66 }
67 st->refill_len = (uint32_t)r;
68 st->refill_pos = 0U;
69 st->fetched += (uint64_t)r;
70 }
71 const uint8_t b = st->refill[st->refill_pos];
72 st->refill_pos += 1U;
73 return b;
74}

194{
195 d->len[0] = 0U;

223static void s_make_tables(ra8_rar5_dtab_t* d, const uint8_t* lengths, uint16_t size)
224{
225 uint32_t count[k_r5_maxbits + 1U] = {};
226 for (uint16_t i = 0U; i < size; ++i) { /* bound: size <= k_ra8_rar5_nc */
227 count[lengths[i] & (uint8_t)k_r5_nibble_mask] += 1U;
228 }
229 count[0] = 0U;
230 s_tab_limits(d, count);
231 (void)memset(d->num, 0, (size_t)size * sizeof(d->num[0]));
232 uint32_t copypos[k_r5_maxbits + 1U];
233 for (uint32_t i = 0U; i < (uint32_t)(k_r5_maxbits + 1U); ++i) { /* bound: 16 */
234 copypos[i] = d->pos[i];
235 }
236 for (uint16_t i = 0U; i < size; ++i) { /* bound: size */
237 const uint8_t bl = lengths[i] & (uint8_t)k_r5_nibble_mask;
238 if (bl != 0U) {
239 d->num[copypos[bl]] = i;
240 copypos[bl] += 1U;
241 }

247uint32_t ra8_rar5_decode_num(ra8_rar5_state_t* st, const ra8_rar5_dtab_t* d)
248{
249 const uint32_t bf = s_peek(st, (uint32_t)k_r5_bf_bits) & (uint32_t)k_r5_bf_mask;
250 uint32_t bits = (uint32_t)k_r5_maxbits;
251 for (uint32_t i = 1U; i < (uint32_t)k_r5_maxbits; ++i) { /* bound: 14 lengths */
252 if (bf < d->len[i]) {
253 bits = i;
254 break;
255 }
256 }
257 s_drop(st, bits);
258 const uint32_t dist = (bf - d->len[bits - 1U]) >> ((uint32_t)k_r5_bf_bits - bits);
259 uint32_t pos = d->pos[bits] + dist;
260 if (pos >= (uint32_t)d->max) {
261 pos = 0U;
262 }
263 return d->num[pos];
264}

284static uint8_t s_checksum(uint32_t flags, uint64_t blocksize)
285{
286 uint32_t x = (uint32_t)k_r5_hdr_chk_seed ^ flags;
287 x ^= (uint32_t)(blocksize & (uint64_t)k_r5_byte_mask);
288 x ^= (uint32_t)((blocksize >> k_r5_byte_bits) & (uint64_t)k_r5_byte_mask);
289 x ^= (uint32_t)((blocksize >> (k_r5_byte_bits * 2U)) & (uint64_t)k_r5_byte_mask);
290 return (uint8_t)(x & (uint32_t)k_r5_byte_mask);
291}

294ra8_err_t ra8_rar5_read_block_header(ra8_rar5_state_t* st, r5_block_t* b)
295{
296 s_align(st);
297 const uint32_t flags = ra8_rar5_get(st, (uint32_t)k_r5_byte_bits);
298 const uint32_t bytecount =
299 ((flags >> (uint32_t)k_r5_bf_bcount_shift) & (uint32_t)k_r5_bf_bcount_mask) + 1U;
300 uint64_t bsz = 0U;
301 for (uint32_t i = 0U; i < bytecount; ++i) { /* bound: bytecount <= 4 */
302 bsz |= (uint64_t)ra8_rar5_get(st, (uint32_t)k_r5_byte_bits) << (i * (uint32_t)k_r5_byte_bits);
303 }
304 const uint32_t saved = ra8_rar5_get(st, (uint32_t)k_r5_byte_bits);
305 if ((uint32_t)s_checksum(flags, bsz) != saved) {
306 return k_ra8_err_validation_failed;
307 }
308 b->size = bsz;
309 b->last_bits = (flags & (uint32_t)k_r5_bf_bitsize_mask) + 1U;
310 b->tables = (flags & (uint32_t)k_r5_bf_tables) != 0U;
311 b->last = (flags & (uint32_t)k_r5_bf_last) != 0U;
312 return k_ra8_ok;
313}

364ra8_err_t ra8_rar5_apply_run(ra8_rar5_state_t* st, uint8_t* tbl, uint32_t* idx, uint32_t num)
365{
366 const bool is_long =
367 (num == (uint32_t)k_r5_tbl_copy_long) || (num == (uint32_t)k_r5_tbl_zero_long);
368 const bool is_zero =
369 (num == (uint32_t)k_r5_tbl_zero_short) || (num == (uint32_t)k_r5_tbl_zero_long);
370 const uint32_t count =
371 is_long ? (ra8_rar5_get(st, (uint32_t)k_r5_run_long_bits) + (uint32_t)k_r5_run_long_add)
372 : (ra8_rar5_get(st, 3U) + 3U);
373 if ((!is_zero) && (*idx == 0U)) {
374 return k_ra8_err_validation_failed;
375 }
376 const uint8_t prev = (*idx > 0U) ? tbl[*idx - 1U] : 0U;
377 const uint8_t fill = is_zero ? 0U : prev;
378 uint32_t i = *idx;
379 for (uint32_t c = 0U; (c < count) && (i < (uint32_t)k_ra8_rar5_huff_total);
380 ++c) { /* bound: count */
381 tbl[i] = fill;
382 i += 1U;
383 }
384 *idx = i;
385 return k_ra8_ok;
386}

406{
407 uint32_t i = 0U;
408 while (i < (uint32_t)k_ra8_rar5_huff_total) { /* bound: i advances each pass */
409 const uint32_t num = ra8_rar5_decode_num(st, &st->bd);

424ra8_err_t ra8_rar5_read_tables(ra8_rar5_state_t* st)
425{
426 uint8_t bdlen[k_ra8_rar5_bc] = {};

433 e = s_read_full_table(st, tbl);

435 return e;

438 s_make_tables(&st->dd, &tbl[k_ra8_rar5_nc], (uint16_t)k_ra8_rar5_dc);
439 s_make_tables(&st->ldd, &tbl[k_ra8_rar5_nc + k_ra8_rar5_dc], (uint16_t)k_ra8_rar5_ldc);
440 s_make_tables(&st->rd,

442 (uint16_t)k_ra8_rar5_rc);
