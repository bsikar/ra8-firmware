41def _string_list(payload: object, key: str) -> tuple[str, ...]:
42 """Read one non-empty duplicate-free string list."""
43 if not isinstance(payload, dict):
44 fail("tracker schema root must be an object")
45 raw = payload.get(key)
46 if not isinstance(raw, list) or not raw:
47 fail(f"tracker schema {key} must be a non-empty list")
48 values = tuple(raw)
49 if any(not isinstance(value, str) or not value or not value.isprintable() for value in values):
50 fail(f"tracker schema {key} contains an invalid value")
51 if len(set(values)) != len(values):
52 fail(f"tracker schema {key} contains a duplicate")
53 if len({value.casefold() for value in values}) != len(values):
54 fail(f"tracker schema {key} contains a case-insensitive duplicate")
55 return values

59def tracker_schema() -> TrackerSchema:
60 """Load and validate ``scripts/dev/work/tracker.json`` once per process."""
61 path = Path(__file__).resolve().parents[1] / "tracker.json"
62 try:
63 payload = json.loads(path.read_text(encoding="ascii"))
64 except (OSError, UnicodeError, json.JSONDecodeError) as exc:
65 fail(f"tracker schema could not be read: {exc}", exc)
66 expected = {
67 "schema_version",
68 "github_host",
69 "repository",
70 "project_owner",
71 "project_number",
72 "statuses",
73 "tracks",
74 }
75 valid_root = isinstance(payload, dict) and set(payload) == expected
76 if not valid_root or payload.get("schema_version") != SCHEMA_VERSION:
77 fail("tracker schema keys or version are unsupported")
78 host = payload["github_host"]
79 repository = payload["repository"]
80 owner = payload["project_owner"]
81 number = payload["project_number"]
82 if host != "github.com":
83 fail("tracker schema must pin github.com")
84 if not isinstance(repository, str) or REPOSITORY_RE.fullmatch(repository) is None:
85 fail("tracker schema repository must be owner/name")

88 if not isinstance(number, int) or isinstance(number, bool) or number < 1:
89 fail("tracker schema project_number must be positive")
90 return TrackerSchema(
91 github_host=host,
92 repository=repository,
93 project_owner=owner,
94 project_number=number,
95 statuses=_string_list(payload, "statuses"),
