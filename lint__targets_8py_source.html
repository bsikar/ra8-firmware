4"""One definition of "which files are first-party code", for the size gates.
5
6Four checkers in this tree have now had the same defect: a hand-written
7``SCAN_ROOTS`` / ``SOURCE_SUFFIXES`` tuple that quietly stopped describing the
8repository. ``check_file_size.py`` and ``check_function_size.py`` were the
9worst of them -- their roots omitted ``scripts/`` and their suffixes covered
10only C/C++, so the documented 1000-line file cap and the 60-line NASA Rule 4
11function cap had never once applied to a Python or shell file (#359).
12
13The failure mode is specific and worth naming: a hardcoded list does not fail
14when it goes stale. It reports success over a shrinking slice of the tree, and
15the gate looks green precisely because it stopped looking. So the enumeration
16is derived instead:
17
18* the file set comes from ``git ls-files`` -- whatever is in the repository is
19 in scope, and a new top-level directory is covered the day it is added;
20* language is decided per file by suffix, by well-known basename, or by
21 shebang, so an extensionless executable cannot escape by having no suffix;
22* the only subtractions are vendored SOUP and generated tables, which
23 CLAUDE.md already exempts by name.
24
25``check_lint_coverage.py`` asks a parallel question ("is every code file
26claimed by some linter?") and this module answers the size gates' half of it
27with the same enumeration, so the two cannot disagree about what code is.
28
29Run::
30
31 lint_targets.py # every first-party code file
32 lint_targets.py c python # only the named languages
33 lint_targets.py --list # the language names this module knows
34
35Prints one repo-relative path per line, sorted. Exits non-zero, printing
36nothing, when a requested language resolves to zero files: a gate must never
37mistake a broken enumeration for a clean tree.
38"""
39

66# ---------------------------------------------------------------------------
67# BUILD OUTPUT -- the single definition, shared by every checker in this tree.
68#
69# This used to be thirteen copies of the substring ``"/build/"``, one per
70# checker, and the substring is the defect (#377). ``"/build/" in path`` cannot
71# tell ``tools/ra8_emulator/build/`` -- genuine CMake output -- from a first-party
72# source directory that happens to be called ``build``. When #359's
73# reorganisation created ``scripts/build/``, every file in it # PATHREF-OK: #359
74# became invisible to shellcheck, shfmt and the rest, while every gate still
75# reported
76# a clean tree. The bare ``build/`` line in .gitignore did the same thing to
77# git, so a NEW file there would never have been added at all; the six that
78# survived did so only because ``git mv`` moves already-tracked files.
79#
80# The replacement is a repo-relative PATH check rather than a substring match.
81# A build directory counts as build output only where a build tree is actually
82# produced: at the repo root, or under one of the roots below. Anywhere else,
83# a directory named ``build`` is ordinary source and is linted like any other.
84#
85# .gitignore carries the matching anchored patterns, and
86# ``check_gitignore_scope.py`` fails on any new unanchored directory pattern,
87# so the two halves cannot drift back apart.
88# ---------------------------------------------------------------------------
89
90# Top-level directories beneath which a per-target build tree legitimately
91# appears, at any depth. Deliberately NOT "any directory anywhere": that is the
92# behaviour being removed. `scripts/`, `libs/`, `src/inc/` and friends are
93# absent because nothing builds into them, so a `build` directory appearing
94# there is source and must stay visible to the checkers.
95BUILD_TREE_ROOTS = frozenset(
96 {
97 "docs", # docs/build/ -- generated Doxygen HTML
98 "examples", # examples/**/<app>/build/ -- per-app CMake output
99 "local-poc", # local-poc/**/build/ -- git-excluded PoC tree
100 "port", # port/**/build/
101 "src", # src/app/build/
102 "tests", # tests/build/, tests/build-cov/, tests/build-fuzz/
103 "tools", # tools/<tool>/build/ -- host tool output
104 }
105)
106
107# Directory names owned by a tool, which can never be a first-party source
108# directory and are therefore matched at ANY depth. This is the ONLY
109# depth-agnostic rule left, and every name in it is reserved by the tool that
110# creates it: CMake writes CMakeFiles/ and _deps/, CPython writes __pycache__/,
111# npm writes node_modules/. Nobody can legitimately author a source directory
112# with one of these names, so matching them anywhere cannot swallow source.
113TOOL_OUTPUT_DIR_NAMES = frozenset({"CMakeFiles", "_deps", "__pycache__", "node_modules"})

116def is_build_dir_name(name: str) -> bool:
117 """True when one path COMPONENT names a build tree.
118
119 Exact ``build``, or a ``build-`` / ``build_`` / ``cmake-build-`` prefix.
120 The separator is required: ``builders`` starts with ``build`` and is NOT a
121 build directory, which is precisely the collision a ``build*`` glob would
122 reintroduce.
123 """
124 return name == "build" or name.startswith(("build-", "build_", "cmake-build-"))

127def is_build_output(rel: str) -> bool:
128 """True when repo-relative `rel` lives inside a build tree.
129
130 Directory components only -- a FILE called ``build`` is not a build tree.
131 """
132 parts = rel.split("/")
133 for index, part in enumerate(parts[:-1]):
134 if part in TOOL_OUTPUT_DIR_NAMES:
135 return True
136 if is_build_dir_name(part) and (index == 0 or parts[0] in BUILD_TREE_ROOTS):
137 return True
138 return False

183def is_build_output_path(path: object) -> bool:
184 """``is_build_output`` for a str or Path that may be absolute.
185
186 The checkers hold a mix of absolute paths, repo-relative paths and
187 slash-wrapped forms. Normalising here keeps every call site a single
188 predicate instead of thirteen hand-rolled substring tuples (#377).
189 """
190 text = str(path).replace("\\", "/").strip("/")
191 root = str(REPO_ROOT).replace("\\", "/").strip("/")
192 if text.startswith(root + "/"):
193 text = text[len(root) + 1 :]
194 elif text.startswith("./"):
195 text = text[2:]
196 return is_build_output(text)

199def _tracked() -> list[str]:
200 """Tracked plus untracked-but-not-ignored paths, from git itself."""
201 proc = subprocess.run(
202 [ # noqa: S607 -- trusted: fixed git argv
203 "git",
204 "ls-files",
205 "-z",
206 "--cached",
207 "--others",
208 "--exclude-standard",
209 ],
210 cwd=REPO_ROOT,
211 capture_output=True,
212 text=True,
213 check=False,
214 )
215 if proc.returncode != 0:
216 sys.stderr.write(proc.stderr)
217 sys.stderr.write("lint_targets.py: FATAL -- `git ls-files` failed\n")
218 sys.exit(2)
219 return [rel for rel in proc.stdout.split("\0") if rel]

222def _excluded(rel: str, lang: str | None = None) -> bool:
223 if rel.startswith(EXCLUDED_PREFIXES) or is_build_output(rel):
224 return True
225 extra = LANGUAGE_EXCLUDED_PREFIXES.get(lang or "", ())
226 return bool(extra) and rel.startswith(extra)

229def _shebang_lang(path: Path) -> str | None:
230 """Language named by a ``#!`` first line, or None.
231
232 This is the half of the enumeration a suffix list cannot do. An executable
233 with no extension is still code, and the git hooks are exactly that.
234 """
235 try:
236 with path.open("rb") as handle:
237 first = handle.readline(200).decode("utf-8", errors="replace")
238 except OSError:
239 return None
240 if not first.startswith("#!"):
241 return None
242 words = first[2:].replace("/usr/bin/env", " ").replace("/", " ").split()
243 for word in words:
244 base = word.split("-")[0]
245 if base in SHEBANG_LANG:
246 return SHEBANG_LANG[base]
247 return None

250def _raw_language(rel: str, root: Path) -> str | None:
251 """The language a path's name implies, before any exclusion is applied."""
252 path = Path(rel)
253 if path.name in BASENAME_LANG:
254 return BASENAME_LANG[path.name]
255 lang = SUFFIX_LANG.get(path.suffix)
256 if lang is not None:
257 return lang
258 if path.suffix:
259 return None # a suffix we know is not code (.md, .json, .pdf, ...)
260 return _shebang_lang(root / rel)

263def language_of(rel: str, root: Path = REPO_ROOT) -> str | None:
264 """The language of one repo-relative path, or None if it is not code.
265
266 The language is resolved BEFORE exclusion, because exclusion is now
267 per-language: a vendored tree can be SOUP for its sources and still hold
268 first-party build glue.
269 """
270 if _excluded(rel):
271 return None
272 lang = _raw_language(rel, root)
273 if lang is None or _excluded(rel, lang):
274 return None
275 return lang

278def files_for(languages: tuple[str, ...] = LANGUAGES) -> dict[str, list[str]]:
279 """Map each requested language to its sorted first-party file list."""
280 out: dict[str, list[str]] = {lang: [] for lang in languages}
281 for rel in _tracked():
282 lang = language_of(rel)
283 if lang in out:
284 out[lang].append(rel)
285 return {lang: sorted(paths) for lang, paths in out.items()}

288# A tree this size cannot legitimately collapse to a handful of files. A checker
289# that enumerates almost nothing reports a clean tree because it looked at
290# almost nothing -- the exact failure this module exists to prevent. Same
291# trip-wire as check_ruff.py and check_lint_coverage.py.
292TRACKED_FLOOR = 1000
293
294
295def first_party_paths(
296 suffixes: tuple[str, ...], *, respect_language_excludes: bool = True
297) -> list[str]:
298 """Every tracked first-party path ending in one of ``suffixes``.
299
300 The derived-scope primitive the policy checkers share (#358). Enumeration
301 is ``git ls-files`` -- never a hardcoded directory list -- so a newly added
302 top-level directory (``tools/`` was the one that had been silently omitted
303 for the life of six checkers) is in scope the day it lands, with no
304 allowlist to forget. The only subtractions are the named SOUP / generated /
305 build-output exemptions this module already defines; with
306 ``respect_language_excludes`` also the per-language vendored trees
307 (``port/threadx/`` is C SOUP), which are not ours to police.
308
309 Args:
310 suffixes: Extensions to keep, e.g. ``(".c", ".h")``. Matched with
311 ``str.endswith``, so pass lower-case dotted forms.
312 respect_language_excludes: When true, also drop a path that is a
313 vendored tree for the language its own suffix implies. Callers
314 scanning text (docs, config) pass false, where it is a no-op.
315
316 Returns:
317 The matching repo-relative paths, sorted.
318
319 Raises:
320 SystemExit: When ``git ls-files`` returns fewer than ``TRACKED_FLOOR``
321 paths -- a collapsed enumeration must fail, never read as clean.
322 """
323 rels = _tracked()
324 if len(rels) < TRACKED_FLOOR:
325 sys.stderr.write(
326 f"lint_targets.py: FATAL -- only {len(rels)} tracked path(s), floor "
327 f"is {TRACKED_FLOOR}. A collapsed enumeration reports a clean tree "
328 "because it enumerated nothing.\n"
329 )
330 sys.exit(2)
331 out: list[str] = []
332 for rel in rels:
333 if not rel.endswith(suffixes):
334 continue
335 if _excluded(rel):
336 continue
337 if respect_language_excludes:
338 lang = _raw_language(rel, REPO_ROOT)
339 if lang is not None and _excluded(rel, lang):
340 continue
341 out.append(rel)
342 return sorted(out)

345def main(argv: list[str]) -> int:
346 """Print the first-party file list, optionally filtered by language.
347
348 Exits non-zero printing NOTHING when a requested language resolves to zero
349 files. That is the contract the size gates depend on: an empty list must
350 be distinguishable from a clean tree, or a broken enumeration reads as
351 success.
352
353 Returns 0 with the paths on stdout, 1 on an unknown or empty language.
354 """
355 args = argv[1:]
356 if "--list" in args:
357 print("\n".join(LANGUAGES))
358 return 0
359 requested = tuple(a for a in args if not a.startswith("-")) or LANGUAGES
360 unknown = [lang for lang in requested if lang not in LANGUAGES]
361 if unknown:
362 sys.stderr.write(f"lint_targets.py: unknown language(s): {unknown}\n")
363 return 2
364 grouped = files_for(requested)
365 empty = [lang for lang, paths in grouped.items() if not paths]
366 if empty:
367 sys.stderr.write(
368 f"lint_targets.py: FATAL -- language(s) {empty} resolved to zero "
369 f"files. The enumeration is broken; refusing to report a clean scope.\n"
370 )
371 return 2
372 for lang in requested:
373 for rel in grouped[lang]:
374 print(rel)
375 return 0
