127RA8_INTERNAL static uint16_t ra8_c6link_frame_sum(uint8_t* frame, uint16_t span)

148RA8_PRIV void ra8_c6link_priv_frame_seal(uint8_t* tx, uint8_t if_type, uint8_t if_num, uint16_t len)

151 return;
152 }
153 const uint16_t span = (uint16_t)((uint16_t)k_ra8_c6link_header_bytes + len);
154
155 struct esp_payload_header hdr = {};
156 hdr.if_type = (uint8_t)(if_type & (uint8_t)k_ra8_c6link_hdr_nibble);
157 hdr.if_num = (uint8_t)(if_num & (uint8_t)k_ra8_c6link_hdr_nibble);
158 hdr.len = len;
159 hdr.offset = (uint16_t)k_ra8_c6link_header_bytes;
160 hdr.seq_num = (uint16_t)k_ra8_c6link_hdr_seq;
161
162 ra8_c6link_frame_clear(tx, span);
163 (void)memcpy(tx, &hdr, sizeof hdr);
164 hdr.checksum = compute_checksum(tx, span);
165 (void)memcpy(tx, &hdr, sizeof hdr);
166}

226RA8_PRIV ra8_c6link_frame_class_t ra8_c6link_priv_frame_classify(uint8_t* rx,
227 ra8_c6link_rx_view_t* view)
228{
229 if ((rx == nullptr) || (view == nullptr)) {
230 return k_ra8_c6link_frame_malformed;
231 }
232 struct esp_payload_header hdr = {};
233 (void)memcpy(&hdr, rx, sizeof hdr);
234
235 if (hdr.len == 0U) {
236 return k_ra8_c6link_frame_idle;
237 }
238 if ((hdr.offset != (uint16_t)k_ra8_c6link_header_bytes) ||
239 (hdr.len > (uint16_t)k_ra8_c6link_max_payload)) {
240 return k_ra8_c6link_frame_malformed;
241 }
242 if (ra8_c6link_frame_sum(rx, (uint16_t)(hdr.offset + hdr.len)) != hdr.checksum) {
243 return k_ra8_c6link_frame_bad_checksum;
244 }
245
246 view->offset = hdr.offset;
247 view->len = hdr.len;
248 view->if_type = (uint8_t)hdr.if_type;
249 view->if_num = (uint8_t)hdr.if_num;
250 return k_ra8_c6link_frame_data;
251}
