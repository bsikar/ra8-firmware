155static uint8_t internal_brr(uint32_t pclk_hz, uint32_t baud)
156{
157 if ((baud == 0U) || (pclk_hz == 0U)) {
158 return 0U;
159 }
160 const uint32_t divisor = k_ra8_sci_brr_async_divisor * baud;
161 const uint32_t n = pclk_hz / divisor;
162 if (n == 0U) {
163 return 0U;
164 }
165 return (uint8_t)(n - 1U);
166}

189static uint32_t internal_ccr1(const ra8_sci_cfg_t* cfg)
190{
191 uint32_t ccr1 = (1U << k_ra8_sci_ccr1_bit_spb2dt) | (1U << k_ra8_sci_ccr1_bit_spb2io);
192 if (cfg->parity != k_ra8_sci_parity_none) {
193 ccr1 |= (1U << k_ra8_sci_ccr1_bit_pe);
194 if (cfg->parity == k_ra8_sci_parity_odd) {
195 ccr1 |= (1U << k_ra8_sci_ccr1_bit_pm);
196 }
197 }
198 return ccr1;

219static uint32_t internal_ccr3(const ra8_sci_cfg_t* cfg)
220{
221 /* LSBF = 1 (LSB-first) is the UART standard wire order. SCI_B's
222 * reset state is MSB-first; without this bit the host receives
223 * each byte bit-reversed (e.g. 'h' = 0x68 transmits as 0x16). FSP
224 * r_sci_b_uart sets LSBF unconditionally for async configs.
225 *
226 * BPEN = 1 (Synchronizer Bypass Enable) is required when the bus
227 * clock (PCLK) is also used as the operation clock (TCLK) -- which
228 * is what we're doing in async mode with the on-chip baud-rate
229 * generator on the synchronized clock. Without BPEN the SCI's
230 * shift state machine waits forever for an independent SCICLK
231 * edge that never arrives, and the chip looks alive at the
232 * register level (TDR latches, CCR0.TE=1) but never advances --
233 * CSR.TDRE and CSR.TEND stay 0 indefinitely. HUM Ch 38.2.8 p 2207
234 * "BPEN bit" is the authoritative source. */
235 uint32_t ccr3 = (1U << k_ra8_sci_ccr3_bit_lsbf) | (1U << k_ra8_sci_ccr3_bit_bpen);
236
237 /* MOD = 000 (Asynchronous) -- already 0. */
238
239 /* CHR[1:0]. 8-bit -> 10b, 7-bit -> 11b. */
240 if (cfg->data_bits == k_ra8_sci_data_7) {
241 ccr3 |= (k_ra8_sci_ccr3_chr_7bit << k_ra8_sci_ccr3_shift_chr);
242 } else {
243 ccr3 |= (k_ra8_sci_ccr3_chr_8bit << k_ra8_sci_ccr3_shift_chr);
244 }
245
246 /* STP -- 1 = 2 stop bits. */
247 if (cfg->stop_bits == k_ra8_sci_stop_2) {
248 ccr3 |= (1U << k_ra8_sci_ccr3_bit_stp);
249 }
250
251 return ccr3;
252}

312static ra8_err_t internal_wait_tx_end(volatile r_sci_regs_t* reg)
313{
314 /* HUM Ch 38.2.17 "CSR : Common Status Register", p 2225 -- TEND (bit 30) goes
315 * high when both the data register and the shift register are empty. The
316 * ra8_fake_mmio host fault seam drives this real poll on the unit-test build, so
317 * the success and timeout legs run on host (T1-01) rather than short-circuit. */
318 const uint32_t mask = (1U << k_ra8_sci_csr_bit_tend);
319 return ra8_hw_wait_flag_set32(&reg->CSR, mask, k_ra8_hw_budget_medium);

340static uint32_t internal_ccr2(const ra8_sci_cfg_t* cfg)
341{
342 const uint8_t brr = internal_brr(cfg->pclk_hz, cfg->baud);

376static void internal_program_ccr_bank(volatile r_sci_regs_t* reg, const ra8_sci_cfg_t* cfg)
377{
378 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 -- disable
379 * TX/RX/IE bits before reconfiguring CCR1..CCR4 and FCR. */
380 reg->CCR0 = 0U;
381
382 /* HUM Ch 38.2.11 "FCR : FIFO Control Register", p 2215 -- non-FIFO
383 * polling mode for the bring-up demo: TFRST/RFRST cleared, all
384 * trigger numbers reset to 0. */
385 reg->FCR = 0U;
386
387 /* HUM Ch 38.2.6 "CCR1 : Common Control Register 1", p 2185 */
388 reg->CCR1 = internal_ccr1(cfg);
389
390 /* HUM Ch 38.2.8 "CCR3 : Common Control Register 3", p 2203 -- mode
391 * + framing must be programmed before TE/RE go high. */
392 reg->CCR3 = internal_ccr3(cfg);
393
394 /* HUM Ch 38.2.7 "CCR2 : Common Control Register 2", p 2189 -- BRR
395 * derived from cfg->pclk_hz and cfg->baud. */
396 reg->CCR2 = internal_ccr2(cfg);
397
398 /* HUM Ch 38.2.9 "CCR4 : Common Control Register 4", p 2210 -- no
399 * sample / transmit timing adjustment for async UART. */
400 reg->CCR4 = 0U;
401
402 /* HUM Ch 38.2.24 "CFCLR : Common Flag Clear Register", p 2238 +
403 * HUM Ch 38.2.26 "FFCLR : FIFO Flag Clear Register", p 2239 -- drop
404 * any latches inherited from a previous boot before TX/RX go live.
405 * Mirrors FSP r_sci_b_uart.c. */
406 internal_clear_csr_flags(reg);
407}

409ra8_err_t ra8_sci_init(uint8_t channel, const ra8_sci_cfg_t* cfg)
410{
411 RA8_CHECK_NULL_PTR(cfg, s_tag, "sci_init: cfg");
412 volatile r_sci_regs_t* reg = internal_reg(channel);
413 if (reg == nullptr) {
414 return k_ra8_err_invalid_arg;
415 }
416
417 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B", p 445 */
418 const ra8_err_t mst_err = ra8_mstp_enable(s_mstp_table[channel]);
419 if (mst_err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
420 /* GCOVR_EXCL_START */
421 ra8_log_error_val(s_tag, "sci_init: mstp enable failed", (uint32_t)mst_err);
422 return k_ra8_err_hw_init_failed;
423 /* GCOVR_EXCL_STOP */
424 }
425
426 internal_program_ccr_bank(reg, cfg);
427
428 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 -- enable
429 * transmitter and receiver. Interrupt-enable bits are toggled
430 * separately by ra8_sci_attach_{rx,tx}_handler. */
431 reg->CCR0 = (1U << k_ra8_sci_ccr0_bit_te) | (1U << k_ra8_sci_ccr0_bit_re);
432
433 s_sci_state[channel].initialized = true;
434 s_sci_state[channel].tx_buf = nullptr;
435 s_sci_state[channel].tx_len = 0U;
436 s_sci_state[channel].tx_idx = 0U;
437 s_sci_state[channel].rx_buf = nullptr;
438 s_sci_state[channel].rx_len = 0U;
439 s_sci_state[channel].rx_idx = 0U;
440 ra8_log_info_val(s_tag, "sci_init channel", (uint32_t)channel);
441 return k_ra8_ok;
442}

444ra8_err_t ra8_sci_deinit(uint8_t channel)
445{
446 volatile r_sci_regs_t* reg = internal_reg(channel);
447 if (reg == nullptr) {
448 return k_ra8_err_invalid_arg;
449 }
450
451 /* Drop CCR0 and tear the async descriptor down atomically w.r.t. the
452 * TXI/RXI ISR. Without the mask, a pending interrupt can still observe a
453 * non-zero tx_len/rx_len after CCR0 is cleared and then dereference the
454 * buffer pointer this path is nulling -- a NULL deref in interrupt
455 * context. Mirrors the ra8_sci_abort teardown. */
456 ra8_register_guard_t guard;
457 ra8_register_guard_enter(&guard);
458 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 */
459 reg->CCR0 = 0U;
460 s_sci_state[channel].rx_fn = nullptr;
461 s_sci_state[channel].rx_ctx = nullptr;
462 s_sci_state[channel].tx_fn = nullptr;
463 s_sci_state[channel].tx_ctx = nullptr;
464 s_sci_state[channel].initialized = false;
465 s_sci_state[channel].tx_buf = nullptr;
466 s_sci_state[channel].tx_len = 0U;
467 s_sci_state[channel].tx_idx = 0U;
468 s_sci_state[channel].rx_buf = nullptr;
469 s_sci_state[channel].rx_len = 0U;
470 s_sci_state[channel].rx_idx = 0U;
471 ra8_register_guard_exit(&guard);
472 return ra8_mstp_disable(s_mstp_table[channel]);
473}

477ra8_err_t ra8_sci_putc_polling(uint8_t channel, uint8_t byte)
478{
479 volatile r_sci_regs_t* reg = internal_reg(channel);
480 if (reg == nullptr) {
481 return k_ra8_err_invalid_arg;
482 }
483 /* HUM Ch 38.2.17 "CSR : Common Status Register", p 2225 -- spin
484 * until TDRE = 1 (transmit data register empty). */
485 const uint32_t mask = (1U << k_ra8_sci_csr_bit_tdre);
486 const ra8_err_t werr = ra8_hw_wait_flag_set32(&reg->CSR, mask, k_ra8_hw_budget_medium);
487 if (werr != k_ra8_ok) {
488 return werr;
489 }
490 /* HUM Ch 38.2.3 "TDR : Transmit Data Register", p 2181 -- write to
491 * TDAT[7:0] (low 8 bits of TDR) launches one frame in non-FIFO
492 * 8-bit async mode. */
493 reg->TDR = (uint32_t)byte;
494 return k_ra8_ok;
495}

517ra8_err_t ra8_sci_write_polling(uint8_t channel, const uint8_t* data, uint32_t len)
518{
519 if ((data == nullptr) && (len != 0U)) {
520 return k_ra8_err_null_ptr;
521 }
522 volatile r_sci_regs_t* reg = internal_reg(channel);
523 if (reg == nullptr) {
524 return k_ra8_err_invalid_arg;
525 }
526 for (uint32_t i = 0U; i < len; ++i) {
527 const ra8_err_t err = ra8_sci_putc_polling(channel, data[i]);
528 if (err != k_ra8_ok) {
529 return err;
530 }
531 }
532 /* HUM Ch 38.2.17 "CSR : Common Status Register", p 2225 -- TDRE
533 * goes high as soon as TDR latches into the shifter, but the byte
534 * may still be on the wire. Wait for TEND so the call only returns
535 * after the last frame is fully transmitted. Mirrors FSP
536 * r_sci_b_uart.c (`R_SCI_B_UART_Close` blocks on TEND for the
537 * same reason before dropping TE). */
538 if (len != 0U) {
539 return internal_wait_tx_end(reg);
540 }
541 return k_ra8_ok;
542}

559ra8_err_t ra8_sci_attach_rx_handler(uint8_t channel, ra8_sci_rx_fn_t fn, void* ctx)
560{
561 volatile r_sci_regs_t* reg = internal_reg(channel);
562 if (reg == nullptr) {
563 return k_ra8_err_invalid_arg;
564 }
565 /* The RXI ISR reads rx_fn/rx_ctx and read-modify-writes CCR0; mask it
566 * while we publish the handler and toggle RIE so neither a torn callback
567 * pointer nor a lost CCR0 update is observable. */
568 const uint32_t rie = (1U << k_ra8_sci_ccr0_bit_rie);
569 ra8_register_guard_t guard;
570 ra8_register_guard_enter(&guard);
571 s_sci_state[channel].rx_fn = fn;
572 s_sci_state[channel].rx_ctx = ctx;
573 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 -- toggle
574 * RIE (bit 16). */
575 if (fn != nullptr) {
576 reg->CCR0 = reg->CCR0 | rie;
577 } else {
578 reg->CCR0 = reg->CCR0 & ~rie;
579 }
580 ra8_register_guard_exit(&guard);
581 return k_ra8_ok;
582}

622 mask |= k_ra8_sci_err_overrun;

625 mask |= k_ra8_sci_err_framing;

627 if ((csr & (1U << k_ra8_sci_csr_bit_per)) != 0U) {
628 mask |= k_ra8_sci_err_parity;
629 }
630 *out_mask = mask;
631 return k_ra8_ok;

649ra8_err_t ra8_sci_set_baud(uint8_t channel, uint32_t baud, uint32_t pclk_hz)
650{
651 volatile r_sci_regs_t* reg = internal_reg(channel);
652 if (reg == nullptr) {
653 return k_ra8_err_invalid_arg;
654 }
655 if (baud == 0U) {
656 return k_ra8_err_invalid_arg;
657 }
658 const uint8_t brr = internal_brr(pclk_hz, baud);
659 /* Guard the CCR2 read-modify-write against any SCI ISR that stores to
660 * this channel's control registers: an interrupt landing between the
661 * CCR2 read and the write-back would otherwise drop the freshly merged
662 * BRR field (lost update). The mask is a no-op when uncontended, so the
663 * final CCR2 value is byte-identical to the unguarded path. */
664 ra8_register_guard_t guard;
665 ra8_register_guard_enter(&guard);
666 /* HUM Ch 38.2.7 "CCR2 : Common Control Register 2", p 2189 -- BRR
667 * lives in CCR2[15:8]; preserve the rest of CCR2. */
668 uint32_t v = reg->CCR2;
669 v &= ~k_ra8_sci_ccr2_mask_brr_field;
670 v |= ((uint32_t)brr << k_ra8_sci_ccr2_shift_brr);
671 reg->CCR2 = v;
672 ra8_register_guard_exit(&guard);
673 return k_ra8_ok;
674}

697/* ---- Async byte-stream TX / RX (FSP Read/Write parity) --------------- */

710typedef enum : uint16_t {
711 k_ra8_sci_baud_brr_max = 255U,
712 k_ra8_sci_baud_cks_max = 3U,
713 k_ra8_sci_baud_div_step = 4U,
714 k_ra8_sci_baud_n0_divisor = 32U,
715} ra8_sci_baud_calc_const_t;
716
717ra8_err_t
718ra8_sci_baud_calculate(uint32_t baud, uint32_t pclk_hz, uint16_t* brr_out, uint8_t* clk_div_out)
719{
720 RA8_CHECK_NULL_PTR(brr_out, s_tag, "baud_calc: brr_out");
721 RA8_CHECK_NULL_PTR(clk_div_out, s_tag, "baud_calc: clk_div_out");
722 if ((baud == 0U) || (pclk_hz == 0U)) {
723 return k_ra8_err_invalid_arg;
724 }
725
726 /* HUM Ch 38.2.7 "CCR2 : Common Control Register 2", p 2189 -- walk
727 * CKS = 0..3 and pick the smallest divider that yields a BRR <= 255.
728 * Mirrors the FSP `R_SCI_B_UART_BaudCalculate` outer loop
729 * (r_sci_b_uart.c) but without the bit-rate-modulation pass
730 * since the project always programs BRME=0. */
731 uint32_t divisor = (uint32_t)k_ra8_sci_baud_n0_divisor;
732 for (uint8_t n = 0U; n <= (uint8_t)k_ra8_sci_baud_cks_max; ++n) {
733 const uint32_t denom = divisor * baud;
734 if (denom == 0U) { /* GCOVR_EXCL_BR_LINE -- baud * divisor cannot
735 overflow to zero for legal inputs. */
736 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
737 }
738 const uint32_t quotient = pclk_hz / denom;
739 if (quotient > 0U) {
740 const uint32_t candidate = quotient - 1U;
741 if (candidate <= (uint32_t)k_ra8_sci_baud_brr_max) {
742 *brr_out = (uint16_t)candidate;
743 *clk_div_out = n;
744 return k_ra8_ok;
745 }
746 }
747 divisor *= (uint32_t)k_ra8_sci_baud_div_step;
748 }
749 return k_ra8_err_invalid_arg;
750}

752ra8_err_t ra8_sci_write(uint8_t channel, const uint8_t* data, uint32_t len)
753{
754 if ((data == nullptr) && (len != 0U)) {
755 return k_ra8_err_null_ptr;
756 }
757 volatile r_sci_regs_t* reg = internal_reg(channel);
758 if (reg == nullptr) {
759 return k_ra8_err_invalid_arg;
760 }
761 if (!s_sci_state[channel].initialized) {
762 return k_ra8_err_invalid_arg;
763 }
764 if (s_sci_state[channel].tx_len != 0U) {
765 return k_ra8_err_busy;
766 }
767 if (len == 0U) {
768 return k_ra8_ok;
769 }
770 /* Publish the async TX descriptor and arm TIE atomically w.r.t. the
771 * TXI ISR: it tests tx_len then dereferences tx_buf, so a torn publish
772 * (or a CCR0 RMW racing the ISR's TIE clear) must not be observable. */
773 ra8_register_guard_t guard;
774 ra8_register_guard_enter(&guard);
775 s_sci_state[channel].tx_buf = data;
776 s_sci_state[channel].tx_len = len;
777 s_sci_state[channel].tx_idx = 0U;
778 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 -- arm
779 * TIE so the next TDRE event fires the dispatcher. Mirrors FSP
780 * r_sci_b_uart.c which sets TE | TIE in a single store. */
781 reg->CCR0 = reg->CCR0 | (1U << k_ra8_sci_ccr0_bit_tie);
782 ra8_register_guard_exit(&guard);
783 return k_ra8_ok;
784}

786ra8_err_t ra8_sci_read(uint8_t channel, uint8_t* buf, uint32_t len)
787{
788 if ((buf == nullptr) && (len != 0U)) {
789 return k_ra8_err_null_ptr;
790 }
791 volatile r_sci_regs_t* reg = internal_reg(channel);
792 if (reg == nullptr) {
793 return k_ra8_err_invalid_arg;
794 }
795 if (!s_sci_state[channel].initialized) {
796 return k_ra8_err_invalid_arg;
797 }
798 if (s_sci_state[channel].rx_len != 0U) {
799 return k_ra8_err_busy;
800 }
801 if (len == 0U) {
802 return k_ra8_ok;
803 }
804 /* Publish the async RX descriptor and arm RIE atomically w.r.t. the
805 * RXI ISR (it tests rx_len then writes through rx_buf). */
806 ra8_register_guard_t guard;
807 ra8_register_guard_enter(&guard);
808 s_sci_state[channel].rx_buf = buf;
809 s_sci_state[channel].rx_len = len;
810 s_sci_state[channel].rx_idx = 0U;
811 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 -- arm
812 * RIE so the next RDRF event fires the dispatcher. Mirrors FSP
813 * r_sci_b_uart.c which stashes p_rx_dest / rx_dest_bytes for
814 * use by `rxi_isr`. */
815 reg->CCR0 = reg->CCR0 | (1U << k_ra8_sci_ccr0_bit_rie);
816 ra8_register_guard_exit(&guard);
817 return k_ra8_ok;
818}

820ra8_err_t ra8_sci_abort(uint8_t channel, ra8_sci_dir_t direction)
821{
822 volatile r_sci_regs_t* reg = internal_reg(channel);
823 if (reg == nullptr) {
824 return k_ra8_err_invalid_arg;
825 }
826 if ((direction != k_ra8_sci_dir_tx) && (direction != k_ra8_sci_dir_rx) &&
827 (direction != k_ra8_sci_dir_both)) {
828 return k_ra8_err_invalid_arg;
829 }
830 /* Disarm and tear down the async descriptor atomically w.r.t. the
831 * TXI/RXI ISR. Without the mask, the ISR can latch tx_len > 0, take an
832 * interrupt while this path nulls tx_buf, then dereference the stale
833 * NULL -- a use-after-free / NULL deref in interrupt context.
834 * HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 */
835 ra8_register_guard_t guard;
836 ra8_register_guard_enter(&guard);
837 if ((direction & k_ra8_sci_dir_tx) != 0U) {
838 const uint32_t tie_teie = (1U << k_ra8_sci_ccr0_bit_tie) | (1U << k_ra8_sci_ccr0_bit_teie);
839 reg->CCR0 = reg->CCR0 & ~tie_teie;
840 s_sci_state[channel].tx_buf = nullptr;
841 s_sci_state[channel].tx_len = 0U;
842 s_sci_state[channel].tx_idx = 0U;
843 }
844 if ((direction & k_ra8_sci_dir_rx) != 0U) {
845 const uint32_t rie = (1U << k_ra8_sci_ccr0_bit_rie);
846 reg->CCR0 = reg->CCR0 & ~rie;
847 s_sci_state[channel].rx_buf = nullptr;
848 s_sci_state[channel].rx_len = 0U;
849 s_sci_state[channel].rx_idx = 0U;
850 }
851 ra8_register_guard_exit(&guard);
852 return k_ra8_ok;
853}

855ra8_err_t ra8_sci_read_stop(uint8_t channel, uint32_t* remaining)
856{
857 RA8_CHECK_NULL_PTR(remaining, s_tag, "read_stop: remaining");
858 volatile r_sci_regs_t* reg = internal_reg(channel);
859 if (reg == nullptr) {
860 return k_ra8_err_invalid_arg;
861 }
862 /* Mirror FSP `R_SCI_B_UART_ReadStop` r_sci_b_uart.c: stash the
863 * pre-stop count, zero state, then disarm RIE. */
864 /* Snapshot the residual count, null the descriptor, and disarm RIE
865 * atomically w.r.t. the RXI ISR so the reported remaining matches the
866 * state we tear down (and the ISR cannot deref a half-nulled rx_buf). */
867 const uint32_t rie = (1U << k_ra8_sci_ccr0_bit_rie);
868 ra8_register_guard_t guard;
869 ra8_register_guard_enter(&guard);
870 const uint32_t pending = (s_sci_state[channel].rx_len > s_sci_state[channel].rx_idx)
871 ? (s_sci_state[channel].rx_len - s_sci_state[channel].rx_idx)
872 : 0U;
873 *remaining = pending;
874 s_sci_state[channel].rx_buf = nullptr;
875 s_sci_state[channel].rx_len = 0U;
876 s_sci_state[channel].rx_idx = 0U;
877 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 */
878 reg->CCR0 = reg->CCR0 & ~rie;
879 ra8_register_guard_exit(&guard);
880 return k_ra8_ok;
881}

884{

886 if (reg == nullptr) {
887 return k_ra8_err_invalid_arg;

900}
