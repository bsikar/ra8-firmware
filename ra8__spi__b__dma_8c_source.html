112static uint32_t internal_round_up_to_cache_line(uint32_t bytes)
113{
114 if (bytes == 0U) {
115 return 0U;
116 }
117 const uint32_t line = ra8_cache_dcache_line_bytes();
118 if (line == 0U) { /* GCOVR_EXCL_BR_LINE */
119 return bytes; /* GCOVR_EXCL_LINE */
120 }
121 return (bytes + (line - 1U)) & ~(line - 1U);
122}

148static void internal_spi_dma_rx_complete(void* ctx)
149{
150 ra8_spi_dma_rx_ctx_t* rxc = (ra8_spi_dma_rx_ctx_t*)ctx;
151 if (rxc == nullptr) { /* GCOVR_EXCL_BR_LINE */
152 return; /* GCOVR_EXCL_LINE */
153 }
154 ra8_cache_dcache_invalidate_by_addr(rxc->rx_buf, internal_round_up_to_cache_line(rxc->rx_len));
155 if (rxc->user_on_complete != nullptr) {
156 rxc->user_on_complete(rxc->user_ctx);
157 }
158}

160ra8_err_t ra8_spi_write_dma(uint8_t channel,
161 const uint8_t* data,
162 uint16_t len,
163 ra8_dma_complete_fn_t on_complete,
164 void* ctx,
165 uint8_t* out_dma_channel)
166{
167 RA8_CHECK_NULL_PTR(data, s_tag, "spi_write_dma: data");
168 RA8_CHECK_NULL_PTR(out_dma_channel, s_tag, "spi_write_dma: out_dma_channel");
169 if ((channel >= k_ra8_spi_b_channel_count) || (len == 0U)) {
170 return k_ra8_err_invalid_arg;
171 }
172 volatile r_spi_regs_t* reg = ra8_spi(channel);
173 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
174 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
175 }
176 /* Cache coherency: the SPI DMA reads CPU-written 'data' straight from
177 * memory, so flush any dirty lines covering it before the engine starts.
178 * No-op when the D-cache is disabled. */
179 ra8_cache_dcache_clean_by_addr(data, internal_round_up_to_cache_line(len));
180 /* Byte-wide DMA writes target the low byte of SPDR; wider-frame
181 * streaming is a future wave. */
182 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
183 ra8_dma_request_t req = {};
184 req.src_addr = (uintptr_t)data;
185 req.dst_addr = (uintptr_t)&reg->SPDR;
186 req.count = len;
187 req.width = k_ra8_dmac_width_byte;
188 req.src_inc = true;
189 req.dst_inc = false;
190 req.trigger = (ra8_elc_event_t)0;
191 req.on_complete = on_complete;
192 req.ctx = ctx;
193 return ra8_dma_request(&req, out_dma_channel);
194}

196ra8_err_t ra8_spi_read_dma(
197 uint8_t channel,
198 uint8_t*
199 out_buf, // NOLINT(readability-non-const-parameter) -- written by the DMAC engine via dst_addr, never through the pointer.
200 uint16_t len,
201 ra8_dma_complete_fn_t on_complete,
202 void* ctx,
203 uint8_t* out_dma_channel)
204{
205 RA8_CHECK_NULL_PTR(out_buf, s_tag, "spi_read_dma: out_buf");
206 RA8_CHECK_NULL_PTR(out_dma_channel, s_tag, "spi_read_dma: out_dma_channel");
207 if ((channel >= k_ra8_spi_b_channel_count) || (len == 0U)) {
208 return k_ra8_err_invalid_arg;
209 }
210 volatile r_spi_regs_t* reg = ra8_spi(channel);
211 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE */
212 return k_ra8_err_invalid_arg; /* GCOVR_EXCL_LINE */
213 }
214 /* Cache coherency: the RX DMA writes out_buf straight to memory. Wrap the
215 * caller's completion so the CPU invalidates those lines once the transfer
216 * is known-complete -- before it (or the caller's callback) reads them.
217 * The wrapper is the only known-complete hook in this asynchronous path
218 * (ra8_dma_request returns immediately; completion fires from the DMAC
219 * transfer-end ISR). No-op when the D-cache is disabled. */
220 ra8_spi_dma_rx_ctx_t* rxc = &s_spi_dma_rx_ctx[channel];
221 rxc->rx_buf = out_buf;
222 rxc->rx_len = len;
223 rxc->user_on_complete = on_complete;
224 rxc->user_ctx = ctx;
225
226 /* HUM Ch 43.2.2 "SPDR : SPI Data Register" p 2881 */
227 ra8_dma_request_t req = {};
228 req.src_addr = (uintptr_t)&reg->SPDR;
229 req.dst_addr = (uintptr_t)out_buf;
230 req.count = len;
231 req.width = k_ra8_dmac_width_byte;
232 req.src_inc = false;
233 req.dst_inc = true;
234 req.trigger = (ra8_elc_event_t)0;
235 req.on_complete = internal_spi_dma_rx_complete;
236 req.ctx = rxc;
237 return ra8_dma_request(&req, out_dma_channel);
238}
