376static void eth_mpsm_exec(uint64_t mpsm_off, uint32_t value)
377{
378 const uint32_t pop =
379 (value >> (uint32_t)k_ra8_rmac_shift_mpsm_pop) & (uint32_t)k_ra8_rmac_mask_mpsm_op;
380 const uint32_t pra =
381 (value >> (uint32_t)k_ra8_rmac_shift_mpsm_pra) & (uint32_t)k_ra8_rmac_mask_mpsm_phy_reg;
382 const uint32_t prd =
383 (value >> (uint32_t)k_ra8_rmac_shift_mpsm_prd) & (uint32_t)k_ra8_rmac_mask_mpsm_data;
384 const bool c45 = (value & (uint32_t)k_eth_mpsm_mff) != 0U;
385 uint32_t out = value & ~(uint32_t)k_ra8_rmac_mpsm_psme; /* PSME self-clears. */
386 if (!c45 && (pop == (uint32_t)k_ra8_rmac_mdio_op_c22_read)) {
387 out = ((uint32_t)eth_phy_read(pra) & (uint32_t)k_ra8_rmac_mask_mpsm_data)
388 << (uint32_t)k_ra8_rmac_shift_mpsm_prd;
389 } else if (!c45 && (pop == (uint32_t)k_ra8_rmac_mdio_op_c22_write)) {
390 eth_phy_write(pra, (uint16_t)prd);
391 }
392 eth_shadow_write(mpsm_off, k_eth_reg32_bytes, out);
393}

419static void eth_tx_kick_queue(uc_engine* uc, uint32_t queue)
420{
421 if (s_eth.linkfix_base == 0U) {
422 return;
423 }
424 uint8_t lf[k_eth_desc_bytes];
425 const uint32_t lf_addr = s_eth.linkfix_base + (queue * (uint32_t)k_eth_desc_bytes);
426 eth_desc_read(uc, lf_addr, lf);
427 const uint32_t chain = eth_desc_ptr(lf);
428 if (chain == 0U) {
429 return;
430 }
431 uint8_t head[k_eth_desc_bytes];
432 eth_desc_read(uc, chain, head);
433 if (eth_desc_dt(head) != (uint32_t)k_ra8_gwdcc_dt_fsingle) {
434 return; /* multi-fragment / empty head is unmodelled here -- honest no-op. */
435 }
436 const uint32_t len = eth_desc_ds(head);
437 const uint32_t buf = eth_desc_ptr(head);
438 if ((len == 0U) || (len > (uint32_t)k_eth_frame_max) || (buf == 0U)) {
439 return;
440 }
441 uint8_t frame[k_eth_frame_max];
442 if (uc_mem_read(uc, (uint64_t)buf, frame, (size_t)len) != UC_ERR_OK) {
443 return;
444 }
445 board_net_on_tx(frame, len);
446 eth_desc_set_dt(uc, chain, (uint32_t)k_ra8_gwdcc_dt_fempty);
447 s_eth.tx_frames++;
448 char ln[k_eth_console_cap];
449 (void)snprintf(ln, sizeof(ln), "ETH tx q%u %uB", (unsigned)queue, (unsigned)len);
450 board_console_push(k_board_console_ch_net, ln);
451}

483static bool eth_rx_find_slot(uc_engine* uc, uint32_t chain, uint32_t* out_slot)
484{
485 uint32_t cur = chain;
486 for (uint32_t i = 0U; i < (uint32_t)k_eth_ring_walk_max; ++i) {
487 uint8_t d8[k_eth_desc_bytes];
488 eth_desc_read(uc, cur, d8);
489 const uint32_t dt = eth_desc_dt(d8);
490 if (dt == (uint32_t)k_ra8_gwdcc_dt_fempty) {
491 *out_slot = cur;
492 return true;
493 }
494 if ((dt == (uint32_t)k_ra8_gwdcc_dt_link) || (dt == (uint32_t)k_ra8_gwdcc_dt_linkfix)) {
495 const uint32_t next = eth_desc_ptr(d8);
496 if ((next == 0U) || (next == cur)) {
497 return false;
498 }
499 cur = next;
500 } else {
501 cur += (uint32_t)k_eth_desc_bytes;
502 }
503 }
504 return false;
505}

526static void eth_rx_drain_peer(uc_engine* uc, uint32_t chain)
527{
528 for (uint32_t n = 0U; n < (uint32_t)k_eth_rx_inject_max; ++n) {
529 uint32_t slot = 0U;
530 if (!eth_rx_find_slot(uc, chain, &slot)) {
531 return; /* ring full -- backpressure, peer frame stays queued. */
532 }
533 uint8_t d8[k_eth_desc_bytes];
534 eth_desc_read(uc, slot, d8);
535 const uint32_t cap = eth_desc_ds(d8);
536 const uint32_t buf = eth_desc_ptr(d8);
537 if ((buf == 0U) || (cap == 0U)) {
538 return;
539 }
540 const uint32_t want = (cap < (uint32_t)k_eth_frame_max) ? cap : (uint32_t)k_eth_frame_max;
541 uint8_t frame[k_eth_frame_max];
542 const uint32_t got = board_net_poll_rx(frame, want);
543 if (got == 0U) {
544 return; /* peer has nothing more. */
545 }
546 (void)uc_mem_write(uc, (uint64_t)buf, frame, (size_t)got);
547 eth_desc_set_ds(uc, slot, got);
548 eth_desc_set_dt(uc, slot, (uint32_t)k_ra8_gwdcc_dt_fsingle);
549 s_eth.rx_frames++;
550 }

554static void eth_tick(uc_engine* uc)
555{
556 if ((s_eth.linkfix_base == 0U) || (s_eth.rx_queue == (uint32_t)k_eth_queue_none)) {
557 return;
558 }
559 const uint64_t gwmc_woff =
560 ((uint64_t)k_ra8_gwca0_base_addr - (uint64_t)k_eth_win_base) + (uint64_t)k_ra8_gwca_off_gwmc;
561 const uint32_t opc = eth_shadow_u32(gwmc_woff) & (uint32_t)k_ra8_gwmc_opc_mask;
562 if (opc != (uint32_t)k_ra8_gwmc_opc_operation) {
563 return; /* rings only live in OPERATION mode. */
564 }
565 uint8_t lf[k_eth_desc_bytes];
566 const uint32_t lf_addr = s_eth.linkfix_base + (s_eth.rx_queue * (uint32_t)k_eth_desc_bytes);
567 eth_desc_read(uc, lf_addr, lf);
568 const uint32_t chain = eth_desc_ptr(lf);
569 if (chain != 0U) {
570 eth_rx_drain_peer(uc, chain);
571 }
572}

593static uint64_t eth_read(uc_engine* uc, uint64_t addr, unsigned size)
594{
595 (void)uc;
596 const uint64_t off = addr - (uint64_t)k_eth_win_base;
597 uint32_t queue = 0U;
598 if (eth_is_etha_reg(addr, (uint64_t)k_ra8_etha_off_eams)) {
599 /* EAMS.OPS mirrors the commanded EAMC.OPC (the mode machine converges). */
600 /* HUM Ch 32.3.1.2 "EAMS : Mode Status Register" p 1631 */
601 const uint64_t eamc = off - ((uint64_t)k_ra8_etha_off_eams - (uint64_t)k_ra8_etha_off_eamc);
602 return eth_shadow_u32(eamc) & (uint32_t)k_ra8_etha_mask_ops;
603 }
604 if (addr == (uint64_t)k_ra8_gwca0_base_addr + (uint64_t)k_ra8_gwca_off_gwms) {
605 /* GWMS.OPS mirrors the commanded GWMC.OPC. */
606 /* HUM Ch 34.3.2 "GWMS : Mode Status Register" p 1798 */
607 const uint64_t gwmc = off - ((uint64_t)k_ra8_gwca_off_gwms - (uint64_t)k_ra8_gwca_off_gwmc);
608 return eth_shadow_u32(gwmc) & (uint32_t)k_ra8_gwmc_opc_mask;
609 }
610 if (addr == (uint64_t)k_ra8_gwca0_base_addr + (uint64_t)k_ra8_gwca_off_gwarirm) {
611 /* GWARIRM.ARR (AXI-init done) follows the ARIOG request. */
612 /* HUM Ch 34 "Ethernet CPU Agent (GWCA)" p 1787 */
613 const uint32_t v = eth_shadow_u32(off);
614 return ((v & (uint32_t)k_eth_gwarirm_ariog) != 0U) ? (v | (uint32_t)k_eth_gwarirm_arr) : v;
615 }
616 if (addr == (uint64_t)k_ra8_coma_base_addr + (uint64_t)k_ra8_coma_off_cabpirm) {
617 /* CABPIRM.BPR self-sets after the BPIOG kick (buffer-pool init completes
618 * ~512 clocks later); modelling it lets the board bring-up's non-emulator
619 * BPR poll converge. */
620 /* HUM Ch 31.3.2.7 "CABPIRM" p 1599 */
621 const uint32_t v = eth_shadow_u32(off);
622 return ((v & (uint32_t)k_ra8_coma_cabpirm_bpiog) != 0U) ? (v | (uint32_t)k_ra8_coma_cabpirm_bpr)
623 : v;
624 }
625 if (eth_is_gwdcc(addr, &queue)) {
626 /* GWDCC.BALR self-clears after the reload completes. */
627 /* HUM Ch 34.3 "GWDCCi" p 1811 */
628 return eth_shadow_u32(off) & ~(uint32_t)k_ra8_gwdcc_balr;
629 }
630 if (eth_is_rmac_reg(addr, (uint64_t)k_ra8_rmac_off_mtgfce)) {
631 return s_eth.tx_frames;
632 }
633 if (eth_is_rmac_reg(addr, (uint64_t)k_eth_rmac_off_mrfc) ||
634 eth_is_rmac_reg(addr, (uint64_t)k_ra8_rmac_off_mrgfce)) {
635 return s_eth.rx_frames;
636 }
637 return eth_shadow_read(off, size);
638}

655static void eth_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
656{
657 const uint64_t off = addr - (uint64_t)k_eth_win_base;
658 uint32_t queue = 0U;
659 if (addr == (uint64_t)k_ra8_gwca0_base_addr + (uint64_t)k_ra8_gwca_off_gwtrc0) {
660 eth_shadow_write(off, size, 0U); /* request register clears when consumed. */
661 eth_gwtrc_kick(uc, (uint32_t)value, 0U);
662 return;
663 }
664 if (addr == (uint64_t)k_ra8_gwca0_base_addr + (uint64_t)k_ra8_gwca_off_gwtrc1) {
665 eth_shadow_write(off, size, 0U);
666 eth_gwtrc_kick(uc, (uint32_t)value, (uint32_t)k_eth_phy_reg_count);
667 return;
668 }
669 if (addr == (uint64_t)k_ra8_gwca0_base_addr + (uint64_t)k_ra8_gwca_off_gwdcbac1) {
670 /* Capture the descriptor-chain base so the DMA can walk the rings. */
671 /* HUM Ch 34 "Ethernet CPU Agent (GWCA)" p 1787 */
672 s_eth.linkfix_base = (uint32_t)value;
673 eth_shadow_write(off, size, value);
674 return;
675 }
676 if (eth_is_rmac_reg(addr, (uint64_t)k_ra8_rmac_off_mpsm) &&
677 ((value & (uint32_t)k_ra8_rmac_mpsm_psme) != 0U)) {
678 eth_mpsm_exec(off, (uint32_t)value);
679 return;
680 }
681 eth_shadow_write(off, size, value);
682 if (eth_is_gwdcc(addr, &queue) && ((value & (uint32_t)k_ra8_gwdcc_dqt) == 0U)) {
683 s_eth.rx_queue = queue; /* GWDCC.DQT == 0 marks the reception queue. */
684 }
685}
