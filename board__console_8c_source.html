123static void ring_push(console_ring_t* ring, const char* line)
124{
125 if (ring == nullptr) {
126 return;
127 }
128 if (ring->head >= (uint32_t)k_board_console_ring_depth) {
129 ring->head = 0U; /* defensive: keep the slot index in range */
130 }
131 slot_copy(ring->lines[ring->head], line);
132 ring->head = (ring->head + 1U) % (uint32_t)k_board_console_ring_depth;
133 if (ring->count < (uint32_t)k_board_console_ring_depth) {
134 ring->count++;
135 }
136 ring->total++;
137}

187void board_console_push(board_console_ch_t ch, const char* line)
188{
189 if ((uint32_t)ch == (uint32_t)k_board_console_ch_all) {
190 return; /* ALL is a mirror only; never pushed to directly. */
191 }
192 if ((uint32_t)ch >= (uint32_t)k_board_console_ch_count) {
193 return;
194 }
195 if (line == nullptr) {
196 return;
197 }
198 ring_push(&s_rings[(uint32_t)ch], line);
199 char all_line[k_board_console_line_cap];
200 format_all_line(ch, line, all_line);
201 ring_push(&s_rings[(uint32_t)k_board_console_ch_all], all_line);
202}

220const char* board_console_line(board_console_ch_t ch, uint32_t back)
221{
222 if ((uint32_t)ch >= (uint32_t)k_board_console_ch_count) {
223 return nullptr;
224 }
225 const console_ring_t* ring = &s_rings[(uint32_t)ch];
226 if (back >= ring->count) {
227 return nullptr; /* older than the retained history */
228 }
229 /* back = 0 is the newest line; head points one past the newest. */
230 const uint32_t idx = (ring->head + (uint32_t)k_board_console_ring_depth - 1U - back) %
231 (uint32_t)k_board_console_ring_depth;
232 return ring->lines[idx];
233}
