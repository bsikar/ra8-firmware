176static void sdhi_demo_setup_or_halt(void)
177{
178 uint32_t cpuclk0_hz = 0U;
179 if (ra8_cgc_init() != k_ra8_ok) {
180 sdhi_demo_panic_halt();
181 }
182 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
183 sdhi_demo_panic_halt();
184 }
185 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
186 sdhi_demo_panic_halt();
187 }
188 if (ra8_board_uart_console_init((uint32_t)k_sdhi_demo_uart_baud) != k_ra8_ok) {
189 sdhi_demo_panic_halt();
190 }
191 if (ra8_board_sdhi_pins_init() != k_ra8_ok) {
192 sdhi_demo_panic_halt();
193 }
194}

253static void sdhi_demo_fill_payload(void)
254{
255 uint32_t state = (uint32_t)k_sdhi_demo_prng_seed;
256 for (uint32_t i = 0U; i < (uint32_t)k_sdhi_demo_payload_bytes; i++) {
257 state = (state * (uint32_t)k_sdhi_demo_prng_mul) + (uint32_t)k_sdhi_demo_prng_add;
258 s_payload[i] = (uint8_t)((state >> k_sdhi_demo_prng_byte_shift) & k_sdhi_demo_byte_mask);
259 }
260}

321[[nodiscard]] static ra8_err_t sdhi_demo_write_payload(const char* path)

356[[nodiscard]] static ra8_err_t sdhi_demo_read_and_verify(const char* path)
357{
358 ra8_fs_file_t* rf = nullptr;
359 RA8_RETURN_ON_ERROR(ra8_io_vfs_open(path, k_ra8_fs_mode_read, &rf), s_tag, "open r");
360 memset(s_readback, 0, sizeof(s_readback));
361 uint32_t got = 0U;
362 ra8_err_t err = ra8_fs_read(rf, s_readback, (uint32_t)k_sdhi_demo_payload_bytes, &got);
363 (void)ra8_fs_close(rf);
364 if (err != k_ra8_ok) {
365 return err;
366 }
367 if (got != (uint32_t)k_sdhi_demo_payload_bytes) {
368 return k_ra8_err_invalid_size;
369 }
370 if (memcmp(s_payload, s_readback, (size_t)k_sdhi_demo_payload_bytes) != 0) {
371 return k_ra8_err_checksum_mismatch;
372 }
373 return k_ra8_ok;
374}
