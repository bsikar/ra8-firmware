107static void bkup_reset(void)
108{
109 s_bkup = (bkup_ctrl_t){};
110 /* HUM Ch 12.2.6 p 504 "Value after reset": VBAE = 1, so the access-enable
111 * bit is already armed out of reset (confirmed by a J-Link read of a live
112 * board). Modelling it as 0 would make a forgotten VBAE arm look fatal in
113 * the emulator when on silicon it is harmless. */
114 s_bkup.vbtber = (uint8_t)k_bkup_vbtber_reset;
115 /* s_bkup_vbtbkr is intentionally NOT cleared: it is the reset-retained
116 * domain that the survival demo depends on across a --reboot. */
117}

120static uint64_t bkup_read(uc_engine* uc, uint64_t addr, unsigned size)
121{
122 (void)uc;
123 const uint64_t off = addr - (uint64_t)k_bkup_base;
124 if (off == (uint64_t)k_bkup_off_vbtber) {
125 return s_bkup.vbtber;
126 }
127 if (off >= (uint64_t)k_bkup_off_vbtbkr0) {
128 const uint64_t idx = off - (uint64_t)k_bkup_off_vbtbkr0;
129 uint64_t v = 0U;
130 for (unsigned i = 0U; (i < size) && ((idx + (uint64_t)i) < (uint64_t)k_bkup_reg_count); ++i) {
131 v |= (uint64_t)s_bkup_vbtbkr[idx + (uint64_t)i] << (8U * i);
132 }
133 return v;
134 }
135 return 0U;
136}

139static void bkup_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
140{
141 (void)uc;
142 const uint64_t off = addr - (uint64_t)k_bkup_base;
143 /* HUM Ch 13.1 Table 13.1 p 521 lists VBTBER and VBTBKRn (with the rest of
144 * the VBATT file) under PRC1. Every write below is discarded while that
145 * group is locked, with no fault and no flag -- the #131 silicon behaviour.
146 * Nested-if (not &&) keeps each decision single-condition. */
147 if (!board_prcr_group_unlocked((uint16_t)k_board_prcr_grp1_lpm)) {
148 if (off >= (uint64_t)k_bkup_off_vbtbkr0) {
149 s_bkup.dropped_prc1++;
150 }
151 return;
152 }
153 if (off == (uint64_t)k_bkup_off_vbtber) {
154 s_bkup.vbtber = (uint8_t)value;
155 return;
156 }
157 if (off >= (uint64_t)k_bkup_off_vbtbkr0) {
158 /* HUM Ch 12.2.6 "VBTBER : VBATT Backup Enable Register" p 504: "You must
159 * write 1 to VBAE before accessing VBTBKR." Drop the write while VBAE = 0
160 * so a cleared-VBAE bug surfaces here (rw=BAD) exactly as on silicon. */
161 if ((s_bkup.vbtber & (uint8_t)k_bkup_vbtber_mask_vbae) == 0U) {
162 s_bkup.dropped_vbae++;
163 return;
164 }
165 const uint64_t idx = off - (uint64_t)k_bkup_off_vbtbkr0;
166 for (unsigned i = 0U; (i < size) && ((idx + (uint64_t)i) < (uint64_t)k_bkup_reg_count); ++i) {
167 s_bkup_vbtbkr[idx + (uint64_t)i] = (uint8_t)(value >> (8U * i));
168 }
169 s_bkup.writes++;
170 }
171}

174static void bkup_report(void)
175{
176 if (s_bkup.dropped_prc1 != 0U) {
177 /* Loud: firmware wrote VBTBKRn without unlocking PRCR.PRC1 (#131). */
178 (void)fprintf(
179 stderr,
180 " VBATT-BKUP : VBTBKRn writes=%u dropped=%u (PRCR.PRC1 locked: unlock 0xA502)\n",
181 s_bkup.writes,
182 s_bkup.dropped_prc1);
183 return;
184 }
185 if (s_bkup.dropped_vbae != 0U) {
186 /* Loud: firmware touched VBTBKRn after clearing VBTBER.VBAE. */
187 (void)fprintf(stderr,
188 " VBATT-BKUP : VBTBKRn writes=%u dropped=%u (VBAE=0: enable VBTBER.VBAE)\n",
189 s_bkup.writes,
190 s_bkup.dropped_vbae);
191 return;
192 }
193 if (s_bkup.writes == 0U) {
194 return; /* Untouched: stay quiet. */
195 }
196 (void)fprintf(stderr, " VBATT-BKUP : VBTBKRn writes=%u (domain retained)\n", s_bkup.writes);
197}
