175static void c6_wifi_setup_or_halt(void)
176{
177 if (ra8_cgc_init() != k_ra8_ok) {
178 c6_wifi_panic_halt();
179 }
180 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &s_c6_wifi_cpuclk_hz) != k_ra8_ok) {

190 c6_wifi_panic_halt();

192 if (ra8_board_uart_console_init((uint32_t)k_c6_wifi_uart_baud) != k_ra8_ok) {

272static bool c6_wifi_phase_identity(const ra8_c6link_fw_version_t* fw_in)
273{
274 if (fw_in == nullptr) {
275 c6_wifi_puts("c6_wifi: identity record missing\r\n");
276 return false;
277 }
278 const ra8_c6link_fw_version_t fw = *fw_in;
279
280 c6_wifi_puts("c6_wifi: coprocessor fw=");
281 c6_wifi_put_u32(fw.major);
282 c6_wifi_puts(".");
283 c6_wifi_put_u32(fw.minor);
284 c6_wifi_puts(".");
285 c6_wifi_put_u32(fw.patch);
286 c6_wifi_puts(" chip_id=0x");
287 c6_wifi_put_hex(fw.chip_id, (uint8_t)k_c6_wifi_hex_byte);
288 c6_wifi_puts(" idf_target=");
289 c6_wifi_put_text(fw.target, (size_t)fw.target_len);
290 c6_wifi_puts(" expected=");
291 c6_wifi_put_u32((uint32_t)ESP_HOSTED_VERSION_MAJOR_1);
292 c6_wifi_puts(".");
293 c6_wifi_put_u32((uint32_t)ESP_HOSTED_VERSION_MINOR_1);
294 c6_wifi_puts(".");
295 c6_wifi_put_u32((uint32_t)ESP_HOSTED_VERSION_PATCH_1);
296 c6_wifi_puts("\r\n");
297
298 const bool matched = (fw.major == (uint32_t)ESP_HOSTED_VERSION_MAJOR_1) &&
299 (fw.minor == (uint32_t)ESP_HOSTED_VERSION_MINOR_1) &&
300 (fw.patch == (uint32_t)ESP_HOSTED_VERSION_PATCH_1) &&
301 (fw.chip_id == (uint32_t)k_c6_wifi_expect_chip);
302 if (!matched) {
303 c6_wifi_puts("c6_wifi: coprocessor identity is not the one this host was built against\r\n");
304 }
305 return matched;
306}

324static bool c6_wifi_phase_station(void)
325{
326 const ra8_err_t started = ra8_c6link_wifi_start(&s_c6_wifi_link);
327 if (started != k_ra8_ok) {
328 c6_wifi_report_fault("wifi_start", started);
329 return false;
330 }
331 c6_wifi_puts("c6_wifi: station started (WifiInit + SetWifiMode + WifiStart accepted)\r\n");
332
333 ra8_c6link_stats_t drain = {};
334 (void)ra8_c6link_poll(&s_c6_wifi_link, (uint16_t)k_c6_wifi_drain_polls, &drain);
335 c6_wifi_print_stats("drain", &drain);
336
337 ra8_c6link_mac_t mac = {};
338 const ra8_err_t readen = ra8_c6link_wifi_mac(&s_c6_wifi_link, &mac);
339 if (readen != k_ra8_ok) {
340 c6_wifi_report_fault("wifi_mac", readen);
341 (void)ra8_c6link_wifi_stop(&s_c6_wifi_link);
342 return false;
343 }
344 c6_wifi_puts("c6_wifi: station mac=");
345 c6_wifi_put_mac(&mac);
346 c6_wifi_puts("\r\n");
347
348 const ra8_err_t stopped = ra8_c6link_wifi_stop(&s_c6_wifi_link);
349 if (stopped != k_ra8_ok) {
350 c6_wifi_report_fault("wifi_stop", stopped);
351 return false;
352 }
353 c6_wifi_puts("c6_wifi: station stopped (WifiStop + WifiDeinit accepted)\r\n");
354 return true;
355}

430static bool c6_wifi_phase_ready(ra8_c6link_fw_version_t* out)
431{
432 if (out == nullptr) {
433 c6_wifi_puts("c6_wifi: FAIL no identity record to fill\r\n");
434 return false;
435 }
436
437 const ra8_err_t opened = c6_wifi_open_link();
438 c6_wifi_puts("c6_wifi: link_open=");
439 c6_wifi_puts(ra8_err_to_str(opened));
440 c6_wifi_puts("\r\n");
441 if (opened != k_ra8_ok) {
442 c6_wifi_puts("c6_wifi: FAIL could not open the link\r\n");
443 return false;
444 }
445
446 const ra8_err_t ready =
447 ra8_c6link_await_ready(&s_c6_wifi_link, (uint16_t)k_ra8_c6link_announce_transfers, out);
448 c6_wifi_puts("c6_wifi: await_ready=");
449 c6_wifi_puts(ra8_err_to_str(ready));
450 c6_wifi_puts("\r\n");
451 if (ready != k_ra8_ok) {
452 c6_wifi_report_fault("await_ready", ready);
453 c6_wifi_puts("c6_wifi: FAIL the coprocessor did not answer the readiness probe\r\n");
454 return false;
455 }
456 return true;
457}

473static void c6_wifi_worker_entry(ULONG thread_input)
474{
475 (void)thread_input;
476
477 c6_wifi_puts("c6_wifi: port_init=");
478 c6_wifi_puts(ra8_err_to_str(s_c6_wifi_init_err));
479 c6_wifi_puts("\r\n");
480 if (s_c6_wifi_init_err != k_ra8_ok) {
481 c6_wifi_puts("c6_wifi: FAIL port init failed -- no transaction attempted\r\n");
482 c6_wifi_heartbeat(false);
483 return;
484 }
485
486 ra8_delay_ms((uint32_t)k_c6_wifi_boot_wait_ms);
487
488 ra8_c6link_fw_version_t fw = {};
489 if (!c6_wifi_phase_ready(&fw)) {
490 c6_wifi_heartbeat(false);
491 return;
492 }
493
494 const bool identity = c6_wifi_phase_identity(&fw);
495 const bool station = identity && c6_wifi_phase_station();
496
497 c6_wifi_puts("c6_wifi: events_seen=");
498 c6_wifi_put_u32(s_c6_wifi_events);
499 c6_wifi_puts("\r\n");
500 if (station) {
501 c6_wifi_puts("c6_wifi: PASS ra8_c6link drove the coprocessor station up and read its "
502 "address\r\n");
503 } else {
504 c6_wifi_puts("c6_wifi: FAIL station bring-up through ra8_c6link did not complete\r\n");
505 }
506 ra8_esp_hosted_mem_dump("c6_wifi_link");
507 c6_wifi_heartbeat(station);
508}
