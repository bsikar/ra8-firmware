123static void iu_print_uint(uint32_t value)
124{
125 uint8_t buf[k_iu_dec_ten];
126 uint32_t n = 0U;
127 if (value == 0U) {
128 buf[n] = '0';
129 n++;
130 }
131 while ((value > 0U) && (n < (uint32_t)k_iu_dec_ten)) {
132 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_iu_dec_ten));
133 n++;
134 value /= (uint32_t)k_iu_dec_ten;
135 }
136 for (uint32_t i = 0U; i < n; i++) {
137 iu_print(&buf[n - 1U - i], 1U);
138 }
139}

151static bool iu_tap(int32_t px, int32_t py, bool want_hit, uint16_t want_act, uint32_t* hits)
152{
153 const uint16_t count = (uint16_t)(sizeof(k_iu_targets) / sizeof(k_iu_targets[0]));
154 uint16_t action = 0U;
155 bool hit = false;
156 if (ra8_ui_hit_test(k_iu_targets, count, px, py, &action, &hit) != k_ra8_ok) {
157 return false;
158 }
159 if (hit) {
160 (*hits)++;
161 }
162 if (hit != want_hit) {
163 return false;
164 }
165 return want_hit ? (action == want_act) : true;
166}

173static bool iu_nav_check(void)
174{
175 ra8_ui_nav_t nav = {};
176 uint16_t top = 0U;
177 if (ra8_ui_nav_init(&nav, (uint16_t)k_iu_screen_library) != k_ra8_ok) {
178 return false;
179 }
180 if ((ra8_ui_nav_top(&nav, &top) != k_ra8_ok) || (top != (uint16_t)k_iu_screen_library)) {
181 return false;
182 }
183 /* Open a book: push the reading screen. */
184 if (ra8_ui_nav_push(&nav, (uint16_t)k_iu_screen_reading) != k_ra8_ok) {
185 return false;
186 }
187 if ((ra8_ui_nav_top(&nav, &top) != k_ra8_ok) || (top != (uint16_t)k_iu_screen_reading)) {
188 return false;
189 }
190 /* Back: pop reveals (and returns) the library beneath. */
191 uint16_t new_top = 0U;
192 if ((ra8_ui_nav_pop(&nav, &new_top) != k_ra8_ok) || (new_top != (uint16_t)k_iu_screen_library)) {
193 return false;
194 }
195 /* The root is never popped: a pop at depth 1 must fail and leave the root. */
196 if (ra8_ui_nav_pop(&nav, &new_top) == k_ra8_ok) {
197 return false;
198 }
199 return (ra8_ui_nav_top(&nav, &top) == k_ra8_ok) && (top == (uint16_t)k_iu_screen_library);
200}

232int32_t main(void)
233{
234 iu_setup_or_halt();
235 ra8_isr_globals_enable();
236 iu_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
237
238 uint32_t taps = 0U;
239 uint32_t hits = 0U;
240 bool ok = true;
241
242 /* Tap each target's centre -> a hit on its action id. */
243 const uint16_t ntargets = (uint16_t)(sizeof(k_iu_targets) / sizeof(k_iu_targets[0]));
244 for (uint16_t t = 0U; t < ntargets; ++t) {
245 const ra8_ui_rect_t* r = &k_iu_targets[t].rect;
246 const int32_t cx = r->x + (r->w / (int32_t)k_iu_tap_half);
247 const int32_t cy = r->y + (r->h / (int32_t)k_iu_tap_half);
248 ok = iu_tap(cx, cy, true, k_iu_targets[t].action_id, &hits) && ok;
249 taps++;
250 }
251 /* Off-target taps -> misses (the column gutter, and far off-screen). */
252 ok = iu_tap((int32_t)k_iu_miss_gutter_x, (int32_t)k_iu_miss_gutter_y, false, 0U, &hits) && ok;
253 taps++;
254 ok = iu_tap((int32_t)k_iu_miss_off_x, (int32_t)k_iu_miss_off_y, false, 0U, &hits) && ok;
255 taps++;
256
257 const bool nav_ok = iu_nav_check();
258 ok = ok && nav_ok && (hits == (uint32_t)ntargets) &&
259 (taps == (uint32_t)(ntargets + (uint16_t)k_iu_miss_count));
260
261 iu_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
262 iu_print_uint(taps);
263 iu_print(k_msg_hits, (uint32_t)sizeof(k_msg_hits) - 1U);
264 iu_print_uint(hits);
265 iu_print(k_msg_nav, (uint32_t)sizeof(k_msg_nav) - 1U);
266 iu_print_uint(nav_ok ? 1U : 0U);
267 if (ok) {
268 iu_print(k_msg_pass, (uint32_t)sizeof(k_msg_pass) - 1U);
269 } else {
270 iu_print(k_msg_failr, (uint32_t)sizeof(k_msg_failr) - 1U);
271 }
272
273 while (1) {
274 __asm__ volatile("wfi");
275 }
276}
