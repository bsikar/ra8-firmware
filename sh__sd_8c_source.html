127static bool sh_sd_classify(const char* name, sh_book_fmt_t* out_fmt)
128{
129 if (sh_sd_has_ext(name, ".RBK")) {
130 *out_fmt = k_sh_fmt_rabook;
131 return true;
132 }
133 if (sh_sd_has_ext(name, ".EPB")) { /* .epub truncates to the 3-char 8.3 ext .EPB */
134 *out_fmt = k_sh_fmt_epub;
135 return true;
136 }
137 /* .cbz / .cbr / .cbt are already 3-char exts, so they keep their name on
138 * FAT 8.3 (no truncation, unlike .epub -> .EPB). Comics route through
139 * sh_comic.c; the ra8_comic magic detect picks the container. */
140 if (sh_sd_has_ext(name, ".CBZ")) {
141 *out_fmt = k_sh_fmt_cbz;
142 return true;
143 }
144 if (sh_sd_has_ext(name, ".CBR")) {
145 *out_fmt = k_sh_fmt_cbr;
146 return true;
147 }
148 if (sh_sd_has_ext(name, ".CBT")) {
149 *out_fmt = k_sh_fmt_cbt;
150 return true;
151 }
152 return false;
153}

168 e->fmt = fmt;

171 /* Placeholder until first open populates real title/author/cover (reading +

174 (void)strncpy(e->author, "SD card -- tap to load", sizeof e->author - 1U);

211ra8_err_t sh_sd_book_read(void* ctx, uint64_t offset, uint8_t* buf, uint32_t len)
212{
213 (void)ctx; /* the held file handle is module state; no per-call context */
214 RA8_CHECK_NULL_PTR(buf, s_sd_tag, "book_read: null buf");
215 if (s_book == nullptr) {
216 return k_ra8_err_invalid_state;
217 }
218 if (offset > (uint64_t)UINT32_MAX) {
219 return k_ra8_err_out_of_range;
220 }
221 const ra8_err_t err = ra8_fs_seek(s_book, (uint32_t)offset);
222 if (err != k_ra8_ok) {
223 return err;
224 }
225 uint32_t got = 0U;
226 const ra8_err_t rerr = ra8_fs_read(s_book, buf, len, &got);
227 if (rerr != k_ra8_ok) {
228 return rerr;
229 }
230 /* ra8_fs_seek clamps to the file size, so a read past EOF surfaces here as a
231 * short read rather than a seek error -- reject it as out-of-range. */
232 if (got != len) {
233 return k_ra8_err_out_of_range;
234 }
235 return k_ra8_ok;
236}

238size_t sh_sd_comic_read(void* ctx, uint64_t offset, void* buf, size_t len)
239{
240 (void)ctx; /* the held file handle is module state; no per-call context */
241 if ((buf == nullptr) || (s_book == nullptr)) {
242 return 0U;
243 }
244 if (offset > (uint64_t)UINT32_MAX) {
245 return 0U; /* EOF for any archive within the ra8_fs 4 GiB offset limit */
246 }
247 const uint32_t want = (len > (size_t)UINT32_MAX) ? UINT32_MAX : (uint32_t)len;
248 if (ra8_fs_seek(s_book, (uint32_t)offset) != k_ra8_ok) {
249 return 0U;
250 }
251 uint32_t got = 0U;
252 /* A read past EOF surfaces as a short read (ra8_fs_seek clamps to size); the
253 * comic backends treat a short read as end-of-file, so return the count. */
254 if (ra8_fs_read(s_book, buf, want, &got) != k_ra8_ok) {
255 return 0U;
256 }
257 return (size_t)got;
258}
