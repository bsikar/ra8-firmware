4"""check_no_gnu_attribute.py -- enforce the C23 [[...]] attribute syntax.
5
6Per CLAUDE.md, first-party code uses the standard C23 attribute-specifier
7form [[...]] (e.g. [[gnu::weak]], [[noreturn]], [[maybe_unused]],
8[[nodiscard]]) instead of the GNU __attribute__((...)) form.
9
10 void f(void) __attribute__((weak)); # REJECTED
11 [[gnu::weak]] void f(void); # OK
12
13Scope:
14 C / C++ sources (.c .h .cpp .hpp) under libs/, src/, tests/, examples/,
15 port/, tools/. Vendored SOUP (libs/third_party/*) and generated font
16 data (libs/ra8_fonts/*) are exempt, as are build trees.
17
18Allowed __attribute__ uses (no portable [[...]] spelling -- clang errors on
19the [[gnu::]] form as an unknown attribute while it silently ignores the GNU
20form, so these MUST stay __attribute__):
21 * interrupt
22 * cmse_nonsecure_entry
23 * cmse_nonsecure_call
24(with or without the __leading_trailing__ underscores).
25
26Other exemptions:
27 * Comment lines / prose mentioning __attribute__ (lines whose attribute
28 token sits inside a // or /* comment).
29 * Any line carrying `ATTR-OK: <reason>` (reason text required).
30
31Exit 0 when clean, 1 on any violation, 2 when the whole-tree sweep collapses
32below FILE_FLOOR (see the constant).
33"""
34

37import re

45ROOTS = ("libs", "src", "tests", "examples", "port", "tools")
46EXTS = (".c", ".h", ".cpp", ".hpp")
47EXEMPT_DIRS = ("/third_party/", "/ra8_fonts/")
48ALLOWED = {"interrupt", "cmse_nonsecure_entry", "cmse_nonsecure_call"}
49
50ATTR_RE = re.compile(r"__attribute__\s*\‍(\‍(")
51WAIVER_RE = re.compile(r"ATTR-OK:\s*\S")
52
53# Length of a bare `____` wrapper (two leading + two trailing underscores); a
54# dunder name must exceed it to carry a body worth stripping.
55_MIN_DUNDER_LEN = 4
56
57# A tree this size cannot legitimately collapse to a handful of files. If the
58# whole-tree sweep returns less than this, something broke (a bad cwd -- ROOTS
59# are relative, so this walks nothing when run from elsewhere -- or a renamed
60# root) and reporting "clean" would be a lie. Measured 2026-07-28: 2125
61# first-party C/C++ files. Same trip-wire as check_ruff.py.
62FILE_FLOOR = 1700
63

65def _strip_us(name: str) -> str:
66 """Strip the surrounding double underscores from an attribute name.
67
68 ``__packed__`` and ``packed`` are the same attribute to GCC, so both
69 spellings must normalise to one before comparison or half of them would
70 slip past.
71 """
72 if len(name) > _MIN_DUNDER_LEN and name.startswith("__") and name.endswith("__"):
73 return name[2:-2]
74 return name

77def _attr_body(line: str, pos: int) -> str | None:
78 """Balanced-paren body of the __attribute__ starting at/after pos."""
79 p = line.find("((", pos)
80 if p < 0:
81 return None
82 p += 2
83 depth, q = 1, p
84 while q < len(line):
85 if line[q] == "(":
86 depth += 1
87 elif line[q] == ")":
88 depth -= 1
89 if depth == 0:
90 return line[p:q]
91 q += 1
92 return None

95def _is_comment_pos(line: str, pos: int) -> bool:
96 """True if column `pos` sits inside a // or /* comment on this line."""
97 stripped = line.lstrip()
98 if stripped.startswith(("*", "//", "/*")):
99 return True
100 before = line[:pos]
101 if "//" in before:
102 return True
103 # crude single-line /* ... */ check
104 return "/*" in before and "*/" not in before

107def discover() -> list[str]:
108 """Every first-party source file, for the whole-tree sweep."""
109 out: list[str] = []
110 for root in ROOTS:
111 base = Path(root)
112 if not base.is_dir():
113 continue
114 for ext in EXTS:
115 for f in base.rglob(f"*{ext}"):
116 s = str(f)
117 if is_build_output_path(s) or any(d in s for d in EXEMPT_DIRS):
118 continue
119 out.append(s)
120 return out

123def check_file(path: str) -> list[tuple[int, str]]:
124 """Report every legacy ``__attribute__((...))`` in one file.
125
126 The tree uses the C23 ``[[...]]`` form. Note the two spellings that CANNOT
127 migrate and are excluded rather than reported -- ``interrupt`` and the
128 CMSE ``cmse_nonsecure_entry`` -- neither of which has a standard-attribute
129 equivalent the toolchain accepts.
130 """
131 findings: list[tuple[int, str]] = []
132 try:
133 text = Path(path).read_text(encoding="utf-8")
134 except (OSError, UnicodeDecodeError):
135 return findings
136 if "__attribute__" not in text:
137 return findings
138 for i, line in enumerate(text.splitlines(), 1):
139 for m in ATTR_RE.finditer(line):
140 if _is_comment_pos(line, m.start()):
141 continue
142 if WAIVER_RE.search(line):
143 continue
144 body = _attr_body(line, m.start())
145 names = {
146 _strip_us(t.strip().split("(")[0].strip())
147 for t in (body.split(",") if body else [])
148 }
149 if names and names <= ALLOWED:
150 continue
151 findings.append((i, line.strip()[:100]))
152 return findings

155def main() -> int:
156 """Fail on GNU ``__attribute__`` syntax where the C23 form is available.
157
158 Position matters for the C23 form -- it must precede the declaration
159 specifiers rather than follow them -- which is why this is a migration
160 worth finishing rather than a style preference: the two spellings do not
161 sit in the same place, so mixed usage reads inconsistently.
162
163 FILE_FLOOR is enforced on the whole-tree sweep only, and exits 2 below it.
164 An explicit argv file list is a deliberately narrowed scope -- not a
165 collapsed one -- so it is exempt; the sweep is where a broken enumeration
166 would silently report the tree clean.
167
168 Returns 1 listing each occurrence, 0 when clean, 2 when the whole-tree
169 sweep enumerated too few files to trust.
170 """
171 args = [a for a in sys.argv[1:] if not a.startswith("-")]
172 whole_tree = not args
173 files = args or discover()
174 if whole_tree and len(files) < FILE_FLOOR:
175 print(
176 f"check_no_gnu_attribute.py: FATAL -- only {len(files)} first-party source "
177 f"file(s) in scope, floor is {FILE_FLOOR}. A collapsed sweep reports a "
178 "clean tree because it scanned nothing.",
179 file=sys.stderr,
180 )
181 return 2
182 total = 0
183 for path in sorted(set(files)):
184 if (
185 not path.endswith(EXTS)
186 or is_build_output_path(path)
187 or any(d in path for d in EXEMPT_DIRS)
188 ):
189 continue
190 for ln, snippet in check_file(path):
191 print(
192 f"{path}:{ln}: GNU __attribute__ -- use the C23 [[...]] form "
193 f"(e.g. [[gnu::weak]]); {snippet}"
194 )
195 total += 1
196 if total:
197 print(
198 f"\ncheck_no_gnu_attribute: {total} violation(s). Migrate to [[...]] "
199 f"(only interrupt / cmse_nonsecure_entry / cmse_nonsecure_call may stay "
200 f"__attribute__; add `ATTR-OK: <reason>` for a justified exception)."
201 )
202 return 1
203 print("check_no_gnu_attribute: clean -- all attributes use the C23 [[...]] form.")
204 return 0
