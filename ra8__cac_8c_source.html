90ra8_err_t ra8_cac_init(uint16_t upper, uint16_t lower)
91{
92 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C", p 446 */
93 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_cac);
94 RA8_RETURN_ON_ERROR(mst_err, s_tag, "cac_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
95
96 volatile r_cac_regs_t* reg = ra8_cac();
97 /* HUM Ch 10.2.1 "CACR0 : CAC Control Register 0" p 421 -- CFME must be
98 * cleared before CACR1/CACR2/limits are programmed (FSP r_cac.c
99 * r_cac_hw_configure() does the same, then waits on CACR0 readback). */
100 reg->CACR0 = 0U;
101 internal_cac_wait_cfme(reg, 0U);
102 /* HUM Ch 10.2.4 "CAICR" p 423 */ /* clear all stale W1C status flags. */
103 reg->CAICR = (uint8_t)(k_ra8_cac_status_mask_all << (uint8_t)k_ra8_cac_castr_to_caicr_shift);
104 /* HUM Ch 10.2.2 "CACR1 : CAC Control Register 1" p 421 */
105 reg->CACR1 = 0U;
106 /* HUM Ch 10.2.3 "CACR2 : CAC Control Register 2" p 422 */
107 reg->CACR2 = 0U;
108 /* HUM Ch 10.2.6 "CAULVR : CAC Upper Limit Value Register" p 425 */
109 reg->CAULVR = upper;
110 /* HUM Ch 10.2.7 "CALLVR : CAC Lower Limit Value Register" p 425 */
111 reg->CALLVR = lower;
112 ra8_log_info(s_tag, "cac_init");
113 return k_ra8_ok;
114}

116ra8_err_t ra8_cac_measure(uint16_t* out_count)
117{
118 RA8_CHECK_NULL_PTR(out_count, s_tag, "out_count must not be nullptr");
119 volatile r_cac_regs_t* reg = ra8_cac();
120
121 /* HUM Ch 10.2.1 "CACR0 : CAC Control Register 0" p 421 -- CFME=1 starts
122 * a measurement; CASTR.MENDF asserts when the window completes. FSP
123 * r_cac.c R_CAC_StartMeasurement() reads CACR0 back to confirm CFME
124 * propagation before relying on the result. */
125 const uint8_t cfme_set = (uint8_t)(1U << k_ra8_cacr0_cfme);
126 reg->CACR0 = cfme_set;
127 internal_cac_wait_cfme(reg, cfme_set);
128
129 for (uint32_t i = 0U; i < k_ra8_cac_poll_limit; i++) { /* GCOVR_EXCL_BR_LINE */
130 /* HUM Ch 10.2.5 "CASTR : CAC Status Register" p 424 */
131 if ((reg->CASTR & (uint8_t)(1U << k_ra8_castr_mendf)) != 0U) { /* GCOVR_EXCL_BR_LINE */
132 /* HUM Ch 10.2.4 "CACNTBR : CAC Counter Buffer Register" p 423 */
133 *out_count = reg->CACNTBR;
134 reg->CACR0 = 0U;
135 internal_cac_wait_cfme(reg, 0U);
136 return k_ra8_ok;
137 }
138 }
139 reg->CACR0 = 0U;
140 internal_cac_wait_cfme(reg, 0U);
141 return k_ra8_err_hw_timeout;
142}

182ra8_err_t ra8_cac_clear_status(uint8_t mask)
183{
184 volatile r_cac_regs_t* reg = ra8_cac();
185 /* HUM Ch 10.2.4 "CAICR : CAC Interrupt Control Register" p 423 -- the
186 * write-1-to-clear bits are FERRFCL=bit4, MENDFCL=bit5, OVFFCL=bit6,
187 * which sit `k_ra8_cac_castr_to_caicr_shift` (= 4) bits above the
188 * matching CASTR FERRF/MENDF/OVFF flags. Cross-verified against FSP
189 * r_cac.c CAC_PRV_CAICR_FERRFCL_OFFSET=4 / MENDFCL=5 / OVFFCL=6. */
190 const uint8_t castr_bits = (uint8_t)(mask & k_ra8_cac_status_mask_all);
191 const uint8_t caicr_clr = (uint8_t)(castr_bits << (uint8_t)k_ra8_cac_castr_to_caicr_shift);
192 reg->CAICR = caicr_clr;
193 return k_ra8_ok;
194}

203void ra8_cac_dispatch(void)
204{
205 volatile r_cac_regs_t* reg = ra8_cac();
206 const uint8_t mask = (uint8_t)(reg->CASTR & k_ra8_cac_status_mask_all);
207 /* HUM Ch 10.2.4 "CAICR" p 423 -- write 1 to FERRFCL/MENDFCL/OVFFCL
208 * (bits 4/5/6) to clear the matching CASTR flag, mirroring the FSP
209 * ISR pattern `R_CAC->CAICR |= CAC_PRV_CAICR_*FCL_MASK`. */
210 reg->CAICR = (uint8_t)(mask << (uint8_t)k_ra8_cac_castr_to_caicr_shift);
211 const ra8_cac_event_fn_t fn = s_cac_state.fn;
212 void* const ctx = s_cac_state.ctx;
213 if (fn != nullptr) {
214 fn(ctx, mask);
215 }
216}
