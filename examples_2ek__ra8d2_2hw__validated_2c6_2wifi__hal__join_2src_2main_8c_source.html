105static void wifi_hal_setup_or_halt(void)
106{
107 if (ra8_cgc_init() != k_ra8_ok) {
108 wifi_hal_panic_halt();
109 }
110 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &s_cpuclk_hz) != k_ra8_ok) {

117 wifi_hal_panic_halt();

120 wifi_hal_panic_halt();
121 }
122 if (ra8_board_uart_console_init((uint32_t)k_wifi_hal_uart_baud) != k_ra8_ok) {

142static ra8_err_t wifi_hal_make_cfg(ra8_wifi_cfg_t* out)
143{
144 ra8_wifi_c6link_cfg_t bcfg = {};
145 bcfg.link = &s_link;
146 bcfg.arena = s_arena;
147 bcfg.arena_bytes = (uint32_t)sizeof(s_arena);
148 bcfg.rx_cb = nx_ether_driver_c6_rx;
149 const ra8_err_t bound = ra8_esp_hosted_c6link_bind(&bcfg.transport);
150 if (bound != k_ra8_ok) {
151 return bound;
152 }
153 const ra8_err_t wired = ra8_wifi_c6link_setup(&s_c6, &bcfg, out);
154 if (wired != k_ra8_ok) {
155 return wired;
156 }
157 out->ip_bind = wifi_hal_ip_bind;
158 out->ip_ctx = &s_link;
159 return k_ra8_ok;
160}

205static bool wifi_hal_run(const ra8_net_credentials_t* credentials)
206{
207 ra8_wifi_cfg_t cfg = {};
208 const ra8_err_t made = wifi_hal_make_cfg(&cfg);
209 if (made != k_ra8_ok) {
210 wifi_hal_puts("wifi_hal: FAIL transport setup err=");
211 wifi_hal_puts(ra8_err_to_str(made));
212 wifi_hal_puts("\r\n");
213 return false;
214 }
215
216 const wifi_hal_run_cfg_t rcfg = {
217 .wifi_cfg = &cfg,
218 .wifi = &s_wifi,
219 .ssid = credentials->ssid,
220 .psk = credentials->psk,
221 .poll_budget = (uint32_t)k_wifi_hal_assoc_polls,
222 };
223 wifi_hal_result_t res = {};
224 const bool joined = wifi_hal_join_run(&rcfg, &res);
225 return wifi_hal_report(&res, joined);
226}

244static bool wifi_hal_report(const wifi_hal_result_t* res, bool joined)
245{
246 wifi_hal_puts("wifi_hal: init=");
247 wifi_hal_puts(ra8_err_to_str(res->init_err));
248 wifi_hal_puts("\r\n");
249 if (!res->init_ok) {
250 return false;
251 }
252
253 if (!res->associated) {
254 wifi_hal_puts("wifi_hal: FAIL association did not complete connect=");
255 wifi_hal_puts(ra8_err_to_str(res->connect_err));
256 wifi_hal_puts("\r\n");
257 return false;
258 }
259
260 wifi_hal_puts("wifi_hal: associated mac=");
261 wifi_hal_put_mac(&res->mac);
262 wifi_hal_puts(" chan=");
263 wifi_hal_put_u32((uint32_t)res->ap.channel);
264 wifi_hal_puts("\r\n");
265 if (!res->ip_bound) {
266 wifi_hal_puts("wifi_hal: FAIL dhcp did not bind err=");
267 wifi_hal_puts(ra8_err_to_str(res->ip_err));
268 wifi_hal_puts("\r\n");
269 return false;
270 }
271 wifi_hal_print_lease(&res->lease);
272 if (!joined) {
273 wifi_hal_puts("wifi_hal: FAIL join_run failed with every stage reported ok\r\n");
274 }
275 return joined;
276}

312static void wifi_hal_worker_entry(ULONG thread_input)
313{
314 (void)thread_input;
315
316 wifi_hal_puts("wifi_hal: port_init=");
317 wifi_hal_puts(ra8_err_to_str(s_init_err));
318 wifi_hal_puts("\r\n");
319 if (s_init_err != k_ra8_ok) {
320 wifi_hal_puts("wifi_hal: FAIL port init failed -- no transaction attempted\r\n");
321 wifi_hal_heartbeat(false);
322 return;
323 }
324
325 ra8_delay_ms((uint32_t)k_wifi_hal_boot_wait_ms);
326
327 ra8_net_credentials_t credentials = {};
328 const ra8_err_t provisioned = ra8_net_provision_receive(&s_provision_uart,
329 (uint32_t)k_ra8_net_provision_timeout_ms,
330 &credentials);
331 if (provisioned != k_ra8_ok) {
332 wifi_hal_puts("wifi_hal: FAIL runtime provisioning err=");
333 wifi_hal_puts(ra8_err_to_str(provisioned));
334 wifi_hal_puts("\r\n");
335 ra8_net_provision_clear(&credentials);
336 wifi_hal_puts(k_wifi_hal_fail_line);
337 wifi_hal_heartbeat(false);
338 return;
339 }
340
341 const bool ok = wifi_hal_run(&credentials);
342 ra8_net_provision_clear(&credentials);
343 wifi_hal_puts(ok ? k_wifi_hal_pass_line : k_wifi_hal_fail_line);
344 wifi_hal_heartbeat(ok);
345}

360void tx_application_define(void* first_unused_memory)
361{

383 }
384}
