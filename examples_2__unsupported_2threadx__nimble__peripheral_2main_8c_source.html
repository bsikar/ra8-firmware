167static void demo_clocks_or_halt(void)
168{

170
171 if (ra8_cgc_init() != k_ra8_ok) {
172 demo_panic_halt();
173 }
174 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {

178 demo_panic_halt();

181 demo_panic_halt();

184 demo_panic_halt();

232static void demo_tick_battery(void)
233{

253static void demo_thread_entry(ULONG arg)
254{
255 (void)arg;
256
257 demo_log("[nimble] starting NimBLE peripheral\r\n");
258 demo_ble_or_halt();
259 demo_log("[nimble] advertising as ");
260 demo_log(k_demo_local_name);
261 demo_log("\r\n");
262
263 while (1) {
264 /* Pump any inbound HCI events / ACL frames through the bridge. */
265 (void)ra8_ble_dispatch();
266 /* Sleep one full tick window before the next battery decrement. */
267 (void)tx_thread_sleep((ULONG)k_demo_tick_ms);
268 demo_tick_battery();
269 }
270}
