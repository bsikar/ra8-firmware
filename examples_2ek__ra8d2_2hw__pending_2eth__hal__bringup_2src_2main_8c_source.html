113static void ehb_write_u32_line(uint32_t val)
114{
115 uint8_t buf[k_ehb_dec_u32_max];
116 uint32_t n = 0U;
117 if (val == 0U) {
118 buf[n] = (uint8_t)'0';
119 n++;
120 }
121 while ((val > 0U) && (n < (uint32_t)k_ehb_dec_u32_max)) {
122 buf[n] = (uint8_t)((uint32_t)'0' + (val % (uint32_t)k_ehb_radix));
123 n++;
124 val /= (uint32_t)k_ehb_radix;
125 }
126 for (uint32_t i = 0U; i < n; i++) {
127 ehb_write(&buf[n - 1U - i], 1U);
128 }
129 ehb_write(k_ehb_msg_crlf, (uint32_t)sizeof(k_ehb_msg_crlf) - 1U);
130}

218[[nodiscard]] static ra8_err_t ehb_run_once(void)
219{
220 ra8_err_t rc = ra8_cgc_eswclk_init();
221 ehb_write(k_ehb_msg_eswclk, (uint32_t)sizeof(k_ehb_msg_eswclk) - 1U);
222 ehb_write_u32_line((uint32_t)rc);
223 if (rc != k_ra8_ok) {
224 return rc;
225 }
226
227 rc = ra8_mstp_enable(k_ra8_mstp_eswm);
228 ehb_write(k_ehb_msg_mstp, (uint32_t)sizeof(k_ehb_msg_mstp) - 1U);
229 ehb_write_u32_line((uint32_t)rc);
230 if (rc != k_ra8_ok) {
231 return rc;
232 }
233
234 rc = ra8_eth_coma_bringup();
235 ehb_write(k_ehb_msg_coma, (uint32_t)sizeof(k_ehb_msg_coma) - 1U);
236 ehb_write_u32_line((uint32_t)rc);
237 if (rc != k_ra8_ok) {
238 return rc;
239 }
240
241 rc = ra8_eth_rgmii_select(k_ra8_eth_mii_port_1);
242 ehb_write(k_ehb_msg_rgmii, (uint32_t)sizeof(k_ehb_msg_rgmii) - 1U);
243 ehb_write_u32_line((uint32_t)rc);
244 return rc;
245}
