154{

159 return k_ra8_err_invalid_arg;

164 return k_ra8_err_invalid_arg;

186static uint8_t internal_encode_cr(const ra8_sram_bank_cfg_t* cfg)
187{
188 uint8_t eccmod_field = k_ra8_sram_eccmod_disabled;
189 if (cfg->ecc_mode == k_ra8_sram_ecc_no_check) {
190 eccmod_field = k_ra8_sram_eccmod_no_check;
191 } else if (cfg->ecc_mode == k_ra8_sram_ecc_with_chk) {
192 eccmod_field = k_ra8_sram_eccmod_with_chk;
193 } else {
194 eccmod_field = k_ra8_sram_eccmod_disabled;
195 }
196
197 uint8_t value = eccmod_field;
198 if (cfg->on_error == k_ra8_sram_on_error_reset) {
199 value |= k_ra8_sram_cr_mask_oad;
200 }
201 if (cfg->enable_1bit_latch) {
202 value |= k_ra8_sram_cr_mask_e1stsen;
203 }
204 return value;
205}

225static void internal_write_cr_locked(uint8_t bank, uint8_t value)
226{
227 volatile r_sram_regs_t* regs = ra8_sram_regs();
228
229 /* HUM Ch 58.2.4 "SRAMPRCR_S : SRAM Protection Control Register
230 * for Secure", p 3530 -- write 0xA501 to enable PR=1. */
231 regs->SRAMPRCR_S = k_ra8_sram_prcr_unlock;
232
233 /* HUM Ch 58.2.7 "SRAMCRn : SRAM Control Register n For ECC RAM",
234 * p 3532 -- per-bank SRAMCR layout (OAD / ECCMOD / E1STSEN). */
235 *ra8_sram_cr_ptr(regs, bank) = value;
236
237 /* Re-lock with PR=0.
238 * HUM Ch 58.2.4 "SRAMPRCR_S", p 3530 */
239 regs->SRAMPRCR_S = k_ra8_sram_prcr_lock;

319static void internal_decode_esr(uint16_t raw, uint8_t* one_bit_mask, uint8_t* two_bit_mask)
320{
321 uint8_t one = 0U;
322 uint8_t two = 0U;
323 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
324 const uint16_t one_bit_pos = (uint16_t)((uint16_t)2U * (uint16_t)bank);
325 const uint16_t two_bit_pos = (uint16_t)(one_bit_pos + 1U);
326 if ((raw & (uint16_t)((uint16_t)1U << one_bit_pos)) != 0U) {
327 one |= (uint8_t)((uint8_t)1U << bank);
328 }
329 if ((raw & (uint16_t)((uint16_t)1U << two_bit_pos)) != 0U) {
330 two |= (uint8_t)((uint8_t)1U << bank);
331 }
332 }
333 *one_bit_mask = one;
334 *two_bit_mask = two;
335}

378static void internal_apply_security(const ra8_sram_security_cfg_t* sec)
379{
380 uint32_t sar = 0U;
381 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
382 if (sec->bank_ns[bank]) {
383 sar |= k_ra8_sram_sar_bit_sa0 << bank;
384 }
385 }
386 if (sec->wtsc_ns) {
387 sar |= k_ra8_sram_sar_bit_wtsa;
388 }
389
390 volatile r_sram_cpscu_regs_t* cpscu = ra8_sram_cpscu_regs();
391
392 /* HUM Ch 58.2.2 "SRAMSAR : SRAM Security Attribution Register",
393 * p 3528 -- per-bank register security + SRAMWTSC security. */
394 cpscu->SRAMSAR = sar;
395
396 /* HUM Ch 58.2.3 "SRAMESAR : SRAM ECC region Security Attribute
397 * Register", p 3529 -- ECC region NS bit. */
398 uint32_t esar = 0U;
399 if (sec->ecc_region_ns) {
400 esar = k_ra8_sram_esar_bit_esa;
401 }
402 cpscu->SRAMESAR = esar;
403
404 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
405 /* HUM Ch 58.2.1 "SRAMSABARn : SRAM Security Attribute Boundary
406 * Address Register", p 3527 -- boundary value, low 13 bits forced
407 * to zero (4 KB aligned). */
408 cpscu->SRAMSABAR[bank] = sec->boundary_offset[bank] & ~k_ra8_sram_sabar_align_mask;
409 }
410}

459static void internal_zero_init_with_no_check(uint8_t bank)
460{
461 /* Step 1: enable ECC encode without checking. */
462 internal_write_cr_locked(bank, k_ra8_sram_eccmod_no_check);

532static void internal_apply_per_bank(const ra8_sram_config_t* cfg)
533{
534 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
535 if (cfg->banks[bank].zero_init) {
536 internal_zero_init_with_no_check(bank);

539 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {

546[[nodiscard]] ra8_err_t ra8_sram_init(const ra8_sram_config_t* cfg)
547{
548 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
549
550 const ra8_err_t v_err = internal_validate_and_ungate(cfg);
551 RA8_RETURN_ON_ERROR(v_err, s_tag, "ra8_sram_init: validate/ungate"); /* GCOVR_EXCL_BR_LINE */
552
553 if (cfg->apply_security) {
554 internal_apply_security(&cfg->security);
555 }
556
557 /* SRAMWTSC is deliberately NOT touched here -- ra8_cgc_init owns it,
558 * derived from ICLK per HUM Ch 58.3.7 p 3540. Clearing it from a
559 * zero-initialised config is how a caller silently takes the memory
560 * system outside guaranteed operation (tracker #524). */
561
562 internal_apply_per_bank(cfg);
563
564 /* Clear any stale error flags. HUM 58.2.13 p 3536. */
565 volatile r_sram_regs_t* regs = ra8_sram_regs();
566 regs->SRAMESCLR = k_ra8_sram_err_all_mask;
567
568 s_initialized = true;
569 ra8_log_info(s_tag, "ra8_sram_init done");
570 return k_ra8_ok;
571}

573[[nodiscard]] ra8_err_t ra8_sram_deinit(void)
574{
575 /* Disable ECC + clear OAD on every bank before clock-gating, so a
576 * spurious bus error during teardown does not latch a fault. */
577 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
578 internal_write_cr_locked(bank, 0U);
579 internal_write_eccrgn_locked(bank, k_ra8_sram_eccrgn_off);
580 }
581 /* SRAMWTSC stays as ra8_cgc_init left it. Tearing down the ECC
582 * configuration says nothing about the clock, and the caller is still
583 * executing out of this SRAM: clearing WTEN here would leave every
584 * subsequent access outside the guarantee of HUM Ch 58.3.7 p 3540. */
585
586 /* HUM Ch 58.3.1 "Module Stop Function", p 3538 -- re-gate the
587 * clock for every bank (HUM 11.2.6 MSTPCRA p 443). */
588 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
589 (void)ra8_mstp_disable(s_sram_mstp_table[bank]);
590 }
591
592 s_sram_on_error = nullptr;
593 s_sram_on_error_ctx = nullptr;
594 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
595 s_sram_on_error_bank[bank] = nullptr;
596 s_sram_on_error_bank_ctx[bank] = nullptr;
597 }
598 s_initialized = false;
599 return k_ra8_ok;
600}

675[[nodiscard]] ra8_err_t ra8_sram_get_status(ra8_sram_status_t* out)
676{
677 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
678
679 volatile r_sram_regs_t* regs = ra8_sram_regs();
680
681 /* HUM Ch 58.2.12 "SRAMESR : SRAM Error Status Register For ECC
682 * RAM", p 3535 -- packed 1-bit/2-bit flags per bank. */
683 const uint16_t raw = regs->SRAMESR;
684 out->raw_esr = raw;
685 internal_decode_esr(raw, &out->one_bit_mask, &out->two_bit_mask);
686
687 /* HUM Ch 58.2.14 "SRAMEARnm : SRAM Error Address Register nm
688 * For ECC RAM", p 3537 -- m=0 is the 1-bit slot, m=1 is the
689 * 2-bit slot. Stored as an offset; this driver presents the
690 * absolute Secure address to callers. */
691 for (uint8_t bank = 0U; bank < k_ra8_sram_bank_count; ++bank) {
692 out->addr_1bit[bank] = internal_ear_to_abs_addr(regs->SRAMEAR[bank][0]);
693 out->addr_2bit[bank] = internal_ear_to_abs_addr(regs->SRAMEAR[bank][1]);
694 }
695 return k_ra8_ok;
696}

711{
712 if ((uint16_t)bank >= (uint16_t)k_ra8_sram_bank_count) {

725 * For ECC RAM" p 3536 */
726 regs->SRAMESCLR = mask;

767static void internal_self_test_inject(uint8_t bank, volatile uint64_t* data, bool inject_two_bit)
768{
769 internal_write_cr_locked(bank, k_ra8_sram_cr_self_test_phase_bypass);
770
771 uint8_t inject_mask = k_ra8_sram_self_test_flip_1bit;
772 if (inject_two_bit) {
773 inject_mask = k_ra8_sram_self_test_flip_2bit;

776 const uint64_t corrupted = syndrome ^ (uint64_t)inject_mask;
777 *data = corrupted;

782[[nodiscard]] ra8_err_t
783ra8_sram_self_test(uint8_t bank, uint32_t probe_offset, bool inject_two_bit, bool* out_caught)
784{
785 RA8_CHECK_NULL_PTR(out_caught, s_tag, "out_caught must not be nullptr");
786 if ((uint16_t)bank >= (uint16_t)k_ra8_sram_bank_count) {
787 return k_ra8_err_invalid_arg;
788 }
789 /* Probe must be 8-byte aligned and inside the bank (HUM 58.4.2
790 * p 3541 -- 8-byte ECC line size). */
791 if ((probe_offset & ((uint32_t)k_ra8_sram_ecc_word_bytes - 1U)) != 0U) {
792 return k_ra8_err_invalid_arg;
793 }
794 if (probe_offset >= ra8_sram_bank_size_bytes(bank)) {
795 return k_ra8_err_invalid_arg;
796 }
797
798 *out_caught = false;
799
800 volatile uint64_t* const bank_base = ra8_sram_bank_data_ptr(bank);
801 const uintptr_t data_addr = (uintptr_t)bank_base + (uintptr_t)probe_offset;
802 volatile uint64_t* const data = (volatile uint64_t*)data_addr;
803
804 /* Step 1: seed the line under ECC encode-only (HUM Ch 58.3.4 p 3539). */
805 internal_write_cr_locked(bank, k_ra8_sram_cr_self_test_phase_write);
806 *data = k_ra8_sram_zero_init_word;
807
808 /* Steps 2-3: bypass-read, inject, then arm verify. */
809 internal_self_test_inject(bank, data, inject_two_bit);
810
811 /* Read the line. On real silicon this triggers the ECC engine, which
812 * latches SRAMESR / SRAMEAR for the faulted slot. The RAM-backed host
813 * register file has no ECC engine, so on the unit-test build the read
814 * is inert and host tests stage SRAMESR before the call to drive both
815 * legs of the caught decision below. */
816 volatile uint64_t scratch = *data;
817 (void)scratch;
818
819 /* Step 4: confirm SRAMESR latched the expected flag. */
820 volatile r_sram_regs_t* check_regs = ra8_sram_regs();
821 const uint16_t esr = check_regs->SRAMESR;
822 uint16_t slot_bit = 0U;
823 if (inject_two_bit) {
824 slot_bit = 1U;
825 }
826 const uint16_t want_bit =
827 (uint16_t)((uint16_t)1U << (uint16_t)(((uint16_t)2U * (uint16_t)bank) + slot_bit));
828 *out_caught = ((esr & want_bit) != 0U);
829
830 return k_ra8_ok;
831}

833[[nodiscard]] ra8_err_t ra8_sram_get_bank_info(uint8_t bank, ra8_sram_bank_info_t* out)
834{
835 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");

843 out->ecc_size = ra8_sram_bank_ecc_size_bytes(bank);

845}
