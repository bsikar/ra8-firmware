141static uint32_t adc_diag_u32_to_dec(uint8_t* buf, uint32_t val)
142{
143 if (val == 0U) {
144 buf[0] = (uint8_t)'0';
145 return 1U;
146 }
147 uint8_t tmp[k_adc_diag_dec_max];
148 uint32_t n = 0U;
149 uint32_t v = val;
150 while (v != 0U) {
151 tmp[n] = (uint8_t)('0' + (uint8_t)(v % (uint32_t)k_adc_diag_radix));
152 v = v / (uint32_t)k_adc_diag_radix;
153 n++;
154 }
155 for (uint32_t i = 0U; i < n; i++) {
156 buf[i] = tmp[n - 1U - i];
157 }
158 return n;
159}

238static bool adc_diag_run_cycle(void)
239{
240 bool healthy = true;
241 for (uint32_t i = 0U; i < (uint32_t)k_adc_diag_num_modes; i++) {
242 uint16_t code = 0U;
243 bool pass = false;
244 const ra8_err_t err = ra8_adc_self_diagnose(k_adc_diag_modes[i], &code, &pass);
245 if (err != k_ra8_ok) {
246 healthy = false;
247 pass = false;
248 }
249 if (!pass) {
250 healthy = false;
251 }
252 adc_diag_emit_mode((uint8_t)(i + 1U), code, pass);
253 }
254
255 int32_t milli = 0;
256 const ra8_err_t temp_err = ra8_tsn_read_die_temp_milli_c(&milli);
257 if (temp_err != k_ra8_ok) {
258 healthy = false;
259 }
260 adc_diag_write(k_adc_diag_tsn_prefix, (uint32_t)(sizeof(k_adc_diag_tsn_prefix) - 1U));
261 adc_diag_write_i32(milli);
262 adc_diag_write(k_adc_diag_crlf, (uint32_t)(sizeof(k_adc_diag_crlf) - 1U));
263
264 uint16_t vref = 0U;
265 const ra8_err_t vref_err = ra8_adc_read_internal_channel(k_ra8_adc_chan_int_ref_volt, &vref);
266 if (vref_err != k_ra8_ok) {
267 healthy = false;
268 }
269 adc_diag_write(k_adc_diag_vref_prefix, (uint32_t)(sizeof(k_adc_diag_vref_prefix) - 1U));
270 adc_diag_write_u32((uint32_t)vref);
271 adc_diag_write(k_adc_diag_crlf, (uint32_t)(sizeof(k_adc_diag_crlf) - 1U));
272
273 return healthy;
274}

285static void adc_diag_setup_or_halt(void)

303 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {

329[[nodiscard]] static ra8_err_t adc_diag_arm(void)
330{
331 const ra8_err_t adc_err = ra8_adc_init();
332 if (adc_err != k_ra8_ok) {
333 return adc_err;
334 }
335 const ra8_tsn_config_t cfg = {
336 .high_ref_degc = k_ra8_tsn_cal_temp_high_125,
337 .low_ref_degc = k_ra8_tsn_cal_temp_low_n40,
338 .stab_us = (uint16_t)k_adc_diag_stab_us,
339 };
340 return ra8_tsn_init(&cfg);
341}

345int32_t main(void)
346{
347 adc_diag_setup_or_halt();
348 ra8_isr_globals_enable();
349
350 if (adc_diag_arm() != k_ra8_ok) {
351 adc_diag_panic_halt();
352 }
353
354 while (1) {
355 const bool healthy = adc_diag_run_cycle();
356 if (healthy) {
357 adc_diag_write(k_adc_diag_verdict_pass, (uint32_t)(sizeof(k_adc_diag_verdict_pass) - 1U));
358 } else {
359 adc_diag_write(k_adc_diag_verdict_fail, (uint32_t)(sizeof(k_adc_diag_verdict_fail) - 1U));
360 }
361 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
362 break;
363 }
364 ra8_delay_ms((uint32_t)k_adc_diag_period_ms);
365 }
366 adc_diag_panic_halt();
367 return 0;
368}
