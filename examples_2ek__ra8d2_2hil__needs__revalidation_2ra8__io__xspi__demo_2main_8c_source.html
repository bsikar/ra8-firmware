177static ra8_err_t demo_run(void)
178{
179 /* Route the 12 OCTA bus pins and strobe the IS25LX512M RESET_L BEFORE the
180 * controller comes up. ra8_xspi_init only programs CGC/MSTP and the OSPI
181 * controller registers -- without this the pins stay GPIO, no clock, CS or
182 * data ever reaches the flash, and the mount reads a floating bus and fails
183 * with Error=1281 (0x501). Silicon-confirmed; threadx_levelx_demo, which
184 * passes on the bench, uses the same helper in the same order. */
185 RA8_RETURN_ON_ERROR(ra8_board_xspi_pins_init(), s_tag, "xspi pins");
186 RA8_RETURN_ON_ERROR(ra8_xspi_init((uint8_t)k_demo_xspi_inst, k_ra8_xspi_lio_1s1s1s),
187 s_tag,
188 "xspi init");
189 RA8_RETURN_ON_ERROR(ra8_io_blockdev_xspi_init(&s_bd,
190 &s_xstate,
191 (uint8_t)k_demo_xspi_inst,
192 (uint32_t)k_demo_xspi_base,
193 (uint32_t)k_demo_disk_blocks,
194 false),
195 s_tag,
196 "blockdev init");
197 ra8_fs_mount_t* mnt = nullptr;
198 RA8_RETURN_ON_ERROR(ra8_io_roundtrip_mount(&s_bd, &s_params, &s_be, &mnt), s_tag, "mount");
199 return ra8_io_roundtrip_subdir_file(&s_params);
200}
