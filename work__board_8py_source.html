179 result: dict[str, object] = {}

233def _labels(value: object, context: str) -> tuple[str, ...]:
234 if value is None:
235 return ()
236 if not isinstance(value, list):
237 _fail_data(f"{context} must be a list or null")
238 kept: dict[str, str] = {}
239 for raw in value:
240 label = _strict_string(raw, f"{context} entry")
241 kept.setdefault(label.casefold(), label)

243 sorted(
244 kept.values(),
245 key=lambda label: (printable(label).casefold(), printable(label)),

261def _parse_project_fields(text: str) -> None:
262 """Reject lossy field-name mappings before item-list serializes values."""
263 root = _strict_object(_strict_json(text), "GitHub project fields response")
264 raw_fields = root.get("fields")
265 total = root.get("totalCount")
266 if not isinstance(raw_fields, list):
267 _fail_data("GitHub project fields must be a list")
268 if not isinstance(total, int) or isinstance(total, bool) or total != len(raw_fields):
269 _fail_data("GitHub project fields totalCount must equal fields length")
270 seen: dict[str, str] = {}
271 for raw in raw_fields:
272 field = _strict_object(raw, "GitHub project field")
273 raw_name = field.get("name")
274 name = _strict_string(raw_name, "GitHub project field name")
275 if name != raw_name:
276 _fail_data(f"GitHub project field name must not have surrounding whitespace: {name}")
277 folded = name.casefold()
278 if folded in seen:
279 _fail_data(
280 f"GitHub project fields collide case-insensitively: {seen[folded]} and {name}"
281 )
282 seen[folded] = name
283 missing = [name for name in REQUIRED_PROJECT_FIELDS if name.casefold() not in seen]
284 if missing:
285 _fail_data(f"GitHub project is missing required fields: {', '.join(missing)}")
286 for expected in CANONICAL_PROJECT_FIELDS:
287 actual = seen.get(expected.casefold())
288 if actual is not None and actual != expected:
289 _fail_data(f"GitHub project field {actual} must use canonical name {expected}")

292def _parse_repository_issue_labels(text: str) -> dict[int, tuple[str, ...]]:
293 """Read labels from every page of the repository Issues endpoint."""
294 pages = _strict_json(text)
295 if not isinstance(pages, list):
296 _fail_data("GitHub repository issues response must be a list of pages")
297 result: dict[int, tuple[str, ...]] = {}
298 for page in pages:
299 if not isinstance(page, list):
300 _fail_data("GitHub repository issues page must be a list")
301 for raw in page:
302 issue = _strict_object(raw, "GitHub open issue")
303 if "pull_request" in issue:
304 continue
305 number = _strict_positive_int(issue.get("number"), "GitHub repository issue number")
306 if number in result:
307 _fail_data(f"GitHub repository issues response contains duplicate issue #{number}")
308 result[number] = _detail_labels(issue.get("labels"))
309 return result

312def _classify_labels(
313 labels: tuple[str, ...], number: int
314) -> tuple[tuple[str, ...], tuple[str, ...], tuple[str, ...]]:
315 priority = tuple(
316 match.group(1).upper()
317 for label in labels
318 if (match := PRIORITY_LABEL_RE.fullmatch(label)) is not None
319 )
320 effort = tuple(
321 match.group(1).strip(" \t")
322 for label in labels
323 if (match := EFFORT_LABEL_RE.fullmatch(label)) is not None
324 )
325 epic = tuple(
326 match.group(1).strip(" \t")
327 for label in labels
328 if (match := EPIC_LABEL_RE.fullmatch(label)) is not None
329 )
330 if any(not value for value in (*effort, *epic)):
331 _fail_data(f"project item #{number} has an empty derived label")
332 return priority, effort, epic

335def _casefold_collision(item: dict[str, object]) -> None:
336 seen: dict[str, str] = {}
337 for key in item:
338 folded = key.casefold()
339 if folded in seen and seen[folded] != key:
340 _fail_data(f"project item keys collide case-insensitively: {seen[folded]} and {key}")
341 seen[folded] = key

344def _repository_name(value: object, schema: TrackerSchema, number: int) -> str:
345 """Return the configured identity for one exact GitHub repository spelling."""
346 raw = _strict_string(value, "project item repository")
347 allowed = (schema.repository, f"https://{schema.github_host}/{schema.repository}")

353def _parse_item(raw: object, schema: TrackerSchema) -> BoardItem:
354 item = _strict_object(raw, "project item")
355 _casefold_collision(item)
356 content = _strict_object(item.get("content"), "project item content")
357 _casefold_collision(content)
358 type_fields = tuple(
359 (context, obj["type"])
360 for context, obj in (("project item type", item), ("project item content type", content))
361 if "type" in obj
362 )
363 if not type_fields:
364 _fail_data("project item type must be Issue")
365 for context, value in type_fields:
366 if _strict_string(value, context).casefold() != "issue":
367 _fail_data("project item type must be Issue")
368 number = _strict_positive_int(content.get("number"), "project item number")
369 repository = _repository_name(content.get("repository"), schema, number)
370 if "repository" in item:
371 _repository_name(item["repository"], schema, number)
372 title_value = item.get("title")
373 if title_value is None or (isinstance(title_value, str) and not title_value.strip(" \t")):
374 title_value = content.get("title")
375 title = _strict_string(title_value, f"project item #{number} title")
376 url = _strict_string(content.get("url", item.get("url")), f"project item #{number} url")
377 labels = _labels(item.get("labels"), f"project item #{number} labels")
378 priority_labels, effort_labels, epic_labels = _classify_labels(labels, number)
379 return BoardItem(
380 number=number,
381 title=title,
382 url=url,
383 repository=repository,
384 content_type="Issue",
385 status=_optional_scalar(item, "status"),
386 track=_optional_scalar(item, "track"),
387 horizon=_optional_scalar(item, "horizon"),
388 project_priority=_optional_scalar(item, "priority"),
389 priority_labels=priority_labels,
390 effort_labels=effort_labels,
391 epic_labels=epic_labels,
392 labels=labels,
393 )

396def _parse_snapshot(text: str, schema: TrackerSchema) -> tuple[BoardItem, ...]:
397 root = _strict_object(_strict_json(text), "GitHub project response")
398 if "items" not in root or "totalCount" not in root:
399 _fail_data("GitHub project response must contain items and totalCount")
400 raw_items = root["items"]
401 if not isinstance(raw_items, list):
402 _fail_data("GitHub project response items must be a list")
403 total = root["totalCount"]
404 if not isinstance(total, int) or isinstance(total, bool) or total != len(raw_items):
405 _fail_data("GitHub project response totalCount must equal items length")
406 items = tuple(_parse_item(raw, schema) for raw in raw_items)
407 numbers = [item.number for item in items]
408 if len(numbers) != len(set(numbers)):
409 _fail_data("GitHub project response contains duplicate issue numbers")
410 return items

413def _fetch_snapshot(schema: TrackerSchema) -> tuple[tuple[BoardItem, ...] | None, int]:
414 try:
415 _parse_project_fields(
416 _read_board_data(schema, _field_command(schema), "error fetching project fields")
417 )
418 items = _parse_snapshot(
419 _read_board_data(schema, _explorer_command(schema), "error fetching board snapshot"),
420 schema,
421 )
422 labels_by_number = _parse_repository_issue_labels(
423 _read_board_data(schema, _label_command(schema), "error fetching issue labels")
424 )
425 hydrated: list[BoardItem] = []
426 for item in items:
427 labels = labels_by_number.get(item.number)
428 if labels is None:
429 _fail_data(
430 f"open project issue #{item.number} is absent from repository issue data"
431 )
432 priority, effort, epic = _classify_labels(labels, item.number)
433 hydrated.append(
434 replace(
435 item,
436 labels=labels,
437 priority_labels=priority,
438 effort_labels=effort,
439 epic_labels=epic,
440 )
441 )
442 return tuple(hydrated), 0
443 except ToolMissingError:
444 _error("error: 'gh' CLI is not installed")
445 return None, 1
446 except WorkError as exc:
447 _error(str(exc))
448 return None, 1
449 except BoardDataError as exc:
450 _error(f"error parsing GitHub board data: {exc}")
451 return None, 1

556def _wide_rows(items: tuple[BoardItem, ...], width: int) -> list[str]:
557 id_width = max(4, *(len(f"#{item.number}") for item in items))
558 title_width = width - 2 - id_width - 9 - 7 - 14 - 8 - 14 - 18 - 7
559 if title_width < MIN_TITLE_WIDTH:
560 return _narrow_rows(items, width)
561 lines = [
562 f" {'ID':<{id_width}} {'PRI':<9} {'EFF':<7} {'TRACK':<14} "
563 f"{'HORIZON':<8} {'BLOCK':<14} {'EPIC':<18} TITLE"
564 ]
565 for item in items:
566 values = (
567 f"#{item.number}",
568 _priority(item),
569 _effort(item),
570 item.track or "-",
571 item.horizon or "-",
572 _block(item),
573 _epic(item),
574 item.title,
575 )
576 fields = (
577 _truncate(values[0], id_width),
578 _truncate(values[1], 9),
579 _truncate(values[2], 7),
580 _truncate(values[3], 14),
581 _truncate(values[4], 8),
582 _truncate(values[5], 14),
583 _truncate(values[6], 18),
584 _truncate(values[7], title_width),
585 )
586 lines.append(
587 f" {fields[0]:<{id_width}} {fields[1]:<9} {fields[2]:<7} "
588 f"{fields[3]:<14} {fields[4]:<8} {fields[5]:<14} "
589 f"{fields[6]:<18} {fields[7]}"
590 )
591 return lines

613def _section(label: str, items: tuple[BoardItem, ...]) -> list[str]:
614 lines = ["", f"[ {_safe(label).upper()} ] ({len(items)})"]
615 if not items:
616 lines.append(" (empty)")
617 return lines
618 ordered = tuple(sorted(items, key=_sort_key))
619 if _width() >= WIDE_WIDTH:
620 lines.extend(_wide_rows(ordered, _width()))
621 else:
622 lines.extend(_narrow_rows(ordered, _width()))
623 return lines

626def _status_sections(items: tuple[BoardItem, ...], schema: TrackerSchema) -> list[str]:
627 lines: list[str] = []
628 consumed: set[int] = set()
629 for configured in schema.statuses:
630 group = tuple(item for item in items if _status(item, schema) == configured)
631 consumed.update(item.number for item in group)
632 lines.extend(_section(configured, group))
633 no_status = tuple(item for item in items if _status(item, schema) is None)
634 if no_status:
635 consumed.update(item.number for item in no_status)
636 lines.extend(_section(NO_STATUS, no_status))
637 unexpected = sorted(
638 {
639 _status(item, schema)
640 for item in items
641 if item.number not in consumed and _status(item, schema) is not None
642 },
643 key=lambda value: (_safe(value).casefold(), _safe(value)),
644 )
645 for status in unexpected:
646 group = tuple(item for item in items if _status(item, schema) == status)
647 lines.extend(_section(status, group))
648 return lines

651def _render_focus(schema: TrackerSchema, items: tuple[BoardItem, ...]) -> list[str]:
652 def status(item: BoardItem) -> str:
653 return (_status(item, schema) or "").casefold()
654
655 has_horizon = any(item.horizon is not None for item in items)
656 if has_horizon:
657 ready = tuple(
658 item
659 for item in items
660 if status(item) == "ready" and (item.horizon or "").casefold() == "now"
661 )
662 rule = "Ready candidate rule: Status=Ready and Horizon=Now."
663 else:
664 ready = tuple(
665 item for item in items if status(item) == "ready" and "P0" in _priority_values(item)
666 )
667 rule = (
668 "Notice: Horizon is unavailable; Ready candidates use explicit P0 field/label evidence."
669 )
670 owner = tuple(item for item in items if status(item) == "needs you")
671 in_flight = tuple(item for item in items if status(item) == "in flight")
672 bench = tuple(item for item in items if status(item) == "bench-blocked")
673 lines = [
674 f"=== Board Focus: {schema.project_owner}/{schema.project_number} ===",
675 f"Open project issues scanned: {len(items)}",
676 rule,
677 ]
678 for label, group in (
679 ("Needs you", owner),
680 ("In flight", in_flight),
681 ("Ready candidates", ready),
682 ("Bench-blocked", bench),
