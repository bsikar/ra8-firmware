266static bool cache_mpu_test_cacheable_rw(void)
267{
268 for (uint32_t i = 0U; i < (uint32_t)k_cache_mpu_buf_bytes; i++) {
269 s_cache_mpu_buf[i] =
270 (uint8_t)((i * (uint32_t)k_cache_mpu_pat_mul) + (uint32_t)k_cache_mpu_pat_add);
271 }
272
273 /* Force the freshly written (dirty) lines back to SRAM and drop them, so the
274 * verify pass below must refill from memory -- the full write-back + refill
275 * round-trip with the D-cache on. */
276 if (ra8_cache_dcache_clean_invalidate_by_addr(s_cache_mpu_buf,
277 (uint32_t)sizeof(s_cache_mpu_buf)) != k_ra8_ok) {
278 return false;
279 }
280
281 for (uint32_t i = 0U; i < (uint32_t)k_cache_mpu_buf_bytes; i++) {
282 const uint8_t expected =
283 (uint8_t)((i * (uint32_t)k_cache_mpu_pat_mul) + (uint32_t)k_cache_mpu_pat_add);
284 if (s_cache_mpu_buf[i] != expected) {
285 return false;
286 }
287 }
288 return true;
289}

477int main(void)
478{
479 ra8_log_init();
480 ra8_log_info("CACHE_MPU", "==== cache_mpu_hil: M85 L1 cache + MPU self-test ====");
481
482 if (!cache_mpu_setup()) {
483 ra8_log_info("CACHE_MPU", "setup FAILED (CGC / console) -- halting");
484 cache_mpu_emit(k_cache_mpu_fail_setup, sizeof(k_cache_mpu_fail_setup) - 1U);
485 cache_mpu_wfi_forever();
486 }
487
488 const cache_mpu_result_t result = cache_mpu_run_selftest();
489
490 switch (result) {
491 case k_cache_mpu_result_pass:
492 ra8_log_info("CACHE_MPU", "cache+mpu PASS (cacheable RW + MRAM const + Device MMIO)");
493 cache_mpu_emit(k_cache_mpu_pass_banner, sizeof(k_cache_mpu_pass_banner) - 1U);
494 break;
495 case k_cache_mpu_result_fail_sram:
496 ra8_log_info("CACHE_MPU", "cacheable-SRAM step FAILED");
497 cache_mpu_emit(k_cache_mpu_fail_sram, sizeof(k_cache_mpu_fail_sram) - 1U);
498 break;
499 case k_cache_mpu_result_fail_mram:
500 ra8_log_info("CACHE_MPU", "MRAM-const step FAILED");
501 cache_mpu_emit(k_cache_mpu_fail_mram, sizeof(k_cache_mpu_fail_mram) - 1U);
502 break;
503 case k_cache_mpu_result_fail_mmio:
504 default:
505 ra8_log_info("CACHE_MPU", "device-MMIO step FAILED");
506 cache_mpu_emit(k_cache_mpu_fail_mmio, sizeof(k_cache_mpu_fail_mmio) - 1U);
507 break;
508 }
509
510 cache_mpu_wfi_forever();
511}
