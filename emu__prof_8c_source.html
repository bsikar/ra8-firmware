38/* ===========================================================================
39 * Firmware profiler (RA8_EMU_PROFILE). Two modes, bucketed by ELF FUNC symbol:
40 * =1 wall-time sample -- charge each chunk's wall time to its start PC.
41 * Cheap (no per-instruction cost); a flat % list of the dominant cost.
42 * =full per-instruction -- a code hook tallies every instruction + call entry
43 * AND reconstructs the live call chain (see below), so the run end emits
44 * an Ozone-style breakdown: a boot timeline, an inclusive/self table,
45 * and a speedscope flamechart file. Accurate but ~10x slower; off by
46 * default. The run auto-stops once boot settles into the idle frame
47 * loop, so =full profiles boot work rather than the idle tail.
48 * ===========================================================================
49 */
50enum : uint32_t {
51 k_prof_max_syms = 8192U,
52 k_prof_top_n = 40U,
53};
54typedef struct {
55 uint32_t lo;
56 uint32_t hi;
57 const char* name;
58 double secs;
59 uint64_t insns;
60 uint64_t calls;
61} prof_sym_t;
62static prof_sym_t s_prof[k_prof_max_syms];
63static uint32_t s_prof_n = 0U;
64static double s_prof_total_s = 0.0;
65static uint64_t s_prof_total_i = 0U;
66static prof_mode_t s_prof_mode = k_prof_off;

68/* ---------------------------------------------------------------------------
69 * Ozone-style call-stack tracing (insn mode only). On top of the per-function
70 * tally above, reconstruct the live call chain straight from the PC stream: a
71 * fresh function entry that is not already on the chain is a call (push), and
72 * re-entering a function already deeper on the chain is a return (pop down to
73 * it). NASA Rule 1 bans recursion in this firmware, so a function appears at
74 * most once on the chain and the "already on the chain" test is unambiguous.
75 * The chain is sampled at a fixed instruction cadence into a bounded,
76 * chronological store and written out as a speedscope "sampled" profile -- open
77 * ra8_emulator_profile.speedscope.json at https://speedscope.app for the
78 * time-ordered flamechart ("what ran when", the Ozone timeline) plus the
79 * sandwich view (self vs total per function). WFI idle naturally weighs ~zero
80 * because a halted core retires no instructions, so the picture is boot work,
81 * not the idle frame loop. ===============================================
82 */
83enum : uint32_t {
84 k_prof_max_depth = 64U,
85 k_prof_max_samples = 16384U,
86 k_prof_samp_every = 256U,
87};
88static uint16_t s_pstk[k_prof_max_depth];
89static uint32_t s_pstk_n = 0U;
90static uint16_t s_samp[k_prof_max_samples][k_prof_max_depth];
91static uint8_t s_samp_d[k_prof_max_samples];
92static uint32_t s_samp_w[k_prof_max_samples];
93static uint32_t s_samp_n = 0U;
94static uint64_t s_samp_every = (uint64_t)k_prof_samp_every;
95static uint64_t s_samp_acc = 0U;
96static uint32_t s_prof_stop_pc = 0U;
97static bool s_prof_stop_hit = false;
98static uint64_t s_incl[k_prof_max_syms];
99static uint64_t s_self[k_prof_max_syms];

102static int prof_cmp(const void* a, const void* b)
103{
104 const uint32_t la = ((const prof_sym_t*)a)->lo;
105 const uint32_t lb = ((const prof_sym_t*)b)->lo;
106 if (la < lb) {
107 return -1;
108 }
109 if (la > lb) {
110 return 1;
111 }
112 return 0;
113}

116static void prof_collect_symtab(const uint8_t* elf,
117 const uint8_t* sh,
118 uint32_t shoff,
119 uint16_t shentsize,
120 uint16_t shnum)
121{
122 uint32_t sym_off = 0U;
123 uint32_t sym_size = 0U;
124 uint32_t sym_link = 0U;
125 uint32_t sym_entsize = 0U;
126 (void)memcpy(&sym_off, sh + 16, 4);
127 (void)memcpy(&sym_size, sh + (uint32_t)k_elf_sh_size_off, 4);
128 (void)memcpy(&sym_link, sh + (uint32_t)k_elf_sh_link_off, 4);
129 (void)memcpy(&sym_entsize, sh + (uint32_t)k_elf_sh_entsize_off, 4);
130 if ((sym_entsize < 16U) || (sym_link >= shnum)) {
131 return;
132 }
133 const uint8_t* strsh = elf + shoff + ((size_t)(uint32_t)sym_link * shentsize);
134 uint32_t str_off = 0U;
135 (void)memcpy(&str_off, strsh + 16, 4);
136 const uint32_t nsym = sym_size / sym_entsize;
137 for (uint32_t s = 0U; (s < nsym) && (s_prof_n < (uint32_t)k_prof_max_syms); s++) {
138 const uint8_t* sym = elf + sym_off + ((size_t)s * sym_entsize);
139 uint32_t st_name = 0U;
140 uint32_t st_value = 0U;
141 uint32_t st_size = 0U;
142 (void)memcpy(&st_name, sym + 0, 4);
143 (void)memcpy(&st_value, sym + 4, 4);
144 (void)memcpy(&st_size, sym + 8, 4);
145 if (((sym[k_elf_sym_info_off] & (uint8_t)k_elf_st_type_mask) != 2U) || (st_size == 0U) ||
146 (st_name == 0U)) { /* STT_FUNC */
147 continue;
148 }
149 s_prof[s_prof_n].lo = st_value & ~1U;
150 s_prof[s_prof_n].hi = (st_value & ~1U) + st_size;
151 s_prof[s_prof_n].name = (const char*)(elf + str_off + st_name);
152 s_prof[s_prof_n].secs = 0.0;
153 s_prof[s_prof_n].insns = 0U;
154 s_prof[s_prof_n].calls = 0U;
155 s_prof_n++;
156 }
157}

160void prof_load(const uint8_t* elf, long len)
161{
162 const char* mode = getenv("RA8_EMU_PROFILE");
163 if (mode == nullptr) {
164 s_prof_mode = k_prof_off;
165 return;
166 }
167 s_prof_mode =
168 ((strcmp(mode, "full") == 0) || (strcmp(mode, "insn") == 0)) ? k_prof_insn : k_prof_wall;
169 if (len < (long)k_elf_ehdr_size) {
170 return;
171 }
172 uint32_t shoff = 0U;
173 (void)memcpy(&shoff, elf + 32, 4);
174 const uint16_t shentsize = (uint16_t)(elf[46] | (elf[47] << 8));
175 const uint16_t shnum = (uint16_t)(elf[48] | (elf[49] << 8));
176 if (shoff == 0U) {
177 return;
178 }
179 for (uint16_t i = 0U; i < shnum; i++) {
180 const uint8_t* sh = elf + shoff + ((size_t)(uint32_t)i * shentsize);
181 uint32_t sh_type = 0U;
182 (void)memcpy(&sh_type, sh + 4, 4);
183 if (sh_type != 2U) { /* SHT_SYMTAB */
184 continue;
185 }
186 prof_collect_symtab(elf, sh, shoff, shentsize, shnum);
187 }
188 qsort(s_prof, (size_t)s_prof_n, sizeof(s_prof[0]), prof_cmp);
189 (void)fprintf(stderr,
190 " [profile] %s; %u FUNC symbols\n",
191 (s_prof_mode == k_prof_insn) ? "per-instruction (exact, slow)" : "wall-time sample",
192 (unsigned)s_prof_n);
193}

196static uint32_t prof_find(uint32_t pc)
197{
198 uint32_t lo = 0U;
199 uint32_t hi = s_prof_n;
200 while (lo < hi) {
201 const uint32_t mid = lo + ((hi - lo) / 2U);
202 if (s_prof[mid].lo <= pc) {
203 lo = mid + 1U;
204 } else {
205 hi = mid;
206 }
207 }
208 if (lo == 0U) {
209 return s_prof_n;
210 }
211 const uint32_t idx = lo - 1U;
212 return (pc < s_prof[idx].hi) ? idx : s_prof_n;
213}

229static void prof_decimate(void)
230{
231 uint32_t dst = 0U;
232 for (uint32_t i = 0U; i < s_samp_n; i += 2U) {
233 const uint32_t w2 = ((i + 1U) < s_samp_n) ? s_samp_w[i + 1U] : 0U;
234 if (dst != i) {
235 (void)memcpy(s_samp[dst], s_samp[i], (size_t)s_samp_d[i] * sizeof(uint16_t));
236 s_samp_d[dst] = s_samp_d[i];
237 }
238 s_samp_w[dst] = s_samp_w[i] + w2; /* merged time keeps the total exact. */
239 dst++;
240 }
241 s_samp_n = dst;
242 s_samp_every *= 2U; /* coarser cadence keeps the next fill the same span. */
243}

246static void prof_sample(uint32_t weight)
247{
248 if (s_samp_n >= (uint32_t)k_prof_max_samples) {
249 prof_decimate();
250 }
251 uint32_t d = s_pstk_n;
252 if (d > (uint32_t)k_prof_max_depth) {
253 d = (uint32_t)k_prof_max_depth;
254 }
255 for (uint32_t i = 0U; i < d; i++) {
256 s_samp[s_samp_n][i] = s_pstk[i];
257 }
258 s_samp_d[s_samp_n] = (uint8_t)d;
259 s_samp_w[s_samp_n] = weight;
260 s_samp_n++;
261}

264static void prof_stack_update(uint32_t f)
265{
266 if (f >= s_prof_n) {
267 return; /* unknown region -- keep the current leaf (it gets the self time). */
268 }
269 if ((s_pstk_n > 0U) && (s_pstk[s_pstk_n - 1U] == (uint16_t)f)) {
270 return; /* still in the same function -- no call/return transition. */
271 }
272 for (uint32_t i = s_pstk_n; i > 0U; i--) {
273 if (s_pstk[i - 1U] == (uint16_t)f) {
274 s_pstk_n = i; /* returned to a frame already on the chain -- unwind to it. */
275 return;
276 }
277 }
278 if (s_pstk_n < (uint32_t)k_prof_max_depth) {
279 s_pstk[s_pstk_n] = (uint16_t)f; /* a fresh call -- push it. */
280 s_pstk_n++;
281 }
282}

286static void prof_insn_hook(uc_engine* uc, uint64_t address, uint32_t size, void* user)
287{
288 (void)size;
289 (void)user;
290 s_prof_total_i++;
291 const uint32_t idx = prof_find((uint32_t)address);
292 if (idx < s_prof_n) {
293 s_prof[idx].insns++;
294 if ((uint32_t)address == s_prof[idx].lo) {
295 s_prof[idx].calls++; /* PC at the entry point -> a fresh call (approx). */
296 }
297 }
298 prof_stack_update(idx);
299 s_samp_acc++;
300 if (s_samp_acc >= s_samp_every) {
301 prof_sample((uint32_t)s_samp_acc);
302 s_samp_acc = 0U;
303 }
304 if ((s_prof_stop_pc != 0U) && ((uint32_t)address == s_prof_stop_pc)) {
305 s_prof_stop_hit = true; /* RA8_EMU_STOP_PC reached -- end the run cleanly. */
306 (void)uc_emu_stop(uc);
307 }

550static uint64_t prof_accumulate_incl_self(void)
551{
552 /* Inclusive (anywhere on the chain) + self (leaf) weights, from the samples.
553 * No recursion (NASA Rule 1) -> each function appears at most once per sample,
554 * so a straight per-frame add needs no dedup. */
555 uint64_t total = 0U;
556 for (uint32_t i = 0U; i < s_prof_n; i++) {
557 s_incl[i] = 0U;
558 s_self[i] = 0U;
559 }
560 for (uint32_t i = 0U; i < s_samp_n; i++) {
561 const uint32_t w = s_samp_w[i];
562 const uint8_t d = s_samp_d[i];
563 total += w;
564 for (uint8_t j = 0U; j < d; j++) {
565 s_incl[s_samp[i][j]] += w;
566 }
567 if (d > 0U) {
568 s_self[s_samp[i][d - 1U]] += w;
569 }
570 }
571 return total;
572}

576{
577 /* Phase timeline: collapse each sample's chain to a fixed shallow depth (the
578 * major subsystem under main) and print each contiguous run as a boot phase,
579 * so the terminal shows "what ran when" even without opening speedscope. */
580 (void)fprintf(stderr,
581 " [profile] boot timeline (phase = call depth %u; start%% .. width%%):\n",
582 (unsigned)k_phase_depth);
583 uint64_t cum = 0U;
584 uint64_t segw = 0U;
585 uint32_t segfn = (uint32_t)k_no_fn;
586 uint32_t lines = 0U;
587 for (uint32_t i = 0U; i <= s_samp_n; i++) {
588 uint32_t fn = (uint32_t)k_no_fn;
589 if (i < s_samp_n) {
590 const uint8_t d = s_samp_d[i];
591 if (d > 0U) {
592 const uint8_t pd = ((uint32_t)(d - 1U) < (uint32_t)k_phase_depth) ? (uint8_t)(d - 1U)
593 : (uint8_t)k_phase_depth;
594 fn = s_samp[i][pd];
595 }
596 }
597 if ((i == s_samp_n) || (fn != segfn)) {
598 const bool show = (segfn != (uint32_t)k_no_fn) &&
599 (((uint32_t)k_phase_pct_x1 * segw) >= total) &&
600 (lines < (uint32_t)k_phase_lines);
601 if (show) {
602 (void)fprintf(stderr,
603 " %5.1f%% +%4.1f%% %s\n",
604 s_percent_scale * (double)cum / (double)total,
605 s_percent_scale * (double)segw / (double)total,
606 (segfn < s_prof_n) ? s_prof[segfn].name : "?");
607 lines++;
608 }
609 cum += segw;
610 segw = 0U;
611 segfn = fn;
612 }
613 if (i < s_samp_n) {
614 segw += s_samp_w[i];
615 }
616 }
617}

620static void prof_print_incl_self_table(const char* html, const char* out, uint64_t total)
621{
622 /* Inclusive/self table -- the "why is it slow" view (sorted by inclusive). */

624 " [profile] flamechart GUI -> %s (interactive: hover/zoom/search)\n"
625 " [profile] %u samples over %llu insns (also %s for speedscope.app)\n"
626 " self%% total%% function\n",
627 html,
628 (unsigned)s_samp_n,
629 (unsigned long long)total,
630 out);
631 for (uint32_t k = 0U; k < (uint32_t)k_prof_top_n; k++) {
632 uint32_t best = s_prof_n;
633 uint64_t bestv = 0U;
634 for (uint32_t i = 0U; i < s_prof_n; i++) {

636 bestv = s_incl[i];

638 }

641 break;

645 s_percent_scale * (double)s_self[best] / (double)total,
646 s_percent_scale * (double)s_incl[best] / (double)total,
647 s_prof[best].name);
648 s_incl[best] = 0U; /* consume so the next pick is the runner-up. */
649 }

675void prof_report(void)
676{
677 const bool insn = (s_prof_mode == k_prof_insn);
678 const double tot = insn ? (double)s_prof_total_i : s_prof_total_s;
679 if ((s_prof_mode == k_prof_off) || (tot <= 0.0)) {
680 return;
681 }
682 if (insn) {
683 (void)fprintf(stderr,
684 " [profile] %llu instructions; hottest (by instruction count):\n"
685 " %%insn instructions calls function\n",
686 (unsigned long long)s_prof_total_i);
687 } else {
688 (void)fprintf(stderr, " [profile] %.2fs wall sampled; hottest (by wall time):\n", tot);
689 }
690 for (uint32_t k = 0U; k < (uint32_t)k_prof_top_n; k++) {
691 uint32_t best = s_prof_n;
692 double bestv = 0.0;
693 for (uint32_t i = 0U; i < s_prof_n; i++) {
694 const double v = insn ? (double)s_prof[i].insns : s_prof[i].secs;
695 if (v > bestv) {
696 bestv = v;
697 best = i;
698 }
699 }
700 if ((best == s_prof_n) || (bestv <= 0.0)) {
701 break;
702 }
703 if (insn) {
704 (void)fprintf(stderr,
705 " %6.2f%% %15llu %10llu %s\n",
706 s_percent_scale * bestv / tot,
707 (unsigned long long)s_prof[best].insns,
708 (unsigned long long)s_prof[best].calls,
709 s_prof[best].name);
710 s_prof[best].insns = 0U;
711 } else {
712 (void)fprintf(stderr,
713 " %6.2f%% %8.2fs %s\n",
714 s_percent_scale * bestv / tot,
715 bestv,
716 s_prof[best].name);
717 s_prof[best].secs = 0.0;
718 }
719 }
720 if (insn && (s_samp_n > 0U)) {
721 prof_report_flamechart(); /* speedscope export + inclusive/self + timeline. */
722 }
723}
