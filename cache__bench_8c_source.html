112static uint32_t cb_hash(cb_key_t k)
113{
114 uint64_t h = ((uint64_t)k.object_id << 32U) | (uint64_t)k.page;
115 h ^= h >> (uint8_t)k_hash_shift;
116 h *= (uint64_t)k_hash_mix_mul;
117 h ^= h >> (uint8_t)k_hash_shift;
118 return (uint32_t)h;
119}

185static int32_t cb_index_find(const cb_index_t* idx, const cb_frame_t* frames, cb_key_t key)
186{

215static void cb_index_remove(cb_index_t* idx, const cb_frame_t* frames, uint32_t frame)
216{
217 const uint32_t ob = cb_hash(frames[frame].key) & idx->mask;
218 int32_t prev = -1;
219 for (int32_t j = idx->bucket[ob]; j != -1; prev = j, j = idx->next[j]) {
220 if (j == (int32_t)frame) {
221 if (prev == -1) {
222 idx->bucket[ob] = idx->next[j];
223 } else {
224 idx->next[prev] = idx->next[j];
225 }
226 break;
227 }
228 }
229}

282static uint32_t cb_replay_take_frame(const cache_policy_t* pol,
283 cb_cache_t* cache,
284 cb_index_t* idx,
285 uint32_t* filled,
286 cb_result_t* out)
287{
288 if (*filled < cache->capacity) {
289 const uint32_t frame = *filled;
290 (*filled)++;
291 return frame;
292 }
293 uint32_t scanned = 0U;
294 const uint32_t frame = pol->pick_victim(cache, &scanned);
295 out->evictions++;
296 out->total_scan += scanned;
297 if (scanned > out->worst_scan) {
298 out->worst_scan = scanned;
299 }
300 cb_index_remove(idx, cache->frames, frame);
301 return frame;
302}

328static bool cb_replay_open(cb_index_t* idx, cb_frame_t** frames, uint32_t capacity)
329{
330 const uint32_t hsize = cb_pow2_ceil(capacity * 4U);
331 *frames = (cb_frame_t*)calloc((size_t)capacity, sizeof(cb_frame_t));
332 *idx = (cb_index_t){.bucket = (int32_t*)malloc((size_t)hsize * sizeof(int32_t)),
333 .next = (int32_t*)malloc((size_t)capacity * sizeof(int32_t)),
334 .mask = hsize - 1U};
335 if ((*frames == nullptr) || (idx->bucket == nullptr) || (idx->next == nullptr)) {
336 return false;
337 }
338 for (uint32_t i = 0U; i < hsize; ++i) {
339 idx->bucket[i] = -1;
340 }
341 return true;
342}

371int cb_replay(const cache_policy_t* pol,
372 const cb_key_t* keys,
373 uint64_t n,
374 uint32_t capacity,
375 cb_result_t* out)
376{
377 *out = (cb_result_t){};
378 if ((pol == nullptr) || (keys == nullptr) || (capacity == 0U)) {
379 return 1;
380 }
381
382 cb_frame_t* frames = nullptr;
383 cb_index_t idx = {};
384 const bool ready = cb_replay_open(&idx, &frames, capacity);
385 cb_cache_t cache = {.frames = frames, .capacity = capacity, .policy_data = nullptr};
386 if (!ready || ((pol->init != nullptr) && (pol->init(&cache) != 0))) {
387 cb_replay_close(&idx, frames);
388 return 1;
389 }
390
391 uint32_t filled = 0U;
392 for (uint64_t i = 0U; i < n; ++i) {
393 const cb_key_t key = keys[i];
394 const int32_t found = cb_index_find(&idx, frames, key);
395 out->accesses++;
396 if (found != -1) {
397 out->hits++;
398 if (pol->on_access != nullptr) {
399 pol->on_access(&cache, (uint32_t)found);
400 }
401 continue;
402 }
403 const uint32_t frame = cb_replay_take_frame(pol, &cache, &idx, &filled, out);
404 frames[frame].key = key;
405 frames[frame].live = true;
406 cb_index_push(&idx, frame, key);
407 if (pol->on_insert != nullptr) {
408 pol->on_insert(&cache, frame);
409 }
410 }
411
412 if (pol->deinit != nullptr) {
413 pol->deinit(&cache);
414 }
415 cb_replay_close(&idx, frames);
416 return 0;
417}
