70static uint32_t cb_hash(cb_key_t k)
71{
72 uint64_t h = ((uint64_t)k.object_id << 32U) | (uint64_t)k.page;
73 h ^= h >> (uint8_t)k_hash_shift;
74 h *= (uint64_t)k_hash_mix_mul;
75 h ^= h >> (uint8_t)k_hash_shift;
76 return (uint32_t)h;
77}

100static int32_t cb_index_find(const cb_index_t* idx, const cb_frame_t* frames, cb_key_t key)
101{
102 for (int32_t j = idx->bucket[cb_hash(key) & idx->mask]; j != -1; j = idx->next[j]) {
103 if (cb_key_eq(frames[j].key, key)) {
104 return j;
105 }
106 }
107 return -1;
108}

111static void cb_index_remove(cb_index_t* idx, const cb_frame_t* frames, uint32_t frame)
112{
113 const uint32_t ob = cb_hash(frames[frame].key) & idx->mask;
114 int32_t prev = -1;
115 for (int32_t j = idx->bucket[ob]; j != -1; prev = j, j = idx->next[j]) {
116 if (j == (int32_t)frame) {
117 if (prev == -1) {
118 idx->bucket[ob] = idx->next[j];
119 } else {
120 idx->next[prev] = idx->next[j];
121 }
122 break;
123 }
124 }
125}

128static void cb_index_push(cb_index_t* idx, uint32_t frame, cb_key_t key)
129{
130 const uint32_t b = cb_hash(key) & idx->mask;
131 idx->next[frame] = idx->bucket[b];

159static uint32_t cb_replay_take_frame(const cache_policy_t* pol,
160 cb_cache_t* cache,
161 cb_index_t* idx,
162 uint32_t* filled,
163 cb_result_t* out)
164{
165 if (*filled < cache->capacity) {
166 const uint32_t frame = *filled;
167 (*filled)++;
168 return frame;
169 }
170 uint32_t scanned = 0U;
171 const uint32_t frame = pol->pick_victim(cache, &scanned);
172 out->evictions++;
173 out->total_scan += scanned;
174 if (scanned > out->worst_scan) {
175 out->worst_scan = scanned;
176 }
177 cb_index_remove(idx, cache->frames, frame);
178 return frame;
179}

203static bool cb_replay_open(cb_index_t* idx, cb_frame_t** frames, uint32_t capacity)
204{
205 const uint32_t hsize = cb_pow2_ceil(capacity * 4U);
206 *frames = (cb_frame_t*)calloc((size_t)capacity, sizeof(cb_frame_t));
207 *idx = (cb_index_t){.bucket = (int32_t*)malloc((size_t)hsize * sizeof(int32_t)),
208 .next = (int32_t*)malloc((size_t)capacity * sizeof(int32_t)),
209 .mask = hsize - 1U};
210 if ((*frames == nullptr) || (idx->bucket == nullptr) || (idx->next == nullptr)) {
211 return false;
212 }
213 for (uint32_t i = 0U; i < hsize; ++i) {
214 idx->bucket[i] = -1;
215 }
216 return true;
217}

220static void cb_replay_close(cb_index_t* idx, cb_frame_t* frames)
221{
222 free(frames);
223 free(idx->bucket);
224 free(idx->next);
225 *idx = (cb_index_t){};

228int cb_replay(const cache_policy_t* pol,
229 const cb_key_t* keys,
230 uint64_t n,
231 uint32_t capacity,
232 cb_result_t* out)
233{
234 *out = (cb_result_t){};
235 if ((pol == nullptr) || (keys == nullptr) || (capacity == 0U)) {
236 return 1;
237 }
238
239 cb_frame_t* frames = nullptr;
240 cb_index_t idx = {};
241 const bool ready = cb_replay_open(&idx, &frames, capacity);
242 cb_cache_t cache = {.frames = frames, .capacity = capacity, .policy_data = nullptr};
243 if (!ready || ((pol->init != nullptr) && (pol->init(&cache) != 0))) {
244 cb_replay_close(&idx, frames);
245 return 1;
246 }
247
248 uint32_t filled = 0U;
249 for (uint64_t i = 0U; i < n; ++i) {
250 const cb_key_t key = keys[i];
251 const int32_t found = cb_index_find(&idx, frames, key);
252 out->accesses++;
253 if (found != -1) {
254 out->hits++;
255 if (pol->on_access != nullptr) {
256 pol->on_access(&cache, (uint32_t)found);
257 }
258 continue;
259 }
260 const uint32_t frame = cb_replay_take_frame(pol, &cache, &idx, &filled, out);
261 frames[frame].key = key;
262 frames[frame].live = true;
263 cb_index_push(&idx, frame, key);
264 if (pol->on_insert != nullptr) {
265 pol->on_insert(&cache, frame);
266 }
267 }
268
269 if (pol->deinit != nullptr) {
270 pol->deinit(&cache);
271 }
272 cb_replay_close(&idx, frames);
273 return 0;
274}
