96{
97 ra8_widget_status_bar_t* sb = (ra8_widget_status_bar_t*)w->ctx;
98 if (sb == nullptr) {
99 return;
100 }
101 if (sb->paint == nullptr) {
102 return;
103 }
104 const ra8_ui_rect_t* r = &w->rect;
105 ra8_widget_priv_fill_box(sb->paint, r, sb->bg, sb->bg, (int16_t)k_ra8_widget_sb_no_rule);
106 internal_sb_label(sb->paint, r, sb->left, sb->pad, k_ra8_widget_align_left, sb->fg, sb->bg);
107 internal_sb_label(sb->paint,
108 r,
109 sb->right,
110 sb->pad,
111 k_ra8_widget_align_right,
112 sb->fg_right,
113 sb->bg);
114 if (sb->rule_h <= (int16_t)k_ra8_widget_sb_no_rule) {

118 return;

121 r->x,

123 r->w,
124 (int32_t)sb->rule_h,
125 sb->rule);
