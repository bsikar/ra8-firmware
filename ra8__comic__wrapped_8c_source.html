113ra8_err_t ra8_comic_open_wrapped(ra8_comic_t* c,
114 ra8_comic_read_fn read,
115 void* ctx,
116 uint64_t size,
117 ra8_comic_page_t* pages,
118 uint32_t page_cap,
119 char* names,
120 uint32_t names_cap,
121 uint8_t* arena,
122 size_t arena_cap,
123 void* xz_scratch,
124 uint32_t xz_scratch_len)
125{
126 RA8_CHECK_NULL_PTR(c, s_tag_wrap, "wrapped open: null c");
127 RA8_CHECK_NULL_PTR((const void*)read, s_tag_wrap, "wrapped open: null read");
128 RA8_CHECK_NULL_PTR(arena, s_tag_wrap, "wrapped open: null arena");
129 if (size == 0U) {
130 return k_ra8_err_invalid_size;
131 }
132 uint8_t sig[k_ra8_comic_magic_len] = {};
133 const size_t got = read(ctx, 0U, sig, sizeof(sig));
134 const bool is_gzip = ra8_unarch_gzip_magic(sig, got);
135 const bool is_xz = ra8_unarch_xz_magic(sig, got);
136 if (!is_gzip && !is_xz) {
137 /* Not a wrapper: the ordinary container detect owns it. */
138 return ra8_comic_open(c, read, ctx, size, pages, page_cap, names, names_cap);
139 }
140 if (((uintptr_t)arena % (uintptr_t)k_ra8_comic_wrap_align) != 0U) {
141 return k_ra8_err_invalid_size; /* descriptor slot needs the alignment */
142 }
143 if (arena_cap <= (size_t)k_wrap_hdr_bytes) {
144 return k_ra8_err_invalid_size; /* no room for descriptor + payload */
145 }
146 uint8_t* payload = &arena[k_wrap_hdr_bytes];
147 const size_t payload_cap = arena_cap - (size_t)k_wrap_hdr_bytes;
148 size_t inner_len = 0U;
149 const ra8_err_t uerr = s_unwrap(read,
150 ctx,
151 size,
152 is_gzip,
153 payload,
154 payload_cap,
155 xz_scratch,
156 xz_scratch_len,
157 &inner_len);
158 if (uerr != k_ra8_ok) {
159 return uerr;
160 }
161 /* Nesting bomb: a wrapper inside the wrapper is rejected before any
162 * inner decode -- the one reachable depth bound in this pipeline. */
163 const bool inner_gzip = ra8_unarch_gzip_magic(payload, inner_len);
164 const bool inner_xz = ra8_unarch_xz_magic(payload, inner_len);
165 if (inner_gzip || inner_xz) {
166 return k_ra8_err_decomp_depth;
167 }
168 /* The flat-memory descriptor lives at the arena start so it out-lives
169 * this call: page extraction keeps reading through it. */
170 void* const hdr_slot = arena;
171 ra8_unarch_mem_t* const mem = (ra8_unarch_mem_t*)hdr_slot;
172 mem->base = payload;
173 mem->len = inner_len;
174 return ra8_comic_open(c,
175 ra8_unarch_mem_read,
176 mem,
177 (uint64_t)inner_len,
178 pages,
179 page_cap,
180 names,
181 names_cap);
182}
