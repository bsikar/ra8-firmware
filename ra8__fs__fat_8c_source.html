134static uint32_t priv_fat_entry_byte_offset(const ra8_fs_mount_t* m, uint32_t cluster)
135{
136 if (m->type == k_ra8_fs_type_fat12) {
137 return cluster + (cluster / 2U);
138 }

140 return cluster * 2U;
141 }

145/* `priv_fat_get()`: see header for the documented contract. */
146ra8_err_t priv_fat_get(const ra8_fs_mount_t* m, uint32_t cluster, uint32_t* out_value)
147{
148 const uint32_t fat_offset = priv_fat_entry_byte_offset(m, cluster);
149 const uint32_t sec_num = m->first_fat_lba + (fat_offset / k_ra8_fs_bytes_per_sector);
150 const uint32_t sec_off = fat_offset % k_ra8_fs_bytes_per_sector;
151
152 uint8_t buf[k_ra8_fs_bytes_per_sector] = {};
153 ra8_err_t err = priv_read_sector(m, sec_num, buf);
154 if (err != k_ra8_ok) {
155 return err;
156 }
157
158 uint32_t v = 0;
159 if (m->type == k_ra8_fs_type_fat12) {
160 /* MS FAT spec sec 4.1: read 16 bits straddling the byte and shift. */
161 uint8_t b0 = buf[sec_off];
162 uint8_t b1 = 0;
163 if (sec_off + 1U < k_ra8_fs_bytes_per_sector) {
164 b1 = buf[sec_off + 1U];
165 } else {
166 uint8_t buf2[k_ra8_fs_bytes_per_sector] = {};
167 err = priv_read_sector(m, sec_num + 1U, buf2);
168 if (err != k_ra8_ok) {
169 return err;
170 }
171 b1 = buf2[0];
172 }
173 uint16_t raw = (uint16_t)((uint16_t)b0 | ((uint16_t)b1 << k_shift_byte));
174 if ((cluster & 1U) != 0U) {
175 v = (uint32_t)(raw >> k_shift_nibble);
176 } else {
177 v = (uint32_t)(raw & k_fat12_value_mask);
178 }
179 } else if (m->type == k_ra8_fs_type_fat16) {
180 v = priv_rd16(&buf[sec_off]);
181 } else {
182 v = priv_rd32(&buf[sec_off]) & k_cluster_mask_fat32;
183 }
184
185 *out_value = v;
186 return k_ra8_ok;
187}

213RA8_INTERNAL
214static ra8_err_t priv_fat12_set_one(const ra8_fs_mount_t* m,
215 uint32_t sec_num,
216 uint32_t sec_off,
217 uint32_t cluster,
218 uint32_t value)
219{
220 uint8_t buf[k_ra8_fs_bytes_per_sector] = {};
221 uint8_t buf2[k_ra8_fs_bytes_per_sector] = {};
222 uint8_t straddle = 0U;
223 ra8_err_t err = priv_read_sector(m, sec_num, buf);
224 if (err != k_ra8_ok) {
225 return err;
226 }
227 uint8_t b0 = buf[sec_off];
228 uint8_t b1 = 0;
229 if (sec_off + 1U < (uint32_t)k_ra8_fs_bytes_per_sector) {
230 b1 = buf[sec_off + 1U];
231 } else {
232 err = priv_read_sector(m, sec_num + 1U, buf2);
233 if (err != k_ra8_ok) {
234 return err;
235 }
236 b1 = buf2[0];
237 straddle = 1U;
238 }
239 uint16_t raw = (uint16_t)((uint16_t)b0 | ((uint16_t)b1 << k_shift_byte));
240 if ((cluster & 1U) != 0U) {
241 raw = (uint16_t)((raw & k_fat12_low_nibble_mask) |
242 (uint16_t)((value & k_fat12_value_mask) << k_shift_nibble));
243 } else {
244 raw = (uint16_t)((raw & k_fat12_high_nibble_mask) | (uint16_t)(value & k_fat12_value_mask));
245 }
246 buf[sec_off] = (uint8_t)(raw & k_byte_mask);
247 if (straddle == 0U) {
248 buf[sec_off + 1U] = (uint8_t)((raw >> k_shift_byte) & k_byte_mask);
249 } else {
250 buf2[0] = (uint8_t)((raw >> k_shift_byte) & k_byte_mask);
251 }
252 err = priv_write_sector(m, sec_num, buf);
253 if (err != k_ra8_ok) {
254 return err;
255 }
256 if (straddle != 0U) {
257 err = priv_write_sector(m, sec_num + 1U, buf2);
258 }
259 return err;
260}

337/* `priv_fat_set()`: see header for the documented contract. */
338ra8_err_t priv_fat_set(const ra8_fs_mount_t* m, uint32_t cluster, uint32_t value)
339{
340 const uint32_t fat_offset = priv_fat_entry_byte_offset(m, cluster);
341 for (uint32_t i = 0; i < m->num_fats; i++) {
342 const uint32_t fat_base = m->first_fat_lba + (i * m->fat_size_sectors);
343 const uint32_t sec_num = fat_base + (fat_offset / k_ra8_fs_bytes_per_sector);
344 const uint32_t sec_off = fat_offset % k_ra8_fs_bytes_per_sector;
345 ra8_err_t err = k_ra8_ok;
346 if (m->type == k_ra8_fs_type_fat12) {
347 err = priv_fat12_set_one(m, sec_num, sec_off, cluster, value);
348 } else if (m->type == k_ra8_fs_type_fat16) {
349 err = priv_fat16_set_one(m, sec_num, sec_off, value);
350 } else {
351 err = priv_fat32_set_one(m, sec_num, sec_off, value);
352 }
353 if (err != k_ra8_ok) {
354 return err;
355 }
356 }
357 return k_ra8_ok;
358}
