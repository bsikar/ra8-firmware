261static void ipc_producer_thread_entry(ULONG thread_input)
262{
263 (void)thread_input;
264
265 ipc_demo_log(k_ipc_demo_banner, (uint32_t)(sizeof(k_ipc_demo_banner) - 1U));
266
267 while (1) {
268 ULONG msg = (ULONG)k_ipc_demo_payload_ping;
269 /* ``tx_queue_send`` arguments: queue ptr, ptr-to-source (NOT the
270 * value), wait option. The source pointer is read for exactly
271 * ``message_size`` ULONGs as declared at create time. */
272 if (tx_queue_send(&s_ipc_queue, &msg, TX_WAIT_FOREVER) == TX_SUCCESS) {
273 ipc_demo_log(k_ipc_demo_send_msg, (uint32_t)(sizeof(k_ipc_demo_send_msg) - 1U));
274 }
275 (void)tx_thread_sleep((ULONG)k_ipc_demo_period_ticks);
276 }
277}

305static void ipc_consumer_thread_entry(ULONG thread_input)
306{
307 (void)thread_input;
308
309 while (1) {
310 ULONG msg = 0U;
311 if (tx_queue_receive(&s_ipc_queue, &msg, TX_WAIT_FOREVER) != TX_SUCCESS) {
312 /* Should not happen with TX_WAIT_FOREVER on a healthy queue,
313 * but if it does we skip the log so the HIL negative-regex
314 * does not trip on spurious output. */
315 continue;
316 }
317 if (msg != (ULONG)k_ipc_demo_payload_ping) {
318 continue;
319 }
320 ipc_demo_log(k_ipc_demo_recv_msg, (uint32_t)(sizeof(k_ipc_demo_recv_msg) - 1U));
321 g_ipc_demo_tick += 1U;
322 }
323}

337void tx_application_define(void* first_unused_memory)
338{
339 (void)first_unused_memory;
340
341 /* ThreadX takes non-const CHAR* for object names; use writable static
342 * buffers to avoid -Wdiscarded-qualifiers / -Wcast-qual. */
343 static CHAR s_queue_name[] = "ipc_demo_q";
344 static CHAR s_producer_name[] = "producer";
345 static CHAR s_consumer_name[] = "consumer";
346
347 /* tx_queue_create: message_size is in 32-bit ULONGs (TX_1_ULONG=1),
348 * queue_size is the total backing-buffer byte count. Deriving the
349 * byte count from sizeof() of the same backing array keeps the two
350 * arguments in sync. */
351 if (tx_queue_create(&s_ipc_queue,
352 s_queue_name,
353 TX_1_ULONG,
354 s_ipc_queue_storage,
355 (ULONG)sizeof(s_ipc_queue_storage)) != TX_SUCCESS) {
356 ipc_demo_panic_halt();
357 }
358
359 /* Consumer goes first so it is parked in tx_queue_receive before the
360 * producer's first send -- avoids a transient queue-not-empty
361 * window where the consumer would have to sleep to catch up. */
362 if (tx_thread_create(&s_consumer_thread,
363 s_consumer_name,
364 ipc_consumer_thread_entry,
365 0U,
366 s_consumer_stack,
367 (ULONG)k_ipc_demo_thread_stack,
368 (UINT)k_ipc_demo_consumer_prio,
369 (UINT)k_ipc_demo_consumer_prio,
370 TX_NO_TIME_SLICE,
371 TX_AUTO_START) != TX_SUCCESS) {
372 ipc_demo_panic_halt();
373 }
374
375 if (tx_thread_create(&s_producer_thread,
376 s_producer_name,
377 ipc_producer_thread_entry,
378 0U,
379 s_producer_stack,
380 (ULONG)k_ipc_demo_thread_stack,
381 (UINT)k_ipc_demo_producer_prio,
382 (UINT)k_ipc_demo_producer_prio,
383 TX_NO_TIME_SLICE,
384 TX_AUTO_START) != TX_SUCCESS) {
385 ipc_demo_panic_halt();
386 }
387}
