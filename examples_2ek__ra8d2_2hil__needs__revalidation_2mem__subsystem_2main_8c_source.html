209static uint32_t mem_crc32_byte(uint32_t crc, uint8_t b)
210{
211 crc ^= (uint32_t)b;
212 for (uint32_t bit = 0U; bit < (uint32_t)k_mem_crc_bits; bit++) {
213 const uint32_t mask = (uint32_t)(-(int32_t)(crc & 1U));
214 crc = (crc >> 1U) ^ ((uint32_t)k_mem_crc_poly & mask);
215 }
216 return crc;
217}

220static uint8_t mem_gen_byte(uint64_t off)
221{
222 const uint32_t x = (uint32_t)off;
223 uint32_t h = x ^ (x >> (uint32_t)k_mem_gen_shift_a) ^ (x >> (uint32_t)k_mem_gen_shift_b);
224 h = (h * (uint32_t)k_mem_gen_mul) + (uint32_t)k_mem_gen_add;
225 return (uint8_t)(h >> (uint32_t)k_mem_gen_shift_out);
226}

229static void mem_print_hex(uint32_t value)
230{
231 uint8_t buf[k_mem_hex_nibbles];
232 for (uint32_t i = 0U; i < (uint32_t)k_mem_hex_nibbles; i++) {
233 const uint32_t shift = ((uint32_t)k_mem_hex_nibbles - 1U - i) * (uint32_t)k_mem_nibble_bits;
234 const uint32_t nib = (value >> shift) & (uint32_t)k_mem_nibble_mask;
235 buf[i] =
236 (uint8_t)((nib < (uint32_t)k_mem_dec_base) ? ('0' + nib) : ('A' + (nib - k_mem_dec_base)));
237 }
238 mem_print(buf, (uint32_t)k_mem_hex_nibbles);
239}

242static void mem_print_uint(uint32_t value)
243{
244 uint8_t buf[k_mem_dec_base];
245 uint32_t n = 0U;
246 if (value == 0U) {
247 buf[n] = '0';
248 n++;
249 }
250 while ((value > 0U) && (n < (uint32_t)k_mem_dec_base)) {
251 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_mem_dec_base));
252 n++;
253 value /= (uint32_t)k_mem_dec_base;
254 }
255 for (uint32_t i = 0U; i < n; i++) {
256 mem_print(&buf[n - 1U - i], 1U);
257 }
258}

266static bool mem_run_slab(mem_report_t* r)
267{
268 ra8_slab_t slab = {};
269 if (ra8_slab_init(&slab,
270 s_slab_pool,
271 (uint32_t)sizeof(s_slab_pool),
272 (uint32_t)k_mem_slab_cell_bytes) != k_ra8_ok) {
273 return false;
274 }
275 void* cells[k_mem_slab_cells] = {};
276 for (uint32_t i = 0U; i < (uint32_t)k_mem_slab_cells; i++) {
277 if (ra8_slab_alloc(&slab, &cells[i]) != k_ra8_ok) {
278 return false;
279 }
280 }
281 void* extra = nullptr;
282 if (ra8_slab_alloc(&slab, &extra) != k_ra8_err_no_mem) {
283 return false;
284 }
285 for (uint32_t i = 0U; i < (uint32_t)k_mem_slab_freed; i++) {
286 if (ra8_slab_free(&slab, cells[i]) != k_ra8_ok) {
287 return false;
288 }
289 }
290 uint32_t free_before = 0U;
291 if ((ra8_slab_stats(&slab, &free_before, nullptr) != k_ra8_ok) ||
292 (free_before != (uint32_t)k_mem_slab_freed)) {
293 return false;
294 }
295 /* Reset: re-init re-threads every cell back onto the freelist. */
296 if (ra8_slab_init(&slab,
297 s_slab_pool,
298 (uint32_t)sizeof(s_slab_pool),
299 (uint32_t)k_mem_slab_cell_bytes) != k_ra8_ok) {
300 return false;
301 }
302 uint32_t total = 0U;
303 if (ra8_slab_stats(&slab, &r->slab_free, &total) != k_ra8_ok) {
304 return false;
305 }
306 return (total == (uint32_t)k_mem_slab_cells) && (r->slab_free == (uint32_t)k_mem_slab_cells);
307}

381static bool mem_run_tiles(mem_report_t* r)
382{
383 ra8_tile_cache_t tc = {};
384 const ra8_tile_cache_cfg_t cfg = {
385 .cell_mem = s_tile_cells,
386 .cell_bytes = (uint32_t)k_mem_tile_cell_bytes,
387 .cell_count = (uint32_t)k_mem_tile_cells,
388 .meta = s_tile_meta,
389 .keys = s_tile_keys,
390 .dims = s_tile_dims,
391 .buckets = s_tile_buckets,
392 .bucket_count = (uint32_t)k_mem_tile_buckets,
393 .decode = mem_tile_decode,
394 .decode_ctx = nullptr,
395 };
396 if (ra8_tile_cache_init(&tc, &cfg) != k_ra8_ok) {
397 return false;
398 }
399 /* Overfill: more distinct tiles than cells forces LRU eviction. */
400 for (uint32_t i = 0U; i < (uint32_t)k_mem_tile_distinct; i++) {
401 if (mem_tile_touch(&tc, (uint16_t)i) != k_ra8_ok) {
402 return false;
403 }
404 }
405 /* The most-recent k_mem_tile_cells tiles are still resident: re-touch them
406 * (MRU-first) so each is a hit and none evicts. */
407 for (uint32_t i = 0U; i < (uint32_t)k_mem_tile_cells; i++) {
408 const uint16_t tx = (uint16_t)((uint32_t)k_mem_tile_distinct - 1U - i);
409 if (mem_tile_touch(&tc, tx) != k_ra8_ok) {
410 return false;
411 }
412 }
413 /* Re-touch the very first tile (long evicted): one more miss + eviction. */
414 if (mem_tile_touch(&tc, 0U) != k_ra8_ok) {
415 return false;
416 }
417 if (ra8_tile_cache_stats(&tc, &r->tile_hits, &r->tile_misses, &r->tile_evictions) != k_ra8_ok) {
418 return false;
419 }
420 return (r->tile_hits == (uint32_t)k_mem_tile_exp_hits) &&
421 (r->tile_misses == (uint32_t)k_mem_tile_exp_miss) &&
422 (r->tile_evictions == (uint32_t)k_mem_tile_exp_evict);
423}

449mem_stream_window(ra8_vmem_stream_t* st, uint64_t off, uint32_t len, uint32_t* crc, size_t* out_got)
450{
451 if (len > (uint32_t)k_mem_win_buf_bytes) {
452 return false;
453 }
454 const size_t got = ra8_vmem_stream_read(st, off, s_win_buf, (size_t)len);
455 for (size_t i = 0U; i < got; i++) {
456 if (s_win_buf[i] != mem_gen_byte(off + (uint64_t)i)) {
457 return false;
458 }
459 *crc = mem_crc32_byte(*crc, s_win_buf[i]);
460 }
461 *out_got = got;
462 return true;
463}
