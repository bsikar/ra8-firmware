4"""Gate: a header under a ``src/`` directory shall be module-private.
5
6The repository splits every module into a public ``inc/`` directory (the
7contract other translation units consume) and a private ``src/`` directory
8(the implementation). A header that lives under ``src/`` is therefore, by
9construction, module-private -- and must announce that with an ``_internal``
10suffix (``*_internal.h``). A non-``_internal`` header under ``src/`` is one of
11two mistakes:
12
13 * it is actually a public interface that was filed in the wrong place and
14 belongs in the module's ``inc/`` directory, or
15 * it is genuinely private but was never given the ``_internal`` name that
16 marks it as such.
17
18Either way it is a defect. This gate walks every first-party ``.h``/``.hpp``
19under a ``src/`` directory and fails on any that do not end in ``_internal``.
20Vendored trees (``libs/third_party/``) and generated font tables
21(``libs/ra8_fonts/``) are out of scope, matching every other repo gate.
22
23There is deliberately NO in-file waiver marker: the fix is to move the header
24to ``inc/`` or rename it ``*_internal.h``, never to annotate an exception.
25
26Run::
27
28 check_header_file_placement.py # scan the whole tree
29 check_header_file_placement.py path/to/file.h ... # scan listed files
30
31Exit 0 if every ``src/`` header is ``*_internal``, exit 1 (with a table)
32otherwise.
33"""
34

47HEADER_SUFFIXES = (".h", ".hpp", ".hh", ".hxx")
48SCAN_ROOTS = ("libs", "src", "port", "examples", "tools")
49EXCLUDE_FRAGMENTS = (
50 "libs/third_party/",
51 "libs/ra8_fonts/",

55# The suffix that marks a src/ header as intentionally module-private.

59def _is_excluded(path: Path) -> bool:
60 return is_build_output_path(path) or any(frag in str(path) for frag in EXCLUDE_FRAGMENTS)
61
62
63def _is_header(path: Path) -> bool:
64 return path.suffix in HEADER_SUFFIXES
65
66
67def _governing_dir(path: Path) -> str | None:
68 """Return the nearest ``inc``/``src`` ancestor component, or None.
69
70 A module may nest an ``inc`` inside a ``src`` tree (e.g.
71 ``src/secure_app/inc/key_vault.h``): the *closest* such component to the
72 file decides whether it is public (``inc``) or private (``src``), so a
73 higher ``src`` does not condemn a header that sits in a deeper ``inc``.
74 """
75 for part in reversed(path.parent.parts):
76 if part in ("inc", "src"):
77 return part
78 return None
79
80
81def _under_src(path: Path) -> bool:
82 """True if the header's nearest inc/src ancestor is a private ``src``."""
83 return _governing_dir(path) == "src"
84
85
86def _is_internal(path: Path) -> bool:
87 """True if the header's stem ends in the ``_internal`` marker."""
88 return path.stem.endswith(INTERNAL_STEM_SUFFIX)
89
90
91def _rel(path: Path) -> str:
92 if path.is_relative_to(REPO_ROOT):
93 return str(path.relative_to(REPO_ROOT))
94 return str(path)

97def _enumerate_targets(arg_paths: Iterable[str]) -> list[Path]:
98 """Resolve the list of headers to scan from CLI arguments."""
99 args = list(arg_paths)
100 if args:
101 out: list[Path] = []
102 for raw in args:
103 path = Path(raw)
104 if not path.is_absolute():
105 path = REPO_ROOT / path
106 if path.is_dir():
107 for suffix in HEADER_SUFFIXES:
108 out.extend(path.rglob("*" + suffix))
109 elif _is_header(path):
110 out.append(path)
111 return [p for p in out if not _is_excluded(p)]
112
113 out = []
114 for root in SCAN_ROOTS:
115 for suffix in HEADER_SUFFIXES:
116 out.extend((REPO_ROOT / root).rglob("*" + suffix))
117 return [p for p in out if not _is_excluded(p)]

120def main(argv: list[str]) -> int:
121 """Fail any header under a ``src/`` directory not named ``*_internal.h``.
122
123 The scanned count reported on success is the number of headers actually
124 UNDER a src/ directory, not the number handed in: everything else is
125 filtered out first, so passing the whole staged file list is normal and
126 the printed total will legitimately be far smaller than argv.
127
128 Returns 1 listing each misplaced header, 0 when every src/ header is
129 module-private or when nothing in scope reached the filter.
130 """
131 targets = _enumerate_targets(argv[1:])
132 if not targets:
133 print("check_header_file_placement.py: no headers to scan", file=sys.stderr)
134 return 0
135
136 scanned = 0
137 offenders = []
138 for path in targets:
139 if not _under_src(path):
140 continue
141 scanned += 1
142 if not _is_internal(path):
143 offenders.append(_rel(path))
144
145 if not offenders:
146 print(
147 f"check_header_file_placement.py: {scanned} src/ header(s) scanned, "
148 "all module-private (*_internal.h)."
149 )
150 return 0
151
152 offenders.sort()
153 print(
154 f"check_header_file_placement.py: {len(offenders)} src/ header(s) are not *_internal.h:\n",
155 file=sys.stderr,
156 )
157 for path in offenders:
158 print(f" {path}", file=sys.stderr)
159 print(
160 "\nA header under a src/ directory is module-private and must say so.\n"
161 "For each offender, decide which it is and fix at the root:\n"
162 " - public interface (consumed outside the module) -> move it to the\n"
163 " module's inc/ directory;\n"
164 " - genuinely module-private -> rename it '*_internal.h'.\n"
165 "Update every #include of the header in the same change. There is no\n"
166 "waiver marker -- placement is the contract.",
167 file=sys.stderr,
168 )
169 return 1

173 sys.exit(main(sys.argv))
