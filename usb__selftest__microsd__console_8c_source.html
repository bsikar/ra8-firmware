125[[nodiscard]] ra8_err_t microsd_print_dec(uint32_t value)
126{
127 uint8_t scratch[k_microsd_dec_chars_u32] = {};
128 uint8_t out[k_microsd_dec_chars_u32] = {};
129 uint8_t count = 0U;
130 uint32_t v = value;
131 if (v == 0U) {
132 out[0] = (uint8_t)'0';
133 return microsd_sci_write(out, 1U);
134 }
135 while (v != 0U) {
136 if (count >= (uint8_t)k_microsd_dec_chars_u32) {
137 break;
138 }
139 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_microsd_dec_radix));
140 v = v / k_microsd_dec_radix;
141 count++;
142 }
143 for (uint8_t i = 0U; i < count; i++) {
144 out[i] = scratch[count - 1U - i];
145 }
146 return microsd_sci_write(out, (uint32_t)count);
147}

150[[nodiscard]] ra8_err_t microsd_print_hex(uint32_t value, uint8_t digits)
151{
152 uint8_t out[k_microsd_hex_chars_u32] = {};
153 uint8_t width = digits;
154 if (width > (uint8_t)k_microsd_hex_chars_u32) {
155 width = (uint8_t)k_microsd_hex_chars_u32;
156 }
157 for (uint8_t i = 0U; i < width; i++) {
158 const uint8_t shift = (uint8_t)((width - 1U - i) * k_microsd_nibble_bits);
159 out[i] = microsd_nibble_to_hex((value >> shift) & k_microsd_nibble_mask);
160 }
161 return microsd_sci_write(out, (uint32_t)width);
162}
