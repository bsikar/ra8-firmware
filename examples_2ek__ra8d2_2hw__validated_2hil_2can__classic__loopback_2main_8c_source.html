183[[nodiscard]] static ra8_err_t can_demo_one_round_trip(uint8_t seq)
184{
185 ra8_canfd_frame_t tx = {
186 .id = (uint32_t)k_can_demo_id,
187 .dlc = (uint8_t)k_can_demo_dlc,
188 .is_extended = 0U,
189 .is_fd = 0U, /* classic */
190 .is_brs = 0U,
191 .data = {seq,
192 (uint8_t)k_can_demo_byte_marker_a,
193 (uint8_t)k_can_demo_byte_marker_b,
194 (uint8_t)k_can_demo_byte_marker_c,
195 (uint8_t)k_can_demo_byte_marker_d,
196 (uint8_t)k_can_demo_byte_marker_e,
197 (uint8_t)k_can_demo_byte_marker_f,
198 (uint8_t)k_can_demo_byte_marker_g},
199 };
200 if (ra8_canfd_transmit((uint8_t)k_can_demo_channel, &tx) != k_ra8_ok) {
201 return k_ra8_err_hw_error;
202 }
203 ra8_canfd_frame_t rx = {};
204 if (ra8_canfd_receive((uint8_t)k_can_demo_channel, &rx) != k_ra8_ok) {
205 return k_ra8_err_no_data;
206 }
207 return k_ra8_ok;
208}

212int32_t main(void)
213{
214 can_demo_setup_or_halt();
215 ra8_isr_globals_enable();
216
217 uint8_t seq = 0U;
218 while (1) {
219 if (can_demo_one_round_trip(seq) == k_ra8_ok) {
220 (void)ra8_board_led_toggle(k_ra8_board_led1);
221 g_can_match += 1U;
222 } else {
223 (void)ra8_board_led_toggle(k_ra8_board_led2);
224 g_can_mismatch += 1U;
225 }
226 seq++;
227 ra8_delay_ms((uint32_t)k_can_demo_period_ms);
228 }
229 can_demo_panic_halt();
230 return 0;
231}
