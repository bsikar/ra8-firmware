138RA8_INTERNAL
139static ra8_err_t internal_lcd_validate_cfg(const display_cfg_t* cfg)
140{
141 RA8_CHECK_NULL_PTR(cfg->framebuffer, s_tag, "cfg->framebuffer");
142 if (cfg->width_px == 0U || cfg->height_px == 0U) {
143 return k_ra8_err_invalid_arg;
144 }
145 if (cfg->pixfmt != k_display_pixfmt_rgb565) {
146 return k_ra8_err_not_supported;
147 }
148 /* The GLCDC backend needs the panel's RGB timing; the board BSP supplies it
149 * (e.g. &k_ra8_panel_ek_ra8d2_timing). Host / e-ink backends leave it null. */
150 if (cfg->panel_timing == nullptr) {
151 return k_ra8_err_invalid_arg;
152 }
153 const uint32_t need_bytes =
154 (uint32_t)cfg->width_px * (uint32_t)cfg->height_px * (uint32_t)k_lcd_rgb565_bpp;
155 if (cfg->framebuffer_bytes < need_bytes) {
156 return k_ra8_err_invalid_arg;
157 }
158 return k_ra8_ok;
159}

189RA8_INTERNAL
190static ra8_err_t internal_lcd_bringup_panel(const display_cfg_t* cfg)
191{
192 ra8_err_t err = ra8_board_lcd_panel_power_on();
193 if (err != k_ra8_ok) {
194 return err;
195 }
196 err = ra8_board_glcdc_init(k_ra8_board_glcdc_fmt_rgb888);
197 if (err != k_ra8_ok) {
198 return err;
199 }
200 ra8_delay_ms((uint32_t)k_lcd_bringup_settle_ms);
201
202 const ra8_glcdc_config_t glcdc_cfg = {
203 .framebuffer_addr = (uint32_t)(uintptr_t)cfg->framebuffer,
204 .width_px = cfg->width_px,
205 .height_px = cfg->height_px,
206 .format = k_ra8_glcdc_fmt_rgb565,
207 .timing = *(const ra8_glcdc_timing_t*)cfg->panel_timing,
208 };
209 err = ra8_glcdc_init(&glcdc_cfg);
210 if (err != k_ra8_ok) {
211 return err;
212 }
213 err = ra8_glcdc_set_background_color((uint32_t)k_lcd_bg_color_black);
214 if (err != k_ra8_ok) {
215 return err;
216 }
217 err = ra8_glcdc_start(true);
218 if (err != k_ra8_ok) {
219 return err;
220 }
221 return ra8_glcdc_layer1_show((uintptr_t)cfg->framebuffer);
222}

336RA8_INTERNAL
337static ra8_err_t lcd_init(const display_cfg_t* cfg, void** out_ctx)
338{
339 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg");
340 RA8_CHECK_NULL_PTR(out_ctx, s_tag, "out_ctx");
341
342 const ra8_err_t v = internal_lcd_validate_cfg(cfg);
343 if (v != k_ra8_ok) {
344 return v;
345 }
346 if (s_lcd_ctx.started) {
347 ra8_log_error(s_tag, "lcd_init: already started");
348 return k_ra8_err_busy;
349 }
350 const ra8_err_t err = internal_lcd_bringup_panel(cfg);
351 if (err != k_ra8_ok) {
352 return err;
353 }
354 internal_lcd_snapshot(cfg);
355 *out_ctx = &s_lcd_ctx;
356 return k_ra8_ok;
357}

449RA8_INTERNAL
450static ra8_err_t lcd_flush(void* ctx, display_rect_t rect, display_refresh_hint_t hint)
451{
452 (void)hint; /* LCD scans continuously; hint is purely informational. */
453 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx");
454 const lcd_ctx_t* c = (const lcd_ctx_t*)ctx;
455 const ra8_err_t v = internal_lcd_check_rect(&c->caps, rect);
456 if (v != k_ra8_ok) {
457 return v;
458 }
459#ifdef RA8_BOOT_ENABLE_CACHE_MPU
460 /* D-cache is enabled for this build: the CPU paints the cacheable SDRAM
461 * framebuffer through the write-back L1 D-cache, but the GLCDC scans SDRAM
462 * directly over AXI and never sees the cache. Clean (write back) the painted
463 * rows so the next scan latches the new pixels instead of stale SDRAM. The
464 * clean covers the flushed rect's contiguous row-span [y, y+h) -- a superset
465 * of the rect, exact for the full-width flushes the reader issues, and safe
466 * for a partial-width rect (whole rows, never a strided w*h slice).
467 * ra8_cache_dcache_clean_by_addr rounds to whole cache lines internally. */
468 {
469 const uint8_t* row_base = (const uint8_t*)c->fb.pixels + (uint32_t)rect.y * c->fb.stride_bytes;
470 const uint32_t row_span = (uint32_t)rect.h * c->fb.stride_bytes;
471 (void)ra8_cache_dcache_clean_by_addr(row_base, row_span);
472 }
473#endif
474 /* Memory barrier so any pixel writes the caller did with NEON / DMA
475 * are visible to the GLCDC's AXI controller before the next scan. */
476 internal_lcd_dsb();
477 return k_ra8_ok;
478}

502RA8_INTERNAL
503static ra8_err_t lcd_clear(void* ctx, uint32_t color)
504{
505 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx");
506 lcd_ctx_t* c = (lcd_ctx_t*)ctx;
507 uint16_t* pixels = (uint16_t*)c->fb.pixels;
508 const uint32_t pixels_total = (uint32_t)c->fb.width_px * (uint32_t)c->fb.height_px;
509 const uint16_t rgb565 = (uint16_t)(color & k_rgb565_mask);
510 for (uint32_t i = 0U; i < pixels_total; ++i) {
511 pixels[i] = rgb565;
512 }
513#ifdef RA8_BOOT_ENABLE_CACHE_MPU
514 /* Write the cleared pixels back from the write-back D-cache to SDRAM so the
515 * GLCDC scan sees them (see lcd_flush for the coherency rationale). The whole
516 * framebuffer was just touched, so clean the whole contiguous span. */
517 (void)ra8_cache_dcache_clean_by_addr(c->fb.pixels,
518 (uint32_t)c->fb.stride_bytes * (uint32_t)c->fb.height_px);
519#endif
520 internal_lcd_dsb();
521 return k_ra8_ok;
522}
