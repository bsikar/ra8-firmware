4"""Wrap a raw application binary into a dfu_bootloader slot image.
5
6Produces the body at the slot base plus the 32-byte image header in the slot's
7LAST page (slot_base + 0x6FFE0), emitted as an Intel HEX that J-Link `loadfile`
8can flash. That stages a slot WITHOUT a USB host -- the bench path for
9validating the bootloader's copy-to-run behaviour.
10
11This is NOT the operator path. Operators use dfu-util, and the bootloader
12writes the header itself on DFU commit, so that path needs only the body
13binary and never this tool.
14
15Copy-to-run is why one image works in either slot: the bootloader copies the
16slot body to the fixed SRAM run base (RUN_BASE) and launches it there, so the
17IDENTICAL app.bin can be staged to slot a or slot b. The consequence is a hard
18requirement on the input -- the application MUST be linked at the SRAM run base
19(ORIGIN = RUN_BASE), NOT at a slot base. An image linked at a slot base stages
20and flashes cleanly and then faults on launch. See this app's README, "One
21image, either slot (copy-to-run)".
22
23The header layout and CRC must stay in lock-step with libs/ra8_dfu
24(header-last)::
25
26 magic(u32 = 0x52413844) seq(u32) img_len(u32, multiple of 32)
27 img_crc32(u32, CRC32 over [slot_base, slot_base + img_len))
28 entry(u32 = RUN_BASE) reserved(12 bytes, zero)
29
30CRC32 is the standard reflected zlib polynomial.
31
32Usage:
33 stage_slot_image.py --payload app.bin --slot a [--seq 1] --out slotA.hex
34"""
35

42# Bank map -- keep in lock-step with libs/ra8_dfu/inc/ra8_dfu.h.
43SLOT_BASE = {"a": 0x02020000, "b": 0x02090000}
44SLOT_SIZE = 0x00070000
45HDR_OFFSET = 0x0006FFE0
46HDR_MAGIC = 0x52413844
47RUN_BASE = 0x22020000 # k_ra8_dfu_run_base: SRAM copy-to-run / payload link base
48PAGE = 0x20

51def ihex_records(data: bytes, base: int) -> list:
52 """Emit Intel-HEX records (ELA + data) placing data at absolute base."""
53 out = []
54 hi = (base >> 16) & 0xFFFF
55 rec = bytes([2, 0, 0, 4, (hi >> 8) & 0xFF, hi & 0xFF])
56 out.append(":" + rec.hex().upper() + f"{((-sum(rec)) & 0xFF):02X}")
57 off = base & 0xFFFF
58 for i in range(0, len(data), 16):
59 chunk = data[i : i + 16]
60 addr = off + i
61 rec = bytes([len(chunk), (addr >> 8) & 0xFF, addr & 0xFF, 0]) + chunk
62 out.append(":" + rec.hex().upper() + f"{((-sum(rec)) & 0xFF):02X}")
63 return out

91def _build_slot_body(args: argparse.Namespace) -> tuple[int, int, bytes] | None:
92 """Return ``(img_len, crc, slot_bytes)``, or None after reporting a fault.
93
94 The two layouts differ in what the header covers. A plain payload is
95 FF-padded to a page multiple and the CRC spans all of it. A signed slot is
96 ``[body][ra8_rot_trailer_t]``: the header's img_len and CRC cover the BODY
97 only, and the trailer is staged contiguously right after it so the RoT
98 launch gate finds it at ``slot_base + img_len`` (see ra8_rot_trailer_after).
99 The bootloader copies only img_len body bytes to the run base.
100 """
101 raw = bytearray(Path(args.payload).read_bytes())
102 if not args.signed:
103 if len(raw) % PAGE != 0:
104 raw += b"\xff" * (PAGE - (len(raw) % PAGE))
105 return len(raw), zlib.crc32(bytes(raw)) & 0xFFFFFFFF, bytes(raw)
106
107 if len(raw) <= args.trailer_size:
108 sys.stderr.write(f"error: signed image {len(raw)}B <= trailer {args.trailer_size}B\n")
109 return None
110 img_len = len(raw) - args.trailer_size
111 if img_len % PAGE != 0:
112 sys.stderr.write(
113 f"error: signed body {img_len}B is not a multiple of {PAGE} "
114 f"-- sign a page-aligned body\n"
115 )
116 return None
117 return img_len, zlib.crc32(bytes(raw[:img_len])) & 0xFFFFFFFF, bytes(raw)

120def main() -> int:
121 """Build the slot image from the command line and write the Intel HEX.
122
123 Pads the payload to a PAGE multiple, computes the CRC over the padded body,
124 and places the header in the slot's last page. Body and header land at
125 absolute addresses derived from `--slot`, so the output HEX is
126 slot-specific even though the payload is not.
127
128 `--seq` selects which slot the bootloader prefers at boot: the higher
129 sequence number wins. Staging an image with a sequence at or below the other
130 slot's produces a valid image the bootloader will simply not choose --
131 which looks exactly like a flash that did not take.
132
133 Returns:
134 0 on success, non-zero on a usage or validation failure.
135 """
136 args = _build_parser().parse_args()
137
138 base = SLOT_BASE[args.slot]
139 built = _build_slot_body(args)
140 if built is None:
141 return 2
142 img_len, crc, slot = built
143
144 if len(slot) > HDR_OFFSET:
145 sys.stderr.write(f"error: slot content {len(slot)} bytes exceeds capacity {HDR_OFFSET}\n")
146 return 2
147
148 # entry == RUN_BASE (copy-to-run): the bootloader copies img_len body bytes
149 # there and launches it; the same image is valid in either slot.
150 hdr = struct.pack("<5I", HDR_MAGIC, args.seq, img_len, crc, RUN_BASE) + b"\x00" * 12
151
152 records = ihex_records(slot, base)
153 records += ihex_records(hdr, base + HDR_OFFSET)
154 records.append(":00000001FF")
155 with Path(args.out).open("w", encoding="ascii") as f:
156 f.write("\n".join(records) + "\n")
157
158 kind = "signed" if args.signed else "plain"
159 print(
160 f"slot {args.slot.upper()} @0x{base:08X} ({kind}): img_len={img_len}B "
161 f"slot={len(slot)}B crc=0x{crc:08X} hdr@0x{base + HDR_OFFSET:08X} "
162 f"seq={args.seq} -> {args.out}"
163 )
164 return 0
