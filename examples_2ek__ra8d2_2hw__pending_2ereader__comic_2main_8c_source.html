375static void cm_print_hex(uint32_t value)
376{
377 uint8_t buf[k_cm_hex_nibbles];
378 for (uint32_t i = 0U; i < (uint32_t)k_cm_hex_nibbles; i++) {
379 const uint32_t shift = ((uint32_t)k_cm_hex_nibbles - 1U - i) * (uint32_t)k_cm_nibble_bits;
380 const uint32_t nib = (value >> shift) & (uint32_t)k_cm_nibble_mask;
381 buf[i] = (uint8_t)((nib < (uint32_t)k_cm_dec_ten) ? ('0' + nib) : ('A' + (nib - k_cm_dec_ten)));
382 }
383 cm_print(buf, (uint32_t)k_cm_hex_nibbles);
384}

387static void cm_print_uint(uint32_t value)
388{
389 uint8_t buf[k_cm_dec_ten];
390 uint32_t n = 0U;
391 if (value == 0U) {
392 buf[n] = '0';
393 n++;
394 }
395 while ((value > 0U) && (n < (uint32_t)k_cm_dec_ten)) {
396 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_cm_dec_ten));
397 n++;
398 value /= (uint32_t)k_cm_dec_ten;
399 }
400 for (uint32_t i = 0U; i < n; i++) {
401 cm_print(&buf[n - 1U - i], 1U);
402 }
403}

826static cm_zone_t cm_zone_at(int32_t x, int32_t y)
827{
828 const int32_t third = (int32_t)k_cm_fb_w / (int32_t)k_cm_thirds;
829 if (x < third) {
830 return k_cm_zone_prev;
831 }
832 if (x >= (2 * third)) {
833 return k_cm_zone_next;
834 }
835 if (y < ((int32_t)k_cm_fb_h / (int32_t)k_cm_halves)) {
836 return k_cm_zone_chrome;
837 }
838 return k_cm_zone_none;
839}

855static bool cm_handle_tap(int32_t x, int32_t y)
856{
857 switch (cm_zone_at(x, y)) {
858 case k_cm_zone_next:
859 if ((s_page + 1U) < s_page_count) {
860 s_page++;
861 return true;
862 }
863 return false;
864 case k_cm_zone_prev:
865 if (s_page > 0U) {
866 s_page--;
867 return true;
868 }
869 return false;
870 case k_cm_zone_chrome:
871 s_chrome_on = !s_chrome_on;
872 return true;
873 case k_cm_zone_none:
874 default:
875 return false;
876 }
877}
