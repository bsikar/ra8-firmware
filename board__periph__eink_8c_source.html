263static void eink_begin_read(void)
264{
265 uint16_t value = 0U;
266 if (s_eink.cur_cmd == (uint16_t)k_eink_cmd_reg_rd) {
267 value = eink_reg_value(s_eink.reg_addr);
268 } else if (s_eink.cur_cmd == (uint16_t)k_eink_cmd_vcom) {
269 /* Only a "get" reads back; the driver never reads after a set. */
270 value = s_eink.vcom_mv;
271 } else if (s_eink.cur_cmd == (uint16_t)k_eink_cmd_get_dev_info) {
272 value = eink_dev_info_word(s_eink.dev_idx);
273 s_eink.dev_idx = (uint16_t)(s_eink.dev_idx + 1U);
274 }
275 s_eink.rd_buf[k_eink_rd_dummy_hi] = (uint8_t)k_eink_idle_byte;
276 s_eink.rd_buf[k_eink_rd_dummy_lo] = (uint8_t)k_eink_idle_byte;
277 s_eink.rd_buf[k_eink_rd_val_hi] =
278 (uint8_t)((value >> (uint16_t)k_eink_byte_shift) & (uint16_t)k_eink_byte_mask);
279 s_eink.rd_buf[k_eink_rd_val_lo] = (uint8_t)(value & (uint16_t)k_eink_byte_mask);
280 s_eink.rd_len = (uint8_t)k_eink_read_bytes;
281 s_eink.rd_pos = 0U;
282 s_eink.reading = true;
283}

286static void eink_consume_data(uint16_t word)
287{
288 if ((s_eink.cur_cmd == (uint16_t)k_eink_cmd_reg_rd) ||
289 (s_eink.cur_cmd == (uint16_t)k_eink_cmd_reg_wr)) {
290 if (s_eink.datawr_idx == (uint16_t)k_eink_idx_reg_addr) {
291 s_eink.reg_addr = word;
292 }
293 } else if (s_eink.cur_cmd == (uint16_t)k_eink_cmd_vcom) {
294 if (s_eink.datawr_idx == (uint16_t)k_eink_idx_vcom_dir) {
295 s_eink.vcom_dir = word;
296 } else if ((s_eink.datawr_idx == (uint16_t)k_eink_idx_vcom_val) &&
297 (s_eink.vcom_dir == (uint16_t)k_eink_vcom_set)) {
298 s_eink.vcom_mv = word;
299 } else {
300 /* A get takes no value word; nothing further to consume. */
301 }
302 } else if (s_eink.cur_cmd == (uint16_t)k_eink_cmd_ld_img_area) {
303 if (s_eink.datawr_idx == (uint16_t)k_eink_idx_ld_arg0) {
304 const uint16_t wire_pf =
305 (uint16_t)((word >> (uint16_t)k_eink_pf_shift) & (uint16_t)k_eink_pf_mask);
306 s_eink.px_per_word = eink_px_per_word(wire_pf);
307 } else if (s_eink.datawr_idx >= (uint16_t)k_eink_idx_ld_px_start) {
308 s_eink.pixels += (uint64_t)s_eink.px_per_word;
309 } else {
310 /* Geometry args: no pixel accounting. */
311 }
312 } else if (s_eink.cur_cmd == (uint16_t)k_eink_cmd_dpy_area) {
313 if (s_eink.datawr_idx == (uint16_t)k_eink_idx_dpy_wf) {
314 s_eink.last_wf = word;
315 s_eink.refreshes = (uint32_t)(s_eink.refreshes + 1U);
316 }
317 }
318}

321static void eink_consume_word(uint16_t word)
322{
323 if (s_eink.st == (eink_state_t)k_eink_st_cmd) {
324 s_eink.cur_cmd = word;
325 s_eink.datawr_idx = 0U;
326 if (word == (uint16_t)k_eink_cmd_get_dev_info) {
327 s_eink.dev_idx = 0U;
328 }
329 s_eink.st = (eink_state_t)k_eink_st_preamble;
330 return;
331 }
332 if (s_eink.st == (eink_state_t)k_eink_st_datawr) {
333 eink_consume_data(word);
334 s_eink.datawr_idx = (uint16_t)(s_eink.datawr_idx + 1U);
335 s_eink.st = (eink_state_t)k_eink_st_preamble;
336 return;
337 }
338 /* k_eink_st_preamble: classify the preamble word. */
339 if (word == (uint16_t)k_eink_pre_cmd) {
340 s_eink.st = (eink_state_t)k_eink_st_cmd;
341 } else if (word == (uint16_t)k_eink_pre_wr) {
342 s_eink.st = (eink_state_t)k_eink_st_datawr;
343 } else if (word == (uint16_t)k_eink_pre_rd) {
344 eink_begin_read(); /* subsequent dummy bytes are answered from rd_buf */
345 }

360uint8_t board_eink_exchange(uint8_t tx)
361{
362 if (!s_eink.attached) {
363 return (uint8_t)k_eink_idle_byte;
364 }
365 if (s_eink.reading) {
366 const uint8_t r =
367 (s_eink.rd_pos < s_eink.rd_len) ? s_eink.rd_buf[s_eink.rd_pos] : (uint8_t)k_eink_idle_byte;
368 s_eink.rd_pos = (uint8_t)(s_eink.rd_pos + 1U);
369 if (s_eink.rd_pos >= s_eink.rd_len) {
370 s_eink.reading = false;
371 }
372 return r;
373 }
374 if (!s_eink.have_hi) {
375 s_eink.hi = tx;
376 s_eink.have_hi = true;
377 return (uint8_t)k_eink_idle_byte;
378 }
379 const uint16_t word =
380 (uint16_t)(((uint16_t)s_eink.hi << (uint16_t)k_eink_byte_shift) | (uint16_t)tx);
381 s_eink.have_hi = false;
382 eink_consume_word(word);
383 return (uint8_t)k_eink_idle_byte;
384}

386void board_eink_reset(void)
387{
388 const bool was_attached = s_eink.attached;
389 const uint64_t pixels = s_eink.pixels;
390 const uint32_t refreshes = s_eink.refreshes;
391 const uint16_t vcom_mv = s_eink.vcom_mv;
392 s_eink = (eink_model_t){};
393 s_eink.attached = was_attached;
394 s_eink.pixels = pixels; /* keep run totals across a mid-run reset */
395 s_eink.refreshes = refreshes; /* (the SPI block resets framing, not the panel) */
396 s_eink.st = (eink_state_t)k_eink_st_preamble;
397 /* VCOM is controller configuration, not bus framing: a mid-run SPI reset
398 * must not clear it, but a cold attach starts from the power-on value. */
399 s_eink.vcom_mv = (vcom_mv != 0U) ? vcom_mv : (uint16_t)k_eink_vcom_power_on_mv;
400 s_eink.px_per_word = eink_px_per_word((uint16_t)k_eink_wire_pf_8bpp);
401}
