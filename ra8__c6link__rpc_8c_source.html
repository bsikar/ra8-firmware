64RA8_INTERNAL static ra8_err_t ra8_c6link_rpc_stage(ra8_c6link_t* link, Rpc* req)
65{
66 link->tx_len = 0U;
67
68 const size_t packed = rpc__get_packed_size(req);
69 if (packed > (size_t)k_ra8_c6link_max_payload) {
70 return k_ra8_err_invalid_size;
71 }
72
73 uint8_t* payload = &link->tx[k_ra8_c6link_header_bytes];
74 uint16_t body_at = 0U;
75 const ra8_err_t opened = ra8_c6link_priv_tlv_open(payload,
76 (uint16_t)k_ra8_c6link_max_payload,
77 (uint16_t)packed,
78 &body_at);
79 if (opened != k_ra8_ok) {
80 return opened;
81 }
82 if (rpc__pack(req, &payload[body_at]) != packed) {
83 return k_ra8_err_validation_failed;
84 }
85
86 link->tx_len = (uint16_t)((uint32_t)body_at + (uint32_t)packed);
87 link->tx_if = (uint8_t)ESP_SERIAL_IF;
88 return k_ra8_ok;
89}

106RA8_PRIV ra8_err_t ra8_c6link_priv_rpc_call(ra8_c6link_t* link,
107 Rpc* req,
108 uint32_t resp_id,
109 ra8_c6link_take_fn_t take,
110 void* take_ctx)
111{
112 if ((link == nullptr) || (req == nullptr) || (take == nullptr)) {
113 return k_ra8_err_null_ptr;
114 }
115 if (!link->open) {
116 return k_ra8_err_not_initialized;
117 }
118 if (link->wait.armed || (link->tx_len != 0U)) {
119 return k_ra8_err_busy;
120 }
121
122 link->next_uid = link->next_uid + 1U;
123 req->uid = link->next_uid;
124
125 const ra8_err_t staged = ra8_c6link_rpc_stage(link, req);
126 if (staged != k_ra8_ok) {
127 return staged;
128 }
129
130 link->wait = (ra8_c6link_wait_t){
131 .uid = req->uid,
132 .resp_id = resp_id,
133 .take = take,
134 .take_ctx = take_ctx,
135 .result = k_ra8_ok,
136 .armed = true,
137 .satisfied = false,
138 };
139
140 ra8_c6link_stats_t stats = {};
141 const ra8_err_t pumped = ra8_c6link_priv_pump(link, (uint16_t)k_ra8_c6link_rpc_transfers, &stats);
142 const bool got = link->wait.satisfied;
143 const ra8_err_t result = link->wait.result;
144 link->wait = (ra8_c6link_wait_t){};
145 link->tx_len = 0U;
146
147 if (pumped != k_ra8_ok) {
148 return pumped;
149 }
150 if (!got) {
151 link->fault.rpc_id = (uint32_t)req->msg_id;
152 link->fault.resp = 0;
153 return k_ra8_err_timeout;
154 }
155 return result;
156}

227RA8_INTERNAL static void ra8_c6link_rpc_event(ra8_c6link_t* link, const Rpc* msg)
228{
229 ra8_c6link_event_t ev = {};
230
231 switch ((int32_t)msg->msg_id) {
232 case (int32_t)RPC_ID__Event_ESPInit:
233 ev.kind = k_ra8_c6link_event_boot;
234 if (msg->event_esp_init != nullptr) {
235 ev.reset_reason = msg->event_esp_init->cp_reset_reason;
236 }
237 break;
238 case (int32_t)RPC_ID__Event_StaConnected:
239 ev.kind = k_ra8_c6link_event_sta_connected;
240 if (msg->event_sta_connected != nullptr) {
241 ra8_c6link_rpc_ev_connected(&ev, msg->event_sta_connected->sta_connected);
242 }
243 break;
244 case (int32_t)RPC_ID__Event_StaDisconnected:
245 ev.kind = k_ra8_c6link_event_sta_disconnected;
246 if (msg->event_sta_disconnected != nullptr) {
247 ra8_c6link_rpc_ev_disconnected(&ev, msg->event_sta_disconnected->sta_disconnected);
248 }
249 break;
250 case (int32_t)RPC_ID__Event_WifiEventNoArgs:
251 ev.kind = k_ra8_c6link_event_wifi;
252 if (msg->event_wifi_event_no_args != nullptr) {
253 ev.wifi_event_id = msg->event_wifi_event_no_args->event_id;
254 }
255 break;
256 default:
257 return;
258 }
259 ra8_c6link_priv_emit(link, &ev);
260}

284RA8_INTERNAL static bool ra8_c6link_rpc_answer(ra8_c6link_t* link, const Rpc* msg)
285{
286 if (!link->wait.armed || (msg->uid != link->wait.uid) ||
287 ((uint32_t)msg->msg_id != link->wait.resp_id)) {
288 return false;
289 }
290 link->wait.result = link->wait.take(link->wait.take_ctx, msg);
291 link->wait.satisfied = true;
292 return true;
293}

295RA8_PRIV bool ra8_c6link_priv_rpc_consume(ra8_c6link_t* link, const uint8_t* payload, uint16_t len)
296{
297 if ((link == nullptr) || (payload == nullptr)) {
298 return false;
299 }
300
301 uint16_t proto_len = 0U;
302 const uint8_t* proto = ra8_c6link_priv_tlv_body(payload, len, &proto_len);
303 if (proto == nullptr) {
304 if (link->stats != nullptr) {
305 link->stats->undecodable++;
306 }
307 return false;
308 }
309
310 ProtobufCAllocator alloc = {};
311 ra8_c6link_priv_arena_bind(&alloc, link);
312 ra8_c6link_priv_arena_reset(link);
313
314 Rpc* msg = rpc__unpack(&alloc, (size_t)proto_len, proto);
315 if (msg == nullptr) {
316 if (link->stats != nullptr) {
317 link->stats->undecodable++;
318 }
319 ra8_c6link_priv_arena_reset(link);
320 return false;
321 }
322 if (link->stats != nullptr) {
323 link->stats->rpc_in++;
324 }
325
326 bool stop = false;
327 if (msg->msg_type == RPC_TYPE__Event) {
328 ra8_c6link_rpc_event(link, msg);
329 } else if (msg->msg_type == RPC_TYPE__Resp) {
330 stop = ra8_c6link_rpc_answer(link, msg);
331 } else {
332 /* A request arriving at a host is a co-processor defect, not a message
333 this side has any handler for. Counted, dropped, never acted on. */
334 if (link->stats != nullptr) {
335 link->stats->undecodable++;
336 }
337 }
338
339 rpc__free_unpacked(msg, &alloc);
340 ra8_c6link_priv_arena_reset(link);
341 return stop;
342}
