159static uint8_t internal_i3c_i2c_half_period(uint32_t bus_hz, uint32_t pclka_hz)
160{
161 /* mcdc-deactivated: both args validated by internal_i3c_i2c_init upstream; defensive duplicate. */
162 if ((bus_hz == 0U) || (pclka_hz == 0U)) {
163 return 0U;
164 }
165 enum : uint32_t {
166 k_ra8_i3c_i2c_period_split = 2U,
167 k_ra8_i3c_i2c_max_field = 0xFFU,
168 };
169 const uint32_t total = pclka_hz / bus_hz;
170 const uint32_t half = total / k_ra8_i3c_i2c_period_split;
171 if (half == 0U) {
172 return 0U;
173 }
174 uint32_t value = half - 1U;
175 if (value > k_ra8_i3c_i2c_max_field) {
176 value = k_ra8_i3c_i2c_max_field;
177 }
178 return (uint8_t)value;
179}

518static ra8_err_t internal_i3c_i2c_block_bringup(volatile r_i3c_i2c_regs_t* reg)
519{
520 /* The I3C/IIC_B shared block needs both MSTPB4 (I3C) AND MSTPB9 */
521 /* (IIC0) ungated; ungating only MSTPB4 leaves the channel-0 IIC */
522 /* controller powered down and any START hangs in BCST before */
523 /* TDBEF0 ever rises. Mirrors */
524 /* ra8_board_ek_ra8d2.c::internal_io_expander_apply. */
525 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B" p 444 */
526 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_i3c);
527 RA8_RETURN_ON_ERROR(mst_err, s_tag, "iic_b_init: mstp i3c"); /* GCOVR_EXCL_BR_LINE */
528 const ra8_err_t iic0_err = ra8_mstp_enable(k_ra8_mstp_iic0);
529 RA8_RETURN_ON_ERROR(iic0_err, s_tag, "iic_b_init: mstp iic0"); /* GCOVR_EXCL_BR_LINE */
530
531 /* HUM Ch 40.2.92 "CECTL : Clock Enable Control Register" p 2543 */
532 reg->CECTL = k_ra8_i3c_i2c_msk_cectl_clke;
533 /* BUSE=0: detach SCL/SDA before reset.
534 * HUM Ch 40.2.2 "BCTL : Bus Control Register" p 2449 */
535 reg->BCTL = 0U;
536
537 const ra8_err_t reset_err = internal_i3c_i2c_reset(reg);
538 RA8_RETURN_ON_ERROR(reset_err, s_tag, "iic_b_init: reset"); /* GCOVR_EXCL_BR_LINE */
539
540 /* HUM Ch 40.2.1 "PRTS : Protocol Selection" p 2449 -- the I3C IP
541 * defaults to PRTMD=0 (I3C FIFO buffer transfer, command-queue
542 * driven). This polling driver writes NTDTBP0 directly, so it must
543 * switch the IP into PRTMD=1 (I2C single-buffer transfer) before
544 * any START / STOP / address byte is issued. Without this write the
545 * peripheral silently swallows writes to NTDTBP0 as stray I3C
546 * FIFO entries and never drives SCL. */
547 reg->PRTS = k_ra8_i3c_i2c_msk_prts_prtmd;
548 return k_ra8_ok;
549}

741 const uint8_t* data,
742 uint32_t len,
743 bool restart)
744{
745 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
746 RA8_CHECK_NULL_PTR(reg, s_tag, "iic_b_write: channel");
747 RA8_CHECK_NULL_PTR(data, s_tag, "iic_b_write: data");
748
749 const bool bus_held = s_iic_b_state[channel].bus_held;

756 internal_i3c_i2c_open_phase(reg, bus_held);

804internal_i3c_i2c_drain_rx(volatile r_i3c_i2c_regs_t* reg, uint8_t* out, uint32_t len)
805{
806 ra8_err_t err = k_ra8_ok;
807 for (uint32_t i = 0U; i < len; i++) {
808 if (i == (len - 1U)) {
809 /* HUM Ch 40.2.24 "ACKCTL : Acknowledge Control Register" p 2473 */
810 reg->ACKCTL = k_ra8_i3c_i2c_msk_ackctl_acktwp | k_ra8_i3c_i2c_msk_ackctl_ackt;
811 }
812 err = internal_i3c_i2c_wait_ntst(reg, k_ra8_i3c_i2c_msk_ntst_rdbff0);
813 if (err != k_ra8_ok) {
814 break;
815 }
816 /* HUM Ch 40.2.35 "NTDTBP0 : Normal Transfer Data Buffer Port 0" p 2481 */
817 out[i] = (uint8_t)(reg->NTDTBP0 & k_ra8_i3c_i2c_byte_mask);

903ra8_err_t internal_i3c_i2c_transfer(uint8_t channel,
904 uint8_t target_7b,
905 const uint8_t* tx,
906 uint32_t tx_len,
907 uint8_t* rx,
908 uint32_t rx_len)
909{
910 if (i3c_i2c_regs(channel) == nullptr) {
911 return k_ra8_err_null_ptr;
912 }
913 if ((tx_len == 0U) && (rx_len == 0U)) {
914 return k_ra8_err_invalid_arg;
915 }
916 if ((tx_len != 0U) && (tx == nullptr)) {
917 return k_ra8_err_null_ptr;
918 }
919 if (internal_i3c_i2c_len_buf_invalid(rx_len, rx)) {
920 return k_ra8_err_null_ptr;
921 }
922
923 /* Phase 1: write (hold the bus for the upcoming read).
924 * Skipped entirely when tx_len == 0 -- this degenerates the call
925 * into a plain read, matching FSP's expectation that a zero-byte
926 * tx phase means "no register-pointer update needed". */
927 if (tx_len != 0U) {
928 const ra8_err_t w_err =
929 internal_i3c_i2c_write(channel, target_7b, tx, tx_len, /*restart=*/(rx_len != 0U));
930 if (w_err != k_ra8_ok) {
931 return w_err;
932 }
933 }
934
935 /* Phase 2: read (close the bus normally). */
936 if (rx_len != 0U) {
937 const ra8_err_t r_err =
938 internal_i3c_i2c_read(channel, target_7b, rx, rx_len, /*restart=*/false);
939 if (r_err != k_ra8_ok) {
940 return r_err;
941 }
