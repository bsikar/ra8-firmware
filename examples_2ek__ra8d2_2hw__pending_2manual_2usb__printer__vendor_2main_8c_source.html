208static ra8_err_t demo_printer_setup(void* ctx, const ra8_usb_setup_t* setup)
209{
210 (void)ctx;
211 if (setup == nullptr) {
212 return k_ra8_err_null_ptr;
213 }
214 if (setup->b_request == (uint8_t)k_pv_pprn_get_port_status) {
215 uint8_t status = 0U;
216 const ra8_err_t st_err = ra8_usb_pprn_get_port_status(&status);
217 if (st_err != k_ra8_ok) {
218 return st_err;
219 }
220 return ra8_usb_dcp_in_data(k_ra8_usb_speed_fs, &status, 1U);
221 }
222 if (setup->b_request == (uint8_t)k_pv_pprn_get_device_id) {
223 const uint16_t want = pv_clamp_len((uint16_t)sizeof(s_device_id), setup->w_length);
224 return ra8_usb_dcp_in_data(k_ra8_usb_speed_fs, s_device_id, want);
225 }
226 return k_ra8_ok; /* SOFT_RESET: no data stage. */
227}

307static void demo_dispatch_setup(const ra8_usb_setup_t* setup)
308{
309 const pv_action_t action = pv_route_setup(setup);
310 ra8_err_t err = k_ra8_ok;
311 switch (action) {
312 case k_pv_action_get_device_desc:
313 case k_pv_action_get_config_desc:
314 case k_pv_action_get_string_desc:
315 err = demo_stage_descriptor(action, setup);
316 break;
317 case k_pv_action_set_address:
318 err = ra8_usb_set_address(k_ra8_usb_speed_fs, (uint8_t)setup->w_value);
319 if (err == k_ra8_ok) {
320 err = ra8_usb_control_response(k_ra8_usb_speed_fs, true);
321 }
322 break;
323 case k_pv_action_set_configuration:
324 err = ra8_usb_control_response(k_ra8_usb_speed_fs, true);
325 break;
326 case k_pv_action_printer_class:
327 err = ra8_usb_pprn_handle_setup(setup);
328 break;
329 case k_pv_action_vendor_class:
330 err = ra8_usb_pvnd_handle_setup(setup);
331 break;
332 case k_pv_action_stall:
333 default:
334 err = k_ra8_err_not_supported;
335 break;
336 }
337 if (err != k_ra8_ok) {
338 (void)ra8_usb_control_response(k_ra8_usb_speed_fs, false); /* STALL EP0. */
339 }
340}

355static void demo_handle_dvst(void)
356{
357 ra8_usb_dev_state_t state = k_ra8_usb_dev_state_powered;
358 const ra8_err_t dev_err = ra8_usb_get_device_state(k_ra8_usb_speed_fs, &state);
359 if (dev_err != k_ra8_ok) {
360 return;
361 }
362 if (state == k_ra8_usb_dev_state_default) {
363 (void)ra8_usb_device_busreset_rearm(k_ra8_usb_speed_fs);
364 } else if ((state == k_ra8_usb_dev_state_configured) && !s_configured) {
365 s_configured = true;
366 demo_puts("USB PRINTER+VENDOR CONFIGURED\r\n");
367 } else {
368 /* Address / Powered / Suspended: nothing to do here. */
369 }
370}

428static void demo_poll_once(void)
429{
430 uint16_t sts = 0U;
431 const ra8_err_t sts_err = ra8_usb_get_status(k_ra8_usb_speed_fs, &sts);
432 if (sts_err != k_ra8_ok) {
433 return;
434 }
435 if ((sts & (uint16_t)k_demo_int_ctrt) != 0U) {
436 (void)ra8_usb_clear_status(k_ra8_usb_speed_fs, (uint16_t)k_demo_int_ctrt);
437 ra8_usb_setup_t setup = {};
438 const ra8_err_t rd_err = ra8_usb_read_setup_if_valid(k_ra8_usb_speed_fs, &setup);
439 if (rd_err == k_ra8_ok) {
440 demo_dispatch_setup(&setup); /* fresh SETUP. */
441 } else {
442 (void)ra8_usb_control_response(k_ra8_usb_speed_fs, true); /* status-stage ACK. */
443 }
444 }
445 if ((sts & (uint16_t)k_demo_int_dvst) != 0U) {
446 (void)ra8_usb_clear_status(k_ra8_usb_speed_fs, (uint16_t)k_demo_int_dvst);
447 demo_handle_dvst();
448 }
449 if ((sts & (uint16_t)k_demo_int_brdy) != 0U) {
450 (void)ra8_usb_clear_status(k_ra8_usb_speed_fs, (uint16_t)k_demo_int_brdy);
451 demo_drain_print_job();
452 demo_drain_vendor();
453 }
454}

527[[nodiscard]] static ra8_err_t demo_usb_bringup(void)
528{
529 const uint8_t* config_desc = nullptr;
530 uint16_t config_len = 0U;
531 if (!pv_descriptor(k_pv_action_get_config_desc, &config_desc, &config_len)) {
532 return k_ra8_err_not_supported;
533 }
534
535 ra8_err_t err = ra8_usb_pprn_init(k_ra8_usb_speed_fs);
536 if (err != k_ra8_ok) {
537 return err;
538 }
539 err = ra8_usb_pprn_set_descriptors(config_desc,
540 config_len,
541 s_device_id,
542 (uint16_t)sizeof(s_device_id));
543 if (err != k_ra8_ok) {
544 return err;
545 }
546 err = ra8_usb_pvnd_init(k_ra8_usb_speed_fs);
547 if (err != k_ra8_ok) {
548 return err;
549 }
550 err = ra8_usb_pvnd_set_descriptors(config_desc, config_len);
551 if (err != k_ra8_ok) {
552 return err;
553 }
554 err = ra8_usb_pprn_attach_setup_handler(demo_printer_setup, nullptr);
555 if (err != k_ra8_ok) {
556 return err;
557 }
558 err = ra8_usb_pvnd_attach_setup_handler(demo_vendor_setup, nullptr);
559 if (err != k_ra8_ok) {
560 return err;
561 }
562 return ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
563}

580int32_t main(void)
581{
582 uint32_t cpuclk0_hz = 0U;
583
584 if (ra8_cgc_init() != k_ra8_ok) {
585 demo_panic_halt();
586 }
587 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
588 demo_panic_halt();
589 }
590 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
591 demo_panic_halt();
592 }
593 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
594 demo_panic_halt();
595 }
596 if (ra8_board_uart_console_init(k_demo_console_baud) != k_ra8_ok) {
597 demo_panic_halt();
598 }
599 if (demo_pins_init() != k_ra8_ok) {
600 demo_panic_halt();
601 }
602
603 ra8_isr_globals_enable();
604
605 if (demo_usb_bringup() != k_ra8_ok) {
606 demo_panic_halt();
607 }
608 demo_puts("USB PRINTER+VENDOR READY\r\n");
609
610 /* Bounded outer supervisor: each pass runs the poll budget, then re-checks
611 * for progress. The device idles here forever answering the host. */
612 for (uint32_t guard = 0U; guard < (uint32_t)k_demo_poll_budget; guard++) {
613 demo_poll_once();
614 }
615
616 /* Poll budget exhausted (host never enumerated): park. */
617 while (1) {
618 demo_poll_once();
619 }
620}
