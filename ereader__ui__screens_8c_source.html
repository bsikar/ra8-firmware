357static bool er_ci_contains(const char* hay, const char* needle)
358{
359 if (needle[0] == '\0') {
360 return true;
361 }
362 for (uint32_t i = 0U; (i < (uint32_t)k_er_match_scan_max) && (hay[i] != '\0'); i++) {
363 uint32_t j = 0U;
364 while ((j < (uint32_t)k_ra8_kbd_text_max) && (needle[j] != '\0') && (hay[i + j] != '\0') &&
365 (er_lc(hay[i + j]) == er_lc(needle[j]))) {
366 j++;
367 }
368 if (needle[j] == '\0') {
369 return true;
370 }
371 }
372 return false;

710static bool er_draw_reading_body_reflow(int32_t body_top, int32_t height)
711{
712 /* Reflow from the SD-loaded font if present, else the Latin-1 face baked into
713 * flash (#66) -- so the Reading body shows real proportional text with no card
714 * at all. Only a reflow-engine failure falls through to the bitmap body. */
715 const uint8_t* font_data = s_have_font ? s_font_buf : g_ra8_font_literata_latin1;
716 const uint32_t font_len = s_have_font ? s_font_len : g_ra8_font_literata_latin1_len;
717 const int32_t body_w = (int32_t)s_fb.width_px - ((int32_t)k_er_margin_x * 2);
718 const int32_t body_h =
719 height - (int32_t)k_er_statusbar_h - (int32_t)k_er_footer_h - ((int32_t)k_er_body_gap * 2);
720 if (body_w <= 0) {
721 return false;
722 }
723 if (body_h <= 0) {
724 return false;
725 }
726 /* Re-flow only when the chapter or body geometry changes. Laying a chapter out
727 * (parse XHTML + paginate every page) dwarfs the cost of rasterising one page,
728 * so a same-chapter page turn reuses the cached layout and only renders the
729 * requested page -- otherwise every turn re-parses + re-paginates the whole
730 * chapter, the multi-second page-turn stall. */
731 const bool need_layout = (!s_reflow_open) || (s_reflow_chapter != s_chapter_idx) ||
732 (s_reflow_w != body_w) || (s_reflow_h != body_h);
733 if (need_layout && !er_reflow_relayout(body_w, body_h, font_data, font_len)) {
734 return false;
735 }
736 /* Clamp the current page (the chapter may paginate shorter than the last one)
737 * and rasterise just that page from the cached layout. */
738 if (s_reading_page >= s_reading_pages) {
739 s_reading_page = s_reading_pages - 1U;
740 }
741 (void)
742 ra8_reflow_render_page_at(&s_reflow_engine, s_reading_page, (int32_t)k_er_margin_x, body_top);
743 return true;
744}
