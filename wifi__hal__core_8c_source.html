34/* Poll the facade until the station is associated or the budget is spent. A
35 * slow association can outlast ra8_wifi_connect's internal wait, so the run
36 * finishes the job here -- the loop the facade lets a caller keep short. */
37static bool wifi_hal_settle(ra8_wifi_t* wifi, uint32_t budget)
38{
39 ra8_wifi_status_t st = {};
40 for (uint32_t i = 0U; i < budget; i++) {
41 if ((ra8_wifi_status(wifi, &st) == k_ra8_ok) && st.associated) {
42 return true;
43 }
44 ra8_wifi_link_t link = k_ra8_wifi_link_down;
45 (void)ra8_wifi_poll(wifi, &link);
46 }
47 return (ra8_wifi_status(wifi, &st) == k_ra8_ok) && st.associated;
48}

50bool wifi_hal_join_run(const wifi_hal_run_cfg_t* cfg, wifi_hal_result_t* out)
51{
52 if (out == nullptr) {
53 return false;
54 }
55 *out = (wifi_hal_result_t){};
56 out->init_err = k_ra8_err_not_initialized;
57 out->connect_err = k_ra8_err_not_initialized;
58 out->ip_err = k_ra8_err_not_initialized;
59 if ((cfg == nullptr) || (cfg->wifi == nullptr) || (cfg->wifi_cfg == nullptr) ||
60 (cfg->ssid == nullptr)) {
61 return false;
62 }
63
64 out->init_err = ra8_wifi_init(cfg->wifi, cfg->wifi_cfg);
65 if (out->init_err != k_ra8_ok) {
66 return false;
67 }
68 out->init_ok = true;
69
70 if (cfg->ssid[0] == '\0') {
71 return false; /* no credentials compiled in */
72 }
73
74 out->connect_err = ra8_wifi_connect(cfg->wifi, cfg->ssid, cfg->psk);
75 out->associated = wifi_hal_settle(cfg->wifi, cfg->poll_budget);
76 if (!out->associated) {
77 return false;
78 }
79
80 (void)ra8_wifi_get_mac(cfg->wifi, &out->mac);
81 (void)ra8_wifi_get_ap(cfg->wifi, &out->ap);
82
83 out->ip_err = ra8_wifi_wait_ip(cfg->wifi, &out->lease);
84 if ((out->ip_err != k_ra8_ok) || !out->lease.bound) {
85 return false;
86 }
87 out->ip_bound = true;
88 out->passed = true;
89 return true;
90}
