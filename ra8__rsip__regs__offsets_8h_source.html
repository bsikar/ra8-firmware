234/* -------------------------------------------------------------------------- *
235 * Layout insurance for ra8_rsip_off_t.
236 *
237 * Two-line policy for every register the HAL touches:
238 *
239 * 1. Pin the exact byte offset (catches accidental edits and rebases).
240 * 2. Verify the offset is 4-byte aligned and lands inside the
241 * documented sub-block window (catches typos that would alias
242 * onto an unrelated sub-block -- the bug pattern that put
243 * ``key_op_status`` on top of the TRNG status word).
244 *
245 * Cross-references:
246 * * File-header table at top of this file (sub-block layout).
247 * * FSP ``RSIP_PRV_ADDR_VAL_*`` anchors in
248 * ``crypto_procedures_protected/.../common/primitive/r_rsip_reg.h``.
249 * * HUM Ch 52 "Renesas Secure IP (RSIP-E50D)" p 3302-3307 (no
250 * register table -- mailbox model).
251 * -------------------------------------------------------------------------- */
252
253/* Every register cell is a 32-bit word. */
255#define RA8_RSIP_ASSERT_ALIGN(off_) \
256 static_assert(((size_t)(off_) & 0x3U) == 0U, #off_ " must be 4-byte aligned")
257
258/* Range check: low <= off <= high. */
260#define RA8_RSIP_ASSERT_IN_WIN(off_, lo_, hi_) \
261 static_assert((size_t)(off_) >= (size_t)(lo_) && (size_t)(off_) <= (size_t)(hi_), \
262 #off_ " outside window [" #lo_ ", " #hi_ "]")
