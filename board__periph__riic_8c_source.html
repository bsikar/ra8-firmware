298static void pi4ioe_write(void* ctx, uint8_t byte)
299{
300 pi4ioe_state_t* p = (pi4ioe_state_t*)ctx;
301 if (!p->ptr_set) {
302 p->reg_ptr = (uint8_t)(byte % (uint8_t)k_pi4ioe_reg_max);
303 p->ptr_set = true;
304 return;
305 }
306 /* Payload byte lands in the addressed register; the pointer auto-increments. */
307 p->file[p->reg_ptr] = byte;
308 p->reg_ptr = (uint8_t)((p->reg_ptr + 1U) % (uint8_t)k_pi4ioe_reg_max);
309 p->writes++;
310}

314{
315 pi4ioe_state_t* p = (pi4ioe_state_t*)ctx;
316 if (max == 0U) {
317 return 0U;
318 }
319 const uint32_t n = (max < (uint32_t)k_pi4ioe_reg_max) ? max : (uint32_t)k_pi4ioe_reg_max;
320 for (uint32_t i = 0U; i < n; i++) {
321 const uint8_t r = (uint8_t)((p->reg_ptr + i) % (uint8_t)k_pi4ioe_reg_max);

323 }
324 return n;

411/* =============================================================================
412 * RIIC controller model -- the transfer state machine the ra8_i2c.c polling
413 * driver drives (START / RESTART / addr / write / read / STOP).
414 * =============================================================================
415 */
416
418static void riic_open_transfer(riic_state_t* s)
419{
420 s->busy = true;
421 s->addr_done = false;
422 s->acked = false;
423 s->reading = false;
424 s->rx_len = 0U;
425 s->rx_pos = 0U;
426 s->rx_primed = false;
427 /* Transmit buffer is empty so the driver can write the address byte; clear
428 * the stale condition / fault flags so the new phase reports its own. */
429 s->icsr2 = (uint32_t)k_riic_icsr2_tdre;
430}
431
433static void riic_close_transfer(riic_state_t* s)
434{
435 if (s->acked) {
436 riic_device_t* dev = riic_device_find(s->target_7b);
437 if (dev != nullptr) {
438 if (dev->stop != nullptr) {
439 dev->stop(dev->ctx);
440 }
441 }
442 /* Console I2C tab: one line per completed (ACKed) RIIC transaction at STOP
443 * (RIIC shares the I2C tab with the IIC_B model). */
444 char ln[k_riic_console_line_cap];
445 if (s->reading) {
446 (void)snprintf(ln,
447 sizeof(ln),
448 "RIIC addr=0x%02X R %uB",
449 (unsigned)s->target_7b,
450 (unsigned)s->rx_len);
451 } else {
452 (void)snprintf(ln, sizeof(ln), "RIIC addr=0x%02X W", (unsigned)s->target_7b);
453 }
454 board_console_push(k_board_console_ch_i2c, ln);
455 }
456 s->busy = false;
457 s->addr_done = false;
458 s->icsr2 = (uint32_t)k_riic_icsr2_stop;
459}

462static void riic_address_phase(riic_state_t* s, uint8_t address_byte)
463{
464 s->target_7b =
465 (uint8_t)((uint32_t)address_byte >> (uint32_t)k_riic_addr_shift) & (uint8_t)k_riic_addr_mask7;
466 s->reading = ((uint32_t)address_byte & (uint32_t)k_riic_addr_rnw) != 0U;
467 s->addr_done = true;
468
469 riic_device_t* dev = riic_device_find(s->target_7b);
470 if (dev == nullptr) {
471 /* No device at this address: NACK it (scan reports ack=0). */
472 s->acked = false;
473 s->icsr2 |=
474 ((uint32_t)k_riic_icsr2_nackf | (uint32_t)k_riic_icsr2_tend | (uint32_t)k_riic_icsr2_tdre);
475 return;
476 }
477 s->acked = true;
478 /* Address ACKed: TEND lets the scan / write proceed, TDRE re-arms TX. */
479 s->icsr2 |= ((uint32_t)k_riic_icsr2_tend | (uint32_t)k_riic_icsr2_tdre);
480 if (s->reading) {
481 /* Pre-fetch the device's response for this read so ICDRR reads serve it. */
482 s->rx_len = dev->read(dev->ctx, s->rx, (uint32_t)k_riic_dev_rx_max);
483 s->rx_pos = 0U;
484 s->rx_primed = false;
485 s->icsr2 |= (uint32_t)k_riic_icsr2_rdrf; /* RX data ready */
486 }
487}

490static void riic_icdrt_write(riic_state_t* s, uint32_t value)
491{
492 const uint8_t byte = (uint8_t)(value & (uint32_t)k_riic_byte_mask);
493 if (!s->addr_done) {
494 riic_address_phase(s, byte);
495 return;
496 }
497 if (!s->acked) {
498 return; /* NACKed address: swallow further writes until STOP */
499 }
500 riic_device_t* dev = riic_device_find(s->target_7b);
501 if ((dev != nullptr) && (dev->write != nullptr)) {
502 dev->write(dev->ctx, byte);
503 }
504 /* Buffer empty again and the byte was accepted: TDRE + TEND for the next. */
505 s->icsr2 |= ((uint32_t)k_riic_icsr2_tdre | (uint32_t)k_riic_icsr2_tend);
506}

509static uint32_t riic_icdrr_read(riic_state_t* s)
510{
511 if (!s->rx_primed) {
512 /* The controller-receive flow does one dummy ICDRR read to start the clock
513 * before the first real byte (HUM Ch 39.3.4). */
514 s->rx_primed = true;
515 return 0U;
516 }
517 uint8_t b = 0U;
518 if (s->rx_pos < s->rx_len) {
519 b = s->rx[s->rx_pos];
520 s->rx_pos++;
521 }
522 s->icsr2 |= (uint32_t)k_riic_icsr2_rdrf; /* keep RX-ready for the next byte */
523 return (uint32_t)b;
524}

527static void riic_iccr2_write(riic_state_t* s, uint64_t off, uint32_t value)
528{
529 if ((value & ((uint32_t)k_riic_iccr2_st | (uint32_t)k_riic_iccr2_rs)) != 0U) {
530 riic_open_transfer(s);
531 /* ST / RS auto-clear once the condition is issued; the driver spins on RS
532 * reading 0 after a repeated-START, so drop both request bits. */
533 s->reg[off] = (uint8_t)(value & ~((uint32_t)k_riic_iccr2_st | (uint32_t)k_riic_iccr2_rs));
534 } else if ((value & (uint32_t)k_riic_iccr2_sp) != 0U) {
535 riic_close_transfer(s);
536 s->reg[off] = (uint8_t)(value & ~(uint32_t)k_riic_iccr2_sp);
537 }
538}

556static void riic_target_begin_read(riic_state_t* s)
557{
558 s->tgt_phase = (uint8_t)k_riic_tgt_read;
559 s->tgt_rd_len = 0U;

564static void riic_target_complete_read(riic_state_t* s)
565{
566 bool ok = (s->tgt_rd_len == (uint32_t)k_riic_tgt_payload_len);
567 for (uint32_t i = 0U; i < (uint32_t)k_riic_tgt_payload_len; i++) {
568 if (s->tgt_rd[i] != k_riic_tgt_payload[i]) {
569 ok = false;
570 }
571 }
572 if (!ok) {
573 s->tgt_echo_bad = true;
574 }
575 s->tgt_cycles++;
576 if (s->tgt_cycles < (uint32_t)k_riic_tgt_cycles) {
577 riic_target_begin_write(s);
578 } else {
579 s->tgt_phase = (uint8_t)k_riic_tgt_done;
580 s->tgt_icsr2 = 0U; /* no match, no STOP: dispatch reports no event forever. */
581 }
582}

585static void riic_target_open(riic_state_t* s, uint32_t icser)
586{
587 if ((icser & (uint32_t)k_riic_icser_slots) == 0U) {
588 s->tgt_armed = false;
589 s->tgt_phase = (uint8_t)k_riic_tgt_idle;
590 return;
591 }
592 uint8_t sarl = s->reg[k_riic_off_sarl0];
593 if ((icser & (uint32_t)k_riic_icser_sar1e) != 0U) {
594 sarl = s->reg[k_riic_off_sarl1];
595 } else if ((icser & (uint32_t)k_riic_icser_sar2e) != 0U) {
596 sarl = s->reg[k_riic_off_sarl2];
597 }
598 s->tgt_own_addr = (uint8_t)((uint32_t)sarl >> (uint32_t)k_riic_addr_shift);
599 s->tgt_armed = true;
600 riic_target_begin_write(s);
601}

616static uint32_t riic_target_iccr2(const riic_state_t* s)
617{

626static uint32_t riic_target_icdrr(riic_state_t* s)
627{
628 if (s->tgt_phase != (uint8_t)k_riic_tgt_write) {
629 return 0U; /* read-phase SCL-release dummy read / idle: no receive data. */
630 }
631 uint8_t b = 0U;
632 if (s->tgt_wr_pos == 0U) {
633 /* Step 1: the matched address byte the driver dummy-reads to discard. */
634 b = (uint8_t)((uint32_t)s->tgt_own_addr << (uint32_t)k_riic_addr_shift);
635 } else if (s->tgt_wr_pos <= (uint32_t)k_riic_tgt_payload_len) {
636 b = k_riic_tgt_payload[s->tgt_wr_pos - 1U];
637 }
638 s->tgt_wr_pos++;
639 if (s->tgt_wr_pos > (uint32_t)k_riic_tgt_payload_len) {
640 s->tgt_icsr2 = (uint32_t)k_riic_icsr2_stop; /* all served: STOP, drop RDRF. */
641 } else {
642 s->tgt_icsr2 = (uint32_t)k_riic_icsr2_rdrf;
643 }
644 return (uint32_t)b;
645}

648static void riic_target_icdrt(riic_state_t* s, uint8_t byte)
649{
650 if (s->tgt_phase != (uint8_t)k_riic_tgt_read) {
651 return;
652 }
653 if (s->tgt_rd_len < (uint32_t)k_riic_tgt_cap) {
654 s->tgt_rd[s->tgt_rd_len] = byte;
655 }
656 s->tgt_rd_len++;
657 /* TDRE|TEND stay asserted (begin_read) so fill_tx keeps pushing to completion. */
658}

661static void riic_target_icsr2_write(riic_state_t* s, uint32_t value)
662{
663 s->tgt_icsr2 &= value; /* condition flags are write-0-to-clear. */
664 if (s->tgt_phase == (uint8_t)k_riic_tgt_write) {
665 riic_target_begin_read(s); /* receive() cleared STOP -> controller now reads. */
666 } else if (s->tgt_phase == (uint8_t)k_riic_tgt_read) {
667 riic_target_complete_read(s); /* finish_tx cleared NACKF/STOP -> cycle done. */
668 }
669}

672static uint64_t riic_target_reg_read(riic_state_t* s, uint64_t off)

674 if (off == (uint64_t)k_riic_off_icsr1) {

690static uint64_t riic_reg_read(riic_state_t* s, uint64_t off)
691{
692 if (s->tgt_armed) {
693 return riic_target_reg_read(s, off);
694 }
695 if (off == (uint64_t)k_riic_off_icsr2) {
696 return s->icsr2;
697 }
698 if (off == (uint64_t)k_riic_off_iccr2) {
699 /* BBSY tracks an open transaction; the request bits already read back 0. */
700 uint32_t v = s->reg[off] & ~(uint32_t)k_riic_iccr2_bbsy;
701 if (s->busy) {
702 v |= (uint32_t)k_riic_iccr2_bbsy;
703 }
704 return v;
705 }
706 if (off == (uint64_t)k_riic_off_icdrr) {
707 return riic_icdrr_read(s);
708 }
709 return s->reg[off]; /* reflect every other register */
710}

713static void riic_reg_write(riic_state_t* s, uint64_t off, uint32_t value)
714{
715 s->reg[off] = (uint8_t)(value & (uint32_t)k_riic_byte_mask);
716 if (off == (uint64_t)k_riic_off_icser) {
717 /* ICSER arms / disarms the own-address responder; it is the target trigger. */
718 riic_target_open(s, value);
719 return;
720 }
721 if (s->tgt_armed) {
722 if (off == (uint64_t)k_riic_off_icdrt) {
723 riic_target_icdrt(s, (uint8_t)(value & (uint32_t)k_riic_byte_mask));
724 } else if (off == (uint64_t)k_riic_off_icsr2) {
725 riic_target_icsr2_write(s, value);
726 }
727 return;
728 }
729 if (off == (uint64_t)k_riic_off_iccr2) {
730 riic_iccr2_write(s, off, value);
731 } else if (off == (uint64_t)k_riic_off_icdrt) {
732 riic_icdrt_write(s, value);
733 } else if (off == (uint64_t)k_riic_off_icsr2) {
734 /* Condition / fault flags are write-0-to-clear: keep only the model's
735 * internal flags whose bit is still written as 1 by the driver's RMW. */
736 s->icsr2 &= value;
737 }
738}

741static uint64_t riic_read(uc_engine* uc, uint64_t addr, unsigned size)
742{
743 (void)uc;
744 (void)size;
745 const uint32_t ch = (uint32_t)((addr - (uint64_t)k_riic_base) / (uint64_t)k_riic_stride);
746 const uint64_t off = (addr - (uint64_t)k_riic_base) % (uint64_t)k_riic_stride;
747 if (off >= (uint64_t)k_riic_reg_bytes) {
748 return 0U; /* outside the modelled register file */
749 }
750 return riic_reg_read(&s_riic[ch], off);
751}

767static void riic_reset(void)
768{
769 for (uint32_t i = 0U; i < (uint32_t)k_riic_count; i++) {
770 s_riic[i] = (riic_state_t){};
771 }
772 s_pi4ioe = (pi4ioe_state_t){};
773 /* Seed the expander's device-id register so a read-back of the part probes
774 * its reset default, matching the PI4IOE5V6408. */
775 s_pi4ioe.file[k_pi4ioe_reg_devid] = (uint8_t)k_pi4ioe_devid_val;
776 /* Populate the modelled I2C bus: the EK-RA8D2 carrier's PI4IOE5V6408 I/O
777 * expander (U15) answers at its 7-bit address, so the firmware's real
778 * ra8_i2c_scan / ra8_i2c_write -> RIIC path ACKs instead of needing a stub. */
779 for (uint32_t i = 0U; i < (uint32_t)k_riic_dev_max; i++) {
780 s_riic_dev[i] = (riic_device_t){};
781 }
782 riic_device_register((uint8_t)k_pi4ioe_addr_7b,
783 pi4ioe_write,
784 pi4ioe_read,
785 pi4ioe_stop,
786 &s_pi4ioe);
787 /* The camera_capture example's OV5640 sensor answers on channel 1 at SCCB
788 0x3C; its chip-ID registers report 0x5640 so the sensor probe passes and
789 the CEU capture path (board_periph_ceu.c) runs headless. */
790 s_ov5640 = (ov5640_state_t){};
791 riic_device_register((uint8_t)k_ov5640_addr_7b,
792 ov5640_write,
793 ov5640_read,
794 ov5640_stop,
795 &s_ov5640);
796}

799static void riic_report(void)
800{
801 if (s_pi4ioe.writes > 0U) {
802 (void)fprintf(stderr,
803 " RIIC PI4IOE : U15 expander 0x%02X acked %u register write(s)\n",
804 (unsigned)k_pi4ioe_addr_7b,
805 s_pi4ioe.writes);
806 }
807 if ((s_ov5640.id_reads > 0U) || (s_ov5640.writes > 0U)) {
808 (void)fprintf(stderr,
809 " RIIC OV5640 : SCCB 0x%02X chip-id 0x5640 (%u id read(s), %u cfg write(s))\n",
810 (unsigned)k_ov5640_addr_7b,
811 s_ov5640.id_reads,
812 s_ov5640.writes);
813 }
814 for (uint32_t ch = 0U; ch < (uint32_t)k_riic_count; ch++) {
815 if (s_riic[ch].tgt_cycles > 0U) {
816 (void)fprintf(stderr,
817 " RIIC target : own 0x%02X serviced %u controller write+read "
818 "cycle(s), echo=%s\n",
819 (unsigned)s_riic[ch].tgt_own_addr,
820 (unsigned)s_riic[ch].tgt_cycles,
821 s_riic[ch].tgt_echo_bad ? "MISMATCH" : "OK");
822 }
823 }
