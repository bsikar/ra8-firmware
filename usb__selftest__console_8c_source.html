126[[nodiscard]] ra8_err_t selftest_print_dec(uint32_t value)
127{
128 uint8_t scratch[k_selftest_dec_chars_u32] = {};
129 uint8_t out[k_selftest_dec_chars_u32] = {};
130 uint8_t count = 0U;
131 uint32_t v = value;
132 if (v == 0U) {
133 out[0] = (uint8_t)'0';
134 return selftest_sci_write(out, 1U);
135 }
136 while (v != 0U) {
137 if (count >= (uint8_t)k_selftest_dec_chars_u32) {
138 break;
139 }
140 scratch[count] = (uint8_t)((uint8_t)'0' + (uint8_t)(v % k_selftest_dec_radix));
141 v = v / k_selftest_dec_radix;
142 count++;
143 }
144 for (uint8_t i = 0U; i < count; i++) {
145 out[i] = scratch[count - 1U - i];
146 }
147 return selftest_sci_write(out, (uint32_t)count);
148}

169[[nodiscard]] ra8_err_t selftest_print_hex(uint32_t value, uint8_t digits)
170{
171 uint8_t out[k_selftest_hex_chars_u32] = {};
172 uint8_t width = digits;
173 if (width > (uint8_t)k_selftest_hex_chars_u32) {
174 width = (uint8_t)k_selftest_hex_chars_u32;
175 }
176 for (uint8_t i = 0U; i < width; i++) {
177 const uint8_t shift = (uint8_t)((width - 1U - i) * k_selftest_nibble_bits);
178 out[i] = selftest_nibble_to_hex((value >> shift) & k_selftest_nibble_mask);
179 }
180 return selftest_sci_write(out, (uint32_t)width);
181}
