342static void tz_sram_ns_boundary(void)
343{
344 /* Open PRC4 so the CPSCU SRAMSABAR writes below land. */
345 /* HUM Ch 13.2.1 "PRCR_S" p 521 */
346 *(volatile uint16_t*)k_tz_prcr_s_addr = (uint16_t)k_tz_prcr_s_open;
347
348 /* HUM Ch 58.2 "SRAMSABARn : SRAM Security Attribute Boundary Address
349 * Register" p 3527 -- boundary = start address of the NS region; below
350 * = Secure, at/above = Non-secure. */
351 tz_write32(k_tz_sramsabar0_addr + (0U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar0_val);
352 tz_write32(k_tz_sramsabar0_addr + (1U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar1_val);
353 tz_write32(k_tz_sramsabar0_addr + (2U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar2_val);
354 tz_write32(k_tz_sramsabar0_addr + (3U * sizeof(uint32_t)), (uint32_t)k_tz_sramsabar3_val);
355
356 /* Re-lock PRC4 (restore CPSCU write-protect). */
357 /* HUM Ch 13.2.1 "PRCR_S" p 521 */
358 *(volatile uint16_t*)k_tz_prcr_s_addr = (uint16_t)k_tz_prcr_s_close;
359}

380static ra8_err_t tz_sau_program(void)
381{
382 /* HUM Ch 51.3.3.3 "Secure Attribution Unit (SAU)" p 3266 -- need at
383 * least 4 implemented regions for this layout. */
384 const uint32_t sau_type = tz_read32(k_tz_sau_type_addr);
385 if ((sau_type & (uint32_t)k_tz_sau_type_mask) < (uint32_t)k_tz_sau_min_regions) {
386 return k_ra8_err_not_supported;
387 }
388
389 /* RLAR limit is the LAST byte of the veneer block (end - 1); the helper
390 * masks it to the 32-byte SAU region quantum. */
391 const uint32_t nsc_base = (uint32_t)(uintptr_t)&g_ra8_ls_sgstubs_start;
392 const uint32_t nsc_limit = (uint32_t)(uintptr_t)&g_ra8_ls_sgstubs_end - 1U;
393 tz_sau_set_region((uint8_t)k_tz_region_nsc, nsc_base, nsc_limit, /*is_nsc=*/true);
394 tz_sau_set_region((uint8_t)k_tz_region_ns_code,
395 (uint32_t)k_tz_ns_code_base,
396 (uint32_t)k_tz_ns_code_limit,
397 /*is_nsc=*/false);
398 tz_sau_set_region((uint8_t)k_tz_region_ns_sram,
399 (uint32_t)k_tz_ns_sram_base,
400 (uint32_t)k_tz_ns_sram_limit,
401 /*is_nsc=*/false);
402 tz_sau_set_region((uint8_t)k_tz_region_ns_per,
403 (uint32_t)k_tz_ns_per_base,
404 (uint32_t)k_tz_ns_per_limit,
405 /*is_nsc=*/false);
406
407 __asm__ volatile("dsb 0xF" ::: "memory");
408 /* Enable the SAU with ALLNS = 0 (default-deny). */
409 /* HUM Ch 51.3.3.3 "Secure Attribution Unit (SAU)" p 3266 */
410 tz_write32(k_tz_sau_ctrl_addr, (uint32_t)k_tz_sau_ctrl_enable);
411 __asm__ volatile("dsb 0xF" ::: "memory");
412 __asm__ volatile("isb" ::: "memory");
413 return k_ra8_ok;
414}

463static ra8_err_t tz_usb_route_pins(void)
464{
465 /* Establish the pin-validator baseline. The Secure boot never runs
466 * ra8_infrastructure_init (its main() is dead -- BLXNS does not return), so
467 * the validator bitmap is in an uninitialised-contract state and warm
468 * resets leave stale claims (SRAM survives SYSRESETREQ; PFS does not), which
469 * would make the USB-pin claims spuriously conflict. Reset it first. */
470 ra8_pin_validator_reset();
471
472 /* Route the USB-FS pins as the DEVICE (Secure owns PFS). P5_00 LOW = dev. */
473 ra8_err_t err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_tz_usb_pin_vbus,
474 k_ra8_psel_usb_fs,
475 "tz_usb.fs_vbus");
476 if (err == k_ra8_ok) {
477 err = ra8_gpio_output_init((ra8_port_pin_t)k_tz_usb_pin_vbusen, k_ra8_level_low);
478 }
479 if (err == k_ra8_ok) {
480 err =
481 ra8_pfs_route_peripheral((ra8_port_pin_t)k_tz_usb_pin_dp, k_ra8_psel_usb_fs, "tz_usb.fs_dp");
482 }
483 if (err == k_ra8_ok) {
484 err =
485 ra8_pfs_route_peripheral((ra8_port_pin_t)k_tz_usb_pin_dm, k_ra8_psel_usb_fs, "tz_usb.fs_dm");
486 }
487
488 /* USBHS host pins: PD07 HIGH (U18 supplies J7 VBUS), route P4_08
489 * USBHS_VBUS. (The host-mode mux is the U15 expander -- handled later.) */
490 if (err == k_ra8_ok) {
491 err = ra8_gpio_output_init((ra8_port_pin_t)k_tz_usb_pin_hs_pwr, k_ra8_level_high);
492 }
493 if (err == k_ra8_ok) {
494 err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_tz_usb_pin_hs_vbus,
495 k_ra8_psel_usb_hs,
496 "tz_usb.hs_vbus");
497 }
498
499 /* Enable the USBHS UTMI PLL (Secure CGC; PLL1 already locked). A warm
500 * reset leaves the CGC PLL domain running, so a re-enable reports
501 * k_ra8_err_busy ("already locked"); the clock is up either way, so treat
502 * busy as success. */
503 if (err == k_ra8_ok) {
504 const ra8_err_t pll_err = ra8_cgc_usbhs_pll_enable();
505 if ((pll_err != k_ra8_ok) && (pll_err != k_ra8_err_busy)) {
506 err = pll_err;
507 }
508 }
509 return err;
510}

530static void tz_usb_mark_ns(void)
531{
532 /* HUM Ch 51.8.1 "PSARB : Peripheral Security Attribution Register B" p 3284
533 * -- PSARB11 = USBFS0, PSARB12 = USBHS (+ their MSTPCRB.MSTPB11/12 bits);
534 * 0 = Secure, 1 = Non-secure. PSARx share the PRCR_S.PRC4 write gate with
535 * SRAMSABARn (HUM Ch 13.2.1 "Association between PRCR bits and use of
536 * registers" p 521), so the gate must be open across the write; HUM "Security
537 * or Privilege Bit Write Timing" p 3301 then requires reading back until the
538 * value matches. */
539 /* Open PRC4. HUM Ch 13.2.1 "PRCR_S" p 521 */
540 *(volatile uint16_t*)k_tz_prcr_s_addr = (uint16_t)k_tz_prcr_s_open;
541 const uint32_t want = tz_read32(k_tz_psarb_addr) | (uint32_t)k_tz_psarb_usb_ns;
542 tz_write32(k_tz_psarb_addr, want);
543 uint32_t seen = 0U;
544 for (uint32_t spin = 0U; spin < (uint32_t)k_tz_psarb_readback_spins; spin += 1U) {
545 seen = tz_read32(k_tz_psarb_addr);
546 if (seen == want) {
547 break;
548 }
549 }
550 /* Re-lock PRC4. HUM Ch 13.2.1 "PRCR_S" p 521 */
551 *(volatile uint16_t*)k_tz_prcr_s_addr = (uint16_t)k_tz_prcr_s_close;
552 g_tz_usb_psarb_readback = seen;
553}

575static void tz_usb_handoff_prepare(void)
576{
577 /* 1+2+3. Route USB pins (FS device + HS host) and enable the USBHS PLL. */
578 const ra8_err_t pins_err = tz_usb_route_pins();
579 g_tz_usb_pins_err = (uint32_t)pins_err;
580
581 /* 4. Set the U15 I/O-expander to USBHS host mode (SW4-8 -> Host). Decoupled
582 * from the deterministic setup above and best-effort: on a cold boot the
583 * single I2C write lands (probe -> success); after a warm reset RIIC1's
584 * BBSY can still be set, reporting k_ra8_err_busy. The external PI4IOE
585 * latches its host-mode output, so the USBHS host role persists across the
586 * MCU warm reset -- the self-loop still enumerates (the HIL gate proves
587 * it). A retry cannot help here: the expander claims the SCL1/SDA1 pins on
588 * the first try and a second try would fault the pin validator. */
589 const ra8_err_t exp_err = ra8_board_io_expander_set_usbhs_host_mode();
590 g_tz_usb_expander_err = (uint32_t)exp_err;
591
592 /* 5. Mark BOTH USB controllers Non-secure in PSARB (bits 11 + 12). */
593 tz_usb_mark_ns();
594}

598void ra8_trustzone_init(void)
599{
600#ifdef RA8_TRUSTZONE_ENABLE
601 /* 0. Hand USB-FS (pins + PSARB NS attribution) to the NS world. */
602 tz_usb_handoff_prepare();
603
604 /* 1. Carve the SRAM2 NS aperture via the runtime SRAMSABAR boundary. */
605 tz_sram_ns_boundary();
606
607 /* 2. Programme the bit[28] SAU (NSC veneers + IDAU-NS ranges), enable. */
608 if (tz_sau_program() != k_ra8_ok) {
609 return; /* Fall through to the S-side main() fallback. */
610 }
611
612#ifdef RA8_EREADER_NS_XIP
613 /* 3. XIP: no image copy. .text/.rodata execute in place from the OSPI NS
614 * alias (0x9000_0000); ns_reset_handler copies only .data into SRAM.
615 * ra8_emulator maps the OSPI window unconditionally, so the fake proof needs
616 * no controller arming here.
617 * TODO(OSPI XIP arming on silicon): before this BLXNS, call
618 * ra8_xspi_xip_enter(instance, enter_code, exit_code) with the EK-RA8D2
619 * OSPI flash's real XIP enter/exit opcodes so the 0x8000_0000/0x9000_0000
620 * window returns flash on hardware, and add a SAU region marking the OSPI
621 * NS alias Non-secure (the IDAU bit[28] alias alone is overridden Secure by
622 * the default SAU). Both are unnecessary in ra8_emulator. */
623#else
624 /* 3. Copy the NS image into the now-Non-secure SRAM alias. */
625 tz_copy_ns_image();
626#endif
627
628 /* 4. Clear the Secure PRIMASK before handing off. SystemInit masked IRQs
629 * (CPSID i) for secure bring-up; a set PRIMASK_S boosts the execution
630 * priority and masks NON-secure exceptions too, so the NS ThreadX
631 * PendSV / SysTick would never fire. The NS side cannot clear PRIMASK_S
632 * (it is Secure-banked), so do it here, right before BLXNS. */
633 __asm__ volatile("cpsie i" ::: "memory");
634
635 /* 5. BLXNS into the NS reset vector (slot 1 of the NS vector table at the
636 * fixed run base 0x3210_0000). Does not return on hardware. */
637 const ra8_err_t jump_err =
638 ra8_tz_secure_boot_jump_ns((const uint32_t*)(uintptr_t)k_tz_ns_run_base);
639 if (jump_err != k_ra8_ok) {
640 /* Reached on hardware ONLY when the NS image was DENIED (vector-table
641 * validation or the root-of-trust gate failed). Latch the verdict for a
642 * J-Link probe and fall through to the S-side main() fallback -- never
643 * treat the NS world as live. */
644 g_tz_jump_ns_err = (uint32_t)jump_err;
645 return;
646 }
647
648 /* On host (RA8_OFF_TARGET) the library stubs BLXNS and returns
649 * k_ra8_ok; on target this point is unreachable. */
650#endif
651}
