50/* Hand-modelled Cortex-M exception state. Unicorn's M33 core has no NVIC /
51 * exception unit, so ra8_emulator takes SysTick / PendSV / SVCall by hand: it
52 * tracks the active-exception priority stack here (everything else -- MSP/PSP/
53 * CONTROL/xPSR/PRIMASK -- is read straight from Unicorn). s_exc_stack holds the
54 * priority of each handler currently active so a higher-priority exception
55 * (e.g. SysTick, prio 0x40) can pre-empt a lower one (PendSV, prio 0xFF) but
56 * not vice-versa, exactly as the real priority logic would nest them. */

58static uint32_t s_exc_depth;
59static uint32_t s_pendsv_takes;

98static uint32_t exc_priority(uc_engine* uc, uint32_t exc_num)
99{
100 if (exc_num == (uint32_t)k_exc_svcall) {
101 return (rd32(uc, (uint64_t)k_scb_shpr2) >> (3U * (uint32_t)k_byte_bits)) &
102 (uint32_t)k_byte_mask;
103 }
104 if (exc_num == (uint32_t)k_exc_pendsv) {
105 return (rd32(uc, (uint64_t)k_scb_shpr3) >> (2U * (uint32_t)k_byte_bits)) &
106 (uint32_t)k_byte_mask;
107 }
108 if (exc_num == (uint32_t)k_exc_systick) {
109 return (rd32(uc, (uint64_t)k_scb_shpr3) >> (3U * (uint32_t)k_byte_bits)) &
110 (uint32_t)k_byte_mask;
111 }
112 return (uint32_t)k_exc_prio_max;

116static void exc_stack_frame(uc_engine* uc, uint32_t sp, bool fp_active, uint32_t frame_xpsr)
117{
118 wr32(uc, (uint64_t)sp + 0U, reg_get(uc, UC_ARM_REG_R0));
119 wr32(uc, (uint64_t)sp + 4U, reg_get(uc, UC_ARM_REG_R1));
120 wr32(uc, (uint64_t)sp + 8U, reg_get(uc, UC_ARM_REG_R2));
121 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_r3, reg_get(uc, UC_ARM_REG_R3));
122 wr32(uc, (uint64_t)sp + 16U, reg_get(uc, UC_ARM_REG_R12));
123 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_lr, reg_get(uc, UC_ARM_REG_LR));
124 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_pc, reg_get(uc, UC_ARM_REG_PC));
125 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_xpsr, frame_xpsr);
126
127 /* Armv8-M FP extended frame: S0-S15 + FPSCR sit directly above the 8-word
128 * basic frame (ThreadX's PendSV adds S16-S31 on top of this). */
129 if (fp_active) {
130 for (uint32_t i = 0U; i < (uint32_t)k_fp_s_words; i++) {
131 wr32(uc,
132 (uint64_t)sp + (uint64_t)k_frame_off_s0 + (uint64_t)(i * (uint32_t)k_word_bytes),
133 reg_get(uc, UC_ARM_REG_S0 + (int)i));
134 }
135 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_fpscr, reg_get(uc, UC_ARM_REG_FPSCR));
136 }
137}

140static uint32_t exc_return_value(bool in_thread, bool use_psp, bool fp_active)
141{
142 /* EXC_RETURN encodes where to unstack: Thread/PSP, Thread/MSP, or (when an
143 * exception pre-empts another) Handler/MSP. */
144 uint32_t exc_ret;
145 if (!in_thread) {
146 exc_ret = (uint32_t)k_exc_ret_handler;
147 } else if (use_psp) {
148 exc_ret = (uint32_t)k_exc_ret_psp;
149 } else {
150 exc_ret = (uint32_t)k_exc_ret_msp;
151 }
152 if (fp_active) {
153 exc_ret &= ~(uint32_t)k_exc_ret_ftype; /* FType=0: an FP frame was stacked. */
154 }
155 return exc_ret;
156}

185void exc_enter(uc_engine* uc, uint32_t exc_num, uint32_t handler)
186{
187 const uint32_t xpsr_in = reg_get(uc, UC_ARM_REG_XPSR);
188 const uint32_t control = reg_get(uc, UC_ARM_REG_CONTROL);
189 const bool in_thread = (xpsr_in & (uint32_t)k_xpsr_ipsr_mask) == 0U;
190 const bool use_psp = in_thread && ((control & (uint32_t)k_control_spsel) != 0U);
191 const bool fp_active = (control & (uint32_t)k_control_fpca) != 0U;
192
193 const int sp_reg = use_psp ? UC_ARM_REG_PSP : UC_ARM_REG_MSP;
194 uint32_t sp = reg_get(uc, sp_reg);
195
196 /* Hardware aligns the stack pointer to 8 bytes on entry and flags the pad in
197 * the stacked xPSR (bit 9) so the matching exception return can remove it. */
198 uint32_t frame_xpsr = xpsr_in;
199 if ((sp & 0x4U) != 0U) {
200 sp -= 4U;
201 frame_xpsr |= (uint32_t)k_xpsr_align9;
202 } else {
203 frame_xpsr &= ~(uint32_t)k_xpsr_align9;
204 }
205 sp -= (uint32_t)k_exc_frame_bytes;
206 if (fp_active) {
207 sp -= (uint32_t)k_fp_frame_extra;
208 }
209
210 exc_stack_frame(uc, sp, fp_active, frame_xpsr);
211
212 /* Commit the new value of whichever stack the frame went onto. */
213 reg_set(uc, sp_reg, sp);
214
215 const uint32_t exc_ret = exc_return_value(in_thread, use_psp, fp_active);
216
217 /* Handler mode always runs on MSP with CONTROL.SPSEL clear. */
218 reg_set(uc, UC_ARM_REG_CONTROL, control & ~(uint32_t)k_control_spsel);
219 reg_set(uc, UC_ARM_REG_SP, reg_get(uc, UC_ARM_REG_MSP));
220
221 uint32_t handler_xpsr =
222 (xpsr_in & ~(uint32_t)k_xpsr_ipsr_mask) | (exc_num & (uint32_t)k_xpsr_ipsr_mask);
223 handler_xpsr |= (uint32_t)k_xpsr_t_bit; /* M-profile is always Thumb. */
224 reg_set(uc, UC_ARM_REG_XPSR, handler_xpsr);
225 reg_set(uc, UC_ARM_REG_LR, exc_ret);
226 reg_set(uc, UC_ARM_REG_PC, handler & ~1U);
227
228 if (s_exc_depth < (uint32_t)k_exc_nest_max) {
229 s_exc_stack[s_exc_depth] = exc_priority(uc, exc_num);
230 s_exc_depth++;
231 }
232}

235static void exc_restore_fp_frame(uc_engine* uc, uint32_t sp)
236{
237 /* Armv8-M FP extended frame (EXC_RETURN bit4 clear): S0-S15 + FPSCR sit above
238 * the basic frame. Restore them so the thread's scalar FP state survives.
239 * PC/xPSR keep their basic-frame offsets (accounted for by the caller). */
240 for (uint32_t i = 0U; i < (uint32_t)k_fp_s_words; i++) {
241 reg_set(
242 uc,
243 UC_ARM_REG_S0 + (int)i,
244 rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_s0 + (uint64_t)(i * (uint32_t)k_word_bytes)));
245 }
246 reg_set(uc, UC_ARM_REG_FPSCR, rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_fpscr));
247}

250static void exc_restore_mode(uc_engine* uc, uint32_t xpsr, bool to_thread, bool to_psp)
251{
252 /* Restore mode: on return to Thread, CONTROL.SPSEL follows EXC_RETURN bit2;
253 * on return to a pre-empted handler the core stays on MSP. */
254 uint32_t control = reg_get(uc, UC_ARM_REG_CONTROL);
255 if (to_thread && to_psp) {
256 control |= (uint32_t)k_control_spsel;
257 } else {
258 control &= ~(uint32_t)k_control_spsel;
259 }
260 reg_set(uc, UC_ARM_REG_CONTROL, control);
261
262 uint32_t new_xpsr = xpsr | (uint32_t)k_xpsr_t_bit;
263 if (to_thread) {
264 new_xpsr &= ~(uint32_t)k_xpsr_ipsr_mask; /* Thread mode: IPSR == 0 */
265 }
266 reg_set(uc, UC_ARM_REG_XPSR, new_xpsr);
267}

293void exc_return(uc_engine* uc, uint32_t exc_ret)
294{
295 const bool to_psp = (exc_ret & (uint32_t)k_exc_ret_spsel) != 0U;
296 const bool to_thread = (exc_ret & (uint32_t)k_exc_ret_mode) != 0U;
297 const bool fp_frame = (exc_ret & (uint32_t)k_exc_ret_ftype) == 0U;
298 const int sp_reg = to_psp ? UC_ARM_REG_PSP : UC_ARM_REG_MSP;
299 uint32_t sp = reg_get(uc, sp_reg);
300
301 const uint32_t r0 = rd32(uc, (uint64_t)sp + 0U);
302 const uint32_t r1 = rd32(uc, (uint64_t)sp + 4U);
303 const uint32_t r2 = rd32(uc, (uint64_t)sp + 8U);
304 const uint32_t r3 = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_r3);
305 const uint32_t r12 = rd32(uc, (uint64_t)sp + 16U);
306 const uint32_t lr = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_lr);
307 const uint32_t pc = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_pc);
308 const uint32_t xpsr = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_xpsr);
309
310 if (fp_frame) {
311 exc_restore_fp_frame(uc, sp);
312 }
313
314 sp += (uint32_t)k_exc_frame_bytes;
315 if (fp_frame) {
316 sp += (uint32_t)k_fp_frame_extra;
317 }
318 if ((xpsr & (uint32_t)k_xpsr_align9) != 0U) {
319 sp += 4U; /* undo the entry-time 8-byte realignment pad */
320 }
321 reg_set(uc, sp_reg, sp);
322
323 reg_set(uc, UC_ARM_REG_R0, r0);
324 reg_set(uc, UC_ARM_REG_R1, r1);
325 reg_set(uc, UC_ARM_REG_R2, r2);
326 reg_set(uc, UC_ARM_REG_R3, r3);
327 reg_set(uc, UC_ARM_REG_R12, r12);
328 reg_set(uc, UC_ARM_REG_LR, lr);
329
330 exc_restore_mode(uc, xpsr, to_thread, to_psp);
331
332 /* Active SP becomes whichever stack the returned-to context uses. */
333 reg_set(uc, UC_ARM_REG_SP, reg_get(uc, to_psp && to_thread ? UC_ARM_REG_PSP : UC_ARM_REG_MSP));
334 reg_set(uc, UC_ARM_REG_PC, pc & ~1U);
335
336 if (s_exc_depth > 0U) {
337 s_exc_depth--;
338 }
339}

409static bool exc_take_periph_irq(uc_engine* uc, uint32_t vtor_base, uint32_t active)
410{
411 uint32_t irq = 0U;
412 if (!board_periph_next_irq(&irq)) {
413 return false;
414 }
415 const uint8_t prio_byte = (uint8_t)rd32(uc, (uint64_t)k_nvic_ipr_base + irq);
416 const uint32_t prio =
417 (uint32_t)(prio_byte >> (uint32_t)k_nvic_prio_shift) & (uint32_t)k_lo4_mask; /* 4 MSBs used */
418 if (prio >= active) {
419 return false; /* an equal/higher-priority handler is active -- defer */
420 }
421 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_irq_vec0 + irq);
422 if (handler == 0U) {
423 return false; /* no ISR installed: drop (default handler would just return) */
424 }
425 /* Clear the NVIC ISPR pending bit on activation, as hardware does. */
426 const uint64_t ispr_word = (uint64_t)k_nvic_ispr_base + ((uint64_t)(irq / 32U) * 4U);
427 uint32_t ispr = rd32(uc, ispr_word);
428 ispr &= ~(1U << (irq % 32U));
429 wr32(uc, ispr_word, ispr);
430
431 exc_enter(uc, (uint32_t)k_exc_irq_vec0 + irq, handler);
432 board_periph_note_irq_taken(irq);
433 return true;
434}

482bool exc_take_pending(uc_engine* uc, uint32_t vtor_base, bool allow_systick)
483{
484 const uint32_t primask = reg_get(uc, UC_ARM_REG_PRIMASK);
485 if ((primask & 1U) != 0U) {
486 return false; /* interrupts masked -- no exception may be taken now */
487 }
488 const uint32_t active = exc_active_prio();
489
490 /* SysTick first: highest-priority of the modelled exceptions, so it can
491 * pre-empt a lower-priority PendSV that is spinning for a runnable thread.
492 * Skipped on a context-switch stop (allow_systick == false) so the tick does
493 * not advance while ready threads still have work to run. */
494 if (allow_systick && s_systick_pending) {
495 const bool armed =
496 (rd32(uc, (uint64_t)k_syst_csr) & (uint32_t)k_syst_csr_run) == (uint32_t)k_syst_csr_run;
497 if (!armed) {
498 s_systick_pending = false; /* disabled SysTick: drop the pended tick */
499 } else if (exc_priority(uc, (uint32_t)k_exc_systick) < active) {
500 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_systick);
501 if (handler != 0U) {
502 s_systick_pending = false;
503 exc_enter(uc, (uint32_t)k_exc_systick, handler);
504 s_systick_fires++;
505 return true;
506 }
507 }
508 }
509
510 /* PendSV: taken when the firmware has requested a context switch. */
511 uint32_t icsr = rd32(uc, (uint64_t)k_scb_icsr);
512 if ((icsr & (1U << (uint32_t)k_icsr_pendsvset)) != 0U) {
513 if (exc_priority(uc, (uint32_t)k_exc_pendsv) < active) {
514 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_pendsv);
515 if (handler != 0U) {
516 /* Hardware clears PENDSVSET when PendSV is activated. */
517 icsr &= ~(1U << (uint32_t)k_icsr_pendsvset);
518 wr32(uc, (uint64_t)k_scb_icsr, icsr);
519 exc_enter(uc, (uint32_t)k_exc_pendsv, handler);
520 s_pendsv_takes++;
521 return true;
522 }
523 }
524 }
525
526 /* Peripheral NVIC IRQs queued by the ICU model (timer overflow / underflow
527 * routed through IELSR). Taken last among the modelled exceptions but via the
528 * identical real entry/return path, with NVIC IPR priority honoured. */
529 return exc_take_periph_irq(uc, vtor_base, active);
530}

553static bool
554on_unmapped(uc_engine* uc, uc_mem_type type, uint64_t addr, int size, int64_t value, void* user)
555{
556 (void)size;
557 (void)value;
558 (void)user;
559 /* A FETCH into the EXC_RETURN range is not a fault -- it is the core taking
560 * an exception return (the handler ran "BX lr" with an EXC_RETURN magic in
561 * LR). Capture it and stop cleanly so the run loop can unstack the frame and
562 * resume; the Unicorn fetch-fault error this produces is expected/handled. */
563 if ((type == UC_MEM_FETCH_UNMAPPED) && is_exc_return(addr)) {
564 s_exc_return_pc = addr;
565 s_exc_return_hit = true;
566 (void)uc_emu_stop(uc);
567 return false;
568 }
569 (void)fprintf(stderr,
570 " UNMAPPED %s @ 0x%08llX (extend the memory/peripheral map)\n",
571 unmapped_access_kind(type),
572 (unsigned long long)addr);
573 return false; /* stop emulation and report */
574}

681static void on_intr(uc_engine* uc, uint32_t int_no, void* user_data)
682{
683 (void)int_no;
684 uint32_t pc = 0U;
685 (void)uc_reg_read(uc, UC_ARM_REG_PC, &pc);
686
687 /* Cortex-M exception RETURN. Unicorn's M-profile core does not pop the
688 * exception frame itself, but it DOES trap a branch to an EXC_RETURN magic
689 * (0xFFFFFFFx) by raising an interrupt with the magic left in PC (Thumb bit
690 * masked off, but the stack/mode selector bits 2/3 intact). Unstack the basic
691 * frame and resume the interrupted context. This is how every handler that
692 * ra8_emulator vectors in (SysTick / PendSV / SVCall) returns. */
693 if (is_exc_return((uint64_t)pc)) {
694 s_exc_return_pc = (uint64_t)pc;
695 s_exc_return_hit = true;
696 (void)uc_emu_stop(uc);
697 return;
698 }
699
700 /* MVE (Helium) contiguous load/store raises a NoCP UsageFault on Unicorn's
701 * M33; ::emu_mve_nocp_emulate documents the decode and why it precedes SVCall. */
702 if (emu_mve_nocp_emulate(uc, pc)) {
703 (void)uc_emu_stop(uc);
704 return;
705 }
706
707 /* Armv8-M secure gateway: every Non-Secure-Callable veneer starts with `SG`
708 * (0xE97FE97F) then `B.W __acle_se_<fn>`. Unicorn's M33 has no Security
709 * Extension, so it raises INTR on the unrecognised SG instead of switching to
710 * Secure state. ra8_emulator has a single flat address space (no S/NS split), so
711 * the faithful model is to treat SG as a NOP and let the following branch reach
712 * the secure entry directly. Without this the SG is mis-taken as an `svc` and
713 * re-taken forever (the firmware has no SVC), looping until the stack
714 * underflows -- the tz_nsc_cgc_usb fault. The matching `BXNS`/`BLXNS` returns
715 * are handled below. */
716 uint32_t insn = 0U;
717 (void)uc_mem_read(uc, (uint64_t)pc, &insn, sizeof(insn));
718
719 /* Armv8-M Security Extension instructions (SG / FPCXTNS) reduce to their plain
720 * effects in this single-domain model; a BKPT is a deliberate firmware halt. */
721 if (on_intr_sec_insn(uc, pc, insn)) {
722 return;
723 }
724 if (on_intr_bkpt(uc, pc, insn)) {
725 return;
726 }
727
728 /* Otherwise it is a synchronous `svc` -- take SVCall (#11). ThreadX in single
729 * mode never issues one, but bare-metal / future RTOS first-thread-start
730 * paths do. The VTOR fallback is the MRAM vector-table base; the live VTOR
731 * (set by SystemInit) is read inside exc_vector. */
732 (void)user_data;
733 const uint32_t vtor_base = (uint32_t)emu_memmap_mram_base();
734 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_svcall);
735 if (handler != 0U) {
736 exc_enter(uc, (uint32_t)k_exc_svcall, handler);
737 s_svc_takes++;
738 }
739 (void)uc_emu_stop(uc);
740}

773static void
774on_icsr_write(uc_engine* uc, uc_mem_type type, uint64_t addr, int size, int64_t value, void* user)
775{
776 (void)type;
777 (void)addr;
778 (void)size;
779 (void)user;
780 if (((uint32_t)value & (1U << (uint32_t)k_icsr_pendsvset)) == 0U) {
781 return; /* not a PendSV request (e.g. PENDSVCLR / status write) */
782 }
783 /* Only end the chunk when PendSV could actually activate now: interrupts
784 * unmasked and no equal/higher-priority handler already running. If masked
785 * (the store sits inside a ThreadX critical section), do NOT stop -- the run
786 * loop would otherwise relaunch from this same store and re-pend forever,
787 * since uc_emu_stop here leaves PC on the store. With those guards, the run
788 * loop's exc_take_pending takes PendSV and moves PC to the handler, so the
789 * store is never re-executed. The masked case is picked up at the next
790 * boundary once TX_RESTORE re-enables interrupts. */
791 uint32_t primask = 0U;
792 (void)uc_reg_read(uc, UC_ARM_REG_PRIMASK, &primask);
793 if ((primask & 1U) != 0U) {
794 return;
795 }
796 if (exc_priority(uc, (uint32_t)k_exc_pendsv) >= exc_active_prio()) {
797 return; /* a higher/equal-priority handler is active -- defer */
798 }
799 /* Advance PC past the storing instruction before stopping so the PendSV we
800 * are about to take stacks the return address of the NEXT instruction -- as
801 * real hardware does (PendSV activates after the store retires), not the store
802 * itself. Re-stacking the store would re-pend PendSV on every return and spin.
803 * Thumb length: a halfword whose top 5 bits are 0b111xx with xx != 00 starts a
804 * 32-bit instruction; otherwise it is 16-bit. */
805 uint32_t pc = 0U;
806 (void)uc_reg_read(uc, UC_ARM_REG_PC, &pc);
807 uint16_t hw0 = 0U;
808 (void)uc_mem_read(uc, pc, &hw0, sizeof(hw0));
809 const uint32_t op5 = (uint32_t)(hw0 >> (uint32_t)k_thumb_op5_shift) & (uint32_t)k_thumb_op5_mask;
810 const uint32_t step = (op5 >= (uint32_t)k_thumb32_op5_min) ? 4U : 2U;
811 uint32_t next = pc + step;
812 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
813 /* Mark this as a context-switch stop so the run loop takes PendSV WITHOUT
814 * advancing the SysTick: a context switch consumes no modelled time, so a
815 * thread that suspends (e.g. tx_thread_sleep) must not have its tick-based
816 * wait expired the instant it yields -- that would starve lower-priority
817 * ready threads. Time (SysTick) advances only on a full-budget run (genuine
818 * execution / idle spin). See the run loop's inner dispatch. */
819 s_pendsv_stop = true;
820 (void)uc_emu_stop(uc);
821}
