51/* Hand-modelled Cortex-M exception state. Unicorn's M33 core has no NVIC /
52 * exception unit, so ra8_emulator takes SysTick / PendSV / SVCall by hand: it
53 * tracks the active-exception priority stack here (everything else -- MSP/PSP/
54 * CONTROL/xPSR/PRIMASK -- is read straight from Unicorn). s_exc_stack holds the
55 * priority of each handler currently active so a higher-priority exception
56 * (e.g. SysTick, prio 0x40) can pre-empt a lower one (PendSV, prio 0xFF) but
57 * not vice-versa, exactly as the real priority logic would nest them. */

59static uint32_t s_exc_depth;
60static uint32_t s_pendsv_takes;

99static uint32_t exc_priority(uc_engine* uc, uint32_t exc_num)
100{
101 if (exc_num == (uint32_t)k_exc_svcall) {
102 return (rd32(uc, (uint64_t)k_scb_shpr2) >> (3U * (uint32_t)k_byte_bits)) &
103 (uint32_t)k_byte_mask;
104 }
105 if (exc_num == (uint32_t)k_exc_pendsv) {
106 return (rd32(uc, (uint64_t)k_scb_shpr3) >> (2U * (uint32_t)k_byte_bits)) &
107 (uint32_t)k_byte_mask;
108 }
109 if (exc_num == (uint32_t)k_exc_systick) {
110 return (rd32(uc, (uint64_t)k_scb_shpr3) >> (3U * (uint32_t)k_byte_bits)) &
111 (uint32_t)k_byte_mask;
112 }
113 return (uint32_t)k_exc_prio_max;

117static void exc_stack_frame(uc_engine* uc, uint32_t sp, bool fp_active, uint32_t frame_xpsr)
118{
119 wr32(uc, (uint64_t)sp + 0U, reg_get(uc, UC_ARM_REG_R0));
120 wr32(uc, (uint64_t)sp + 4U, reg_get(uc, UC_ARM_REG_R1));
121 wr32(uc, (uint64_t)sp + 8U, reg_get(uc, UC_ARM_REG_R2));
122 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_r3, reg_get(uc, UC_ARM_REG_R3));
123 wr32(uc, (uint64_t)sp + 16U, reg_get(uc, UC_ARM_REG_R12));
124 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_lr, reg_get(uc, UC_ARM_REG_LR));
125 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_pc, reg_get(uc, UC_ARM_REG_PC));
126 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_xpsr, frame_xpsr);
127
128 /* Armv8-M FP extended frame: S0-S15 + FPSCR sit directly above the 8-word
129 * basic frame (ThreadX's PendSV adds S16-S31 on top of this). */
130 if (fp_active) {
131 for (uint32_t i = 0U; i < (uint32_t)k_fp_s_words; i++) {
132 wr32(uc,
133 (uint64_t)sp + (uint64_t)k_frame_off_s0 + (uint64_t)(i * (uint32_t)k_word_bytes),
134 reg_get(uc, UC_ARM_REG_S0 + (int)i));
135 }
136 wr32(uc, (uint64_t)sp + (uint64_t)k_frame_off_fpscr, reg_get(uc, UC_ARM_REG_FPSCR));
137 }
138}

141static uint32_t exc_return_value(bool in_thread, bool use_psp, bool fp_active)
142{
143 /* EXC_RETURN encodes where to unstack: Thread/PSP, Thread/MSP, or (when an
144 * exception pre-empts another) Handler/MSP. */
145 uint32_t exc_ret;
146 if (!in_thread) {
147 exc_ret = (uint32_t)k_exc_ret_handler;
148 } else if (use_psp) {
149 exc_ret = (uint32_t)k_exc_ret_psp;
150 } else {
151 exc_ret = (uint32_t)k_exc_ret_msp;
152 }
153 if (fp_active) {
154 exc_ret &= ~(uint32_t)k_exc_ret_ftype; /* FType=0: an FP frame was stacked. */
155 }
156 return exc_ret;
157}

186void exc_enter(uc_engine* uc, uint32_t exc_num, uint32_t handler)
187{
188 const uint32_t xpsr_in = reg_get(uc, UC_ARM_REG_XPSR);
189 const uint32_t control = reg_get(uc, UC_ARM_REG_CONTROL);
190 const bool in_thread = (xpsr_in & (uint32_t)k_xpsr_ipsr_mask) == 0U;
191 const bool use_psp = in_thread && ((control & (uint32_t)k_control_spsel) != 0U);
192 const bool fp_active = (control & (uint32_t)k_control_fpca) != 0U;
193
194 const int sp_reg = use_psp ? UC_ARM_REG_PSP : UC_ARM_REG_MSP;
195 uint32_t sp = reg_get(uc, sp_reg);
196
197 /* Hardware aligns the stack pointer to 8 bytes on entry and flags the pad in
198 * the stacked xPSR (bit 9) so the matching exception return can remove it. */
199 uint32_t frame_xpsr = xpsr_in;
200 if ((sp & 0x4U) != 0U) {
201 sp -= 4U;
202 frame_xpsr |= (uint32_t)k_xpsr_align9;
203 } else {
204 frame_xpsr &= ~(uint32_t)k_xpsr_align9;
205 }
206 sp -= (uint32_t)k_exc_frame_bytes;
207 if (fp_active) {
208 sp -= (uint32_t)k_fp_frame_extra;
209 }
210
211 exc_stack_frame(uc, sp, fp_active, frame_xpsr);
212
213 /* Commit the new value of whichever stack the frame went onto. */
214 reg_set(uc, sp_reg, sp);
215
216 const uint32_t exc_ret = exc_return_value(in_thread, use_psp, fp_active);
217
218 /* Handler mode always runs on MSP with CONTROL.SPSEL clear. */
219 reg_set(uc, UC_ARM_REG_CONTROL, control & ~(uint32_t)k_control_spsel);
220 reg_set(uc, UC_ARM_REG_SP, reg_get(uc, UC_ARM_REG_MSP));
221
222 uint32_t handler_xpsr =
223 (xpsr_in & ~(uint32_t)k_xpsr_ipsr_mask) | (exc_num & (uint32_t)k_xpsr_ipsr_mask);
224 handler_xpsr |= (uint32_t)k_xpsr_t_bit; /* M-profile is always Thumb. */
225 reg_set(uc, UC_ARM_REG_XPSR, handler_xpsr);
226 reg_set(uc, UC_ARM_REG_LR, exc_ret);
227 reg_set(uc, UC_ARM_REG_PC, handler & ~1U);
228
229 if (s_exc_depth < (uint32_t)k_exc_nest_max) {
230 s_exc_stack[s_exc_depth] = exc_priority(uc, exc_num);
231 s_exc_depth++;
232 }
233}

236static void exc_restore_fp_frame(uc_engine* uc, uint32_t sp)
237{
238 /* Armv8-M FP extended frame (EXC_RETURN bit4 clear): S0-S15 + FPSCR sit above
239 * the basic frame. Restore them so the thread's scalar FP state survives.
240 * PC/xPSR keep their basic-frame offsets (accounted for by the caller). */
241 for (uint32_t i = 0U; i < (uint32_t)k_fp_s_words; i++) {
242 reg_set(
243 uc,
244 UC_ARM_REG_S0 + (int)i,
245 rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_s0 + (uint64_t)(i * (uint32_t)k_word_bytes)));
246 }
247 reg_set(uc, UC_ARM_REG_FPSCR, rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_fpscr));
248}

251static void exc_restore_mode(uc_engine* uc, uint32_t xpsr, bool to_thread, bool to_psp)
252{
253 /* Restore mode: on return to Thread, CONTROL.SPSEL follows EXC_RETURN bit2;
254 * on return to a pre-empted handler the core stays on MSP. */
255 uint32_t control = reg_get(uc, UC_ARM_REG_CONTROL);
256 if (to_thread && to_psp) {
257 control |= (uint32_t)k_control_spsel;
258 } else {
259 control &= ~(uint32_t)k_control_spsel;
260 }
261 reg_set(uc, UC_ARM_REG_CONTROL, control);
262
263 uint32_t new_xpsr = xpsr | (uint32_t)k_xpsr_t_bit;
264 if (to_thread) {
265 new_xpsr &= ~(uint32_t)k_xpsr_ipsr_mask; /* Thread mode: IPSR == 0 */
266 }
267 reg_set(uc, UC_ARM_REG_XPSR, new_xpsr);
268}

294void exc_return(uc_engine* uc, uint32_t exc_ret)
295{
296 const bool to_psp = (exc_ret & (uint32_t)k_exc_ret_spsel) != 0U;
297 const bool to_thread = (exc_ret & (uint32_t)k_exc_ret_mode) != 0U;
298 const bool fp_frame = (exc_ret & (uint32_t)k_exc_ret_ftype) == 0U;
299 const int sp_reg = to_psp ? UC_ARM_REG_PSP : UC_ARM_REG_MSP;
300 uint32_t sp = reg_get(uc, sp_reg);
301
302 const uint32_t r0 = rd32(uc, (uint64_t)sp + 0U);
303 const uint32_t r1 = rd32(uc, (uint64_t)sp + 4U);
304 const uint32_t r2 = rd32(uc, (uint64_t)sp + 8U);
305 const uint32_t r3 = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_r3);
306 const uint32_t r12 = rd32(uc, (uint64_t)sp + 16U);
307 const uint32_t lr = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_lr);
308 const uint32_t pc = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_pc);
309 const uint32_t xpsr = rd32(uc, (uint64_t)sp + (uint64_t)k_frame_off_xpsr);
310
311 if (fp_frame) {
312 exc_restore_fp_frame(uc, sp);
313 }
314
315 sp += (uint32_t)k_exc_frame_bytes;
316 if (fp_frame) {
317 sp += (uint32_t)k_fp_frame_extra;
318 }
319 if ((xpsr & (uint32_t)k_xpsr_align9) != 0U) {
320 sp += 4U; /* undo the entry-time 8-byte realignment pad */
321 }
322 reg_set(uc, sp_reg, sp);
323
324 reg_set(uc, UC_ARM_REG_R0, r0);
325 reg_set(uc, UC_ARM_REG_R1, r1);
326 reg_set(uc, UC_ARM_REG_R2, r2);
327 reg_set(uc, UC_ARM_REG_R3, r3);
328 reg_set(uc, UC_ARM_REG_R12, r12);
329 reg_set(uc, UC_ARM_REG_LR, lr);
330
331 exc_restore_mode(uc, xpsr, to_thread, to_psp);
332
333 /* Active SP becomes whichever stack the returned-to context uses. */
334 reg_set(uc, UC_ARM_REG_SP, reg_get(uc, to_psp && to_thread ? UC_ARM_REG_PSP : UC_ARM_REG_MSP));
335 reg_set(uc, UC_ARM_REG_PC, pc & ~1U);
336
337 if (s_exc_depth > 0U) {
338 s_exc_depth--;
339 }
340}

410static bool exc_take_periph_irq(uc_engine* uc, uint32_t vtor_base, uint32_t active)
411{
412 uint32_t irq = 0U;
413 if (!board_periph_next_irq(&irq)) {
414 return false;
415 }
416 const uint8_t prio_byte = (uint8_t)rd32(uc, (uint64_t)k_nvic_ipr_base + irq);
417 const uint32_t prio =
418 (uint32_t)(prio_byte >> (uint32_t)k_nvic_prio_shift) & (uint32_t)k_lo4_mask; /* 4 MSBs used */
419 if (prio >= active) {
420 return false; /* an equal/higher-priority handler is active -- defer */
421 }
422 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_irq_vec0 + irq);
423 if (handler == 0U) {
424 return false; /* no ISR installed: drop (default handler would just return) */
425 }
426 /* Clear the NVIC ISPR pending bit on activation, as hardware does. */
427 const uint64_t ispr_word = (uint64_t)k_nvic_ispr_base + ((uint64_t)(irq / 32U) * 4U);
428 uint32_t ispr = rd32(uc, ispr_word);
429 ispr &= ~(1U << (irq % 32U));
430 wr32(uc, ispr_word, ispr);
431
432 exc_enter(uc, (uint32_t)k_exc_irq_vec0 + irq, handler);
433 board_periph_note_irq_taken(irq);
434 return true;
435}

483bool exc_take_pending(uc_engine* uc, uint32_t vtor_base, bool allow_systick)
484{
485 const uint32_t primask = reg_get(uc, UC_ARM_REG_PRIMASK);
486 if ((primask & 1U) != 0U) {
487 return false; /* interrupts masked -- no exception may be taken now */
488 }
489 const uint32_t active = exc_active_prio();
490
491 /* SysTick first: highest-priority of the modelled exceptions, so it can
492 * pre-empt a lower-priority PendSV that is spinning for a runnable thread.
493 * Skipped on a context-switch stop (allow_systick == false) so the tick does
494 * not advance while ready threads still have work to run. */
495 if (allow_systick && s_systick_pending) {
496 const bool armed =
497 (rd32(uc, (uint64_t)k_syst_csr) & (uint32_t)k_syst_csr_run) == (uint32_t)k_syst_csr_run;
498 if (!armed) {
499 s_systick_pending = false; /* disabled SysTick: drop the pended tick */
500 } else if (exc_priority(uc, (uint32_t)k_exc_systick) < active) {
501 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_systick);
502 if (handler != 0U) {
503 s_systick_pending = false;
504 exc_enter(uc, (uint32_t)k_exc_systick, handler);
505 s_systick_fires++;
506 return true;
507 }
508 }
509 }
510
511 /* PendSV: taken when the firmware has requested a context switch. */
512 uint32_t icsr = rd32(uc, (uint64_t)k_scb_icsr);
513 if ((icsr & (1U << (uint32_t)k_icsr_pendsvset)) != 0U) {
514 if (exc_priority(uc, (uint32_t)k_exc_pendsv) < active) {
515 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_pendsv);
516 if (handler != 0U) {
517 /* Hardware clears PENDSVSET when PendSV is activated. */
518 icsr &= ~(1U << (uint32_t)k_icsr_pendsvset);
519 wr32(uc, (uint64_t)k_scb_icsr, icsr);
520 exc_enter(uc, (uint32_t)k_exc_pendsv, handler);
521 s_pendsv_takes++;
522 return true;
523 }
524 }
525 }
526
527 /* Peripheral NVIC IRQs queued by the ICU model (timer overflow / underflow
528 * routed through IELSR). Taken last among the modelled exceptions but via the
529 * identical real entry/return path, with NVIC IPR priority honoured. */
530 return exc_take_periph_irq(uc, vtor_base, active);
531}

554static bool
555on_unmapped(uc_engine* uc, uc_mem_type type, uint64_t addr, int size, int64_t value, void* user)
556{
557 (void)size;
558 (void)value;
559 (void)user;
560 /* A FETCH into the EXC_RETURN range is not a fault -- it is the core taking
561 * an exception return (the handler ran "BX lr" with an EXC_RETURN magic in
562 * LR). Capture it and stop cleanly so the run loop can unstack the frame and
563 * resume; the Unicorn fetch-fault error this produces is expected/handled. */
564 if ((type == UC_MEM_FETCH_UNMAPPED) && is_exc_return(addr)) {
565 s_exc_return_pc = addr;
566 s_exc_return_hit = true;
567 (void)uc_emu_stop(uc);
568 return false;
569 }
570 (void)fprintf(stderr,
571 " UNMAPPED %s @ 0x%08llX (extend the memory/peripheral map)\n",
572 unmapped_access_kind(type),
573 (unsigned long long)addr);
574 return false; /* stop emulation and report */
575}

682static void on_intr(uc_engine* uc, uint32_t int_no, void* user_data)
683{
684 (void)int_no;
685 uint32_t pc = 0U;
686 (void)uc_reg_read(uc, UC_ARM_REG_PC, &pc);
687
688 /* Cortex-M exception RETURN. Unicorn's M-profile core does not pop the
689 * exception frame itself, but it DOES trap a branch to an EXC_RETURN magic
690 * (0xFFFFFFFx) by raising an interrupt with the magic left in PC (Thumb bit
691 * masked off, but the stack/mode selector bits 2/3 intact). Unstack the basic
692 * frame and resume the interrupted context. This is how every handler that
693 * ra8_emulator vectors in (SysTick / PendSV / SVCall) returns. */
694 if (is_exc_return((uint64_t)pc)) {
695 s_exc_return_pc = (uint64_t)pc;
696 s_exc_return_hit = true;
697 (void)uc_emu_stop(uc);
698 return;
699 }
700
701 /* MVE (Helium) contiguous load/store raises a NoCP UsageFault on Unicorn's
702 * M33; ::emu_mve_nocp_emulate documents the decode and why it precedes SVCall. */
703 if (emu_mve_nocp_emulate(uc, pc)) {
704 (void)uc_emu_stop(uc);
705 return;
706 }
707
708 /* Armv8-M secure gateway: every Non-Secure-Callable veneer starts with `SG`
709 * (0xE97FE97F) then `B.W __acle_se_<fn>`. Unicorn's M33 has no Security
710 * Extension, so it raises INTR on the unrecognised SG instead of switching to
711 * Secure state. ra8_emulator has a single flat address space (no S/NS split), so
712 * the faithful model is to treat SG as a NOP and let the following branch reach
713 * the secure entry directly. Without this the SG is mis-taken as an `svc` and
714 * re-taken forever (the firmware has no SVC), looping until the stack
715 * underflows -- the tz_nsc_cgc_usb fault. The matching `BXNS`/`BLXNS` returns
716 * are handled below. */
717 uint32_t insn = 0U;
718 (void)uc_mem_read(uc, (uint64_t)pc, &insn, sizeof(insn));
719
720 /* Armv8-M Security Extension instructions (SG / FPCXTNS) reduce to their plain
721 * effects in this single-domain model; a BKPT is a deliberate firmware halt. */
722 if (on_intr_sec_insn(uc, pc, insn)) {
723 return;
724 }
725 if (on_intr_bkpt(uc, pc, insn)) {
726 return;
727 }
728
729 /* Otherwise it is a synchronous `svc` -- take SVCall (#11). ThreadX in single
730 * mode never issues one, but bare-metal / future RTOS first-thread-start
731 * paths do. The VTOR fallback is the MRAM vector-table base; the live VTOR
732 * (set by SystemInit) is read inside exc_vector. */
733 (void)user_data;
734 const uint32_t vtor_base = (uint32_t)emu_memmap_mram_base();
735 const uint32_t handler = exc_vector(uc, vtor_base, (uint32_t)k_exc_svcall);
736 if (handler != 0U) {
737 exc_enter(uc, (uint32_t)k_exc_svcall, handler);
738 s_svc_takes++;
739 }
740 (void)uc_emu_stop(uc);
741}

774static void
775on_icsr_write(uc_engine* uc, uc_mem_type type, uint64_t addr, int size, int64_t value, void* user)
776{
777 (void)type;
778 (void)addr;
779 (void)size;
780 (void)user;
781 if (((uint32_t)value & (1U << (uint32_t)k_icsr_pendsvset)) == 0U) {
782 return; /* not a PendSV request (e.g. PENDSVCLR / status write) */
783 }
784 /* Only end the chunk when PendSV could actually activate now: interrupts
785 * unmasked and no equal/higher-priority handler already running. If masked
786 * (the store sits inside a ThreadX critical section), do NOT stop -- the run
787 * loop would otherwise relaunch from this same store and re-pend forever,
788 * since uc_emu_stop here leaves PC on the store. With those guards, the run
789 * loop's exc_take_pending takes PendSV and moves PC to the handler, so the
790 * store is never re-executed. The masked case is picked up at the next
791 * boundary once TX_RESTORE re-enables interrupts. */
792 uint32_t primask = 0U;
793 (void)uc_reg_read(uc, UC_ARM_REG_PRIMASK, &primask);
794 if ((primask & 1U) != 0U) {
795 return;
796 }
797 if (exc_priority(uc, (uint32_t)k_exc_pendsv) >= exc_active_prio()) {
798 return; /* a higher/equal-priority handler is active -- defer */
799 }
800 /* Advance PC past the storing instruction before stopping so the PendSV we
801 * are about to take stacks the return address of the NEXT instruction -- as
802 * real hardware does (PendSV activates after the store retires), not the store
803 * itself. Re-stacking the store would re-pend PendSV on every return and spin.
804 * Thumb length: a halfword whose top 5 bits are 0b111xx with xx != 00 starts a
805 * 32-bit instruction; otherwise it is 16-bit. */
806 uint32_t pc = 0U;
807 (void)uc_reg_read(uc, UC_ARM_REG_PC, &pc);
808 uint16_t hw0 = 0U;
809 (void)uc_mem_read(uc, pc, &hw0, sizeof(hw0));
810 const uint32_t op5 = (uint32_t)(hw0 >> (uint32_t)k_thumb_op5_shift) & (uint32_t)k_thumb_op5_mask;
811 const uint32_t step = (op5 >= (uint32_t)k_thumb32_op5_min) ? 4U : 2U;
812 uint32_t next = pc + step;
813 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
814 /* Mark this as a context-switch stop so the run loop takes PendSV WITHOUT
815 * advancing the SysTick: a context switch consumes no modelled time, so a
816 * thread that suspends (e.g. tx_thread_sleep) must not have its tick-based
817 * wait expired the instant it yields -- that would starve lower-priority
818 * ready threads. Time (SysTick) advances only on a full-budget run (genuine
819 * execution / idle spin). See the run loop's inner dispatch. */
820 s_pendsv_stop = true;
821 (void)uc_emu_stop(uc);
822}
