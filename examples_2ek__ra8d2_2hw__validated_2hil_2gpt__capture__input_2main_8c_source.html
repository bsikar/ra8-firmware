132static bool gpt_capture_edge(ra8_board_sw_state_t* prev)
133{
134 ra8_board_sw_state_t now = k_ra8_board_sw_released;
135 if (ra8_board_sw_read(k_ra8_board_sw1, &now) != k_ra8_ok) {
136 return false;
137 }
138 bool fell = (*prev == k_ra8_board_sw_released) && (now == k_ra8_board_sw_pressed);
139 *prev = now;
140 return fell;
141}

145int32_t main(void)
146{
147 gpt_capture_setup_or_halt();
148 ra8_isr_globals_enable();
149
150 ra8_board_sw_state_t last_state = k_ra8_board_sw_released;
151 uint32_t last_tick = 0U;
152 uint32_t prev_count = 0U;
153 bool have_prev = false;
154
155 while (1) {
156 if (gpt_capture_edge(&last_state)) {
157 const uint32_t now = gpt_capture_read();
158 const uint32_t last_delta = now - last_tick;
159 last_tick = now;
160 (void)last_delta;
161 (void)ra8_board_led_toggle(k_ra8_board_led1);
162 }
163 /* Liveness check independent of SW1 -- proves GTCNT is counting
164 * even when the button isn't pressed (which it never is on the
165 * unattended HIL bench). */
166 const uint32_t now_count = gpt_capture_read();
167 if (now_count != UINT32_MAX) {
168 if (have_prev && (now_count != prev_count)) {
169 g_gpt_capture_tick += 1U;
170 }
171 prev_count = now_count;
172 have_prev = true;
173 }
174 ra8_delay_ms(k_gpt_capture_poll_period_ms);
175 }
176 gpt_capture_panic_halt();
177 return 0;
178}
