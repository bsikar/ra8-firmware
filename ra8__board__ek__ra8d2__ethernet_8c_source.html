396RA8_INTERNAL static ra8_err_t internal_eth_rmac_program(uint32_t eswclk_hz)
397{
398 /* NOLINTBEGIN(clang-analyzer-optin.core.EnumCastOutOfRange) -- OR-combined MRAFC flags and the board RMAC port are valid values outside the enumerator lists. */
399 /* HUM Ch 33.4 "MRAFC : MAC Reception Address Filter Configuration
400 * Register" p 1717: each frame class has an ENABLE bit (UCENE/BCENE
401 * /MCENE in the [10:0] half + matching UCENP/BCENP/MCENP in the
402 * [26:16] half) AND a separate ACCEPT bit (BCACE bit 6, MCACE bit 5)
403 * that controls whether the MAC actually forwards the matched frame
404 * to the descriptor ring. Without BCACE the chip latches the ARP
405 * broadcast in the perfect-match comparator and then drops it before
406 * the GWCA agent ever sees it, so the host's ARP responder never
407 * sees a "who-has 192.168.1.42" request and never replies. FSP
408 * r_rmac.c::rmac_configure_reception_filter ORs both bits into the
409 * canonical promiscuous mask for the same reason. */
410 /* HUM Table 29.11: for an external RGMII link the RMAC's internal
411 * xMII (MPIC.PIS) is MII for 10/100 Mbps and GMII for 1 Gbps -- the
412 * ESWM media mux (MIICR1.MIISEL = 01b) does the RGMII conversion.
413 * Default to MII (10/100 Mbps) at init so the on-chip RGMII RX
414 * sample timing matches the most common PHY auto-neg outcome on
415 * this board. Issue #1 bench: starting in GMII (1 Gbps) silently
416 * dropped every RX frame when the PHY negotiated to 100 Mbps with
417 * the Pi USB-Ethernet adapter, and the link-up-gated resync inside
418 * ra8_eth_open never fired because chip-side MDIO reads of BMSR
419 * return 0x0000 (the PHY's BMSR.link bit is invisible until skew
420 * + clock are correct). Starting in MII makes RX work for any
421 * 10/100 link the PHY auto-negotiates; for 1 Gbps links the
422 * resync inside ra8_eth_link_status will promote PIS to GMII once
423 * MDIO becomes usable. */
424 const ra8_rmac_config_t rmac_cfg = {
425 .rx_filter = (ra8_rmac_mrafc_t)(k_ra8_rmac_mrafc_unicast_match | k_ra8_rmac_mrafc_broadcast |
426 k_ra8_rmac_mrafc_bc_accept),
427 .err_irq_enable = 0U,
428 .mon0_irq_enable = 0U,
429 .mon1_irq_enable = 0U,
430 .mon2_irq_enable = 0U,
431 .phy_interface = k_ra8_rmac_pis_mii,
432 .link_speed = k_ra8_rmac_lsc_100mbit,
433 .duplex = k_ra8_rmac_duplex_full,
434 .eswclk_hz = eswclk_hz,
435 .mdc_hz = (uint32_t)k_ra8_rmac_mdc_default_hz,
436 };
437 return ra8_rmac_init((ra8_rmac_port_t)k_ra8_board_eth_rmac_port, &rmac_cfg);
438 /* NOLINTEND(clang-analyzer-optin.core.EnumCastOutOfRange) */
439}

635ra8_err_t ra8_board_ethernet_init(void)
636{
637 /* Step 0: hardware-reset the on-board PEF7071 PHY before anything
638 * else touches MDIO. */
639 ra8_err_t err = internal_eth_phy_hw_reset();
640 if (err != k_ra8_ok) {
641 return err;
642 }
643 /* Step 1: route every remaining Ethernet pin to its RGMII alt mux. */
644 err = internal_eth_route_alt_pins();
645 if (err != k_ra8_ok) {
646 return err;
647 }
648 /* Step 2: bring up ESWCLK + ESWPHYCLK, release PDCTRESWM, lift the
649 * ETHPHYCLK / ESWM MSTP gates, and pulse the COMA agent out of
650 * reset. After this the per-port RMAC / ETHA register windows
651 * answer reads and accept writes. */
652 uint32_t eswclk_hz = 0U;
653 err = internal_eth_eswm_bring_up(&eswclk_hz);
654 if (err != k_ra8_ok) {
655 return err;
656 }
657 /* Step 3: ETHA1 RESET -> DISABLE -> CONFIG so RMAC.MPIC is writable. */
658 err = internal_eth_etha_to_config();
659 if (err != k_ra8_ok) {
660 return err;
661 }
662 /* Step 4: select RGMII on the media-interface mux + release the
663 * per-port reset, via the chip-generic ETH HAL primitive. The EK-RA8D2
664 * wires its PEF7071 / GPY111 PHY to RMAC port 1 -- the canonical FSP
665 * example ``ethernet_ek_ra8d2_ep`` configures every Ethernet pin as
666 * ``eswm_rgmii1`` and instantiates r_rmac on Channel 1; RMAC0 / ETHA0
667 * is unused on this board. ra8_eth_rgmii_select programs MIICR1 =
668 * TXCIDE | RGMII and sets MIIRR.RGRST1 = 1 (HUM 29.2.1.2: an enable,
669 * not an active-high reset). Without the RGRST1 enable, TXC is never
670 * generated, the RMAC RX state machine is unclocked, and every RMAC RX
671 * counter stays at 0 (bench-confirmed on EK-RA8D2). */
672 err = ra8_eth_rgmii_select((ra8_eth_mii_port_t)k_ra8_board_eth_rmac_port);
673 if (err != k_ra8_ok) {
674 /* ra8_eth_rgmii_select only fails on an out-of-range port; the board
675 * RMAC port constant is always valid. */
676 return err; /* GCOVR_EXCL_LINE */
677 }
678 /* Step 5: RMAC1 bring-up. Programs MPIC (PSMCS / PIS / LSC / PIPP),
679 * MAC address filter, and IRQ block. */
680 err = internal_eth_rmac_program(eswclk_hz);
681 if (err != k_ra8_ok) {
682 return err;
683 }
684 /* Step 6: promote ETHA1 to OPERATION so MDIO can drive MDC. */
685 err = internal_eth_etha_to_operation();
686 if (err != k_ra8_ok) {
687 return err;
688 }
689 /* Step 7: GPY111 PHY chip-init -- soft-reset, program the RGMII
690 * receive-timing skew (1.0 ns), then restart auto-negotiation so
691 * the link re-converges with the skew applied. Mirrors FSP
692 * R_RMAC_PHY_ChipInit + R_RMAC_PHY_StartAutoNegotiate. */
693 return internal_eth_phy_chip_init();
694}
