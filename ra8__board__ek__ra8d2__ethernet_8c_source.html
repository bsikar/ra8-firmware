281RA8_INTERNAL static ra8_err_t internal_eth_coma_reset(void)
282{
283 /* Faithful port of FSP r_layer3_switch_reset_coma. */
284
285 /* Step 1: RRC pulse + ~1 ms delay. */
286 *ra8_coma_rrc() = (uint32_t)k_ra8_coma_rrc_rr;
287 *ra8_coma_rrc() = 0U;
288 for (volatile uint32_t i = 0U; i < (uint32_t)k_ra8_board_eth_coma_delay_iters; ++i) {
289 __asm__ volatile("nop");
290 }
291
292 /* Step 2: enable the switch clock (RCE) alone, then settle. */
293 *ra8_coma_rcec() = (uint32_t)k_ra8_coma_rcec_rce;
294 for (volatile uint32_t i = 0U; i < (uint32_t)k_ra8_board_eth_coma_delay_iters; ++i) {
295 __asm__ volatile("nop");
296 }
297
298 /* Step 3: kick the COMA buffer-pool init and wait for BPR.
299 * HUM Ch 31.3.2.7 "CABPIRM" p 1599: writing BPIOG=1 starts the pool
300 * reset; BPR self-sets clk_period x 512 later. On the host build the
301 * CABPIRM register is mmap'd RAM with no hardware to self-set BPR, so
302 * the shared bounded waiter consults the ra8_fake_mmio fault seam --
303 * first-poll success unless a test arms a timeout on this register. */
304 *ra8_coma_cabpirm() = (uint32_t)k_ra8_coma_cabpirm_bpiog;
305 /* HUM Ch 31.3.2.7 "CABPIRM" p 1599 */
306 const ra8_err_t bpr_err = ra8_hw_wait_flag_set32(ra8_coma_cabpirm(),
307 (uint32_t)k_ra8_coma_cabpirm_bpr,
308 (uint32_t)k_ra8_board_eth_coma_bpr_poll_max);
309 if (bpr_err != k_ra8_ok) {
310 return bpr_err;
311 }
312
313 /* Step 4: fan out every per-agent clock (RCE | ACE[6:0]=ALL) so
314 * RMAC0/1 + ETHA0/1 + GWCA + MFWD + GPTP all become accessible. */
315 *ra8_coma_rcec() = (uint32_t)k_ra8_coma_rcec_rce | (uint32_t)k_ra8_coma_rcec_ace_mask;
316 for (volatile uint32_t i = 0U; i < (uint32_t)k_ra8_board_eth_coma_delay_iters; ++i) {
317 __asm__ volatile("nop");
318 }
319 return k_ra8_ok;
320}

356RA8_INTERNAL static ra8_err_t internal_eth_eswm_select_rgmii(void)
357{
358 /* HUM Ch 29 "ESWM" + FSP r_rmac_phy_set_mii_type_configuration:
359 * write MIICR before enabling the per-port RGMII block so the pin
360 * mux is in the correct mode the instant the data pins go live.
361 * MIICR / MIIRR live at the +0x19400 sub-block of the ESWM window
362 * (see ra8_ether_regs.h ra8_eswm_off_miicr1 / ra8_eswm_off_miirr). */
363 *ra8_eswm_miicr1() = (uint32_t)k_ra8_eswm_miicr_txcide | (uint32_t)k_ra8_eswm_miicr_miisel_rgmii;
364
365 /* HUM Ch 29.2.1.2 "MIIRR : Media-independent Interface Reset
366 * Register" p 1289: RGRST1 is **0 = Reset, 1 = Enable** -- it is an
367 * enable, not an active-high reset. Bench-confirmed on EK-RA8D2:
368 * with RGRST1 = 0 the RGMII1 block stays in reset, TXC is never
369 * generated, the RMAC RX state machine is unclocked, and every
370 * RMAC RX counter (MRFC, MRGFCE ...) stays at 0. FSP's
371 * r_rmac_phy_set_mii_type_configuration sets this bit with the
372 * comment "Enable TXC generation". SET it. */
373 uint32_t miirr = *ra8_eswm_miirr();
374 miirr |= (uint32_t)k_ra8_eswm_miirr_rgrst1;
375 *ra8_eswm_miirr() = miirr;
376 return k_ra8_ok;
377}

530RA8_INTERNAL static ra8_err_t internal_eth_rmac_program(uint32_t eswclk_hz)
531{
532 /* NOLINTBEGIN(clang-analyzer-optin.core.EnumCastOutOfRange) -- OR-combined MRAFC flags and the board RMAC port are valid values outside the enumerator lists. */
533 /* HUM Ch 33.4 "MRAFC : MAC Reception Address Filter Configuration
534 * Register" p 1707: each frame class has an ENABLE bit (UCENE/BCENE
535 * /MCENE in the [10:0] half + matching UCENP/BCENP/MCENP in the
536 * [26:16] half) AND a separate ACCEPT bit (BCACE bit 6, MCACE bit 5)
537 * that controls whether the MAC actually forwards the matched frame
538 * to the descriptor ring. Without BCACE the chip latches the ARP
539 * broadcast in the perfect-match comparator and then drops it before
540 * the GWCA agent ever sees it, so the host's ARP responder never
541 * sees a "who-has 192.168.1.42" request and never replies. FSP
542 * r_rmac.c::rmac_configure_reception_filter ORs both bits into the
543 * canonical promiscuous mask for the same reason. */
544 /* HUM Table 29.11: for an external RGMII link the RMAC's internal
545 * xMII (MPIC.PIS) is MII for 10/100 Mbps and GMII for 1 Gbps -- the
546 * ESWM media mux (MIICR1.MIISEL = 01b) does the RGMII conversion.
547 * Default to MII (10/100 Mbps) at init so the on-chip RGMII RX
548 * sample timing matches the most common PHY auto-neg outcome on
549 * this board. Issue #1 bench: starting in GMII (1 Gbps) silently
550 * dropped every RX frame when the PHY negotiated to 100 Mbps with
551 * the Pi USB-Ethernet adapter, and the link-up-gated resync inside
552 * ra8_eth_open never fired because chip-side MDIO reads of BMSR
553 * return 0x0000 (the PHY's BMSR.link bit is invisible until skew
554 * + clock are correct). Starting in MII makes RX work for any
555 * 10/100 link the PHY auto-negotiates; for 1 Gbps links the
556 * resync inside ra8_eth_link_status will promote PIS to GMII once
557 * MDIO becomes usable. */
558 const ra8_rmac_config_t rmac_cfg = {
559 .rx_filter = (ra8_rmac_mrafc_t)(k_ra8_rmac_mrafc_unicast_match | k_ra8_rmac_mrafc_broadcast |
560 k_ra8_rmac_mrafc_bc_accept),
561 .err_irq_enable = 0U,
562 .mon0_irq_enable = 0U,
563 .mon1_irq_enable = 0U,
564 .mon2_irq_enable = 0U,
565 .phy_interface = k_ra8_rmac_pis_mii,
566 .link_speed = k_ra8_rmac_lsc_100mbit,
567 .duplex = k_ra8_rmac_duplex_full,
568 .eswclk_hz = eswclk_hz,
569 .mdc_hz = (uint32_t)k_ra8_rmac_mdc_default_hz,
570 };
571 return ra8_rmac_init((ra8_rmac_port_t)k_ra8_board_eth_rmac_port, &rmac_cfg);
572 /* NOLINTEND(clang-analyzer-optin.core.EnumCastOutOfRange) */
573}

769ra8_err_t ra8_board_ethernet_init(void)
770{
771 /* Step 0: hardware-reset the on-board PEF7071 PHY before anything
772 * else touches MDIO. */
773 ra8_err_t err = internal_eth_phy_hw_reset();
774 if (err != k_ra8_ok) {
775 return err;
776 }
777 /* Step 1: route every remaining Ethernet pin to its RGMII alt mux. */
778 err = internal_eth_route_alt_pins();
779 if (err != k_ra8_ok) {
780 return err;
781 }
782 /* Step 2: bring up ESWCLK + ESWPHYCLK, release PDCTRESWM, lift the
783 * ETHPHYCLK / ESWM MSTP gates, and pulse the COMA agent out of
784 * reset. After this the per-port RMAC / ETHA register windows
785 * answer reads and accept writes. */
786 uint32_t eswclk_hz = 0U;
787 err = internal_eth_eswm_bring_up(&eswclk_hz);
788 if (err != k_ra8_ok) {
789 return err;
790 }
791 /* Step 3: ETHA1 RESET -> DISABLE -> CONFIG so RMAC.MPIC is writable. */
792 err = internal_eth_etha_to_config();
793 if (err != k_ra8_ok) {
794 return err;
795 }
796 /* Step 4: select RGMII on the media-interface mux + release the
797 * per-port reset. */
798 err = internal_eth_eswm_select_rgmii();
799 if (err != k_ra8_ok) {
800 return err;
801 }
802 /* Step 5: RMAC1 bring-up. Programs MPIC (PSMCS / PIS / LSC / PIPP),
803 * MAC address filter, and IRQ block. */
804 err = internal_eth_rmac_program(eswclk_hz);

815 * the link re-converges with the skew applied. Mirrors FSP
816 * R_RMAC_PHY_ChipInit + R_RMAC_PHY_StartAutoNegotiate. */
817 return internal_eth_phy_chip_init();
