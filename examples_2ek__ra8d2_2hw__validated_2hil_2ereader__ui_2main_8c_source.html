542/* ===========================================================================
543 * App framework integration (#146)
544 *
545 * Each screen is re-expressed as an ``ra8_app``: the screen-id stack (``s_nav``)
546 * remains the source of truth for *which* screen is active, and that id is the
547 * active app's id, so ::er_render_current focuses the matching app and renders
548 * its tree. The render callbacks are the existing ``er_render_*`` functions, so
549 * the composited chrome is byte-identical (``make ereader-golden`` stays green)
550 * while the control flow is now an app composition rather than a hardcoded
551 * if/else. ``RA8_APP_SETTINGS`` shows the build-time "uninstallable core" knob: a
552 * Settings app is only registered when the macro is defined.
553 * ===========================================================================
554 */

619void er_render_current(void)
620{
621 uint16_t top = (uint16_t)k_er_screen_library;
622 (void)ra8_ui_nav_top(&s_nav, &top);
623 /* The nav top is the active screen-app's id: focus it (idempotent if already
624 * focused, so no flicker) and render its widget tree. */
625 (void)ra8_app_launch(&s_app_reg, top);
626 (void)ra8_app_render(&s_app_reg);
627 /* Low-battery nag (#145/#146): an app-framework-level overlay widget drawn
628 * over whichever screen-app is active. A no-op at a healthy SOC, so the chrome
629 * golden (rendered at the default battery) stays byte-identical. */
630 if (s_batt_nag != k_ra8_batt_nag_none) {
631 ra8_widget_t nag = {.vt = &k_er_nag_vt, .visible = true, .dirty = true};
632 (void)ra8_widget_render_dirty(&nag, 1U);
633 }
634}

658{

668 g_er_last_hint = (uint32_t)dec.hint;
669 g_er_cur_page = s_reading_page;

672#pragma GCC diagnostic push
673#pragma GCC diagnostic ignored "-Wmain"
674int32_t main(void)
675{
676 app_bringup_clocks();
677 app_bringup_panel();
678 app_bringup_gfx();
679 app_bringup_touch();
680 (void)ra8_batt_monitor_init(&s_batt_mon); /* low-battery nag policy (#145/#146) */
681 /* User switches SW1/SW2 as inputs with internal pull-ups (active-low). Best-
682 * effort: a config failure just leaves page-turn on the touch path. */
683 (void)ra8_board_sw_init(k_ra8_board_sw1);
684 (void)ra8_board_sw_init(k_ra8_board_sw2);
685 er_try_load_font(); /* Best-effort SD font for the ra8_reflow Reading body. */
686
687 /* Default refresh cadence: fast A2 turns with a periodic GC16 clean (#78). */
688 (void)display_policy_init(&s_policy, k_display_policy_fast_clean, (uint16_t)k_er_clean_every);
689 (void)ra8_ui_nav_init(&s_nav, (uint16_t)k_er_screen_library);
690 er_apps_init(); /* re-express the screens as ra8_app instances (#146) */
691 er_render_current();
692 er_flush_event(k_display_event_open); /* boot -> a clean INIT panel update */
693
694 while (1) {
695 er_poll_touch(); /* fast cadence so taps feel responsive */
696 er_poll_buttons(); /* page-turn switches */
697 er_poll_battery(); /* self-throttled (~1 Hz) low-battery nag */
698 /* Heartbeat LED toggles on a slow sub-cadence so it blinks (~1 Hz) instead
699 * of strobing at the input-poll rate. */
700 if ((g_er_loop_ticks % (uint32_t)k_er_led_every) == 0U) {
701 (void)ra8_board_led_toggle(k_ra8_board_led_blue);
702 }
703 /* Free-running liveness counter: the jlink_memprobe HIL reads it twice across
704 * a window and asserts the render/poll loop advanced (it never WFIs). */
705 g_er_loop_ticks++;
706 ra8_delay_ms((uint32_t)k_er_frame_ms);
707 }
708 return 0;
709}
