123uint8_t priv_path_to_83(const char* path, uint8_t* out11)
124{
125 if (path == nullptr || out11 == nullptr) {
126 return 0U;
127 }
128 while (*path == '/') {
129 path++;
130 }
131 for (uint32_t i = 0; i < (uint32_t)k_max_8_3_name; i++) {
132 out11[i] = ' ';
133 }
134 if (priv_pack_base(&path, out11) == 0U) {
135 return 0U;

138 return 0U;

141 out11[0] = k_dir_marker_kanji_e5;
142 }
143 return 1U;

146/* `priv_83_to_str()`: see header for the documented contract. */
147void priv_83_to_str(const uint8_t* in11, char* out12)
148{
149 uint32_t i = 0;
150 uint32_t j = 0;
151 for (i = 0; i < (uint32_t)k_filename_base_len; i++) {
152 if (in11[i] == ' ') {
153 break;
154 }
155 out12[j++] = (char)in11[i];
156 }
157 /* Restore kanji escape. */
158 /* mcdc-deactivated: 3-condition AND on Shift-JIS kanji-escape directory entry; only reachable from a kanji-named FAT image, none of which exist in the test corpus. */
159 if (j > 0 && (uint8_t)out12[0] == k_dir_marker_kanji_e5 && in11[0] == k_dir_marker_kanji_e5) {
160 out12[0] = (char)k_dir_marker_free_used;
161 }
162 uint8_t has_ext = 0;
163 for (i = 0; i < k_filename_ext_len; i++) {
164 if (in11[k_filename_base_len + i] != ' ') {
165 has_ext = 1;
166 break;
167 }
168 }
169 if (has_ext != 0U) {
170 out12[j++] = '.';
171 for (i = 0; i < k_filename_ext_len; i++) {
172 if (in11[k_filename_base_len + i] == ' ') {
173 break;
174 }
175 out12[j++] = (char)in11[k_filename_base_len + i];
176 }
177 }
178 out12[j] = '\0';
179}

243/* `priv_dir_walk_next_sector()`: see header for the documented contract. */
244ra8_err_t priv_dir_walk_next_sector(const ra8_fs_mount_t* m, dir_walk_t* w, uint8_t* out_eod)
245{
246 *out_eod = 0;
247 if (w->is_root_fixed != 0U) {
248 if (w->fixed_remaining <= 1U) {
249 *out_eod = 1;
250 return k_ra8_ok;
251 }
252 w->fixed_remaining--;
253 w->cur_lba++;
254 w->entry_idx = 0;
255 return k_ra8_ok;
256 }
257 /* FAT32 cluster-chain root. */
258 w->sector_in_cluster++;
259 if (w->sector_in_cluster >= m->sectors_per_cluster) {
260 uint32_t next = 0;
261 ra8_err_t err = priv_fat_get(m, w->cluster, &next);
262 if (err != k_ra8_ok) {
263 return err;
264 }
265 if (priv_is_eoc(m, next) != 0U) {
266 *out_eod = 1;
267 return k_ra8_ok;
268 }
269 /* Cycle guard (NASA Rule 2): a healthy chain visits at most
270 * count_of_clusters distinct clusters; more means a corrupt loop. */
271 w->cluster_hops++;
272 if (w->cluster_hops > m->count_of_clusters) {
273 return k_ra8_err_protocol_error;
274 }
275 w->cluster = next;
276 w->sector_in_cluster = 0;
277 w->cur_lba = priv_cluster_to_lba(m, w->cluster);
278 } else {
279 w->cur_lba++;
280 }
281 w->entry_idx = 0;
282 return k_ra8_ok;
283}

286ra8_err_t priv_dir_find(const ra8_fs_mount_t* m,
287 const dir_loc_t* loc,
288 const uint8_t* name83,
289 uint32_t* out_lba,
290 uint32_t* out_entry_off,
291 uint8_t out_entry[k_ra8_fs_dir_entry_bytes])
292{
293 dir_walk_t w = {};
294 priv_dir_walk_init_loc(m, loc, &w);
295 uint8_t eod = 0;
296 uint8_t buf[k_ra8_fs_bytes_per_sector] = {};
297 while (eod == 0U) {
298 ra8_err_t err = priv_read_sector(m, w.cur_lba, buf);
299 if (err != k_ra8_ok) {
300 return err;
301 }
302 for (uint32_t e = 0; e < k_dir_entries_per_sector; e++) {
303 uint8_t* ent = &buf[(size_t)e * (size_t)k_ra8_fs_dir_entry_bytes];
304 if (ent[k_dir_off_name] == k_dir_marker_free_perm) {
305 return k_ra8_err_not_found;
306 }
307 if (ent[k_dir_off_name] == k_dir_marker_free_used) {
308 continue;
309 }
310 if (ent[k_dir_off_attr] == k_ra8_fs_attr_lfn) {
311 continue;
312 }
313 if (priv_byte_equal(ent, name83, k_dir_name_field_len) != 0U) {
314 *out_lba = w.cur_lba;
315 *out_entry_off = e * (uint32_t)k_ra8_fs_dir_entry_bytes;
316 priv_byte_copy(out_entry, ent, k_ra8_fs_dir_entry_bytes);
317 return k_ra8_ok;
318 }
319 }
320 err = priv_dir_walk_next_sector(m, &w, &eod);
321 if (err != k_ra8_ok) {
322 return err;
323 }
324 }
325 return k_ra8_err_not_found;
