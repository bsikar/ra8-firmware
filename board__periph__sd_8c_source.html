104static uint16_t board_sd_crc16(const uint8_t* data, uint32_t len)
105{
106 uint16_t crc = 0U;
107 for (uint32_t i = 0U; i < len; i++) {
108 crc ^= (uint16_t)((uint16_t)data[i] << (uint16_t)k_sd_byte_bits);
109 for (uint8_t b = 0U; b < (uint8_t)k_sd_byte_bits; b++) {
110 const bool msb = (crc & (uint16_t)k_sd_crc_msb) != 0U;
111 crc = (uint16_t)(crc << 1U);
112 if (msb) {
113 crc = (uint16_t)(crc ^ (uint16_t)k_sd_crc_poly);
114 }
115 }
116 }
117 return crc;
118}

160static void board_sd_stage_block(board_sd_state_t* c, const uint8_t* payload, uint32_t len)
161{
162 c->resp[0] = (uint8_t)k_sd_r1_ready;
163 c->resp[1] = (uint8_t)k_sd_tok_data;
164 memcpy(&c->resp[2], payload, len);
165 const uint16_t crc = board_sd_crc16(&c->resp[2], len);
166 c->resp[2U + len] = (uint8_t)(crc >> (uint16_t)k_sd_byte_bits);
167 c->resp[2U + len + 1U] = (uint8_t)(crc & (uint16_t)k_sd_byte_mask);
168 c->resp_len = 2U + len + 2U;
169 /* Console SD tab: anti-flood -- the first block and then every Nth, so a
170 * book-sized multi-block read shows activity without swamping the ring. */
171 s_sd_spi_block_reads++;
172 bool sd_report = (s_sd_spi_block_reads == 1U);
173 if ((s_sd_spi_block_reads % (uint32_t)k_sd_console_rd_every) == 0U) {
174 sd_report = true;
175 }
176 if (sd_report) {
177 char ln[k_sd_console_line_cap];
178 (void)
179 snprintf(ln, sizeof(ln), "SDSPI rd %uB (#%u)", (unsigned)len, (unsigned)s_sd_spi_block_reads);
180 board_console_push(k_board_console_ch_sd, ln);
181 }
182}

453static void board_sd_cmd_send_csd(board_sd_state_t* c)
454{
455 uint8_t csd[k_sd_csd_len];
456 memset(csd, 0, sizeof(csd));
457 csd[0] = (uint8_t)k_sd_csd_v2;
458 uint32_t csize = (uint32_t)k_sd_csd_csize; /* fallback: 8 MiB. */
459 if (c->image_len >= (uint32_t)k_sd_csd_unit) {
460 csize = (c->image_len / (uint32_t)k_sd_csd_unit) - 1U;
461 }
462 csd[k_sd_csd_csize_b7] =
463 (uint8_t)((csize >> (uint32_t)k_sd_arg_sh1) & (uint32_t)k_sd_csize_b7_mask);
464 csd[k_sd_csd_csize_b8] =
465 (uint8_t)((csize >> (uint32_t)k_sd_byte_bits) & (uint16_t)k_sd_byte_mask);
466 csd[k_sd_csd_off] = (uint8_t)(csize & (uint16_t)k_sd_byte_mask);
467 board_sd_stage_block(c, csd, (uint32_t)k_sd_csd_len);
468}

492static bool board_sd_dispatch_ident(board_sd_state_t* c, uint8_t idx, uint8_t r1)
493{
494 switch (idx) {
495 case (uint8_t)k_sd_idx_cmd0:
496 c->resp[0] = (uint8_t)k_sd_r1_idle;
497 c->resp_len = 1U;
498 break;
499 case (uint8_t)k_sd_idx_cmd8: /* R7: R1 + 0x000001AA echo. */
500 c->resp[0] = (uint8_t)k_sd_r1_idle;
501 c->resp[1] = 0U;
502 c->resp[2] = 0U;
503 c->resp[3] = 1U;
504 c->resp[4] = (uint8_t)k_sd_cmd8_echo;
505 c->resp_len = (uint8_t)k_sd_r7_len;
506 break;
507 case (uint8_t)k_sd_idx_cmd55:
508 c->app_cmd = true;
509 c->resp[0] = (uint8_t)k_sd_r1_idle;
510 c->resp_len = 1U;
511 break;
512 case (uint8_t)k_sd_idx_cmd58: /* R3: R1 + OCR (CCS=1 -> SDHC). */
513 c->resp[0] = r1;
514 c->resp[1] = (uint8_t)k_sd_ocr_pwrccs;
515 c->resp[2] = (uint8_t)k_sd_idle;
516 c->resp[3] = (uint8_t)k_sd_ocr_volt;
517 c->resp[4] = 0U;
518 c->resp_len = (uint8_t)k_sd_r7_len;
519 break;
520 case (uint8_t)k_sd_idx_cmd16:
521 c->resp[0] = (uint8_t)k_sd_r1_ready;
522 c->resp_len = 1U;
523 break;
524 default:
525 return false;
526 }
527 return true;
528}

610static void board_sd_dispatch_data(board_sd_state_t* c, uint8_t idx, uint32_t arg, uint8_t r1)
611{
612 switch (idx) {
613 case (uint8_t)k_sd_idx_cmd9: /* SEND_CSD: 16-byte CSD v2.0 data block. */
614 board_sd_cmd_send_csd(c);
615 break;
616 case (uint8_t)k_sd_idx_cmd17: /* READ_SINGLE_BLOCK (SDHC: arg = block). */
617 case (uint8_t)k_sd_idx_cmd18: /* READ_MULTIPLE_BLOCK -- streams to CMD12. */
618 board_sd_begin_read(c, idx, arg);
619 break;
620 case (uint8_t)k_sd_idx_cmd12: /* STOP_TRANSMISSION: end a CMD18 stream. */
621 board_sd_stop_read(c, r1);
622 break;
623 case (uint8_t)k_sd_idx_cmd24: /* WRITE_SINGLE_BLOCK */
624 case (uint8_t)k_sd_idx_cmd25: /* WRITE_MULTIPLE_BLOCK */
625 board_sd_begin_write(c, idx, arg, r1);
626 break;
627 case (uint8_t)k_sd_idx_cmd32: /* ERASE_WR_BLK_START (SDHC: arg = block). */
628 board_sd_cmd_erase_bound(c, &c->erase_start, arg, r1);
629 break;
630 case (uint8_t)k_sd_idx_cmd33: /* ERASE_WR_BLK_END (SDHC: arg = block). */
631 board_sd_cmd_erase_bound(c, &c->erase_end, arg, r1);
632 break;
633 case (uint8_t)k_sd_idx_cmd38: /* ERASE: zero the latched [start, end] range. */
634 board_sd_cmd_erase(c, r1);
635 break;
636 default:
637 c->resp[0] = r1;
638 c->resp_len = 1U;
639 break;
640 }
641}

655static void board_sd_process_cmd(board_sd_state_t* c)
656{
657 const uint8_t idx = (uint8_t)(c->cmd[0] & (uint8_t)k_sd_idx_mask);
658 const uint32_t arg = ((uint32_t)c->cmd[1] << (uint32_t)k_sd_arg_sh0) |
659 ((uint32_t)c->cmd[2] << (uint32_t)k_sd_arg_sh1) |
660 ((uint32_t)c->cmd[3] << (uint32_t)k_sd_byte_bits) | (uint32_t)c->cmd[4];
661 const bool was_app = c->app_cmd;
662 const uint8_t r1 = c->ready ? (uint8_t)k_sd_r1_ready : (uint8_t)k_sd_r1_idle;
663 c->app_cmd = false;
664 c->resp_pos = 0U;
665 c->resp_len = 0U;
666
667 if ((idx == (uint8_t)k_sd_idx_acmd41) && was_app) { /* ACMD41 -- init done. */
668 c->ready = true;
669 c->resp[0] = (uint8_t)k_sd_r1_ready;
670 c->resp_len = 1U;
671 return;
672 }
673 if (board_sd_dispatch_ident(c, idx, r1)) {
674 return;
675 }
676 board_sd_dispatch_data(c, idx, arg, r1);
677}

679uint8_t board_sd_exchange(uint8_t tx)
680{
681 if (s_sd.resp_pos < s_sd.resp_len) {
682 return s_sd.resp[s_sd.resp_pos++];
683 }
684 if (s_sd.wr_phase != k_sd_wr_idle) {
685 return board_sd_write_byte(&s_sd, tx);
686 }
687 if (s_sd.collecting) {
688 s_sd.cmd[s_sd.cmd_idx] = tx;
689 s_sd.cmd_idx++;
690 if (s_sd.cmd_idx >= (uint32_t)k_sd_cmd_len) {
691 s_sd.collecting = false;
692 board_sd_process_cmd(&s_sd);
693 }
694 return (uint8_t)k_sd_idle;
695 }
696 if ((tx & (uint8_t)k_sd_cmd_mask) == (uint8_t)k_sd_cmd_start) {
697 s_sd.collecting = true;
698 s_sd.cmd[0] = tx;
699 s_sd.cmd_idx = 1U;
700 return (uint8_t)k_sd_idle;
701 }
702 if (s_sd.rd_multi) {
703 /* Open CMD18 stream: the host clocking idle is fetching the next block.
704 * The command-start check above stays first so a CMD12 frame interrupts
705 * the stream instead of being swallowed as read clocking. */
706 board_sd_read_stream_next(&s_sd);
707 return s_sd.resp[s_sd.resp_pos++];
708 }
709 return (uint8_t)k_sd_idle;
710}

712bool board_sd_read_block(uint32_t lba, uint8_t* dst)
713{
714 if (!s_sd.attached || (s_sd.image == nullptr) || (dst == nullptr)) {
715 return false;
716 }
717 /* Same SDHC block-addressing as the CMD17 path: arg is a block index. */
718 const uint64_t off = (uint64_t)lba * (uint64_t)k_sd_block;
719 if (off >= s_sd.image_len) {
720 return false; /* past the end of the card -- let the real driver error out. */
721 }
722 board_sd_fill_block(&s_sd, off, dst);
723 return true;
724}

726bool board_sd_write_block(uint32_t lba, const uint8_t* src)
727{
728 if (!s_sd.attached || (s_sd.image == nullptr) || (src == nullptr)) {
729 return false;
730 }
731 /* Same SDHC block-addressing as the CMD24 path: arg is a block index. */
732 const uint64_t off = (uint64_t)lba * (uint64_t)k_sd_block;
733 if (off >= s_sd.image_len) {
734 return false; /* past the end of the card -- let the real driver error out. */
735 }
736 if ((off + (uint64_t)k_sd_block) > s_sd.image_len) {
737 return false; /* a full block would run off the end of the image. */
738 }
739 for (uint32_t i = 0U; i < (uint32_t)k_sd_block; ++i) {
740 s_sd.image[off + (uint64_t)i] = src[i];
741 }
742 return true;
743}
