3"""End-to-end checks over the dispatch layer, without a client.
4
5Drives real JSON-RPC requests through :func:`mcp_protocol.dispatch` and
6asserts both directions: every advertised capability answers, and malformed or
7unknown requests produce the right JSON-RPC error rather than an exception or
8a silent success.
9
10The schema-vs-handler cross-check is the one that earns its keep. A tool whose
11advertised schema and handler disagree still starts, still lists, and fails
12only when a client happens to call it -- so the two are compared here.
13"""
14

30def _req(method: str, **params: object) -> dict[str, object]:
31 """One JSON-RPC request. The id is filled in by the runner.
32
33 Values are typed ``object`` rather than ``Any``: they are stored and later
34 JSON-serialised, never inspected here, so ``object`` states the real
35 contract while ``Any`` would merely switch checking off.
36 """
37 out: dict[str, Any] = {"jsonrpc": "2.0", "method": method}
38 if params:
39 out["params"] = params
40 return out

43def _call(name: str, **arguments: object) -> dict[str, object]:
44 """A tools/call request for ``name``."""
45 return _req("tools/call", name=name, arguments=arguments)

48def _text(reply: dict[str, Any]) -> str:
49 """The first text block of a tools/call result."""
50 return reply["result"]["content"][0]["text"]

53#: (label, request, predicate over the reply). Every advertised capability is
54#: exercised, and the two error paths -- an unknown method and an unknown app
55#: -- are asserted to FAIL rather than to quietly succeed.
56_CASES: tuple[tuple[str, dict[str, Any], Callable[[dict[str, Any]], bool]], ...] = (
57 (
58 "initialize returns serverInfo",
59 _req("initialize", protocolVersion=PROTOCOL_VERSION_DEFAULT),
60 lambda r: bool(r) and r["result"]["serverInfo"]["name"] == SERVER_NAME,
61 ),
62 (
63 "tools/list advertises every tool",
64 _req("tools/list"),
65 lambda r: {t["name"] for t in r["result"]["tools"]} == set(TOOL_INDEX),
66 ),
67 (
68 "resources/list advertises every resource",
69 _req("resources/list"),
70 lambda r: {x["uri"] for x in r["result"]["resources"]} == set(RESOURCE_INDEX),
71 ),
72 (
73 "prompts/list advertises every prompt",
74 _req("prompts/list"),
75 lambda r: {p["name"] for p in r["result"]["prompts"]} == set(PROMPT_INDEX),
76 ),
77 (
78 "list_apps finds firmware apps",
79 _call("list_apps"),
80 lambda r: r["result"]["isError"] is False and "firmware app" in _text(r),
81 ),
82 (
83 "repo_overview mentions the target MCU",
84 _call("repo_overview"),
85 lambda r: "R7KA8D2KFLCAC" in _text(r),
86 ),
87 (
88 "flash_app withholds the write without confirm",
89 _call("flash_app", app="blink"),
90 lambda r: "[dry run]" in _text(r),
91 ),
92 (
93 "hum_citation emits a HUM skeleton",
94 _call("hum_citation", chapter="38.2.3"),
95 lambda r: "HUM Ch 38.2.3" in _text(r),
96 ),
97 (
98 "emu_app rejects an unknown app (no build)",
99 _call("emu_app", app="no_such_app"),
100 lambda r: r["result"]["isError"] is True,
101 ),
102 (
103 "resources/read returns CLAUDE.md",
104 _req("resources/read", uri="ra8d2://doc/claude-md"), # AI-OK: serves CLAUDE.md
105 lambda r: "CLAUDE" in r["result"]["contents"][0]["text"], # AI-OK: served doc is CLAUDE.md
106 ),
107 (
108 "prompts/get fills the template",
109 _req("prompts/get", name="audit_register_access", arguments={"code": "x"}),
110 lambda r: "HUM" in r["result"]["messages"][0]["content"]["text"],
111 ),
112 (
113 "unknown method -> method-not-found",
114 _req("no/such/method"),
115 lambda r: r["error"]["code"] == ERR_METHOD_NOT_FOUND,
116 ),
117)

120def _schema_matches_handlers() -> list[tuple[str, bool]]:
121 """Every advertised tool must have a handler, and vice versa.
122
123 A tool whose schema and handler disagree still starts and still lists; it
124 fails only when a client happens to call it. Comparing the two tables is
125 the only thing that catches it before a user does.
126 """
127 missing = sorted(n for n, t in TOOL_INDEX.items() if not callable(t.get("handler")))
128 label = "every advertised tool has a handler"
129 if missing:
130 label = f"{label} (missing: {', '.join(missing)})"
131 return [(label, not missing)]

134def selftest() -> int:
135 """Exercise the dispatcher in-process (no client, no hardware, no build)."""
136 checks: list[tuple[str, bool]] = []
137 for idx, (label, request, predicate) in enumerate(_CASES, start=1):
138 reply = dispatch({**request, "id": idx})
139 try:
140 passed = bool(predicate(reply))
141 except (KeyError, IndexError, TypeError):
142 passed = False
143 checks.append((label, passed))
144 checks.extend(_schema_matches_handlers())
145
146 ok = True
147 for label, passed in checks:
148 print(f" [{'PASS' if passed else 'FAIL'}] {label}")
149 ok = ok and passed
150 print(f"\nselftest: {'PASS' if ok else 'FAIL'} ({sum(p for _, p in checks)}/{len(checks)})")
151 return 0 if ok else 1
