168RA8_INTERNAL static void internal_lpm_dpsby2_add_offset(const ra8_rtc_datetime_t* now,
169 ra8_rtc_datetime_t* out)
170{
171 *out = *now;
172 const uint16_t s = (uint16_t)now->second + (uint16_t)k_lpm_dpsby2_alarm_offset_s;
173 out->second = (uint8_t)(s % (uint16_t)k_lpm_dpsby2_secs_per_min);
174 const uint16_t carm = (uint16_t)(s / (uint16_t)k_lpm_dpsby2_secs_per_min);
175 const uint16_t m = (uint16_t)now->minute + carm;
176 out->minute = (uint8_t)(m % (uint16_t)k_lpm_dpsby2_mins_per_hour);
177 const uint16_t carh = (uint16_t)(m / (uint16_t)k_lpm_dpsby2_mins_per_hour);
178 const uint16_t h = (uint16_t)now->hour + carh;
179 out->hour = (uint8_t)(h % (uint16_t)k_lpm_dpsby2_hours_per_day);
180}

210[[nodiscard]] RA8_INTERNAL static ra8_err_t
211internal_lpm_dpsby2_arm_wake(const ra8_rtc_datetime_t* now)
212{
213 ra8_rtc_datetime_t a = {};
214 internal_lpm_dpsby2_add_offset(now, &a);
215 ra8_err_t err = ra8_rtc_set_alarm(&a);
216 if (err != k_ra8_ok) {
217 return err;
218 }
219 err = ra8_rtc_set_irq_enable((uint8_t)k_ra8_rtc_irq_alarm);
220 if (err != k_ra8_ok) {
221 return err;
222 }
223 err = ra8_lpm_arm_dpsier(k_ra8_lpm_dpsier_idx_2, (uint8_t)k_ra8_lpm_dpsier2_drtcaie_mask);
224 if (err != k_ra8_ok) {
225 return err;
226 }
227 return ra8_lpm_arm_wupen0_bits((uint32_t)k_ra8_lpm_wupen0_rtcalm);
228}

230void main(void)
231{
232 internal_lpm_dpsby2_setup_or_halt();
233 ra8_isr_globals_enable();
234
235 /* Emit boot banner before the deep-standby entry so the HIL gate
236 * confirms the build + bring-up path even if the sub-clock crystal
237 * is silent (see SOSC caveat in the file header). */
238 (void)ra8_board_uart_console_write(s_lpm_dpsby2_boot_msg,
239 (size_t)(sizeof(s_lpm_dpsby2_boot_msg) - 1U));
240
241 while (1) {
242 ra8_rtc_datetime_t now = {};
243 if (ra8_rtc_get(&now) != k_ra8_ok) {
244 break;
245 }
246 if (internal_lpm_dpsby2_arm_wake(&now) != k_ra8_ok) {
247 break;
248 }
249 /* On hardware this never returns -- Deep Software Standby resets
250 * the CPU on wake, so control re-enters Reset_Handler. On the
251 * host (RA8_OFF_TARGET) WFI is a no-op so the loop simply
252 * iterates. */
253 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_deep_standby_2) != k_ra8_ok) {
254 break;
255 }
256 }
257 internal_lpm_dpsby2_panic_halt();
258}
