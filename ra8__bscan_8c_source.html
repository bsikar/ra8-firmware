109[[nodiscard]] ra8_err_t ra8_bscan_init(void)
110{
111 /* HUM Ch 50.2 "Register Descriptions", Table 50.3 p 3258 -- JTIDR
112 * is hardwired to 0x085D_A447 and the JTIR power-on value is 0xE
113 * (BYPASS path selected by the TAP controller on reset). */
114 s_bscan_state.initialized = true;
115 s_bscan_state.last_instruction = k_ra8_bscan_instr_bypass;
116 s_bscan_state.expected_idcode = k_ra8_bscan_jtidr_reset;
117 ra8_log_info(s_tag, "bscan init");
118 return k_ra8_ok;
119}

121[[nodiscard]] ra8_err_t ra8_bscan_deinit(void)
122{
123 /* HUM Ch 50.3 "Operation", p 3259 -- TAP returns to its reset
124 * state when the external fixture holds TMS=1 for >=5 TCK clocks.
125 * From a CPU-side bookkeeping viewpoint we just drop our cached
126 * fields back to the power-on equivalents. */
127 s_bscan_state.initialized = false;
128 s_bscan_state.last_instruction = k_ra8_bscan_instr_bypass;
129 s_bscan_state.expected_idcode = 0U;
130 return k_ra8_ok;

133[[nodiscard]] ra8_err_t ra8_bscan_get_idcode(uint32_t* out)
134{
135 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
136 if (!s_bscan_state.initialized) {

158[[nodiscard]] ra8_err_t ra8_bscan_clear_status(uint32_t mask)
159{
160 if (mask != k_ra8_bscan_clear_mask_none) {
161 return k_ra8_err_invalid_arg;
162 }
163 if (!s_bscan_state.initialized) {
164 return k_ra8_err_not_initialized;
165 }
166 /* HUM Ch 50.2.1 "JTIR : Instruction Register", p 3258 -- the TAP
167 * reset value 0xE selects the BYPASS path, so "cleared" maps to
168 * BYPASS in our bookkeeping. */
169 s_bscan_state.last_instruction = k_ra8_bscan_instr_bypass;
170 return k_ra8_ok;
171}

173[[nodiscard]] ra8_err_t ra8_bscan_set_instruction(ra8_bscan_instr_t instr)
174{
175 if (!s_bscan_state.initialized) {
176 return k_ra8_err_not_initialized;
177 }
178 if (!internal_is_known_instruction(instr)) {
179 return k_ra8_err_invalid_arg;
180 }
181 /* HUM Ch 50.2.1 "JTIR : Instruction Register", p 3258 -- valid
182 * opcodes are EXTEST / SAMPLE_PRELOAD / IDCODE / CLAMP / HIGHZ /
183 * BYPASS. All other 4-bit codes are reserved and rejected above. */
184 s_bscan_state.last_instruction = instr;
185 ra8_log_info_val(s_tag, "set instruction", (uint32_t)instr);
186 return k_ra8_ok;
