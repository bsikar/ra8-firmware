96static uint64_t ssie_read(uc_engine* uc, uint64_t addr, unsigned size)
97{
98 (void)uc;
99 (void)size;
100 const uint32_t ch = (uint32_t)((addr - (uint64_t)k_ssie_base) / (uint64_t)k_ssie_stride);
101 const uint64_t off = (addr - (uint64_t)k_ssie_base) % (uint64_t)k_ssie_stride;
102 if (ch >= (uint32_t)k_ssie_count) {
103 return 0U;
104 }
105 const ssie_state_t* s = &s_ssie[ch];
106 if (off == (uint64_t)k_ssie_off_ssisr) {
107 /* IIRQ asserts while the channel is idle (REN and TEN both clear), so the
108 * driver's "must be idle before enabling" gate in ra8_ssie_start passes. */
109 const bool running = (s->reg[k_ssie_off_ssicr / 4U] & (uint32_t)k_ssie_ren_ten) != 0U;
110 return running ? 0U : (uint64_t)k_ssie_iirq;
111 }
112 if (off == (uint64_t)k_ssie_off_ssifsr) {
113 return (uint64_t)k_ssie_tde; /* TX FIFO always has room; RX empty. */
114 }
115 if (off == (uint64_t)k_ssie_off_ssifrdr) {
116 return 0U; /* No RX audio source in the model. */
117 }
118 if ((off % 4UL) != 0UL) {
119 return 0U;
120 }
121 const uint32_t idx = (uint32_t)(off / 4UL);
122 return (idx < (uint32_t)k_ssie_regs) ? (uint64_t)s->reg[idx] : 0U;
123}

126static void ssie_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
127{
128 (void)uc;
129 (void)size;
130 const uint32_t ch = (uint32_t)((addr - (uint64_t)k_ssie_base) / (uint64_t)k_ssie_stride);
131 const uint64_t off = (addr - (uint64_t)k_ssie_base) % (uint64_t)k_ssie_stride;
132 if (ch >= (uint32_t)k_ssie_count) {
133 return;
134 }
135 ssie_state_t* s = &s_ssie[ch];
136 if (off == (uint64_t)k_ssie_off_ssiftdr) {
137 s->tx_samples++; /* Transmit FIFO drains immediately in the model. */
138 return;
139 }
140 if ((off % 4UL) != 0UL) {
141 return;
142 }
143 const uint32_t idx = (uint32_t)(off / 4UL);
144 if (idx < (uint32_t)k_ssie_regs) {
145 s->reg[idx] = (uint32_t)value;
146 }
147}

150static void ssie_report(void)
151{
152 for (uint32_t ch = 0U; ch < (uint32_t)k_ssie_count; ch++) {
153 const uint32_t ssicr = s_ssie[ch].reg[k_ssie_off_ssicr / 4U];
154 if ((ssicr == 0U) && (s_ssie[ch].tx_samples == 0U)) {
155 continue; /* Untouched channel: stay quiet. */
156 }
157 /* Console I2S tab: per-channel TX-sample tally at end-of-run. I2S streams a
158 * sample per SSIFTDR write (far too fast to log live), so the bounded
159 * transfer summary is the whole-run tally -- the same shape spi_report uses. */
160 char ln[k_ssie_console_line_cap];
161 (void)snprintf(ln,
162 sizeof(ln),
163 "I2S ch%u tx_samples=%u",
164 (unsigned)ch,
165 (unsigned)s_ssie[ch].tx_samples);
166 board_console_push(k_board_console_ch_i2s, ln);
167 (void)fprintf(stderr,
168 " SSIE%u : TEN=%u REN=%u tx_samples=%u\n",
169 ch,
170 (unsigned)((ssicr >> 1) & 1U),
171 (unsigned)(ssicr & 1U),
172 s_ssie[ch].tx_samples);
173 }
174}
