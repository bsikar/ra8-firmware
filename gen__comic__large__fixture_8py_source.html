4"""Generate the baked oversized single-page CBZ fixture for ereader_comic (#344).
5
6Emits a pure 7-bit-ASCII C header (comic_large_fixture.h) holding one CBZ with a
7single deliberately-large grayscale PNG page. Decoded at full resolution the page
8far exceeds the app's whole-decode arena, so the reader cannot open it that way;
9it exercises the tile path (ra8_comic_tiles -> JOF atlas -> ra8_tile_cache)
10instead. The art is flat-shaded horizontal bands plus a frame and a left gutter,
11so it DEFLATEs to a few hundred bytes and the baked array stays tiny on disk
12while the decoded image is multiple megabytes.
13
14The output is deterministic (fixed art, fixed zlib level), so the tile digest the
15app prints is identical on host, ra8_emulator, and silicon.
16
17Usage:
18 python3 scripts/gen/gen_comic_large_fixture.py \
19 examples/ek_ra8d2/hw_pending/ereader_comic/comic_large_fixture.h
20"""

29# Oversized on purpose: 1024x1400 decodes to ~4.3 MiB RGB (or ~5.7 MiB at the
30# reader's 4 bpp footprint estimate), well past the 2 MiB whole-decode arena.
31PAGE_W = 1024
32PAGE_H = 1400
33
34PNG_BIT_DEPTH_8 = 8
35PNG_COLOR_GRAY = 0
36ZIP_LOCAL_MAGIC = 0x04034B50
37ZIP_CENTRAL_MAGIC = 0x02014B50

53def gray(x: int, y: int) -> int:
54 """Deterministic, DEFLATE-friendly grayscale sample for one pixel."""
55 if x < BORDER or x >= PAGE_W - BORDER or y < BORDER or y >= PAGE_H - BORDER:
56 return 0 # black frame
57 if x < GUTTER:
58 return GUTTER_GRAY # dark left gutter
59 band = (y * BAND_COUNT) // PAGE_H
60 return min(255, BAND_BASE + BAND_STEP * band)

63def render_page() -> bytearray:
64 """Render the whole page into a flat grayscale buffer."""
65 out = bytearray(PAGE_W * PAGE_H)
66 for y in range(PAGE_H):
67 row = y * PAGE_W
68 for x in range(PAGE_W):
69 out[row + x] = gray(x, y)
70 return out

73def png_chunk(tag: bytes, data: bytes) -> bytes:
74 """Wrap a payload in one PNG chunk: length, type, data, CRC-32."""
75 out = struct.pack(">I", len(data)) + tag + data
76 crc = zlib.crc32(tag + data) & 0xFFFFFFFF
77 return out + struct.pack(">I", crc)

80def encode_png(g: bytes | bytearray) -> bytes:
81 """8-bit grayscale PNG (color type 0), one filter-0 byte per scanline."""
82 sig = b"\x89PNG\r\n\x1a\n"
83 ihdr = struct.pack(">IIBBBBB", PAGE_W, PAGE_H, PNG_BIT_DEPTH_8, PNG_COLOR_GRAY, 0, 0, 0)
84 raw = bytearray()
85 for y in range(PAGE_H):
86 raw.append(0)
87 raw += g[y * PAGE_W : (y + 1) * PAGE_W]
88 idat = zlib.compress(bytes(raw), ZLIB_BEST)
89 return sig + png_chunk(b"IHDR", ihdr) + png_chunk(b"IDAT", idat) + png_chunk(b"IEND", b"")

92def build_cbz(png: bytes) -> bytes:
93 """Minimal deterministic single-member ZIP (DEFLATE) that miniz reads."""
94 name = b"01_large.png"
95 comp = zlib.compressobj(ZLIB_BEST, zlib.DEFLATED, DEFLATE_RAW_WBITS)
96 body = comp.compress(png) + comp.flush()
97 crc = zlib.crc32(png) & 0xFFFFFFFF
98 local = (
99 struct.pack(
100 "<IHHHHHIIIHH",
101 ZIP_LOCAL_MAGIC,
102 ZIP_VERSION,
103 0,
104 ZIP_METHOD_DEFLATE,
105 0,
106 0,
107 crc,
108 len(body),
109 len(png),
110 len(name),
111 0,
112 )
113 + name
114 + body
115 )
116 central = (
117 struct.pack(
118 "<IHHHHHHIIIHHHHHII",
119 ZIP_CENTRAL_MAGIC,
120 ZIP_VERSION,
121 ZIP_VERSION,
122 0,
123 ZIP_METHOD_DEFLATE,
124 0,
125 0,
126 crc,
127 len(body),
128 len(png),
129 len(name),
130 0,
131 0,
132 0,
133 0,
134 0,
135 0,
136 )
137 + name
138 )
139 eocd = struct.pack(
140 "<IHHHHIIH",
141 ZIP_EOCD_MAGIC,
142 0,
143 0,
144 1,
145 1,
146 len(central),
147 len(local),
148 0,
149 )

153def emit_header(path: str | Path, cbz: bytes) -> None:
154 """Write the generated pure-ASCII C header holding the CBZ byte array."""
155 header = (
156 "/**\n"
157 " * @file comic_large_fixture.h\n"
158 " * @brief Baked oversized single-page CBZ for the ereader_comic tile self-check\n"
159 " * (#344; generated by scripts/gen/gen_comic_large_fixture.py -- do not edit).\n"
160 " *\n"
161 f" * @details A miniz-decodable ZIP of one {PAGE_W}x{PAGE_H} grayscale PNG page whose\n"
162 " * decoded size far exceeds the reader's whole-decode arena, so it can only\n"
163 " * be opened through the JOF tile path (ra8_comic_tiles). Flat-shaded art\n"
164 " * keeps the baked array tiny while the decoded image is multiple MiB.\n"
165 " *\n"
166 " * @copyright Copyright (c) 2026 Brighton Sikarskie\n"
167 " * SPDX-License-Identifier: MIT\n"
168 " */\n"
169 "#pragma once\n\n"
170 "#include <stdint.h>\n\n"
171 f"/** @brief Baked {len(cbz)}-byte single-page oversized CBZ. */\n"
172 "static const uint8_t k_comic_large_cbz[] = {\n"
173 )
174 rows = []
175 for start in range(0, len(cbz), BYTES_PER_ROW):
176 chunk = cbz[start : start + BYTES_PER_ROW]
177 rows.append(" " + " ".join(f"0x{byte:02X}," for byte in chunk))
178 footer = (
179 "\n};\n\n"
180 "/** @brief Length of ::k_comic_large_cbz in bytes. */\n"
181 f"static const uint32_t k_comic_large_cbz_len = {len(cbz)}U;\n\n"
182 "/** @brief Decoded width of the baked oversized page, pixels. */\n"
183 f"static const uint32_t k_comic_large_w = {PAGE_W}U;\n\n"
184 "/** @brief Decoded height of the baked oversized page, pixels. */\n"
185 f"static const uint32_t k_comic_large_h = {PAGE_H}U;\n"
186 )
187 Path(path).write_text(header + "\n".join(rows) + footer, encoding="ascii")
