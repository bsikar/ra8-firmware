193struct v2_state {
194 /* Field order matters for destruction: the document holds a raw
195 * pointer to the container (passed via container.get() into
196 * createFromString), so the document must be destroyed first.
197 * In C++ struct destruction runs in reverse declaration order,
198 * so list the container last. */
199 litehtml::document::ptr document;
200 std::unique_ptr<v2_container> container;
201};
202
203/* Explicit reset helper: clear the document before the container so
204 * raw-pointer back-references inside the document are not dangling
205 * during its destructor. Default-assignment of the whole struct does
206 * NOT guarantee this order on all libstdc++ versions, so we do it by
207 * hand. */
208inline void v2_state_clear(v2_state& s)
209{
210 s.document.reset();
211 s.container.reset();
212}

216struct v2_cache_slot {
217 const ra8_reflow_t* engine = nullptr;
218 v2_state state;
219};

221v2_cache_slot s_v2_cache[k_v2_max_engines];
222
223v2_state* v2_state_for(const ra8_reflow_t* engine, bool create)
224{
225 for (uint8_t i = 0; i < (uint8_t)k_v2_max_engines; ++i) {
226 if (s_v2_cache[i].engine == engine) {
227 return &s_v2_cache[i].state;
228 }
229 }
230 if (!create) {
231 return nullptr;
232 }
233 for (uint8_t i = 0; i < (uint8_t)k_v2_max_engines; ++i) {
234 if (s_v2_cache[i].engine == nullptr) {
235 s_v2_cache[i].engine = engine;
236 v2_state_clear(s_v2_cache[i].state);
237 return &s_v2_cache[i].state;
238 }
239 }
240 return nullptr;
241}
242
243void v2_state_drop(const ra8_reflow_t* engine)
244{
245 for (uint8_t i = 0; i < (uint8_t)k_v2_max_engines; ++i) {
246 if (s_v2_cache[i].engine == engine) {
247 v2_state_clear(s_v2_cache[i].state);
248 s_v2_cache[i].engine = nullptr;
249 return;
250 }
251 }

265/* Build the litehtml document for the engine's current xhtml buffer. Resets in
266 * document-first order before allocating new ones so a stale document holding a
267 * raw container pointer is destroyed before its target. */
268ra8_err_t v2_build_document(v2_state* state, const ra8_reflow_t* engine)
269{
270 state->document.reset();
271 state->container.reset();
272
273 state->container = std::make_unique<v2_container>(static_cast<int>(engine->viewport_w),
274 static_cast<int>(engine->viewport_h),
275 static_cast<int>(engine->font_px));
276
277 const std::string body(reinterpret_cast<const char*>(engine->xhtml_buf), engine->xhtml_len);
278 std::string html;
279 html.reserve(body.size() + k_v2_html_wrap_reserve);
280 html.append("<!DOCTYPE html><html><head></head><body>");
281 html.append(body);
282 html.append("</body></html>");
283
284 const litehtml::estring src(html);
285 state->document = litehtml::document::createFromString(src, state->container.get());
286 return state->document ? k_ra8_ok : k_ra8_err_validation_failed;
287}

289/* Split the rendered content height into the engine's page array, clamped to
290 * the reflow page-count bound. */
291void v2_paginate(ra8_reflow_t* engine, int content_height)
292{
293 const int vh = static_cast<int>(engine->viewport_h);
294 int pages = (content_height + vh - 1) / vh;
295 if (pages < 1) {
296 pages = 1;
297 }
298 if (pages > static_cast<int>(k_ra8_reflow_max_pages)) {
299 pages = static_cast<int>(k_ra8_reflow_max_pages);
300 }
301 engine->page_count = static_cast<uint32_t>(pages);
302 for (int i = 0; i < pages; ++i) {
303 engine->pages[i].glyph_first = 0U;
304 engine->pages[i].glyph_count = 0U;
305 }
306}

308ra8_err_t v2_run_layout(ra8_reflow_t* engine, uint32_t* out_total_pages)
309{
310 v2_state* state = v2_state_for(engine, true);
311 if (state == nullptr) {
312 return k_ra8_err_no_mem;
313 }
314 const ra8_err_t err = v2_build_document(state, engine);
315 if (err != k_ra8_ok) {
316 return err;
317 }
318
319 (void)state->document->render(static_cast<litehtml::pixel_t>(engine->viewport_w));
320 int content_height = static_cast<int>(state->document->height());
321 if (content_height < 0) {
322 content_height = 0;
323 }
324
325 v2_paginate(engine, content_height);
326 if (out_total_pages != nullptr) {
327 *out_total_pages = engine->page_count;
328 }
329 return k_ra8_ok;
330}

336[[nodiscard]] ra8_err_t ra8_reflow_init(uint16_t viewport_w,
337 uint16_t viewport_h,
338 const uint8_t* font_data,
339 size_t font_len,
340 uint16_t font_px,
341 uint32_t body_color,
342 uint32_t link_color,
343 ra8_reflow_t* out_engine)
344{
345 if ((font_data == nullptr) || (out_engine == nullptr)) {
346 return k_ra8_err_null_ptr;
347 }
348 if ((viewport_w == 0U) || (viewport_h == 0U) || (font_px < (uint16_t)k_ra8_reflow_min_font_px) ||
349 (font_px > (uint16_t)k_ra8_reflow_max_font_px)) {
350 (void)std::memset(out_engine, 0, sizeof(*out_engine));
351 return k_ra8_err_invalid_arg;
352 }
353 if (font_len < 16U) {
354 (void)std::memset(out_engine, 0, sizeof(*out_engine));
355 return k_ra8_err_invalid_size;
356 }
357 (void)std::memset(out_engine, 0, sizeof(*out_engine));
358 out_engine->viewport_w = viewport_w;
359 out_engine->viewport_h = viewport_h;
360 out_engine->font_px = font_px;
361 out_engine->body_color = body_color;
362 out_engine->link_color = link_color;
363 out_engine->font_data = font_data;
364 out_engine->font_len = font_len;
365 out_engine->in_use = 1U;
366 return k_ra8_ok;
367}
