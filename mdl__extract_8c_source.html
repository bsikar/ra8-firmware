73read_eq_value(const char* tag, size_t tag_len, size_t k, char* out, size_t out_cap)
74{
75 k = skip_ws(tag, tag_len, k);
76 if (!((k < tag_len) && (tag[k] == '='))) {
77 return false;
78 }
79 k = skip_ws(tag, tag_len, k + 1U);
80 if (!((k < tag_len) && ((tag[k] == '"') || (tag[k] == '\'')))) {
81 return false;

83 const char quote = tag[k];
84 ++k;

167RA8_INTERNAL static bool
168resolve_root_rel(const char* auth, const char* raw, char* out, size_t out_cap)
169{
170 /* The snprintf return is the truncation test: n >= out_cap is exactly the
171 * over-long case a separate strlen guard would reject, and using the return
172 * (rather than discarding it) is also what keeps -Wformat-truncation quiet. */
173 const int n = snprintf(out, out_cap, "%s%s", auth, raw);
174 return (n >= 0) && ((size_t)n < out_cap);

178RA8_INTERNAL static bool
179resolve_path_rel(const char* base, const char* auth, const char* raw, char* out, size_t out_cap)
180{
181 const char* q = strpbrk(base, "?#");
182 const size_t blen = (q == nullptr) ? strlen(base) : (size_t)(q - base);
183 size_t slash = 0U;
184 for (size_t i = 0U; i < blen; ++i) {
185 if (base[i] == '/') {
186 slash = i;
187 }
188 }
189 if (slash < strlen(auth)) {
190 /* As in resolve_root_rel: the snprintf return is the truncation test, and
191 * using it keeps -Wformat-truncation quiet where a discarded return does not. */
192 const int n = snprintf(out, out_cap, "%s/%s", auth, raw);
193 return (n >= 0) && ((size_t)n < out_cap);
194 }
195 if ((slash + 1U + strlen(raw) + 1U) > out_cap) {
196 return false;
197 }
198 memcpy(out, base, slash + 1U);
199 memcpy(out + slash + 1U, raw, strlen(raw) + 1U);
200 return true;
201}

204RA8_INTERNAL static bool resolve_url(const char* base, const char* raw, char* out, size_t out_cap)
205{
206 while ((*raw == ' ') || (*raw == '\t') || (*raw == '\n') || (*raw == '\r')) {
207 ++raw;
208 }
209 if ((raw[0] == '\0') || (raw[0] == '#') || (strncmp(raw, "data:", sizeof("data:") - 1U) == 0)) {
210 return false;
211 }
212 if ((strncmp(raw, "http://", sizeof("http://") - 1U) == 0) ||
213 (strncmp(raw, "https://", sizeof("https://") - 1U) == 0)) {
214 return copy_fits(out, out_cap, raw);
215 }
216
217 char auth[k_authority_max];
218 if (!authority_of(base, auth, sizeof(auth))) {
219 return false;
220 }
221 if ((raw[0] == '/') && (raw[1] == '/')) {
222 return resolve_scheme_rel(base, raw, out, out_cap);
223 }
224 if (raw[0] == '/') {
225 return resolve_root_rel(auth, raw, out, out_cap);
226 }
227 return resolve_path_rel(base, auth, raw, out, out_cap);
228}

362RA8_INTERNAL static bool
363decode_entity(const char* s, size_t len, size_t* i, char* out_ch, bool* space)
364{
365 static const mdl_entity_t table[] = {
366 {"amp;", '&'},
367 {"lt;", '<'},
368 {"gt;", '>'},
369 {"quot;", '"'},
370 {"#39;", '\''},
371 {"apos;", '\''},
372 {"nbsp;", '\0'},
373 };
374 const size_t start = *i + 1U; /* byte after '&' */
375 for (size_t e = 0U; e < (sizeof(table) / sizeof(table[0])); ++e) {
376 const size_t nlen = strlen(table[e].name);
377 if (((start + nlen) <= len) && (memcmp(s + start, table[e].name, nlen) == 0)) {
378 *i = start + nlen - 1U; /* leave *i on the ';'; the loop's ++ steps past */
379 *space = (table[e].ch == '\0');
380 *out_ch = table[e].ch;
381 return true;
382 }
383 }
384 return false;
385}

392RA8_INTERNAL static void clean_inner_text(const char* text, size_t len, char* out, size_t cap)
393{
394 size_t n = 0U;
395 bool pending = false; /* a space is owed before the next visible char */
396 for (size_t i = 0U; (i < len) && ((n + 1U) < cap); ++i) {
397 const char c = text[i];
398 if (c == '<') {
399 while ((i < len) && (text[i] != '>')) {
400 ++i; /* skip a nested tag such as <span> */
401 }
402 continue;
403 }
404 char emit = c;
405 bool space = is_ws(c);
406 if (c == '&') {
407 char decoded = '\0';
408 if (decode_entity(text, len, &i, &decoded, &space)) {
409 emit = decoded;
410 }
411 }
412 if (space) {
413 pending = (n > 0U); /* collapse; never a leading space */
414 continue;
415 }
416 if (pending && ((n + 1U) < cap)) {
417 out[n] = ' ';
418 ++n;
419 }
420 pending = false;
421 if ((n + 1U) < cap) {
422 out[n] = emit;
423 ++n;
424 }
425 }
426 out[n] = '\0';
427}

430RA8_INTERNAL static void url_slug(const char* url, char* out, size_t cap)
431{
432 const char* q = strpbrk(url, "?#");
433 const size_t end = (q == nullptr) ? strlen(url) : (size_t)(q - url);
434 size_t beg = 0U;
435 size_t fin = end;
436 while ((fin > 0U) && (url[fin - 1U] == '/')) {
437 --fin; /* drop trailing slashes */
438 }
439 beg = fin;
440 while ((beg > 0U) && (url[beg - 1U] != '/')) {
441 --beg; /* walk back to the segment start */
442 }
443 size_t n = 0U;
444 for (size_t i = beg; (i < fin) && ((n + 1U) < cap); ++i) {
445 out[n] = url[i];
446 ++n;
447 }
448 out[n] = '\0';
449}

512RA8_INTERNAL static ra8_err_t emit_hit(const char* html,
513 size_t html_len,
514 const char* base_url,
515 const char* keep,
516 const char* tag,
517 size_t tag_len,
518 size_t gt_off,
519 mdl_hit_list_t* out,
520 bool* real)
521{
522 char raw[k_mdl_url_max];
523 if (!find_attr_value(tag, tag_len, "href", raw, sizeof(raw))) {
524 return k_ra8_ok; /* an <a> with no href is not a link */
525 }
526 char abs[k_mdl_url_max];
527 if (!resolve_url(base_url, raw, abs, sizeof(abs))) {
528 return k_ra8_ok; /* unresolvable (e.g. javascript:, fragment): not counted */
529 }
530 out->anchors_seen++;
531 if (!contains_ok(abs, keep)) {
532 return k_ra8_ok; /* a real link, but not a result: counted, not stored */
533 }
534 char title[k_mdl_hit_title_max];
535 const bool title_real = anchor_title(html, html_len, tag, tag_len, gt_off, title, sizeof(title));
536 if (!title_real) {
537 url_slug(abs, title, sizeof(title));
538 }
539 return merge_hit(out, real, abs, title, title_real);
