40RA8_INTERNAL static bool internal_authority_of(const char* base, char* auth, size_t cap)
41{
42 const char* sep = strstr(base, "://");
43 if (sep == nullptr) {
44 return false;
45 }
46 const char* host = sep + (sizeof("://") - 1U);
47 const char* end = strchr(host, '/');
48 const size_t len = (end == nullptr) ? strlen(base) : (size_t)(end - base);
49 if (len + 1U > cap) {
50 return false;
51 }
52 memcpy(auth, base, len);
53 auth[len] = '\0';
54 return true;
55}

105RA8_INTERNAL static bool
106internal_resolve_root_rel(const char* auth, const char* raw, char* out, size_t out_cap)
107{
108 /* The snprintf return is the truncation test: n >= out_cap is exactly the
109 * over-long case a separate strlen guard would reject, and using the return
110 * (rather than discarding it) is also what keeps -Wformat-truncation quiet.
111 */
112 const int n = snprintf(out, out_cap, "%s%s", auth, raw);
113 return (n >= 0) && ((size_t)n < out_cap);

134RA8_INTERNAL static bool internal_resolve_path_rel(const char* base,
135 const char* auth,
136 const char* raw,
137 char* out,
138 size_t out_cap)
139{
140 const char* q = strpbrk(base, "?#");
141 const size_t blen = (q == nullptr) ? strlen(base) : (size_t)(q - base);
142 size_t slash = 0U;
143 for (size_t i = 0U; i < blen; ++i) {
144 if (base[i] == '/') {
145 slash = i;
146 }
147 }
148 if (slash < strlen(auth)) {
149 /* As in internal_resolve_root_rel: the snprintf return is the truncation test, and
150 * using it keeps -Wformat-truncation quiet where a discarded return does
151 * not. */
152 const int n = snprintf(out, out_cap, "%s/%s", auth, raw);
153 return (n >= 0) && ((size_t)n < out_cap);
154 }
155 if ((slash + 1U + strlen(raw) + 1U) > out_cap) {
156 return false;
157 }
158 memcpy(out, base, slash + 1U);
159 memcpy(out + slash + 1U, raw, strlen(raw) + 1U);
160 return true;
161}

164bool mdl_extract_resolve_url(const char* base, const char* raw, char* out, size_t out_cap)
165{
166 if ((base == nullptr) || (raw == nullptr) || (out == nullptr) || (out_cap == 0U)) {
167 return false;
168 }
169 while ((*raw == ' ') || (*raw == '\t') || (*raw == '\n') || (*raw == '\r')) {
170 ++raw;
171 }
172 if ((raw[0] == '\0') || (raw[0] == '#') || (strncmp(raw, "data:", sizeof("data:") - 1U) == 0)) {
173 return false;
174 }
175 if ((strncmp(raw, "http://", sizeof("http://") - 1U) == 0) ||
176 (strncmp(raw, "https://", sizeof("https://") - 1U) == 0)) {
177 return internal_copy_fits(out, out_cap, raw);
178 }
179
180 char auth[k_authority_max];
181 if (!internal_authority_of(base, auth, sizeof(auth))) {
182 return false;
183 }
184 if ((raw[0] == '/') && (raw[1] == '/')) {
185 return internal_resolve_scheme_rel(base, raw, out, out_cap);
186 }
187 if (raw[0] == '/') {
188 return internal_resolve_root_rel(auth, raw, out, out_cap);
189 }
190 return internal_resolve_path_rel(base, auth, raw, out, out_cap);
191}

240RA8_INTERNAL static ra8_err_t internal_emit_tag_url(const char* base_url,
241 const char* attr1,
242 const char* attr2,
243 const char* keep,
244 const char* tag,
245 size_t tag_len,
246 mdl_url_list_t* out)
247{
248 char raw[k_mdl_url_max];
249 bool got = internal_find_attr_value(tag, tag_len, attr1, raw, sizeof(raw));
250 if (!got && (attr2 != nullptr)) {
251 got = internal_find_attr_value(tag, tag_len, attr2, raw, sizeof(raw));
252 }
253 if (!got) {
254 return k_ra8_ok;
255 }
256 char abs[k_mdl_url_max];

262 return k_ra8_err_no_mem;
263 }
264 (void)internal_copy_fits(out->urls[out->count], k_mdl_url_max, abs);
265 out->count++;
266 return k_ra8_ok;
267}

292RA8_INTERNAL static ra8_err_t internal_scan_tags(const char* html,
293 size_t html_len,
294 const char* base_url,
295 const char* tagname,
296 const char* attr1,
297 const char* attr2,
298 const char* keep,
299 mdl_url_list_t* out)
300{
301 if ((html == nullptr) || (base_url == nullptr) || (attr1 == nullptr) || (out == nullptr)) {
302 return k_ra8_err_invalid_arg;
303 }
304 out->count = 0U;
305 const size_t name_len = strlen(tagname);
306
307 size_t pos = 0U;
308 while (pos < html_len) {
309 const char* tag = internal_find_ci(html + pos, html_len - pos, tagname);
310 if (tag == nullptr) {
311 break;
312 }
313 const size_t tag_off = (size_t)(tag - html);
314 const size_t after = tag_off + name_len;
315 if ((after < html_len) && !internal_is_name_end(html[after])) {
316 pos = tag_off + 1U; /* false match, e.g. "<article" while seeking "<a" */
317 continue;
318 }
319 const char* gt = memchr(tag, '>', html_len - tag_off);
320 const size_t tag_len = (gt == nullptr) ? (html_len - tag_off) : (size_t)(gt - tag);
321 if (internal_emit_tag_url(base_url, attr1, attr2, keep, tag, tag_len, out) ==
322 k_ra8_err_no_mem) {
323 return k_ra8_err_no_mem;
324 }
325 pos = tag_off + ((gt == nullptr) ? tag_len : (tag_len + 1U));
326 }
327 return k_ra8_ok;
328}
