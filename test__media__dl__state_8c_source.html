85static void test_hash_determinism(void)
86{
87 TEST_BEGIN("hash determinism");
88 /* V2/V3: an empty or NULL range returns the seed (here, the offset basis). */
89 TEST_ASSERT_EQ((uint64_t)k_mdl_fnv_offset, mdl_hash_bytes(nullptr, k_probe_bytes));
90 TEST_ASSERT_EQ((uint64_t)k_mdl_fnv_offset, mdl_hash_bytes("abc", 0U));
91 /* V1: equal inputs hash equally; a one-byte change diverges. */
92 const uint64_t h_abc = mdl_hash_str("abc");
93 TEST_ASSERT_EQ(h_abc, mdl_hash_str("abc"));
94 TEST_ASSERT(h_abc != mdl_hash_str("abd"));
95 /* A chunked fold reproduces the single-shot digest. */
96 uint64_t chained = (uint64_t)k_mdl_fnv_offset;
97 chained = mdl_hash_bytes_seed("ab", 2U, chained);
98 chained = mdl_hash_bytes_seed("c", 1U, chained);
99 TEST_ASSERT_EQ(mdl_hash_bytes("abc", 3U), chained);
100 TEST_END("hash determinism");
101}

114static void test_hash_file(void)
115{
116 TEST_BEGIN("hash file");
117 char path[k_tmp_path_bytes];
118 make_tmp(path, sizeof(path));
119 const char payload[] = "hello";
120 FILE* fp = fopen(path, "wb");
121 TEST_ASSERT_NOT_NULL(fp);
122 (void)fwrite(payload, 1U, strlen(payload), fp);
123 (void)fclose(fp);
124
125 uint64_t fh = 0U;
126 /* V1 control: the file hash equals the same bytes hashed in memory. */
127 TEST_ASSERT_EQ((int64_t)k_ra8_ok, mdl_hash_file(path, &fh));
128 TEST_ASSERT_EQ(mdl_hash_bytes(payload, strlen(payload)), fh);
129 /* V2/V3: NULL path / out are refused. */
130 TEST_ASSERT_EQ((int64_t)k_ra8_err_invalid_arg, mdl_hash_file(nullptr, &fh));
131 TEST_ASSERT_EQ((int64_t)k_ra8_err_invalid_arg, mdl_hash_file(path, nullptr));
132 (void)unlink(path);
133 /* A now-absent file fails cleanly. */
134 TEST_ASSERT_EQ((int64_t)k_ra8_fail, mdl_hash_file(path, &fh));
135 TEST_END("hash file");
136}

181static void test_state_chapters_and_pages(void)
182{
183 TEST_BEGIN("state chapters + pages");
184 mdl_state_init(&s_a);
185 TEST_ASSERT_EQ((uint16_t)0, s_a.chapter_count);
186
187 mdl_chapter_rec_t* c1 = mdl_state_add_chapter(&s_a, "c1", "http://s/c1", (long)k_ch_five);
188 TEST_ASSERT_NOT_NULL(c1);
189 TEST_ASSERT(mdl_state_find_chapter(&s_a, "c1") == c1);
190 /* A matched-incomplete chapter is not "complete"; pages unknown. */
191 TEST_ASSERT(!mdl_state_chapter_complete(&s_a, "c1"));
192 TEST_ASSERT_EQ((uint16_t)0, mdl_state_chapter_pages(&s_a, "c1"));
193 /* Adding the same id twice returns the SAME record (idempotent). */
194 TEST_ASSERT(mdl_state_add_chapter(&s_a, "c1", "http://s/c1", (long)k_ch_five) == c1);
195 TEST_ASSERT_EQ((uint16_t)1, s_a.chapter_count);
196
197 c1->complete = true;
198 c1->page_count = (uint16_t)k_pages_three;
199 TEST_ASSERT(mdl_state_chapter_complete(&s_a, "c1"));
200 TEST_ASSERT_EQ((uint16_t)k_pages_three, mdl_state_chapter_pages(&s_a, "c1"));
201 /* An absent id is neither complete nor found. */
202 TEST_ASSERT(!mdl_state_chapter_complete(&s_a, "nope"));
203 TEST_ASSERT_NULL(mdl_state_find_chapter(&s_a, "nope"));
204
205 TEST_ASSERT(mdl_state_add_page(&s_a, (uint64_t)k_uh_a, (uint64_t)k_ch_a, "c1/page_0001.jpg"));
206 const mdl_page_rec_t* p = mdl_state_find_page(&s_a, (uint64_t)k_uh_a);
207 TEST_ASSERT_NOT_NULL(p);
208 TEST_ASSERT_EQ((uint64_t)k_ch_a, p->content_hash);
209 TEST_ASSERT_NULL(mdl_state_find_page(&s_a, (uint64_t)k_uh_z));
210 TEST_END("state chapters + pages");
211}
