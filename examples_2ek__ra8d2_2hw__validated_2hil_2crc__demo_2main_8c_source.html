169static uint32_t crc_demo_sw_crc32(const uint8_t* data, uint32_t len)
170{
171 uint32_t crc = (uint32_t)k_crc_demo_sw_init;
172 for (uint32_t i = 0U; i < len; ++i) {
173 crc ^= (uint32_t)data[i];
174 for (uint8_t b = 0U; b < (uint8_t)k_crc_demo_bits_per_byte; ++b) {
175 if ((crc & (uint32_t)k_crc_demo_lsb_mask) != 0U) {
176 crc = (crc >> 1) ^ (uint32_t)k_crc_demo_sw_poly_rev;
177 } else {
178 crc = (crc >> 1);
179 }
180 }
181 }
182 return crc ^ (uint32_t)k_crc_demo_sw_xor_out;
183}

232[[nodiscard]] static ra8_err_t crc_demo_one_iter(uint8_t* out_match)
233{
234 uint32_t hw = 0U;
235 ra8_crc_reset();
236 if (ra8_crc_compute(k_crc_demo_payload, (uint32_t)k_crc_demo_payload_len, &hw) != k_ra8_ok) {
237 return k_ra8_err_hw_error;
238 }
239 const uint32_t sw = crc_demo_sw_crc32(k_crc_demo_payload, (uint32_t)k_crc_demo_payload_len);
240
241 uint8_t hwhex[k_crc_demo_hex_per_word] = {};
242 uint8_t swhex[k_crc_demo_hex_per_word] = {};
243 crc_demo_word_to_hex(hw, hwhex);
244 crc_demo_word_to_hex(sw, swhex);
245
246 (void)ra8_board_uart_console_write(k_crc_demo_prefix, (size_t)(sizeof(k_crc_demo_prefix) - 1U));
247 (void)ra8_board_uart_console_write(hwhex, (size_t)k_crc_demo_hex_per_word);
248 (void)ra8_board_uart_console_write(k_crc_demo_sw_tag, (size_t)(sizeof(k_crc_demo_sw_tag) - 1U));
249 (void)ra8_board_uart_console_write(swhex, (size_t)k_crc_demo_hex_per_word);
250 *out_match = (hw == sw) ? 1U : 0U;
251 if (*out_match != 0U) {
252 (void)ra8_board_uart_console_write(k_crc_demo_ok_tag, (size_t)(sizeof(k_crc_demo_ok_tag) - 1U));
253 (void)ra8_board_uart_console_write(k_crc_demo_pass_msg,
254 (size_t)(sizeof(k_crc_demo_pass_msg) - 1U));
255 return k_ra8_ok;
256 }
257 (void)ra8_board_uart_console_write(k_crc_demo_bad_tag, (size_t)(sizeof(k_crc_demo_bad_tag) - 1U));
258 return k_ra8_ok;
259}

266 ra8_isr_globals_enable();
267

274 if (match != 0U) {
275 (void)ra8_board_led_toggle(k_ra8_board_led1);
276 } else {
277 (void)ra8_board_led_toggle(k_ra8_board_led2);
278 }
279 ra8_delay_ms(k_crc_demo_period_ms);
280 }
281 crc_demo_panic_halt();
282 return 0;
283}
