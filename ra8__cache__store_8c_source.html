61static uint16_t cs_index_used(const ra8_cache_store_t* store)
62{
63 if (store == nullptr) {
64 return 0U;
65 }
66 if (store->index == nullptr) {
67 return 0U;
68 }
69 uint16_t used = 0U;
70 for (uint16_t i = 0U; i < store->index_cap; i++) {
71 if ((store->index[i].flags & (uint8_t)k_ra8_cache_store_flag_in_use) != 0U) {
72 used++;
73 }
74 }
75 return used;
76}

94static bool cs_run_free(const ra8_cache_store_t* store, uint32_t start, uint32_t count)
95{
96 if (store == nullptr) {
97 return false;
98 }
99 if (store->index == nullptr) {
100 return false;
101 }
102 uint32_t end = start + count;
103 for (uint16_t i = 0U; i < store->index_cap; i++) {
104 const ra8_cache_store_entry_t* e = &store->index[i];
105 if ((e->flags & (uint8_t)k_ra8_cache_store_flag_in_use) == 0U) {
106 continue;
107 }
108 uint32_t es = e->start_sector;
109 uint32_t ee = es + e->sector_count;
110 if (start < ee) {
111 if (es < end) {
112 return false;
113 }
114 }
115 }
116 return true;
117}

174static ra8_err_t cs_write_entry(ra8_cache_store_t* store,
175 uint32_t start,
176 uint32_t seq,
177 uint32_t key,
178 const uint8_t* data,
179 uint32_t len,
180 uint16_t count)
181{
182 RA8_CHECK_NULL_PTR(store, s_tag, "store");
183 RA8_CHECK_NULL_PTR(data, s_tag, "data");
184 uint16_t data_sectors = (uint16_t)(count - 1U);
185 uint32_t off = 0U;
186 const uint32_t sec_sz = (uint32_t)k_ra8_cache_store_sector_bytes;
187 for (uint16_t i = 0U; i < data_sectors; i++) {
188 uint32_t remain = len - off;
189 uint32_t chunk = (remain < sec_sz) ? remain : sec_sz;
190 (void)memset(store->staging, 0, (size_t)sec_sz);
191 (void)memcpy(store->staging, &data[off], (size_t)chunk);
192 RA8_RETURN_ON_ERROR(ra8_cs_sector_write(store, start + 1U + (uint32_t)i, store->staging),
193 s_tag,
194 "payload");
195 off += chunk;
196 }
197 ra8_cs_entry_hdr_t h = {.magic = (uint32_t)k_ra8_cs_entry_magic,
198 .seq = seq,
199 .key = key,
200 .byte_len = len,
201 .start_sector = start,
202 .sector_count = count,
203 .flags = 0U,
204 .hdr_crc = 0U};
205 h.hdr_crc = ra8_cs_crc32((const uint8_t*)&h, (uint32_t)(sizeof(h) - sizeof(h.hdr_crc)));
206 (void)memset(store->staging, 0, (size_t)sec_sz);
207 (void)memcpy(store->staging, &h, sizeof(h));
208 return ra8_cs_sector_write(store, start, store->staging);
209}

292static ra8_err_t cs_put_check(ra8_cache_store_t* store,
293 uint32_t key,
294 const uint8_t* data,
295 uint32_t len,
296 uint32_t* out_count)
297{
298 RA8_CHECK_NULL_PTR(store, s_tag, "store");
299 RA8_CHECK_NULL_PTR(data, s_tag, "data");
300 RA8_VALIDATE_INIT(store->inited, s_tag, "store");
301 if (len == 0U) {
302 return k_ra8_err_invalid_size;
303 }
304 if (ra8_cs_index_find(store, key) >= 0) {
305 return k_ra8_err_exists;
306 }
307 if (cs_index_used(store) >= store->index_cap) {
308 return k_ra8_err_no_mem;
309 }
310 const uint32_t sec_sz = (uint32_t)k_ra8_cache_store_sector_bytes;
311 uint32_t count = 1U + ((len + (sec_sz - 1U)) / sec_sz);
312 if (count > (uint32_t)k_ra8_cs_max_run) {
313 return k_ra8_err_invalid_size;
314 }
315 if ((store->live_sectors + count) > store->data_capacity) {
316 return k_ra8_err_no_mem;
317 }
318 *out_count = count;
319 return k_ra8_ok;
320}
