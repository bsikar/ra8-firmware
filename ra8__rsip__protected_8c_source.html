204ra8_err_t ra8_rsip_protected_aes_init(const uint8_t* wrapped_key,
205 ra8_rsip_aes_key_bits_t key_bits,
206 ra8_rsip_aes_mode_t mode,
207 const uint8_t* iv)
208{
209 /* cppcheck-suppress unknownMacro -- cppcheck runs without include paths and cannot expand RA8_STACK_BUDGET. */
210 RA8_STACK_BUDGET(k_unwrap_key_stack_bytes); /* unwrapped-key scratch; scrubbed on unwind */
211 RA8_CHECK_NULL_PTR(wrapped_key, s_tag, "p_aes_init: wrapped_key");
212
213 ra8_err_t rc = ra8_rsip_key_validate(wrapped_key, k_ra8_rsip_wrapped_type_aes);
214 if (rc != k_ra8_ok) {
215 return rc;
216 }
217
218 uint8_t raw_key[k_ra8_rsip_p_aes_max_bytes] = {};
219 const uint32_t n_bytes = (uint32_t)key_bits / (uint32_t)k_ra8_rsip_p_bits_per_byte;
220 if (n_bytes > (uint32_t)k_ra8_rsip_p_aes_max_bytes) {
221 return k_ra8_err_invalid_arg;
222 }
223 for (uint32_t i = 0U; i < n_bytes; ++i) {
224 raw_key[i] = wrapped_key[(uint32_t)k_ra8_rsip_p_off_payload + i];
225 }
226
227 ra8_rsip_key_handle_t handle = {};
228 rc = p_aes_install(raw_key, key_bits, &handle);
229 p_scrub(raw_key, n_bytes);
230 if (rc != k_ra8_ok) {
231 return rc;
232 }
233
234 s_p_aes_handle = handle;
235 s_p_aes_mode = mode;
236 s_p_aes_iv_set = (iv != nullptr);
237 if (s_p_aes_iv_set) {
238 for (uint32_t i = 0U; i < (uint32_t)k_ra8_rsip_p_iv_bytes; ++i) {
239 s_p_aes_iv[i] = iv[i];
240 }
241 } else {
242 p_scrub(s_p_aes_iv, (uint32_t)k_ra8_rsip_p_iv_bytes);
243 }
244 s_p_aes_active = true;
245 return k_ra8_ok;
246}

437RA8_INTERNAL
438static ra8_err_t internal_rsa_install_priv(const uint8_t* wrapped_priv,
439 ra8_rsip_rsa_size_t size,
440 uint32_t mod_bytes,
441 ra8_rsip_key_handle_t* out_handle)
442{
443 uint8_t modulus[k_ra8_rsip_wrapped_max_payload] = {};
444 for (uint32_t i = 0U; i < mod_bytes; ++i) {
445 modulus[i] = wrapped_priv[(uint32_t)k_ra8_rsip_p_off_payload + i];
446 }
447 /* Use the OEM install path for RSA private keys: the wrapped layout
448 * fed in here is already the engine-acceptable blob plus a 16-byte
449 * IV that the stub validates trivially. */
450 uint8_t install_iv[k_ra8_rsip_p_iv_bytes] = {};
451 const ra8_rsip_oem_cmd_t install_cmd = internal_rsa_install_cmd(size);
452 const ra8_err_t rc =
453 ra8_rsip_oem_install(install_cmd, install_iv, modulus, mod_bytes, out_handle);
454 p_scrub(modulus, mod_bytes);
455 return rc;
456}

466 RA8_STACK_BUDGET(k_rsa4096_priv_stack_bytes); /* RSA-4096 modulus scratch; scrubbed on unwind */
467 RA8_CHECK_NULL_PTR(wrapped_priv, s_tag, "p_rsa_decrypt: wrapped_priv");
468 RA8_CHECK_NULL_PTR(ciphertext, s_tag, "p_rsa_decrypt: ciphertext");

472 if (rc != k_ra8_ok) {
473 return rc;
474 }
475
476 uint32_t mod_bytes = 0U;
477 rc = internal_rsa_mod_bytes(size, &mod_bytes);
478 if (rc != k_ra8_ok) {
479 return rc;
480 }
481 if (plaintext_cap < mod_bytes) {
482 return k_ra8_err_invalid_arg;
483 }
484 if (ciphertext_len > mod_bytes) {
485 return k_ra8_err_invalid_arg;
486 }
487
488 ra8_rsip_key_handle_t handle = {};
489 rc = internal_rsa_install_priv(wrapped_priv, size, mod_bytes, &handle);
490 if (rc != k_ra8_ok) {
491 return rc;
492 }
493
494 /* RSA "private decrypt" maps onto ra8_rsip_rsa_sign in the stub --
495 * both drive the engine's modular-exponentiation path. */
496 return ra8_rsip_rsa_sign(&handle, size, ciphertext, ciphertext_len, plaintext_out);

550ra8_err_t ra8_rsip_protected_ecdsa_sign(const uint8_t* wrapped_priv,
551 ra8_rsip_curve_t curve,
552 const uint8_t* hash,
553 uint32_t hash_len,
554 uint8_t* sig_out)
555{
556 /* cppcheck-suppress unknownMacro -- cppcheck runs without include paths and cannot expand RA8_STACK_BUDGET. */
557 RA8_STACK_BUDGET(k_ecc_priv_stack_bytes); /* ECC private scalar scratch; scrubbed on unwind */
558 RA8_CHECK_NULL_PTR(wrapped_priv, s_tag, "p_ecdsa_sign: wrapped_priv");
559 RA8_CHECK_NULL_PTR(hash, s_tag, "p_ecdsa_sign: hash");
560 RA8_CHECK_NULL_PTR(sig_out, s_tag, "p_ecdsa_sign: sig_out");
561
562 ra8_err_t rc = ra8_rsip_key_validate(wrapped_priv, k_ra8_rsip_wrapped_type_ecc_priv);
563 if (rc != k_ra8_ok) {
564 return rc;
565 }
566
567 /* Build a minimal ECC private handle for the engine. The stub's
568 * ``ra8_rsip_ecdsa_sign`` routes through ``internal_asym_run`` which
569 * just needs a non-NULL handle whose ``alg`` carries an ECC opcode;
570 * the wrapped body holds the private scalar bytes copied from the
571 * payload. */
572 ra8_rsip_key_handle_t handle = {};
573 uint32_t priv_bytes = 0U;
574 rc = internal_ecc_priv_params(curve, &handle.alg, &priv_bytes);
575 if (rc != k_ra8_ok) {
576 return rc;
577 }
578 handle.body_words = priv_bytes / sizeof(uint32_t);
579
580 /* Copy the private scalar into the handle body word-aligned. */
581 for (uint32_t i = 0U; i < priv_bytes; ++i) {
582 ((uint8_t*)handle.body)[i] = wrapped_priv[(uint32_t)k_ra8_rsip_p_off_payload + i];
583 }
584
585 rc = ra8_rsip_ecdsa_sign(&handle, curve, hash, hash_len, sig_out);
586 p_scrub((uint8_t*)handle.body, priv_bytes);
587 return rc;
588}
