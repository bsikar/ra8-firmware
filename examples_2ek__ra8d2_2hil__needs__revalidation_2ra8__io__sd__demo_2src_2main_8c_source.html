213RA8_INTERNAL static void internal_sd_demo_setup_or_halt(uint32_t* out_pclka_hz)
214{
215 uint32_t cpuclk0_hz = 0U;
216 uint32_t pclka_hz = 0U;
217 if (ra8_cgc_init() != k_ra8_ok) {
218 internal_sd_demo_panic_halt();
219 }
220 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
221 internal_sd_demo_panic_halt();
222 }
223 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
224 internal_sd_demo_panic_halt();
225 }
226 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
227 internal_sd_demo_panic_halt();
228 }
229 if (ra8_board_uart_console_init((uint32_t)k_sd_demo_uart_baud) != k_ra8_ok) {
230 internal_sd_demo_panic_halt();
231 }
232 *out_pclka_hz = pclka_hz;
233}

256RA8_INTERNAL static void internal_sd_demo_init_card_or_halt(uint32_t pclka_hz)
257{
258 const ra8_sdmmc_spi_sci_pins_t pins = {
259 .sck = s_sd_demo_pin_sck,
260 .cipo = s_sd_demo_pin_cipo,
261 .copi = s_sd_demo_pin_copi,
262 .cs = s_sd_demo_pin_cs,
263 };
264 ra8_sdmmc_spi_transport_t transport = {};
265 if (ra8_sdmmc_spi_transport_sci((uint8_t)k_sd_demo_spi_channel, pclka_hz, &pins, &transport) !=
266 k_ra8_ok) {
267 SD_DEMO_PUTS(s_msg_init_fail);
268 internal_sd_demo_panic_halt();
269 }
270 if (ra8_sdmmc_spi_init(&transport) != k_ra8_ok) {
271 SD_DEMO_PUTS(s_msg_init_fail);
272 internal_sd_demo_panic_halt();
273 }
274 SD_DEMO_PUTS(s_msg_card_ok);
275}

304[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_sd_demo_roundtrip(void)

311}

334void main(void)
335{
336 uint32_t pclka_hz = 0U;
337 internal_sd_demo_setup_or_halt(&pclka_hz);
338 ra8_isr_globals_enable();
339 ra8_log_init();
340 SD_DEMO_PUTS(s_msg_boot);
341
342 internal_sd_demo_init_card_or_halt(pclka_hz);
343
344 const ra8_err_t r = internal_sd_demo_roundtrip();
345 if (r != k_ra8_ok) {
346 SD_DEMO_PUTS(s_msg_fail);
347 internal_sd_demo_panic_halt();
348 }
349 SD_DEMO_PUTS(s_msg_pass);
350 (void)ra8_board_uart_console_flush();
351
352 while (true) {
353 __asm__ volatile("wfi");
354 }
355}
