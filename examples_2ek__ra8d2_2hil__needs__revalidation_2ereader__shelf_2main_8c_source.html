95static uint32_t sh_fb_hash(void)
96{
97 uint32_t h = (uint32_t)k_sh_fnv_offset;
98 const uint8_t* p = (const uint8_t*)s_framebuffer;
99 const size_t bytes = sizeof s_framebuffer;
100 for (size_t i = 0U; i < bytes; ++i) {
101 h = (h ^ p[i]) * (uint32_t)k_sh_fnv_prime;
102 }
103 return h;
104}

363static bool sh_loupe_held(int32_t x, int32_t y, bool* out_lens_only)
364{
365 if (g_sh.loupe_active) {
366 int32_t cx = g_sh.loupe_cx;
367 int32_t cy = g_sh.loupe_cy;
368 if (sh_cover_loupe_map(x, y, &cx, &cy)) {
369 g_sh.loupe_cx = cx;
370 g_sh.loupe_cy = cy;
371 }
372 *out_lens_only = true; /* lens is a fixed window; only its contents pan */
373 return true;
374 }
375 if ((ra8_time_ms() - g_sh.touch_down_ms) < (uint32_t)k_sh_loupe_hold_ms) {
376 return false; /* not held long enough yet */
377 }
378 int32_t cx = 0;
379 int32_t cy = 0;
380 if (!sh_cover_loupe_map(g_sh.touch_down_x, g_sh.touch_down_y, &cx, &cy)) {
381 return false; /* press did not start over a magnifiable cover image */
382 }
383 g_sh.loupe_active = true;
384 g_sh.loupe_cx = cx;
385 g_sh.loupe_cy = cy;
386 return true;

402static bool sh_cover_gesture(bool down, int32_t x, int32_t y, bool was_down, bool* out_lens_only)
403{
404 *out_lens_only = false;
405 if (down && !was_down) {
406 g_sh.touch_down_ms = ra8_time_ms();
407 g_sh.touch_down_x = x;
408 g_sh.touch_down_y = y;
409 return false; /* defer: could still become a tap or a hold */
410 }
411 if (down) {
412 return sh_loupe_held(x, y, out_lens_only);
413 }
414 if (was_down && g_sh.loupe_active) {
415 g_sh.loupe_active = false;
416 return true; /* release closes the lens */
417 }
418 if (was_down) {
419 return sh_tap_cover(g_sh.touch_down_x,
420 g_sh.touch_down_y,
421 g_sh.touch_down_y < (int32_t)k_sh_bar_h);
422 }
423 return false; /* idle: no finger down this poll or last */
424}

427static bool sh_handle_button(bool is_sw2)
428{
429 switch (g_sh.screen) {
430 case k_sh_screen_shelf:
431 if (g_sh.book_count == 0U) {
432 return false;
433 }
434 if (is_sw2) {
435 return sh_select_book(g_sh.selected);
436 }
437 g_sh.selected = (uint16_t)((g_sh.selected + g_sh.book_count - 1U) % g_sh.book_count);
438 return true;
439 case k_sh_screen_cover:
440 if (is_sw2) {
441 sh_start_reading();
442 return true;
443 }
444 g_sh.screen = k_sh_screen_shelf;
445 return true;
446 case k_sh_screen_toc:
447 sh_toc_scroll(is_sw2 ? 1 : -1);
448 return true;
449 case k_sh_screen_comic:
450 /* SW2 = next page, SW1 = previous page (in the active reading order). */
451 return sh_comic_turn(is_sw2 ? 1 : -1);
452 case k_sh_screen_reader:
453 default:
454 return sh_reader_turn(is_sw2 ? 1 : -1);
455 }

618static bool
619sh_pump_input(uint8_t* prev_touch, ra8_board_sw_state_t* prev1, ra8_board_sw_state_t* prev2)
620{
621 bool changed = false;
622 bool acted = false;
623 bool lens_only = false;
624
625 ra8_touch_point_t pts[k_sh_poll_pts] = {};
626 uint8_t got = 0U;
627 if (ra8_touch_read(pts, (uint8_t)k_sh_poll_pts, &got) == k_ra8_ok) {
628 const bool down = (got > 0U);
629 const bool was_down = (*prev_touch > 0U);
630 /* The cover screen defers its tap so a press-hold can open the loupe; every
631 * other screen keeps the unchanged tap-on-press path (page-turn thirds). */
632 if (g_sh.screen == k_sh_screen_cover) {
633 const int32_t x = down ? (int32_t)pts[0].x : g_sh.touch_down_x;
634 const int32_t y = down ? (int32_t)pts[0].y : g_sh.touch_down_y;
635 changed = sh_cover_gesture(down, x, y, was_down, &lens_only);
636 acted = down || was_down;
637 } else if (down && !was_down) {
638 acted = true;
639 changed = sh_handle_tap((int32_t)pts[0].x, (int32_t)pts[0].y);
640 }
641 *prev_touch = got;
642 }
643
644 ra8_board_sw_state_t s1 = k_ra8_board_sw_released;
645 ra8_board_sw_state_t s2 = k_ra8_board_sw_released;
646 if ((ra8_board_sw_read(k_ra8_board_sw1, &s1) == k_ra8_ok) && (s1 == k_ra8_board_sw_pressed) &&
647 (*prev1 == k_ra8_board_sw_released)) {
648 acted = true;
649 changed = sh_handle_button(false) || changed;
650 }
651 if ((ra8_board_sw_read(k_ra8_board_sw2, &s2) == k_ra8_ok) && (s2 == k_ra8_board_sw_pressed) &&
652 (*prev2 == k_ra8_board_sw_released)) {
653 acted = true;
654 changed = sh_handle_button(true) || changed;
655 }
656 *prev1 = s1;
657 *prev2 = s2;
658
659 if (changed) {
660 if (lens_only) {
661 sh_present_loupe();
662 } else {
663 sh_present();
664 }
665 }
666 return acted;
667}

743 (void)ra8_board_sw_read(k_ra8_board_sw1, &sw1_boot);
744 uint8_t prev_touch = 0U;
745 ra8_board_sw_state_t prev1 = sw1_boot;
746 ra8_board_sw_state_t prev2 = k_ra8_board_sw_released;
747 bool demo = (sw1_boot == k_ra8_board_sw_pressed);
748 uint32_t demo_ticks = 0U;
749 uint32_t demo_step = 0U;
750 uint32_t idle_ref_ms = ra8_time_ms(); /* timestamp of last activity. */
751 bool backlit = true; /* backlight currently lit. */
752 sh_wdt_arm_or_halt();
753 while (1) {
754 (void)ra8_wdt_refresh_for(k_ra8_wdt0); /* heartbeat: loop is alive. */
755 const bool acted = sh_pump_input(&prev_touch, &prev1, &prev2);
756 if (acted) {
757 demo = false; /* a real touch / button takes over */
758 } else if (demo && (++demo_ticks >= (uint32_t)k_sh_demo_period)) {
759 demo_ticks = 0U;
760 sh_demo_step(demo_step++);
761 sh_present();
762 }
763 if (acted || demo) {
764 idle_ref_ms = ra8_time_ms();
765 if (!backlit) {
766 (void)ra8_board_backlight_set(true);
767 backlit = true;
768 }
769 } else if (backlit && ((ra8_time_ms() - idle_ref_ms) >= (uint32_t)k_sh_idle_dim_ms)) {
770 (void)ra8_board_backlight_set(false);
771 backlit = false;
772 }
773 (void)ra8_delay_ms((uint32_t)k_sh_poll_ms);
774 }
775}
