264RA8_PRIV ra8_err_t ra8_c6link_priv_bare_req(ra8_c6link_t* link, uint32_t req_id)
265{
266 if (link == nullptr) {
267 return k_ra8_err_null_ptr;
268 }
269
270 Rpc req;
271 rpc__init(&req);
272 req.msg_type = RPC_TYPE__Req;
273 req.msg_id = (RpcId)req_id;
274
275 ra8_c6link_bare_body_t body;
276 uint32_t resp_id = 0U;
277 if (!ra8_c6link_wifi_bare_body(&req, &body, req_id, &resp_id)) {
278 return k_ra8_err_not_supported;
279 }
280
281 ra8_c6link_take_ctx_t take = {.link = link, .out = nullptr, .rpc_id = req_id};
282 return ra8_c6link_priv_rpc_call(link, &req, resp_id, ra8_c6link_priv_take_resp, &take);
283}

389ra8_err_t ra8_c6link_wifi_start(ra8_c6link_t* link)
390{
391 if (link == nullptr) {
392 return k_ra8_err_null_ptr;
393 }
394 if (!ra8_c6link_is_open(link)) {
395 return k_ra8_err_not_initialized;
396 }
397
398 const ra8_err_t inited = ra8_c6link_wifi_do_init(link);
399 if (inited != k_ra8_ok) {
400 return inited;
401 }
402 const ra8_err_t moded = ra8_c6link_wifi_do_mode(link);
403 if (moded != k_ra8_ok) {
404 return moded;
405 }
406 return ra8_c6link_priv_bare_req(link, (uint32_t)RPC_ID__Req_WifiStart);
407}
