248static void selftest_route_usb_or_halt(void)
249{
250 /* HS port: device role. PD07 LOW so U18 does not back-feed VBUS. */
251 if (ra8_pfs_route_peripheral(k_selftest_pin_hs_vbus, k_ra8_psel_usb_hs, "selftest.hs_vbus") !=
252 k_ra8_ok) {
253 selftest_panic_halt();
254 }
255 if (ra8_gpio_output_init(k_selftest_pin_hs_role, k_ra8_level_low) != k_ra8_ok) {
256 selftest_panic_halt();
257 }
258 /* FS port: host role. P5_00 VBUSEN peripheral-routed sources J11. */
259 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_vbus, k_ra8_psel_usb_fs, "selftest.fs_vbus") !=
260 k_ra8_ok) {
261 selftest_panic_halt();
262 }
263 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_vbusen, k_ra8_psel_usb_fs, "selftest.fs_vbusen") !=
264 k_ra8_ok) {
265 selftest_panic_halt();
266 }
267 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_dp, k_ra8_psel_usb_fs, "selftest.fs_dp") !=
268 k_ra8_ok) {
269 selftest_panic_halt();
270 }
271 if (ra8_pfs_route_peripheral(k_selftest_pin_fs_dm, k_ra8_psel_usb_fs, "selftest.fs_dm") !=
272 k_ra8_ok) {
273 selftest_panic_halt();
274 }
275}

292static void selftest_setup_or_halt(void)
293{
294 uint32_t cpuclk0_hz = 0U;
295 if (ra8_cgc_init() != k_ra8_ok) {
296 selftest_panic_halt();
297 }
298 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
299 selftest_panic_halt();
300 }
301 if (ra8_cgc_usbhs_pll_enable() != k_ra8_ok) {
302 selftest_panic_halt();
303 }
304 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
305 selftest_panic_halt();
306 }
307 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
308 selftest_panic_halt();
309 }
310 if (ra8_board_uart_console_init((uint32_t)k_selftest_baud) != k_ra8_ok) {
311 selftest_panic_halt();
312 }
313 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
314 selftest_panic_halt();
315 }
316 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
317 selftest_panic_halt();
318 }
319 selftest_route_usb_or_halt();
320}
