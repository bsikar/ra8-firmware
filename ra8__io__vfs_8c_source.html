214{
215 uint32_t i = 0;

217 const char c = path[i];
218 if (c == '\0') {
219 return k_ra8_err_invalid_arg;

224 out_name[i] = c;

227 return k_ra8_err_invalid_arg;

230 *out_sub = &path[i + 1U];
231 return k_ra8_ok;

284ra8_err_t ra8_io_vfs_mount(const char* name, ra8_fs_mount_t* mount)
285{
286 RA8_CHECK_NULL_PTR(name, s_tag, "name must not be nullptr");
287 RA8_CHECK_NULL_PTR(mount, s_tag, "mount must not be nullptr");
288 if (!internal_name_ok(name)) {
289 return k_ra8_err_invalid_arg;
290 }
291 if (internal_find(name) != nullptr) {
292 return k_ra8_err_exists;
293 }
294 for (uint32_t i = 0; i < (uint32_t)k_ra8_io_vfs_max_mounts; ++i) {
295 if (s_table[i].in_use) {
296 continue;
297 }
298 internal_copy_name(s_table[i].name, name);
299 s_table[i].mount = mount;
300 s_table[i].in_use = true;
301 return k_ra8_ok;
302 }
303 return k_ra8_err_no_mem;
304}
