204static void enc_emit_u8(ra8_jpeg_enc_ctx_t* e, uint8_t b)
205{
206 if (e->pos >= e->cap) {
207 e->overflow = true;
208 return;
209 }
210 e->dst[e->pos] = b;
211 e->pos++;
212}

222RA8_PRIV void ra8_jpeg_sw_priv_enc_put_bits(ra8_jpeg_enc_ctx_t* e, uint32_t code, uint8_t n)
223{
224 e->bit_buf = (e->bit_buf << n) | (code & ((1U << n) - 1U));
225 e->bit_cnt = (uint8_t)(e->bit_cnt + n);
226 while (e->bit_cnt >= (uint8_t)k_ra8_jpeg_byte_shift) {
227 uint8_t b = (uint8_t)(e->bit_buf >> (e->bit_cnt - k_ra8_jpeg_byte_shift));
228 enc_emit_u8(e, b);
229 if (b == (uint8_t)k_ra8_jpeg_marker_byte) {
230 enc_emit_u8(e, 0U);
231 }
232 e->bit_cnt = (uint8_t)(e->bit_cnt - k_ra8_jpeg_byte_shift);
233 }
234}

237RA8_PRIV void ra8_jpeg_sw_priv_enc_flush_bits(ra8_jpeg_enc_ctx_t* e)
238{
239 if (e->bit_cnt > 0U) {
240 uint32_t pad = (1U << (k_ra8_jpeg_byte_shift - e->bit_cnt)) - 1U;
241 ra8_jpeg_sw_priv_enc_put_bits(e,
242 ((e->bit_buf << (k_ra8_jpeg_byte_shift - e->bit_cnt)) | pad) &
243 k_jpeg_byte_mask,
244 (uint8_t)(k_ra8_jpeg_byte_shift - e->bit_cnt));
245 }

276static void enc_build_codes(const uint8_t* bits,
277 const uint8_t* vals,
278 uint16_t* codes,
279 uint8_t* sizes,
280 uint16_t total)
281{
282 uint8_t huffsize[k_ra8_jpeg_huff_max + 1U];
283 uint16_t k = 0U;
284 for (uint8_t i = 0U; i < (uint8_t)k_ra8_jpeg_huff_lengths; i++) {
285 for (uint8_t j = 0U; j < bits[i]; j++) {
286 huffsize[k] = (uint8_t)(i + 1U);
287 k++;
288 }
289 }
290 huffsize[k] = 0U;
291
292 uint16_t huffcode[k_ra8_jpeg_huff_max + 1U];
293 uint16_t code = 0U;
294 uint8_t si = (k > 0U) ? huffsize[0] : 0U;
295 uint16_t kk = 0U;
296 while (huffsize[kk] != 0U) {
297 while (huffsize[kk] == si) {
298 huffcode[kk] = code;
299 code++;
300 kk++;
301 }
302 if (huffsize[kk] == 0U) {
303 break;
304 }
305 do {
306 code = (uint16_t)(code << 1U);
307 si++;
308 } while (huffsize[kk] != si);
309 }
310 for (uint16_t i = 0U; i < total; i++) {
311 codes[vals[i]] = huffcode[i];
312 sizes[vals[i]] = huffsize[i];
313 }
314}
