83static uint8_t lpm_demo_nibble_to_hex(uint8_t nibble)
84{
85 uint8_t n = (uint8_t)(nibble & (uint8_t)k_lpm_demo_nibble_mask);
86 if (n < (uint8_t)k_lpm_demo_alpha_thresh) {
87 return (uint8_t)('0' + n);
88 }
89 return (uint8_t)('a' + (n - (uint8_t)k_lpm_demo_alpha_thresh));
90}

102static void lpm_demo_word_to_hex(uint32_t v, uint8_t* dst)
103{
104 for (uint8_t i = 0U; i < (uint8_t)k_lpm_demo_hex_per_word; ++i) {
105 const uint8_t shift =
106 (uint8_t)(((uint8_t)k_lpm_demo_hex_per_word - 1U - i) * (uint8_t)k_lpm_demo_nibble_shift);
107 const uint8_t nibble = (uint8_t)((v >> shift) & (uint32_t)k_lpm_demo_nibble_mask);
108 dst[i] = lpm_demo_nibble_to_hex(nibble);
109 }
110}

113static void lpm_demo_setup_or_halt(void)
114{
115 uint32_t cpuclk0_hz = 0U;
116
117 if (ra8_cgc_init() != k_ra8_ok) {
118 lpm_demo_panic_halt();
119 }
120 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
121 lpm_demo_panic_halt();
122 }
123 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
124 lpm_demo_panic_halt();
125 }
126 if (ra8_board_uart_console_init((uint32_t)k_lpm_demo_baud) != k_ra8_ok) {
127 lpm_demo_panic_halt();
128 }
129 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
130 lpm_demo_panic_halt();
131 }
132
133 /* LPM block uses the cold-reset defaults: OPE=1, IOKEEP=0, no
134 * fast-return / SS2LP options. ra8_lpm_init takes care of the
135 * PRC1 unlock + SBYCR / DPSBYCR / SSCR1 programming for us. */
136 const ra8_lpm_config_t lpm_cfg = {
137 .io_port_keep = false,
138 .opa_bus_keep = true,
139 .sscr_fast_return = false,
140 .dcdc_softstart = k_ra8_lpm_dcssmode_128us,
141 .sscr_low_power = k_ra8_lpm_ss2lp_default,
142 };
143 if (ra8_lpm_init(&lpm_cfg) != k_ra8_ok) {
144 lpm_demo_panic_halt();
145 }

165[[nodiscard]] static ra8_err_t lpm_demo_one_wake(void)
166{
167 /* Sleep mode -- WFI returns on the next SysTick IRQ ~1 ms later. */
168 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_sleep) != k_ra8_ok) {
169 return k_ra8_err_hw_error;
170 }
171 /* Continue idling until ~100 ms have accumulated. */
172 ra8_delay_ms((uint32_t)k_lpm_demo_period_ms);
173 s_wake_count++;
174
175 uint8_t hex[k_lpm_demo_hex_per_word] = {};
176 lpm_demo_word_to_hex(s_wake_count, hex);
177
178 (void)ra8_board_uart_console_write(k_lpm_demo_prefix, (size_t)(sizeof(k_lpm_demo_prefix) - 1U));
179 (void)ra8_board_uart_console_write(hex, (size_t)k_lpm_demo_hex_per_word);
180 (void)ra8_board_uart_console_write(k_lpm_demo_eol, (size_t)(sizeof(k_lpm_demo_eol) - 1U));
181 return k_ra8_ok;
182}
