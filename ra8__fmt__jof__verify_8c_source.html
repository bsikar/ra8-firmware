99static ra8_err_t
100priv_write_ppm(const char* path, const uint8_t* px, uint16_t w, uint16_t h, uint8_t bpp)
101{
102 if ((bpp != (uint8_t)k_fmt_ver_gray) && (bpp != (uint8_t)k_fmt_ver_rgb) &&
103 (bpp != (uint8_t)k_fmt_ver_rgba)) {
104 return k_ra8_err_not_supported;
105 }
106 FILE* f = fopen(path, "wb");
107 if (f == nullptr) {
108 return k_ra8_fail;
109 }
110 (void)fprintf(f,
111 "P%c\n%u %u\n%u\n",
112 (bpp == (uint8_t)k_fmt_ver_gray) ? '5' : '6',
113 (unsigned)w,
114 (unsigned)h,
115 (unsigned)k_fmt_ppm_max);
116 const size_t npx = (size_t)w * (size_t)h;
117 bool ok = true;
118 if (bpp == (uint8_t)k_fmt_ver_rgba) {
119 /* Drop alpha one pixel at a time: PPM carries colour channels only. */
120 for (size_t i = 0U; (i < npx) && ok; ++i) {
121 ok = (fwrite(&px[i * (size_t)k_fmt_ver_rgba], 1U, (size_t)k_fmt_ver_rgb, f) ==
122 (size_t)k_fmt_ver_rgb);
123 }
124 } else {
125 const size_t n = npx * (size_t)bpp;
126 ok = (fwrite(px, 1U, n, f) == n);
127 }
128 const bool cl = (fclose(f) == 0);
129 if (!ok || !cl) {
130 (void)remove(path);
131 return k_ra8_fail;
132 }
133 return k_ra8_ok;
134}

156RA8_INTERNAL
157static size_t priv_diff_rasters(const uint8_t* a,
158 const uint8_t* b,
159 size_t len,
160 size_t stride,
161 uint8_t bpp,
162 FILE* out)
163{
164 size_t diffs = 0U;
165 size_t shown = 0U;
166 for (size_t i = 0U; i < len; ++i) { /* bounded by len */
167 if (a[i] == b[i]) {
168 continue;
169 }
170 diffs++;
171 if (shown < (size_t)k_fmt_ver_max_diffs) {
172 const size_t y = i / stride;
173 const size_t x = (i % stride) / (size_t)bpp;
174 (void)fprintf(out,
175 " DIFF at pixel (%zu, %zu): reference=0x%02X banded=0x%02X\n",
176 x,
177 y,
178 (unsigned)a[i],
179 (unsigned)b[i]);
180 shown++;
181 }
182 }
183 return diffs;
184}

251priv_encode_pair(const ra8_fmt_blob_t* src, const ra8_fmt_opts_t* opts, fmt_ver_pair_t* pair)
252{
253 uint16_t w = 0U;
254 uint16_t h = 0U;

257 (void)fprintf(opts->report, "verify: cannot read source dimensions (rc=%d)\n", (int)rc);
258 return rc;
259 }

261 if (rc != k_ra8_ok) {
262 (void)fprintf(opts->report, "verify: reference encode failed (rc=%d)\n", (int)rc);
263 return rc;
264 }

266 ? pair->rinfo.height
267 : (uint16_t)k_fmt_ver_band_h;
268 rc = priv_encode_roundtrip(src,
269 w,
270 h,
271 pair->rinfo.width,
272 band,
273 &pair->got,
274 &pair->gotn,
275 &pair->ginfo);
276 if (rc != k_ra8_ok) {
277 (void)fprintf(opts->report, "verify: banded encode failed (rc=%d)\n", (int)rc);
278 free(pair->ref);
279 pair->ref = nullptr;
280 return rc;
281 }
282 (void)fprintf(opts->report,
283 "verify: %ux%u bpp=%u | reference 1 tile | banded %u tiles of %u rows\n",
284 (unsigned)pair->rinfo.width,
285 (unsigned)pair->rinfo.height,
286 (unsigned)pair->rinfo.bpp,
287 (unsigned)pair->ginfo.tile_count,
288 (unsigned)pair->ginfo.tile_h);
289 return k_ra8_ok;

292ra8_err_t ra8_fmt_jof_verify(const ra8_fmt_blob_t* src, const ra8_fmt_opts_t* opts)
293{
294 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
295 RA8_CHECK_NULL_PTR(opts, s_tag, "opts must not be nullptr");
296 fmt_ver_pair_t pair = {};
297 const ra8_err_t rc = priv_encode_pair(src, opts, &pair);
298 if (rc != k_ra8_ok) {
299 return rc;
300 }
301 uint8_t* const ref = pair.ref;
302 uint8_t* const got = pair.got;
303 const size_t refn = pair.refn;
304 const size_t gotn = pair.gotn;
305 const ra8_jof_info_t rinfo = pair.rinfo;
306 const ra8_jof_info_t ginfo = pair.ginfo;
307 size_t diffs = (refn != gotn) ? refn : 0U;
308 if (refn == gotn) {
309 diffs = priv_diff_rasters(ref,
310 got,
311 refn,
312 (size_t)rinfo.width * (size_t)rinfo.bpp,
313 rinfo.bpp,
314 opts->report);
315 } else {
316 (void)fprintf(opts->report, " raster size mismatch: %zu vs %zu bytes\n", refn, gotn);
317 }
318 if (opts->out_path != nullptr) {
319 const ra8_err_t prc = priv_write_ppm(opts->out_path, got, ginfo.width, ginfo.height, ginfo.bpp);
320 (void)fprintf(opts->report,
321 " wrote reassembled raster to %s (%s)\n",
322 opts->out_path,
323 (prc == k_ra8_ok) ? "ok" : "FAILED");
324 }
325 free(ref);
326 free(got);
327 (void)fprintf(opts->report,
328 "verdict: %s (%zu differing bytes)\n",
329 (diffs == 0U) ? "ROUND-TRIP EXACT -- the produced file is correct" : "MISMATCH",
330 diffs);
331 return (diffs == 0U) ? k_ra8_ok : k_ra8_err_validation_failed;
332}
