54static size_t priv_fs_stream_read(void* ctx, uint64_t offset, void* buf, size_t len)
55{
56 ra8_epub_stream_fs_ctx_t* io = (ra8_epub_stream_fs_ctx_t*)ctx;
57 if (io == nullptr || io->file == nullptr || buf == nullptr) {
58 return 0U; /* GCOVR_EXCL_LINE -- open binds a live ctx+file; miniz never reads a NULL buf */
59 }
60 if (offset > (uint64_t)UINT32_MAX) {
61 return 0U; /* GCOVR_EXCL_LINE -- ra8_fs is 32-bit; a > 4 GiB single .epub is rejected before here */
62 }
63 if (ra8_fs_seek(io->file, (uint32_t)offset) != k_ra8_ok) {
64 return 0U; /* GCOVR_EXCL_LINE -- seek clamps to size and cannot fail on a live handle */
65 }
66 const uint32_t want =
67 (len > (size_t)UINT32_MAX)
68 ? UINT32_MAX
69 : (uint32_t)len; /* GCOVR_EXCL_BR_LINE -- len is a bounded miniz IO chunk, never > 4 GiB */
70 uint32_t got = 0U;
71 if (ra8_fs_read(io->file, (uint8_t*)buf, want, &got) != k_ra8_ok) {
72 return 0U; /* GCOVR_EXCL_LINE -- read cannot fail on a live read-mode handle */
73 }
74 return (size_t)got;
75}

77[[nodiscard]] ra8_err_t ra8_epub_open_streamed_fs(ra8_fs_mount_t* mount,
78 const char* path,
79 ra8_epub_stream_fs_ctx_t* io,
80 ra8_epub_book_t* out_book)
81{

85 io->file = nullptr;

93 /* Size the archive for miniz. ra8_fs_size cannot fail on the handle ra8_fs_open
94 * just returned (in_use set, both out-params non-NULL); on any future contract
95 * change `size` stays 0 and ra8_epub_open_streamed rejects it as invalid_arg,
96 * so no separate (untestable) error branch is needed here. */
97 uint32_t size = 0U;
98 (void)ra8_fs_size(file, &size);
99
100 io->file = file;
101 const ra8_epub_stream_media_t media = {
102 .read = priv_fs_stream_read,
103 .ctx = io,
104 .size = (uint64_t)size,
105 };
106 err = ra8_epub_open_streamed(&media, path, out_book);
107 if (err != k_ra8_ok) {
108 (void)ra8_fs_close(file);
109 io->file = nullptr;
110 return err;
111 }
112 return k_ra8_ok;
113}
