194[[nodiscard]] static ra8_err_t pdg_demo_sample(uint8_t* out_ok)
195{
196 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
197
198 uint8_t code = 0U;
199 const ra8_err_t derr =
200 ra8_pdg_get_delay((uint8_t)k_pdg_demo_channel, k_ra8_pdg_pin_a, k_ra8_pdg_edge_rising, &code);
201 if (derr != k_ra8_ok) {
202 return derr;
203 }
204 ra8_pdg_status_full_t st = {};
205 const ra8_err_t err = ra8_pdg_get_status_full(&st);
206 if (err != k_ra8_ok) {
207 return err;
208 }
209 /* GTDLYRnA is not read-exposed on RA8D2 silicon: a write stages the delay
210 * but the register reads back its 0x0000 reset value (confirmed by a debugger
211 * probe; FSP never reads it back either). ra8_emulator shadows it as plain R/W,
212 * which is why the read-back only appears to work there. So this is recorded
213 * for the emulator but NOT gated on. */
214 g_pdg_delay_readback = (uint32_t)code;
215 g_pdg_dll = (st.dll_enabled != 0U) ? 1U : 0U;
216
217 const uint8_t powered = (st.per_channel_powered[k_pdg_demo_channel] != 0U) ? 1U : 0U;
218 const uint8_t bypass_n = (st.per_channel_bypass_off[k_pdg_demo_channel] != 0U) ? 1U : 0U;
219 /* Validate the software-observable bring-up: DLL locked + channel 0 powered
220 * and un-bypassed. The delay's actual edge-shift effect needs a scope. */
221 *out_ok = (st.dll_enabled != 0U && powered != 0U && bypass_n != 0U) ? 1U : 0U;
222 return k_ra8_ok;
223}

226#pragma GCC diagnostic ignored "-Wmain"
227int32_t main(void)
228{
229 pdg_demo_setup_or_halt();
230 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
231 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No NVIC sources
232 * are armed by this demo. */
233 ra8_isr_globals_enable();
234
235 if (pdg_demo_configure() != k_ra8_ok) {
236 pdg_demo_panic_halt();
237 }
238
239 while (1) {
240 uint8_t ok = 0U;
241 const ra8_err_t err = pdg_demo_sample(&ok);
242 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
243 g_pdg_cfg_ok = (uint32_t)good;
244 if (good != 0U) {
245 (void)ra8_board_uart_console_write(k_pdg_demo_ok_msg,
246 (size_t)(sizeof(k_pdg_demo_ok_msg) - 1U));
247 (void)ra8_board_led_toggle(k_ra8_board_led1);
248 } else {
249 (void)ra8_board_uart_console_write(k_pdg_demo_bad_msg,
250 (size_t)(sizeof(k_pdg_demo_bad_msg) - 1U));
251 (void)ra8_board_led_toggle(k_ra8_board_led2);
252 }
253 ++g_pdg_heartbeat;
254 ra8_delay_ms(k_pdg_demo_period_ms);
255 }
256 pdg_demo_panic_halt();
257 return 0;
258}
