119static void erb_print_hex(uint32_t value)
120{
121 uint8_t buf[k_erb_hex_nibbles];
122 for (uint32_t i = 0U; i < (uint32_t)k_erb_hex_nibbles; i++) {
123 const uint32_t shift = ((uint32_t)k_erb_hex_nibbles - 1U - i) * (uint32_t)k_erb_nibble_bits;
124 const uint32_t nib = (value >> shift) & (uint32_t)k_erb_nibble_mask;
125 buf[i] =
126 (uint8_t)((nib < (uint32_t)k_erb_dec_ten) ? ('0' + nib) : ('A' + (nib - k_erb_dec_ten)));
127 }
128 erb_print(buf, (uint32_t)k_erb_hex_nibbles);
129}

132static void erb_print_uint(uint32_t value)
133{
134 uint8_t buf[k_erb_dec_ten];
135 uint32_t n = 0U;
136 if (value == 0U) {
137 buf[n] = '0';
138 n++;
139 }
140 while ((value > 0U) && (n < (uint32_t)k_erb_dec_ten)) {
141 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_erb_dec_ten));
142 n++;
143 value /= (uint32_t)k_erb_dec_ten;
144 }
145 for (uint32_t i = 0U; i < n; i++) {
146 erb_print(&buf[n - 1U - i], 1U);
147 }
148}

155static uint32_t erb_hash_fb(uint32_t seed)
156{
157 const size_t nbytes = (size_t)k_erb_fb_w * (size_t)k_erb_fb_h * sizeof(uint16_t);
158 const uint8_t* fb = (const uint8_t*)s_framebuffer;
159 uint32_t hsh = seed;
160 for (size_t i = 0U; i < nbytes; i++) {
161 hsh = (hsh ^ (uint32_t)fb[i]) * (uint32_t)k_erb_fnv_prime;
162 }
163 return hsh;
164}

172{

233static uint32_t erb_distinct_levels(const uint8_t* px, uint32_t n)
234{
235 bool seen[k_erb_level_count] = {};
236 uint32_t distinct = 0U;
237 const uint32_t lim = (n < (uint32_t)k_erb_img_max_px) ? n : (uint32_t)k_erb_img_max_px;
238 for (uint32_t i = 0U; i < lim; i++) {
239 if (!seen[px[i]]) {
240 seen[px[i]] = true;
241 distinct++;
242 }
243 }
244 return distinct;
245}

263static void erb_render_image(void)
264{
265 const void* img_book = (const void*)k_rabook_gray8_fixture;
266 if (ra8_book_validate(img_book, (size_t)k_rabook_gray8_fixture_len) != k_ra8_ok) {
267 erb_panic_halt(k_msg_berr, (uint32_t)sizeof(k_msg_berr) - 1U);
268 }
269 const ra8_book_header_t* hdr = ra8_book_header(img_book);
270 if (hdr->image_count == 0U) {
271 erb_panic_halt(k_msg_berr, (uint32_t)sizeof(k_msg_berr) - 1U);
272 }
273 const ra8_book_image_t* im = &ra8_book_images(img_book)[0];
274 const uint32_t npx = (uint32_t)im->width * (uint32_t)im->height;
275 /* Must be a gray8 raster (8bpp: one byte per pixel), within the scan bound. */
276 if ((im->format != (uint8_t)k_ra8_book_image_gray4) ||
277 (ra8_book_image_pixfmt(im) != k_ra8_book_pixfmt_gray8) || (im->raw_size != npx) ||
278 (npx == 0U) || (npx > (uint32_t)k_erb_img_max_px)) {
279 erb_panic_halt(k_msg_berr, (uint32_t)sizeof(k_msg_berr) - 1U);
280 }
281 /* Continuous tone survived: more than 16 distinct levels is impossible for a
282 * 4bpp store, so this proves the source was kept at full depth, not quantised. */
283 const uint8_t* px = ra8_book_image_data(img_book, im);
284 if (erb_distinct_levels(px, npx) <= (uint32_t)k_erb_gray4_levels) {
285 erb_panic_halt(k_msg_berr, (uint32_t)sizeof(k_msg_berr) - 1U);
286 }
287
288 /* Render the retained pixels straight (1 byte/pixel, no unpack) at full res. */
289 (void)ra8_gfx_clear((uint32_t)k_erb_bg);
290 (void)ra8_gfx_blit_gray8(px, (int32_t)im->width, (int32_t)im->height, 0, 0);
291
292 erb_print(k_msg_img, (uint32_t)sizeof(k_msg_img) - 1U);
293 erb_print_uint((uint32_t)im->width);
294 erb_print(k_msg_x, (uint32_t)sizeof(k_msg_x) - 1U);
295 erb_print_uint((uint32_t)im->height);
296 erb_print(k_msg_g8, (uint32_t)sizeof(k_msg_g8) - 1U);
297}
