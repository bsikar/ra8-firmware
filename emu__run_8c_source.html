221static loop_action_t run_loop_prologue(run_loop_t* st)
222{
223 const emu_run_cfg_t* cfg = st->cfg;
224 /* RA8_EMU_PROFILE: charge the wall time of the previous chunk to the
225 * function its execution started in (so a cheap WFI-halt chunk and an
226 * expensive compute chunk are weighted by real time, not by chunk count). */
227 if (emu_prof_mode() == k_prof_wall) {
228 const double now = board_now_s();
229 if (st->prof_prev_t > 0.0) {
230 prof_add(st->prof_prev_pc, now - st->prof_prev_t);
231 }
232 st->prof_prev_pc = st->run_pc;
233 st->prof_prev_t = now;
234 }
235 emu_view_publish(st->run_pc, st->chunks); /* PC + chunk counter for the board view. */
236
237 /* --reboot N: after each boot's settle window force a power-on warm reboot;
238 * the reset-retained models (VBATT backup) survive, proving reset-survival. */
239 if ((st->reboot_count > 0) && ((st->chunks - st->last_boot_chunk) >= (uint32_t)k_reboot_settle)) {
240 st->reboot_count--;
241 board_periph_reset_set_cause(true, false, false, false); /* power-on reboot */
242 st->run_pc = warm_reboot(cfg->uc, cfg->elf, cfg->elf_len, cfg->want_trace);
243 if (st->run_pc == 0U) {
244 return k_loop_break; /* reboot failed to reload the image -- end the run */
245 }
246 st->last_boot_chunk = st->chunks;
247 return k_loop_continue;
248 }
249 /* A peripheral (the watchdog) may have requested a reset on its tick: latch
250 * the watchdog cause and warm-reboot so the next boot reads reset_by=wdt. */
251 bool wdt_rst = false;
252 bool iwdt_rst = false;
253 if (board_periph_reset_take_request(&wdt_rst, &iwdt_rst)) {
254 board_periph_reset_set_cause(false, false, wdt_rst, iwdt_rst);
255 st->run_pc = warm_reboot(cfg->uc, cfg->elf, cfg->elf_len, cfg->want_trace);
256 if (st->run_pc == 0U) {
257 return k_loop_break; /* reboot failed to reload the image -- end the run */
258 }
259 st->last_boot_chunk = st->chunks;
260 return k_loop_continue;
261 }
262 return k_loop_next;
263}

282static void run_loop_tick_inputs(run_loop_t* st)
283{
284 const emu_run_cfg_t* cfg = st->cfg;
285 emu_exc_arm_systick(); /* one SysTick period per outer chunk. */
286 /* DWT cycle counter: keep a masked-context ra8_delay_ms (spins on CYCCNT with
287 * PRIMASK set) making progress. Inert unless the firmware enabled it. */
288 dwt_cyccnt_advance(cfg->uc);
289 board_periph_tick(cfg->uc); /* advance timers; pend any wrap IRQ like SysTick. */
290 board_net_tick();
291
292 /* Drain buffered keystrokes into the console UART RX (the SCI channel --input
293 * targets), from the live window (keyDown) and the headless --keys injector. */
294 char key_byte = 0;
295 while (board_input_pop_key(&key_byte)) {
296 const uint8_t kb = (uint8_t)key_byte;
297 board_periph_sci_feed_rx(board_periph_sci_console_channel(), &kb, 1U);
298 }
299
300 /* Headless --click: keep one contact armed in the GT911 model until the
301 * firmware's real ra8_touch_read drains it; re-arming each chunk is needed
302 * because ra8_touch_open clears the GT911 status byte during bring-up. */
303 if (cfg->want_click && (st->click_btn != k_board_overlay_btn_none)) {
304 if (!st->button_fired) {
305 if ((st->click_btn == k_board_overlay_btn_battery) ||
306 (st->click_btn == k_board_overlay_btn_batt_chg) ||
307 (st->click_btn == k_board_overlay_btn_lowpower)) {
308 apply_battery_click(st->click_btn, (uint16_t)cfg->click_x, st->disp_w); /* SOC/CHG/LP. */
309 } else {
310 set_switch(st->click_btn, true); /* headless --click SW1/SW2: press + hold. */
311 }
312 st->button_fired = true;
313 }
314 } else if (cfg->want_click && !st->click_was_tab && (board_periph_touch_reported() == 0U)) {
315 uint16_t cnx = (uint16_t)cfg->click_x;
316 uint16_t cny = (uint16_t)cfg->click_y;
317 unrotate_click((uint16_t)cfg->click_x,
318 (uint16_t)cfg->click_y,
319 st->panel_w,
320 st->panel_h,
321 cfg->rotate_deg,
322 &cnx,
323 &cny);
324 board_periph_touch_inject(cnx, cny);
325 }
326}

346static loop_action_t run_loop_run_chunk(run_loop_t* st)
347{
348 const emu_run_cfg_t* cfg = st->cfg;
349 if (run_inner(cfg->uc, cfg->vtor_base, &st->run_pc, &st->err)) {
350 return k_loop_break;
351 }
352 if (emu_prof_stop_hit()) {
353 st->prof_stopped = true; /* RA8_EMU_STOP_PC reached -- end the profiled run. */
354 return k_loop_break;
355 }
356 /* AIRCR.SYSRESETREQ: latch a software-reset cause and warm reboot the firmware
357 * from its reset vector, then keep running. */
358 if (emu_exc_reboot_requested()) {
359 board_periph_reset_set_cause(false, true, false, false); /* software reset */
360 st->run_pc = warm_reboot(cfg->uc, cfg->elf, cfg->elf_len, cfg->want_trace);
361 emu_exc_clear_reboot_request();
362 if (st->run_pc == 0U) {
363 return k_loop_break; /* reboot failed to reload the image -- end the run */
364 }
365 return k_loop_continue;
366 }
367 return k_loop_next;
368}

386static void run_loop_record(run_loop_t* st)
387{
388 const emu_run_cfg_t* cfg = st->cfg;
389 /* Dual-core: step cpu1 interleaved with cpu0; a cpu1 fault just stops cpu1. */
390 emu_cpu1_step();
391 /* --record: dump the composite every k_record_every chunks as a numbered PPM. */
392 if ((cfg->record_dir != nullptr) && (st->composite != nullptr) &&
393 ((st->chunks % (uint32_t)k_record_every) == 0U)) {
394 build_composite(cfg->uc,
395 st->panel_fb,
396 st->rot_fb,
397 st->composite,
398 st->panel_w,
399 st->panel_h,
400 st->disp_w,
401 st->disp_h,
402 cfg->rotate_deg,
403 cfg->win_title);
404 enum : uint16_t {
405 k_rec_path_cap = 1024U,
406 };
407 char fpath[k_rec_path_cap];
408 (void)snprintf(fpath,
409 sizeof(fpath),
410 "%s/frame_%06u.ppm",
411 cfg->record_dir,
412 (unsigned)st->rec_frames);
413 if (write_ppm(fpath, st->composite, st->comp_w, st->comp_h) == 0) {

480static loop_action_t run_loop_view(run_loop_t* st)
481{
482 uint16_t cx = 0U;
483 uint16_t cy = 0U;
484 if (board_view_poll_click(st->view, &cx, &cy)) {
485 const board_overlay_btn_t hit =
486 route_click(cx, cy, st->panel_w, st->panel_h, st->disp_w, st->cfg->rotate_deg);
487 st->slider_grab = (hit == k_board_overlay_btn_battery);
488 st->held_btn = ((hit == k_board_overlay_btn_sw1) || (hit == k_board_overlay_btn_sw2))
489 ? hit
490 : k_board_overlay_btn_none;
491 }
492 if (board_view_poll_release(st->view)) { /* mouse-up: release a held button + grab. */
493 if (st->held_btn != k_board_overlay_btn_none) {
494 set_switch(st->held_btn, false);
495 st->held_btn = k_board_overlay_btn_none;
496 }
497 st->slider_grab = false;
498 }
499 uint16_t dx = 0U;
500 uint16_t dy = 0U;
501 if (board_view_poll_drag(st->view, &dx, &dy) && st->slider_grab) {
502 apply_battery_click(k_board_overlay_btn_battery, dx, st->disp_w);
503 }
504 emu_view_wheel(board_view_poll_scroll(st->view)); /* wheel pages the scrollback. */
505 if ((st->chunks % (uint32_t)k_view_present_every) == 0U) {
506 run_view_maybe_present(st);
507 }
508 if (board_view_pump(st->view)) {
509 st->closed = true;
510 return k_loop_break;
511 }
512 return k_loop_next;

533{
534 const bool click_acted = (st->click_btn != k_board_overlay_btn_none)
535 ? st->button_fired

537 if (click_acted) {

547 return k_loop_break;

609static bool run_stop_idle(run_loop_t* st)
610{
611 if (!((st->guards.idle_stop_chunks > 0U) && (st->cfg->record_dir == nullptr))) {
612 return false;
613 }
614 const uint64_t idle_sig = (uint64_t)emu_mmio_reads() + (uint64_t)emu_mmio_writes() +
615 (uint64_t)emu_exc_pendsv_takes() + (uint64_t)emu_exc_svc_takes() +
616 (uint64_t)board_periph_irq_total();
617 if (idle_sig == st->idle_sig_prev) {
618 st->idle_run++;
619 if (st->idle_run >= st->guards.idle_stop_chunks) {
620 st->idle_stopped = true;
621 return true;
622 }
623 } else {
624 st->idle_sig_prev = idle_sig;
625 st->idle_run = 0U;
626 }
627 return false;
628}

835static void run_loop(run_loop_t* st)
836{
837 for (; st->chunks < st->guards.max_chunks; st->chunks++) {
838 const loop_action_t pro = run_loop_prologue(st);
839 if (pro == k_loop_break) {
840 break;
841 }
842 if (pro == k_loop_continue) {
843 continue;
844 }
845 run_loop_tick_inputs(st);
846 const loop_action_t chunk = run_loop_run_chunk(st);
847 if (chunk == k_loop_break) {
848 break;
849 }
850 if (chunk == k_loop_continue) {
851 continue;
852 }
853 run_loop_record(st);
854 if (run_loop_present_and_stops(st) == k_loop_break) {
855 break;
856 }
857 }
858}
