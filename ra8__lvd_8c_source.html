379static ra8_err_t internal_validate_cfg(const ra8_lvd_channel_map_t* map, const ra8_lvd_cfg_t* cfg)
380{
381 ra8_err_t err = internal_validate_threshold(cfg->threshold);
382 if (err != k_ra8_ok) {
383 return err;
384 }
385 err = ra8_lvd_internal_validate_div(cfg->filter_div);
386 if (err != k_ra8_ok) {
387 return err;
388 }
389 if (map->has_irq) {
390 err = internal_validate_edge(cfg->edge);
391 if (err != k_ra8_ok) {
392 return err;
393 }
394 } else {
395 /* HUM Ch 8.2.5 "PVDnCR0 : Voltage Monitor n Circuit Control Register 0" p 306 */
396 /* n channels only have RE -- reject IRQ / NMI responses. */
397 if ((cfg->response == k_ra8_lvd_response_interrupt) ||
398 (cfg->response == k_ra8_lvd_response_nmi)) {
399 return k_ra8_err_not_supported;
400 }
401 }
402 /* HUM Ch 8.2.4 "PVDmCR0 : Voltage Monitor m Circuit Control Register 0" p 306 */
403 /* RN=1 prohibited when RHSEL=1. */
404 if (ra8_lvd_internal_reject_hvd_after((uint32_t)k_ra8_lvd_hysteresis_hvd,
405 (uint32_t)k_ra8_lvd_negate_after_assert,
406 (uint32_t)cfg->hysteresis,
407 (uint32_t)cfg->negate)) {
408 return k_ra8_err_invalid_arg;
409 }
410 return k_ra8_ok;
411}

436static uint8_t internal_compose_cr0(const ra8_lvd_channel_map_t* map, const ra8_lvd_cfg_t* cfg)
437{
438 uint8_t cr0 = (uint8_t)((uint8_t)cfg->filter_div << k_ra8_lvd_cr0_shift_fsamp);
439 cr0 &= k_ra8_lvd_cr0_mask_fsamp;
440 cr0 |= k_ra8_lvd_cr0_mask_dfdis; /* DFDIS = 1 while writing FSAMP. */
441 if (map->has_irq) {
442 if (cfg->negate == k_ra8_lvd_negate_after_assert) {
443 cr0 |= k_ra8_lvd_cr0_mask_rn;
444 }
445 if (ra8_lvd_internal_set_ri_bit((uint32_t)k_ra8_lvd_response_reset,
446 (uint32_t)k_ra8_lvd_response_reset_on_rise,
447 (uint32_t)cfg->response)) {
448 cr0 |= k_ra8_lvd_cr0_mask_ri;
449 }
450 }
451 return cr0;
452}

472static void internal_program_cr1(const ra8_lvd_channel_map_t* map, const ra8_lvd_cfg_t* cfg)
473{
474 /* HUM Ch 8.2.6 "PVDmCR1 : Voltage Monitor m Circuit Control Register" p 307 */
475 uint8_t cr1 = (uint8_t)((uint8_t)cfg->edge & k_ra8_lvd_cr1_mask_idtsel);
476 if (cfg->irq_type == k_ra8_lvd_irq_maskable) {
477 cr1 |= k_ra8_lvd_cr1_mask_irqsel;
478 }
479 *ra8_lvd_reg8(map->cr1) = cr1;
480
481 if (cfg->clear_status) {
482 /* HUM Ch 8.2.7 "PVDmSR : Voltage Monitor m Circuit Status Register" p 307 */
483 *ra8_lvd_reg8(map->sr) = 0U;
484 }

505static void internal_lvd_program_cmpcr(const ra8_lvd_channel_map_t* map, const ra8_lvd_cfg_t* cfg)
506{
507 /* Step 1: disable everything; HUM Ch 8.2.4 / 8.2.2 */
508 *ra8_lvd_reg8(map->cr0) = internal_cr0_apply_reserved(map, 0U);
509 *ra8_lvd_reg8(map->cmpcr) = 0U;
510
511 /* Step 2: PVDLVL with PVDE = 0 */
512 uint8_t cmpcr = (uint8_t)((uint8_t)cfg->threshold & k_ra8_lvd_cmpcr_mask_pvdlvl);
513 *ra8_lvd_reg8(map->cmpcr) = cmpcr;
514
515 /* Step 3: HUM Ch 8.2.8 / 8.2.9 -- RHSEL on FCR */
516 *ra8_lvd_reg8(map->fcr) = (uint8_t)((uint8_t)cfg->hysteresis & k_ra8_lvd_fcr_mask_rhsel);
517
518 /* Step 4: re-assert PVDE */
519 cmpcr |= k_ra8_lvd_cmpcr_mask_pvde;
520 *ra8_lvd_reg8(map->cmpcr) = cmpcr;
521}

542static void internal_lvd_program_cr0_chain(const ra8_lvd_channel_map_t* map,
543 const ra8_lvd_cfg_t* cfg)
544{
545 uint8_t cr0 = internal_compose_cr0(map, cfg);
546 *ra8_lvd_reg8(map->cr0) = internal_cr0_apply_reserved(map, cr0);
547
548 if (cfg->filter_en) {
549 cr0 &= (uint8_t)~k_ra8_lvd_cr0_mask_dfdis;
550 *ra8_lvd_reg8(map->cr0) = internal_cr0_apply_reserved(map, cr0);
551 }
552
553 if (map->has_irq) {
554 internal_program_cr1(map, cfg);
555 }
556
557 if (cfg->irq_enable && (cfg->response != k_ra8_lvd_response_none)) {
558 /* RIE (m-channel: IRQ/reset enable) and RE (n-channel: reset-only)
559 * sit at the same CR0 bit position (0x01) on RA8D2; the semantic
560 * distinction lives in `map->has_irq`. Either mask gates bit 0. */
561 cr0 |= k_ra8_lvd_cr0_mask_rie;
562 *ra8_lvd_reg8(map->cr0) = internal_cr0_apply_reserved(map, cr0);
563 }
564
565 cr0 |= k_ra8_lvd_cr0_mask_cmpe;
566 *ra8_lvd_reg8(map->cr0) = internal_cr0_apply_reserved(map, cr0);
567}

594ra8_err_t ra8_lvd_channel_init(ra8_lvd_channel_t channel, const ra8_lvd_cfg_t* cfg)
595{
596 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
597
598 uint8_t idx = 0U;
599 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
600 RA8_RETURN_ON_ERROR(map_err, s_tag, "lvd_init: bad channel"); /* GCOVR_EXCL_BR_LINE */
601
602 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
603 const ra8_err_t cfg_err = internal_validate_cfg(&map, cfg);
604 RA8_RETURN_ON_ERROR(cfg_err, s_tag, "lvd_init: bad cfg"); /* GCOVR_EXCL_BR_LINE */
605
606 internal_lvd_program_cmpcr(&map, cfg);
607 internal_lvd_program_cr0_chain(&map, cfg);
608
609 ra8_log_info_val(s_tag, "lvd_init ch", (uint32_t)channel);
610 return k_ra8_ok;
611}

635ra8_err_t ra8_lvd_channel_deinit(ra8_lvd_channel_t channel)
636{
637 uint8_t idx = 0U;
638 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
639 RA8_RETURN_ON_ERROR(map_err, s_tag, "lvd_deinit: bad channel"); /* GCOVR_EXCL_BR_LINE */
640
641 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
642
643 /* === HUM Table 8.5 step 1 / Table 8.7 step 1 -- drop CMPE. */
644 ra8_lvd_internal_cr0_rmw(&map, k_ra8_lvd_cr0_mask_cmpe, 0U);
645
646 /* === HUM Table 8.5 step 3 / Table 8.7 step 3 -- drop RIE / RE. */
647 uint8_t enable_bit = k_ra8_lvd_cr0_mask_re;
648 if (map.has_irq) {
649 enable_bit = k_ra8_lvd_cr0_mask_rie;
650 }
651 ra8_lvd_internal_cr0_rmw(&map, enable_bit, 0U);
652
653 /* === HUM Table 8.5 step 4 / Table 8.7 step 4 -- DFDIS = 1 (filter off). */
654 ra8_lvd_internal_cr0_rmw(&map, 0U, k_ra8_lvd_cr0_mask_dfdis);
655
656 /* === HUM Table 8.5 step 5 / Table 8.7 step 5 -- drop PVDE. */
657 /* HUM Ch 8.2.2 "PVDmCMPCR" p 303 */
658 *ra8_lvd_reg8(map.cmpcr) = 0U;
659
660 /* Cleanly drop the rest of the per-channel state. */
661 *ra8_lvd_reg8(map.cr0) = internal_cr0_apply_reserved(&map, 0U);
662 *ra8_lvd_reg8(map.fcr) = 0U;
663 if (map.has_irq) {
664 /* HUM Ch 8.2.6 "PVDmCR1 : Voltage Monitor m Circuit Control Register" p 307 */
665 *ra8_lvd_reg8(map.cr1) = 0U;
666 /* HUM Ch 8.2.7 "PVDmSR : Voltage Monitor m Circuit Status Register" p 307 */
667 *ra8_lvd_reg8(map.sr) = 0U;
668 }
669 return k_ra8_ok;
670}

699ra8_err_t ra8_lvd_set_threshold(ra8_lvd_channel_t channel, ra8_lvd_pvdlvl_t threshold)
700{
701 uint8_t idx = 0U;
702 const ra8_err_t map_err = ra8_lvd_internal_channel_to_idx(channel, &idx);
703 RA8_RETURN_ON_ERROR(map_err, s_tag, "lvd_set_threshold: bad channel"); /* GCOVR_EXCL_BR_LINE */
704
705 const ra8_err_t lvl_err = internal_validate_threshold(threshold);
706 RA8_RETURN_ON_ERROR(lvl_err, s_tag, "lvd_set_threshold: bad threshold"); /* GCOVR_EXCL_BR_LINE */
707
708 const ra8_lvd_channel_map_t map = s_lvd_map[idx];
709
710 /* HUM Ch 8.2.2 "PVDmCMPCR : Voltage Monitor m Comparator Control Register" p 303 */
711 /* Preserve the PVDE bit, drop it around the PVDLVL write, restore it afterwards. */
712 const uint8_t prev = *ra8_lvd_reg8(map.cmpcr);
713 const uint8_t pvde_was = (uint8_t)(prev & k_ra8_lvd_cmpcr_mask_pvde);
714
715 *ra8_lvd_reg8(map.cmpcr) = 0U;
716
717 uint8_t cmpcr = (uint8_t)((uint8_t)threshold & k_ra8_lvd_cmpcr_mask_pvdlvl);
718 cmpcr |= pvde_was;
719 *ra8_lvd_reg8(map.cmpcr) = cmpcr;
720 return k_ra8_ok;
721}
