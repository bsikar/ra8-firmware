51RA8_PRIV ra8_err_t ra8_jof_png_priv_pull_exact(ra8_png_state_t* st, uint8_t* buf, uint32_t len)
52{
53 uint32_t done = 0U;
54 while (done < len) {
55 size_t got = 0U;
56 const ra8_err_t err = st->pull(st->pull_ctx, &buf[done], (size_t)(len - done), &got);
57 if (err != k_ra8_ok) {
58 return err;
59 }
60 if (got == 0U) {
61 return k_ra8_err_protocol_error;
62 }
63 done += (uint32_t)got;

84RA8_PRIV ra8_err_t ra8_jof_png_priv_skip(ra8_png_state_t* st, uint32_t len)
85{
86 uint8_t scratch[k_ra8_png_skip_chunk];
87 uint32_t left = len;
88 while (left > 0U) {
89 const uint32_t take =
90 (left < (uint32_t)k_ra8_png_skip_chunk) ? left : (uint32_t)k_ra8_png_skip_chunk;
91 const ra8_err_t err = ra8_jof_png_priv_pull_exact(st, scratch, take);
92 if (err != k_ra8_ok) {
93 return err;
94 }
95 left -= take;
96 }
97 return k_ra8_ok;

117RA8_PRIV ra8_err_t ra8_jof_png_priv_chunk_hdr(ra8_png_state_t* st,
118 uint32_t* out_len,
119 uint32_t* out_type)
120{
121 uint8_t hdr[k_ra8_png_chunk_hdr] = {};

195png_check_ihdr(ra8_png_state_t* st, const uint8_t* ihdr, uint16_t max_w, uint16_t max_h)
196{
197 const uint32_t w32 = ((uint32_t)ihdr[0] << k_ra8_png_be_sh24) |
198 ((uint32_t)ihdr[1] << k_ra8_png_be_sh16) |
199 ((uint32_t)ihdr[2] << k_ra8_png_be_sh8) | (uint32_t)ihdr[3];
200 const uint32_t h32 = ((uint32_t)ihdr[4] << k_ra8_png_be_sh24) |
201 ((uint32_t)ihdr[5] << k_ra8_png_be_sh16) |
202 ((uint32_t)ihdr[6] << k_ra8_png_be_sh8) | (uint32_t)ihdr[7];
203 if ((w32 == 0U) || (w32 > (uint32_t)max_w)) {
204 return k_ra8_err_invalid_size;
205 }
206 if ((h32 == 0U) || (h32 > (uint32_t)max_h)) {
207 return k_ra8_err_invalid_size;
208 }
209 if (ihdr[k_ra8_png_ihdr_ofs_depth] != (uint8_t)k_ra8_png_depth_8) {
210 return k_ra8_err_not_supported;
211 }
212 st->src_ch = png_src_channels(ihdr[k_ra8_png_ihdr_ofs_color]);
213 if (st->src_ch == 0U) {
214 return k_ra8_err_not_supported;
215 }
216 if (ihdr[k_ra8_png_ihdr_ofs_compress] != 0U) {
217 return k_ra8_err_not_supported;
218 }
219 if (ihdr[k_ra8_png_ihdr_ofs_filter] != 0U) {
220 return k_ra8_err_not_supported;
221 }
222 if (ihdr[k_ra8_png_ihdr_ofs_interlace] != 0U) {
223 return k_ra8_err_not_supported; /* Adam7 breaks streaming; fail closed */
224 }
225 st->w = (uint16_t)w32;
226 st->h = (uint16_t)h32;
227 st->color_type = ihdr[k_ra8_png_ihdr_ofs_color];
228 return k_ra8_ok;
229}

251static ra8_err_t png_parse_ihdr(ra8_png_state_t* st, uint16_t max_w, uint16_t max_h)
252{
253 uint32_t len = 0U;
254 uint32_t type = 0U;
255 ra8_err_t err = ra8_jof_png_priv_chunk_hdr(st, &len, &type);
256 if (err != k_ra8_ok) {
257 return err;
258 }
259 if ((type != (uint32_t)k_ra8_png_type_ihdr) || (len != (uint32_t)k_ra8_png_ihdr_len)) {
260 return k_ra8_err_protocol_error;
261 }
262 uint8_t ihdr[k_ra8_png_ihdr_len] = {};
263 err = ra8_jof_png_priv_pull_exact(st, ihdr, sizeof(ihdr));
264 if (err != k_ra8_ok) {
265 return err;
266 }
267 err = png_check_ihdr(st, ihdr, max_w, max_h);
268 if (err != k_ra8_ok) {
269 return err;
270 }
271 return ra8_jof_png_priv_skip(st, (uint32_t)k_ra8_png_crc_bytes);
272}

421RA8_PRIV ra8_err_t ra8_jof_png_priv_finish(ra8_png_state_t* st)
422{
423 uint32_t len = st->pending_len;
424 uint32_t type = st->pending_type;
425 if (st->pending_valid == 0U) {
426 ra8_err_t err = ra8_jof_png_priv_skip(st, (uint32_t)k_ra8_png_crc_bytes); /* last IDAT's CRC */
427 if (err != k_ra8_ok) {
428 return err;
429 }
430 err = ra8_jof_png_priv_chunk_hdr(st, &len, &type);
431 if (err != k_ra8_ok) {
432 return err;
433 }
434 }
435 for (uint32_t guard = 0U; guard < (uint32_t)k_ra8_png_max_chunks; guard++) {
436 if (type == (uint32_t)k_ra8_png_type_iend) {
437 return k_ra8_ok;
438 }
439 if (type == (uint32_t)k_ra8_png_type_idat) {
440 return k_ra8_err_protocol_error; /* IDAT after the stream completed */
441 }
442 ra8_err_t err = ra8_jof_png_priv_skip(st, len + (uint32_t)k_ra8_png_crc_bytes);
443 if (err != k_ra8_ok) {
444 return err;
445 }
446 err = ra8_jof_png_priv_chunk_hdr(st, &len, &type);
447 if (err != k_ra8_ok) {
448 return err;
449 }
450 }
451 return k_ra8_err_protocol_error;
