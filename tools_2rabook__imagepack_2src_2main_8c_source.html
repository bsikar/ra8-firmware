146RA8_INTERNAL
147static bool priv_parse_opts(int argc, char** argv, ra8_fmt_opts_t* opts, const char** fmt)
148{
149 for (int i = 2; i < argc; ++i) { /* bounded by argc */
150 if ((strcmp(argv[i], "--verbose") == 0) || (strcmp(argv[i], "-v") == 0)) {
151 opts->verbose = true;
152 } else if ((strcmp(argv[i], "--format") == 0) && ((i + 1) < argc)) {
153 ++i;
154 *fmt = argv[i];
155 } else if ((strcmp(argv[i], "--in") == 0) && ((i + 1) < argc)) {
156 ++i;
157 opts->in_path = argv[i];
158 } else if ((strcmp(argv[i], "--out") == 0) && ((i + 1) < argc)) {
159 ++i;
160 opts->out_path = argv[i];
161 } else if ((argv[i][0] != '-') && (opts->in_path == nullptr)) {
162 opts->in_path = argv[i];
163 } else {
164 (void)fprintf(stderr, "ra8_fmt: unrecognised argument '%s'\n", argv[i]);
165 return false;
166 }
167 }
168 return true;
169}

210RA8_INTERNAL
211static ra8_err_t priv_dispatch(ra8_fmt_verb_t verb,
212 const ra8_fmt_desc_t* desc,
213 const ra8_fmt_blob_t* src,
214 const ra8_fmt_opts_t* opts)
215{
216 ra8_fmt_convert_fn fn = nullptr;
217 if (verb == k_fmt_verb_convert) {
218 fn = desc->convert;
219 } else if (verb == k_fmt_verb_inspect) {
220 fn = desc->inspect;
221 } else {
222 fn = desc->verify;
223 }
224 if (fn == nullptr) {
225 (void)fprintf(opts->report, "ra8_fmt: format '%s' does not support this verb\n", desc->name);
226 return k_ra8_err_not_supported;
227 }
228 return fn(src, opts);
229}

231int main(int argc, char** argv)
232{
233 ra8_log_set_byte_sink(fmt_log_sink, nullptr);
234 if (argc < 2) {
235 priv_usage(stderr);
236 return (int)k_fmt_exit_usage;
237 }
238 const ra8_fmt_verb_t verb = priv_parse_verb(argv[1]);
239 if (verb == k_fmt_verb_none) {
240 priv_usage(stderr);
241 return (int)k_fmt_exit_usage;
242 }
243 ra8_fmt_opts_t opts = {.in_path = nullptr,
244 .out_path = nullptr,
245 .verbose = false,
246 .report = stdout};
247 const char* fmt = nullptr;
248 if (!priv_parse_opts(argc, argv, &opts, &fmt)) {
249 priv_usage(stderr);
250 return (int)k_fmt_exit_usage;
251 }
252 if (opts.in_path == nullptr) {
253 (void)fprintf(stderr, "ra8_fmt: no input file given\n");
254 priv_usage(stderr);
255 return (int)k_fmt_exit_usage;
256 }
257 ra8_fmt_blob_t src = {};
258 const ra8_err_t src_rc = ra8_fmt_slurp(opts.in_path, &src);
259 if (src_rc != k_ra8_ok) {
260 (void)fprintf(stderr, "ra8_fmt: cannot read '%s' (rc=%d)\n", opts.in_path, (int)src_rc);
261 return (int)k_fmt_exit_fail;
262 }
263 const ra8_fmt_desc_t* desc = priv_resolve(verb, fmt, &src);
264 if (desc == nullptr) {
265 (void)fprintf(stderr, "ra8_fmt: unknown or undetectable format; pass --format\n");
266 ra8_fmt_blob_free(&src);
267 priv_usage(stderr);
268 return (int)k_fmt_exit_usage;
269 }
270 const ra8_err_t rc = priv_dispatch(verb, desc, &src, &opts);
271 ra8_fmt_blob_free(&src);
272 return (rc == k_ra8_ok) ? (int)k_fmt_exit_ok : (int)k_fmt_exit_fail;
273}
