202static uint32_t audio_loopback_u32_to_dec(uint32_t value, uint8_t* buf)
203{
204 enum : uint8_t {
205 k_ascii_zero = 0x30U,
206 k_radix = 10U,
207 k_max_digits = 10U,
208 };
209 uint8_t tmp[k_max_digits];
210 uint32_t n = 0U;
211 if (value == 0U) {
212 buf[0] = k_ascii_zero;
213 return 1U;
214 }
215 while (value > 0U) {
216 tmp[n] = (uint8_t)(k_ascii_zero + (uint8_t)(value % k_radix));
217 n++;
218 value /= k_radix;
219 }
220 for (uint32_t i = 0U; i < n; i++) {
221 buf[i] = tmp[n - 1U - i];
222 }
223 return n;
224}

268int32_t main(void)
269{
270 audio_loopback_init_clocks_and_led();

272 audio_loopback_init_codec();
273
274 ra8_isr_globals_enable();
275
276 uint32_t block_count = 0U;
277
278 while (1) {

280 k_audio_loopback_block_samples) != k_ra8_ok) {
281 break;
282 }
283 block_count++;
284 if ((block_count % k_audio_loopback_print_period) == 0U) {
285 audio_loopback_print_count(block_count);
286 (void)ra8_board_led_toggle(k_ra8_board_led1);
287 }
288 }
289
290 audio_loopback_panic_halt();
291 return 0;
292}
