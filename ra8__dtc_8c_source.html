54/* Retained DTC vector-table base. ra8_dtc_enable() has no vector_base
55 * argument, yet the DTC fetches the table from RAM at DTCST=1; keep the
56 * base from ra8_dtc_init()/ra8_dtc_reconfigure() so the pre-enable D-cache
57 * clean can flush the caller-populated entries out to memory. */
58static void* s_dtc_vector_base;

60ra8_err_t ra8_dtc_init(void* vector_base)
61{
62 RA8_CHECK_NULL_PTR(vector_base, s_tag, "vector_base must not be nullptr");
63
64 /* DTC0 + DMAC0 share MSTPA22; ra8_mstp keeps the ref count so
65 * a follow-up ra8_dmac_start does not flip the bit again.
66 * HUM Ch 11.2.6 "MSTPCRA : Module Stop Control Register A" p 443 */
67 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_dmac0_dtc0);
68 RA8_RETURN_ON_ERROR(mst_err, s_tag, "dtc_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
69
70 volatile r_dtc_regs_t* reg = ra8_dtc();
71 /* HUM 18.2.1 DTCCR p 786 / 18.2.2 DTCVBR p 787 / 18.2.3 DTCST p 787. On a
72 * TrustZone part the secure (and flat-secure) DTC engine fetches its vector
73 * table from DTCVBR_SEC (+0x14, HUM 18.2.6 p 789); the plain DTCVBR (+0x04)
74 * is the Non-secure alias and a secure write to it is dropped. Program both
75 * so the table is found in either world (and so the ra8_emulator DTC model,
76 * which shadows DTCVBR, still sees the base). */
77 reg->DTCCR = 0U;
78 reg->DTCVBR = (uint32_t)(uintptr_t)vector_base;
79 reg->DTCVBR_SEC = (uint32_t)(uintptr_t)vector_base;
80 reg->DTCST = 0U;
81 s_dtc_vector_base = vector_base; /* flushed to RAM by ra8_dtc_enable() */
82
83 ra8_log_info(s_tag, "dtc_init");
84 return k_ra8_ok;
85}

87ra8_err_t ra8_dtc_deinit(void)
88{
89 volatile r_dtc_regs_t* reg = ra8_dtc();
90 /* HUM 18.2.3 DTCST p 787 / 18.2.1 DTCCR p 786 / 18.2.2 DTCVBR p 787 /
91 * 18.2.6 DTCVBR_SEC p 789. Clear both vector bases (see ra8_dtc_init). */
92 reg->DTCST = 0U;
93 reg->DTCCR = 0U;
94 reg->DTCVBR = 0U;
95 reg->DTCVBR_SEC = 0U;
96 s_dtc_fn = nullptr;
97 s_dtc_ctx = nullptr;
98 s_dtc_vector_base = nullptr;
99 return ra8_mstp_disable(k_ra8_mstp_dmac0_dtc0);
100}

102ra8_err_t ra8_dtc_enable(void)
103{
104 /* Cache coherency (M85 L1 D-cache): the DTC fetches its vector table --
105 * and, one indirection on, each per-source TI block -- straight from RAM.
106 * Vector-table entries the CPU wrote between ra8_dtc_init() and here may
107 * still sit in dirty cache lines the engine cannot see, so clean (write
108 * back) the table region before DTCST=1. Clean is non-destructive and an
109 * architectural no-op when the D-cache is off (every current app), so it
110 * is added unconditionally. The TI blocks and the source/destination data
111 * buffers are NOT reachable from this driver: the DTC is direction-blind
112 * (like the DMAC), so the owning driver cleans the TI/source and
113 * invalidates the destination. */
114 if (s_dtc_vector_base != nullptr) {
115 (void)ra8_cache_dcache_clean_by_addr(s_dtc_vector_base, (uint32_t)k_ra8_dtc_vector_table_align);
116 }
117 /* HUM 18.2.3 DTCST p 787. */
118 ra8_dtc()->DTCST = k_ra8_dtcst_dtcst_msk;
119 return k_ra8_ok;
120}

129ra8_err_t ra8_dtc_reconfigure(void* vector_base)
130{
131 RA8_CHECK_NULL_PTR(vector_base, s_tag, "vector_base must not be nullptr");
132 volatile r_dtc_regs_t* reg = ra8_dtc();
133 /* HUM 18.2.3 DTCST p 787 / 18.2.2 DTCVBR p 787 / 18.2.1 DTCCR p 786.
134 * Mirrors FSP r_dtc_set_info(): drop RRS before touching the table,
135 * rewrite the base, then re-enable RRS so the read-skip cache picks
136 * up the fresh entries. Bit 3 of DTCCR is reserved-write-1. */
137 reg->DTCST = 0U;
138 reg->DTCVBR = (uint32_t)(uintptr_t)vector_base;
139 reg->DTCVBR_SEC = (uint32_t)(uintptr_t)vector_base;
140 s_dtc_vector_base = vector_base;
141 /* Cache coherency (M85 L1 D-cache): the RRS toggle below drops the DTC's
142 * internal read-skip cache and forces it to re-read its descriptors from
143 * RAM. Clean the CPU's vector-table edits out to memory first so the
144 * re-read sees them, not stale dirty lines. Non-destructive and a no-op
145 * when the D-cache is off. The indirectly-referenced TI blocks live one
146 * pointer-hop away and are flushed by the owning driver (DTC is
147 * direction-blind, like the DMAC). */
148 (void)ra8_cache_dcache_clean_by_addr(vector_base, (uint32_t)k_ra8_dtc_vector_table_align);
149 reg->DTCCR = k_ra8_dtccr_rrs_disable;
150 reg->DTCCR = k_ra8_dtccr_rrs_enable;
151 return k_ra8_ok;
152}
