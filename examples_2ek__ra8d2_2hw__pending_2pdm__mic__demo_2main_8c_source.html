239static uint32_t pdm_demo_i32_to_dec(int32_t value, uint8_t* out)
240{
241 uint8_t tmp[k_pdm_demo_int_str_cap] = {};
242 uint32_t n = 0U;
243 bool neg = value < 0;
244 /* Work in int64 so INT32_MIN negates without overflow. */
245 int64_t v = value;
246 if (neg) {
247 v = -v;
248 }
249 if (v == 0) {
250 tmp[n++] = (uint8_t)'0';
251 } else {
252 while (v > 0 && n < (uint32_t)k_pdm_demo_int_str_cap) {
253 tmp[n++] = (uint8_t)('0' + (uint8_t)(v % (int64_t)k_pdm_demo_decimal));
254 v /= (int64_t)k_pdm_demo_decimal;
255 }
256 }
257 if (neg && n < (uint32_t)k_pdm_demo_int_str_cap) {
258 tmp[n++] = (uint8_t)'-';
259 }
260 for (uint32_t i = 0U; i < n; ++i) {
261 out[i] = tmp[n - 1U - i];
262 }
263 if (n < (uint32_t)k_pdm_demo_int_str_cap) {
264 out[n] = 0U;
265 }
266 return n;
267}

286static uint32_t pdm_demo_isqrt(uint64_t value)
287{
288 uint64_t rem = 0U;
289 uint64_t root = 0U;
290 for (int32_t i = 0; i < (int32_t)k_pdm_isqrt_steps; ++i) {
291 root <<= 1;
292 rem = (rem << 2) | (value >> (uint32_t)k_pdm_isqrt_top2);
293 value <<= 2;
294 ++root;
295 if (root <= rem) {
296 rem -= root;
297 ++root;
298 } else {
299 --root;
300 }
301 }
302 return (uint32_t)(root >> 1);
303}

480static uint32_t pdm_demo_capture(int32_t* buf, uint32_t want)
481{
482 uint32_t filled = 0U;
483 for (uint32_t tries = 0U; tries < (uint32_t)k_pdm_demo_poll_max; ++tries) {
484 if (filled >= want) {
485 break;
486 }
487 uint32_t got = 0U;
488 if (ra8_pdm_read(k_pdm_demo_channel, &buf[filled], want - filled, &got) != k_ra8_ok) {
489 break;
490 }
491 filled += got;
492 }
493 return filled;
494}

516static void pdm_demo_analyze(const int32_t* buf, uint32_t n, pdm_metrics_t* m)
517{
518 int64_t sum = 0;
519 int32_t lo = buf[0];
520 int32_t hi = buf[0];
521 for (uint32_t i = 0U; i < n; ++i) {
522 const int32_t s = buf[i];
523 sum += s;
524 lo = (s < lo) ? s : lo;
525 hi = (s > hi) ? s : hi;
526 }
527 const int32_t mean = (int32_t)(sum / (int64_t)n);
528
529 uint64_t sumsq = 0U;
530 int32_t peak = 0;
531 for (uint32_t i = 0U; i < n; ++i) {
532 const int32_t d = buf[i] - mean;
533 const int32_t mag = (d < 0) ? -d : d;
534 peak = (mag > peak) ? mag : peak;
535 sumsq += (uint64_t)((int64_t)d * (int64_t)d);
536 }
537 m->min = lo;
538 m->max = hi;
539 m->mean = mean;
540 m->peak = peak;
541 m->rms = (int32_t)pdm_demo_isqrt(sumsq / (uint64_t)n);
542 m->n = n;
543}

565{
566 const int32_t span = m->max - m->min;

568 if (m->rms >= (int32_t)k_pdm_demo_rms_floor) {
569 return true;
570 }
571 }

645static bool pdm_demo_run_window(int32_t vary, pdm_metrics_t* out_m)
646{
647 const uint32_t got = pdm_demo_capture(s_pdm_demo_buf, (uint32_t)k_pdm_demo_window);
648 if (got == 0U) {
649 pdm_demo_tx(k_pdm_demo_banner_nodat, (uint32_t)(sizeof(k_pdm_demo_banner_nodat) - 1U));
650 return false;
651 }
652 pdm_demo_analyze(s_pdm_demo_buf, got, out_m);
653 const bool active = pdm_demo_is_active(out_m);
654 pdm_demo_publish(out_m, vary, active);
655 return true;
656}

677int32_t main(void)
678{
679 pdm_demo_clocks_or_halt();
680 pdm_demo_io_or_halt();
681 pdm_demo_pdm_or_halt();
682
683 /* Flush the sinc/high-pass filter transient before measuring. */
684 pdm_metrics_t m = {};
685 for (uint32_t i = 0U; i < (uint32_t)k_pdm_demo_discard; ++i) {
686 (void)pdm_demo_capture(s_pdm_demo_buf, (uint32_t)k_pdm_demo_window);
687 }
688
689 /* Sweep windows for the verdict, tracking the RMS spread. */
690 int32_t rms_lo = INT32_MAX;
691 int32_t rms_hi = 0;
692 for (uint32_t w = 0U; w < (uint32_t)k_pdm_demo_report_win; ++w) {
693 const int32_t vary = (rms_hi > rms_lo) ? (rms_hi - rms_lo) : 0;
694 if (!pdm_demo_run_window(vary, &m)) {
695 pdm_demo_panic_halt();
696 }
697 rms_lo = (m.rms < rms_lo) ? m.rms : rms_lo;
698 rms_hi = (m.rms > rms_hi) ? m.rms : rms_hi;
699 }
700
701 /* Live stream so a tap near the underside mics is visible on the wire. */
702 const int32_t vary = (rms_hi > rms_lo) ? (rms_hi - rms_lo) : 0;
703 while (1) {
704 if (pdm_demo_run_window(vary, &m)) {
705 (void)ra8_board_led_toggle(k_ra8_board_led1);
706 }
707 ra8_delay_ms((uint32_t)k_pdm_demo_live_ms);
708 }
709
710 pdm_demo_panic_halt();
711 return 0;
712}
