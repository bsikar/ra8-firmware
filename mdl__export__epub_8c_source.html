59RA8_INTERNAL static ra8_err_t internal_epub_media_type_from_sniff(mdl_storage_t* storage,
60 const char* dir,
61 const char* name,
62 const char** out)
63{
64 char src[k_fw_fs_path_cap];
65 if (priv_mdl_export_path_join(src, sizeof(src), dir, name) != k_ra8_ok) {
66 return k_ra8_err_invalid_size;
67 }
68 char mime[64];
69 const ra8_err_t sniff =
70 mdl_urlname_sniff_file(storage, src, nullptr, nullptr, 0U, mime, sizeof(mime));
71 if (sniff != k_ra8_ok) {
72 return (sniff == k_ra8_err_validation_failed) ? k_ra8_err_not_found : sniff;
73 }
74 if (strcmp(mime, "image/png") == 0) {
75 *out = "image/png";
76 return k_ra8_ok;
77 }
78 if (strcmp(mime, "image/gif") == 0) {
79 *out = "image/gif";
80 return k_ra8_ok;
81 }
82 if (strcmp(mime, "image/webp") == 0) {
83 *out = "image/webp";
84 return k_ra8_ok;
85 }
86 if (strcmp(mime, "image/jpeg") == 0) {
87 *out = "image/jpeg";
88 return k_ra8_ok;
89 }
90 return k_ra8_err_not_found;
91}

107RA8_INTERNAL static void internal_epub_media_type_from_suffix(const char* name, const char** out)
108{
109 const char* dot = strrchr(name, '.');
110 if (dot != nullptr) {
111 if ((strcmp(dot, ".png") == 0) || (strcmp(dot, ".PNG") == 0)) {
112 *out = "image/png";
113 return;
114 }
115 if ((strcmp(dot, ".gif") == 0) || (strcmp(dot, ".GIF") == 0)) {
116 *out = "image/gif";
117 return;
118 }
119 if ((strcmp(dot, ".webp") == 0) || (strcmp(dot, ".WEBP") == 0)) {
120 *out = "image/webp";
121 return;
122 }
123 if ((strcmp(dot, ".bmp") == 0) || (strcmp(dot, ".BMP") == 0)) {
124 *out = "image/bmp";
125 return;
126 }
127 }
128 *out = "image/jpeg";
129}

584RA8_PRIV ra8_err_t priv_mdl_export_epub(mdl_storage_t* storage,
585 const char* dir,
586 char names[][k_name_max],
587 size_t count,
588 mdl_export_output_t* output,
589 const mdl_export_meta_t* meta,
590 mdl_export_workspace_t* ws)
591{
592 mdl_external_cover_t cover;
593 ra8_err_t rc = priv_mdl_export_prepare_cover(storage, meta, names, count, &cover);
594 if (rc != k_ra8_ok) {
595 return rc;
596 }
597 const size_t cap = (size_t)k_epub_base_bytes + (count * (size_t)k_epub_per_page_bytes);
598 char* mani;
599 char* spine;
600 char* nav;
601 char* opf;
602 char* navdoc;
603 rc = internal_epub_carve_workspace(ws, cap, &mani, &spine, &nav, &opf, &navdoc);
604 if (rc != k_ra8_ok) {
605 return rc;
606 }
607 mz_zip_archive zip;
608 mdl_zip_allocator_t zip_alloc;
609 priv_mdl_zip_workspace_bind(&zip, &zip_alloc, ws);
610 zip.m_pWrite = priv_mdl_export_zip_write;
611 zip.m_pIO_opaque = output;
612 const bool zip_open = (mz_zip_writer_init(&zip, 0) != MZ_FALSE);
613 rc = zip_open ? k_ra8_ok : priv_mdl_zip_workspace_error(&zip_alloc);
614 if ((rc == k_ra8_ok) &&
615 (!priv_mdl_epub_add_str(&zip, "mimetype", "application/epub+zip") ||
616 !priv_mdl_epub_add_str(&zip, "META-INF/container.xml", s_epub_container_xml))) {
617 rc = k_ra8_fail;
618 }
619 if (rc == k_ra8_ok) {
620 rc = internal_epub_add_external_cover(&zip, storage, &cover, mani, cap);
621 }
622 mdl_export_meta_t page_meta;
623 const mdl_export_meta_t* page_meta_ptr = internal_epub_page_meta(&cover, meta, &page_meta);
624 for (size_t i = 0U; (rc == k_ra8_ok) && (i < count); ++i) {
625 rc =
626 internal_epub_add_page(storage, &zip, dir, names[i], i, mani, spine, nav, cap, page_meta_ptr);
627 }
628 if (rc == k_ra8_ok) {
629 rc = priv_mdl_epub_add_meta(&zip, mani, spine, nav, count, meta, opf, cap, navdoc, cap);
630 }
631 /* The coordinator aborts this stage unless writer completion and verification pass. */
632 return internal_epub_finish_writer(&zip, zip_open, &zip_alloc, rc, output);
633}
