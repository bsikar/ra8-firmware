92ra8_err_t internal_i3c_i2c_peripheral_open(uint8_t channel, const ra8_i3c_i2c_peripheral_cfg_t* cfg)
93{
94 RA8_CHECK_NULL_PTR(cfg, s_tag, "internal_i3c_i2c_peripheral_open: cfg null");
95 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
96 if (reg == nullptr) {
97 ra8_log_error(s_tag, "internal_i3c_i2c_peripheral_open: channel out of range");
98 return k_ra8_err_invalid_arg;
99 }
100 if (cfg->peripheral_addr_7b > (uint8_t)k_ra8_i3c_i2c_peripheral_max_addr_7b) {
101 return k_ra8_err_invalid_arg;
102 }
103
104 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B", p 445 -- the R_I3C
105 * block is clock-gated OFF at reset; cancel its module-stop (MSTPB4) before the
106 * first responder register write or the peripheral never sees the bus. Omitting
107 * this ran clean in ra8_emulator yet was inert on silicon until #405 modelled the
108 * gate, which surfaced the miss here (the controller path already ungated). The
109 * writes are guarded so a failed ungate does not touch a still-gated block, and
110 * the outcome is returned from the single exit below (no extra return path). */
111 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_i3c);
112 if (mst_err == k_ra8_ok) { /* GCOVR_EXCL_BR_LINE */
113 reg->RSTCTL = 0U;
114 reg->MSDVAD = (uint32_t)cfg->peripheral_addr_7b << k_ra8_i3c_i2c_peripheral_msdvad_shift_dvad;
115 reg->SVCTL = (cfg->general_call != 0U) ? k_ra8_i3c_i2c_peripheral_msk_svctl_gcae : 0U;
116 reg->BCTL = k_ra8_i3c_i2c_msk_bctl_buse;
117 ra8_log_info_val(s_tag, "internal_i3c_i2c_peripheral_open ch", (uint32_t)channel);
118 }
119 return mst_err;
120}

176ra8_err_t internal_i3c_i2c_peripheral_status(uint8_t channel, uint8_t* out_mask)
177{
178 RA8_CHECK_NULL_PTR(out_mask, s_tag, "internal_i3c_i2c_peripheral_status: out_mask null");
179 volatile r_i3c_i2c_regs_t* reg = i3c_i2c_regs(channel);
180 if (reg == nullptr) {
181 return k_ra8_err_invalid_arg;
182 }
183 const uint32_t ntst = reg->NTST;
184 const uint32_t bst = reg->BST;
185 uint8_t mask = 0U;
186 if ((ntst & k_ra8_i3c_i2c_peripheral_msk_ntst_rdbff0) != 0U) {
187 mask |= k_ra8_i3c_i2c_peripheral_status_rx_full;
188 }
189 if ((ntst & k_ra8_i3c_i2c_peripheral_msk_ntst_tdbef0) != 0U) {
190 mask |= k_ra8_i3c_i2c_peripheral_status_tx_empty;
191 }
192 if ((bst & k_ra8_i3c_i2c_msk_bst_spcnddf) != 0U) {
193 mask |= k_ra8_i3c_i2c_peripheral_status_stop;
194 }
195 if ((bst & k_ra8_i3c_i2c_msk_bst_nackdf) != 0U) {
196 mask |= k_ra8_i3c_i2c_peripheral_status_nack;
197 }
198 if (reg->MSDVAD != 0U) {
199 mask |= k_ra8_i3c_i2c_peripheral_status_aas;
200 }
201 *out_mask = mask;
202 return k_ra8_ok;
