139static void lsm6dso_write(void* ctx, uint8_t byte)
140{
141 lsm6dso_state_t* s = (lsm6dso_state_t*)ctx;
142 if (!s->ptr_set) {
143 s->reg_ptr = byte;
144 s->ptr_set = true;
145 return;
146 }
147 if (s->reg_ptr < (uint16_t)k_lsm6dso_reg_file) {
148 s->regs[s->reg_ptr] = byte; /* CTRL config writes land harmlessly. */
149 }
150 s->reg_ptr = (uint16_t)(s->reg_ptr + 1U);
151}

154static uint32_t lsm6dso_read(void* ctx, uint8_t* buf, uint32_t max)
155{
156 lsm6dso_state_t* s = (lsm6dso_state_t*)ctx;

159 s->reg_ptr = (uint16_t)(s->reg_ptr + 1U);
160 }
161 s->reads++;

173static void fuelgauge_put16(max17048_state_t* s, uint8_t reg, uint16_t val)
174{
175 s->regs[reg] = (uint8_t)(val >> (uint16_t)k_battery_byte_shift);
176 s->regs[(uint8_t)(reg + 1U)] = (uint8_t)(val & (uint16_t)k_battery_byte_mask);

180static void fuelgauge_seed(max17048_state_t* s)
181{
182 for (uint32_t i = 0U; i < (uint32_t)k_max17048_reg_file; i++) {
183 s->regs[i] = 0U;
184 }
185 fuelgauge_put16(s, (uint8_t)k_max17048_reg_vcell, (uint16_t)k_battery_vcell_default);
186 /* SOC: high byte is the integer percent, low byte the 1/256 fraction (0). */
187 fuelgauge_put16(s,
188 (uint8_t)k_max17048_reg_soc,
189 (uint16_t)((uint16_t)s_battery.soc_pct << (uint16_t)k_battery_byte_shift));
190 fuelgauge_put16(s, (uint8_t)k_max17048_reg_version, (uint16_t)k_battery_version_val);
191 /* CRATE is signed: positive while charging, negative while discharging. */
192 const int32_t crate =
193 s_battery.charging ? (int32_t)k_battery_crate_mag : -(int32_t)k_battery_crate_mag;
194 fuelgauge_put16(s, (uint8_t)k_max17048_reg_crate, (uint16_t)(int16_t)crate);
195}
