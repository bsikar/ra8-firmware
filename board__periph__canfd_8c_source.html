166static void canfd_apply_global_mode(canfd_inst_t* c, uint32_t gctr)
167{
168 uint32_t gsts = c->reg[canfd_word((uint64_t)k_canfd_off_gsts)];
169 gsts &= ~(uint32_t)(k_gsts_grststs | k_gsts_ghltsts | k_gsts_graminit);
170 const uint32_t mode = gctr & (uint32_t)k_mode_mask;
171 if (mode == (uint32_t)k_mode_reset) {
172 gsts |= (uint32_t)k_gsts_grststs;
173 } else if (mode == (uint32_t)k_mode_halt) {
174 gsts |= (uint32_t)k_gsts_ghltsts;
175 }
176 c->reg[canfd_word((uint64_t)k_canfd_off_gsts)] = gsts;
177}

209static bool canfd_frame_accepted(const canfd_inst_t* c, uint32_t tx_id)
210{
211 bool any_active = false;
212 for (uint32_t slot = 0U; slot < (uint32_t)k_canfd_afl_count; slot++) {
213 const uint64_t ent =
214 (uint64_t)k_canfd_off_afl + ((uint64_t)slot * (uint64_t)k_canfd_afl_stride);
215 const uint32_t mask = c->reg[canfd_word(ent + 4UL)] & (uint32_t)k_canfd_id_ext;
216 if (mask == 0U) {
217 continue; /* unprogrammed slot */
218 }
219 any_active = true;
220 const uint32_t accept_id = c->reg[canfd_word(ent)] & (uint32_t)k_canfd_id_ext;
221 if ((tx_id & mask) == (accept_id & mask)) {
222 return true;
223 }
224 }
225 return !any_active;
226}

263static void canfd_loopback_deliver(uc_engine* uc, uint32_t inst)
264{
265 canfd_inst_t* c = &s_canfd[inst];
266 const uint32_t tm = canfd_word((uint64_t)k_canfd_off_tm0);
267 const uint32_t rf = canfd_word((uint64_t)k_canfd_off_rf0);
268 const uint32_t tx_id = c->reg[tm] & (uint32_t)k_canfd_id_ext;
269 const uint32_t ptr = c->reg[tm + (uint32_t)k_canfd_ptr_word];
270 const uint32_t dlc = (ptr >> (uint32_t)k_canfd_ptr_dlc_shift) & (uint32_t)k_canfd_ptr_dlc_mask;
271 /* The frame is transmitted regardless of the filter; mark TX complete.
272 * CFDTMSTS[0] is a byte at a word-aligned offset, so its low byte carries
273 * TMTRF = 10b ("transmission complete"). */
274 c->reg[canfd_word((uint64_t)k_canfd_off_tmsts0)] = (uint32_t)k_tmsts_done;
275 if (!canfd_frame_accepted(c, tx_id)) {
276 if (board_periph_trace()) {
277 (void)fprintf(stderr,
278 " [trace] CANFD%u loopback: TX id=0x%X filtered (no AFL match)\n",
279 inst,
280 tx_id);
281 }
282 canfd_console_frame(inst, tx_id, dlc, "filtered");
283 return; /* filtered out: transmitted but not received */
284 }
285 for (uint32_t w = 0U; w < (uint32_t)k_canfd_frame_words; w++) {
286 c->reg[rf + w] = c->reg[tm + w];
287 }
288 /* RX FIFO 0 now holds one frame: clear empty, raise the RX flag. */
289 uint32_t rfsts = c->reg[canfd_word((uint64_t)k_canfd_off_rfsts0)];
290 rfsts &= ~(uint32_t)k_rfsts_empty;
291 rfsts |= (uint32_t)k_rfsts_if;
292 c->reg[canfd_word((uint64_t)k_canfd_off_rfsts0)] = rfsts;
293 c->loopbacks++;
294 if (inst == 0U) {
295 board_periph_icu_raise_event(uc, (uint16_t)k_event_can0_rxf);
296 }
297 if (board_periph_trace()) {
298 (void)fprintf(stderr,
299 " [trace] CANFD%u loopback: TX MB0 -> RX FIFO0 (id=0x%X)\n",
300 inst,
301 c->reg[rf] & (uint32_t)k_canfd_id_ext);
302 }
303 canfd_console_frame(inst, tx_id, dlc, "-> RXF0");
304}

307static uint64_t canfd_read(uc_engine* uc, uint64_t addr, unsigned size)
308{
309 (void)uc;
310 (void)size;

323static void canfd_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
324{
325 (void)size;
326 const uint32_t inst = canfd_instance(addr);
327 if (inst >= (uint32_t)k_canfd_count) {
328 return;
329 }
330 const uint64_t off = canfd_offset(inst, addr);
331 if (off >= (uint64_t)k_canfd_span) {
332 return;
333 }
334 canfd_inst_t* c = &s_canfd[inst];
335 /* CFDTMC[0] is a byte register; a TMTR request triggers the loopback. */
336 if (off == (uint64_t)k_canfd_off_tmc0) {
337 c->reg[canfd_word(off)] = (uint32_t)value;
338 if (((uint8_t)value & (uint8_t)k_tmc_txreq) != 0U) {
339 canfd_loopback_deliver(uc, inst);
340 }
341 return;
342 }
343 /* Popping RX FIFO 0 (CFDRFPCTR write) returns it to the empty state. */
344 if (off == (uint64_t)k_canfd_off_rfpctr0) {
345 c->reg[canfd_word(off)] = (uint32_t)value;
346 uint32_t rfsts = c->reg[canfd_word((uint64_t)k_canfd_off_rfsts0)];
347 rfsts |= (uint32_t)k_rfsts_empty;
348 rfsts &= ~(uint32_t)k_rfsts_if;
349 c->reg[canfd_word((uint64_t)k_canfd_off_rfsts0)] = rfsts;
350 return;
351 }
352 c->reg[canfd_word(off)] = (uint32_t)value;
353 if (off == (uint64_t)k_canfd_off_gctr) {
354 canfd_apply_global_mode(c, (uint32_t)value);
355 } else if (off == (uint64_t)k_canfd_off_cnctr) {
356 canfd_apply_channel_mode(c, (uint32_t)value);
357 }
358}

362{
363 for (uint32_t i = 0U; i < (uint32_t)k_canfd_count; i++) {
364 (void)memset(&s_canfd[i], 0, sizeof(s_canfd[i]));
