297[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_canfd_filter_one_round(uint32_t id)
298{
299 ra8_canfd_frame_t tx = {.id = id,
300 .dlc = (uint8_t)k_canfd_filter_dlc,
301 .is_extended = 0U,
302 .is_fd = 0U,
303 .is_brs = 0U,
304 .data = {(uint8_t)id,
305 (uint8_t)(id >> k_canfd_filter_byte_shift),
306 (uint8_t)k_canfd_filter_payload_b2,
307 (uint8_t)k_canfd_filter_payload_b3,
308 (uint8_t)k_canfd_filter_payload_b4,
309 (uint8_t)k_canfd_filter_payload_b5,
310 (uint8_t)k_canfd_filter_payload_b6,
311 (uint8_t)k_canfd_filter_payload_b7}};
312 if (ra8_canfd_transmit((uint8_t)k_canfd_filter_channel, &tx) != k_ra8_ok) {
313 return k_ra8_err_hw_error;
314 }
315 /* Poll for the loopback frame: 500 kbit/s + 8 data bytes is ~240 us
316 * round-trip, so wait up to ~10 ms before declaring no_data. */
317 ra8_canfd_frame_t rx = {};
318 for (uint32_t i = 0U; i < (uint32_t)k_canfd_filter_rx_spin; i++) {
319 if (ra8_canfd_receive((uint8_t)k_canfd_filter_channel, &rx) == k_ra8_ok) {
320 return k_ra8_ok;
321 }
322 }
323 return k_ra8_err_no_data;
324}

326void main(void)
327{
328 internal_canfd_filter_setup_or_halt();
329 ra8_isr_globals_enable();
330
331 while (1) {
332 /* Drain any stale frames left in RX FIFO 0 before this iteration's
333 * three sub-rounds. Without this, a late-arriving frame from the
334 * mask round can land in the FIFO while the no-match round is
335 * spinning on ra8_canfd_receive and report a false "match" for
336 * what should be a rejected frame. */
337 ra8_canfd_frame_t scratch = {};
338 for (uint8_t i = 0U; i < (uint8_t)k_canfd_filter_drain_max; i++) {
339 if (ra8_canfd_receive((uint8_t)k_canfd_filter_channel, &scratch) != k_ra8_ok) {
340 break;
341 }
342 }
343
344 /* Two IDs the filters accept and one they should reject. The
345 * loopback path mirrors every TX, but only filter-matched frames
346 * land in the RX FIFO. */
347 if (internal_canfd_filter_one_round((uint32_t)k_canfd_filter_id_exact) == k_ra8_ok) {
348 (void)ra8_board_led_toggle(k_ra8_board_led1);
349 g_canfd_filter_exact_match += 1U;
350 g_canfd_filter_match += 1U;
351 } else {
352 g_canfd_filter_exact_mismatch += 1U;
353 g_canfd_filter_mismatch += 1U;
354 }
355 if (internal_canfd_filter_one_round((uint32_t)k_canfd_filter_id_mask) == k_ra8_ok) {
356 (void)ra8_board_led_toggle(k_ra8_board_led1);
357 g_canfd_filter_mask_match += 1U;
358 g_canfd_filter_match += 1U;
359 } else {
360 g_canfd_filter_mask_mismatch += 1U;
361 g_canfd_filter_mismatch += 1U;
362 }
363 /* No-match: receive should report no_data; if it doesn't, latch LED2. */
364 if (internal_canfd_filter_one_round((uint32_t)k_canfd_filter_id_nomatch) == k_ra8_ok) {
365 (void)ra8_board_led_toggle(k_ra8_board_led2);
366 g_canfd_filter_nomatch_mismatch += 1U;
367 g_canfd_filter_mismatch += 1U;
368 } else {
369 g_canfd_filter_nomatch_match += 1U;
370 g_canfd_filter_match += 1U;
371 }
372 ra8_delay_ms(k_canfd_filter_period_ms);
373 }
374 internal_canfd_filter_panic_halt();
375}
