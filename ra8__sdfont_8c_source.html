187static ra8_err_t sdfont_bringup_spi(const ra8_sdfont_cfg_t* cfg)
188{
189 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg");
190
191 s_ctx.channel = cfg->spi_channel;
192 s_ctx.pclka_hz = cfg->pclka_hz;
193 s_ctx.cs = cfg->cs;
194
195 ra8_err_t err = ra8_pfs_route_peripheral(cfg->sck, k_ra8_psel_sci_async, "sdfont.sck");
196 if (err != k_ra8_ok) {
197 return err;
198 }
199 err = ra8_pfs_route_peripheral(cfg->cipo, k_ra8_psel_sci_async, "sdfont.cipo");
200 if (err != k_ra8_ok) {
201 return err;
202 }
203 err = ra8_pfs_route_peripheral(cfg->copi, k_ra8_psel_sci_async, "sdfont.copi");
204 if (err != k_ra8_ok) {
205 return err;
206 }
207 err = ra8_gpio_output_init(cfg->cs, k_ra8_level_high);
208 if (err != k_ra8_ok) {
209 return err;
210 }
211 const ra8_sci_spi_cfg_t spi_cfg = {
212 .baud_hz = (uint32_t)k_ra8_sdmmc_spi_clock_init_hz,
213 .pclk_hz = cfg->pclka_hz,
214 .mode = k_ra8_spi_mode_0,
215 .lsb_first = false,
216 };
217 return ra8_sci_spi_init(cfg->spi_channel, &spi_cfg);
218}

283static ra8_err_t sdfont_open_or_provision(const ra8_sdfont_cfg_t* cfg,
284 ra8_fs_mount_t* mount,
285 const char* name,
286 ra8_fs_file_t** out_file,
287 ra8_sdfont_source_t* out_source)
288{
289 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg");
290 RA8_CHECK_NULL_PTR(mount, s_tag, "mount");
291 RA8_CHECK_NULL_PTR(out_file, s_tag, "out_file");
292
293 *out_source = k_ra8_sdfont_source_card;
294 ra8_err_t err = ra8_fs_open(mount, name, k_ra8_fs_mode_read, out_file);
295
296 /* Self-provision only when the file is genuinely absent and a blob is
297 * configured. Kept as separate single-condition guards (not one compound
298 * decision) so each branch is independently covered without MC/DC vectors. */
299 if (err != k_ra8_err_not_found) {
300 return err;
301 }
302 if (cfg->provision_blob == nullptr) {
303 return err; /* provisioning disabled -- propagate k_ra8_err_not_found */
304 }
305 if (cfg->provision_len == 0U) {
306 return err; /* nothing to write -- propagate k_ra8_err_not_found */
307 }
308 err = ra8_fs_write_file(mount, name, cfg->provision_blob, cfg->provision_len);
309 if (err != k_ra8_ok) {
310 return err;
311 }
312 *out_source = k_ra8_sdfont_source_provisioned;
313 return ra8_fs_open(mount, name, k_ra8_fs_mode_read, out_file);
314}

346 (void)ra8_fs_close(file);

391ra8_err_t ra8_sdfont_load(const ra8_sdfont_cfg_t* cfg,
392 uint8_t* buf,
393 uint32_t cap,
394 uint32_t* out_len,
395 ra8_sdfont_source_t* out_source)
396{
397 ra8_err_t err = sdfont_validate(cfg, buf, out_len, cap);
398 if (err != k_ra8_ok) {
399 return err;
400 }
401
402 err = sdfont_bringup_spi(cfg);
403 if (err != k_ra8_ok) {
404 return err;
405 }
406 ra8_fs_mount_t* mount = nullptr;
407 err = sdfont_mount(&mount);
408 if (err != k_ra8_ok) {
409 return err;
410 }
411
412 const char* name = (cfg->filename != nullptr) ? cfg->filename : s_default_name;
413 ra8_fs_file_t* file = nullptr;
414 ra8_sdfont_source_t src = k_ra8_sdfont_source_card;
415 err = sdfont_open_or_provision(cfg, mount, name, &file, &src);
416 if (err == k_ra8_ok) {
417 err = sdfont_read_font(file, buf, cap, out_len);
418 }
419 (void)ra8_fs_unmount(mount);
420 if (err != k_ra8_ok) {
421 return err;
422 }
423
424 if (out_source != nullptr) {
425 *out_source = src;
426 }
427 return k_ra8_ok;
428}
