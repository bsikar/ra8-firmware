249static bool demo_read_verify(ra8_fs_mount_t* mount, const char* path, const char* expected)
250{
251 ra8_fs_file_t* file = nullptr;
252 if (ra8_fs_open(mount, path, k_ra8_fs_mode_read, &file) != k_ra8_ok) {
253 demo_print("[fs] ra8_fs_open(read) failed\r\n");
254 return false;
255 }
256 uint8_t buf[k_demo_file_chunk];
257 uint32_t off = 0U;
258 bool ok = true;
259 uint32_t got = 0U;
260 do {
261 got = 0U;
262 if (ra8_fs_read(file, buf, (uint32_t)k_demo_file_chunk, &got) != k_ra8_ok) {
263 ok = false;
264 break;
265 }
266 if (got > 0U) {
267 if (memcmp(buf, &expected[off], (size_t)got) != 0) {
268 ok = false;
269 }
270 off += got;
271 }
272 } while (got > 0U);
273 (void)ra8_fs_close(file);
274 ok = ok && (off == (uint32_t)strlen(expected));
275 return ok;
276}

310static void demo_file_ops(ra8_fs_mount_t* mount)
311{
312 const uint8_t* payload = (const uint8_t*)s_readme_text;
313 uint32_t len = (uint32_t)strlen(s_readme_text);
314 if (ra8_fs_write_file(mount, s_readme_path, payload, len) != k_ra8_ok) {
315 demo_print("[fs] ra8_fs_write_file failed\r\n");
316 return;
317 }
318 if (ra8_fs_write_file(mount, s_scratch_path, payload, len) != k_ra8_ok) {
319 demo_print("[fs] ra8_fs_write_file failed\r\n");
320 return;
321 }
322 demo_print("[fs] root listing:\r\n");
323 if (ra8_fs_listdir(mount, "/", demo_list_entry, nullptr) != k_ra8_ok) {
324 demo_print("[fs] ra8_fs_listdir failed\r\n");
325 return;
326 }
327 if (!demo_read_verify(mount, s_readme_path, s_readme_text)) {
328 demo_print("[fs] readback mismatch\r\n");
329 return;
330 }
331 demo_print("[fs] readback verified\r\n");
332 if (ra8_fs_unlink(mount, s_scratch_path) != k_ra8_ok) {
333 demo_print("[fs] ra8_fs_unlink failed\r\n");
334 return;
335 }
336 ra8_fs_stat_t st = {};
337 if (ra8_fs_stat(mount, s_scratch_path, &st) != k_ra8_err_not_found) {
338 demo_print("[fs] delete did not remove the file\r\n");
339 return;
340 }
341 demo_print("[fs] ospi FAT roundtrip ok\r\n");
342}

352static void demo_thread_entry(ULONG thread_input)
353{
354 (void)thread_input;
355
356 demo_print("[fs] formatting + opening LevelX on OSPI flash\r\n");
357 if (!demo_lx_open()) {
358 return;
359 }
360 demo_print("[fs] formatting + mounting FAT volume\r\n");
361 ra8_fs_backend_t backend = {};
362 ra8_fs_mount_t* mount = nullptr;
363 if (!demo_fs_format_mount(&backend, &mount)) {
364 (void)lx_nor_flash_close(&s_nor_flash);
365 return;
366 }
367 demo_file_ops(mount);
368 (void)ra8_fs_unmount(mount);
369 (void)lx_nor_flash_close(&s_nor_flash);
370 demo_print("[fs] done\r\n");
371}

386void tx_application_define(void* first_unused_memory)
