85void* ra8_unarch_xz_pool_alloc(uint32_t size)
86{
87 if (s_pool_base == nullptr) {
88 return nullptr;
89 }
90 if (size == 0U) {
91 return nullptr;
92 }
93 const uint32_t align = (uint32_t)k_ra8_unarch_xz_pool_align;
94 if (size > (UINT32_MAX - (align - 1U))) {
95 return nullptr; /* rounding would wrap: refuse */
96 }
97 const uint32_t need = (size + (align - 1U)) & ~(align - 1U);
98 if (need > (s_pool_len - s_pool_off)) {
99 return nullptr; /* arena exhausted: xz-embedded aborts init cleanly */
100 }
101 void* const p = &s_pool_base[s_pool_off];
102 s_pool_off += need;
103 return p;
104}
