31/* Renesas peripheral quirks that the generic sparse model cannot reproduce.
32 *
33 * MRMS frequency latches: the CGC driver (libs/ra8_hal/src/ra8_cgc.c,
34 * internal_wait_mrm_freq) writes ``key | freq_mhz`` to MRCFREQ / MREFREQ and
35 * spins until the register reads back == freq_mhz. Real silicon validates the
36 * upper key byte then strips it, so the readback is the bare frequency. The
37 * generic model reflects the full written word (key still in bits[31:24]), so
38 * the readback never equals freq and the poll runs to its 0x40000 timeout ->
39 * lcd_panic_halt. Model the hardware: on readback of these two registers,
40 * return the stored value with the key byte masked off. */
41typedef enum : uint64_t {
42 k_mrms_mrcfreq = 0x4013C004UL,
43 k_mrms_mrefreq = 0x4013C008UL,
44 k_mrms_freq_mask = 0x00FFFFFFUL,
45} mrms_quirk_t;

77static void bgc_track(uint32_t value)
78{
79 s_bgc_writes++;
80 for (uint32_t i = 0U; i < s_bgc_distinct_n; i++) {
81 if (s_bgc_distinct[i] == value) {
82 return;
83 }
84 }
85 if (s_bgc_distinct_n < (uint32_t)k_bgc_track_max) {
86 s_bgc_distinct[s_bgc_distinct_n++] = value;
87 }

91static int mmio_index(uint64_t addr)
92{
93 if ((s_mmio_cache >= 0) && (s_mmio_addr[s_mmio_cache] == addr)) {
94 return s_mmio_cache;
95 }
96 for (uint32_t i = 0U; i < s_mmio_n; i++) {
97 if (s_mmio_addr[i] == addr) {
98 s_mmio_cache = (int)i;
99 return (int)i;
100 }
101 }
102 if (s_mmio_n < (uint32_t)k_mmio_slots) {
103 s_mmio_addr[s_mmio_n] = addr;
104 s_mmio_cache = (int)s_mmio_n;
105 return (int)(s_mmio_n++);
106 }
107 return -1;
108}

110uint64_t mmio_read(uc_engine* uc, uint64_t offset, unsigned size, void* user)
111{
112 (void)user;
113 s_mmio_reads++;
114 /* A modelled peripheral block answers first; the sparse fallback below is
115 * only reached for addresses no block in board_periph owns. */
116 bool handled = false;
117 const uint64_t modeled = board_periph_read(uc, (uint64_t)k_periph_base + offset, size, &handled);
118 if (handled) {
119 return modeled;
120 }
121 const int idx = mmio_index((uint64_t)k_periph_base + offset);
122 if (idx >= 0) {
123 s_mmio_rcount[idx]++;
124 if (idx == s_mmio_run_slot) {
125 s_mmio_run++;
126 } else {
127 s_mmio_run_slot = idx;
128 s_mmio_run = 1U;
129 }
130 /* Reflect a written control value until a spin-poll forces it to settle. */
131 if (s_mmio_written[idx] && (s_mmio_run <= (uint32_t)k_mmio_settle)) {
132 const uint64_t addr = (uint64_t)k_periph_base + offset;
133 /* MRMS frequency latches strip the write key byte on readback so the
134 * driver's "wait until reg == freq" poll completes (see mrms_quirk_t). */
135 if ((addr == (uint64_t)k_mrms_mrcfreq) || (addr == (uint64_t)k_mrms_mrefreq)) {
136 return (uint64_t)(s_mmio_val[idx] & (uint32_t)k_mrms_freq_mask);
137 }
138 return (uint64_t)s_mmio_val[idx];
139 }
140 }
141 s_mmio_toggle ^= (uint32_t)k_u32_all_ones;
142 return (uint64_t)s_mmio_toggle;
143}

168void mmio_write(uc_engine* uc, uint64_t offset, unsigned size, uint64_t value, void* user)
169{
170 (void)user;
171 s_mmio_writes++;
172 const uint64_t mmio_abs = (uint64_t)k_periph_base + offset;
173 /* Dual-core release: the firmware stages cpu1's vector table in
174 * CPU1INITVTOR, then asserts CPU1ACTCSR.ACTREQ to start cpu1. The watcher
175 * captures both so the run loop can boot the second engine (see emu_cpu1). */
176 emu_cpu1_notify_mmio_write(mmio_abs, value);
177 if (mmio_abs == (uint64_t)k_glcdc_bg_bgc) {
178 bgc_track((uint32_t)value);
179 }
180 /* A modelled peripheral block consumes the write first (so GPIO latches,
181 * timer control, and ICU event links take real effect); the sparse fallback
182 * still records the write for the MMIO table and unmodelled blocks. */
183 bool handled = false;
184 board_periph_write(uc, (uint64_t)k_periph_base + offset, size, value, &handled);
185 if (handled) {
186 return;
187 }
188 const int idx = mmio_index((uint64_t)k_periph_base + offset);
189 if (idx >= 0) {
190 s_mmio_wcount[idx]++;
191 s_mmio_val[idx] = (uint32_t)value;
192 s_mmio_written[idx] = true;
193 if (idx == s_mmio_run_slot) {
194 s_mmio_run++; /* same-addr read-modify-write spin accumulates toward settle */
195 } else {
196 s_mmio_run_slot = idx; /* new register -> following reads should see its value */
197 s_mmio_run = 1U;
198 }
199 }
200}
