1#!/usr/bin/env python3
2# SPDX-License-Identifier: MIT

4"""Gate: every first-party shell script carries an ``#!/usr/bin/env <interp>``.
5
6Why
7---
8A hardcoded interpreter path is a portability assertion the tree cannot keep.
9``#!/bin/bash`` is wrong on any host whose bash is not at that path -- NixOS,
10a Homebrew bash 5 on macOS (``/bin/bash`` there is the 3.2 that ships with the
11OS and lacks ``mapfile``), FreeBSD, and a container built ``FROM scratch`` with
12a busybox layout. ``#!/usr/bin/env bash`` resolves through PATH instead, which
13is the same discipline ``scripts/ci/lib/tool_env.sh`` applies to every other
14tool this project runs: decide the interpreter by resolution, not by a literal
15path baked into 15 files.
16
17Two questions, two scopes
18-------------------------
19* **Presence** -- is there a shebang at all? Required of every first-party
20 *shell script* (``*.sh`` / ``*.bash``, and the extensionless files whose own
21 shebang already declares them shell -- the ``scripts/git/*`` hooks). The
22 scope is :func:`lint_targets.files_for`'s ``shell`` set, the one definition
23 this repo shares. A ``.sh`` with no shebang is a real gap even when the file
24 is only ever *sourced*: the shebang is what tells an editor and ShellCheck
25 the dialect, and the moment someone runs it directly the interpreter is
26 whatever the parent shell happens to be.
27* **Form** -- is the shebang, if present, the env form? Asked of *any*
28 in-scope file that opens with ``#!`` (a Python or Perl helper as much as a
29 shell one), because a hardcoded path is wrong wherever it appears.
30
31Four defect shapes are rejected, all of which this tree has carried:
32
33* **missing** -- a first-party shell script with no shebang on line 1. The
34 gate bodies under ``scripts/ci/gates/`` and the sourced libs under
35 ``scripts/ci/lib/`` were exactly this until they were normalised.
36* **non-env** -- ``#!/bin/bash``, ``#!/bin/sh``, ``#!/usr/bin/python3``.
37* **malformed** -- ``# !/bin/bash``. A space between ``#`` and ``!`` means the
38 kernel never sees a shebang at all; the file is a comment followed by code,
39 and executing it directly falls back to the caller's shell (or fails). It
40 reads as a shebang to a human and is not one, which is why it survives.
41* **argument-bearing env** -- ``#!/usr/bin/env bash -x``. Linux passes the
42 whole remainder of the shebang line as ONE argument, so the interpreter is
43 looked up as the literal name ``bash -x`` and the exec fails. Put the flag
44 in the script body (``set -x``) instead.
45
46Scope is derived, not hardcoded
47-------------------------------
48``git ls-files --cached --others --exclude-standard`` enumerates every tracked
49path AND every untracked-but-not-ignored one, and each is judged on its own
50first line. So a script cannot escape by living in a directory nobody listed,
51by having no extension, or by not being committed yet -- which is the moment a
52wrong shebang is actually introduced, and the moment the pre-commit hook asks.
53Vendored SOUP, generated font tables, the ThreadX port and build output are
54excluded, matching :mod:`check_shell`'s scope.
55
56The presence rule does NOT require the executable bit: a sourced-only library
57correctly carries a shebang and is correctly not executable, because nothing
58execs it. The shebang is a dialect declaration here, not a promise the file
59is a program.
60
61Non-vacuity
62-----------
63``--selftest`` asserts both directions over crafted fixtures -- every rejected
64shape must fire and every accepted shape must stay silent -- and the scan
65enforces a floor on both how many shebang-bearing files and how many shell
66scripts it found, because a scope that collapses reports a clean tree having
67examined almost nothing.
68
69Run::
70
71 check_shebangs.py # gate (fail on any finding)
72 check_shebangs.py --selftest # prove the rules fire and stay quiet
73
74Exit 0 if clean, 1 on findings or a failing selftest, 2 on a broken scan.
75"""

100# Floor on shebang-bearing first-party files. The tree has 270 today. This is
101# not kept in step file by file -- it is a trip-wire for a scope that collapses
102# wholesale (a broken `git ls-files`, a runaway exclusion). Lower it
103# deliberately, with a reason, if first-party scripting genuinely shrinks.
104SHEBANG_FLOOR = 150
105
106# Floor on first-party shell scripts subject to the PRESENCE rule. Measured
107# 2026-08-02: 128. Same trip-wire, for the other scope: if the `shell` set
108# collapses, the presence rule silently stops requiring anything and the gate
109# reports a clean tree having demanded a shebang of almost nothing.
110SHELL_FLOOR = 100

112# The one accepted form: `#!/usr/bin/env` plus exactly one interpreter token.
113# The trailing `$` is load-bearing -- it is what rejects `env bash -x`.
114_GOOD_RE = re.compile(r"^#!/usr/bin/env [A-Za-z_][A-Za-z0-9_.+-]*$")
115
116# A near-shebang: `#` then whitespace then `!`. Never a real shebang.
117_MALFORMED_RE = re.compile(r"^#\s+!\s*\S")

119# Bytes read from each file: enough for any shebang, cheap over 13k files.

123def _git_ls() -> list[str]:
124 """Return every tracked-or-untracked-but-not-ignored path, repo-relative.
125
126 ``--others --exclude-standard`` is what puts a BRAND NEW script in scope.
127 Tracked-only enumeration cannot see the file a developer just wrote, which
128 is precisely the moment a wrong shebang gets introduced and the moment the
129 pre-commit hook is asked about it. Ignored paths stay out, so locally
130 present vendor drops and build trees never enter the gate.
131
132 Returns:
133 Paths as reported by ``git ls-files``.
134 """
135 proc = subprocess.run(
136 ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard"], # noqa: S607
137 cwd=REPO_ROOT,
138 capture_output=True,
139 text=True,
140 check=False,
141 )
142 if proc.returncode != 0:
143 sys.stderr.write(proc.stderr)
144 sys.stderr.write("check_shebangs.py: FATAL -- `git ls-files` failed\n")
145 sys.exit(EXIT_CONFIG)
146 return [p for p in proc.stdout.split("\0") if p]

163def _first_line(path: Path) -> str | None:
164 """Return the first line of `path`, or None when it cannot be read as text.
165
166 Args:
167 path: File to probe.
168
169 Returns:
170 The first line without its newline, or None for unreadable/binary data.
171 """
172 try:
173 with path.open("rb") as handle:
174 raw = handle.readline(_PROBE_BYTES)
175 except OSError:
176 return None
177 if b"\0" in raw:
178 return None
179 return raw.decode("utf-8", errors="replace").rstrip("\r\n")

182def classify(line: str, *, require_shebang: bool) -> str | None:
183 """Judge one first line, returning a violation reason or None when clean.
184
185 Args:
186 line: The file's first line, newline stripped.
187 require_shebang: True for a first-party shell script, where the absence
188 of a shebang is itself a finding. False elsewhere, where a file
189 with no ``#!`` is simply not a script and is left alone.
190
191 Returns:
192 A short reason when the line is a bad (or, for a shell script, missing)
193 shebang, else None. A malformed near-shebang fires regardless of
194 `require_shebang`, because it is never a legitimate first line.
195 """
196 if _MALFORMED_RE.match(line):
197 return "malformed: a space between '#' and '!' is not a shebang at all"
198 if not line.startswith("#!"):
199 if require_shebang:
200 return "no shebang: a shell script must start with `#!/usr/bin/env <interp>`"
201 return None
202 if _GOOD_RE.match(line):
203 return None
204 if line.startswith("#!/usr/bin/env "):
205 return (
206 "env shebang with arguments: Linux passes them as one token; use `set -x` in the body"
207 )
208 return "hardcoded interpreter path: use `#!/usr/bin/env <interp>`"

211def scan() -> tuple[list[tuple[str, str, str]], int, int]:
212 """Judge every in-scope tracked file.
213
214 The presence rule applies only to the first-party ``shell`` set from
215 :func:`lint_targets.files_for` -- the single shared definition of "this is
216 a shell script". The form rule applies to any in-scope file that opens with
217 a ``#!``.
218
219 Returns:
220 A ``(findings, shebang_count, shell_count)`` triple, where each finding
221 is ``(path, line, reason)``.
222 """
223 shell_set = set(files_for(("shell",))["shell"])
224 findings: list[tuple[str, str, str]] = []
225 shebangs = 0
226 shell_seen = 0
227 for rel in _git_ls():
228 if not _in_scope(rel):
229 continue
230 require = rel in shell_set
231 if require:
232 shell_seen += 1
233 line = _first_line(REPO_ROOT / rel)
234 if line is None:
235 if require:
236 findings.append((rel, "<unreadable>", "shell script with an unreadable first line"))
237 continue
238 if line.startswith("#!"):
239 shebangs += 1
240 reason = classify(line, require_shebang=require)
241 if reason is not None:
242 findings.append((rel, line, reason))
243 return findings, shebangs, shell_seen

246# ---------------------------------------------------------------------------
247# Selftest -- both directions, over crafted first lines. Nothing is written
248# into the tree: a deliberately bad fixture stored as a real file would be
249# picked up by this gate's own scan and fail it.
250# ---------------------------------------------------------------------------
251
252# Each case is (label, first_line, require_shebang): the third field is the
253# scope flag the scan would pass, True where the line stands for a shell script.
254MUST_FIRE: tuple[tuple[str, str, bool], ...] = (
255 # form -- a bad shebang fires whether or not a shebang was required
256 ("plain bash path", "#!/bin/bash", False),
257 ("plain sh path", "#!/bin/sh", False),
258 ("usr-bin python path", "#!/usr/bin/python3", False),
259 ("space after the bang", "#! /bin/bash", False),
260 ("space before the bang", "# !/bin/bash", False),
261 ("env with a flag", "#!/usr/bin/env bash -x", False),
262 ("env with nothing after it", "#!/usr/bin/env", False),
263 ("env with a pathful interpreter", "#!/usr/bin/env /bin/bash", False),
264 # presence -- a shell script with no shebang at all
265 ("shell script missing its shebang", "# shellcheck shell=bash", True),
266 ("shell script that opens with code", "echo hi", True),
267 ("shell script with an empty first line", "", True),
268)
269
270MUST_STAY_QUIET: tuple[tuple[str, str, bool], ...] = (
271 ("env bash", "#!/usr/bin/env bash", False),
272 ("env sh", "#!/usr/bin/env sh", False),
273 ("env python3", "#!/usr/bin/env python3", False),
274 ("env perl", "#!/usr/bin/env perl", False),
275 ("env bash where a shebang was required", "#!/usr/bin/env bash", True),
276 ("a non-script file with no shebang", "# SPDX-License-Identifier: MIT", False),
277 ("a C comment", "// SPDX-License-Identifier: MIT", False),
278 ("a markdown heading", "# Title", False),
279 ("an empty first line off the shell scope", "", False),
280)

289def _selftest_scope() -> list[str]:
290 """Assert the scope helper excludes vendored trees and keeps first-party.
291
292 Returns:
293 Failure descriptions; empty when the scope rules behave.
294 """
295 excluded = [
296 f" scope: {rel} should be excluded but is in scope"
297 for rel in SCOPE_MUST_EXCLUDE
298 if _in_scope(rel)
299 ]
300 included = [
301 f" scope: {rel} should be in scope but is excluded"
302 for rel in SCOPE_MUST_INCLUDE
303 if not _in_scope(rel)
304 ]
305 return excluded + included

308def selftest() -> int:
309 """Prove every rejected shape fires and every accepted shape stays silent.
310
311 Returns:
312 EXIT_OK when the classifier and the scope rules behave in both
313 directions, EXIT_FAIL otherwise.
314 """
315 failures = [
316 f" must-fire: {label} was accepted: {line!r}"
317 for label, line, req in MUST_FIRE
318 if classify(line, require_shebang=req) is None
319 ]
320 failures += [
321 f" must-stay-quiet: {label} was rejected ({classify(line, require_shebang=req)}): {line!r}"
322 for label, line, req in MUST_STAY_QUIET
323 if classify(line, require_shebang=req) is not None
324 ]
325 failures += _selftest_scope()
326
327 if failures:
328 sys.stderr.write("check_shebangs.py --selftest: FAILED\n\n")
329 sys.stderr.write("\n".join(failures) + "\n")
330 sys.stderr.write("\nThe rules do not judge shebangs as claimed.\n")
331 return EXIT_FAIL
332
333 total = len(MUST_FIRE) + len(MUST_STAY_QUIET)
334 print(
335 f"check_shebangs.py --selftest: OK ({total} cases: "
336 f"{len(MUST_FIRE)} must fire, {len(MUST_STAY_QUIET)} must stay quiet; scope both ways)."
337 )
338 return EXIT_OK

341def main(argv: list[str]) -> int:
342 """Run the selftest or scan the tree for missing / non-env shebangs.
343
344 Returns:
345 EXIT_OK when clean, EXIT_FAIL on any finding or a failing selftest,
346 EXIT_CONFIG when either scan scope collapsed below its non-vacuity floor.
347 """
348 if "--selftest" in argv[1:]:
349 return selftest()
350
351 findings, shebangs, shell_seen = scan()
352
353 if shebangs < SHEBANG_FLOOR or shell_seen < SHELL_FLOOR:
354 sys.stderr.write(
355 f"check_shebangs.py: FATAL -- {shebangs} shebang(s) (floor {SHEBANG_FLOOR}) "
356 f"and {shell_seen} shell script(s) (floor {SHELL_FLOOR}) in scope.\n"
357 " A collapsed scope reports a clean tree because it checked nothing.\n"
358 )
359 return EXIT_CONFIG
360
361 if not findings:
362 print(
363 f"check_shebangs.py: {shell_seen} shell script(s) all carry a shebang; "
364 f"{shebangs} shebang(s) all use the env form."
365 )
366 return EXIT_OK
367
368 sys.stderr.write(f"check_shebangs.py: {len(findings)} bad shebang(s):\n")
369 for rel, line, reason in findings:
370 sys.stderr.write(f" {rel}: {line!r}\n {reason}\n")
371 sys.stderr.write(
372 "\nEvery shell script needs `#!/usr/bin/env bash` (or sh) on line 1. The\n"
373 "interpreter must be a bare name resolved through PATH, with no arguments\n"
374 "on the shebang line.\n"
375 )
376 return EXIT_FAIL
