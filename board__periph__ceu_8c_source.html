144static bool ceu_do_capture(uc_engine* uc)
145{
146 uint32_t line_bytes = ceu_line_bytes();
147 uint32_t lines = ceu_line_count();
148 if ((line_bytes == 0U) || (lines == 0U)) {
149 return false; /* No geometry programmed -> no capture. */
150 }
151 if (line_bytes > (uint32_t)k_ceu_max_line) {
152 line_bytes = (uint32_t)k_ceu_max_line;
153 }
154 if (lines > (uint32_t)k_ceu_max_lines) {
155 lines = (uint32_t)k_ceu_max_lines;
156 }
157 const uint32_t dst = s_ceu.reg[ceu_word(k_ceu_off_cdayr)];
158 if (dst == 0U) {
159 return false; /* No destination buffer -> no capture. */
160 }
161 for (uint32_t y = 0U; y < lines; y++) {
162 for (uint32_t x = 0U; x < line_bytes; x++) {
163 s_ceu_line[x] = (uint8_t)((x + y) & (uint32_t)k_ceu_pattern_mask);
164 }
165 (void)uc_mem_write(uc,
166 (uint64_t)dst + ((uint64_t)y * (uint64_t)line_bytes),
167 s_ceu_line,
168 (size_t)line_bytes);
169 }
170 const uint32_t total = line_bytes * lines;
171 s_ceu.reg[ceu_word(k_ceu_off_cetcr)] |= (uint32_t)k_ceu_cetcr_cpe;
172 s_ceu.reg[ceu_word(k_ceu_off_cdssr)] = total;
173 s_ceu.reg[ceu_word(k_ceu_off_capsr)] &= ~(uint32_t)k_ceu_capsr_ce;
174 s_ceu.frames++;
175 s_ceu.last_bytes = total;
176 s_ceu.last_w = line_bytes;
177 s_ceu.last_h = lines;
178
179 char ln[k_ceu_console_line_cap];
180 (void)snprintf(ln,
181 sizeof(ln),
182 "CEU frame %ux%u -> %u bytes",
183 (unsigned)line_bytes,
184 (unsigned)lines,
185 (unsigned)total);
186 board_console_push(k_board_console_ch_dma, ln);
187 if (board_periph_trace()) {
188 (void)fprintf(stderr, " [trace] CEU capture %ux%u -> 0x%08X\n", line_bytes, lines, dst);
189 }
190 return true;
191}

194static uint64_t ceu_read(uc_engine* uc, uint64_t addr, unsigned size)
195{
196 (void)uc;
197 (void)size;
198 const uint64_t off = addr - (uint64_t)k_ceu_base;
199 if (off >= (uint64_t)k_ceu_span) {
200 return 0U;
201 }
202 if (off == (uint64_t)k_ceu_off_cstsr) {
203 return 0U; /* Capture is instantaneous: CPTON always reads idle. */
204 }
205 if (off == (uint64_t)k_ceu_off_capsr) {
206 /* CPKIL self-clears (reset is instantaneous in the model). */
207 return (uint64_t)(s_ceu.reg[ceu_word(off)] & ~(uint32_t)k_ceu_capsr_cpkil);
208 }
209 return (uint64_t)s_ceu.reg[ceu_word(off)];
210}

213static void ceu_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
214{
215 (void)size;
216 const uint64_t off = addr - (uint64_t)k_ceu_base;
217 if (off >= (uint64_t)k_ceu_span) {
218 return;
219 }
220 s_ceu.reg[ceu_word(off)] = (uint32_t)value;
221 if (off != (uint64_t)k_ceu_off_capsr) {
222 return;
223 }
224 if (((uint32_t)value & (uint32_t)k_ceu_capsr_cpkil) != 0U) {
225 /* Software reset: drop pending events and CE; CPKIL self-clears. */
226 s_ceu.reg[ceu_word(k_ceu_off_cetcr)] = 0U;
227 s_ceu.reg[ceu_word(k_ceu_off_capsr)] =
228 (uint32_t)value & ~((uint32_t)k_ceu_capsr_cpkil | (uint32_t)k_ceu_capsr_ce);
229 return;
230 }
231 if (((uint32_t)value & (uint32_t)k_ceu_capsr_ce) != 0U) {
232 (void)ceu_do_capture(uc); /* Arm -> synthesise a frame + latch CETCR.CPE. */
233 }
234}
