4"""Gate: every insecure placeholder-crypto body shall be guarded fail-closed.
5
6Several secure-side translation units ship an INSECURE placeholder body that is
7only safe under an off-target build or an explicitly-declared insecure dev/eval
8image (issue #180):
9
10 - src/secure_app/secure_trng.c deterministic xorshift PRNG as a "TRNG"
11 - src/secure_app/key_vault.c plain secure-SRAM key store (no HW vault)
12 - libs/ra8_hal/src/ra8_rsip_key_injection.c non-cryptographic xorshift key-wrap
13 - libs/ra8_hal/src/ra8_rsip_ecc.c fiction-opcode ECDSA / ECDH / Ed25519 asym
14 - libs/ra8_hal/src/ra8_rsip_cipher.c fiction-opcode AES / ChaCha / key-install
15 - libs/ra8_hal/src/ra8_rsip_rsa.c fiction-opcode RSA sign / verify / enc / dec
16 - libs/ra8_hal/src/ra8_rsip_asym.c fiction-opcode hash / HMAC / key vault / wrap / KDF
17 - libs/ra8_hal/src/ra8_rsip_devsec.c fiction-register lifecycle / debug / tamper / DPA
18
19Each such body MUST be wrapped in the guard::
20
21 #if defined(RA8_INSECURE_STUB_CRYPTO) || defined(RA8_OFF_TARGET)
22 <insecure placeholder body>
23 #else
24 <fail-closed: every entry point returns a hard error, never k_ra8_ok>
25 #endif
26
27so a real production/HIL image that sets NEITHER flag compiles the fail-closed
28#else and cannot silently ship the stub. This gate is the compile-time-of-CI
29guarantee the audit asked for: it FAILS if, for any listed TU,
30
31 1. the guard `#if defined(RA8_INSECURE_STUB_CRYPTO) || defined(RA8_OFF_TARGET)`
32 is absent, or has no matching #else / #endif;
33 2. the #else branch is not fail-closed (no `#error` and no `k_ra8_err_` return);
34 3. the TU's insecure signature token escapes the guarded region (appears before
35 the guard, in the #else, or after the #endif) -- i.e. an insecure body that
36 is not actually behind the guard.
37
38There is no allowlist: either guard the insecure body fail-closed, or delete it
39in favour of a real backend.
40
41Run::
42
43 check_stub_crypto_guarded.py
44
45Exit status: 0 if every stub TU is guarded fail-closed, 1 otherwise.
46"""
47

57# Each stub TU -> a distinctive token that appears ONLY in its insecure body
58# (never in the fail-closed #else nor in surrounding prose). The token must live
59# inside the guarded #if region; if it escapes, the insecure body is unguarded.
60STUB_TUS = {
61 "src/secure_app/secure_trng.c": "internal_xorshift64",
62 "src/secure_app/key_vault.c": "s_vault",
63 "libs/ra8_hal/src/ra8_rsip_key_injection.c": "ki_compute_mac",
64 "libs/ra8_hal/src/ra8_rsip_ecc.c": "k_ra8_rsip_asym_op_eddsa_sign",
65 "libs/ra8_hal/src/ra8_rsip_cipher.c": "internal_sym_run",
66 "libs/ra8_hal/src/ra8_rsip_rsa.c": "internal_rsa_dispatch",
67 "libs/ra8_hal/src/ra8_rsip_asym.c": "internal_hash_pull_digest",
68 "libs/ra8_hal/src/ra8_rsip_devsec.c": "k_ra8_rsip_off_life_state",
69}
70
71_RE_IF = re.compile(r"^\s*#\s*if(n?def)?\b")
72_RE_ELSE = re.compile(r"^\s*#\s*else\b")
73_RE_ENDIF = re.compile(r"^\s*#\s*endif\b")
74_RE_ERROR = re.compile(r"^\s*#\s*error\b")

77def is_guard_open(line: str) -> bool:
78 """Whether ``line`` is the stub-crypto guard opener (either flag order)."""
79 return bool(
80 re.match(r"^\s*#\s*if\b", line)
81 and "RA8_INSECURE_STUB_CRYPTO" in line
82 and "RA8_OFF_TARGET" in line
83 )

86def find_guard_region(lines: list[str]) -> tuple[int, int, int] | None:
87 """Locate the guard's (#if, #else, #endif) 0-based line indices.
88
89 Returns ``(if_idx, else_idx, endif_idx)`` or ``None`` when the guard opener
90 is absent or its matching #else / #endif cannot be resolved. Nesting of
91 inner #if/#endif is tracked so a nested conditional does not confuse the
92 match.
93 """
94 if_idx = next((i for i, ln in enumerate(lines) if is_guard_open(ln)), None)
95 if if_idx is None:
96 return None
97 depth = 1
98 else_idx = None
99 for i in range(if_idx + 1, len(lines)):
100 ln = lines[i]
101 if _RE_IF.match(ln):
102 depth += 1
103 elif _RE_ENDIF.match(ln):
104 depth -= 1
105 if depth == 0:
106 return (if_idx, else_idx, i) if else_idx is not None else None
107 elif _RE_ELSE.match(ln) and depth == 1:
108 else_idx = i
109 return None

112def check_file(rel: str, token: str) -> list[str]:
113 """Return a list of violation strings for one stub TU (empty when clean)."""
114 path = REPO_ROOT / rel
115 if not path.is_file():
116 return [f"{rel}: file not found (expected an insecure stub TU here)"]
117 lines = path.read_text(encoding="utf-8").splitlines()
118
119 region = find_guard_region(lines)
120 if region is None:
121 return [
122 f"{rel}: missing the stub-crypto guard "
123 f"'#if defined(RA8_INSECURE_STUB_CRYPTO) || defined(RA8_OFF_TARGET)' "
124 f"with a matching #else / #endif"
125 ]
126 if_idx, else_idx, endif_idx = region
127
128 problems: list[str] = []
129
130 # (2) the #else branch must be fail-closed: a compile-time #error or a hard
131 # k_ra8_err_ return (never k_ra8_ok).
132 else_body = lines[else_idx + 1 : endif_idx]
133 fail_closed = any(_RE_ERROR.match(ln) or "k_ra8_err_" in ln for ln in else_body)
134 if not fail_closed:
135 problems.append(
136 f"{rel}: the #else branch is not fail-closed "
137 f"(needs a #error or a k_ra8_err_* hard return, not k_ra8_ok)"
138 )
139
140 # (3) the insecure signature token must live INSIDE the guarded #if region
141 # and never escape it.
142 hits = [i for i, ln in enumerate(lines) if token in ln]
143 inside = [i for i in hits if if_idx < i < else_idx]
144 escaped = [i for i in hits if not (if_idx < i < else_idx)]
145 if not inside:
146 problems.append(
147 f"{rel}: insecure signature '{token}' not found inside the guarded "
148 f"#if region (is the insecure body still present and guarded?)"
149 )
150 if escaped:
151 where = ", ".join(f"line {i + 1}" for i in escaped)
152 problems.append(
153 f"{rel}: insecure signature '{token}' appears OUTSIDE the guard "
154 f"({where}) -- the insecure body must be fully inside the "
155 f"#if defined(RA8_INSECURE_STUB_CRYPTO) || defined(RA8_OFF_TARGET) block"
156 )
157
158 return problems

161def main() -> int:
162 """Verify every placeholder-crypto TU is guarded fail-closed.
163
164 These TUs hold deliberately insecure stand-ins for real crypto. The guard
165 must make the ``#else`` half -- the branch taken when the real
166 implementation is absent -- refuse to build or fail closed, so a
167 misconfiguration cannot silently ship the placeholder as if it were the
168 algorithm.
169
170 Returns 1 listing each unguarded TU, 0 when all are fail-closed.
171 """
172 all_problems: list[str] = []
173 for rel, token in STUB_TUS.items():
174 all_problems.extend(check_file(rel, token))
175
176 if all_problems:
177 print("check_stub_crypto_guarded.py: insecure placeholder crypto not guarded fail-closed:")
178 for p in all_problems:
179 print(f" {p}")
180 print("Fix each at the root -- wrap the insecure body in")
181 print(" #if defined(RA8_INSECURE_STUB_CRYPTO) || defined(RA8_OFF_TARGET)")
182 print("and make the #else fail closed (return k_ra8_err_* / #error), or replace")
183 print("the placeholder with a real crypto backend.")
184 return 1
185
186 count = len(STUB_TUS)
187 print(f"check_stub_crypto_guarded.py: PASS -- {count} stub crypto TU(s) guarded fail-closed.")
188 return 0
