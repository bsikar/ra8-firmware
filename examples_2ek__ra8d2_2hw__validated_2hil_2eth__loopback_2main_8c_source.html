102[[nodiscard]] static ra8_err_t eth_loopback_run_once(ra8_etha_port_stats_t* out)
103{
104 const ra8_etha_config_t cfg = {
105 .initial_mode = k_ra8_etha_opc_config,
106 .eaeie0_mask = 0U,
107 .eaeie1_mask = 0U,
108 .eaeie2_mask = 0U,
109 };
110 ra8_err_t err = ra8_etha_init(k_ra8_etha_port_0, &cfg);
111 if (err != k_ra8_ok) {
112 return err;
113 }
114 err = ra8_etha_descriptor_ring_init(k_ra8_etha_port_0,
115 (uint16_t)k_eth_loopback_ring_tx,
116 (uint16_t)k_eth_loopback_ring_rx,
117 (uint16_t)k_eth_loopback_buffer_size);
118 if (err != k_ra8_ok) {
119 return err;
120 }
121 err = ra8_etha_set_mode(k_ra8_etha_port_0, k_ra8_etha_opc_operation);
122 if (err != k_ra8_ok) {
123 return err;
124 }
125 for (uint8_t i = 0U; i < (uint8_t)k_eth_loopback_burst_count; i++) {
126 err = ra8_etha_account_traffic(k_ra8_etha_port_0, 1U, 0U, 1U, 0U, 0U);
127 if (err != k_ra8_ok) {
128 return err;
129 }
130 }
131 return ra8_etha_get_stats(k_ra8_etha_port_0, out);
132}

136int32_t main(void)
137{
138 eth_loopback_setup_or_halt();
139 ra8_isr_globals_enable();

141 /* Boot banner -- emit before the loopback runs so the HIL host can
142 * confirm the firmware booted even when the ETHA loopback fails for
143 * board-specific reasons (PHY not negotiated, missing pull-ups, etc.). */
144 (void)ra8_board_uart_console_write(k_eth_loopback_boot_msg,
145 (size_t)(sizeof(k_eth_loopback_boot_msg) - 1U));
146
147 ra8_etha_port_stats_t stats = {};
148 if (eth_loopback_run_once(&stats) != k_ra8_ok) {
149 eth_loopback_panic_halt();
150 }
151 if (ra8_board_uart_console_write(k_eth_loopback_log_msg,
152 (size_t)(sizeof(k_eth_loopback_log_msg) - 1U)) != k_ra8_ok) {
153 eth_loopback_panic_halt();
154 }
155 (void)ra8_etha_deinit(k_ra8_etha_port_0);
156
157 while (1) {
158 __asm__ volatile("wfi");
159 }
160 return 0;
161}
