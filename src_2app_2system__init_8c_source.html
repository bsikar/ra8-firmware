403[[maybe_unused]] static void internal_mpu_init(void)
404{
405 enum : uint32_t {
406 k_ra8_mpu_ctrl_enable = 1UL << 0,
407 k_ra8_mpu_ctrl_privdefena = 1UL << 2,
408 k_ra8_mair0_default = 0x000404FFUL,
409 };

411 /* MAIR0: idx 0 = WB/WA, idx 1 = non-cacheable, idx 2 = device-nGnRE. */
412 internal_write32(k_ra8_mpu_mair0_addr, k_ra8_mair0_default);
413 internal_write32(k_ra8_mpu_mair1_addr, 0U);

416 (k_ra8_mpu_mram_base | k_ra8_rbar_attr_ro_x),
417 (k_ra8_mpu_mram_limit | k_ra8_mpu_rlar_enable));
418 internal_mpu_set_region(1U,
419 (k_ra8_mpu_sram_base | k_ra8_rbar_attr_rw_xn),
420 (k_ra8_mpu_sram_limit | k_ra8_mpu_rlar_enable));

464/* =============================================================================
465 * Public entry point
466 * =============================================================================
467 *
468 * Called from `Reset_Handler` after the C runtime is live (.data
469 * copied, .bss zeroed). It may therefore read and write globals
470 * freely; the Secure clock + TrustZone bring-up below relies on that.
471 */
472
473/* NOLINTNEXTLINE(readability-identifier-naming) -- CMSIS-mandated name. */
474void SystemInit(void)
475{
476 /* Mask global IRQs so the application gets a deterministic init
477 * window. main() must call ``ra8_isr_globals_enable`` once every
478 * driver / handler is wired up before any IRQ source can fire. */
479 internal_disable_irq();
480
481 internal_set_vtor();
482 internal_enable_fpu();
483 internal_enable_fpu_lazy_stack();
484 /* Cache enable, MPU init, and TrustZone bring-up are temporarily
485 * disabled -- they HardFault on first reset because the CCSIDR-driven
486 * invalidate-by-set/way loop is not yet implemented and the MPU
487 * regions are unconfigured. Re-enable once those code paths land. */
488 (void)internal_enable_icache;
489 (void)internal_enable_dcache;
490 (void)internal_enable_branch_predictor;
491 (void)internal_mpu_init;
492 internal_set_priority_grouping();
493
494 /* Bring up the Secure clock tree (XTAL -> PLL1 -> CPUCLK0 = 1 GHz) BEFORE the
495 * SAU + BLXNS below. ra8_trustzone_init() does NOT return on hardware (BLXNS
496 * leaves Secure thread mode), so anything sequenced after it never runs on the
497 * Secure side. The NS-side NSC CGC veneers (ra8_nsc_cgc_pll2_enable etc.) trap
498 * back into this Secure CGC driver, which needs PLL1 already locked -- without
499 * it the first PLL2 enable fails because its PLL1 source is missing. */
500 if (ra8_cgc_init() != k_ra8_ok) {
501 /* NASA Rule 7: a failed Secure clock bring-up must not BLXNS into NS code
502 * on a dead clock tree -- park at a named symbol instead of switching. */
503 internal_halt_secure_clock_fault();
504 }
505
506#ifdef RA8_TRUSTZONE_ENABLE
507 /* Programme the SAU, then BLXNS into the NS image. The NSC veneers depend on
508 * the SAU being live so the cmse_nonsecure_entry traps land on the Secure
509 * side. Does not return on hardware. */
510 ra8_trustzone_init();
511#else
512 (void)ra8_trustzone_init;
513#endif
514}
