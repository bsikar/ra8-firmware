110RA8_INTERNAL
111static ra8_err_t priv_classify(ra8_epub_book_t* book, const char* href, bool* out_is_atlas)
112{
113 *out_is_atlas = false;
114 uint8_t magic[k_ra8_epub_import_magic_len] = {};
115 size_t got = 0U;
116 const ra8_err_t err = ra8_epub_entry_pread(book, href, 0U, magic, sizeof(magic), &got);
117 if (err == k_ra8_err_not_supported) {
118 return k_ra8_ok; /* deflated entry: cannot be an in-place atlas */
119 }
120 if (err != k_ra8_ok) {
121 return err;
122 }
123 if ((got == sizeof(magic)) && (memcmp(magic, s_atlas_magic, sizeof(magic)) == 0)) {
124 *out_is_atlas = true;
125 }
126 return k_ra8_ok;
127}

147RA8_INTERNAL
148static ra8_err_t priv_transcode(ra8_epub_book_t* book,
149 const char* href,
150 const ra8_epub_atlas_import_cfg_t* cfg,
151 ra8_jof_info_t* out_info)
152{
153 ra8_epub_entry_pull_t pull = {};
154 uint64_t size = 0U;
155 ra8_err_t err = ra8_epub_entry_open(book, href, &pull.reader, &size);
156 if (err != k_ra8_ok) {
157 return err;
158 }
159 const ra8_jof_produce_cfg_t pcfg = {
160 .pull = priv_entry_pull,
161 .pull_ctx = &pull,
162 .sink = cfg->store.sink,
163 .sink_ctx = cfg->store.sink_ctx,
164 .tile_w = cfg->tile_w,
165 .tile_h = cfg->tile_h,
166 .codec = cfg->codec,
167 .max_width = cfg->max_width,
168 .max_height = cfg->max_height,
169 .work = cfg->work,
170 .work_cap = cfg->work_cap,
171 .webp_work = cfg->webp_work,
172 .webp_work_cap = cfg->webp_work_cap,
173 };
174 err = ra8_jof_produce(&pcfg, out_info);
175 const ra8_err_t close_err = ra8_epub_entry_close(&pull.reader);
176 return (err != k_ra8_ok) ? err : close_err;
177}

211ra8_err_t ra8_epub_tile_binder_import(ra8_epub_tile_binder_t* binder,
212 ra8_epub_book_t* book,
213 const char* href,
214 uint32_t image_id,
215 const ra8_epub_atlas_import_cfg_t* cfg)
216{
217 const ra8_err_t nz = priv_import_args_ok(binder, book, href, cfg);
218 if (nz != k_ra8_ok) {
219 return nz;
220 }
221 const size_t hlen = strlen(href);
222 if ((hlen == 0U) || (hlen >= (size_t)k_ra8_epub_max_path_len)) {
223 return k_ra8_err_invalid_arg;
224 }
225 bool is_atlas = false;
226 ra8_err_t err = priv_classify(book, href, &is_atlas);
227 if (err != k_ra8_ok) {
228 return err;
229 }
230 if (is_atlas) {
231 return ra8_epub_tile_binder_add(binder, book, href, image_id); /* in-place */
232 }
233 ra8_jof_info_t info = {};
234 err = priv_transcode(book, href, cfg, &info);
235 if (err != k_ra8_ok) {
236 return err;
237 }
238 return ra8_epub_tile_binder_add_ext(binder,
239 cfg->store.pread,
240 cfg->store.pread_ctx,
241 (uint64_t)info.total_size,
242 image_id);
243}
