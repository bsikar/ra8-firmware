4"""Generate the baked multi-page CBZ fixture for the viewable ereader_comic demo.
5
6Emits a pure 7-bit-ASCII C header (comic_pages_fixture.h) holding one CBZ
7(a miniz-decodable ZIP of grayscale PNG pages) as a byte array. The pages are
8near-panel-size (aspect ~ 1024:600) and each is visually distinct -- a unique
9paper tint, a different comic panel layout, and a huge 7-segment page number --
10so a page turn on the panel is unmistakable. Grayscale PNGs of flat-shaded art
11DEFLATE to a few hundred bytes each, so the whole archive stays small enough to
12bake as an ASCII array under the repo's per-file line cap.
13
14The output is deterministic (fixed art, fixed zlib level), so the render hash
15the app prints is identical on host, ra8_emulator, and silicon.
16
17Usage:
18 python3 scripts/gen/gen_comic_fixture.py \
19 examples/ek_ra8d2/hw_pending/ereader_comic/comic_pages_fixture.h
20"""

29PAGE_W = 480
30PAGE_H = 240
31PAGE_COUNT = 5

33# Grayscale palette (0 = black ink, 255 = white paper).
34INK = 0
35
36# Per-page paper tint + panel-fill grays, chosen far apart so the page turn is
37# visually obvious and every page framebuffer hashes differently.
38PAGE_PAPER = [244, 232, 250, 224, 238]
39PAGE_FILL = [200, 150, 120, 175, 95]

41# PNG / ZIP magic constants (named so the comparisons are self-documenting).
42PNG_BIT_DEPTH_8 = 8
43PNG_COLOR_GRAY = 0
44ZIP_LOCAL_MAGIC = 0x04034B50
45ZIP_CENTRAL_MAGIC = 0x02014B50
46ZIP_EOCD_MAGIC = 0x06054B50
47ZIP_VERSION = 20
48ZIP_METHOD_DEFLATE = 8
49DEFLATE_RAW_WBITS = -15
50ZLIB_BEST = 9
51BYTES_PER_ROW = 16

53# 7-segment layouts: single-vertical / mid-horizontal grids per page index.
54LAYOUT_VBAR = 0
55LAYOUT_HBAR = 1
56LAYOUT_CROSS = 2
57LAYOUT_TRIPTYCH = 3

59# Active 7-segment segments per decimal digit: (a, b, c, d, e, f, g).
60SEG_MAP = {
61 0: (1, 1, 1, 1, 1, 1, 0),
62 1: (0, 1, 1, 0, 0, 0, 0),
63 2: (1, 1, 0, 1, 1, 0, 1),
64 3: (1, 1, 1, 1, 0, 0, 1),
65 4: (0, 1, 1, 0, 0, 1, 1),

74class Canvas:
75 """A flat 8-bit grayscale page the drawing helpers paint into."""
76
77 def __init__(self, fill: int) -> None:
78 """Allocate a full page pre-filled with one gray level (the paper tint)."""
79 self.px = bytearray([fill]) * (PAGE_W * PAGE_H)

81 def fill_rect(self, rect: tuple[int, int, int, int], v: int) -> None:
82 """Paint an ``(x, y, w, h)`` rectangle, clipped to the page.
83
84 Clipping is applied per axis rather than rejecting an out-of-range
85 rect, so callers may compute a rectangle that runs off the edge and get
86 the visible part -- ``frame_rect`` and ``draw_digit`` both rely on that
87 and would otherwise need bounds checks of their own.
88 """
89 x0, y0, w, h = rect
90 for y in range(max(0, y0), min(PAGE_H, y0 + h)):
91 row = y * PAGE_W
92 for x in range(max(0, x0), min(PAGE_W, x0 + w)):
93 self.px[row + x] = v
94
95 def frame_rect(self, rect: tuple[int, int, int, int], t: int, v: int) -> None:
96 """Draw a hollow rectangle of border thickness ``t``, inset into ``rect``.
97
98 The four sides are drawn as overlapping bars, so corners are painted
99 twice; harmless here because every bar uses the same value ``v``.
100 """
101 x0, y0, w, h = rect
102 self.fill_rect((x0, y0, w, t), v)
103 self.fill_rect((x0, y0 + h - t, w, t), v)
104 self.fill_rect((x0, y0, t, h), v)
105 self.fill_rect((x0 + w - t, y0, t, h), v)
106
107 def draw_digit(self, digit: int, rect: tuple[int, int, int, int], t: int) -> None:
108 """Render one decimal digit as a 7-segment figure filling ``rect``.
109
110 Always drawn in INK, so the digit stays legible over any paper tint or
111 panel fill; the segment thickness ``t`` is the caller's, not derived
112 from the rect, which keeps the stroke weight identical across pages.
113 """
114 a, b, c, d, e, f, g = SEG_MAP[digit]
115 x, y, w, h = rect
116 mid = y + (h - t) // 2
117 half = (h // 2) + t
118 if a:
119 self.fill_rect((x, y, w, t), INK)
120 if g:
121 self.fill_rect((x, mid, w, t), INK)
122 if d:
123 self.fill_rect((x, y + h - t, w, t), INK)
124 if f:
125 self.fill_rect((x, y, t, half), INK)
126 if b:
127 self.fill_rect((x + w - t, y, t, half), INK)
128 if e:
129 self.fill_rect((x, mid, t, half), INK)
130 if c:
131 self.fill_rect((x + w - t, mid, t, half), INK)

134def draw_layout(canvas: Canvas, page_idx: int, fill: int) -> None:
135 """Draw a distinct comic panel arrangement (ink gutters) over a gray fill."""
136 gut = 8
137 x0, y0 = 24, 24
138 w = PAGE_W - 2 * x0
139 h = PAGE_H - 2 * y0
140 canvas.fill_rect((x0, y0, w, h), fill)
141 layout = page_idx % PAGE_COUNT
142 if layout in (LAYOUT_VBAR, LAYOUT_CROSS):
143 canvas.fill_rect((x0 + w // 2 - gut // 2, y0, gut, h), INK)
144 if layout in (LAYOUT_HBAR, LAYOUT_CROSS):
145 canvas.fill_rect((x0, y0 + h // 2 - gut // 2, w, gut), INK)
146 if layout == LAYOUT_TRIPTYCH:
147 canvas.fill_rect((x0 + w // 3 - gut // 2, y0, gut, h), INK)
148 canvas.fill_rect((x0 + 2 * w // 3 - gut // 2, y0, gut, h), INK)
149 if layout not in (LAYOUT_VBAR, LAYOUT_HBAR, LAYOUT_CROSS, LAYOUT_TRIPTYCH):
150 canvas.fill_rect((x0, y0 + h // 3 - gut // 2, w, gut), INK)
151 canvas.fill_rect((x0 + w // 2 - gut // 2, y0 + h // 3, gut, 2 * h // 3), INK)

154def render_page(page_idx: int) -> bytearray:
155 """Build one distinct grayscale page: border, panels, and a huge number."""
156 canvas = Canvas(PAGE_PAPER[page_idx])
157 canvas.frame_rect((6, 6, PAGE_W - 12, PAGE_H - 12), 6, INK)
158 draw_layout(canvas, page_idx, PAGE_FILL[page_idx])
159 dw, dh, dt = 90, 150, 20
160 canvas.draw_digit(page_idx + 1, ((PAGE_W - dw) // 2, (PAGE_H - dh) // 2, dw, dh), dt)
161 return canvas.px

164def png_chunk(tag: bytes, data: bytes) -> bytes:
165 """Wrap a payload in one PNG chunk: length, type, data, CRC-32.
166
167 The CRC covers the type tag AND the data but NOT the length prefix, which
168 is what the PNG spec requires and the easiest detail to get wrong -- a
169 chunk whose CRC included the length decodes as corrupt in every reader.
170 """
171 out = struct.pack(">I", len(data)) + tag + data
172 crc = zlib.crc32(tag + data) & 0xFFFFFFFF
173 return out + struct.pack(">I", crc)

176def encode_png(gray: bytes | bytearray) -> bytes:
177 """8-bit grayscale PNG (color type 0), one filter-0 byte per scanline."""
178 sig = b"\x89PNG\r\n\x1a\n"
179 ihdr = struct.pack(">IIBBBBB", PAGE_W, PAGE_H, PNG_BIT_DEPTH_8, PNG_COLOR_GRAY, 0, 0, 0)
180 raw = bytearray()
181 for y in range(PAGE_H):
182 raw.append(0)
183 raw += gray[y * PAGE_W : (y + 1) * PAGE_W]
184 idat = zlib.compress(bytes(raw), ZLIB_BEST)
185 return sig + png_chunk(b"IHDR", ihdr) + png_chunk(b"IDAT", idat) + png_chunk(b"IEND", b"")

188def _zip_local_entry(idx: int, data: bytes) -> tuple[bytes, tuple[bytes, int, int, int, int]]:
189 """Deflate one page and return its local-header record plus its directory entry.
190
191 The offset in the returned entry is relative to the start of the local
192 section, so the caller must add records in the same order it later writes
193 the central directory -- a ZIP whose offsets disagree is unreadable.
194 """
195 name = f"page{idx + 1:02d}.png".encode("ascii")
196 comp = zlib.compressobj(ZLIB_BEST, zlib.DEFLATED, DEFLATE_RAW_WBITS)
197 body = comp.compress(data) + comp.flush()
198 crc = zlib.crc32(data) & 0xFFFFFFFF
199 record = (
200 struct.pack(
201 "<IHHHHHIIIHH",
202 ZIP_LOCAL_MAGIC,
203 ZIP_VERSION,
204 0,
205 ZIP_METHOD_DEFLATE,
206 0,
207 0,
208 crc,
209 len(body),
210 len(data),
211 len(name),
212 0,
213 )
214 + name
215 + body
216 )
217 return record, (name, crc, len(body), len(data), 0)

263def build_cbz(pngs: list[bytes]) -> bytes:
264 """Minimal deterministic ZIP writer (DEFLATE members) that miniz reads."""
265 files: list[tuple[bytes, int, int, int, int]] = []
266 local = bytearray()
267 for idx, data in enumerate(pngs):
268 offset = len(local)
269 record, entry = _zip_local_entry(idx, data)
270 local += record
271 files.append((*entry[:4], offset))
272 central = bytearray()
273 for entry in files:
274 central += _zip_central_record(entry)
275 cd_off = len(local)
276 return bytes(local) + bytes(central) + _zip_eocd(len(files), len(central), cd_off)

279def emit_header(path: str | Path, cbz: bytes) -> None:
280 """Write the generated pure-ASCII C header holding the CBZ byte array."""
281 header = (
282 "/**\n"
283 " * @file comic_pages_fixture.h\n"
284 " * @brief Baked multi-page CBZ fixture for the viewable ereader_comic demo\n"
285 " * (generated by scripts/gen/gen_comic_fixture.py -- do not edit).\n"
286 " *\n"
287 f" * @details A miniz-decodable ZIP of {PAGE_COUNT} near-panel grayscale PNG pages\n"
288 f" * ({PAGE_W}x{PAGE_H} each). Every page carries a distinct paper tint,\n"
289 " * panel layout, and a huge 7-segment page number, so a page turn on the\n"
290 " * 1024x600 panel is unmistakable. Pure 7-bit ASCII byte array, like the\n"
291 " * bundled font / cover blobs.\n"
292 " *\n"
293 " * @copyright Copyright (c) 2026 Brighton Sikarskie\n"
294 " * SPDX-License-Identifier: MIT\n"
295 " */\n"
296 "#pragma once\n\n"
297 "#include <stdint.h>\n\n"
298 f"/** @brief Baked {len(cbz)}-byte multi-page CBZ ({PAGE_COUNT} pages). */\n"
299 "static const uint8_t k_comic_cbz[] = {\n"
300 )
301 rows = []
302 for start in range(0, len(cbz), BYTES_PER_ROW):
303 chunk = cbz[start : start + BYTES_PER_ROW]
304 rows.append(" " + " ".join(f"0x{byte:02X}," for byte in chunk))
305 footer = (
306 "\n};\n\n"
307 "/** @brief Length of ::k_comic_cbz in bytes. */\n"
308 f"static const uint32_t k_comic_cbz_len = {len(cbz)}U;\n\n"
309 "/** @brief Page count baked into ::k_comic_cbz. */\n"
310 f"static const uint32_t k_comic_page_count = {PAGE_COUNT}U;\n"
311 )
312 Path(path).write_text(header + "\n".join(rows) + footer, encoding="ascii")

315def main() -> None:
316 """Render every page, pack them into a CBZ, and write the C header.
317
318 Takes the output path from argv[1], defaulting to the bare filename in the
319 current directory. Returns None and exits 0 unconditionally: every failure
320 mode here (unwritable path, non-ASCII output) raises, and there is no
321 partial-success state a status code could describe.
322
323 The progress line goes to stderr so stdout stays free for the generated
324 header if this is ever redirected.
325 """
326 out = sys.argv[1] if len(sys.argv) > 1 else "comic_pages_fixture.h"
327 pngs = [encode_png(render_page(i)) for i in range(PAGE_COUNT)]
328 cbz = build_cbz(pngs)
329 emit_header(out, cbz)
330 sys.stderr.write(f"wrote {out} ({len(cbz)} bytes CBZ, {PAGE_COUNT} pages)\n")
