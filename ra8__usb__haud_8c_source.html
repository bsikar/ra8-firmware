591 case k_ra8_haud_step_get_dev_desc:

593 case k_ra8_haud_step_get_cfg_desc:

691 return k_ra8_ok;

730/* USB Audio 1.0 sec 5.2.3.2.3.1 "Sampling Frequency Control":
731 * bmRequestType = 0x22 (H2D | Class | Endpoint)
732 * bRequest = 0x01 (SET_CUR)
733 * wValue.high = 0x01 (SAMPLING_FREQ_CONTROL)
734 * wValue.low = 0x00
735 * wIndex = endpoint address
736 * wLength = 3 (24-bit LE sample rate)
737 */
738ra8_err_t
739ra8_usb_haud_set_format(uint8_t channel_count, uint8_t bits_per_sample, uint32_t sample_rate_hz)
740{
741 const ra8_err_t pre = internal_class_preflight();
742 if (pre != k_ra8_ok) {
743 return pre;
744 }
745 if (!internal_format_ok(channel_count, bits_per_sample, sample_rate_hz)) {
746 return k_ra8_err_invalid_arg;
747 }
748 /* Update the cached snapshot so the recipient's
749 * `ra8_usb_haud_device_t` reflects the new format. */
750 s_state.device.channel_count = channel_count;
751 s_state.device.bits_per_sample = bits_per_sample;
752 s_state.device.sample_rate_min_hz = sample_rate_hz;
753 s_state.device.sample_rate_max_hz = sample_rate_hz;
754
755 const uint8_t target_ep =
756 (s_state.device.iso_out_ep != 0U) ? s_state.device.iso_out_ep : s_state.device.iso_in_ep;
757 const ra8_usb_setup_t setup = {
758 .bm_request_type = k_ra8_haud_bm_class_ep_out,
759 .b_request = k_ra8_haud_req_set_cur,
760 .w_value = (uint16_t)((uint16_t)k_ra8_haud_ep_control_sampling_freq << k_ra8_haud_shift_byte1),
761 .w_index = (uint16_t)target_ep,
762 .w_length = k_ra8_haud_sample_rate_payload,
763 };
764 return ra8_usb_host_setup_request(s_state.speed, &setup);
765}

767/* USB Audio 1.0 sec 5.2.2.4.3.2 "Volume Control":
768 * bmRequestType = 0x21 (H2D | Class | Interface)
769 * bRequest = 0x01 (SET_CUR)
770 * wValue.high = 0x02 (VOLUME_CONTROL)
771 * wValue.low = channel
772 * wIndex.high = bUnitID (Feature Unit)
773 * wIndex.low = bInterfaceNumber (AC interface)
774 * wLength = 2 (signed 16-bit dB in 1/256 step)
775 */
776ra8_err_t ra8_usb_haud_set_volume(uint8_t channel, int16_t volume)
777{
778 const ra8_err_t pre = internal_class_preflight();
779 if (pre != k_ra8_ok) {
780 return pre;
781 }
782 const uint16_t w_value =
783 (uint16_t)(((uint16_t)k_ra8_haud_fu_control_volume << k_ra8_haud_shift_byte1) |
784 (uint16_t)channel);
785 const uint16_t w_index =
786 (uint16_t)(((uint16_t)s_state.device.feature_unit_id << k_ra8_haud_shift_byte1) |
787 (uint16_t)s_state.device.ac_interface);
788 const ra8_usb_setup_t setup = {
789 .bm_request_type = k_ra8_haud_bm_class_iface_out,
790 .b_request = k_ra8_haud_req_set_cur,
791 .w_value = w_value,
792 .w_index = w_index,
793 .w_length = k_ra8_haud_volume_payload,
794 };
795 /* Production path queues the 2-byte signed dB payload on the DCP
796 * data stage. The starter just ensures the SETUP envelope hits the
797 * wire. */
798 (void)volume;
799 return ra8_usb_host_setup_request(s_state.speed, &setup);
800}
