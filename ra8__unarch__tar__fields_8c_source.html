73static ra8_err_t s_octal(const uint8_t* field, size_t len, uint64_t* out)
74{
75 size_t i = 0U;
76 while (i < len) { /* bound: len */
77 if (field[i] != (uint8_t)' ') {
78 break;
79 }
80 i += 1U;
81 }
82 uint64_t v = 0U;
83 size_t digits = 0U;
84 while (i < len) { /* bound: len */
85 const uint8_t c = field[i];
86 if (c == 0U) {
87 break;
88 }
89 if (c == (uint8_t)' ') {
90 break;
91 }
92 if (c < (uint8_t)'0') {
93 return k_ra8_err_validation_failed;
94 }
95 if (c > (uint8_t)'7') {
96 return k_ra8_err_validation_failed;
97 }
98 const uint64_t d = (uint64_t)(c - (uint8_t)'0');
99 if (v > ((UINT64_MAX - d) / (uint64_t)k_tar_octal_base)) {
100 return k_ra8_err_validation_failed; /* would overflow uint64 */
101 }
102 v = (v * (uint64_t)k_tar_octal_base) + d;
103 digits += 1U;
104 i += 1U;
105 }
106 if (digits == 0U) {
107 return k_ra8_err_validation_failed; /* empty field */
108 }
109 *out = v;
110 return k_ra8_ok;
111}

132static ra8_err_t s_base256(const uint8_t* field, size_t len, uint64_t* out)
133{
134 if ((field[0] & (uint8_t)k_tar_b256_negative) != 0U) {
135 return k_ra8_err_validation_failed; /* negative size: hostile */
136 }
137 if ((field[0] & (uint8_t)k_tar_b256_payload) != 0U) {
138 return k_ra8_err_validation_failed; /* > 2^(8*(len-1)) - way over uint64 */
139 }
140 uint64_t v = 0U;
141 for (size_t i = 1U; i < len; ++i) { /* bound: len */
142 const size_t remaining = len - i;
143 if (remaining > (size_t)k_tar_b256_fit) {
144 if (field[i] != 0U) {
145 return k_ra8_err_validation_failed; /* over 64 bits */
146 }
147 continue;
148 }
149 v = (v << (uint64_t)k_tar_shift_byte) | (uint64_t)field[i];
150 }
151 *out = v;
152 return k_ra8_ok;
153}

179bool ra8_unarch_tar_checksum_ok(const uint8_t* block)
180{
181 uint32_t sum = 0U;
182 for (uint32_t i = 0U; i < (uint32_t)k_ra8_unarch_tar_block; ++i) { /* bound: block size */
183 const bool in_chksum = (i >= (uint32_t)k_ra8_tar_off_chksum) &&
184 (i < ((uint32_t)k_ra8_tar_off_chksum + (uint32_t)k_ra8_tar_len_chksum));
185 sum += in_chksum ? (uint32_t)k_tar_chksum_space : (uint32_t)block[i];
186 }
187 uint64_t stored = 0U;
188 if (s_octal(&block[k_ra8_tar_off_chksum], (size_t)k_ra8_tar_len_chksum, &stored) != k_ra8_ok) {
189 return false;
190 }
191 return stored == (uint64_t)sum;
192}

250static ra8_err_t s_pax_reclen(const uint8_t* data, size_t len, size_t* reclen, size_t* body)
251{
252 uint64_t v = 0U;
253 size_t digits = 0U;
254 size_t i = 0U;
255 while (i < len) { /* bound: len */
256 const uint8_t c = data[i];
257 if (c == (uint8_t)' ') {
258 break;
259 }
260 if (c < (uint8_t)'0') {
261 return k_ra8_err_validation_failed;
262 }
263 if (c > (uint8_t)'9') {
264 return k_ra8_err_validation_failed;
265 }
266 if (digits >= (size_t)k_tar_pax_len_max) {
267 return k_ra8_err_validation_failed; /* absurd length prefix */
268 }
269 v = (v * (uint64_t)k_tar_decimal_base) + (uint64_t)(c - (uint8_t)'0');
270 digits += 1U;
271 i += 1U;
272 }
273 if (digits == 0U) {
274 return k_ra8_err_validation_failed; /* no digits before the space */
275 }
276 if (i >= len) {
277 return k_ra8_err_validation_failed; /* ran out before the space */
278 }
279 *reclen = (size_t)v;
280 *body = i + 1U;
281 return k_ra8_ok;
282}

352static ra8_err_t s_pax_apply(const uint8_t* key,
353 size_t key_len,
354 const uint8_t* val,
355 size_t val_len,
356 char* name_buf,
357 uint16_t name_cap,
358 uint16_t* name_len,
359 bool* have_path,
360 uint64_t* size_ovr,
361 bool* have_size)
362{
363 static const char k_key_path[] = "path";
364 static const char k_key_size[] = "size";
365 const bool is_path =
366 (key_len == (sizeof(k_key_path) - 1U)) && (memcmp(key, k_key_path, key_len) == 0);
367 if (is_path) {
368 const size_t want = (val_len > (size_t)name_cap) ? (size_t)name_cap : val_len;
369 if (want > 0U) {
370 (void)memcpy(name_buf, val, want);
371 }
372 *name_len = (uint16_t)want;
373 *have_path = true;
374 return k_ra8_ok;
375 }
376 const bool is_size =
377 (key_len == (sizeof(k_key_size) - 1U)) && (memcmp(key, k_key_size, key_len) == 0);
378 if (is_size) {
379 const ra8_err_t verr = s_pax_size_value(val, val_len, size_ovr);
380 if (verr != k_ra8_ok) {
381 return verr;
382 }
383 *have_size = true;
384 }
385 return k_ra8_ok;

388ra8_err_t ra8_unarch_tar_pax_parse(const uint8_t* data,
389 size_t len,
390 char* name_buf,
391 uint16_t name_cap,
392 uint16_t* name_len,
393 bool* have_path,
394 uint64_t* size_ovr,
395 bool* have_size)
396{
397 RA8_CHECK_NULL_PTR(data, s_tag_tar_f, "pax: null data");
398 RA8_CHECK_NULL_PTR(name_len, s_tag_tar_f, "pax: null name_len");
399 RA8_CHECK_NULL_PTR(have_path, s_tag_tar_f, "pax: null have_path");
400 RA8_CHECK_NULL_PTR(size_ovr, s_tag_tar_f, "pax: null size_ovr");
401 RA8_CHECK_NULL_PTR(have_size, s_tag_tar_f, "pax: null have_size");
402 if (name_cap > 0U) {
403 RA8_CHECK_NULL_PTR(name_buf, s_tag_tar_f, "pax: null name_buf");
404 }
405 size_t off = 0U;
406 while (off < len) { /* bound: every record advances by reclen >= body + 1 */
407 size_t reclen = 0U;
408 size_t body = 0U;
409 const ra8_err_t lerr = s_pax_reclen(&data[off], len - off, &reclen, &body);
410 if (lerr != k_ra8_ok) {
411 return lerr;
412 }
413 if (reclen <= body) {
414 return k_ra8_err_validation_failed; /* length cannot cover its prefix */
415 }
416 if (reclen > (len - off)) {
417 return k_ra8_err_validation_failed; /* record overruns the data */
418 }
419 if (data[off + reclen - 1U] != (uint8_t)'\n') {
420 return k_ra8_err_validation_failed; /* record must end in newline */
421 }
422 const uint8_t* kv = &data[off + body];
423 const size_t kv_len = reclen - body - 1U; /* strip the newline */
424 const void* eq = memchr(kv, (int)'=', kv_len);
425 if (eq == nullptr) {
426 return k_ra8_err_validation_failed; /* no key/value delimiter */
427 }
428 const size_t key_len = (size_t)((const uint8_t*)eq - kv);
429 const ra8_err_t aerr = s_pax_apply(kv,
430 key_len,
431 &kv[key_len + 1U],
432 kv_len - key_len - 1U,
433 name_buf,
434 name_cap,
435 name_len,
436 have_path,
437 size_ovr,
438 have_size);
439 if (aerr != k_ra8_ok) {
440 return aerr;
441 }
442 off += reclen;
443 }
444 return k_ra8_ok;
445}
