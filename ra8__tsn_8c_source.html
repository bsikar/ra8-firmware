129static ra8_err_t internal_validate_cfg(const ra8_tsn_config_t* cfg)
130{
131 bool high_ok = false;
132 if ((cfg->high_ref_degc == k_ra8_tsn_cal_temp_high_125) ||
133 (cfg->high_ref_degc == k_ra8_tsn_cal_temp_high_105)) {
134 high_ok = true;
135 }
136 bool low_ok = false;
137 if (cfg->low_ref_degc == k_ra8_tsn_cal_temp_low_n40) {
138 low_ok = true;
139 }
140 if (!high_ok || !low_ok) {
141 return k_ra8_err_invalid_arg;
142 }
143 if (cfg->stab_us < k_ra8_tsn_min_stab_us) {
144 return k_ra8_err_invalid_arg;
145 }
146 return k_ra8_ok;
147}

149ra8_err_t ra8_tsn_init(const ra8_tsn_config_t* cfg)
150{
151 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
152 const ra8_err_t cfg_err = internal_validate_cfg(cfg);
153 RA8_RETURN_ON_ERROR(cfg_err, s_tag, "tsn_init: cfg invalid"); /* GCOVR_EXCL_BR_LINE */
154
155 /* HUM Ch 11.2.9 "MSTPCRD : Module Stop Control Register D", p 449 */
156 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_tsn);
157 RA8_RETURN_ON_ERROR(mst_err, s_tag, "tsn_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
158
159 /* HUM Ch 55.2.1 "TSCR : Temperature Sensor Control Register", p 3498 */
160 volatile r_tsn_ctrl_regs_t* reg = ra8_tsn();
161 reg->TSCR = k_ra8_tscr_mask_tsen;
162
163 /* HUM Ch 55.3.2 "Procedures for Measuring Temperature Sensor", p 3502
164 * (tTSTBL = 30 us per Figure 55.2 -- wait before enabling output). */
165 internal_busy_wait_us(cfg->stab_us);
166
167 /* HUM Ch 55.2.1 "TSCR : Temperature Sensor Control Register", p 3498 */
168 reg->TSCR = (uint8_t)(k_ra8_tscr_mask_tsen | k_ra8_tscr_mask_tsoe);
169
170 s_ra8_tsn_high_ref_degc = (int16_t)cfg->high_ref_degc;
171 s_ra8_tsn_low_ref_degc = (int16_t)cfg->low_ref_degc;
172 s_ra8_tsn_initialized = true;
173
174 ra8_log_info(s_tag, "tsn_init");
175 return k_ra8_ok;
176}

178ra8_err_t ra8_tsn_deinit(void)
179{
180 /* HUM Ch 55.2.1 "TSCR : Temperature Sensor Control Register", p 3498 */
181 volatile r_tsn_ctrl_regs_t* reg = ra8_tsn();
182 /* HUM Ch 55.3.2 "Procedures for Measuring Temperature Sensor", p 3502
183 * (clear TSOE first, then TSEN). */
184 reg->TSCR = k_ra8_tscr_mask_tsen;
185 /* HUM Ch 55.2.1 "TSCR : Temperature Sensor Control Register", p 3498 */
186 reg->TSCR = 0U;
187
188 /* Clear the init flag unconditionally so that callers cannot reach a
189 * "looks-initialized but isn't" state if deinit is invoked from a
190 * partially-torn-down context (e.g. mstp refcount already 0 because
191 * the test harness reset it). The mstp refcount underflow is logged
192 * by ra8_mstp itself and is harmless here. */
193 s_ra8_tsn_initialized = false;
194
195 /* HUM Ch 11.2.9 "MSTPCRD : Module Stop Control Register D", p 449 */
196 (void)ra8_mstp_disable(k_ra8_mstp_tsn);
197 return k_ra8_ok;
198}

212ra8_err_t ra8_tsn_convert_to_milli_c(uint16_t raw_code, int32_t* out_milli_c)
213{
214 RA8_CHECK_NULL_PTR(out_milli_c, s_tag, "out_milli_c must not be nullptr");
215 if (!s_ra8_tsn_initialized) {
216 return k_ra8_err_invalid_state;
217 }
218
219 /* HUM Ch 55.2.2 "TSCDR : Temperature Sensor Calibration Data Register", p 3498-3499 */
220 volatile const r_tsn_cal_regs_t* cal = ra8_tsn_cal();
221 const uint32_t cal_hi = cal->TSCDR & (uint32_t)k_ra8_tscdr_data_mask;
222 const uint32_t cal_lo = cal->TSCDR2 & (uint32_t)k_ra8_tscdr_data_mask;
223
224 if (cal_hi == cal_lo) {
225 /* Either an unprogrammed part or both trim words read 0; both
226 * mean "no slope is computable" and we must refuse rather than
227 * divide by zero (HUM Ch 55.3.1 p 3499). */
228 return k_ra8_err_invalid_state;
229 }
230
231 /* All math in 64-bit signed integers, microvolt scale. */
232 const int64_t avcc_uv = (int64_t)k_ra8_tsn_adc_avcc_uv;
233 const int64_t full_scale = (int64_t)k_ra8_tsn_adc_full_scale;
234 const int64_t v1_uv = (avcc_uv * (int64_t)cal_hi) / full_scale;
235 const int64_t v2_uv = (avcc_uv * (int64_t)cal_lo) / full_scale;
236 const int64_t vs_uv = (avcc_uv * (int64_t)raw_code) / full_scale;
237 const int64_t t1_milli_c = (int64_t)s_ra8_tsn_high_ref_degc * (int64_t)k_ra8_tsn_uv_per_mv;
238 const int64_t t2_milli_c = (int64_t)s_ra8_tsn_low_ref_degc * (int64_t)k_ra8_tsn_uv_per_mv;
239 const int64_t v_delta_uv = v1_uv - v2_uv; /* != 0 by guard above */
240 const int64_t numerator_uv = (vs_uv - v1_uv) * (t1_milli_c - t2_milli_c);
241 const int64_t result_milli = (numerator_uv / v_delta_uv) + t1_milli_c;
242
243 *out_milli_c = (int32_t)result_milli;
244 return k_ra8_ok;
245}

247ra8_err_t ra8_tsn_read_die_temp_milli_c(int32_t* out_milli_c)
248{
249 RA8_CHECK_NULL_PTR(out_milli_c, s_tag, "out_milli_c must not be nullptr");

251 return k_ra8_err_invalid_state;

254 /* Drive the ADC temperature-sensor channel (CNVCS = 0x64) and read the
255 * raw code back from its ADEXDR4 result register (HUM Ch 53.2.13.2
256 * p 3391). The ADC side lives entirely in adc.c. */
257 uint16_t adc_raw = 0U;

264
265 return ra8_tsn_convert_to_milli_c(code, out_milli_c);
266}
