244static uint32_t demo_fb_hash(void)
245{
246 const uint8_t* bytes = (const uint8_t*)g_dlist_fb;
247 uint32_t hash = (uint32_t)k_demo_fnv_offset;
248 for (uint32_t i = 0U; i < sizeof(g_dlist_fb); ++i) {
249 hash ^= (uint32_t)bytes[i];
250 hash *= (uint32_t)k_demo_fnv_prime;
251 }
252 return hash;
253}

256static void demo_fmt_hex(uint32_t crc, uint8_t* out)
257{
258 for (uint32_t i = 0U; i < (uint32_t)k_demo_hex_digits; ++i) {
259 const uint32_t shift = ((uint32_t)k_demo_hex_digits - 1U - i) * (uint32_t)k_demo_nibble_bits;
260 out[i] = k_demo_hex_lut[(crc >> shift) & (uint32_t)k_demo_nibble_mask];
261 }
262}

265static void demo_report(uint32_t crc, bool match)
266{
267 uint8_t hex[(uint32_t)k_demo_hex_digits];
268 demo_fmt_hex(crc, hex);
269 (void)ra8_board_uart_console_write(k_demo_prefix, (size_t)(sizeof(k_demo_prefix) - 1U));
270 (void)ra8_board_uart_console_write(hex, (size_t)k_demo_hex_digits);
271 if (match) {
272 (void)ra8_board_uart_console_write(k_demo_ok_msg, (size_t)(sizeof(k_demo_ok_msg) - 1U));
273 (void)ra8_board_led_toggle(k_ra8_board_led1);
274 } else {
275 (void)ra8_board_uart_console_write(k_demo_bad_msg, (size_t)(sizeof(k_demo_bad_msg) - 1U));
276 (void)ra8_board_led_toggle(k_ra8_board_led2);
277 }

282int32_t main(void)
283{
284 demo_setup_or_halt();
285 /* Clear PRIMASK so SysTick can dispatch ra8_delay_ms(). No NVIC sources are
286 * armed by this demo. */
287 ra8_isr_globals_enable();
288
289 if (demo_configure() != k_ra8_ok) {
290 demo_panic_halt();
291 }
292
293 ra8_drw_dlist_t dl;
294 if (demo_build_dlist(&dl) != k_ra8_ok) {
295 demo_panic_halt();
296 }
297
298 /* Poison the framebuffer ONCE so the first display-list clear must zero it;
299 * the DRW owns the framebuffer from here on. */
300 for (uint32_t i = 0U; i < (uint32_t)k_demo_fb_dim * (uint32_t)k_demo_fb_dim; ++i) {
301 g_dlist_fb[i] = (uint32_t)k_demo_poison;
302 }
303
304 while (1) {
305 if (ra8_drw_dlist_run(&dl) == k_ra8_ok) {
306 (void)ra8_drw_wait_idle((uint32_t)k_demo_idle_budget);
307 }
308 uint32_t status = 0U;
309 if (ra8_drw_get_status(&status) == k_ra8_ok) {
310 g_dlist_buserr |= (status & (uint32_t)k_ra8_drw_status_buserr_mask);
311 }
312 const uint32_t crc = demo_fb_hash();
313 const bool match = (crc == (uint32_t)k_demo_golden_crc);
314 g_dlist_crc = crc;
315 g_dlist_match = match ? 1U : 0U;
316 demo_report(crc, match);
317 ++g_dlist_heartbeat;
318 ra8_delay_ms((uint32_t)k_demo_period_ms);
319 }
320 demo_panic_halt();
321 return 0;
322}
