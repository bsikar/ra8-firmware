135static void internal_set_vtor(void)
136{
137 /* Vector table sits at the start of the .vectors section, which
138 * the linker pins to the start of MRAM (`0x02000000`). The SCB
139 * VTOR register stores the absolute address. */

293[[maybe_unused]] static void internal_mpu_init(void)
294{
295 enum : uint32_t {
296 k_ra8_mpu_ctrl_enable = 1UL << 0,
297 k_ra8_mpu_ctrl_privdefena = 1UL << 2,
298 k_ra8_mair0_default = 0x000404FFUL,
299 };
300
301 /* MAIR0: idx 0 = WB/WA, idx 1 = non-cacheable, idx 2 = device-nGnRE. */
302 internal_write32(k_ra8_mpu_mair0_addr, k_ra8_mair0_default);
303 internal_write32(k_ra8_mpu_mair1_addr, 0U);
304
305 internal_mpu_set_region(0U,
306 (k_ra8_mpu_mram_base | k_ra8_rbar_attr_ro_x),
307 (k_ra8_mpu_mram_limit | k_ra8_mpu_rlar_enable));
308 internal_mpu_set_region(1U,
309 (k_ra8_mpu_sram_base | k_ra8_rbar_attr_rw_xn),
310 (k_ra8_mpu_sram_limit | k_ra8_mpu_rlar_enable));
311 internal_mpu_set_region(2U,
312 (k_ra8_mpu_sdram_base | k_ra8_rbar_attr_rw_xn),
313 (k_ra8_mpu_sdram_limit | k_ra8_mpu_rlar_enable));
314 internal_mpu_set_region(3U,
315 (k_ra8_mpu_peri_base | k_ra8_rbar_attr_device_rw),
316 (k_ra8_mpu_peri_limit | k_ra8_mpu_rlar_enable));

318 /* Enable MPU with PRIVDEFENA so unclassified privileged accesses

320 internal_dsb();

354/* =============================================================================
355 * Public entry point
356 * =============================================================================
357 *
358 * Called from `Reset_Handler` before the .data copy and .bss zero.
359 * Must therefore not touch any global variables -- everything here
360 * runs on the stack and writes to CPU / SCB memory only.
361 */
362
363/* NOLINTNEXTLINE(readability-identifier-naming) -- CMSIS-mandated name. */
364void SystemInit(void)
365{
366 /* Mask global IRQs so the application gets a deterministic init
367 * window. main() must call ``ra8_isr_globals_enable`` once every
368 * driver / handler is wired up before any IRQ source can fire. */
369 internal_disable_irq();
370
371 internal_set_vtor();
372 internal_enable_fpu();
373 internal_enable_fpu_lazy_stack();
374 /* Cache enable, MPU init, and TrustZone bring-up are temporarily
375 * disabled -- they HardFault on first reset because the CCSIDR-driven
376 * invalidate-by-set/way loop is not yet implemented and the MPU
377 * regions are unconfigured. Re-enable once those code paths land. */
378 (void)internal_enable_icache;
379 (void)internal_enable_dcache;
380 (void)internal_enable_branch_predictor;
381 (void)internal_mpu_init;
382 internal_set_priority_grouping();
383
384 /* Bring up the Secure clock tree (XTAL -> PLL1 -> CPUCLK0 = 1 GHz) BEFORE the
385 * SAU + BLXNS below. ra8_trustzone_init() does NOT return on hardware (BLXNS
386 * leaves Secure thread mode), so anything sequenced after it never runs on the
387 * Secure side. The NS-side NSC CGC veneers (ra8_nsc_cgc_pll2_enable etc.) trap
388 * back into this Secure CGC driver, which needs PLL1 already locked -- without
389 * it the first PLL2 enable fails because its PLL1 source is missing. */
390 if (ra8_cgc_init() != k_ra8_ok) {
391 /* NASA Rule 7: a failed Secure clock bring-up must not BLXNS into NS code
392 * on a dead clock tree -- park at a named symbol instead of switching. */
393 internal_halt_secure_clock_fault();
394 }
395
396#ifdef RA8_TRUSTZONE_ENABLE
397 /* Programme the SAU, then BLXNS into the NS image. The NSC veneers depend on
398 * the SAU being live so the cmse_nonsecure_entry traps land on the Secure
399 * side. Does not return on hardware. */
400 ra8_trustzone_init();
401#else
402 (void)ra8_trustzone_init;
403#endif
404}
