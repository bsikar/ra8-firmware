433{

557RA8_ISR_SAFE
558void ra8_mipi_phy_dispatch(void)
559{
560 /* HUM Ch 64.2.6 "DPHYSFR : D-PHY Status Flag Register", p 3826 */
561 const uint32_t sfr = *ra8_mipi_phy_reg32(k_ra8_mipi_phy_off_sfr);
562 const uint32_t prev = s_last_sfr;
563 s_last_sfr = sfr;
564 const ra8_mipi_phy_event_fn_t fn = s_mipi_phy_fn;
565 void* const ctx = s_mipi_phy_ctx;
566
567 if (fn == nullptr) {
568 return;
569 }
570
571 const uint32_t pwr_now = sfr & k_ra8_mipi_phy_sfr_pwrsf;
572 const uint32_t pwr_prev = prev & k_ra8_mipi_phy_sfr_pwrsf;
573 const uint32_t pll_now = sfr & k_ra8_mipi_phy_sfr_pllsf;
574 const uint32_t pll_prev = prev & k_ra8_mipi_phy_sfr_pllsf;
575
576 /* Emit at most one event per dispatch -- power edges take priority
577 * over PLL edges, with status_chg as the fallback when nothing
578 * transitioned. Multiple emits per call would over-count callers
579 * that key off the cb count. */
580 if ((pwr_now != 0U) && (pwr_prev == 0U)) {
581 fn(ctx, k_ra8_mipi_phy_event_ldo_ready, sfr);
582 } else if ((pwr_now == 0U) && (pwr_prev != 0U)) {
583 fn(ctx, k_ra8_mipi_phy_event_ldo_lost, sfr);
584 } else if ((pll_now != 0U) && (pll_prev == 0U)) {
585 fn(ctx, k_ra8_mipi_phy_event_pll_locked, sfr);
586 } else if ((pll_now == 0U) && (pll_prev != 0U)) {
587 fn(ctx, k_ra8_mipi_phy_event_pll_lost, sfr);
588 } else {
589 fn(ctx, k_ra8_mipi_phy_event_status_chg, sfr);
590 }
591}
