68static int32_t priv_find(const ra8_epub_tile_binder_t* binder, uint32_t image_id)
69{
70 for (uint8_t i = 0U; i < binder->source_count; ++i) {

74 }

159RA8_INTERNAL
160static ra8_err_t priv_tile_decode(void* ctx,
161 const ra8_tile_key_t* key,
162 uint8_t* cell,
163 uint32_t cell_bytes,
164 uint16_t* out_w,
165 uint16_t* out_h)
166{
167 ra8_epub_tile_binder_t* binder = (ra8_epub_tile_binder_t*)ctx;
168 const int32_t si = priv_find(binder, key->image_id);
169 if (si < 0) {
170 return k_ra8_err_not_found; /* GCOVR_EXCL_LINE -- get() rejects unregistered ids before cache */
171 }
172 ra8_epub_tile_source_t* src = &binder->sources[si];
173 const ra8_jof_pread_fn pread = (src->book != nullptr) ? priv_entry_pread : src->pread;
174 void* pctx = (src->book != nullptr) ? (void*)src : src->pread_ctx;
175 const ra8_err_t err = ra8_jof_read_tile(pread,
176 pctx,
177 &src->info,
178 key->tile_x,
179 key->tile_y,
180 binder->scratch,
181 binder->scratch_cap,
182 cell,
183 cell_bytes,
184 out_w,
185 out_h);
186 if (err == k_ra8_err_invalid_size) {
187 return k_ra8_err_no_mem; /* tile exceeds the cell / scratch budget */
188 }
189 return err;
190}

475ra8_err_t ra8_epub_reflow_img_load(void* ctx,
476 const char* href,
477 uint32_t href_len,
478 const uint8_t** out_bytes,
479 size_t* out_len)
480{
481 const ra8_err_t nz = priv_loader_null_ok(ctx, href, out_bytes, out_len);
482 if (nz != k_ra8_ok) {
483 return nz;
484 }
485 *out_len = 0U;
486 ra8_epub_img_loader_t* ld = (ra8_epub_img_loader_t*)ctx;
487 const ra8_err_t ok = priv_loader_args_ok(ld, href_len);
488 if (ok != k_ra8_ok) {
489 return ok;
490 }
491 char path[k_ra8_epub_max_path_len];
492 (void)memcpy(path, href, (size_t)href_len);
493 path[href_len] = '\0';
494
495 size_t got = 0U;
496 const ra8_err_t err = ra8_epub_get_resource(ld->book, path, ld->scratch, ld->scratch_cap, &got);
497 if (err != k_ra8_ok) {
498 return err;
499 }
500 *out_bytes = ld->scratch;
501 *out_len = got;
502 return k_ra8_ok;
503}
