121static void internal_lin_program_mode(volatile r_sci_regs_t* reg,
122 ra8_sci_lin_role_t role,
123 uint32_t tcss,
124 uint16_t break_len)
125{
126 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0" p 2182 -- drop TE/RE
127 * before changing the operating mode. */
128 reg->CCR0 = 0U;
129
130 /* HUM Ch 38.2.8 "CCR3 : Common Control Register 3" p 2203 -- select
131 * Simple LIN (MOD = 110b), preserving LSBF / BPEN / CHR / STP set by
132 * ra8_sci_init. */
133 uint32_t ccr3 = reg->CCR3;
134 ccr3 &= ~(uint32_t)k_ra8_sci_ccr3_mask_mod;
135 ccr3 |= ((uint32_t)k_ra8_sci_ccr3_mod_simple_lin << k_ra8_sci_ccr3_shift_mod);
136 reg->CCR3 = ccr3;
137
138 /* HUM Ch 38.2.14 "XCR0 : Simple LIN Control Register 0" p 2220 -- select
139 * the break-field timer clock (TCSS) and enable break-field logic (BFE);
140 * BFE gates the responder's break detector as well as commander output. */
141 reg->XCR0 = (tcss << k_ra8_sci_xcr0_shift_tcss) | (1U << k_ra8_sci_xcr0_bit_bfe);
142
143 /* HUM Ch 38.2.16 "XCR2 : Simple LIN Control Register 2" p 2224 -- break
144 * dominant time = (BFLW + 1) x timer clock. */
145 reg->XCR2 = ((uint32_t)break_len << k_ra8_sci_xcr2_shift_bflw);
146
147 /* HUM Ch 38.2.15 "XCR1 : Simple LIN Control Register 1" p 2223 -- a
148 * commander leaves the trigger / detection fields idle (TCST is pulsed per
149 * frame by ra8_sci_lin_send_break); a responder arms Start-Frame detection
150 * (SDST) and bit-rate measurement (BMEN) so an inbound break sets XSR0.BFDF
151 * (HUM Ch 38.11.2 "Simple LIN Start Frame Reception" p 2341). */
152 uint32_t xcr1 = 0U;
153 if (role == k_ra8_sci_lin_role_responder) {
154 xcr1 = (1U << k_ra8_sci_xcr1_bit_sdst) | (1U << k_ra8_sci_xcr1_bit_bmen);
155 }
156 reg->XCR1 = xcr1;
157
158 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0" p 2182 -- re-enable
159 * the transmitter and receiver. */
160 reg->CCR0 = (1U << k_ra8_sci_ccr0_bit_te) | (1U << k_ra8_sci_ccr0_bit_re);
161}

186static ra8_err_t internal_lin_wait_break_done(volatile const r_sci_regs_t* reg)
187{
188 /* HUM Ch 38.2.15 "XCR1 : Simple LIN Control Register 1" p 2223 -- TCST holds 1
189 * during break output and clears on completion. The ra8_fake_mmio host seam
190 * drives this real poll on the unit-test build (T1-01): default RAM (TCST
191 * clear) takes the success leg, a staged TCST runs the loop to its timeout. */
192 const uint32_t mask = (1U << k_ra8_sci_xcr1_bit_tcst);
193 return ra8_hw_wait_flag_clear32(&reg->XCR1, mask, k_ra8_hw_budget_long);
194}

218static uint8_t internal_lin_fold_complement(uint16_t sum)
219{
220 for (uint8_t i = 0U; i < (uint8_t)k_ra8_sci_lin_fold_passes; ++i) {
221 const uint16_t low = (uint16_t)(sum & (uint16_t)k_ra8_sci_lin_byte_mask);
222 const uint16_t high = (uint16_t)(sum >> (uint16_t)k_ra8_sci_lin_byte_bits);
223 sum = (uint16_t)(low + high);
224 }
225 return (uint8_t)(~sum & (uint16_t)k_ra8_sci_lin_byte_mask);
226}

331ra8_err_t ra8_sci_lin_send_break(uint8_t channel)
332{
333 volatile r_sci_regs_t* reg = ra8_sci(channel);
334 RA8_CHECK_NULL_PTR(reg, s_tag, "lin_send_break: channel out of range");
335 /* HUM Ch 38.2.15 "XCR1 : Simple LIN Control Register 1" p 2223 -- writing
336 * 1 to TCST starts break-field output on TXD; the bit self-clears when the
337 * BFLW dominant time elapses. The other XCR1 fields stay idle in commander
338 * mode, so a full-word write is deterministic. */
339 reg->XCR1 = (1U << k_ra8_sci_xcr1_bit_tcst);
340 return internal_lin_wait_break_done(reg);

445uint8_t ra8_sci_lin_pid(uint8_t id)
446{
447 /* Mask any caller bits above bit 5 so parity is computed over a legal
448 * 6-bit identifier (the public range check lives in send_header). */
449 const uint8_t id6 = (uint8_t)(id & (uint8_t)k_ra8_sci_lin_id_mask);
450 const uint8_t b0 = (uint8_t)((id6 >> k_ra8_sci_lin_id_bit0) & (uint8_t)k_ra8_sci_lin_bit_mask);
451 const uint8_t b1 = (uint8_t)((id6 >> k_ra8_sci_lin_id_bit1) & (uint8_t)k_ra8_sci_lin_bit_mask);
452 const uint8_t b2 = (uint8_t)((id6 >> k_ra8_sci_lin_id_bit2) & (uint8_t)k_ra8_sci_lin_bit_mask);
453 const uint8_t b3 = (uint8_t)((id6 >> k_ra8_sci_lin_id_bit3) & (uint8_t)k_ra8_sci_lin_bit_mask);
454 const uint8_t b4 = (uint8_t)((id6 >> k_ra8_sci_lin_id_bit4) & (uint8_t)k_ra8_sci_lin_bit_mask);
455 const uint8_t b5 = (uint8_t)((id6 >> k_ra8_sci_lin_id_bit5) & (uint8_t)k_ra8_sci_lin_bit_mask);
456 /* P0 = ID0 ^ ID1 ^ ID2 ^ ID4 (even parity over the LIN-specified set). */
457 const uint8_t p0 = (uint8_t)(b0 ^ b1 ^ b2 ^ b4);
458 /* P1 = NOT(ID1 ^ ID3 ^ ID4 ^ ID5) (odd parity over the LIN set). */
459 const uint8_t p1 = (uint8_t)((b1 ^ b3 ^ b4 ^ b5) ^ (uint8_t)k_ra8_sci_lin_bit_mask);
460 return (uint8_t)(id6 | (uint8_t)(p0 << k_ra8_sci_lin_pid_p0_shift) |
461 (uint8_t)(p1 << k_ra8_sci_lin_pid_p1_shift));
462}

464ra8_err_t ra8_sci_lin_checksum(ra8_sci_lin_checksum_mode_t mode,
465 uint8_t pid,
466 const uint8_t* data,
467 uint8_t len,
468 uint8_t* out_checksum)
469{
470 RA8_CHECK_NULL_PTR(out_checksum, s_tag, "lin_checksum: out_checksum");
471 if (data == nullptr) {
472 if (len != 0U) {
473 return k_ra8_err_null_ptr;
474 }
475 }
476 if (mode > k_ra8_sci_lin_checksum_enhanced) {
477 return k_ra8_err_invalid_arg;
478 }
479
480 uint16_t sum = 0U;
481 /* Enhanced (LIN 2.x) folds the protected identifier into the sum;
482 * classic (LIN 1.x) sums the data bytes only. */
483 if (mode == k_ra8_sci_lin_checksum_enhanced) {
484 sum = (uint16_t)pid;
485 }
486 for (uint8_t i = 0U; i < len; ++i) {
487 sum = (uint16_t)(sum + (uint16_t)data[i]);
488 }
489 *out_checksum = internal_lin_fold_complement(sum);
490 return k_ra8_ok;
491}

493ra8_err_t ra8_sci_lin_check_header(uint8_t sync, uint8_t pid, uint8_t* out_id, bool* out_valid)
494{
495 RA8_CHECK_NULL_PTR(out_id, s_tag, "lin_check_header: out_id");
496 RA8_CHECK_NULL_PTR(out_valid, s_tag, "lin_check_header: out_valid");
497
498 const uint8_t id = (uint8_t)(pid & (uint8_t)k_ra8_sci_lin_id_mask);
499 *out_id = id;
500 /* A responder accepts the header only when the SYNC field is exactly 0x55
501 * AND the received PID matches the parity recomputed from its low 6 bits. */
502 const bool sync_ok = (sync == (uint8_t)k_ra8_sci_lin_sync_byte);
503 const bool pid_ok = (ra8_sci_lin_pid(id) == pid);
504 *out_valid = sync_ok && pid_ok;
505 return k_ra8_ok;
506}
