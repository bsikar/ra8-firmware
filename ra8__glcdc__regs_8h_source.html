189/* =============================================================================
190 * Gamma correction register block (HUM Ch 63 "GLCDC" p 3789-3793)
191 * =============================================================================
192 *
193 * Each of the three color channels (R=GAM[0], G=GAM[1], B=GAM[2]) occupies
194 * exactly 64 bytes starting at the k_ra8_glcdc_off_gamN base. The block
195 * contains:
196 * - LUT[8] at offset 0x00: eight 32-bit registers, each holding two 11-bit
197 * gain coefficients (GAIN0 in bits[26:16], GAIN1 in bits[10:0]).
198 * - AREA[8] at offset 0x20: eight 32-bit registers, each holding two 10-bit
199 * area/threshold values (TH1 in bits[25:16], TH2 in bits[9:0]).
200 * The global gamma-enable switch (GAMSW.GAMON, bit 0) lives in the shared
201 * OUT control block at k_ra8_glcdc_off_out_gamsw (0x13D8).
202 */
203

217 volatile uint32_t LUT[8];
