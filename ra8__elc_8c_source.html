91ra8_err_t ra8_elc_init(void)
92{
93 ra8_log_info(s_tag, "ra8_elc_init");
94
95 /* HUM Ch 11.2.8 "MSTPCRC : Module Stop Control Register C", p 447 */
96 const ra8_err_t mst_err = ra8_mstp_enable(k_ra8_mstp_elc);
97 RA8_RETURN_ON_ERROR(mst_err, s_tag, "elc_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
98
99 /* Clear every ELSR slot before flipping ELCON so stale routes
100 * don't fire spuriously.
101 * HUM Ch 19.2.3 "ELSRn : Event Link Setting Register n", p 817 */
102 for (uint8_t i = 0U; i < k_ra8_elc_elsr_count; ++i) {
103 *internal_elsr(i) = 0U;
104 }
105
106 /* HUM Ch 19.2.2 "ELSEGRn : Event Link Software Event Generation",
107 * p 817 -- writing 0 clears WI/WE/SEG so subsequent
108 * ra8_elc_software_trigger calls start from a clean state. */
109 for (uint8_t g = 0U; g < k_ra8_elc_segr_count; ++g) {
110 *internal_elsegr(g) = k_ra8_elc_elsegr_step_unlock;
111 }
112
113 /* HUM Ch 19.2.1 "ELCR : Event Link Control Register", p 817 */
114 *internal_elcr() = (uint8_t)(1U << k_ra8_elcr_bit_elcon);
115 return k_ra8_ok;
116}

145ra8_err_t ra8_elc_software_trigger(uint8_t event_index)
146{
147 if (event_index >= k_ra8_elc_segr_count) {
148 return k_ra8_err_invalid_arg;
149 }
150 /* HUM Ch 19.2.2 "ELSEGRn : Event Link Software Event Generation
151 * Register n", p 817. ELSEGRn is write-protected by WI/WE; SEG
152 * latches only after the documented three-write sequence. The
153 * same sequence is used by FSP R_ELC_SoftwareEventGenerate
154 * (`ELC_ELSEGRN_STEP1..3`). */
155 volatile uint8_t* elsegr = internal_elsegr(event_index);
156 *elsegr = k_ra8_elc_elsegr_step_unlock;
157 *elsegr = k_ra8_elc_elsegr_step_arm;
158 *elsegr = k_ra8_elc_elsegr_step_trigger;
159 return k_ra8_ok;
160}

162ra8_err_t ra8_elc_is_enabled(bool* out_enabled)
163{
164 RA8_CHECK_NULL_PTR(out_enabled, s_tag, "is_enabled out");
165 /* HUM Ch 19.2.1 "ELCR : Event Link Control Register", p 817 */
166 const uint8_t val = *internal_elcr();
167 *out_enabled = ((val & (uint8_t)(1U << k_ra8_elcr_bit_elcon)) != 0U);
168 return k_ra8_ok;
