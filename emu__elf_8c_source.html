32uint8_t* read_file(const char* path, long* out_len)
33{
34 FILE* f = fopen(path, "rb");
35 if (f == nullptr) {
36 return nullptr;
37 }
38 (void)fseek(f, 0, SEEK_END);
39 const long len = ftell(f);
40 (void)fseek(f, 0, SEEK_SET);
41 uint8_t* buf = (uint8_t*)malloc((size_t)len);
42 if ((buf != nullptr) && (fread(buf, 1U, (size_t)len, f) != (size_t)len)) {
43 free(buf);
44 buf = nullptr;
45 }
46 (void)fclose(f);
47 *out_len = len;
48 return buf;
49}

51int load_elf(uc_engine* uc, const uint8_t* elf, long len)
52{
53 if ((len < (long)k_elf_ehdr_size) ||
54 (memcmp(elf,
55 "\x7F"
56 "ELF",
57 4) != 0) ||
58 (elf[4] != 1) /* ELFCLASS32 */) {
59 (void)fprintf(stderr, "not a 32-bit ELF\n");
60 return -1;
61 }
62 const uint16_t e_machine =
63 (uint16_t)(elf[k_elf_e_machine_off] | (elf[k_elf_e_machine_off + 1U] << 8));
64 if (e_machine != (uint16_t)k_elf_em_arm) {
65 (void)fprintf(stderr, "ELF e_machine %u != ARM(40)\n", e_machine);
66 return -1;
67 }
68 uint32_t phoff = 0U;
69 (void)memcpy(&phoff, elf + (uint32_t)k_elf_e_phoff_off, 4);
70 uint16_t phentsize =
71 (uint16_t)(elf[k_elf_e_phentsize_off] | (elf[k_elf_e_phentsize_off + 1U] << 8));
72 uint16_t phnum = (uint16_t)(elf[k_elf_e_phnum_off] | (elf[k_elf_e_phnum_off + 1U] << 8));
73 int loaded = 0;
74 for (uint16_t i = 0U; i < phnum; i++) {
75 const uint8_t* ph = elf + phoff + ((size_t)(uint32_t)i * phentsize);
76 uint32_t p_type;
77 uint32_t p_offset;
78 uint32_t p_paddr;
79 uint32_t p_filesz;
80 (void)memcpy(&p_type, ph + 0, 4);
81 (void)memcpy(&p_offset, ph + 4, 4);
82 (void)memcpy(&p_paddr, ph + (uint32_t)k_elf_ph_paddr_off, 4); /* load address (LMA) */
83 (void)memcpy(&p_filesz, ph + 16, 4);
84 if ((p_type != 1U /* PT_LOAD */) || (p_filesz == 0U)) {
85 continue;
86 }
87 if (uc_mem_write(uc, p_paddr, elf + p_offset, p_filesz) != UC_ERR_OK) {
88 (void)fprintf(stderr, "uc_mem_write seg @0x%08X (%u bytes) failed\n", p_paddr, p_filesz);
89 return -1;
90 }
91 (void)fprintf(stderr, " loaded %u bytes @ 0x%08X\n", p_filesz, p_paddr);
92 loaded++;
93 }
94 return (loaded > 0) ? 0 : -1;
95}

133uint32_t elf_foreach_exec_segment(const uint8_t* elf, long len, elf_exec_segment_fn fn, void* ctx)
134{
135 if ((elf == nullptr) || (fn == nullptr) || (len < (long)k_elf_ehdr_size)) {
136 return 0U;
137 }
138 uint32_t phoff = 0U;
139 (void)memcpy(&phoff, elf + (uint32_t)k_elf_e_phoff_off, 4);
140 const uint16_t phentsize = (uint16_t)(elf[42] | (elf[43] << 8));
141 const uint16_t phnum = (uint16_t)(elf[44] | (elf[45] << 8));
142 if (((uint64_t)phoff + ((uint64_t)phnum * (uint64_t)phentsize)) > (uint64_t)len) {
143 return 0U; /* truncated program-header table -- refuse to walk past the file. */
144 }
145
146 uint32_t visited = 0U;
147 for (uint16_t i = 0U; i < phnum; i++) {
148 const uint8_t* ph = elf + phoff + ((size_t)(uint32_t)i * phentsize);
149 uint32_t p_type = 0U;
150 uint32_t p_offset = 0U;
151 uint32_t p_vaddr = 0U;
152 uint32_t p_filesz = 0U;
153 uint32_t p_flags = 0U;
154 (void)memcpy(&p_type, ph + 0, 4);
155 (void)memcpy(&p_offset, ph + (uint32_t)k_elf_ph_offset_off, 4);
156 (void)memcpy(&p_vaddr, ph + (uint32_t)k_elf_ph_vaddr_off, 4);
157 (void)memcpy(&p_filesz, ph + (uint32_t)k_elf_ph_filesz_off, 4);
158 (void)memcpy(&p_flags, ph + (uint32_t)k_elf_ph_flags_off, 4);
159 const bool executable = ((p_flags & (uint32_t)k_elf_pf_x) != 0U);
160 if ((p_type != (uint32_t)k_elf_pt_load) || (p_filesz == 0U) || !executable) {
161 continue;
162 }
163 if (((uint64_t)p_offset + (uint64_t)p_filesz) > (uint64_t)len) {
164 continue; /* truncated / out-of-file segment -- skip. */
165 }
166 const elf_exec_segment_t seg = {.bytes = elf + p_offset, .vaddr = p_vaddr, .filesz = p_filesz};
167 visited++;
168 if (!fn(&seg, ctx)) {
169 break;
170 }
171 }
172 return visited;

267static uint32_t elf_symtab_find(const uint8_t* elf,
268 long len,
269 const char* name,
270 size_t nlen,
271 const elf_symtab_t* st,
272 uint32_t* size_out)
273{
274 for (uint32_t s = 0U; s < st->count; s++) {
275 const uint8_t* sym = elf + st->sym_off + ((size_t)s * st->entsize);
276 if (((size_t)(sym - elf) + 16U) > (size_t)len) {
277 break;
278 }
279 uint32_t st_name = 0U;
280 uint32_t st_value = 0U;
281 uint32_t st_size = 0U;
282 (void)memcpy(&st_name, sym + 0, 4);
283 (void)memcpy(&st_value, sym + 4, 4);
284 (void)memcpy(&st_size, sym + 8, 4);
285 const size_t pos = (size_t)st->str_off + (size_t)st_name;
286 if ((st_name == 0U) || ((pos + nlen) > (size_t)len)) {
287 continue;
288 }
289 if (memcmp(elf + pos, name, nlen) == 0) {
290 if (size_out != nullptr) {
291 *size_out = st_size;
292 }
293 return st_value & ~1U; /* clear the Thumb bit for the hook address. */
294 }
295 }

299uint32_t elf_sym_addr(const uint8_t* elf, long len, const char* name, uint32_t* size_out)
300{
301 if (size_out != nullptr) {
302 *size_out = 0U;
303 }
304 if (len < (long)k_elf_ehdr_size) {
305 return 0U;
306 }
307 uint32_t shoff = 0U;
308 (void)memcpy(&shoff, elf + 32, 4);
309 const uint16_t shentsize = (uint16_t)(elf[46] | (elf[47] << 8));
310 const uint16_t shnum = (uint16_t)(elf[48] | (elf[49] << 8));
311 const size_t nlen = strlen(name) + 1U;
312 if ((shoff == 0U) || (shentsize < (uint32_t)k_elf_shentsize_min)) {
313 return 0U;
314 }
315 for (uint16_t i = 0U; i < shnum; i++) {
316 const uint8_t* sh = elf + shoff + ((size_t)(uint32_t)i * shentsize);
317 if (((size_t)(sh - elf) + (size_t)k_elf_shentsize_min) > (size_t)len) {
318 break;
319 }
320 elf_symtab_t st = {};
321 if (!elf_symtab_decode(elf, sh, shoff, shentsize, shnum, &st)) {
322 continue;
323 }
324 const uint32_t hit = elf_symtab_find(elf, len, name, nlen, &st, size_out);
325 if (hit != 0U) {
326 return hit;
327 }
328 }
329 return 0U;

333uint32_t warm_reboot(uc_engine* uc, const uint8_t* elf, long len, bool trace)
334{
335 /* 1. Restore the code + .data initial image from the ELF PT_LOAD segments.
336 * The same elf/len loaded successfully at startup, so a failure here means the
337 * engine's memory writes started failing -- report and return PC=0 so the
338 * caller ends the run rather than execute a stale image. */
339 if (load_elf(uc, elf, len) != 0) {
340 (void)fprintf(stderr, "ra8_emulator: warm_reboot: load_elf failed -- ending run\n");
341 return 0U;
342 }
343
344 /* 2. Reset the peripheral + network models (RSTSRn / VBATT backup survive). */
345 board_periph_init(trace);
346 board_net_init(trace);
347
348 /* 3. Clear host-side exception / scheduler bookkeeping. */
349 emu_exc_reset();
350 emu_mpu_clear_fault();
351 emu_div0_clear_fault();
352 emu_div0_disarm(); /* a warm reboot re-loads the image, un-patching sites. */
353 /* Clear the multi-channel console store + the in-flight ITM line, and reset
354 * the tabbed-console view so the rebooted firmware starts with an empty
355 * console on the ALL tab (the SCI model's own reset clears its line buffers). */
356 board_console_reset();
357 emu_console_reset();
358 emu_view_reset_console();
359
360 /* 4. Re-read the Cortex-M reset vector (SP = vectors[0], PC = vectors[1]). */
361 uint32_t sp = 0U;
362 uint32_t pc = 0U;
363 (void)uc_mem_read(uc, emu_memmap_mram_base() + 0U, &sp, 4);
364 (void)uc_mem_read(uc, emu_memmap_mram_base() + 4U, &pc, 4);
365 pc |= 1U; /* Thumb */
366 uint32_t xpsr = (uint32_t)k_xpsr_t_bit;
367 (void)uc_reg_write(uc, UC_ARM_REG_SP, &sp);
368 (void)uc_reg_write(uc, UC_ARM_REG_PC, &pc);
369 (void)uc_reg_write(uc, UC_ARM_REG_XPSR, &xpsr);
370 (void)fprintf(stderr, "ra8_emulator: warm reboot -- reset SP=0x%08X PC=0x%08X\n", sp, pc);
371 return pc;
372}
