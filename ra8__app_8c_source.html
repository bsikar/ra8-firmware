45[[nodiscard]] ra8_err_t ra8_app_register(ra8_app_registry_t* reg, ra8_app_t* app)
46{
47 RA8_CHECK_NULL_PTR(reg, s_tag, "reg must not be nullptr");
48 RA8_CHECK_NULL_PTR(app, s_tag, "app must not be nullptr");
49 RA8_CHECK_NULL_PTR(app->vt, s_tag, "app->vt must not be nullptr");
50 /* A duplicate id is a conflict regardless of capacity (check it first so a
51 * full registry still reports the more precise error). */
52 int16_t existing = (int16_t)k_ra8_app_none;
53 const ra8_err_t ferr = ra8_app_find(reg, app->id, &existing);
54 if (ferr != k_ra8_ok) {
55 return ferr;
56 }
57 if (existing != (int16_t)k_ra8_app_none) {
58 return k_ra8_err_conflict;
59 }
60 if (reg->count >= reg->cap) {
61 return k_ra8_err_no_mem;
62 }
63 if (app->vt->init != nullptr) {
64 const ra8_err_t ierr = app->vt->init(app);
65 if (ierr != k_ra8_ok) {
66 return ierr; /* leave unregistered */
67 }
68 }
69 app->initialized = true;
70 reg->apps[reg->count] = app;
71 reg->count = (uint16_t)(reg->count + 1U);
72 return k_ra8_ok;
73}

75[[nodiscard]] ra8_err_t ra8_app_launch(ra8_app_registry_t* reg, uint16_t id)
76{
77 RA8_CHECK_NULL_PTR(reg, s_tag, "reg must not be nullptr");
78 int16_t target = (int16_t)k_ra8_app_none;
79 const ra8_err_t ferr = ra8_app_find(reg, id, &target);
80 if (ferr != k_ra8_ok) {
81 return ferr;
82 }
83 if (target == (int16_t)k_ra8_app_none) {
84 return k_ra8_err_not_found;
85 }
86 if (target == reg->active) {
87 return k_ra8_ok; /* already focused: idempotent */
88 }
89 if (reg->active != (int16_t)k_ra8_app_none) {
90 ra8_app_t* cur = reg->apps[reg->active];
91 if ((cur != nullptr) && (cur->vt->on_leave != nullptr)) {
92 cur->vt->on_leave(cur);
93 }
94 }
95 reg->active = target;
96 ra8_app_t* next = reg->apps[target];
97 /*
98 * `target` is a live registry index returned by ra8_app_find() above (the
99 * function already returned k_ra8_err_not_found for k_ra8_app_none), and
100 * ra8_app_register() never stores a nullptr slot, so reg->apps[target] is
101 * provably non-null here. The (next != nullptr) clause is a defensive guard
102 * that cannot be driven false on any public-API path, so it cannot be given
103 * independent MC/DC influence; only (next->vt->on_enter != nullptr) varies.
104 */
105 /* mcdc-deactivated: next=reg->apps[target] with target a validated ra8_app_find index; (next!=nullptr) is always true here. */
106 if ((next != nullptr) && (next->vt->on_enter != nullptr)) {
107 next->vt->on_enter(next);
108 }
109 return k_ra8_ok;
110}

198[[nodiscard]] ra8_err_t ra8_app_nav_go(ra8_app_nav_t* nav, uint16_t id)
199{
200 RA8_CHECK_NULL_PTR(nav, s_tag, "nav must not be nullptr");
201 RA8_CHECK_NULL_PTR(nav->reg, s_tag, "nav->reg must not be nullptr");
202 ra8_app_t* cur = nullptr;
203 const ra8_err_t aerr = ra8_app_active(nav->reg, &cur);
204 if (aerr != k_ra8_ok) {
205 return aerr;
206 }
207 /* Only a switch *away from* a different app records a trail entry. The first
208 * launch (no prior focus) and an idempotent re-tap push nothing. The two
209 * single-condition guards below avoid a compound decision (each is plain
210 * branch coverage, not MC/DC). */
211 bool will_push = false;
212 uint16_t prev_id = 0U;
213 if (cur != nullptr) {
214 if (cur->id != id) {
215 will_push = true;
216 prev_id = cur->id;
217 }
218 }
219 if (will_push) {
220 if (nav->depth >= nav->cap) {
221 return k_ra8_err_no_mem;
222 }
223 }
224 const ra8_err_t lerr = ra8_app_launch(nav->reg, id);
225 if (lerr != k_ra8_ok) {
226 return lerr;
227 }
228 if (will_push) {
229 nav->stack[nav->depth] = prev_id;
230 nav->depth = (uint16_t)(nav->depth + 1U);
231 }
232 return k_ra8_ok;
233}

326[[nodiscard]] ra8_err_t ra8_app_uninstall(ra8_app_registry_t* reg, uint16_t id)
327{
328 RA8_CHECK_NULL_PTR(reg, s_tag, "reg must not be nullptr");
329 int16_t idx = (int16_t)k_ra8_app_none;
330 const ra8_err_t ferr = ra8_app_find(reg, id, &idx);
331 if (ferr != k_ra8_ok) {
332 return ferr;
333 }
334 if (idx == (int16_t)k_ra8_app_none) {
335 return k_ra8_err_not_found;
336 }
337 ra8_app_t* app = reg->apps[idx];
338 RA8_CHECK_NULL_PTR(app, s_tag, "registry slot must not be nullptr");
339 if (!app->removable) {
340 return k_ra8_err_not_supported; /* core app: uninstall refused */
341 }
342 if (idx == reg->active) {
343 return k_ra8_err_busy; /* the focused app cannot be unmounted */
344 }
345 if (app->vt->deinit != nullptr) {
346 app->vt->deinit(app);
347 }
348 app->initialized = false;
349 internal_app_remove_at(reg, (uint16_t)idx);
350 return k_ra8_ok;
