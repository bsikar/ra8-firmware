72static ra8_err_t s_source_read(void* ctx, uint64_t offset, uint8_t* buf, uint32_t len)
73{
74 RA8_CHECK_NULL_PTR(ctx, s_tag, "source_read: ctx");
75 RA8_CHECK_NULL_PTR(buf, s_tag, "source_read: buf");
76 if (offset > (uint64_t)UINT32_MAX) {
77 return k_ra8_err_out_of_range; /* GCOVR_EXCL_LINE -- the object size is an ra8_fs_size result (32-bit) and the loader clamps offsets within it */
78 }
79 ra8_fs_file_t* file = (ra8_fs_file_t*)ctx;
80 ra8_err_t err = ra8_fs_seek(file, (uint32_t)offset);
81 if (err != k_ra8_ok) {
82 return err; /* GCOVR_EXCL_LINE -- seek clamps to size and cannot fail on a live handle */
83 }
84 uint32_t got = 0U;
85 err = ra8_fs_read(file, buf, len, &got);
86 if (err != k_ra8_ok) {
87 return err; /* corrupt FAT chain / backend fault mid-stream */
88 }
89 if (got != len) {
90 return k_ra8_err_hw_error; /* GCOVR_EXCL_LINE -- defensive: ra8_fs_read reports ok only after producing every requested in-bounds byte */
91 }
92 return k_ra8_ok;
93}

174static ra8_err_t s_stream_open(const ra8_rabook_import_compiler_ctx_t* ctx,
175 ra8_fs_mount_t* mount,
176 const char* epub_path,
177 import_stream_t* ss)
178{
179 RA8_CHECK_NULL_PTR(ctx->epub, s_tag, "epub");
180 RA8_CHECK_NULL_PTR(ss, s_tag, "stream state");
181 ra8_err_t err = ra8_fs_open(mount, epub_path, k_ra8_fs_mode_read, &ss->file);
182 if (err != k_ra8_ok) {
183 return err;
184 }
185 /* ra8_fs_size cannot fail on the handle ra8_fs_open just returned (in_use
186 * set, both out-params non-NULL); on any future contract change `size`
187 * stays 0 and s_cache_bind rejects it as invalid_size. */
188 uint32_t size = 0U;
189 (void)ra8_fs_size(ss->file, &size);
190 err = s_cache_bind(ctx, ss, size);
191 if (err == k_ra8_ok) {
192 const ra8_epub_stream_media_t media = {
193 .read = ra8_vmem_stream_read,
194 .ctx = &ss->st,
195 .size = (uint64_t)size,
196 };
197 err = ra8_epub_open_streamed(&media, epub_path, ctx->epub);
198 }
199 if (err != k_ra8_ok) {
200 (void)ra8_fs_close(ss->file);
201 ss->file = nullptr;
202 }
203 return err;

206ra8_err_t ra8_rabook_import_compile_adapter(void* compile_ctx,
207 ra8_fs_mount_t* mount,
208 const char* epub_path,
209 const char* out_path)
210{
211 RA8_CHECK_NULL_PTR(compile_ctx, s_tag, "compile_ctx");
212 RA8_CHECK_NULL_PTR(mount, s_tag, "mount");
213 RA8_CHECK_NULL_PTR(epub_path, s_tag, "epub_path");
214 RA8_CHECK_NULL_PTR(out_path, s_tag, "out_path");
215
216 /* The cookie's epub/cache fields are validated by the stream-open helpers and
217 * the bufs/scr by `ra8_rabook_compile_from_epub`, so a NULL cookie field still
218 * yields k_ra8_err_null_ptr without restating those guards here. */
219 const ra8_rabook_import_compiler_ctx_t* ctx =
220 (const ra8_rabook_import_compiler_ctx_t*)compile_ctx;
221
222 /* The book streams through `ss` for the whole compile, so it lives in this
223 * frame, spanning open -> compile -> close. */
224 import_stream_t ss = {};
225 ra8_err_t err = s_stream_open(ctx, mount, epub_path, &ss);
226 if (err != k_ra8_ok) {
227 return err;
228 }
229
230 err = ra8_rabook_compile_from_epub(ctx->epub, ctx->bufs, ctx->scr, mount, out_path);
231 ra8_err_t cerr = ra8_epub_close(ctx->epub);
232 (void)ra8_fs_close(ss.file); /* always release the source handle */
233 if (err != k_ra8_ok) {
234 return err;
235 }
236 return cerr;
237}

265static ra8_err_t s_read_whole_file(ra8_fs_mount_t* mount,
266 const char* path,
267 uint8_t* buf,
268 uint32_t cap,
269 uint32_t* out_len)
270{
271 ra8_fs_file_t* file = nullptr;
272 ra8_err_t err = ra8_fs_open(mount, path, k_ra8_fs_mode_read, &file);
273 if (err != k_ra8_ok) {
274 return err;
275 }
276 /* ra8_fs_size cannot fail on a just-opened handle; a lying 0 simply reads 0
277 * bytes below and the dispatch rejects the empty source. */
278 uint32_t size = 0U;
279 (void)ra8_fs_size(file, &size);
280 if (size > cap) {
281 /* Transport overflow: the source cannot fit the cross-core buffer. Report
282 * no_mem (an offload-failure class) so the streamed in-core fallback --
283 * which has no size ceiling (#230) -- can still import the book. */
284 (void)ra8_fs_close(file);
285 return k_ra8_err_no_mem;
286 }
287 err = ra8_fs_read(file, buf, cap, out_len);
288 ra8_err_t cerr = ra8_fs_close(file);
289 if (err != k_ra8_ok) {
290 return err;
291 }
292 return cerr;
293}

440ra8_err_t ra8_rabook_import_compile_adapter_m33(void* compile_ctx,
441 ra8_fs_mount_t* mount,
442 const char* epub_path,
443 const char* out_path)
444{
445 RA8_CHECK_NULL_PTR(compile_ctx, s_tag, "compile_ctx");
446 RA8_CHECK_NULL_PTR(mount, s_tag, "mount");
447 RA8_CHECK_NULL_PTR(epub_path, s_tag, "epub_path");
448 RA8_CHECK_NULL_PTR(out_path, s_tag, "out_path");
449
450 ra8_rabook_import_compiler_m33_ctx_t* ctx = (ra8_rabook_import_compiler_m33_ctx_t*)compile_ctx;
451 RA8_CHECK_NULL_PTR(ctx->dispatch, s_tag, "dispatch");
452
453 /* ctx->epub_load_buf / ctx->blob_buf are caller-owned per the cookie invariant
454 * and reached through ra8_fs_read / the dispatch, so a NULL there is a caller
455 * bug -- matching how the in-core adapter leans on its downstream guards. */
456
457 /* Read the source .epub off the mount (the M85 owns the FS), then offload. A
458 * read-stage transport overflow (source larger than the load buffer, #230)
459 * is an offload failure like any other: the streamed in-core fallback still
460 * imports the book. */
461 uint32_t epub_len = 0U;
462 ra8_err_t err =
463 s_read_whole_file(mount, epub_path, ctx->epub_load_buf, ctx->epub_load_cap, &epub_len);
464 if (err != k_ra8_ok) {
465 return s_fallback_or_propagate(ctx, err, mount, epub_path, out_path);
466 }
467 return s_offload_or_fallback(ctx, epub_len, mount, epub_path, out_path);
468}
