144[[nodiscard]] ra8_err_t ra8_ipc_sem_try_take(uint8_t sem_id)
145{
146 if ((uint16_t)sem_id >= (uint16_t)k_ra8_ipc_sem_count) {
147 return k_ra8_err_invalid_arg;
148 }
149 volatile uint32_t* sem = ra8_ipc_sem(sem_id);
150 RA8_CHECK_NULL_PTR(sem, s_tag, "sem mapping failed");
151 /* HUM Ch 3.2.3 "IPCSEMn" p 210 -- 32-bit read sets LOCK; the read
152 * value reports the *previous* state. 0 -> we just acquired,
153 * 1 -> someone else already owned it. */
154 const uint32_t prev = internal_ra8_ipc_sem_read_take(sem);
155 if (prev != 0U) {
156 return k_ra8_err_busy;
157 }
158 /* Acquire barrier: the critical section the caller is about to enter
159 * must not be reordered ahead of taking the lock. */
160 ra8_ipc_barrier();
161 return k_ra8_ok;
162}

164[[nodiscard]] ra8_err_t ra8_ipc_sem_take_timeout(uint8_t sem_id, uint16_t max_spins)
165{
166 if ((uint16_t)sem_id >= (uint16_t)k_ra8_ipc_sem_count) {
167 return k_ra8_err_invalid_arg;
168 }
169 if (max_spins == 0U) {
170 return k_ra8_err_invalid_arg;
171 }
172 if (max_spins > k_ra8_ipc_sem_take_max) {
173 max_spins = k_ra8_ipc_sem_take_max;
174 }
175 volatile uint32_t* sem = ra8_ipc_sem(sem_id);
176 RA8_CHECK_NULL_PTR(sem, s_tag, "sem mapping failed");
177 /* NASA Rule 2: bounded by ``max_spins``. */
178 for (uint16_t i = 0U; i < max_spins; ++i) {
179 /* HUM Ch 3.2.3 "IPCSEMn" p 210 */
180 const uint32_t prev = internal_ra8_ipc_sem_read_take(sem);
181 if (prev == 0U) {
182 /* Acquire barrier: order the caller's critical section after the
183 * successful take. */
184 ra8_ipc_barrier();
185 return k_ra8_ok;
186 }
187 }
188 return k_ra8_err_hw_timeout;
189}

191[[nodiscard]] ra8_err_t ra8_ipc_sem_release(uint8_t sem_id)
192{
193 if ((uint16_t)sem_id >= (uint16_t)k_ra8_ipc_sem_count) {
194 return k_ra8_err_invalid_arg;
195 }
196 volatile uint32_t* sem = ra8_ipc_sem(sem_id);
197 RA8_CHECK_NULL_PTR(sem, s_tag, "sem mapping failed");
198 /* Release barrier: every store the caller made inside the critical
199 * section must be visible to the peer before the lock is dropped. */
200 ra8_ipc_barrier();
201 internal_ra8_ipc_sem_release_write(sem);
202 return k_ra8_ok;
203}

205[[nodiscard]] ra8_err_t ra8_ipc_sem_is_locked(uint8_t sem_id, bool* out_locked)
206{
207 RA8_CHECK_NULL_PTR(out_locked, s_tag, "out_locked must not be nullptr");
208 if ((uint16_t)sem_id >= (uint16_t)k_ra8_ipc_sem_count) {
209 return k_ra8_err_invalid_arg;
210 }
211 volatile uint32_t* sem = ra8_ipc_sem(sem_id);
212 RA8_CHECK_NULL_PTR(sem, s_tag, "sem mapping failed");
213 /* HUM Ch 3.2.3 "IPCSEMn" p 210 -- the 32-bit read takes the lock
214 * as a side-effect, so we sample then restore the previous state. */
215 const uint32_t prev = internal_ra8_ipc_sem_read_take(sem);
216 *out_locked = (prev != 0U);
217 if (prev == 0U) {
218 /* The caller observed "unlocked", but the diagnostic read itself
219 * took the lock (HUM Ch 3.2.3 set condition). Undo the probe's own
220 * take with the write-1-to-clear release command so the register
221 * reads back 0 and the side effect stays invisible. */
222 internal_ra8_ipc_sem_release_write(sem);
223 }
224 return k_ra8_ok;
225}

275void ra8_ipc_dispatch_nmi(uint8_t unit)
276{
277 volatile r_ipc_nmi_regs_t* nmi = internal_ra8_ipc_get_nmi(unit);
278 if (nmi == nullptr) {
279 return;
280 }
281 /* HUM Ch 3.2.4 "IPC0NMISTA.NMI" p 210 -- snapshot before invoking
282 * the callback so a re-entrant SET on the peer side does not
283 * confuse our acknowledge. */
284 if ((nmi->NMISTA & k_ra8_ipc_nmi_mask_bit) == 0U) {
285 return;
286 }
287 const ra8_ipc_nmi_fn_t fn = s_ipc_nmi_callback;
288 void* const ctx = s_ipc_nmi_context;
289 if (fn != nullptr) {
290 fn(ctx, unit);
291 }
292 /* HUM Ch 3.2.6 "IPC0NMICLR.CLR" p 212 -- ack so the ICU drops the
293 * line. */
294 nmi->NMICLR = k_ra8_ipc_nmi_mask_bit;
295}

321{
322 if (ring->capacity == 0U) {
323 return k_ra8_err_invalid_arg;
324 }
325 if ((ring->capacity & (ring->capacity - 1U)) != 0U) {
326 /* Power-of-two capacity keeps the modulo cheap and unambiguous. */
327 return k_ra8_err_invalid_arg;

330 return k_ra8_err_invalid_arg;
331 }
332 if ((uint16_t)ring->sem_id >= (uint16_t)k_ra8_ipc_sem_count) {
333 return k_ra8_err_invalid_arg;
334 }
335 if ((uint16_t)ring->notify_id >= (uint16_t)k_ra8_ipc_irq_event_count) {
336 return k_ra8_err_invalid_arg;
337 }
338 return k_ra8_ok;

356[[nodiscard]] ra8_err_t ra8_ipc_ring_produce(ra8_ipc_ring_t* ring, uint32_t payload)
357{
358 RA8_CHECK_NULL_PTR(ring, s_tag, "ring must not be nullptr");
359 ra8_err_t err = ra8_ipc_sem_try_take(ring->sem_id);
360 if (err != k_ra8_ok) {
361 return err;
362 }
363 const uint32_t head = *ring->head;
364 const uint32_t tail = *ring->tail;
365 if ((head - tail) >= ring->capacity) {
366 /* Ring full -- release the semaphore before reporting busy so a
367 * later consumer can acquire and drain. */
368 (void)ra8_ipc_sem_release(ring->sem_id);
369 return k_ra8_err_busy;
370 }
371 ring->slots[head & (ring->capacity - 1U)] = payload;
372 *ring->head = head + 1U;
373 err = ra8_ipc_sem_release(ring->sem_id);
374 if (err != k_ra8_ok) {
375 return err; /* GCOVR_EXCL_LINE -- ring->sem_id already validated */
376 }
377 /* Wake the consumer through the configured IRQ event line. */
378 return ra8_ipc_send_event(ring->channel, ring->notify_id);
379}

382{
383 RA8_CHECK_NULL_PTR(ring, s_tag, "ring must not be nullptr");
384 RA8_CHECK_NULL_PTR(out_payload, s_tag, "out_payload must not be nullptr");
385 ra8_err_t err = ra8_ipc_sem_try_take(ring->sem_id);
386 if (err != k_ra8_ok) {
387 return err;
388 }
389 const uint32_t head = *ring->head;
390 const uint32_t tail = *ring->tail;
391 if (head == tail) {
392 (void)ra8_ipc_sem_release(ring->sem_id);
393 return k_ra8_err_no_data;
394 }
395 *out_payload = ring->slots[tail & (ring->capacity - 1U)];
396 *ring->tail = tail + 1U;
397 return ra8_ipc_sem_release(ring->sem_id);
398}
