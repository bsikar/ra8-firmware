111static void ehb_write_u32_line(uint32_t val)
112{
113 uint8_t buf[k_ehb_dec_u32_max];
114 uint32_t n = 0U;
115 if (val == 0U) {
116 buf[n] = (uint8_t)'0';
117 n++;
118 }
119 while ((val > 0U) && (n < (uint32_t)k_ehb_dec_u32_max)) {
120 buf[n] = (uint8_t)((uint32_t)'0' + (val % (uint32_t)k_ehb_radix));
121 n++;
122 val /= (uint32_t)k_ehb_radix;
123 }
124 for (uint32_t i = 0U; i < n; i++) {
125 ehb_write(&buf[n - 1U - i], 1U);
126 }
127 ehb_write(k_ehb_msg_crlf, (uint32_t)sizeof(k_ehb_msg_crlf) - 1U);
128}

216[[nodiscard]] static ra8_err_t ehb_run_once(void)
217{
218 ra8_err_t rc = ra8_cgc_eswclk_init();
219 ehb_write(k_ehb_msg_eswclk, (uint32_t)sizeof(k_ehb_msg_eswclk) - 1U);
220 ehb_write_u32_line((uint32_t)rc);
221 if (rc != k_ra8_ok) {
222 return rc;
223 }
224
225 rc = ra8_mstp_enable(k_ra8_mstp_eswm);
226 ehb_write(k_ehb_msg_mstp, (uint32_t)sizeof(k_ehb_msg_mstp) - 1U);
227 ehb_write_u32_line((uint32_t)rc);
228 if (rc != k_ra8_ok) {
229 return rc;
230 }
231
232 rc = ra8_eth_coma_bringup();
233 ehb_write(k_ehb_msg_coma, (uint32_t)sizeof(k_ehb_msg_coma) - 1U);
234 ehb_write_u32_line((uint32_t)rc);
235 if (rc != k_ra8_ok) {
236 return rc;
237 }
238
239 rc = ra8_eth_rgmii_select(k_ra8_eth_mii_port_1);
240 ehb_write(k_ehb_msg_rgmii, (uint32_t)sizeof(k_ehb_msg_rgmii) - 1U);
241 ehb_write_u32_line((uint32_t)rc);
242 return rc;
243}

247int32_t main(void)
248{
249 ehb_setup_or_halt();
250 ra8_isr_globals_enable();
251
252 ehb_write(k_ehb_msg_boot, (uint32_t)sizeof(k_ehb_msg_boot) - 1U);
253
254 if (ehb_run_once() != k_ra8_ok) {
255 ehb_write(k_ehb_msg_fail, (uint32_t)sizeof(k_ehb_msg_fail) - 1U);
256 } else {
257 ehb_write(k_ehb_msg_pass, (uint32_t)sizeof(k_ehb_msg_pass) - 1U);
258 }
259
260 while (1) {
261 __asm__ volatile("wfi");
262 }
263 return 0;
264}
