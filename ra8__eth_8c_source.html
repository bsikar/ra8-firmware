377static ra8_err_t internal_resolve_sizes(const ra8_eth_cfg_t* cfg,
378 uint16_t* tx_count,
379 uint16_t* rx_count,
380 uint16_t* buf_size)
381{
382 uint16_t tx = cfg->num_tx_descriptors;
383 uint16_t rx = cfg->num_rx_descriptors;
384 uint16_t bs = cfg->buffer_size;
385 if (tx == 0U) {
386 tx = k_ra8_eth_num_tx_desc;
387 }
388 if (rx == 0U) {
389 rx = k_ra8_eth_num_rx_desc;
390 }
391 if (bs == 0U) {
392 bs = k_ra8_eth_buf_size;
393 }
394 /* mcdc-deactivated: tx normalized to nonzero above; first OR-condition unreachable. */
395 if ((tx == 0U) || (tx > k_ra8_eth_num_tx_desc)) {
396 return k_ra8_err_invalid_arg;
397 }
398 /* mcdc-deactivated: rx normalized to nonzero above; first OR-condition unreachable. */
399 if ((rx == 0U) || (rx > k_ra8_eth_num_rx_desc)) {
400 return k_ra8_err_invalid_arg;
401 }
402 if ((bs < k_ra8_eth_min_frame) || (bs > k_ra8_eth_buf_size)) {
403 return k_ra8_err_invalid_arg;
404 }
405 *tx_count = tx;
406 *rx_count = rx;
407 *buf_size = bs;
408 return k_ra8_ok;
409}

427RA8_INTERNAL
428static ra8_err_t internal_bring_up_rmac(const ra8_eth_cfg_t* cfg)
429{
430 /* The board-level ``ra8_board_ethernet_init`` (or its equivalent for
431 * non-EK boards) is expected to have already programmed RMAC's MPIC
432 * (PIS / LSC / PIPP / PSMCS), enabled the MSTP gate, and brought the
433 * matching ETHA into OPERATION mode. Overwriting MPIC here would
434 * clobber the carefully-computed MDC clock divider (PSMCS) and force
435 * the wrong PHY interface mode (MII instead of RGMII). The only
436 * RMAC-level state ``ra8_eth_open`` still has to programme is the
437 * MAC address the application chose.
438 *
439 * HUM Ch 33.4 "MRMAC0 / MRMAC1" p 1707 -- the MAC address registers
440 * are paired with MPIC under the same write-once gate: silent on
441 * MAC writes while ETHA is in OPERATION. Bench-confirmed on
442 * EK-RA8D2: without the DISABLE -> CONFIG -> {MAC write} -> DISABLE
443 * -> OPERATION bracket, MRMAC0 / MRMAC1 read back as 0 after
444 * ra8_eth_open completes, and every wire-side ARP "who-has 192.168.
445 * .1.42" gets dropped at the RMAC perfect-match comparator before
446 * the GWCA descriptor ring ever sees it.
447 *
448 * The board layer ran the bracket once during ra8_rmac_init; we have
449 * to repeat it here so the application-supplied MAC actually lands
450 * in MRMAC0/MRMAC1. */
451 const ra8_rmac_port_t rmac_port = ra8_eth_channel_to_port(cfg->channel);
452 const ra8_etha_port_t etha_port =
453 (cfg->channel == 0U) ? (ra8_etha_port_t)k_ra8_etha_port_0 : (ra8_etha_port_t)k_ra8_etha_port_1;
454 uint8_t mac_copy[k_ra8_eth_mac_len];
455 for (uint8_t i = 0U; i < (uint8_t)k_ra8_eth_mac_len; ++i) {
456 mac_copy[i] = cfg->mac_address[i];
457 }
458
459 /* HUM Ch 32.3.1.1 "EAMC : Mode Command Register" p 1631 -- ETHA
460 * mode-transition sequence. DISABLE before CONFIG so the ESWM
461 * accepts the request even if the port was already in OPERATION. */
462 ra8_err_t err = ra8_etha_set_mode(etha_port, k_ra8_etha_opc_disable);
463 if (err != k_ra8_ok) {
464 return err;
465 }
466 err = ra8_etha_set_mode(etha_port, k_ra8_etha_opc_config);
467 if (err != k_ra8_ok) {
468 return err;
469 }
470
471 /* HUM Ch 33.4 "MRMAC0 / MRMAC1" p 1707 -- writes only stick while
472 * the paired ETHA is in CONFIG. */
473 const ra8_err_t mac_err = ra8_rmac_set_mac_address(rmac_port, mac_copy);
474
475 /* Always return to OPERATION even on MAC write failure so the chip
476 * doesn't end up parked in CONFIG. */
477 const ra8_err_t dis_err = ra8_etha_set_mode(etha_port, k_ra8_etha_opc_disable);
478 const ra8_err_t op_err = ra8_etha_set_mode(etha_port, k_ra8_etha_opc_operation);
479 if (mac_err != k_ra8_ok) {
480 return mac_err;
481 }
482 if (dis_err != k_ra8_ok) {
483 return dis_err;
484 }
485 return op_err;
486}

723RA8_INTERNAL
724static ra8_err_t internal_open_gwca_path(void)
725{
726 /* Note: ``ra8_board_ethernet_init`` already runs its own COMA reset
727 * (RR pulse + RCEC|ACE enable) via ``internal_eth_coma_reset`` --
728 * the full FSP CABPIRM.BPIOG/BPR handshake is intentionally skipped
729 * there because BPR never asserts on EK-RA8D2 silicon unless GWCA
730 * is also being brought up, and the board can't depend on that. */
731
732 /* Per FSP r_layer3_switch_open: program the per-port forwarding
733 * destination masks BEFORE bringing the GWCA up. 0x7F = allow all
734 * destinations (permissive baseline; tightens once L3 filtering
735 * lands). */
736 static const uint8_t s_fwpbfc_masks[3] = {k_eth_fwpbfc_mask,
737 k_eth_fwpbfc_mask,
738 k_eth_fwpbfc_mask};
739 (void)ra8_eth_mfwd_set_forwarding_masks(s_fwpbfc_masks);
740
741 const ra8_err_t err = ra8_eth_gwca_default_open(&s_gwca_state);
742 if (err != k_ra8_ok) {
743 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_fail_3;
744 return err;
745 }
746 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_ok_3;
747 const ra8_err_t mfwd_err =
748 ra8_eth_mfwd_route_queue(s_gwca_state.mac_port, (uint8_t)s_gwca_state.rx_queue_index);
749 if (mfwd_err != k_ra8_ok) {
750 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_fail_4;
751 return mfwd_err;
752 }
753 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_ok_4;
754 return k_ra8_ok;
755}

757ra8_err_t ra8_eth_open(const ra8_eth_cfg_t* cfg)
758{
759 RA8_CHECK_NULL_PTR(cfg, s_tag, "open: cfg must not be nullptr");
760
761 g_ra8_eth_open_step = 0U;
762 const ra8_err_t prep_rc = internal_open_prep(cfg);
763 if (prep_rc != k_ra8_ok) {
764 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_fail_1;
765 return prep_rc;
766 }
767 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_ok_1;
768
769 internal_populate_gwca_state(cfg);
770 g_ra8_eth_open_step = (uint32_t)k_ra8_eth_step_ok_2;
771
772 const ra8_err_t bring_up_rc = internal_open_gwca_path();
773 if (bring_up_rc != k_ra8_ok) {
774 return bring_up_rc;
775 }
776
777 /* Force a fresh MPIC re-sync on the next ::ra8_eth_link_status call
778 * so the on-chip RMAC's MPIC.LSC matches whatever speed the PHY
779 * actually negotiated (HUM Ch 33.4.1.2 "MPIC : PHY Interfaces
780 * Configuration Register" p 1707). */
781 s_eth_mac_speed_resynced = false;
782
783 internal_capture_state(cfg);
784
785 /* Bench-confirmed (issue #1 follow-up): leaving MPIC.PIS at the
786 * default GMII (1000mbit) configured by ra8_board_ethernet_init
787 * silently drops every RX frame when the actual link auto-
788 * negotiates to 100 Mbps. The PIS must match the *internal* xMII
789 * timing (MII for 10/100, GMII for 1G). Force a resync at the end
790 * of open so the MAC is ready for RX before the IP stack hits the
791 * driver -- without it MRGFCE stays at zero until something else
792 * calls ra8_eth_link_status, which NetX never does at boot. */
793 ra8_eth_link_t link = {.link_up = 0U, .speed_mbps = 0U, .full_duplex = 0U, .bmsr = 0U};
794 (void)ra8_eth_link_status(&link);
795
796 ra8_log_info(s_tag, "eth_open: gwca rings ready");
797 return k_ra8_ok;
798}

800ra8_err_t ra8_eth_close(void)
801{
802 if (s_eth_state.opened == 0U) {
803 return k_ra8_err_not_initialized;
804 }
805
806 /* Park the GWCA state machine in DISABLE. Best-effort -- ignore
807 * the err so teardown always reaches the MSTP-gate balance below.
808 * HUM Ch 34.3.1 "GWMC : Mode Configuration Register" p 1797 */
809 (void)ra8_eth_gwca_set_operation_mode(k_ra8_gwmc_opc_disable);
810
811 /* Tear down the local RMAC port (MSTP-gate ref-counted). */
812 const ra8_rmac_port_t rmac_port = ra8_eth_channel_to_port(s_eth_state.cfg.channel);
813 (void)ra8_rmac_deinit(rmac_port);
814
815 /* Balance the second ESWM-MSTP enable that ::ra8_eth_gwca_default_open
816 * acquired via ::ra8_eth_gwca_init. The first enable, owned by
817 * ::internal_open_prep, is balanced by the final ra8_mstp_disable
818 * call below. */
819 (void)ra8_eth_gwca_deinit();
820
821 s_eth_state.opened = 0U;
822 s_eth_mac_speed_resynced = false;
823 return ra8_mstp_disable(k_ra8_mstp_eswm);
824}

953{
954 RA8_CHECK_NULL_PTR(payload, s_tag, "test_inject_rx: payload nullptr");

958 const uint32_t slot = internal_test_rx_slot();

960 uint32_t copy_len = len;
