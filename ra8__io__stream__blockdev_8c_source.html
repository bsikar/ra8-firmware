112static uint32_t bdsink_fill_chunk(ra8_io_stream_blockdev_state_t* st,
113 const uint8_t* buf,
114 uint32_t len,
115 uint32_t* done)
116{
117 const uint32_t room = (uint32_t)k_ra8_io_block_size_bytes - st->fill;
118 const uint32_t rem = len - *done;
119 const uint32_t chunk = (rem < room) ? rem : room;
120 (void)memcpy(&st->sector[st->fill], &buf[*done], (size_t)chunk);
121 st->fill += chunk;
122 *done += chunk;
123 return chunk;
124}

185RA8_INTERNAL
186static ra8_err_t bdsink_write(void* ctx, const uint8_t* buf, uint32_t len, uint32_t* out_written)
187{
188 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
189 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
190 ra8_io_stream_blockdev_state_t* st = (ra8_io_stream_blockdev_state_t*)ctx;
191 uint32_t done = 0;
192 while (done < len) {
193 (void)bdsink_fill_chunk(st, buf, len, &done);
194 const ra8_err_t e = bdsink_commit_if_full(st);
195 if (e != k_ra8_ok) {
196 if (out_written != nullptr) {
197 *out_written = done;
198 }
199 return e;
200 }
201 }
202 if (out_written != nullptr) {
203 *out_written = len;
204 }
205 return k_ra8_ok;
206}

231RA8_INTERNAL
232static ra8_err_t bdsink_flush(void* ctx)
233{
234 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
235 ra8_io_stream_blockdev_state_t* st = (ra8_io_stream_blockdev_state_t*)ctx;
236 if (st->fill == 0U) {
237 return k_ra8_ok;
238 }
239 const size_t pad = (size_t)((uint32_t)k_ra8_io_block_size_bytes - st->fill);
240 (void)memset(&st->sector[st->fill], (int)k_bd_pad_byte, pad);
241 return bdsink_commit_sector(st);
242}
