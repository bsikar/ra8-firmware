229static void tc_print_uint(uint32_t value)
230{
231 uint8_t buf[k_tc_dec_ten];
232 uint32_t n = 0U;
233 if (value == 0U) {
234 buf[n] = (uint8_t)'0';
235 n++;
236 }
237 while ((value > 0U) && (n < (uint32_t)k_tc_dec_ten)) {
238 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_tc_dec_ten));
239 n++;
240 value /= (uint32_t)k_tc_dec_ten;
241 }
242 for (uint32_t i = 0U; i < n; i++) {
243 tc_print(&buf[n - 1U - i], 1U);
244 }
245}

405static ra8_err_t tc_draw_target(void* ctx, ra8_touch_cal_point_t target)
406{
407 tc_capture_t* cap = (tc_capture_t*)ctx;

409 return k_ra8_err_invalid_arg;

457static ra8_err_t tc_read_raw(void* ctx, ra8_touch_cal_point_t* out_raw)
458{
459 tc_capture_t* cap = (tc_capture_t*)ctx;
460 if ((cap == nullptr) || (out_raw == nullptr) ||
461 (cap->n_read >= (uint8_t)k_ra8_touch_cal_n_targets)) {
462 return k_ra8_err_invalid_arg;
463 }
464 ra8_touch_point_t pts[k_tc_max_points] = {};
465 for (uint32_t i = 0U; i < (uint32_t)k_tc_poll_max; i++) {
466 uint8_t got = 0U;
467 const ra8_err_t rc = ra8_touch_read(pts, (uint8_t)k_tc_max_points, &got);
468 if (tc_sample_usable(rc, got)) {
469 out_raw->x = (int32_t)pts[0].x;
470 out_raw->y = (int32_t)pts[0].y;
471 cap->raws[cap->n_read] = *out_raw;
472 cap->n_read++;
473 return k_ra8_ok;
474 }
475 }
476 return k_ra8_err_hw_error;
477}

528static bool tc_blob_roundtrip(const ra8_touch_cal_matrix_t* m)
529{
530 uint8_t blob0[k_tc_blob_cap] = {};
531 uint8_t blob1[k_tc_blob_cap] = {};
532 ra8_touch_cal_matrix_t back = {};
533 if (ra8_touch_cal_save(m, blob0, sizeof(blob0)) != k_ra8_ok) {
534 return false;
535 }
536 if (ra8_touch_cal_load(blob0, sizeof(blob0), &back) != k_ra8_ok) {
537 return false;
538 }
539 if (ra8_touch_cal_save(&back, blob1, sizeof(blob1)) != k_ra8_ok) {
540 return false;
541 }
542 return memcmp(blob0, blob1, sizeof(blob0)) == 0;
543}

559static ra8_err_t
560tc_apply_and_measure(const ra8_touch_cal_matrix_t* m, tc_capture_t* cap, int32_t* out_max)
561{
562 if ((m == nullptr) || (cap == nullptr) || (out_max == nullptr)) {
563 return k_ra8_err_null_ptr;
564 }
565 int32_t worst = 0;
566 for (uint8_t i = 0U; i < cap->n_read; i++) {
567 ra8_touch_cal_point_t screen = {};
568 const ra8_err_t rc =
569 ra8_touch_cal_apply(cap->raws[i], m, (uint16_t)k_tc_fb_w, (uint16_t)k_tc_fb_h, &screen);
570 if (rc != k_ra8_ok) {
571 return rc;
572 }
573 const int32_t ex = tc_axis_err(screen.x, cap->targets[i].x);
574 const int32_t ey = tc_axis_err(screen.y, cap->targets[i].y);
575 worst = (ex > worst) ? ex : worst;
576 worst = (ey > worst) ? ey : worst;
577 tc_draw_cross(screen.x, screen.y, (uint16_t)k_tc_color_corrected);
578 }
579 if (cap->disp != nullptr) {
580 (void)display_flush(cap->disp, display_full_rect(cap->disp), k_display_refresh_fast);
581 }
582 *out_max = worst;
583 return k_ra8_ok;
584}

640static void tc_calibrate_and_report(display_handle_t* disp)
641{
642 ra8_touch_cal_matrix_t matrix = {};
643 const ra8_err_t run_rc = tc_run_calibration(disp, &matrix);
644 if (run_rc != k_ra8_ok) {
645 tc_report_skip(s_capture.n_read);
646 return;
647 }
648 int32_t maxerr = 0;
649 if (tc_apply_and_measure(&matrix, &s_capture, &maxerr) != k_ra8_ok) {
650 tc_report_ok(maxerr, false);
651 return;
652 }
653 const bool blob_ok = tc_blob_roundtrip(&matrix);
654 const bool verify_ok = tc_cal_pass(run_rc, maxerr, (int32_t)k_tc_verify_tol, blob_ok);
655 tc_report_ok(maxerr, verify_ok);
656}
