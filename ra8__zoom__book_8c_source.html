71ra8_err_t
72ra8_zoom_book_src_init(ra8_zoom_book_src_t* bs, const ra8_book_src_t* src, uint32_t img_idx)
73{
74 RA8_CHECK_NULL_PTR(bs, s_tag, "binding must not be nullptr");
75 RA8_CHECK_NULL_PTR(src, s_tag, "book source must not be nullptr");
76 ra8_book_image_t img = {};
77 RA8_RETURN_ON_ERROR(ra8_book_src_image(src, img_idx, &img), s_tag, "image descriptor read");
78 if (img.format != (uint8_t)k_ra8_book_image_gray4) {
79 ra8_log_error(s_tag, "vector images are not magnifiable by the raster viewer");
80 return k_ra8_err_not_supported;
81 }
82 /* Decision: a raster with no extent cannot be magnified (2 conditions). */
83 if ((img.width == 0U) || (img.height == 0U)) {
84 ra8_log_error(s_tag, "image descriptor has a zero extent");
85 return k_ra8_err_invalid_arg;
86 }
87 bs->src = src;
88 bs->img = img;
89 return k_ra8_ok;
90}
