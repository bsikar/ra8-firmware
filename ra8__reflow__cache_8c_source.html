220RA8_INTERNAL
221static uint32_t priv_fnv1a(const uint8_t* data, size_t len)
222{
223 uint32_t hash = (uint32_t)k_priv_fnv_offset;
224 for (size_t i = 0U; i < len; ++i) {
225 hash ^= (uint32_t)data[i];
226 hash *= (uint32_t)k_priv_fnv_prime;
227 }
228 return hash;
229}

503RA8_INTERNAL
504static bool priv_key_matches(const ra8_reflow_t* engine,
505 const priv_cache_key_t* key,
506 const uint8_t* content,
507 size_t content_len)
508{
509 /* One simple decision per attribute (no compound boolean): any single
510 * mismatch makes the cached layout stale. */
511 if (key->viewport_w != engine->viewport_w) {
512 return false;
513 }
514 if (key->viewport_h != engine->viewport_h) {
515 return false;
516 }
517 if (key->font_px != engine->font_px) {
518 return false;
519 }
520 if (key->body_color != engine->body_color) {
521 return false;
522 }
523 if (key->link_color != engine->link_color) {
524 return false;
525 }
526 if (key->font_len != (uint32_t)engine->font_len) {
527 return false;
528 }
529 if (key->font_hash != priv_font_hash(engine)) {
530 return false;
531 }
532 if (key->content_len != (uint32_t)content_len) {
533 return false;
534 }
535 if (key->content_hash != priv_fnv1a(content, content_len)) {
536 return false;
537 }
538 return true;
539}

574RA8_INTERNAL
575static ra8_err_t
576priv_parse_header(const uint8_t* buf, size_t len, priv_cache_key_t* key, size_t* need)
577{
578 if (len < (size_t)k_ra8_reflow_cache_header_bytes) {
579 return k_ra8_err_invalid_size;
580 }
581 priv_get_header(buf, key);
582 if (key->magic != (uint32_t)k_ra8_reflow_cache_magic) {
583 return k_ra8_err_not_found;
584 }
585 if (key->version != (uint32_t)k_ra8_reflow_cache_version) {
586 return k_ra8_err_invalid_state;
587 }
588 if (key->glyph_count > (uint32_t)k_ra8_reflow_max_glyphs) {
589 return k_ra8_err_invalid_size;
590 }
591 if (key->page_count > (uint32_t)k_ra8_reflow_max_pages) {
592 return k_ra8_err_invalid_size;
593 }
594 *need = priv_blob_size(key->glyph_count, key->page_count);
595 if (len < *need) {
596 return k_ra8_err_invalid_size;
597 }
598 return k_ra8_ok;
599}

663[[nodiscard]] ra8_err_t ra8_reflow_cache_serialize(const ra8_reflow_t* engine,
664 const uint8_t* content,
665 size_t content_len,
666 uint8_t* out_buf,
667 size_t out_cap,
668 size_t* out_len)
669{
670 if ((engine == nullptr) || (out_buf == nullptr) || (out_len == nullptr)) {
671 return k_ra8_err_null_ptr;
672 }
673 const ra8_err_t verr = priv_cache_precheck(engine, content, content_len);
674 if (verr != k_ra8_ok) {
675 return verr;
676 }
677 const size_t need = priv_blob_size(engine->glyph_count, engine->page_count);
678 if (out_cap < need) {
679 return k_ra8_err_invalid_size;
680 }
681
682 priv_put_header(out_buf, engine, content_len, priv_fnv1a(content, content_len));
683
684 size_t off = (size_t)k_ra8_reflow_cache_header_bytes;
685 for (uint32_t i = 0U; i < engine->glyph_count; ++i) {
686 priv_put_glyph(out_buf, &off, &engine->glyphs[i]);
687 }
688 for (uint32_t i = 0U; i < engine->page_count; ++i) {
689 priv_put_page(out_buf, &off, &engine->pages[i]);
690 }
691
692 const uint32_t checksum = priv_fnv1a(&out_buf[(size_t)k_ra8_reflow_cache_header_bytes],
693 need - (size_t)k_ra8_reflow_cache_header_bytes);
694 size_t ck_off = (size_t)k_priv_checksum_off;
695 priv_put_u32(out_buf, &ck_off, checksum);
696 *out_len = need;
697 return k_ra8_ok;
698}

700[[nodiscard]] ra8_err_t ra8_reflow_cache_load(ra8_reflow_t* engine,
701 const uint8_t* content,
702 size_t content_len,
703 const uint8_t* buf,
704 size_t len)
705{
706 if ((engine == nullptr) || (buf == nullptr)) {
707 return k_ra8_err_null_ptr;
708 }
709 const ra8_err_t verr = priv_cache_precheck(engine, content, content_len);
710 if (verr != k_ra8_ok) {
711 return verr;
712 }
713
714 priv_cache_key_t key = {};
715 size_t need = 0U;
716 const ra8_err_t perr = priv_parse_header(buf, len, &key, &need);
717 if (perr != k_ra8_ok) {
718 return perr;
719 }
720 if (!priv_key_matches(engine, &key, content, content_len)) {
721 return k_ra8_err_invalid_state;
722 }
723
724 const uint32_t checksum = priv_fnv1a(&buf[(size_t)k_ra8_reflow_cache_header_bytes],
725 need - (size_t)k_ra8_reflow_cache_header_bytes);
726 if (checksum != key.body_checksum) {
727 return k_ra8_err_checksum_mismatch;
728 }
729
730 size_t off = (size_t)k_ra8_reflow_cache_header_bytes;
731 for (uint32_t i = 0U; i < key.glyph_count; ++i) {
732 priv_get_glyph(buf, &off, &engine->glyphs[i]);
733 }
734 for (uint32_t i = 0U; i < key.page_count; ++i) {
735 priv_get_page(buf, &off, &engine->pages[i]);
736 }
737 engine->glyph_count = key.glyph_count;
738 engine->page_count = key.page_count;
739 engine->xhtml_buf = content;
740 engine->xhtml_len = content_len;
741 return k_ra8_ok;
