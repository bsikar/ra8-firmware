111 * the console derives its baud from PCLKA, so bringing the clock up second
112 * would fix the baud against the reset clocks. Best-effort: a failure only
113 * changes the busy-loop cadence, not the correctness of the blink, so the
114 * verdict is carried forward and narrated once logging is available. */
115 const ra8_err_t clk_err = ra8_cgc_init();
116
117 ra8_log_init();
118 ra8_log_info("M85", "==== RA8D2 dual-core blink (HAL PCNTR primitive) ====");
119 ra8_log_info("M85", "Cortex-M85 primary core online");
120 if (clk_err != k_ra8_ok) {
121 ra8_log_info("M85", "clock init failed -- continuing at reset clocks");
122 }
123
124 ra8_log_info("M85", "releasing Cortex-M33 secondary core ...");
125 const ra8_err_t err = ra8_cpu1_release(&g_ra8_ls_cpu1_mram_start, &g_ra8_ls_cpu1_stack_top);
126 ra8_log_info_val("M85", "ra8_cpu1_release rc (0 = ok)", (uint32_t)err);
127 if (err != k_ra8_ok) {
128 ra8_log_info("M85", "release FAILED -- halting");
129 park_forever();
130 }
131
132 ra8_log_info("M85", "M33 released -- it now blinks LED1 via ra8_pcntr_set_output()");
133 ra8_log_info("M85", "M85 entering WFI idle -- low-power co-processor model");
134 idle_forever();
