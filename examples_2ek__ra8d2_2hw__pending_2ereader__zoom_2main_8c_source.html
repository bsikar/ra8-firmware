331static void ez_print_hex(uint32_t value)
332{
333 uint8_t buf[k_ez_hex_nibbles];
334 for (uint32_t i = 0U; i < (uint32_t)k_ez_hex_nibbles; ++i) {
335 const uint32_t shift = ((uint32_t)k_ez_hex_nibbles - 1U - i) * (uint32_t)k_ez_nibble_bits;
336 const uint32_t nib = (value >> shift) & (uint32_t)k_ez_nibble_mask;
337 buf[i] = (uint8_t)((nib < (uint32_t)k_ez_dec_ten) ? ('0' + nib) : ('A' + (nib - k_ez_dec_ten)));
338 }
339 ez_print(buf, (uint32_t)k_ez_hex_nibbles);
340}

343static void ez_print_uint(uint32_t value)
344{
345 uint8_t buf[k_ez_dec_ten];
346 uint32_t n = 0U;
347 if (value == 0U) {
348 buf[n] = '0';
349 n++;
350 }
351 while ((value > 0U) && (n < (uint32_t)k_ez_dec_ten)) {
352 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_ez_dec_ten));
353 n++;
354 value /= (uint32_t)k_ez_dec_ten;
355 }
356 for (uint32_t i = 0U; i < n; ++i) {
357 ez_print(&buf[n - 1U - i], 1U);
358 }
359}

427static void ez_flush(const ez_present_t* plan)
428{
429 const display_rect_t rect = {
430 .x = (uint16_t)plan->rect.x,
431 .y = (uint16_t)plan->rect.y,
432 .w = (uint16_t)plan->rect.w,
433 .h = (uint16_t)plan->rect.h,
434 };
435 /* The engine's fast/quality split IS the e-ink waveform choice: a pan burst
436 * flushes A2 (bi-level, ~10x faster than GC16) and the settle repaints the
437 * same rectangle in all 16 levels. On this GLCDC panel the hint is inert;
438 * the e-ink backend honours it. */
439 (void)display_flush(s_display,
440 rect,
441 plan->quality ? k_display_refresh_quality : k_display_refresh_fast);
442}

491static void ez_poll_touch(bool* was_touching)
492{
493 ra8_touch_point_t pt = {};
494 uint8_t got = 0U;
495 if (ra8_touch_read(&pt, 1U, &got) != k_ra8_ok) {
496 return;
497 }
498 const bool touching = (got > 0U);
499 if (touching && !(*was_touching)) {
500 if (ez_scene_tap(&s_scene, (int32_t)pt.x, (int32_t)pt.y, ra8_time_ms())) {
501 (void)ez_redraw();
502 }
503 }
504 *was_touching = touching;
505}

537int32_t main(void)
538{
539 ez_bringup_clocks();
540 ez_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
541 ez_bringup_panel();
542 ez_bringup_touch();
543
544 if (!ez_setup_scene()) {
545 ez_panic_halt(k_msg_scene, (uint32_t)sizeof(k_msg_scene) - 1U);
546 }
547
548 ez_selftest_t st = {};
549 if (ez_scene_selftest(&s_scene, &st) != k_ra8_ok) {
550 ez_panic_halt(k_msg_rend, (uint32_t)sizeof(k_msg_rend) - 1U);
551 }
552 /* The self-check consumed every pending plan, so publish its final frame
553 * explicitly: one clean full-quality flush of the content area. */
554 const ez_present_t final_plan = {
555 .rect = ez_content_rect((int32_t)k_ez_fb_w, (int32_t)k_ez_fb_h),
556 .quality = true,
557 .present = true,
558 };
559 ez_flush(&final_plan);
560 ez_print_banner(&st);
561
562 bool was_touching = false;
563 uint32_t ticks = 0U;
564 while (1) {
565 ez_poll_touch(&was_touching);
566 /* Settle promotion: once a gesture burst has been still for the view's
567 * settle window, repaint the same rectangle at full 16-level quality. */
568 if (ez_scene_tick(&s_scene, ra8_time_ms())) {
569 (void)ez_redraw();
570 }
571 if ((ticks % (uint32_t)k_ez_led_every) == 0U) {
572 (void)ra8_board_led_toggle(k_ra8_board_led_blue);
573 }
574 ticks++;
575 ra8_delay_ms((uint32_t)k_ez_frame_ms);
576 }
577 return 0;
578}
