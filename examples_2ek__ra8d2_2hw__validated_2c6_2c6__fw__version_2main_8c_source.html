310static void c6_fwver_heartbeat(bool passed)
311{
312 uint32_t beat = 0U;
313 while (1) {
314 c6_fwver_puts("c6_fwver: heartbeat n=");
315 c6_fwver_put_u32(beat);
316 c6_fwver_puts(passed ? " verdict=PASS\r\n" : " verdict=FAIL\r\n");
317 beat++;
318 tx_thread_sleep((ULONG)k_c6_fwver_heartbeat_ms);
319 }

347 c6_fwver_heartbeat(false);

350
351 ra8_delay_ms((uint32_t)k_c6_fwver_boot_wait_ms);
352 c6_fwver_phase_caps();
353 c6_fwver_phase_request();
354 const bool passed = c6_fwver_rpc_report();
355 ra8_esp_hosted_mem_dump("c6_fw_version");
356 c6_fwver_heartbeat(passed);
357}
