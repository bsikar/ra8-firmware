145static ra8_err_t demo_read_once(const uint8_t* expect)
146{
147 RA8_CHECK_NULL_PTR(expect, s_tag, "expect");
148 ra8_fs_file_t* f = nullptr;
149 RA8_RETURN_ON_ERROR(ra8_io_vfs_open("ram:/HELLO.TXT", k_ra8_fs_mode_read, &f), s_tag, "open");
150 uint8_t got[(size_t)k_demo_payload] = {};
151 uint32_t got_len = 0;
152 RA8_RETURN_ON_ERROR(ra8_fs_read(f, got, (uint32_t)k_demo_payload, &got_len), s_tag, "read");
153 RA8_RETURN_ON_ERROR(ra8_fs_close(f), s_tag, "close");
154 if (got_len != (uint32_t)k_demo_payload) {
155 return k_ra8_err_invalid_size;
156 }
157 if (memcmp(got, expect, (size_t)k_demo_payload) != 0) {
158 return k_ra8_err_checksum_mismatch;
159 }
160 return k_ra8_ok;
161}

164static ra8_err_t demo_run(uint32_t* out_hits, uint32_t* out_misses)
165{
166 RA8_CHECK_NULL_PTR(out_hits, s_tag, "out_hits");
167 RA8_CHECK_NULL_PTR(out_misses, s_tag, "out_misses");
168 ra8_fs_mount_t* mnt = nullptr;
169 RA8_RETURN_ON_ERROR(demo_mount(&mnt), s_tag, "mount stage");
170
171 uint8_t data[(size_t)k_demo_payload];
172 for (uint32_t i = 0; i < (uint32_t)k_demo_payload; ++i) {
173 data[i] = (uint8_t)(i * (uint32_t)k_demo_seed_mul + 1U);
174 }
175 RA8_RETURN_ON_ERROR(ra8_fs_write_file(mnt, "HELLO.TXT", data, (uint32_t)k_demo_payload),
176 s_tag,
177 "write");
178
179 for (uint32_t pass = 0; pass < (uint32_t)k_demo_reads; ++pass) {
180 RA8_RETURN_ON_ERROR(demo_read_once(data), s_tag, "reread");
181 }
182
183 RA8_RETURN_ON_ERROR(ra8_io_blockdev_cache_stats(&s_cstate, out_hits, out_misses), s_tag, "stats");
184 if (*out_hits == 0U) {
185 return k_ra8_err_checksum_mismatch;
186 }
187 return k_ra8_ok;
188}

217 while (true) {
218 }
