284[[nodiscard]] static ra8_err_t fileops_run_ladder(ra8_usb_speed_t speed)
285{
286 ra8_err_t err = fileops_print("ra8d2 fileops: probing ");
287 if (err != k_ra8_ok) {
288 return err;
289 }
290 err = fileops_print(fileops_port_name(speed));
291 if (err != k_ra8_ok) {
292 return err;
293 }
294 err = fileops_print("\r\n");
295 if (err != k_ra8_ok) {
296 return err;
297 }
298 err = ra8_usb_hmsc_init(speed);
299 if (err != k_ra8_ok) {
300 (void)fileops_print_fail("host init", err);
301 return err;
302 }
303
304 ra8_usb_hmsc_device_t device = {};
305 err = ra8_usb_hmsc_enumerate(&device);
306 if (err != k_ra8_ok) {
307 (void)ra8_usb_hmsc_close();
308 return err;
309 }
310 err = fileops_print_attach(&device, speed);
311 if (err != k_ra8_ok) {
312 (void)ra8_usb_hmsc_close();
313 return err;
314 }
315
316 ra8_fs_mount_t* mount = nullptr;
317 err = fileops_mount_volume(&mount);
318 if (err != k_ra8_ok) {
319 fileops_probe_layout();
320 (void)ra8_usb_hmsc_close();
321 return err;
322 }
323
324 err = fileops_suite_create(mount);
325 if (err == k_ra8_ok) {
326 err = fileops_suite_mutate(mount);
327 }
328 (void)ra8_fs_unmount(mount);
329 if (err != k_ra8_ok) {
330 (void)ra8_usb_hmsc_close();
331 return err;
332 }
333
334 err = fileops_print("ra8d2 fileops: ALL FILE OPS PASSED\r\n");
335 if (err != k_ra8_ok) {
336 return err;
337 }
338 (void)ra8_board_led_on(k_ra8_board_led1);
339 return k_ra8_ok;
340}

362int32_t main(void)
363{
364 fileops_setup_or_halt();
365
366 ra8_isr_globals_enable();
367
368 if (fileops_print("ra8d2 fileops: ready, plug a USB drive into either USB jack\r\n") !=
369 k_ra8_ok) {
370 fileops_panic_halt();
371 }
372
373 /* Alternate ports until the suite fully passes: a drive inserted in
374 * either jack (or reseated) is picked up by a following cycle. */
375 ra8_usb_speed_t speed = k_ra8_usb_speed_hs;
376 ra8_err_t err = fileops_run_ladder(speed);
377 while (err != k_ra8_ok) {
378 (void)fileops_print_fail("ladder", err);
379 (void)fileops_print("ra8d2 fileops: trying the other port in 5 s\r\n");
380 ra8_delay_ms(k_fileops_retry_ms);
381 if (speed == k_ra8_usb_speed_hs) {
382 speed = k_ra8_usb_speed_fs;
383 } else {
384 speed = k_ra8_usb_speed_hs;
385 }
386 err = fileops_run_ladder(speed);
387 }
388
389 while (1) {
390 ra8_delay_ms(k_fileops_idle_ms);
391 __asm__ volatile("wfi");
392 }
393
394 return 0;
395}
