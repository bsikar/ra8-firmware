209static void motor_3phase_build_sine(void)
210{
211 /* Use a Q16 representation of pi and 2*pi for the Bhaskara math. */
212 enum : uint64_t {
213 k_q16_pi = 205887ULL,
214 k_q16_two_pi = 411775ULL,
215 k_q16_5_pi_sq = 3236018ULL,
216 k_q16_one = 65536ULL,
217 k_q16_numer_k = 16ULL,
218 k_q16_denom_k = 4ULL,
219 };
220 const uint64_t period_u64 = k_motor_3phase_period;
221 const uint64_t half_period = k_motor_3phase_half_period;
222
223 for (uint16_t i = 0U; i < k_motor_3phase_sine_size; i++) {
224 const uint64_t theta = ((uint64_t)i * k_q16_two_pi) / k_motor_3phase_sine_size;
225 bool neg = false;
226 uint64_t x = theta;
227 if (x > k_q16_pi) {
228 x = x - k_q16_pi;
229 neg = true;
230 }
231 const uint64_t pi_minus_x = k_q16_pi - x;
232 const uint64_t numer = k_q16_numer_k * x * pi_minus_x;
233 const uint64_t denom_part = k_q16_denom_k * x * pi_minus_x;
234 const uint64_t denom = k_q16_5_pi_sq - denom_part;
235 const uint64_t sin_q16 = (denom > 0ULL) ? (numer / denom) : 0ULL;
236 const uint64_t scaled = (sin_q16 * half_period) / k_q16_one;
237 uint64_t duty = neg ? (half_period - scaled) : (half_period + scaled);
238 if (duty > period_u64) {
239 duty = period_u64;
240 }
241 s_motor_3phase_sine[i] = (uint32_t)duty;
242 }
243}

441static uint32_t motor_3phase_u32_to_dec(uint32_t value, uint8_t* buf)
442{
443 enum : uint8_t {
444 k_ascii_zero = 0x30U,
445 k_radix = 10U,
446 k_max_digits = 10U,
447 };
448 uint8_t tmp[k_max_digits];
449 uint32_t n = 0U;
450 if (value == 0U) {
451 buf[0] = k_ascii_zero;
452 return 1U;
453 }
454 while (value > 0U) {
455 tmp[n] = (uint8_t)(k_ascii_zero + (uint8_t)(value % k_radix));
456 n++;
457 value /= k_radix;
458 }
459 for (uint32_t i = 0U; i < n; i++) {
460 buf[i] = tmp[n - 1U - i];
461 }
462 return n;
463}

518static void motor_3phase_advance(uint32_t* out_u, uint32_t* out_v, uint32_t* out_w)
519{
520 const uint16_t i_u = s_motor_3phase_step;
521 const uint16_t i_v =
522 (uint16_t)((s_motor_3phase_step + k_motor_3phase_phase_120) & k_motor_3phase_sine_mask);
523 const uint16_t i_w =
524 (uint16_t)((s_motor_3phase_step + k_motor_3phase_phase_240) & k_motor_3phase_sine_mask);
525 *out_u = s_motor_3phase_sine[i_u];
526 *out_v = s_motor_3phase_sine[i_v];
527 *out_w = s_motor_3phase_sine[i_w];
528 s_motor_3phase_step = (uint16_t)((s_motor_3phase_step + 1U) & k_motor_3phase_sine_mask);
529}

546int32_t main(void)
547{
548 motor_3phase_init_clocks_and_led();
549 motor_3phase_init_console();
550 motor_3phase_init_pwm();
551
552 ra8_isr_globals_enable();
553
554 uint32_t print_accum_ms = 0U;
555
556 while (1) {
557 uint32_t u_duty = 0U;
558 uint32_t v_duty = 0U;
559 uint32_t w_duty = 0U;
560
561 motor_3phase_advance(&u_duty, &v_duty, &w_duty);
562
563 if (ra8_gpt_three_phase_set_duty(u_duty, v_duty, w_duty) != k_ra8_ok) {
564 break;
565 }
566
567 print_accum_ms += k_motor_3phase_update_ms;
568 if (print_accum_ms >= k_motor_3phase_print_period) {
569 print_accum_ms = 0U;
570 motor_3phase_print_duty(u_duty, v_duty, w_duty);
571 (void)ra8_board_led_toggle(k_ra8_board_led1);
572 }
573
574 ra8_delay_ms(k_motor_3phase_update_ms);
575 }
576
577 motor_3phase_panic_halt();
578 return 0;
579}
