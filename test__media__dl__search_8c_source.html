75static void test_hits_parse(void)
76{
77 TEST_BEGIN("search hits parse");
78 const ra8_err_t rc = mdl_extract_hits(k_search_html,
79 sizeof(k_search_html) - 1U,
80 "https://manhwaus.net/?s=solo",
81 "/webtoon/",
82 &s_hits);
83 TEST_ASSERT(rc == k_ra8_ok);
84 TEST_ASSERT_EQ((uint16_t)k_hits_expected, s_hits.count);
85 TEST_ASSERT_EQ((uint16_t)k_anchors_expected, s_hits.anchors_seen);
86 /* First card: thumbnail link (no text) then the titled link to the SAME URL;
87 * the real title upgrades the slug fallback captured from the thumbnail. */
88 TEST_ASSERT(strcmp(s_hits.hits[0].url, "https://manhwaus.net/webtoon/solo-leveling/") == 0);
89 TEST_ASSERT(strcmp(s_hits.hits[0].title, "Solo Leveling") == 0);
90 /* Second card: title comes from the `title=` attribute. */
91 TEST_ASSERT(strcmp(s_hits.hits[1].url, "https://manhwaus.net/webtoon/tower-of-god/") == 0);
92 TEST_ASSERT(strcmp(s_hits.hits[1].title, "Tower of God") == 0);
93 /* External hit: absolute URL kept, `&amp;` decoded in the title. */
94 TEST_ASSERT(strcmp(s_hits.hits[2].url, "https://cdn.example/webtoon/leak/") == 0);
95 TEST_ASSERT(strcmp(s_hits.hits[2].title, "External & Leak") == 0);
96 TEST_ASSERT(mdl_search_classify(&s_hits) == k_mdl_search_have_results);
97 TEST_END("search hits parse");
98}

122static void test_zero_vs_markup(void)
123{
124 TEST_BEGIN("search zero vs markup");
125 /* Links present, none match: a genuine zero-result. */
126 ra8_err_t rc =
127 mdl_extract_hits(k_zero_html, sizeof(k_zero_html) - 1U, "https://s.net/", "/webtoon/", &s_hits);

132 /* No links at all: the markup changed or the request was blocked. */
133 rc = mdl_extract_hits(k_markup_html,
134 sizeof(k_markup_html) - 1U,
135 "https://s.net/",
136 "/webtoon/",
137 &s_hits);
138 TEST_ASSERT(rc == k_ra8_ok);
139 TEST_ASSERT_EQ((uint16_t)0, s_hits.count);
140 TEST_ASSERT_EQ((uint16_t)0, s_hits.anchors_seen);
141 TEST_ASSERT(mdl_search_classify(&s_hits) == k_mdl_search_markup_changed);
142 /* A NULL list is treated as unreadable, never as a clean empty result. */
143 TEST_ASSERT(mdl_search_classify(nullptr) == k_mdl_search_markup_changed);
144 TEST_END("search zero vs markup");

160static void test_query_encode(void)
161{
162 TEST_BEGIN("query encode");
163 char buf[k_buf];
164 TEST_ASSERT(mdl_query_encode("hello world", buf, sizeof(buf)));
165 TEST_ASSERT(strcmp(buf, "hello%20world") == 0);
166 TEST_ASSERT(mdl_query_encode("a&b#c+d/e?f", buf, sizeof(buf)));
167 TEST_ASSERT(strcmp(buf, "a%26b%23c%2Bd%2Fe%3Ff") == 0);
168 TEST_ASSERT(mdl_query_encode("Az0-._~", buf, sizeof(buf)));
169 TEST_ASSERT(strcmp(buf, "Az0-._~") == 0); /* unreserved set copied verbatim */
170 /* A UTF-8 term (source stays 7-bit ASCII via hex escapes): each byte -> %HH. */
171 TEST_ASSERT(mdl_query_encode("caf\xC3\xA9", buf, sizeof(buf)));
172 TEST_ASSERT(strcmp(buf, "caf%C3%A9") == 0);
173 /* Too small to hold "a%20b" (needs 6 incl NUL): a clean false, not a truncation. */
174 TEST_ASSERT(!mdl_query_encode("a b", buf, k_tiny_buf));
175 TEST_ASSERT(buf[0] == '\0');
176 TEST_ASSERT(!mdl_query_encode(nullptr, buf, sizeof(buf)));
177 TEST_END("query encode");
178}

194static void test_build_url(void)
195{
196 TEST_BEGIN("search build url");
197 char buf[k_buf];
198 TEST_ASSERT(mdl_search_build_url("https://s.net/?s={q}", "solo%20x", buf, sizeof(buf)));
199 TEST_ASSERT(strcmp(buf, "https://s.net/?s=solo%20x") == 0);
200 /* Every placeholder is substituted, not just the first. */
201 TEST_ASSERT(mdl_search_build_url("https://s.net/{q}/?s={q}", "ab", buf, sizeof(buf)));
202 TEST_ASSERT(strcmp(buf, "https://s.net/ab/?s=ab") == 0);
203 /* A template with no {q} could never carry the term: rejected, not silent. */
204 TEST_ASSERT(!mdl_search_build_url("https://s.net/search", "ab", buf, sizeof(buf)));
205 TEST_ASSERT(buf[0] == '\0');
206 /* Overflow is a clean false, never a truncated request. */
207 TEST_ASSERT(!mdl_search_build_url("https://s.net/?s={q}", "abcdefghij", buf, k_tiny_buf));
208 /* Each NULL / zero-cap condition independently forces a false (MC/DC above). */
209 TEST_ASSERT(!mdl_search_build_url(nullptr, "ab", buf, sizeof(buf)));
210 TEST_ASSERT(!mdl_search_build_url("https://s.net/?s={q}", nullptr, buf, sizeof(buf)));
211 TEST_ASSERT(!mdl_search_build_url("https://s.net/?s={q}", "ab", nullptr, sizeof(buf)));
212 TEST_ASSERT(!mdl_search_build_url("https://s.net/?s={q}", "ab", buf, 0U));
213 TEST_END("search build url");
214}
