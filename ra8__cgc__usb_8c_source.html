258ra8_err_t ra8_cgc_pll2_enable(uint8_t mul_int, uint8_t mul_quarters, ra8_plodiv_t p_div_code)
259{
260 if (mul_int == 0U) {
261 return k_ra8_err_invalid_arg;
262 }
263 if ((uint16_t)mul_quarters > (uint16_t)k_ra8_pll2_max_quarters) {
264 return k_ra8_err_invalid_arg;
265 }
266
267 ra8_log_info_val(s_tag, "pll2 enable mul_int", (uint32_t)mul_int);
268
269 /* Idempotency (HUM Ch 9): never re-program a running PLL2 -- if FS
270 * path already locked it, the HS path's "same VCO" request is a no-op. */
271 if ((*ra8_sys_oscsf() & (uint8_t)(1U << k_ra8_oscsf_bit_pll2sf)) != 0U) {
272 ra8_log_info(s_tag, "pll2 already locked -- skip re-program");
273 return k_ra8_ok;
274 }
275
276 /* PLL2CCR (32-bit, HUM Ch 9.2.9): PLL2MULNF[7:6] | PLL2MUL[16:8]. */
277 const uint32_t mul_quarters_field =
278 ((uint32_t)mul_int * (uint32_t)k_ra8_cgc_quarters_per_unit) + (uint32_t)mul_quarters;
279 const uint32_t pll2ccr =
280 ((mul_quarters_field & (uint32_t)k_ra8_pll2ccr_mask_quarters) << k_ra8_pllccr_shift_quarters) |
281 ((uint32_t)k_ra8_plsrcsel_main << k_ra8_pllccr_shift_plsrcsel) |
282 ((uint32_t)k_ra8_plidiv_div2 & (uint32_t)k_ra8_pll2ccr_mask_plidiv);
283
284 /* PLL2CCR2: P from caller; Q/R fixed at /6. Code 0 is prohibited
285 * per HUM Ch 9.2.10/9.2.12 (drops the whole 16-bit write). */
286 const uint16_t pll2ccr2 =
287 (uint16_t)(((uint16_t)k_ra8_plodiv_div6 << k_ra8_pllccr2_shift_plodivr) |
288 ((uint16_t)k_ra8_plodiv_div6 << k_ra8_pllccr2_shift_plodivq) |
289 ((uint16_t)p_div_code << k_ra8_pllccr2_shift_plodivp));
290
291 const ra8_err_t err = internal_pll2_program_protected(pll2ccr, pll2ccr2);
292 if (err != k_ra8_ok) {
293 return err;
294 }
295 ra8_log_info(s_tag, "pll2 locked");
296 return k_ra8_ok;
297}

364ra8_err_t ra8_cgc_usbfs_clock_enable(
365 void) // NOLINT(readability-function-size,readability-function-cognitive-complexity)
366{
367 ra8_log_info(s_tag, "usbfs clock enable");
368
369 /* Step 1: bring up PLL2 to land 240 MHz on PLL2P. */
370 const ra8_err_t pll2_err = ra8_cgc_pll2_enable((uint8_t)k_ra8_pll2_usbfs_mul,
371 (uint8_t)k_ra8_pll2_usbfs_quarters,
372 k_ra8_plodiv_div4);
373 if (pll2_err != k_ra8_ok) {
374 ra8_log_error_val(s_tag, "usbfs: pll2 enable failed", (uint32_t)pll2_err);
375 return pll2_err;
376 }
377
378 /* Step 2a: Force USBFS into module-stop (MSTPB11 = 1) BEFORE the SREQ
379 * handshake. HUM Ch 9 "Clock selection switching procedure" step 1
380 * (p367 / p370 / p373 / p378 / p381) is mandatory whenever USBCKDIVCR
381 * is moved off 1/1; we are programming /5 below. MSTPCR is NOT
382 * PRCR-protected. ra8_usb_device_init() releases MSTPB11 again later
383 * via the ref-counted ra8_mstp_enable() path. */
384 ra8_mstp()->MSTPCRB |= (uint32_t)k_ra8_mstpb11_usbfs_mask;
385 (void)ra8_mstp()->MSTPCRB; /* HUM 11.2.7 Note 2: read-back. */
386
387 /* Step 2a': Ensure HOCO is running. USBCKCR's reset-default source is
388 * HOCO; the SREQ -> SRDY synchronizer crosses from ICLK into the
389 * CURRENT UCK source's clock domain (HOCO). If HOCO is stopped
390 * (HCSTP=1, the chip reset state) the handshake never completes -- we
391 * observed USBCKCR=0x40 (SREQ=1, SRDY=0) with HOCOCR=0x01 and
392 * OSCSF.HOCOSF=0 on real silicon. ra8_cgc_init() leaves HOCO stopped
393 * because it switches SCKSCR to PLL1; we have to start HOCO here so
394 * the source clock for USBCKCR is alive when SREQ is asserted.
395 * HOCOCR IS PRCR-protected (HUM Ch 9; FSP wraps writes inside its
396 * BSP_PRV_PRCR_UNLOCK / BSP_PRV_PRCR_LOCK window -- writes outside
397 * the window are silently dropped, which is what we observed when
398 * an earlier attempt left HOCOCR at 0x01 after the store). */
399 ra8_err_t hoco_err = k_ra8_ok;
400 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
401 {
402 volatile uint8_t* const hococr = ra8_sys_hococr();
403 if ((*hococr & (uint8_t)(1U << k_ra8_hococr_hcstp)) != 0U) {
404 *hococr = (uint8_t)((uint8_t)*hococr & (uint8_t)~(1U << k_ra8_hococr_hcstp));
405 }
406 }
407 /* OSCSF.HOCOSF poll outside PRCR window (read-only register). */
408 hoco_err = ra8_cgc_wait_oscsf_set(k_ra8_oscsf_bit_hocosf);
409 if (hoco_err != k_ra8_ok) {
410 ra8_log_error(s_tag, "usbfs: HOCO stabilization timeout");
411 return hoco_err;
412 }
413 ra8_log_info(s_tag, "usbfs: HOCO running for USBCKCR source");
414
415 /* Step 2b: USBCKCR / USBCKDIVCR handshake. */
416 const ra8_err_t err = internal_usbckcr_switch_to_pll2p_div5();
417 if (err != k_ra8_ok) {
418 return err;
419 }
420 ra8_log_info(s_tag, "usbfs clock ready (PLL2P/5 = 48 MHz)");
421 return k_ra8_ok;
422}

518static ra8_err_t internal_usb60ckcr_switch_to_pll2p_div4(void)
519{
520 ra8_err_t err = k_ra8_ok;
521 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
522 {
523 const uint8_t sreq_mask = (uint8_t)(1U << k_ra8_usbckcr_bit_sreq);
524 const uint8_t srdy_mask = (uint8_t)(1U << k_ra8_usbckcr_bit_srdy);
525 /* FSP `bsp_clocks.c` ::bsp_peripheral_clock_set sequence (HUM Ch 9
526 * "Clock selection switching procedure"):
527 *
528 * 1. Set SREQ=1 with read-modify-write so the current SEL field
529 * is preserved (the reset-default source is HOCO; clobbering
530 * SEL to 0 also writes "HOCO" but loses the invariant that
531 * SREQ is asserted on top of the EXISTING source -- this
532 * matters when the previous source is no longer running).
533 * 2. Spin-wait SRDY=1 (handshake acknowledge).
534 * 3. Programme USB60CKDIVCR (peripheral clock now stopped).
535 * 4. Write `source | SREQ | SRDY` -- the SRDY bit being set in
536 * the same write is mandatory per FSP (line 2896 of
537 * `bsp_clocks.c`); on RA8 Gen2 silicon, omitting SRDY here
538 * makes the subsequent SREQ-clear hang at step 6.
539 * 5. Clear SREQ via read-modify-write (peripheral clock starts).
540 * 6. Spin-wait SRDY=0 (start acknowledged). */
541 volatile uint8_t* const ckcr = ra8_sys_usb60ckcr();
542 *ckcr = (uint8_t)(*ckcr | sreq_mask);
543 err = internal_wait_usb60cksrdy(1U);
544 if (err != k_ra8_ok) {
545 ra8_log_error(s_tag, "usbhs: SRDY=1 timeout");
546 break;
547 }
548 *ra8_sys_usb60ckdivcr() = (uint8_t)k_ra8_usbhs_div4_code;
549 const uint8_t src = (uint8_t)((uint8_t)k_ra8_usbcksel_pll2p & k_ra8_usbckcr_mask_sel);
550 *ckcr = (uint8_t)(src | sreq_mask | srdy_mask);
551 *ckcr = (uint8_t)(*ckcr & (uint8_t)~sreq_mask);
552 err = internal_wait_usb60cksrdy(0U);
553 if (err != k_ra8_ok) {
554 ra8_log_error(s_tag, "usbhs: SRDY=0 timeout");
555 break;
556 }
557 }
558 /* Diagnostic witnesses -- read OUTSIDE the PRCR window since
559 * USB60CKCR / OSCSF reads are not protected. */
560 s_usb60ckcr_probe = *ra8_sys_usb60ckcr();
561 s_pll2_status_probe = *ra8_sys_oscsf();
562 return err;
563}

625ra8_err_t ra8_cgc_usbhs_pll_enable(
626 void) // NOLINT(readability-function-size,readability-function-cognitive-complexity)
627{
628 ra8_log_info(s_tag, "usbhs phy clock enable");
629
630 /* Step 1: HUM Ch 9.2.21 "OSCSF : Oscillation Stabilization Flag
631 * Register", p 344. The USBHS PHY's 12 MHz reference is derived
632 * from the main XTAL; without MOSCSF=1 the PHY cannot lock. */
633 const ra8_err_t osc_err = ra8_cgc_wait_oscsf_set(k_ra8_oscsf_bit_moscsf);
634 if (osc_err != k_ra8_ok) {
635 ra8_log_error_val(s_tag, "usbhs: main xtal not stable", (uint32_t)osc_err);
636 return osc_err;
637 }
638
639 /* Step 2: bring up PLL2 to land 240 MHz on PLL2P. Reuse the FS path's
640 * configuration -- 24 MHz XTAL / PL2IDIV(/2) * MUL(80) = 960 MHz VCO,
641 * PL2ODIVP=/4 -> PLL2P = 240 MHz. PLL2P / 4 = 60.000 MHz on USB60CLK
642 * exactly (0 ppm vs the USBHS 60 MHz spec target). ra8_cgc_pll2_enable
643 * is idempotent w.r.t. the PRCR window and tolerates being called
644 * once per USB controller bring-up (FS path also calls it). */
645 const ra8_err_t pll2_err = ra8_cgc_pll2_enable((uint8_t)k_ra8_pll2_usbfs_mul,
646 (uint8_t)k_ra8_pll2_usbfs_quarters,
647 k_ra8_plodiv_div4);
648 if (pll2_err != k_ra8_ok) {
649 ra8_log_error_val(s_tag, "usbhs: pll2 enable failed", (uint32_t)pll2_err);
650 return pll2_err;
651 }
652
653 /* Step 3: Force USBHS into module-stop (MSTPB12 = 1) BEFORE the SREQ
654 * handshake. HUM Ch 9 "Clock selection switching procedure" step 1 is
655 * mandatory whenever USB60CKDIVCR is moved off 1/1 (we are programming
656 * /4 below). MSTPCR is NOT PRCR-protected. ra8_usb_device_init() will
657 * release MSTPB12 again later via the ref-counted ra8_mstp_enable
658 * path for k_ra8_mstp_usbhs. */
659 ra8_mstp()->MSTPCRB |= (uint32_t)k_ra8_mstpb12_usbhs_mask;
660 (void)ra8_mstp()->MSTPCRB; /* HUM 11.2.7 Note 2: read-back. */
661
662 /* Step 4: Ensure HOCO is running so the SREQ->SRDY synchronizer can
663 * drain (USB60CKCR's reset-default source is HOCO). */
664 const ra8_err_t hoco_err = ra8_cgc_ensure_hoco_running_for_usb_ck();
665 if (hoco_err != k_ra8_ok) {
666 ra8_log_error(s_tag, "usbhs: HOCO stabilization timeout");
667 return hoco_err;
668 }
669
670 /* Step 5: USB60CKCR / USB60CKDIVCR SREQ/SRDY handshake. */
671 const ra8_err_t err = internal_usb60ckcr_switch_to_pll2p_div4();
672 if (err != k_ra8_ok) {
673 return err;
674 }
675
676 ra8_log_info(s_tag, "usbhs phy clock ready (PLL2P/4 = 60 MHz)");
677 return k_ra8_ok;
678}
