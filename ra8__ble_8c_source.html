235ra8_err_t ra8_ble_hci_send_command(uint16_t opcode, const uint8_t* params, uint8_t params_len)
236{
237 if (s_state.open == 0U) {
238 return k_ra8_err_not_initialized;
239 }
240 if ((params == nullptr) && (params_len > 0U)) {
241 return k_ra8_err_null_ptr;
242 }
243 /* params_len is uint8_t so it cannot exceed k_ra8_ble_max_cmd_params (255). */
244
245 internal_tx_byte(k_ra8_ble_pkt_cmd);
246 internal_tx_byte((uint8_t)(opcode & k_ra8_ble_byte_mask));
247 internal_tx_byte((uint8_t)((opcode >> k_ra8_ble_byte_shift) & k_ra8_ble_byte_mask));
248 internal_tx_byte(params_len);
249 for (uint16_t i = 0U; i < params_len; ++i) {
250 internal_tx_byte(params[i]);
251 }
252 return k_ra8_ok;
253}

310static ra8_err_t internal_dispatch_event(void)
311{
312 uint8_t code = 0U;
313 uint8_t plen = 0U;
314 // mcdc-deactivated: TU-local helper internal_dispatch_event; HCI byte-stream RX guard against truncated event headers. Reaching this requires the producer to enqueue a partial header, which the HCI transport delivers as atomic packet boundaries -- the second short-circuit condition is unreachable on any well-formed link.
315 if ((internal_rx_byte(&code) == 0U) || (internal_rx_byte(&plen) == 0U)) {
316 return k_ra8_err_invalid_arg;
317 }
318 static uint8_t s_evt_scratch[k_ra8_ble_max_evt_params];
319 for (uint8_t i = 0U; i < plen; ++i) {
320 if (internal_rx_byte(&s_evt_scratch[i]) == 0U) {
321 return k_ra8_err_invalid_arg;
322 }
323 }
324 if (s_state.evt_fn != nullptr) {
325 s_state.evt_fn(s_state.evt_ctx, code, s_evt_scratch, plen);
326 }
327 return k_ra8_ok;

343static ra8_err_t internal_dispatch_acl(void)
344{
345 uint8_t hdl_lo = 0U;
346 uint8_t hdl_hi = 0U;
347 uint8_t len_lo = 0U;
348 uint8_t len_hi = 0U;
349 // mcdc-deactivated: TU-local helper internal_dispatch_acl; 4-byte HCI ACL header is delivered atomically by the transport. Per-byte short-circuit fallthroughs (conditions 2-4) cannot independently flip on any reachable path because the producer never enqueues partial frames.
350 if ((internal_rx_byte(&hdl_lo) == 0U) || (internal_rx_byte(&hdl_hi) == 0U) ||
351 (internal_rx_byte(&len_lo) == 0U) || (internal_rx_byte(&len_hi) == 0U)) {
352 return k_ra8_err_invalid_arg;
353 }
354 const uint16_t handle = (uint16_t)(((uint16_t)hdl_hi << k_ra8_ble_byte_shift) | (uint16_t)hdl_lo);
355 const uint16_t len = (uint16_t)(((uint16_t)len_hi << k_ra8_ble_byte_shift) | (uint16_t)len_lo);
356 if (len > k_ra8_ble_max_acl_payload) {
357 return k_ra8_err_invalid_arg;
358 }
359 static uint8_t s_acl_scratch[k_ra8_ble_max_acl_payload];
360 for (uint16_t i = 0U; i < len; ++i) {
361 if (internal_rx_byte(&s_acl_scratch[i]) == 0U) {
362 return k_ra8_err_invalid_arg;
363 }
364 }
365 if (s_state.acl_fn != nullptr) {
366 s_state.acl_fn(s_state.acl_ctx, handle, s_acl_scratch, len);
367 }
368 return k_ra8_ok;
369}

371ra8_err_t ra8_ble_dispatch(void)
372{
373 if (s_state.open == 0U) {
374 return k_ra8_err_not_initialized;
375 }
376 for (uint32_t budget = 0U; budget < k_ra8_ble_dispatch_budget;
377 ++budget) { /* GCOVR_EXCL_BR_LINE */
378 uint8_t pkt_type = 0U;
379 if (internal_rx_byte(&pkt_type) == 0U) { /* GCOVR_EXCL_BR_LINE */
380 return k_ra8_ok;
381 }
382 if (pkt_type == k_ra8_ble_pkt_event) {
383 const ra8_err_t err = internal_dispatch_event();
384 if (err != k_ra8_ok) {
385 return err;
386 }
387 } else if (pkt_type == k_ra8_ble_pkt_acl_data) {
388 const ra8_err_t err = internal_dispatch_acl();
389 if (err != k_ra8_ok) {
390 return err;
391 }
392 } else {
393 return k_ra8_err_invalid_arg;
394 }
395 }
396 return k_ra8_ok;
397}

410ra8_err_t ra8_ble_set_advertising_data(const uint8_t* data, uint8_t len)
411{
412 if (len > k_ra8_ble_adv_data_max) {
413 return k_ra8_err_invalid_arg;
414 }
415 if ((data == nullptr) && (len > 0U)) {
416 return k_ra8_err_null_ptr;
417 }
418 /* Spec frames the command as: [len:1][data:31][padding:31-len].
419 * Most stacks pad with zeros; here we forward only the meaningful
420 * prefix to keep the framing test simple and let the controller
421 * pad if it cares. */
422 uint8_t buf[1U + k_ra8_ble_adv_data_max] = {};
423 buf[0] = len;
424 if (len > 0U) {
425 internal_byte_copy(&buf[1], data, len);
426 }
427 return ra8_ble_hci_send_command(k_ra8_ble_op_le_set_adv_data, buf, (uint8_t)(1U + len));
428}

436ra8_err_t ra8_ble_scan_start(uint8_t active, uint16_t interval, uint16_t window)
437{
438 if ((interval < k_ra8_ble_scan_min) || (interval > k_ra8_ble_scan_max)) {
439 return k_ra8_err_invalid_arg;
440 }
441 if ((window < k_ra8_ble_scan_min) || (window > k_ra8_ble_scan_max)) {
442 return k_ra8_err_invalid_arg;
443 }
444 if (window > interval) {
445 return k_ra8_err_invalid_arg;
446 }
447 if (s_state.open == 0U) {
448 return k_ra8_err_not_initialized;
449 }
450
451 /* Bluetooth Core 5.3 Vol 4 Part E 7.8.10 LE_Set_Scan_Parameters
452 * params: scan_type(1) | interval(2) | window(2) | own_addr(1) | filter(1) */
453 uint8_t params[k_ra8_ble_scan_param_bytes];
454 params[k_ra8_ble_scan_off_type] = (active != 0U) ? 1U : 0U;
455 params[k_ra8_ble_scan_off_interval_lo] = (uint8_t)(interval & k_ra8_ble_byte_mask);
456 params[k_ra8_ble_scan_off_interval_hi] =
457 (uint8_t)((interval >> k_ra8_ble_byte_shift) & k_ra8_ble_byte_mask);
458 params[k_ra8_ble_scan_off_window_lo] = (uint8_t)(window & k_ra8_ble_byte_mask);
459 params[k_ra8_ble_scan_off_window_hi] =
460 (uint8_t)((window >> k_ra8_ble_byte_shift) & k_ra8_ble_byte_mask);
461 params[k_ra8_ble_scan_off_own_addr] = k_ra8_ble_scan_own_pub;
462 params[k_ra8_ble_scan_off_filter] = k_ra8_ble_scan_filt_basic;
463
464 ra8_err_t err =
465 ra8_ble_hci_send_command(k_ra8_ble_op_le_set_scan_params, params, k_ra8_ble_scan_param_bytes);
466 if (err != k_ra8_ok) {
467 return err;
468 }
469
470 /* 7.8.11 LE_Set_Scan_Enable: enable(1) | filter_dup(1) */
471 const uint8_t enable_params[k_ra8_ble_scan_enable_bytes] = {1U, 0U};
472 return ra8_ble_hci_send_command(k_ra8_ble_op_le_set_scan_enable,
473 enable_params,
474 (uint8_t)sizeof(enable_params));
475}
