112static ra8_err_t internal_wait_cksrdy(volatile uint8_t* ckcr_reg, uint8_t expected)
113{
114 const uint8_t mask = (uint8_t)(1U << k_ra8_eswckcr_bit_srdy);
115 /* Bounded wait through ra8_hw_err.h: on host tests the ra8_fake_mmio
116 * seam decides the poll (first-poll success unless a test arms a
117 * fault), so the real timeout leg is reachable everywhere. */
118 if (expected != 0U) {
119 return ra8_hw_wait_flag_set8(ckcr_reg, mask, (uint32_t)k_ra8_cgc_eswclk_poll_limit);
120 }
121 return ra8_hw_wait_flag_clear8(ckcr_reg, mask, (uint32_t)k_ra8_cgc_eswclk_poll_limit);
122}

155static ra8_err_t internal_switch_eswcr_to_pll1p(volatile uint8_t* ckcr_reg,
156 volatile uint8_t* divcr_reg,
157 uint8_t div_code)
158{
159 const uint8_t sreq_mask = (uint8_t)(1U << k_ra8_eswckcr_bit_sreq);
160 const uint8_t srdy_mask = (uint8_t)(1U << k_ra8_eswckcr_bit_srdy);
161 /* FSP `bsp_clocks.c` ::bsp_peripheral_clock_set sequence:
162 * 1. Set SREQ=1 (preserve current SEL via read-modify-write).
163 * 2. Spin-wait SRDY=1.
164 * 3. Programme CKDIVCR (peripheral clock now stopped).
165 * 4. Write `source | SREQ | SRDY`. Writing SRDY in the same word
166 * is mandatory per FSP (bsp_clocks.c line 2896); without it
167 * the subsequent SREQ-clear hangs.
168 * 5. Clear SREQ -> peripheral clock starts.
169 * 6. Spin-wait SRDY=0 (start acknowledged). */
170 *ckcr_reg = (uint8_t)(*ckcr_reg | sreq_mask);
171 ra8_err_t err = internal_wait_cksrdy(ckcr_reg, 1U);
172 if (err != k_ra8_ok) {
173 return err;
174 }
175 *divcr_reg = div_code;
176 const uint8_t src = (uint8_t)((uint8_t)k_ra8_eswcksel_pll1p & k_ra8_eswckcr_mask_sel);
177 *ckcr_reg = (uint8_t)(src | sreq_mask | srdy_mask);
178 *ckcr_reg = (uint8_t)(*ckcr_reg & (uint8_t)~sreq_mask);
179 return internal_wait_cksrdy(ckcr_reg, 0U);
180}

231static ra8_err_t internal_eswclk_power_on_domain(void)
232{
233 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_lpm)
234 {
235 volatile uint8_t* const pd = ra8_sys_pdctreswm();
236 /* Only kick the power-on path if the domain is currently gated --
237 * PDCSF=0 and PDPGSF=1 is the FSP-canonical "power off" signature. */
238 const uint8_t state = *pd;
239 if (((state & (1U << k_ra8_pdctr_bit_pdcsf)) == 0U) &&
240 ((state & (1U << k_ra8_pdctr_bit_pdpgsf)) != 0U)) {
241 *pd = (uint8_t)(state & (uint8_t)~(1U << k_ra8_pdctr_bit_pdde));
242 }
243 }
244 /* Wait for power-on outside the PRCR window (the status flags are
245 * read-only and not protected). */
246 ra8_err_t err = internal_wait_pdctreswm_clear(k_ra8_pdctr_bit_pdcsf);
247 if (err != k_ra8_ok) {
248 ra8_log_error(s_tag, "eswclk: PDCSF stuck");
249 return err;
250 }
251 err = internal_wait_pdctreswm_clear(k_ra8_pdctr_bit_pdpgsf);
252 if (err != k_ra8_ok) {
253 ra8_log_error(s_tag, "eswclk: PDPGSF stuck");
254 return err;
255 }
256 return k_ra8_ok;
257}

278static ra8_err_t internal_eswclk_program_cks(void)
279{
280 ra8_err_t err = k_ra8_ok;
281 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_cgc)
282 {
283 err = internal_switch_eswcr_to_pll1p(ra8_sys_eswckcr(),
284 ra8_sys_eswckdivcr(),
285 (uint8_t)k_ra8_eswckdivcr_div4);
286 if (err != k_ra8_ok) {
287 ra8_log_error(s_tag, "eswclk: ESWCKCR handshake timeout");
288 break;
289 }
290 err = internal_switch_eswcr_to_pll1p(ra8_sys_eswpckcr(),
291 ra8_sys_eswpckdivcr(),
292 (uint8_t)k_ra8_eswckdivcr_div2);
293 if (err != k_ra8_ok) {
294 ra8_log_error(s_tag, "eswclk: ESWPCKCR handshake timeout");
295 break;
296 }
297 }
298 return err;
299}

301ra8_err_t ra8_cgc_eswclk_init(void)
302{
303 ra8_log_info(s_tag, "eswclk init (PLL1P/4 = 250 MHz, eswphyclk PLL1P/2 = 500 MHz)");
304
305 /* Step 1: ESWCKCR / ESWPCKCR / USBCKCR / USB60CKCR all share the same
306 * CKSEL reset-default (HOCO) and the same SREQ -> SRDY handshake
307 * plumbing -- HOCO MUST be running before the SREQ is raised. */
308 const ra8_err_t hoco_err = ra8_cgc_ensure_hoco_running_for_usb_ck();
309 if (hoco_err != k_ra8_ok) {
310 ra8_log_error(s_tag, "eswclk: HOCO stabilization timeout");
311 return hoco_err;
312 }
313 /* Step 2: turn on the ESWM peripheral power domain. */
314 const ra8_err_t pd_err = internal_eswclk_power_on_domain();
315 if (pd_err != k_ra8_ok) {
316 return pd_err;
317 }
318 /* Step 3: release the MSTPC28 ETHPHYCLK module-stop gate. */
319 const ra8_err_t ethphy_mst_err = ra8_mstp_enable(k_ra8_mstp_ethphyclk);
320 if (ethphy_mst_err != k_ra8_ok) {
321 ra8_log_error(s_tag, "eswclk: ethphyclk MSTP release failed");
322 return ethphy_mst_err;
323 }
324 /* Step 4: switch ESWCKCR + ESWPCKCR to PLL1P. */
325 const ra8_err_t cks_err = internal_eswclk_program_cks();
326 if (cks_err != k_ra8_ok) {
327 return cks_err;
328 }
329 s_eswclk_hz = (uint32_t)k_ra8_cgc_eswclk_target_hz;
330 ra8_log_info(s_tag, "eswclk ready");
331 return k_ra8_ok;
332}
