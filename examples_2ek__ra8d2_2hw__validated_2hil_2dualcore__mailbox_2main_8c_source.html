227static bool ping_once(volatile dualcore_mailbox_t* mb, uint32_t seq)
228{
229 mb->request = seq;
230 __asm volatile("dsb" ::: "memory");
231 mb->request_seq = seq;
232 if (!wait_for_reply(mb, seq)) {
233 return false;
234 }
235 const uint32_t expected = (seq * (uint32_t)k_dualcore_m33_mul) + (uint32_t)k_dualcore_m33_add;
236 return (mb->reply == expected);
237}

311[[noreturn]] static void heartbeat_loop(volatile dualcore_mailbox_t* mb, uint32_t seq0)
312{
313 uint32_t seq = seq0;
314 while (1) {
315 for (volatile uint32_t d = 0U; d < (uint32_t)k_m85_heartbeat_nop; d++) {

318 seq++;

366int main(void)
367{
368 ra8_log_init();
369 ra8_log_info("M85", "==== RA8D2 dual-core mailbox demo ====");
370 ra8_log_info("M85", "Cortex-M85 primary core online");
371 ra8_log_info("M85", "mailbox in shared SRAM at 0x22100000");
372
373 /* Bring up the VCOM console so the success path can emit the HIL banner.
374 * Best-effort: a failure is narrated over ITM and the demo continues. */
375 if (!dualcore_mailbox_hil_console_init()) {
376 ra8_log_info("M85", "VCOM console init failed -- HIL banner unavailable");
377 }
378
379 volatile dualcore_mailbox_t* mb = dualcore_mailbox();
380 zero_mailbox(mb);
381
382 ra8_log_info("M85", "releasing Cortex-M33 secondary core ...");
383 const ra8_err_t err = ra8_cpu1_release(&g_ra8_ls_cpu1_mram_start, &g_ra8_ls_cpu1_stack_top);
384 ra8_log_info_val("M85", "ra8_cpu1_release rc (0 = ok)", (uint32_t)err);
385 if (err != k_ra8_ok) {
386 ra8_log_info("M85", "release FAILED -- halting");
387 park_forever();
388 }
389
390 if (wait_for_m33_boot(mb)) {
391 ra8_log_info_val("M85", "M33 is up; boot signature", mb->m33_signature);
392 } else {
393 ra8_log_info("M85", "M33 signature not seen -- did it boot?");
394 }
395
396 ra8_log_info("M85", "-- exchanging messages with the M33 --");
397 uint32_t seq = 0U;
398 uint32_t ok_rounds = 0U;
399 for (uint32_t r = 0U; r < (uint32_t)k_m85_demo_rounds; r++) {
400 seq++;
401 if (do_round(mb, seq)) {
402 ok_rounds++;
403 }
404 }
405
406 ra8_log_info_val("M85", "demo done; total M33 replies", mb->m33_replies);
407 ra8_log_info("M85", "both cores confirmed alive -- idle heartbeat follows");
408 /* Additive HIL banner: emit only when EVERY round returned the M33's
409 * operand*3+1 reply (a value only the second core executing can produce), so
410 * the Pi rig's uart_scrape gates the full dual-core handshake over VCOM. */
411 if (ok_rounds == (uint32_t)k_m85_demo_rounds) {
412 dualcore_mailbox_hil_emit_pass();
413 }
414 heartbeat_loop(mb, seq);
415}
