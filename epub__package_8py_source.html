3"""Reading the EPUB package: the OPF manifest/spine and the navigation document.
4
5Namespace-tolerant on purpose. Real EPUBs in the wild bind the OPF and NCX
6namespaces inconsistently (and some omit them), so every element is matched on
7its LOCAL name -- a book that will not open because of a namespace prefix is a
8bug in the tool, not in the book.

10@copyright Copyright (c) 2026 Brighton Sikarskie

25def opf_localname(tag: str | None) -> str | None:
26 """Strip the `{namespace}` prefix ElementTree prepends to a tag name.
27
28 OPF, NCX and XHTML documents all declare namespaces, and publishers use
29 different prefixes and even different namespace URIs for the same vocabulary.
30 Matching on the local name alone is what lets the parsers below accept real
31 EPUBs instead of only spec-perfect ones.
32
33 Args:
34 tag: An ElementTree tag, namespaced or not. None and "" pass through.
35
36 Returns:
37 The local name, or `tag` unchanged when it carries no namespace.
38 """
39 return tag.split("}", 1)[1] if tag and tag[0] == "{" else tag

42def parse_opf(
43 zf: ZipFile,
44) -> tuple[str, dict[str, str], dict[str, tuple[str, str, str]], list[str], str | None]:
45 """Read container.xml and the OPF package it points at.
46
47 Walks the OPF once with `iter()` rather than following its schema, so
48 elements in unexpected parents are still found. Metadata fields take the
49 FIRST value seen and ignore later ones, which is how a book with several
50 `<dc:creator>` entries yields one author.
51
52 Cover resolution follows the device's precedence exactly: an EPUB3 manifest
53 item whose `properties` contains "cover-image" wins, and the legacy EPUB2
54 `<meta name="cover">` is only the fallback. The substring test mirrors
55 `ra8_epub_xml_shim.cpp` `find_cover_by_properties()` byte for byte, so the
56 host `.rabook` and an on-device compile agree on the cover -- which matters
57 for EPUB3-only fixed-layout comics that ship no legacy meta (issue #196).
58
59 Args:
60 zf: Open ZipFile for the EPUB.
61
62 Returns:
63 Tuple of (opf_dir, meta, manifest, spine, cover_id): the OPF's directory
64 for resolving relative hrefs, the four metadata strings (each "" if
65 absent), manifest id -> (href, media_type, properties), spine idrefs in
66 reading order, and the cover's manifest id or None.
67
68 Raises:
69 KeyError: The EPUB has no META-INF/container.xml.
70 xml.etree.ElementTree.ParseError: container.xml or the OPF is not
71 well-formed XML.
72 """
73 container = ET.fromstring(zf.read("META-INF/container.xml")) # noqa: S314 # trusted local EPUB input
74 rootfile = None
75 for el in container.iter():
76 if opf_localname(el.tag) == "rootfile":
77 rootfile = el.get("full-path")
78 break
79 opf = ET.fromstring(zf.read(rootfile)) # noqa: S314 # trusted local EPUB input
80 opf_dir = posixpath.dirname(rootfile)
81
82 meta = {"title": "", "author": "", "language": "", "identifier": ""}
83 manifest = {} # id -> (href, media_type, properties)
84 spine = [] # list of idref
85 cover_id_meta = None # legacy EPUB2 <meta name="cover" content="ID">
86 cover_id_props = None # EPUB3 manifest <item properties="cover-image">
87 for el in opf.iter():
88 tag = opf_localname(el.tag)
89 if tag == "title" and not meta["title"]:
90 meta["title"] = (el.text or "").strip()
91 elif tag == "creator" and not meta["author"]:
92 meta["author"] = (el.text or "").strip()
93 elif tag == "language" and not meta["language"]:
94 meta["language"] = (el.text or "").strip()
95 elif tag == "identifier" and not meta["identifier"]:
96 meta["identifier"] = (el.text or "").strip()
97 elif tag == "meta" and el.get("name") == "cover":
98 cover_id_meta = el.get("content")
99 elif tag == "item":
100 props = el.get("properties", "") or ""
101 manifest[el.get("id")] = (el.get("href"), el.get("media-type", ""), props)
102 # First manifest item carrying the EPUB3 cover-image property. The
103 # substring test mirrors ra8_epub_xml_shim.cpp find_cover_by_properties()
104 # (a strstr over the space-separated properties list) byte-for-byte, so
105 # an EPUB3-only book -- no legacy meta, how modern fixed-layout comics
106 # ship (issue #196) -- resolves the same cover host-side and on-device.
107 if (cover_id_props is None) and ("cover-image" in props):
108 cover_id_props = el.get("id")
109 elif tag == "itemref":
110 spine.append(el.get("idref"))
111 # EPUB3 properties="cover-image" wins; the legacy <meta name="cover"> is the
112 # fallback -- the exact precedence ra8_epub_xml_shim.cpp uses on device
113 # (find_cover_by_properties() then find_cover_by_meta()), so the desktop
114 # .rabook and the on-device compile agree on the cover image index.
115 cover_id = cover_id_props if cover_id_props is not None else cover_id_meta
116 return opf_dir, meta, manifest, spine, cover_id

119def parse_toc(
120 zf: ZipFile, opf_dir: str, manifest: dict[str, tuple[str, str, str]]
121) -> dict[str, str]:
122 """Map a normalized doc path -> TOC label, from the EPUB3 nav or EPUB2 NCX."""
123 labels = {}
124
125 def resolve(href: str) -> str:
126 return posixpath.normpath(posixpath.join(opf_dir, href.split("#", 1)[0]))
127
128 nav_href = next((h for (h, _m, p) in manifest.values() if "nav" in p), None)
129 ncx_href = next(
130 (h for (h, m, _p) in manifest.values() if m == "application/x-dtbncx+xml"), None
131 )
132 try:
133 if nav_href:
134 tree = ET.fromstring(zf.read(resolve(nav_href))) # noqa: S314 # trusted local EPUB input
135 base = posixpath.dirname(resolve(nav_href))
136 for a in tree.iter():
137 if opf_localname(a.tag) == "a" and a.get("href"):
138 tgt = posixpath.normpath(posixpath.join(base, a.get("href").split("#", 1)[0]))
139 labels.setdefault(tgt, "".join(a.itertext()).strip())
140 elif ncx_href:
141 tree = ET.fromstring(zf.read(resolve(ncx_href))) # noqa: S314 # trusted local EPUB input
142 base = posixpath.dirname(resolve(ncx_href))
143 for nav in tree.iter():
144 if opf_localname(nav.tag) != "navPoint":
145 continue
146 label = ""
147 href = None
148 for sub in nav.iter():
149 ln = opf_localname(sub.tag)
150 if ln == "text" and not label:
151 label = (sub.text or "").strip()
152 elif ln == "content":
153 href = sub.get("src")
154 if href:
155 tgt = posixpath.normpath(posixpath.join(base, href.split("#", 1)[0]))
156 labels.setdefault(tgt, label)
157 except (ET.ParseError, KeyError):
158 pass
159 return labels
