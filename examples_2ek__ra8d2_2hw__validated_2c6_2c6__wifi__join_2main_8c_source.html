94/* Bring up clocks, MSTP, the delay timer and the console, or halt trying. */
95static void c6_join_setup_or_halt(void)
96{
97 if (ra8_cgc_init() != k_ra8_ok) {

100 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &s_cpuclk_hz) != k_ra8_ok) {

112 if (ra8_board_uart_console_init((uint32_t)k_c6_join_uart_baud) != k_ra8_ok) {

117/* Record one announcement; the two station events flip the association flags. */
118static void c6_join_on_event(void* ctx, const ra8_c6link_event_t* ev)
119{
120 (void)ctx;
121 if (ev == nullptr) {
122 return;
123 }
124 s_events++;
125 if (ev->kind == k_ra8_c6link_event_sta_connected) {
126 s_connected = 1U;
127 } else if (ev->kind == k_ra8_c6link_event_sta_disconnected) {
128 s_disconnected = 1U;
129 s_reason = ev->reason;
130 } else {
131 /* boot / bare wifi events: counted above, nothing else to latch. */
132 }
133}

147/* Bind the transport, open the link with the driver's RX callback wired in. */
148static ra8_err_t c6_join_open_link(void)
149{
150 ra8_c6link_cfg_t cfg = {};
151 const ra8_err_t seam = ra8_esp_hosted_c6link_bind(&cfg.transport);
152 if (seam != k_ra8_ok) {
153 return seam;
154 }
155 cfg.arena = s_arena;
156 cfg.arena_bytes = (uint32_t)sizeof(s_arena);
157 cfg.event_cb = c6_join_on_event;

159 cfg.cb_ctx = nullptr;
160 return ra8_c6link_open(&s_link, &cfg);

164static bool c6_join_phase_ready(ra8_c6link_fw_version_t* out)

182 c6_join_put_u32(out->minor);
183 c6_join_puts(".");

185 c6_join_puts(" chip_id=");

187 c6_join_puts("\r\n");

191/* Pump the link until the station-connected event arrives or the budget ends. */
192static bool c6_join_wait_connected(void)
193{
194 for (uint32_t i = 0U; i < (uint32_t)k_c6_join_assoc_tries; i++) {
195 ra8_c6link_stats_t stats = {};
196 (void)ra8_c6link_poll(&s_link, (uint16_t)k_ra8_c6link_announce_transfers, &stats);
197 if (s_connected != 0U) {
198 return true;
199 }
200 if (s_disconnected != 0U) {
201 return false;
202 }
203 tx_thread_sleep((ULONG)k_c6_join_assoc_gap_ms);
204 }
205 return false;
206}

208/* Start the radio, read the MAC, and ask to associate with the bench network. */
209static bool c6_join_phase_associate(ra8_c6link_mac_t* mac_out)
210{
211 /* cppcheck-suppress knownConditionTrueFalse ; build-dependent: s_ssid is the
212 * empty string only in a no-credentials build, non-empty once compiled with
213 * RA8_C6_WIFI_SSID -- cppcheck sees only the default-build literal. */
214 if (s_ssid[0] == '\0') {
215 c6_join_puts("c6_join: FAIL no Wi-Fi credentials compiled in! Set RA8_C6_WIFI_SSID / "
216 "RA8_C6_WIFI_PSK in env, or in coprocessor/esp32c6/wifi.env, or fetch via "
217 "OpenBao (python3 scripts/secrets/openbao_client.py)\r\n");
218 return false;
219 }
220 const ra8_err_t started = ra8_c6link_wifi_start(&s_link);
221 if (started != k_ra8_ok) {
222 c6_join_report_fault("wifi_start", started);
223 return false;
224 }
225 const ra8_err_t got_mac = ra8_c6link_wifi_mac(&s_link, mac_out);
226 if (got_mac != k_ra8_ok) {
227 c6_join_report_fault("wifi_mac", got_mac);
228 return false;
229 }
230 c6_join_puts("c6_join: station mac=");
231 c6_join_put_mac(mac_out);
232 c6_join_puts(" ssid=");
233 c6_join_puts(s_ssid);
234 c6_join_puts("\r\n");
235
236 ra8_c6link_sta_cfg_t sta = {};
237 const ra8_err_t set = ra8_c6link_sta_cfg_set(&sta, s_ssid, s_psk);
238 if (set != k_ra8_ok) {
239 c6_join_report_fault("sta_cfg_set", set);
240 return false;
241 }
242 const ra8_err_t joined = ra8_c6link_wifi_join(&s_link, &sta);
243 if (joined != k_ra8_ok) {
244 c6_join_report_fault("wifi_join", joined);
245 return false;
246 }
247 if (!c6_join_wait_connected()) {
248 c6_join_puts("c6_join: FAIL association did not complete (reason=");
249 c6_join_put_u32((uint32_t)s_reason);
250 c6_join_puts(")\r\n");
251 return false;
252 }
253 c6_join_puts("c6_join: associated\r\n");
254 return true;
255}

257/* Bring the IP layer up, print the lease, and judge the run. */
258static bool c6_join_phase_ip(const ra8_c6link_mac_t* mac)
259{
260 c6_join_lease_t lease = {};
261 const ra8_err_t up = c6_join_net_up(&s_link, mac, &lease);
262 if (up != k_ra8_ok) {
263 c6_join_puts("c6_join: FAIL dhcp did not bind err=");
264 c6_join_puts(ra8_err_to_str(up));
265 c6_join_puts("\r\n");
266 return false;
267 }
268 c6_join_puts("c6_join: dhcp bound ip=");
269 c6_join_put_ip(lease.ip);
270 c6_join_puts(" mask=");
271 c6_join_put_ip(lease.mask);
272 c6_join_puts(" gw=");
273 c6_join_put_ip(lease.gateway);
274 c6_join_puts(" server=");
275 c6_join_put_ip(lease.dhcp_server);
276 c6_join_puts("\r\n");
277 c6_join_puts("c6_join: reachability gateway ping=");
278 c6_join_puts(lease.ping_ok ? "ok" : "no-reply");
279 c6_join_puts("\r\n");
280 return (lease.ip != 0U);
281}

283/* Print a heartbeat forever so the verdict stays visible on the console. */
284static void c6_join_heartbeat(bool passed)
285{
286 while (true) {

288 tx_thread_sleep((ULONG)k_c6_join_heartbeat_ms);

292/* Worker thread: bring up, join, get an IP, judge, then heartbeat. */
293static void c6_join_worker_entry(ULONG thread_input)
294{
295 (void)thread_input;
296
297 c6_join_puts("c6_join: port_init=");
298 c6_join_puts(ra8_err_to_str(s_init_err));
299 c6_join_puts("\r\n");
300 if (s_init_err != k_ra8_ok) {
301 c6_join_puts("c6_join: FAIL port init failed -- no transaction attempted\r\n");
302 c6_join_heartbeat(false);
303 return;
304 }
305
306 ra8_delay_ms((uint32_t)k_c6_join_boot_wait_ms);
307
308 ra8_c6link_fw_version_t fw = {};
309 ra8_c6link_mac_t mac = {};
310 bool ok = c6_join_phase_ready(&fw);
311 ok = ok && c6_join_phase_associate(&mac);
312 ok = ok && c6_join_phase_ip(&mac);
313
314 c6_join_puts("c6_join: events_seen=");
315 c6_join_put_u32((uint32_t)s_events);
316 c6_join_puts("\r\n");
317 if (ok) {
318 c6_join_puts("c6_join: PASS ra8_c6link joined the bench Wi-Fi and DHCP leased an "
319 "address\r\n");
320 } else {
321 c6_join_puts("c6_join: FAIL Wi-Fi join to a DHCP lease did not complete\r\n");
322 }
323 c6_join_heartbeat(ok);
324}
