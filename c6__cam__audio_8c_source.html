146RA8_INTERNAL static int16_t internal_c6_cam_audio_s16(int32_t sample)
147{
148 const int64_t amplified = (int64_t)sample * (int64_t)k_c6_cam_audio_gain;
149 int32_t scaled = (int32_t)(amplified / 16);
150 if (scaled > INT16_MAX) {
151 scaled = INT16_MAX;
152 } else if (scaled < INT16_MIN) {
153 scaled = INT16_MIN;
154 }
155 return (int16_t)scaled;
156}

170RA8_INTERNAL static void internal_c6_cam_audio_on_frame(void* ctx, const ra8_audio_frame_t* frame)
171{
172 (void)ctx;
173 if ((frame == nullptr) || (frame->data == nullptr) ||
174 (frame->format != k_ra8_audio_format_pcm_s32le)) {
175 return;
176 }
177 const uint8_t bank = s_audio_active_bank;
178 c6_cam_audio_bank_t* meta = &s_audio_banks[bank];
179 const int32_t* samples = (const int32_t*)frame->data;
180 for (uint32_t i = 0U; i < frame->sample_count; ++i) {
181 s_audio_ring[bank][meta->write_index] = samples[i];
182 meta->write_index += 1U;
183 if (meta->write_index == (uint32_t)k_c6_cam_audio_ring_samples) {
184 meta->write_index = 0U;
185 }
186 }
187 if (meta->sample_count < (uint32_t)k_c6_cam_audio_ring_samples) {
188 const uint32_t room = (uint32_t)k_c6_cam_audio_ring_samples - meta->sample_count;
189 meta->sample_count += (frame->sample_count < room) ? frame->sample_count : room;
190 }
191 const uint32_t retained_ms =
192 (meta->sample_count * (uint32_t)k_c6_cam_ms_per_second) / (uint32_t)k_c6_cam_audio_rate_hz;
193 const uint32_t frame_end_ms =
194 frame->timestamp_ms +
195 ((frame->sample_count * (uint32_t)k_c6_cam_ms_per_second) / (uint32_t)k_c6_cam_audio_rate_hz);
196 meta->oldest_timestamp_ms = frame_end_ms - retained_ms;
197}

210RA8_INTERNAL static void internal_c6_cam_audio_write_header(uint32_t samples)
211{
212 const uint32_t data_bytes = samples * (uint32_t)k_c6_cam_wav_sample_bytes;
213 (void)memcpy(&s_audio_wav[0], s_wav_tag_riff, sizeof(s_wav_tag_riff));
214 internal_c6_cam_audio_put_u32(&s_audio_wav[4], (uint32_t)k_c6_cam_wav_riff_overhead + data_bytes);
215 (void)memcpy(&s_audio_wav[8], s_wav_tag_wavefmt, sizeof(s_wav_tag_wavefmt));
216 internal_c6_cam_audio_put_u32(&s_audio_wav[16], 16U);
217 internal_c6_cam_audio_put_u16(&s_audio_wav[k_c6_cam_wav_audio_format_off], 1U);
218 internal_c6_cam_audio_put_u16(&s_audio_wav[k_c6_cam_wav_channels_off], 1U);
219 internal_c6_cam_audio_put_u32(&s_audio_wav[24], (uint32_t)k_c6_cam_audio_rate_hz);
220 internal_c6_cam_audio_put_u32(&s_audio_wav[k_c6_cam_wav_byte_rate_off],
221 (uint32_t)k_c6_cam_audio_rate_hz *
222 (uint32_t)k_c6_cam_wav_sample_bytes);
223 internal_c6_cam_audio_put_u16(&s_audio_wav[32], (uint16_t)k_c6_cam_wav_sample_bytes);
224 internal_c6_cam_audio_put_u16(&s_audio_wav[k_c6_cam_wav_bits_off], 16U);
225 (void)memcpy(&s_audio_wav[36], s_wav_tag_data, sizeof(s_wav_tag_data));
226 internal_c6_cam_audio_put_u32(&s_audio_wav[k_c6_cam_wav_data_size_off], data_bytes);
227}

251 .valid_bits = board.valid_bits,

253 };

255 if (err != k_ra8_ok) {

272ra8_err_t
273c6_cam_audio_snapshot_wav(const uint8_t** out_wav, uint32_t* out_bytes, uint32_t* out_timestamp_ms)
274{
275 if ((out_wav == nullptr) || (out_bytes == nullptr) || (out_timestamp_ms == nullptr)) {
276 return k_ra8_err_null_ptr;
277 }
278 *out_wav = nullptr;
279 *out_bytes = 0U;
280 *out_timestamp_ms = 0U;
281 ra8_register_guard_t guard;
282 ra8_register_guard_enter(&guard);
283 const uint8_t snapshot_bank = s_audio_active_bank;
284 const uint32_t samples = s_audio_banks[snapshot_bank].sample_count;
285 if (samples == 0U) {
286 ra8_register_guard_exit(&guard);
287 return k_ra8_err_no_data;
288 }
289 const uint32_t write = s_audio_banks[snapshot_bank].write_index;
290 const uint32_t timestamp_ms = s_audio_banks[snapshot_bank].oldest_timestamp_ms;
291 const uint8_t capture_bank = snapshot_bank ^ 1U;
292 s_audio_banks[capture_bank] = (c6_cam_audio_bank_t){};
293 s_audio_active_bank = capture_bank;
294 ra8_register_guard_exit(&guard);
295
296 const uint32_t oldest = (samples == (uint32_t)k_c6_cam_audio_ring_samples) ? write : 0U;
297 internal_c6_cam_audio_write_header(samples);
298 for (uint32_t i = 0U; i < samples; ++i) {
299 uint32_t ring_index = oldest + i;
300 if (ring_index >= (uint32_t)k_c6_cam_audio_ring_samples) {
301 ring_index -= (uint32_t)k_c6_cam_audio_ring_samples;
302 }
303 const uint16_t pcm =
304 (uint16_t)internal_c6_cam_audio_s16(s_audio_ring[snapshot_bank][ring_index]);
305 internal_c6_cam_audio_put_u16(
306 &s_audio_wav[k_c6_cam_wav_header_bytes + (i * (uint32_t)k_c6_cam_wav_sample_bytes)],
307 pcm);
308 }
309 *out_wav = s_audio_wav;
310 *out_bytes =
311 (uint32_t)k_c6_cam_wav_header_bytes + (samples * (uint32_t)k_c6_cam_wav_sample_bytes);
312 *out_timestamp_ms = timestamp_ms;
313 return k_ra8_ok;
