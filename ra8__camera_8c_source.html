61ra8_err_t ra8_camera_frame_validate(const ra8_camera_frame_t* frame)
62{
63 if (frame == nullptr) {
64 return k_ra8_err_null_ptr;
65 }
66 if (frame->data == nullptr) {
67 return k_ra8_err_null_ptr;
68 }
69 if (frame->width == 0U) {
70 return k_ra8_err_invalid_arg;
71 }
72 if (frame->height == 0U) {
73 return k_ra8_err_invalid_arg;
74 }
75 if (frame->format == k_ra8_camera_format_jpeg) {
76 if (frame->bytes == 0U) {
77 return k_ra8_err_invalid_size;
78 }
79 if (frame->stride_bytes != 0U) {
80 return k_ra8_err_invalid_size;
81 }
82 return k_ra8_ok;
83 }
84 uint32_t row_bytes = 0U;
85 const ra8_err_t row = internal_frame_row_bytes(frame, &row_bytes);
86 if (row != k_ra8_ok) {
87 return row;
88 }
89 if (frame->stride_bytes < row_bytes) {
90 return k_ra8_err_invalid_size;
91 }
92 if ((uint32_t)frame->height > (UINT32_MAX / frame->stride_bytes)) {
93 return k_ra8_err_invalid_size;
94 }
95 return (frame->bytes >= (frame->stride_bytes * (uint32_t)frame->height)) ? k_ra8_ok
96 : k_ra8_err_invalid_size;
97}
