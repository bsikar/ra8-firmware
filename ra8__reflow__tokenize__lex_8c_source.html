46size_t ra8_reflow_tok_utf8_encode(uint32_t cp, uint8_t* dst)
47{
48 if (cp > (uint32_t)k_priv_uc_max) {
49 cp = (uint32_t)k_priv_uc_max;
50 }
51 if (cp < (uint32_t)k_priv_uc_2byte) {
52 dst[0] = (uint8_t)cp;
53 return 1U;
54 }
55 if (cp < (uint32_t)k_priv_uc_3byte) {
56 dst[0] = (uint8_t)((uint32_t)k_priv_utf8_lead2 | (cp >> (uint32_t)k_priv_utf8_sh6));
57 dst[1] = (uint8_t)((uint32_t)k_priv_utf8_cont | (cp & (uint32_t)k_priv_utf8_mask));
58 return 2U;
59 }
60 if (cp < (uint32_t)k_priv_uc_4byte) {
61 dst[0] = (uint8_t)((uint32_t)k_priv_utf8_lead3 | (cp >> (uint32_t)k_priv_utf8_sh12));
62 dst[1] = (uint8_t)((uint32_t)k_priv_utf8_cont |
63 ((cp >> (uint32_t)k_priv_utf8_sh6) & (uint32_t)k_priv_utf8_mask));
64 dst[2] = (uint8_t)((uint32_t)k_priv_utf8_cont | (cp & (uint32_t)k_priv_utf8_mask));
65 return 3U;
66 }
67 dst[0] = (uint8_t)((uint32_t)k_priv_utf8_lead4 | (cp >> (uint32_t)k_priv_utf8_sh18));
68 dst[1] = (uint8_t)((uint32_t)k_priv_utf8_cont |
69 ((cp >> (uint32_t)k_priv_utf8_sh12) & (uint32_t)k_priv_utf8_mask));
70 dst[2] = (uint8_t)((uint32_t)k_priv_utf8_cont |
71 ((cp >> (uint32_t)k_priv_utf8_sh6) & (uint32_t)k_priv_utf8_mask));
72 dst[3] = (uint8_t)((uint32_t)k_priv_utf8_cont | (cp & (uint32_t)k_priv_utf8_mask));
73 return 4U;
74}

96static size_t priv_local_lower(const char* name, size_t len, char* dst)
97{
98 size_t start = 0U;
99 for (size_t k = 0U; k < len; ++k) {
100 if (name[k] == ':') {
101 start = k + 1U;
102 }
103 }
104 size_t n = 0U;
105 for (size_t k = start; (k < len) && (n + 1U < (size_t)k_priv_tag_name_cap); ++k) {
106 char c = name[k];
107 if ((c >= 'A') && (c <= 'Z')) {
108 c = (char)(c + ('a' - 'A'));
109 }
110 dst[n] = c;
111 ++n;
112 }
113 dst[n] = '\0';
114 return n;

182static bool priv_decode_numeric(const char* src, size_t avail, uint32_t* out_cp, size_t* out_used)
183{
184 size_t i = 2U; /* past "&#" */
185 uint32_t base = (uint32_t)k_priv_base_dec;
186 /* mcdc-deactivated: the sole caller ra8_reflow_tok_decode_entity guarantees window >= k_priv_entity_min (4) before delegating, so i == 2 < avail always holds; the (i < avail) bound cannot be flipped false on any public path. */
187 if ((i < avail) && ((src[i] == 'x') || (src[i] == 'X'))) {
188 base = (uint32_t)k_priv_base_hex;
189 ++i;
190 }
191 uint32_t cp = 0U;
192 size_t digits = 0U;
193 while ((i < avail) && (src[i] != ';')) {
194 char c = src[i];
195 uint32_t d; /* assigned on every non-returning path below */
196 if ((c >= '0') && (c <= '9')) {
197 d = (uint32_t)(c - '0');
198 } else if ((base == (uint32_t)k_priv_base_hex) && (c >= 'a') && (c <= 'f')) {
199 d = (uint32_t)(c - 'a') + (uint32_t)k_priv_hex_offset;
200 } else if ((base == (uint32_t)k_priv_base_hex) && (c >= 'A') && (c <= 'F')) {
201 d = (uint32_t)(c - 'A') + (uint32_t)k_priv_hex_offset;
202 } else {
203 return false;
204 }
205 cp = (cp * base) + d;
206 ++digits;
207 ++i;
208 }
209 /* mcdc-deactivated: the scan loop above exits with i < avail only when src[i] == ';', so (src[i] != ';') is co-determined by (i >= avail) and can never independently flip; its independence pair is structurally unreachable. */
210 if ((digits == 0U) || (i >= avail) || (src[i] != ';')) {
211 return false;
212 }
213 *out_cp = cp;
214 *out_used = i + 1U;
215 return true;
216}

218bool ra8_reflow_tok_decode_entity(const char* src, size_t avail, uint32_t* out_cp, size_t* out_used)
219{
220 const size_t window =
221 (avail < (size_t)k_priv_entity_window) ? avail : (size_t)k_priv_entity_window;
222 if ((window < (size_t)k_priv_entity_min) || (src[1] == '\0')) {
223 return false;
224 }
225 if (src[1] == '#') {
226 return priv_decode_numeric(src, window, out_cp, out_used);
227 }
228 static const struct {
229 const char* word;
230 uint32_t cp;
231 } k_named[] = {
232 {"amp", (uint32_t)'&'},
233 {"lt", (uint32_t)'<'},
234 {"gt", (uint32_t)'>'},
235 {"quot", (uint32_t)'"'},
236 {"apos", (uint32_t)'\''},
237 };
238 for (size_t e = 0U; e < (sizeof(k_named) / sizeof(k_named[0])); ++e) {
239 const size_t wlen = strlen(k_named[e].word);
240 if (((wlen + 2U) <= window) && (src[1U + wlen] == ';') &&
241 (strncmp(&src[1], k_named[e].word, wlen) == 0)) {
242 *out_cp = k_named[e].cp;
243 *out_used = wlen + 2U;
244 return true;
245 }
246 }
247 return false;
248}

467static bool priv_stash_one(ra8_reflow_t* engine,
468 const uint8_t* buf,
469 size_t i,

471 bool* last_ws,
472 size_t* consumed)
473{

508bool ra8_reflow_tok_stash_run(ra8_reflow_t* engine,
509 const uint8_t* buf,
510 size_t start,
511 size_t end,
512 uint32_t* out_off,
513 uint32_t* out_len)
514{
515 *out_off = engine->text_pool_used;
516 bool last_ws = true; /* drop leading whitespace */
517 size_t i = start;
518 while (i < end) {
519 size_t consumed = 0U;
520 if (!priv_stash_one(engine, buf, i, end, &last_ws, &consumed)) {
521 return false;
522 }
523 i += consumed;
524 }
525 *out_len = engine->text_pool_used - *out_off;
526 return true;
527}
