108static void test_gov_retry_after_parse(void)
109{
110 TEST_BEGIN("gov retry-after parse");
111 uint32_t ms = 0U;
112 /* delta-seconds form */
113 TEST_ASSERT(mdl_retry_after_parse("120", 0, &ms));
114 TEST_ASSERT_EQ((int64_t)k_gov_retry_120s_ms, (int64_t)ms);
115 TEST_ASSERT(mdl_retry_after_parse("0", 0, &ms));
116 TEST_ASSERT_EQ((int64_t)0, (int64_t)ms);
117 TEST_ASSERT(mdl_retry_after_parse(" 30", 0, &ms)); /* leading spaces tolerated */
118 TEST_ASSERT_EQ((int64_t)30000, (int64_t)ms);
119 /* IMF-fixdate: two minutes past the epoch, measured from the epoch. */
120 TEST_ASSERT(mdl_retry_after_parse("Thu, 01 Jan 1970 00:02:00 GMT", 0, &ms));
121 TEST_ASSERT_EQ((int64_t)k_gov_retry_120s_ms, (int64_t)ms);
122 /* A past HTTP-date clamps to zero. */
123 TEST_ASSERT(mdl_retry_after_parse("Thu, 01 Jan 1970 00:00:00 GMT", 100, &ms));
124 TEST_ASSERT_EQ((int64_t)0, (int64_t)ms);
125 /* RFC 850 (obsolete) form is also recognised. */
126 ms = 0U;
127 TEST_ASSERT(mdl_retry_after_parse("Sunday, 06-Nov-94 08:49:37 GMT", 0, &ms));
128 TEST_ASSERT(ms > 0U);
129 /* Guard / invalid cases. */
130 TEST_ASSERT(!mdl_retry_after_parse("not-a-date", 0, &ms));
131 TEST_ASSERT(!mdl_retry_after_parse("", 0, &ms));
132 TEST_ASSERT(!mdl_retry_after_parse(nullptr, 0, &ms));
133 TEST_ASSERT(!mdl_retry_after_parse("120", 0, nullptr));
134 TEST_END("gov retry-after parse");
135}

145static void test_gov_rate_limit(void)
146{
147 TEST_BEGIN("gov rate limit token bucket");
148 gov_clock_t clk = {};
149 mdl_gov_cfg_t cfg = mdl_gov_cfg_default();
150 cfg.rate_per_min = (uint32_t)k_gov_rate_60; /* 1000 ms interval */
151 cfg.burst = (uint32_t)k_gov_burst_3;
152 cfg.max_inflight = (uint16_t)k_gov_inflight_1;
153 mdl_governor_init_clock(&g_gov,
154 &cfg,
155 (uint64_t)k_gov_seed,
156 gov_clk_now,
157 &clk,
158 gov_clk_sleep,
159 &clk);
160
161 /* The first `burst` (3) requests are admitted immediately -- no sleeping. */
162 for (uint16_t i = 0U; i < (uint16_t)k_gov_burst_3; ++i) {
163 TEST_ASSERT(mdl_governor_acquire(&g_gov, "h", 0U, 0U) == k_ra8_ok);
164 mdl_governor_release(&g_gov, "h");
165 }
166 TEST_ASSERT_EQ((int64_t)0, clk.total_slept);
167 TEST_ASSERT_EQ((int64_t)0, (int64_t)clk.calls);
168
169 /* The 4th and 5th are paced one interval apart. */
170 TEST_ASSERT(mdl_governor_acquire(&g_gov, "h", 0U, 0U) == k_ra8_ok);
171 TEST_ASSERT_EQ((int64_t)k_gov_interval_ms, (int64_t)clk.last_slept);
172 mdl_governor_release(&g_gov, "h");
173 TEST_ASSERT(mdl_governor_acquire(&g_gov, "h", 0U, 0U) == k_ra8_ok);
174 TEST_ASSERT_EQ((int64_t)k_gov_interval_ms, (int64_t)clk.last_slept);
175 mdl_governor_release(&g_gov, "h");
176 TEST_ASSERT(clk.total_slept >= (int64_t)k_gov_two_secs);
177 TEST_END("gov rate limit token bucket");
178}

192static void test_gov_backoff_growth_and_decay(void)
193{
194 TEST_BEGIN("gov backoff growth + decay");
195 gov_clock_t clk = {};
196 mdl_gov_cfg_t cfg = mdl_gov_cfg_default();
197 cfg.rate_per_min = 0U; /* isolate backoff from the rate limiter */
198 cfg.backoff_base_ms = (uint32_t)k_gov_backoff_base;
199 cfg.backoff_max_ms = (uint32_t)k_gov_backoff_cap;
200 cfg.decay_after = (uint16_t)k_gov_decay_2;
201 mdl_governor_init_clock(&g_gov,
202 &cfg,
203 (uint64_t)k_gov_seed,
204 gov_clk_now,
205 &clk,
206 gov_clk_sleep,
207 &clk);
208
209 uint16_t level = 0U;
210 int64_t gate = 0;
211 for (uint16_t k = 1U; k <= (uint16_t)k_gov_throttles; ++k) {
212 mdl_governor_observe(&g_gov, "h", (long)k_http_unavail, nullptr);
213 TEST_ASSERT(mdl_governor_peek(&g_gov, "h", &level, &gate));
214 TEST_ASSERT_EQ((int64_t)k, (int64_t)level); /* level grows 1..5 */
215 TEST_ASSERT(gate >= 0);
216 TEST_ASSERT(gate <= (int64_t)k_gov_backoff_cap); /* full jitter within cap */
217 }
218
219 /* Two successes drop one level; the streak resets between drops. */
220 mdl_governor_observe(&g_gov, "h", (long)k_http_ok, nullptr); /* streak 1: no drop */
221 TEST_ASSERT(mdl_governor_peek(&g_gov, "h", &level, nullptr));
222 TEST_ASSERT_EQ((int64_t)k_gov_throttles, (int64_t)level);
223 mdl_governor_observe(&g_gov, "h", (long)k_http_ok, nullptr); /* streak 2: drop to 4 */
224 TEST_ASSERT(mdl_governor_peek(&g_gov, "h", &level, nullptr));
225 TEST_ASSERT_EQ((int64_t)4, (int64_t)level);
226 mdl_governor_observe(&g_gov, "h", (long)k_http_ok, nullptr);
227 mdl_governor_observe(&g_gov, "h", (long)k_http_ok, nullptr); /* drop to 3 */
228 TEST_ASSERT(mdl_governor_peek(&g_gov, "h", &level, nullptr));
229 TEST_ASSERT_EQ((int64_t)3, (int64_t)level);
230 TEST_END("gov backoff growth + decay");
231}

265 mdl_governor_release(&g_gov, "a");

294{

297 mdl_gov_cfg_t cfg = mdl_gov_cfg_default();
298 cfg.rate_per_min = 0U; /* no pacing noise */
299 cfg.max_inflight = (uint16_t)k_gov_inflight_1;

301 &cfg,
302 (uint64_t)k_gov_seed,
303 gov_clk_now,
304 &clk,

307
308 TEST_ASSERT(mdl_governor_acquire(&g_gov, "a", 0U, 0U) == k_ra8_ok); /* below cap */
309 TEST_ASSERT(mdl_governor_acquire(&g_gov, "a", 0U, 0U) == k_ra8_err_would_block); /* at cap */
310 TEST_ASSERT(mdl_governor_acquire(&g_gov, "b", 0U, 0U) == k_ra8_ok); /* other host */
311 mdl_governor_release(&g_gov, "a");
312 TEST_ASSERT(mdl_governor_acquire(&g_gov, "a", 0U, 0U) == k_ra8_ok); /* slot freed */
313 mdl_governor_release(&g_gov, "a");
314 mdl_governor_release(&g_gov, "b");
315
316 /* A cap of 2 admits two in flight and refuses the third. */
317 gov_clock_t clk2 = {};
318 mdl_gov_cfg_t c2 = cfg;
319 c2.max_inflight = (uint16_t)k_gov_inflight_2;
320 mdl_governor_init_clock(&g_gov,
321 &c2,
322 (uint64_t)k_gov_seed,
323 gov_clk_now,
324 &clk2,
325 gov_clk_sleep,
326 &clk2);
327 TEST_ASSERT(mdl_governor_acquire(&g_gov, "c", 0U, 0U) == k_ra8_ok);
328 TEST_ASSERT(mdl_governor_acquire(&g_gov, "c", 0U, 0U) == k_ra8_ok);
329 TEST_ASSERT(mdl_governor_acquire(&g_gov, "c", 0U, 0U) == k_ra8_err_would_block);
330 mdl_governor_release(&g_gov, "c");
331 TEST_ASSERT(mdl_governor_acquire(&g_gov, "c", 0U, 0U) == k_ra8_ok);
332 TEST_END("gov concurrency cap");

356 nullptr,
357 (uint64_t)k_gov_seed,
358 gov_clk_now,
359 &clk,
360 gov_clk_sleep,
361 &clk);
362 /* NULL host: no per-host record; still spaced by the fixed jitter. */
363 TEST_ASSERT(

368 TEST_END("gov null + untracked host");
