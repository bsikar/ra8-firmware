173static bool cache_hal_test_cacheable_rw(void)
174{
175 for (uint32_t i = 0U; i < (uint32_t)k_cache_hal_buf_bytes; i++) {
176 s_cache_hal_buf[i] =
177 (uint8_t)((i * (uint32_t)k_cache_hal_pat_mul) + (uint32_t)k_cache_hal_pat_add);
178 }
179
180 /* Force the freshly written (dirty) lines back to SRAM and drop them, so the
181 * verify pass below must refill from memory -- the full write-back + refill
182 * round-trip with the HAL-enabled D-cache on. */
183 if (ra8_cache_dcache_clean_invalidate_by_addr(s_cache_hal_buf,
184 (uint32_t)sizeof(s_cache_hal_buf)) != k_ra8_ok) {
185 return false;
186 }
187
188 for (uint32_t i = 0U; i < (uint32_t)k_cache_hal_buf_bytes; i++) {
189 const uint8_t expected =
190 (uint8_t)((i * (uint32_t)k_cache_hal_pat_mul) + (uint32_t)k_cache_hal_pat_add);
191 if (s_cache_hal_buf[i] != expected) {
192 return false;
193 }
194 }
195 return true;
196}
