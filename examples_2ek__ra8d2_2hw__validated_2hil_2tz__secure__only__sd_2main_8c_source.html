144static void sd_demo_print_u32(uint32_t value)
145{
146 uint8_t buf[11];
147 uint32_t pos = (uint32_t)sizeof(buf);
148 if (value == 0U) {
149 buf[--pos] = (uint8_t)'0';
150 } else {
151 while ((value > 0U) && (pos > 0U)) {
152 buf[--pos] = (uint8_t)('0' + (uint8_t)(value % k_sd_decimal_base));
153 value = value / k_sd_decimal_base;
154 }
155 }
156 sd_demo_print(&buf[pos], (uint32_t)sizeof(buf) - pos);
157}

253static void sd_demo_setup_or_halt(uint32_t* out_pclka_hz)
254{
255 uint32_t cpuclk0_hz = 0U;
256 uint32_t pclka_hz = 0U;
257
258 if (ra8_cgc_init() != k_ra8_ok) {
259 sd_demo_panic_halt();
260 }
261 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
262 sd_demo_panic_halt();
263 }
264 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
265 sd_demo_panic_halt();
266 }
267 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
268 sd_demo_panic_halt();
269 }
270 if (ra8_board_uart_console_init((uint32_t)k_sd_demo_uart_baud) != k_ra8_ok) {
271 sd_demo_panic_halt();
272 }
273 if (sd_demo_spi_pins_init() != k_ra8_ok) {
274 sd_demo_panic_halt();
275 }
276 const ra8_sci_spi_cfg_t spi_cfg = {
277 .baud_hz = (uint32_t)k_ra8_sdmmc_spi_clock_init_hz,
278 .pclk_hz = pclka_hz,
279 .mode = k_ra8_spi_mode_0,
280 .lsb_first = false,
281 };
282 if (ra8_sci_spi_init((uint8_t)k_sd_demo_spi_channel, &spi_cfg) != k_ra8_ok) {
283 sd_demo_panic_halt();
284 }
285 *out_pclka_hz = pclka_hz;
286}

300static void sd_demo_fill_payload(uint8_t* buf, uint32_t len)
301{
302 uint32_t state = (uint32_t)k_sd_demo_prng_seed;
303 for (uint32_t i = 0U; i < len; i++) {
304 state = state * (uint32_t)k_sd_demo_prng_mul + (uint32_t)k_sd_demo_prng_add;
305 buf[i] = (uint8_t)((state >> k_sd_prng_byte_shift) & k_sd_byte_mask);
306 }
307}

362static void sd_demo_compare_and_report(const uint8_t* a, const uint8_t* b, uint32_t got)
363{
364 if (got != (uint32_t)k_sd_demo_payload_bytes) {
365 sd_demo_print(k_sd_demo_msg_cmp_pre, (uint32_t)sizeof(k_sd_demo_msg_cmp_pre) - 1U);
366 sd_demo_print_u32(got);
367 sd_demo_print(k_sd_demo_msg_eol, (uint32_t)sizeof(k_sd_demo_msg_eol) - 1U);
368 sd_demo_panic_halt();
369 }
370 for (uint32_t i = 0U; i < (uint32_t)k_sd_demo_payload_bytes; i++) {
371 if (a[i] != b[i]) {
372 sd_demo_print(k_sd_demo_msg_cmp_pre, (uint32_t)sizeof(k_sd_demo_msg_cmp_pre) - 1U);
373 sd_demo_print_u32(i);
374 sd_demo_print(k_sd_demo_msg_eol, (uint32_t)sizeof(k_sd_demo_msg_eol) - 1U);
375 sd_demo_panic_halt();
376 }
377 }
378 sd_demo_print(k_sd_demo_msg_ok, (uint32_t)sizeof(k_sd_demo_msg_ok) - 1U);
379}

437int32_t main(void)
438{
439 uint32_t pclka_hz = 0U;
440 sd_demo_setup_or_halt(&pclka_hz);
441 ra8_isr_globals_enable();
442 ra8_log_init();
443 sd_demo_print(k_sd_demo_msg_boot, (uint32_t)sizeof(k_sd_demo_msg_boot) - 1U);
444
445 sd_demo_init_card_or_halt(&pclka_hz);
446 sd_demo_print_capacity();
447 ra8_fs_mount_t* mount = sd_demo_mount_or_halt();
448
449 static uint8_t s_payload[k_sd_demo_payload_bytes];
450 static uint8_t s_readback[k_sd_demo_payload_bytes];
451 sd_demo_fill_payload(s_payload, (uint32_t)k_sd_demo_payload_bytes);
452 memset(s_readback, 0, sizeof(s_readback));
453
454 if (sd_demo_write_payload(mount, s_payload, (uint32_t)k_sd_demo_payload_bytes) != k_ra8_ok) {
455 sd_demo_print(k_sd_demo_msg_write_fail, (uint32_t)sizeof(k_sd_demo_msg_write_fail) - 1U);
456 sd_demo_panic_halt();
457 }
458 uint32_t got = 0U;
459 if (sd_demo_read_payload(mount, s_readback, (uint32_t)k_sd_demo_payload_bytes, &got) !=
460 k_ra8_ok) {
461 sd_demo_print(k_sd_demo_msg_read_fail, (uint32_t)sizeof(k_sd_demo_msg_read_fail) - 1U);
462 sd_demo_panic_halt();
463 }
464 sd_demo_compare_and_report(s_payload, s_readback, got);
465
466 while (1) {
467 __asm__ volatile("wfi");
468 }
469}
