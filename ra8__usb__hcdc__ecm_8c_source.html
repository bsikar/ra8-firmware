398RA8_INTERNAL
399static ra8_err_t internal_hex_nibble(char c, uint8_t* out_nibble)
400{
401 if ((c >= '0') && (c <= '9')) {
402 *out_nibble = (uint8_t)(c - '0');
403 return k_ra8_ok;
404 }
405 if ((c >= 'a') && (c <= 'f')) {
406 *out_nibble = (uint8_t)(k_ra8_hcdc_ecm_hex_alpha + (c - 'a'));
407 return k_ra8_ok;
408 }
409 if ((c >= 'A') && (c <= 'F')) {
410 *out_nibble = (uint8_t)(k_ra8_hcdc_ecm_hex_alpha + (c - 'A'));
411 return k_ra8_ok;
412 }
413 return k_ra8_err_invalid_arg;
414}

731ra8_err_t ra8_usb_hcdc_ecm_init(ra8_usb_speed_t speed)
732{
733 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
734 return k_ra8_err_invalid_arg;
735 }
736 const ra8_err_t usb_err = ra8_usb_host_init(speed);
737 if (usb_err != k_ra8_ok) {
738 ra8_log_error_val(s_tag, "ra8_usb_host_init failed", (uint32_t)usb_err);
739 return k_ra8_err_hw_init_failed;
740 }
741
742 s_state.speed = speed;
743 s_state.step = k_ra8_hcdc_ecm_step_idle;
744 s_state.attached = false;
745 s_state.attach_cb = nullptr;
746 s_state.attach_ctx = nullptr;
747 s_state.device = (ra8_usb_hcdc_ecm_device_t){};
748 s_state.link = k_ra8_hcdc_ecm_link_down;
749 s_state.initialized = true;
750
751 ra8_log_info_val(s_tag, "host-CDC-ECM ready", (uint32_t)speed);
752 return k_ra8_ok;
753}

755ra8_err_t ra8_usb_hcdc_ecm_close(void)
756{
757 if (!s_state.initialized) {
758 return k_ra8_err_invalid_state;
759 }
760 /* Bus power down: drop UACT before tearing the controller. */
761 (void)ra8_usb_host_set_uact(s_state.speed, false);
762 const ra8_err_t err = ra8_usb_host_deinit(s_state.speed);
763 s_state.initialized = false;
764 s_state.attached = false;
765 s_state.attach_cb = nullptr;
766 s_state.attach_ctx = nullptr;
767 s_state.step = k_ra8_hcdc_ecm_step_idle;
768 return err;
769}

792{
793 if (!s_state.initialized) {
794 return k_ra8_err_invalid_state;

797 return k_ra8_err_invalid_state;

800 return k_ra8_err_invalid_arg;

808ra8_err_t ra8_usb_hcdc_ecm_recv_frame(uint8_t* buf, uint16_t max_len, uint16_t* got_len)
809{
810 RA8_CHECK_NULL_PTR(buf, s_tag, "ecm_recv: buf");
811 RA8_CHECK_NULL_PTR(got_len, s_tag, "ecm_recv: got_len");
812 if (!s_state.initialized) {
813 return k_ra8_err_invalid_state;
814 }
815 if (!s_state.attached) {
816 return k_ra8_err_invalid_state;
817 }
818 if (max_len == 0U) {
819 return k_ra8_err_invalid_arg;

821 uint16_t inout_len = max_len;
822 const ra8_err_t err =
823 ra8_usb_queue_out(s_state.speed, k_ra8_hcdc_ecm_pipe_bulk_in, buf, &inout_len, true);
824 if (err == k_ra8_ok) {
825 *got_len = inout_len;
826 } else {
827 *got_len = 0U;
828 }
829 return err;
830}

852{
853 RA8_CHECK_NULL_PTR(out_link, s_tag, "ecm_link: out_link");
854 if (!s_state.initialized) {

861 /* See USB CDC ECM 1.20 sec 6.2.5 "GetEthernetStatistic": class IN

864 * path updates `s_state.link` when it lands. */

867 .b_request = k_ra8_hcdc_ecm_req_get_statistic,

870 .w_length = 1U,

873 if (err == k_ra8_ok) {
874 *out_link = s_state.link;

879/* =============================================================================
880 * MAC parse helper
881 * =============================================================================
882 */
883
884ra8_err_t ra8_usb_hcdc_ecm_parse_mac(const char* chars, uint8_t* out_mac)
885{
886 RA8_CHECK_NULL_PTR(chars, s_tag, "parse_mac: chars");
887 RA8_CHECK_NULL_PTR(out_mac, s_tag, "parse_mac: out_mac");
888
889 /* See USB CDC ECM 1.20 sec 5.4 "iMACAddress": 12 ASCII hex digits
890 * encoded UTF-16LE in the string descriptor. The caller is expected
891 * to have peeled the UTF-16 wrapper down to the 12 ASCII bytes. */
892 for (size_t i = 0U; i < (size_t)k_ra8_hcdc_ecm_mac_bytes; ++i) {
893 uint8_t hi = 0U;
894 uint8_t lo = 0U;
895 const ra8_err_t hi_err = internal_hex_nibble(chars[i * 2U], &hi);
896 if (hi_err != k_ra8_ok) {
897 return hi_err;
898 }
899 const ra8_err_t lo_err = internal_hex_nibble(chars[(i * 2U) + 1U], &lo);
900 if (lo_err != k_ra8_ok) {
901 return lo_err;
902 }
903 out_mac[i] = (uint8_t)((hi << k_ra8_hcdc_ecm_shift_nibble) | lo);
904 }
905 return k_ra8_ok;
906}
