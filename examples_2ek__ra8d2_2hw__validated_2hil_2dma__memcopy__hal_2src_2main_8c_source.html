238 const ra8_err_t stop_err = ra8_dmac_stop((uint8_t)k_dma_hal_channel);

245void main(void)
246{
247 internal_setup_or_halt();
248 ra8_isr_globals_enable();
249
250 while (1) {
251 internal_fill_buffers();
252 const ra8_err_t err = internal_run_copy();
253 uint8_t ok = 0U;
