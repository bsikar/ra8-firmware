80ra8_err_t ra8_eth_coma_bringup(void)
81{
82 /* Step 1: pulse RRC.RR (1 then 0) to reset the ESWM IP, then settle. */
83 /* HUM Ch 31 "Ethernet Common Agent (COMA)" p 1590 */
84 *ra8_coma_rrc() = (uint32_t)k_ra8_coma_rrc_rr;
85 *ra8_coma_rrc() = 0U;
86 for (volatile uint32_t i = 0U; i < (uint32_t)k_ra8_eth_coma_delay_iters; ++i) {
87 __asm__ volatile("nop");
88 }
89
90 /* Step 2: enable the switch clock (RCE) alone, then settle. */
91 /* HUM Ch 31 "Ethernet Common Agent (COMA)" p 1590 */
92 *ra8_coma_rcec() = (uint32_t)k_ra8_coma_rcec_rce;
93 for (volatile uint32_t i = 0U; i < (uint32_t)k_ra8_eth_coma_delay_iters; ++i) {
94 __asm__ volatile("nop");
95 }
96
97 /* Step 3: kick the buffer-pool init and wait for BPR. Writing BPIOG = 1
98 * starts the pool reset; BPR self-sets clk_period x 512 later. On the host
99 * build CABPIRM is mmap'd RAM with no hardware to self-set BPR, so the
100 * shared bounded waiter consults the ra8_fake_mmio fault seam -- first-poll
101 * success unless a test arms a timeout on this register. */
102 /* HUM Ch 31.3.2.7 "CABPIRM" p 1599 */
103 *ra8_coma_cabpirm() = (uint32_t)k_ra8_coma_cabpirm_bpiog;
104 /* HUM Ch 31.3.2.7 "CABPIRM" p 1599 */
105 const ra8_err_t bpr_err = ra8_hw_wait_flag_set32(ra8_coma_cabpirm(),
106 (uint32_t)k_ra8_coma_cabpirm_bpr,
107 (uint32_t)k_ra8_eth_coma_bpr_poll_max);
108 if (bpr_err != k_ra8_ok) {
109 ra8_log_error(s_tag, "coma_bringup: CABPIRM.BPR timeout");
110 return bpr_err;
111 }
112
113 /* Step 4: fan out every per-agent clock (RCE | ACE[6:0] = ALL) so
114 * RMAC0/1 + ETHA0/1 + GWCA + MFWD + GPTP all become accessible. */
115 /* HUM Ch 31 "Ethernet Common Agent (COMA)" p 1590 */
116 *ra8_coma_rcec() = (uint32_t)k_ra8_coma_rcec_rce | (uint32_t)k_ra8_coma_rcec_ace_mask;
117 for (volatile uint32_t i = 0U; i < (uint32_t)k_ra8_eth_coma_delay_iters; ++i) {
118 __asm__ volatile("nop");
119 }
120 ra8_log_info(s_tag, "coma_bringup");
121 return k_ra8_ok;
122}
