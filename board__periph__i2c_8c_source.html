355static void gt911_write(void* ctx, uint8_t byte)
356{
357 gt911_state_t* g = (gt911_state_t*)ctx;
358 if (g->ptr_bytes < (uint8_t)k_gt911_ptr_bytes) {
359 /* 16-bit pointer arrives MSB first: byte 0 is the high byte, byte 1 the low. */
360 g->reg_ptr = (uint16_t)(((uint32_t)g->reg_ptr << 8U) | (uint32_t)byte);
361 g->ptr_bytes++;
362 return;
363 }
364 /* Payload after the pointer: a 0 written to STATUS acks the current frame so
365 * the IC can latch the next one (ra8_touch's priv_ack_frame). */
366 if ((g->reg_ptr == (uint16_t)k_gt911_reg_status) && (byte == 0U)) {
367 g->click_pending = false;
368 }
369 if (g->reg_ptr == (uint16_t)k_gt911_reg_command) {
370 /* Wake / soft-reset commands are accepted (no observable side effect). */
371 }
372}

375static uint32_t gt911_read_status(gt911_state_t* g, uint8_t* buf)
376{
377 /* Auto-arm the next queued raw point (multi-tap sequence path) so a status
378 * read reports buffer-ready while the FIFO is non-empty. A single-shot
379 * --click (seq_len == 0) is unaffected: the arm is a no-op and the directly
380 * injected contact still reports ready on its own. */
381 gt911_arm_next_from_seq(g);
382 buf[0] = g->click_pending ? (uint8_t)(k_gt911_status_ready | k_gt911_status_one) : 0U;
383 return 1U;
384}

387static uint32_t gt911_read_point0(gt911_state_t* g, uint8_t* buf, uint32_t max)
388{
389 if (max < (uint32_t)k_gt911_point_bytes) {
390 return 0U;
391 }
392 for (uint32_t i = 0U; i < (uint32_t)k_gt911_point_bytes; i++) {
393 buf[i] = 0U;
394 }
395 buf[k_gt911_pt_track] = 0U;
396 buf[k_gt911_pt_x_lsb] = (uint8_t)(g->click_x & (uint16_t)k_i3c_byte_mask);
397 buf[k_gt911_pt_x_msb] = (uint8_t)((uint32_t)g->click_x >> 8U);
398 buf[k_gt911_pt_y_lsb] = (uint8_t)(g->click_y & (uint16_t)k_i3c_byte_mask);
399 buf[k_gt911_pt_y_msb] = (uint8_t)((uint32_t)g->click_y >> 8U);
400 buf[k_gt911_pt_sz_lsb] = (uint8_t)k_gt911_press;
401 /* The coordinate has now been delivered to the firmware -- count it and drop
402 * the contact so the next frame reads "not ready", exactly as the GT911 does
403 * once a tap is drained. */
404 g->click_pending = false;
405 g->reported++;
406 return (uint32_t)k_gt911_point_bytes;
407}

410static uint32_t gt911_read(void* ctx, uint8_t* buf, uint32_t max)
411{
412 gt911_state_t* g = (gt911_state_t*)ctx;
413 if (g->reg_ptr == (uint16_t)k_gt911_reg_product) {
414 const uint8_t id[k_gt911_id_bytes] = {(uint8_t)k_gt911_id0,
415 (uint8_t)k_gt911_id1,
416 (uint8_t)k_gt911_id2,
417 (uint8_t)k_gt911_id3};
418 const uint32_t n = (max < (uint32_t)k_gt911_id_bytes) ? max : (uint32_t)k_gt911_id_bytes;
419 for (uint32_t i = 0U; i < n; i++) {
420 buf[i] = id[i];
421 }
422 return n;
423 }
424 if (g->reg_ptr == (uint16_t)k_gt911_reg_status) {
425 return gt911_read_status(g, buf);
426 }
427 if (g->reg_ptr == (uint16_t)k_gt911_reg_point0) {
428 return gt911_read_point0(g, buf, max);
429 }
430 return 0U;
431}

458static void i3c_open_transfer(void)
459{
460 s_i3c.busy = true;
461 s_i3c.addr_done = false;
462 s_i3c.acked = false;
463 s_i3c.reading = false;
464 s_i3c.rx_len = 0U;
465 s_i3c.rx_pos = 0U;
466 s_i3c.rx_primed = false;
467 /* TX buffer is empty so the driver can write the address byte; clear stale
468 * NACK/TEND so the new address phase reports its own outcome. */
469 s_i3c.ntst = (uint32_t)k_i3c_ntst_tdbef0;
470 s_i3c.bst &= ~((uint32_t)k_i3c_bst_nackdf | (uint32_t)k_i3c_bst_tendf);
471}

479static void i3c_close_transfer(void)
480{
481 if (s_i3c.acked) {
482 i2c_device_t* dev = i2c_device_find(s_i3c.target_7b);
483 if (dev != nullptr) {
484 if (dev->stop != nullptr) {
485 dev->stop(dev->ctx);
486 }
487 }
488 /* Console I2C tab: one line per completed (ACKed) transaction at STOP --
489 * 7-bit address + R/W + read byte count. Bounded to one push per STOP. */
490 char ln[k_i2c_console_line_cap];
491 if (s_i3c.reading) {
492 (void)snprintf(ln,
493 sizeof(ln),
494 "IIC addr=0x%02X R %uB",
495 (unsigned)s_i3c.target_7b,
496 (unsigned)s_i3c.rx_len);
497 } else {
498 (void)snprintf(ln, sizeof(ln), "IIC addr=0x%02X W", (unsigned)s_i3c.target_7b);
499 }
500 board_console_push(k_board_console_ch_i2c, ln);
501 }
502 s_i3c.busy = false;
503 s_i3c.addr_done = false;
504 s_i3c.ntst = (uint32_t)k_i3c_ntst_tdbef0;
505}

508static void i3c_address_phase(uint8_t address_byte)
509{
510 s_i3c.target_7b =
511 (uint8_t)((uint32_t)address_byte >> (uint32_t)k_i3c_addr_shift) & (uint8_t)k_i3c_addr_mask7;
512 s_i3c.reading = ((uint32_t)address_byte & (uint32_t)k_i3c_addr_rnw) != 0U;
513 s_i3c.addr_done = true;
514
515 i2c_device_t* dev = i2c_device_find(s_i3c.target_7b);
516 if (dev == nullptr) {
517 /* No device at this address: NACK the address (scan reports ack=0). */
518 s_i3c.acked = false;
519 s_i3c.bst |= (uint32_t)k_i3c_bst_nackdf;
520 return;
521 }
522 s_i3c.acked = true;
523 s_i3c.bst |= (uint32_t)k_i3c_bst_tendf; /* address ACKed -> scan sees TENDF */
524 if (s_i3c.reading) {
525 /* Pre-fetch the device's response for this read so NTDTBP0 reads serve it. */
526 s_i3c.rx_len = dev->read(dev->ctx, s_i3c.rx, (uint32_t)k_i3c_dev_rx_max);
527 s_i3c.rx_pos = 0U;
528 s_i3c.rx_primed = false;
529 s_i3c.ntst |= (uint32_t)k_i3c_ntst_rdbff0; /* RX data ready */
530 } else {
531 s_i3c.ntst |= (uint32_t)k_i3c_ntst_tdbef0; /* ready for the first data byte */
532 }
533}

535/* =============================================================================
536 * Target (peripheral) mode -- the model plays the EXTERNAL I2C controller that
537 * drives the firmware's IIC_B target (i3c_i2c_peripheral_demo). The firmware
538 * polls NTST and moves bytes through NTDTBP0; the controller path here is idle
539 * (the firmware issues no CNDCTL START), so these handlers own NTST/NTDTBP0 for
540 * the channel whenever target mode is active. See ::i3c_periph_phase_t.
541 * =============================================================================
542 */

545static void i3c_periph_open(uint32_t msdvad)
546{
547 const uint8_t addr =
548 (uint8_t)(((uint32_t)msdvad >> (uint32_t)k_i3c_addr_shift) & (uint32_t)k_i3c_addr_mask7);
549 s_i3c.periph_addr_7b = addr;
550 s_i3c.periph_mode = (addr != 0U); /* MSDVAD == 0 (close) leaves target mode. */
551 s_i3c.periph_phase = (uint8_t)k_i3c_periph_idle;
552}

560static uint32_t i3c_periph_ntst(void)
561{
562 if (s_i3c.periph_phase == (uint8_t)k_i3c_periph_idle) {
563 s_i3c.periph_rx_byte =
564 (uint8_t)(((uint32_t)k_i3c_periph_seed + (s_i3c.periph_xfers * (uint32_t)k_i3c_periph_step)) &
565 (uint32_t)k_i3c_byte_mask);
566 s_i3c.periph_phase = (uint8_t)k_i3c_periph_rx_armed;
567 }
568 if (s_i3c.periph_phase == (uint8_t)k_i3c_periph_rx_armed) {
569 return (uint32_t)k_i3c_ntst_rdbff0; /* controller-written byte is waiting */
570 }
571 return (uint32_t)k_i3c_ntst_tdbef0; /* tx_armed: controller is reading the echo */
572}

592static void i3c_ntdtbp0_write(uint32_t value)
593{
594 const uint8_t byte = (uint8_t)(value & (uint32_t)k_i3c_byte_mask);
595 if (s_i3c.periph_mode) {
596 /* Target mode: the firmware is the peripheral pushing its echo byte. */
597 i3c_periph_tx_write(byte);
598 return;
599 }
600 if (!s_i3c.addr_done) {
601 i3c_address_phase(byte);
602 return;
603 }
604 if (!s_i3c.acked) {
605 return; /* NACKed address: swallow further writes until STOP */
606 }
607 i2c_device_t* dev = i2c_device_find(s_i3c.target_7b);
608 if ((dev != nullptr) && (dev->write != nullptr)) {
609 dev->write(dev->ctx, byte);
610 }
611 s_i3c.ntst |= (uint32_t)k_i3c_ntst_tdbef0; /* buffer empty again for the next */
612}

615static uint32_t i3c_ntdtbp0_read(void)
616{
617 if (s_i3c.periph_mode) {
618 /* Target mode: the firmware is the peripheral draining the written byte. */
619 return i3c_periph_rx_read();
620 }
621 if (!s_i3c.rx_primed) {
622 /* FSP's controller RXI handler drops the first RDBFF0 read before real payload. */
623 s_i3c.rx_primed = true;
624 return 0U;
625 }
626 uint8_t b = 0U;
627 if (s_i3c.rx_pos < s_i3c.rx_len) {
628 b = s_i3c.rx[s_i3c.rx_pos];
629 s_i3c.rx_pos++;
630 }
631 s_i3c.ntst |= (uint32_t)k_i3c_ntst_rdbff0; /* keep RX-ready for the next byte */
632 return (uint32_t)b;
633}

646static uint64_t i3c_reg_read(uint64_t off)
647{
648 if (off == (uint64_t)k_i3c_off_ntst) {
649 if (s_i3c.periph_mode) {
650 return i3c_periph_ntst();
651 }
652 return s_i3c.ntst;
653 }
654 if (off == (uint64_t)k_i3c_off_bst) {
655 return s_i3c.bst;
656 }
657 if (off == (uint64_t)k_i3c_off_bcst) {
658 /* BFREF: 1 when the bus is free. The driver gates new transactions on it. */
659 return s_i3c.busy ? 0U : (uint32_t)k_i3c_bcst_bfref;
660 }
661 if (off == (uint64_t)k_i3c_off_ntdtbp0) {
662 return i3c_ntdtbp0_read();
663 }
664 return s_i3c.reg[i3c_word(off)]; /* reflect every other register */
665}

668static void i3c_reg_write(uint64_t off, uint32_t value)
669{
670 s_i3c.reg[i3c_word(off)] = value; /* shadow keeps "configure then verify" working */
671 if (off == (uint64_t)k_i3c_off_msdvad) {
672 /* Own-address programming = the firmware is coming up as an addressed
673 * target. The controller path never writes MSDVAD, so this cleanly selects
674 * target mode (the external-controller stimulus in i3c_periph_*). */
675 i3c_periph_open(value);
676 } else if (off == (uint64_t)k_i3c_off_cndctl) {
677 i3c_cndctl_write(value);
678 } else if (off == (uint64_t)k_i3c_off_ntdtbp0) {
679 i3c_ntdtbp0_write(value);
680 } else if (off == (uint64_t)k_i3c_off_bst) {
681 /* BST condition / fault flags are write-0-to-clear: keep only bits still
682 * written as 1 (the driver clears by reading then masking the bit out). */
683 s_i3c.bst &= value;
684 }
685}

704static void i3c_reset(void)
705{
706 s_i3c = (i3c_state_t){};
707 s_gt911 = (gt911_state_t){};
708 board_i2c_imu_fuel_reset(); /* lay the IMU + fuel-gauge register files (CLI-set SOC) */
709 /* Populate the modelled I2C bus: the EK-RA8D2 carrier's GT911 touch
710 * controller answers at its default 7-bit address on I3C/IIC_B channel 0, so
711 * the firmware's real ra8_touch -> ra8_i3c_transfer -> GT911 path returns data
712 * instead of needing a function-level touch stub. The LSM6DSO IMU answers at
713 * 0x6B so imu_lsm6dso_demo's WHO_AM_I probe + sample reads succeed too. */
714 for (uint32_t i = 0U; i < (uint32_t)k_i3c_dev_max; i++) {
715 s_i2c_dev[i] = (i2c_device_t){};
716 }
717 i2c_device_register((uint8_t)k_gt911_addr_7b, gt911_write, gt911_read, gt911_stop, &s_gt911);
718 /* LSM6DSO IMU at 0x6B + MAX17048-class fuel gauge at 0x36 (battery SOC +
719 * charge direction) register themselves from the devices TU. */
720 board_i2c_imu_fuel_register();
721}

724static void i3c_report(void)
725{
726 if (s_gt911.reported > 0U) {
727 (void)fprintf(stderr,
728 " I3C/I2C GT911 : %u touch frame(s) drained via ra8_touch -> I3C\n",
729 s_gt911.reported);
730 }
731 if (board_i2c_imu_reads() > 0U) {
732 (void)fprintf(stderr,
733 " I3C/I2C LSM6DSO: %u register read(s) answered (WHO_AM_I + samples)\n",
734 board_i2c_imu_reads());
735 }
736 if (s_i3c.periph_mode) {
737 /* Target mode: report how many controller write+read round-trips the
738 * firmware's IIC_B peripheral accepted, and whether every byte it echoed
739 * back matched the byte the synthetic controller wrote. */
740 const bool echo_ok = (s_i3c.periph_xfers > 0U) && !s_i3c.periph_echo_bad;
741 (void)fprintf(stderr,
742 " I3C/I2C target: addr=0x%02X %u peripheral write+read xfer(s) accepted,"
743 " echo=%s\n",
744 (unsigned)s_i3c.periph_addr_7b,
745 (unsigned)s_i3c.periph_xfers,
746 echo_ok ? "Y" : "N");
747 }
748}
