236static void usbhs_do_setup(uc_engine* uc)
237{
238 s_hs.last_setup_in = false;
239 s_hs.sie_status_done = false;
240 s_hs.ccpl_armed = false;
241 s_hs.status_zlp = false;
242 const uint32_t w1 = usbhs_word((uint64_t)k_ra8_usb_off_intsts1);
243 if (!board_usb_loop_attached()) {
244 s_hs.reg[w1] = (uint16_t)(s_hs.reg[w1] | (uint16_t)k_usbhs_int1_sign);
245 return;
246 }
247 const uint16_t req = s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_usbreq)];
248 const uint16_t val = s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_usbval)];
249 const uint16_t indx = s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_usbindx)];
250 const uint16_t leng = s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_usbleng)];
251 s_hs.last_setup_in = ((req & (uint16_t)k_usbhs_setup_dir_in) != 0U);
252 s_hs.sie_status_done = board_usb_loop_setup(uc, req, val, indx, leng);
253 s_hs.reg[w1] = (uint16_t)(s_hs.reg[w1] | (uint16_t)k_usbhs_int1_sack);
254 s_hs.setups++;
255 if (s_hs_trace) {
256 (void)fprintf(stderr,
257 " USBHS host : SETUP req=0x%04X wVal=0x%04X wLen=%u%s\n",
258 (unsigned)req,
259 (unsigned)val,
260 (unsigned)leng,
261 s_hs.sie_status_done ? " (SIE status)" : "");
262 }

285static void usbhs_do_ccpl(uc_engine* uc)
286{
287 const uint32_t wb = usbhs_word((uint64_t)k_ra8_usb_off_bempsts);
288 if (s_hs.last_setup_in) {
289 board_usb_loop_status_out_zlp(uc);
290 s_hs.reg[wb] = (uint16_t)(s_hs.reg[wb] | (uint16_t)k_usbhs_dcp_bit);
291 return;
292 }
293 if (s_hs.sie_status_done) {
294 s_hs.status_zlp = true; /* SET_ADDRESS: the SIE completed the transfer. */
295 return;

297 s_hs.ccpl_armed = true; /* status ZLP follows the device firmware's CCPL. */

320static uint16_t usbhs_brdysts_value(uc_engine* uc)
321{
322 if (s_hs.ccpl_armed && board_usb_loop_take_ccpl(uc)) {
323 s_hs.ccpl_armed = false;
324 s_hs.status_zlp = true;
325 }
326 uint16_t v = 0U;
327 if (s_hs.status_zlp || (board_usb_loop_ctrl_in_avail() > 0U)) {
328 v |= (uint16_t)k_usbhs_dcp_bit;
329 }
330 for (uint8_t pipe = 1U; pipe < (uint8_t)k_ra8_usb_pipe_count; pipe++) {
331 if (s_hs.pipecfg[pipe] == 0U) {
332 continue;
333 }
334 if (usbhs_pipe_is_tx(pipe)) {
335 continue;
336 }
337 if (board_usb_loop_bulk_in_avail(usbhs_pipe_ep(pipe)) > 0U) {
338 v = (uint16_t)(v | (uint16_t)(1U << pipe));
339 }
340 }
341 return v;

364{

367 uint16_t mps = (uint16_t)k_usbhs_mps_fall;

402static uint64_t usbhs_cfifo_read(uc_engine* uc, unsigned size)
403{
404 uint8_t bytes[sizeof(uint32_t)] = {};
405 const uint16_t cap = (size < sizeof(bytes)) ? (uint16_t)size : (uint16_t)sizeof(bytes);
406 const uint8_t pipe = usbhs_cur_pipe();
407 uint16_t got = 0U;
408 if (pipe == 0U) {
409 got = board_usb_loop_ctrl_in_read(bytes, cap);
410 } else if (!usbhs_pipe_is_tx(pipe)) {
411 got = board_usb_loop_bulk_in_read(uc, usbhs_pipe_ep(pipe), bytes, cap);
412 if (got > 0U) {
413 s_hs.bulk_in++;
414 }
415 } else {
416 got = 0U;
417 }
418 (void)got; /* short reads leave the upper packed bytes zero (don't-care). */
419 uint64_t v = 0U;
420 for (uint32_t i = 0U; i < (uint32_t)cap; i++) {
421 v |= ((uint64_t)bytes[i]) << (i * (uint32_t)k_usbhs_byte_bits);
422 }
423 return v;
424}

474static void usbhs_cfifoctr_write(uc_engine* uc, uint16_t value)
475{
476 const uint8_t pipe = usbhs_cur_pipe();
477 if ((value & (uint16_t)k_ra8_fifoctr_bclr) != 0U) {
478 if (pipe == 0U) {
479 if (usbhs_cfifo_isel()) {
480 s_hs.stage_len = 0U;
481 } else {
482 board_usb_loop_ctrl_in_flush();
483 s_hs.status_zlp = false;
484 }
485 } else if (usbhs_pipe_is_tx(pipe)) {
486 s_hs.stage_len = 0U;
487 } else {
488 board_usb_loop_bulk_in_flush(usbhs_pipe_ep(pipe));
489 }
490 }
491 if ((value & (uint16_t)k_ra8_fifoctr_bval) != 0U) {
492 const uint32_t wb = usbhs_word((uint64_t)k_ra8_usb_off_bempsts);
493 if ((pipe == 0U) && usbhs_cfifo_isel() && (s_hs.stage_len > 0U)) {
494 board_usb_loop_ctrl_out(uc, s_hs.stage, s_hs.stage_len);
495 s_hs.reg[wb] = (uint16_t)(s_hs.reg[wb] | (uint16_t)k_usbhs_dcp_bit);
496 s_hs.stage_len = 0U;
497 } else if ((pipe != 0U) && (s_hs.stage_len > 0U)) {
498 board_usb_loop_bulk_out(uc, usbhs_pipe_ep(pipe), s_hs.stage, s_hs.stage_len);
499 s_hs.reg[wb] = (uint16_t)(s_hs.reg[wb] | (uint16_t)(1U << pipe));
500 s_hs.stage_len = 0U;
501 s_hs.bulk_out++;
502 }
503 }

601static uint64_t usbhs_reg_read(uc_engine* uc, uint64_t off, unsigned size)
602{
603 const uint8_t sel = (uint8_t)(s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_pipesel)] &
604 (uint16_t)k_ra8_fifosel_curpipe);
605 switch ((uint16_t)off) {
606 case (uint16_t)k_ra8_usb_off_syssts0:
607 return (board_usb_loop_attached()) ? (uint64_t)k_usbhs_lnst_j_state : 0U;
608 case (uint16_t)k_ra8_usbhs_off_pllsta:
609 return (uint16_t)k_ra8_pllsta_plllock;
610 case (uint16_t)k_ra8_usb_off_dvstctr0:
611 return (uint16_t)(s_hs.reg[usbhs_word(off)] | s_hs.rhst);
612 case (uint16_t)k_ra8_usb_off_cfifo:
613 return usbhs_cfifo_read(uc, size); /* full width: 32-bit MBW drains. */
614 case (uint16_t)k_ra8_usb_off_cfifoctr:
615 return usbhs_cfifoctr_value();
616 case (uint16_t)k_ra8_usb_off_brdysts:
617 return usbhs_brdysts_value(uc);
618 case (uint16_t)k_ra8_usb_off_nrdysts:
619 return 0U;
620 case (uint16_t)k_ra8_usb_off_pipecfg:
621 return s_hs.pipecfg[sel % k_ra8_usb_pipe_count];
622 case (uint16_t)k_ra8_usb_off_pipebuf:
623 return s_hs.pipebuf[sel % k_ra8_usb_pipe_count];
624 case (uint16_t)k_ra8_usb_off_pipemaxp:
625 return s_hs.pipemaxp[sel % k_ra8_usb_pipe_count];
626 case (uint16_t)k_ra8_usb_off_pipeperi:
627 return s_hs.pipeperi[sel % k_ra8_usb_pipe_count];
628 default:
629 return s_hs.reg[usbhs_word(off)];
630 }
631}

655static void usbhs_reg_write(uc_engine* uc, uint64_t off, unsigned size, uint64_t value64)
656{
657 const uint16_t value = (uint16_t)value64;
658 const uint8_t sel = (uint8_t)(s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_pipesel)] &
659 (uint16_t)k_ra8_fifosel_curpipe);
660 /* The CFIFO port spans +0x14..+0x17: 32-bit head words at +0x14, the
661 * CFIFOH (+0x16) / CFIFOHH (+0x17) residual aliases for the 2- / 1-byte
662 * tails. All append to the same OUT staging. */
663 if ((off >= (uint64_t)k_ra8_usb_off_cfifo) && (off < (uint64_t)(k_ra8_usb_off_cfifo + 4U))) {
664 usbhs_cfifo_write(value64, size); /* full width: 32-bit MBW fills. */
665 return;
666 }
667 switch ((uint16_t)off) {
668 case (uint16_t)k_ra8_usb_off_cfifoctr:
669 usbhs_cfifoctr_write(uc, value);
670 return;
671 case (uint16_t)k_ra8_usb_off_dcpctr:
672 usbhs_dcpctr_write(uc, value);
673 return;
674 case (uint16_t)k_ra8_usb_off_dvstctr0:
675 usbhs_dvstctr_write(uc, value);
676 return;
677 case (uint16_t)k_ra8_usb_off_intsts1:
678 case (uint16_t)k_ra8_usb_off_bempsts:
679 /* W0C: a written 0 clears, a written 1 preserves. */
680 s_hs.reg[usbhs_word(off)] &= value;
681 return;
682 case (uint16_t)k_ra8_usb_off_brdysts:
683 /* Live bits are composed; only the sticky status-ZLP flag is W0C. */
684 if ((value & (uint16_t)k_usbhs_dcp_bit) == 0U) {
685 s_hs.status_zlp = false;
686 }
687 return;
688 case (uint16_t)k_ra8_usb_off_nrdysts:
689 return; /* never latched -- nothing to clear. */
690 case (uint16_t)k_ra8_usb_off_pipecfg:
691 s_hs.pipecfg[sel % k_ra8_usb_pipe_count] = value;
692 return;
693 case (uint16_t)k_ra8_usb_off_pipebuf:
694 s_hs.pipebuf[sel % k_ra8_usb_pipe_count] = value;
695 return;
696 case (uint16_t)k_ra8_usb_off_pipemaxp:
697 s_hs.pipemaxp[sel % k_ra8_usb_pipe_count] = value;
698 return;
699 case (uint16_t)k_ra8_usb_off_pipeperi:
700 s_hs.pipeperi[sel % k_ra8_usb_pipe_count] = value;
701 return;
702 default:
703 break;
704 }
705 if ((off >= (uint64_t)k_ra8_usb_off_pipectr) &&
706 (off < (uint64_t)(k_ra8_usb_off_pipectr +
707 ((uint64_t)k_ra8_usb_pipectr_count * sizeof(uint16_t))))) {
708 /* PIPECTR: PID persists; the SQSET / SQCLR pulses never store. */
709 s_hs.reg[usbhs_word(off)] = (uint16_t)(value & (uint16_t)~(uint16_t)k_usbhs_ctr_pulse_mask);
710 return;
711 }
712 s_hs.reg[usbhs_word(off)] = value;
713}

742void board_usb_host_write(uc_engine* uc,
743 uint64_t addr,
744 unsigned size,
745 uint64_t value,
746 bool* handled)
747{
748 if ((addr < (uint64_t)k_usbhs_base) ||
749 (addr >= ((uint64_t)k_usbhs_base + (uint64_t)k_usbhs_span))) {
750 *handled = false;
751 return;
752 }
753 const uint64_t off = addr - (uint64_t)k_usbhs_base;
754 if (!s_hs.engaged) {
755 /* Dormant: watch only the SYSCFG host-role select. The write still
756 * reports unhandled so the sparse fallback records it and every app that
757 * never engages keeps its exact current behaviour. */
758 *handled = false;
759 if (s_hs.allowed && (off == (uint64_t)k_ra8_usb_off_syscfg) &&
760 (((uint16_t)value & (uint16_t)(1U << k_ra8_syscfg_bit_dcfm)) != 0U)) {
761 s_hs.engaged = true;
762 s_hs.reg[usbhs_word((uint64_t)k_ra8_usb_off_syscfg)] = (uint16_t)value;
763 board_usb_loop_latch();
764 (void)fprintf(stderr,
765 " USBHS host : host mode engaged (SYSCFG.DCFM) -- self-loop to USBFS\n");
766 *handled = true;
767 }
768 return;
769 }
770 *handled = true;
771 usbhs_reg_write(uc, off, size, value);
772}
