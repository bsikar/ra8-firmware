36/* Set true when a real-firmware USB host (the emulated USBHS controller in
37 * board_periph_usbhs_host.c) drives THIS device model through the bridge below;
38 * the built-in virtual chapter-9 host then stands down (board_usb_tick returns
39 * early) so exactly one host drives the device -- the chip-internal self-loop. */

42/* Self-loop role polarity (see the role-routing section of board_usb.h).
43 * false: USBFS window = this device model, USBHS window = host model (Config
44 * A). true: swapped (Config B) -- the firmware declared the FS controller the
45 * host (SYSCFG.DCFM=1 written to the FS window) or the HS controller the
46 * device (SYSCFG.DPRPU=1 written to the HS window), so the FS window routes to
47 * the host model and the HS window routes here. Sticky until board_usb_init. */
48bool s_roles_swapped;

55/* DCP control-OUT "wire" holding buffer (bridge path). The polled host delivers
56 * a control-write data stage (e.g. a DFU_DNLOAD block) before the device -- off
57 * its own IRQ on the shared core -- has armed its DCP to receive it, and the
58 * device's arm-time CFIFO BCLR would discard data landed too early. So the
59 * bytes are held here as "in flight on the wire" and handed to the device DCP
60 * only once it has armed (PID=BUF + DCP BRDY enabled), exactly as the SIE would
61 * deliver an OUT packet the device is finally ready to ACK. */

90void usb_log_line(const char* msg)
91{
92 /* Mirror this completed USB event (one SETUP step / state change / bulk
93 * buffer -- already coalesced by the caller, never per byte) onto the board
94 * view's USB console tab. This is an in-memory store only and does not touch
95 * stdout, so the smoke / golden output stays byte-identical. Pushed outside
96 * the s_log cap below so the live tab keeps flowing past the report cap. */
97 board_console_push(k_board_console_ch_usb, msg);
98 if (s_log_n < (uint32_t)k_usb_log_cap) {
99 (void)snprintf(s_log[s_log_n], sizeof(s_log[0]), "%s", msg);
100 if (s_trace) {
101 (void)fprintf(stderr, " [usb] %s\n", s_log[s_log_n]);
102 }
103 s_log_n++;
104 }
105}

115void board_usb_tick(uc_engine* uc)
116{
117 /* Latch the CONFIGURED marker as soon as the device state reaches it -- the
118 * host sets DVSQ, but record it here so a single source of truth drives both
119 * the success assertion and the report marker. */
120 if (s_usb.dvsq == (uint16_t)k_ra8_dvsq_configured) {
121 s_configured = true;
122 }
123 /* A real-firmware USB host is driving the device through the bridge (the
124 * chip-internal self-loop): the built-in virtual host must not also drive it,
125 * or two hosts would race the device's control endpoint. Stand down -- but
126 * still pump the device's level-triggered DCP-OUT receive so a control-write
127 * data stage the host delivered before the device armed still lands. */
128 if (s_external_host) {
129 bridge_pump_device(uc);
130 return;
131 }
132 /* Self-loop bench (register-level USBHS host model, board_usb_host.c): the
133 * firmware's own host drives this device over the loop cable
134 * (board_usb_loop_*), so the built-in virtual host must not compete for the
135 * control pipe or drain the bulk-IN data. */
136 if (s_loop_latched) {
137 return;
138 }
139 switch ((usb_host_phase_t)s_host_phase) {
140 case k_phase_idle:
141 host_run_idle_phase(uc);
142 break;
143 case k_phase_reset:
144 host_run_reset_phase(uc);
145 break;
146 case k_phase_setup:
147 host_run_setup_phase(uc);
148 break;
149 case k_phase_configured:
150 host_run_configured_phase(uc);
151 break;
152 case k_phase_done:
153 default:
154 host_echo_read_in(uc); /* keep draining any late echo. */
155 break;
156 }
157}

249void board_usb_report(void)
250{
251 if ((s_usb_irqs == 0U) && (s_log_n == 0U)) {
252 return; /* USB never came up in this run; stay silent. */
253 }
254 (void)fprintf(stderr,
255 " USB-FS : %u IRQ(s), device state %s\n",
256 s_usb_irqs,
257 usb_dvsq_name(s_usb.dvsq));
258 for (uint32_t i = 0U; i < s_log_n; i++) {
259 (void)fprintf(stderr, " usb: %s\n", s_log[i]);
260 }
261 if (s_configured) {
262 (void)fprintf(stderr, " USB: device CONFIGURED (%s)\n", usb_class_active_str());
263 } else {
264 (void)fprintf(stderr, " USB: enumeration INCOMPLETE (device did not reach CONFIGURED)\n");
265 }
266 if (s_hid_reports > 0U) {
267 (void)fprintf(stderr,
268 " USB HID : %u mouse report(s), cursor (%d,%d) buttons 0x%02X\n",
269 s_hid_reports,
270 (int)s_hid_cx,
271 (int)s_hid_cy,
272 (unsigned)s_hid_buttons);
273 }
274 if ((s_msc_blocks > 0U) || (s_msc_read_ok > 0U)) {
275 (void)fprintf(
276 stderr,
277 " USB MSC : capacity %u blocks x %uB, INQUIRY %s, sector read %u byte(s)\n",
278 s_msc_blocks,
279 s_msc_block_len,
280 s_msc_inquiry_ok ? "ok" : "--",
281 s_msc_read_ok);
282 }
283 if (s_echo_out_len > 0U) {
284 (void)fprintf(stderr,
285 " USB CDC echo : sent %u byte(s) OUT, read %u byte(s) IN\n",
286 s_echo_out_sent,
287 s_echo_in_got);
288 }
289 if (s_loop_latched) {
290 (void)fprintf(stderr,
291 " USB self-loop : fw host drove %u SETUP(s), %u bulk-OUT, %u bulk-IN pkt(s)\n",
292 s_loop_setups,
293 s_loop_bulk_out_pkts,
294 s_loop_bulk_in_pkts);
295 }
