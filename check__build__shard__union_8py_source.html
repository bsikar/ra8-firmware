4"""Gate: the cross-build shards together covered every app, exactly once.
5
6``scripts/builders/all_examples.sh`` builds a stride slice of the discovered
7app list when ``RA8_BUILD_SHARDS``/``RA8_BUILD_SHARD`` are set, so the
8``build-cross`` gate can fan out across parallel CI jobs. That fan-out is only
9sound if the shards, taken together, build the same set the unsharded gate
10would have built. Nothing about a parallel matrix guarantees that: a shard
11whose job was skipped, cancelled, or silently slice-computed to the empty set
12produces no output and no error, and the downstream ``stack-usage`` aggregate
13would simply measure fewer ``.su`` files and still clear its floor on the
14shards that DID run. That is a gate quietly checking less than it claims --
15the failure mode this repository has hit repeatedly -- so it gets its own
16check rather than an assumption.
17
18The proof is a set comparison against an INDEPENDENTLY re-derived truth:
19
20* This checker re-runs app discovery itself (:func:`discover_apps`) rather
21 than trusting the ``all-apps.txt`` a shard wrote. A shard vouching for its
22 own idea of what the tree contains proves nothing -- if its discovery broke,
23 its manifest and its self-report break together and agree.
24* Every shard ``1..N`` must have left a manifest. A missing one is a shard
25 that did not run.
26* The union of the manifests must equal the discovered set exactly: a missing
27 app FAILS (it was never built), and an app claimed by two shards FAILS (the
28 stride is broken, so some other app is missing too).
29* ``all-apps.txt`` must agree with the re-derived set, which catches the shards
30 having discovered a *different* tree than this checker sees.
31
32Run::
33
34 check_build_shard_union.py --shards N # the gate
35 check_build_shard_union.py --selftest # prove it still detects violations
36
37Exit 0 when the shards cover the tree exactly, 1 (naming every discrepancy)
38otherwise.
39"""

60def discover_apps(repo_root: Path) -> list[str]:
61 """Re-derive the buildable app list exactly as all_examples.sh does.
62
63 Mirrors the shell discovery: every ``examples/**/main.c`` whose directory
64 also holds a ``Makefile``, excluding ``examples/host/`` (macOS-only dev
65 tools, never cross-compiled), keyed by the bare directory name and sorted
66 byte-wise. ``sorted()`` on ``str`` is a C-collation sort, matching the
67 ``LC_ALL=C sort`` the shell pins for exactly this reason.
68
69 :param repo_root: Repository root to discover under.
70 :returns: Sorted app names.
71 """
72 examples = repo_root / "examples"
73 if not examples.is_dir():
74 return []
75 apps = set()
76 for main_c in examples.rglob("main.c"):
77 app_dir = main_c.parent
78 rel = app_dir.relative_to(repo_root).as_posix()
79 if rel.startswith("examples/host/"):
80 continue
81 if not (app_dir / "Makefile").is_file():
82 continue
83 apps.add(app_dir.name)
84 return sorted(apps)

87def read_manifest(path: Path) -> list[str]:
88 """Read one newline-delimited manifest, dropping blank lines.
89
90 :param path: Manifest file.
91 :returns: The app names it lists, in file order.
92 """
93 if not path.is_file():
94 return []
95 return [ln.strip() for ln in path.read_text(encoding="ascii").splitlines() if ln.strip()]

98def check_union(repo_root: Path, shards: int) -> tuple[int, list[str]]:
99 """Compare the union of the shard manifests against a fresh discovery.
100
101 :param repo_root: Repository root holding the shard directory.
102 :param shards: The number of shards that were scheduled.
103 :returns: ``(exit_code, problem_lines)``.
104 """
105 problems: list[str] = []
106 shard_dir = repo_root / SHARD_SUBDIR
107 expected = discover_apps(repo_root)
108 if not expected:
109 return RC_VIOLATION, [
110 f"no apps discovered under {repo_root / 'examples'} -- this checker "
111 "cannot vouch for a union it has no truth to compare against.",
112 ]
113 if not shard_dir.is_dir():
114 return RC_VIOLATION, [f"shard manifest directory {shard_dir} does not exist."]
115
116 seen: dict[str, list[int]] = {}
117 for k in range(1, shards + 1):
118 manifest = shard_dir / f"shard-{k}-of-{shards}.txt"
119 names = read_manifest(manifest)
120 if not names:
121 problems.append(
122 f"shard {k}/{shards}: manifest {manifest.name} is missing or empty "
123 "-- that shard never built anything.",
124 )
125 continue
126 for name in names:
127 seen.setdefault(name, []).append(k)
128
129 missing = sorted(set(expected) - set(seen))
130 extra = sorted(set(seen) - set(expected))
131 duplicated = sorted(n for n, ks in seen.items() if len(ks) > 1)
132
133 problems.extend(f"app '{name}' was discovered but no shard built it." for name in missing)
134 problems.extend(f"app '{name}' was built but is not in the discovered set." for name in extra)
135 problems.extend(
136 f"app '{name}' was built by shards {seen[name]} -- the stride is broken."
137 for name in duplicated
138 )
139
140 all_apps = read_manifest(shard_dir / ALL_APPS_NAME)
141 if all_apps and sorted(all_apps) != expected:
142 shard_only = sorted(set(all_apps) - set(expected))
143 here_only = sorted(set(expected) - set(all_apps))
144 problems.append(
145 f"the shards discovered a different tree than this checker: "
146 f"{len(shard_only)} app(s) only they saw {shard_only[:5]}, "
147 f"{len(here_only)} only this checker saw {here_only[:5]}.",
148 )
149
150 if problems:
151 return RC_VIOLATION, problems
152 print(
153 f"check_build_shard_union.py: {shards} shard(s) covered all "
154 f"{len(expected)} discovered app(s) exactly once.",
155 )
156 return RC_OK, []

159def _write_tree(root: Path, apps: list[str]) -> None:
160 """Materialise a throwaway examples/ tree of buildable apps.

162 :param root: Fake repo root.
163 :param apps: App directory names to create.

172def _shard_manifests(root: Path, shards: int, slices: list[list[str]]) -> None:
173 """Write per-shard manifests plus all-apps.txt into a fake tree.
174
175 :param root: Fake repo root.
176 :param shards: Declared shard count.
177 :param slices: Per-shard app-name lists, index 0 == shard 1.
178 """
179 d = root / SHARD_SUBDIR
180 d.mkdir(parents=True, exist_ok=True)
181 every = sorted({a for s in slices for a in s})
182 (d / ALL_APPS_NAME).write_text("".join(f"{a}\n" for a in every), encoding="ascii")
183 for i, names in enumerate(slices, start=1):
184 (d / f"shard-{i}-of-{shards}.txt").write_text(
185 "".join(f"{a}\n" for a in names),
186 encoding="ascii",
187 )

190def selftest() -> int:
191 """Assert the union check fires on every break and stays quiet when whole.
192
193 A completeness checker that stopped detecting an incomplete union would
194 report success forever, so both directions are asserted here and CI runs
195 this before trusting a clean verdict.
196
197 :returns: 0 when every case behaves, 1 otherwise.
198 """
199 cases: list[tuple[str, list[str], int, list[list[str]], bool]] = [
200 # (label, tree apps, shards, slices, expect_pass)
201 ("complete 2-way stride", ["a", "b", "c", "d"], 2, [["a", "c"], ["b", "d"]], True),
202 ("complete 1-way", ["a", "b"], 1, [["a", "b"]], True),
203 ("a shard built nothing", ["a", "b", "c", "d"], 2, [["a", "c"], []], False),
204 ("an app fell through", ["a", "b", "c", "d"], 2, [["a", "c"], ["b"]], False),
205 ("an app built twice", ["a", "b", "c"], 2, [["a", "c"], ["b", "c"]], False),
206 ("an unknown app appeared", ["a", "b"], 2, [["a"], ["b", "ghost"]], False),
207 ]
208 failures = 0
209 for label, tree, shards, slices, expect_pass in cases:
210 with tempfile.TemporaryDirectory() as td:
211 root = Path(td)
212 _write_tree(root, tree)
213 _shard_manifests(root, shards, slices)
214 rc, problems = check_union(root, shards)
215 ok = (rc == RC_OK) if expect_pass else (rc == RC_VIOLATION)
216 verdict = "ok" if ok else "FAIL"
217 want = "pass" if expect_pass else "fire"
218 print(f" [{verdict}] {label}: expected to {want}, rc={rc}")
219 if not ok:
220 failures += 1
221 for p in problems:
222 print(f" {p}")
223
224 # A missing shard directory must fail rather than vacuously pass.
225 with tempfile.TemporaryDirectory() as td:
226 root = Path(td)
227 _write_tree(root, ["a"])
228 rc, _ = check_union(root, 1)
229 ok = rc == RC_VIOLATION
230 print(f" [{'ok' if ok else 'FAIL'}] absent manifest dir: expected to fire, rc={rc}")
231 failures += 0 if ok else 1
232
233 # An empty tree must fail rather than report "0 of 0 covered".
234 with tempfile.TemporaryDirectory() as td:
235 root = Path(td)
236 (root / SHARD_SUBDIR).mkdir(parents=True)
237 rc, _ = check_union(root, 1)
238 ok = rc == RC_VIOLATION
239 print(f" [{'ok' if ok else 'FAIL'}] empty tree: expected to fire, rc={rc}")
240 failures += 0 if ok else 1
241
242 if failures:
243 print(f"check_build_shard_union.py --selftest: {failures} case(s) FAILED", file=sys.stderr)
244 return RC_VIOLATION
245 print("check_build_shard_union.py --selftest: all cases pass.")
246 return RC_OK

250 """Entry point.

252 :param argv: Command-line arguments without the program name.

266 if args.selftest:
267 return selftest()
268 if args.shards is None or args.shards < 1:

277 for p in problems:
278 print(f" {p}", file=sys.stderr)
279 print(
280 "\n Every app must be built by exactly one shard. Do NOT relax this to\n"

282 " stack-usage aggregate downstream would still clear its floor on the\n"
