249[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_dotf_demo_sample(uint8_t* out_ok)
250{
251 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
252
253 uint32_t st0 = 0U;
254 const ra8_err_t st0_err = ra8_dotf_run_self_test((uint8_t)k_dotf_demo_ch0, &st0);
255 g_dotf_st0_snap = st0;
256
257 uint32_t st1 = 0U;
258 const ra8_err_t st1_err = ra8_dotf_run_self_test((uint8_t)k_dotf_demo_ch1, &st1);
259 g_dotf_st1_snap = st1;
260
261 uint32_t reg00 = 0U;
262 const ra8_err_t stat_err = ra8_dotf_get_status((uint8_t)k_dotf_demo_ch0, &reg00);
263 g_dotf_reg00 = reg00;
264
265 *out_ok = internal_dotf_demo_verdict(st0_err, st1_err, stat_err);
266 if (st0_err != k_ra8_ok) {
267 return st0_err;
268 }
269 if (st1_err != k_ra8_ok) {
270 return st1_err;
271 }
272 return stat_err;
273}

275void main(void)
276{
277 internal_dotf_demo_setup_or_halt();
278 /* Clear PRIMASK so SysTick can dispatch and ra8_delay_ms() uses the
279 * SysTick path (ra8_emulator does not advance DWT_CYCCNT). No maskable NVIC
280 * source is armed; DOTF raises no IRQ of its own. */
281 ra8_isr_globals_enable();
282
283 const ra8_err_t init_err = ra8_dotf_init();
284 g_dotf_init_err = (uint32_t)init_err;
285 if (init_err != k_ra8_ok) {
286 internal_dotf_demo_panic_halt();
287 }
288
289 while (1) {
290 uint8_t ok = 0U;
291 const ra8_err_t err = internal_dotf_demo_sample(&ok);
292 const uint8_t good = (err == k_ra8_ok && ok != 0U) ? 1U : 0U;
293 g_dotf_ok = (uint32_t)good;
294 if (good != 0U) {
295 (void)ra8_board_uart_console_write(s_dotf_demo_ok_msg,
296 (size_t)(sizeof(s_dotf_demo_ok_msg) - 1U));
297 (void)ra8_board_led_toggle(k_ra8_board_led1);
298 } else {
299 (void)ra8_board_uart_console_write(s_dotf_demo_bad_msg,
300 (size_t)(sizeof(s_dotf_demo_bad_msg) - 1U));
301 (void)ra8_board_led_toggle(k_ra8_board_led2);
302 }
303 ++g_dotf_heartbeat;
304 ra8_delay_ms(k_dotf_demo_period_ms);
305 }
306 internal_dotf_demo_panic_halt();
307}
