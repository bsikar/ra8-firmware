150static void internal_write_adprc(uint8_t vch, uint32_t adprc)
151{
152 volatile uint32_t* reg = ra8_adc_b_addopcrc(vch);
153 if (reg == nullptr) {
154 return;
155 }
156 const uint32_t adprc_field =
157 (adprc << (uint32_t)k_ra8_addopcrc_bit_adprc) & k_ra8_addopcrc_mask_adprc;
158 /* HUM Ch 53.2.3.4 "ADDOPCRCn : A/D Conversion Data Operation Control C Register n" p 3339 */
159 *reg = (*reg & ~k_ra8_addopcrc_mask_adprc) | adprc_field;

348ra8_err_t ra8_adc_read_channel(uint8_t channel, uint16_t* out_raw)
349{
350 RA8_CHECK_NULL_PTR(out_raw, s_tag, "out_raw must not be nullptr");
351
352 volatile uint32_t* chcr = ra8_adc_b_adchcr(channel);
353 volatile uint32_t* addr = ra8_adc_b_addr(channel);
354 if ((chcr == nullptr) || (addr == nullptr)) {
355 return k_ra8_err_out_of_range;
356 }
357
358 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
359 /* Map this slot onto physical channel == ``channel`` in the
360 * default scan group (driver convention: virtual_ch == physical_ch). */
361 internal_program_channel(channel, channel);
362
363 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
364 /* Kick the conversion via per-group ADSTR[default_group].ADST. */
365 volatile uint32_t* adstr = ra8_adc_b_adstr(k_ra8_adc_default_group);
366 if (adstr != nullptr) {
367 *adstr = k_ra8_adstr_mask_adst;
368 }
369
370 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
371 /* Poll ADSR.ADACT0 -- hardware clears it when the unit becomes idle. */
372 for (uint32_t i = 0U; i < k_ra8_adc_busy_wait_limit; ++i) { /* GCOVR_EXCL_BR_LINE */
373 if ((*ra8_adc_b_adsr() & k_ra8_adsr_mask_adact0) == 0U) { /* GCOVR_EXCL_BR_LINE */
374 *out_raw = (uint16_t)(*addr & k_ra8_addr_mask_data);
375 return k_ra8_ok;
376 }
377 }
378
379 *out_raw = 0U;
380 ra8_log_error(s_tag, "ADACT0 poll timeout");
381 return k_ra8_err_hw_timeout;
382}

752{

754 return k_ra8_err_out_of_range;

764 *ra8_adc_b_admdr() = mdr_now | admd0;

768ra8_err_t ra8_adc_set_compare_window(uint8_t channel, uint16_t low, uint16_t high)
769{
770 if (channel >= k_ra8_adc_b_cmp_tables) {
771 return k_ra8_err_out_of_range;
772 }
773 if (high < low) {
774 return k_ra8_err_invalid_arg;
775 }
776 volatile uint32_t* tbl = ra8_adc_b_adcmptbr(channel);
777 if (tbl == nullptr) {
778 return k_ra8_err_out_of_range;
779 }
780 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
781 /* Pack (low,high) into a single 32-bit ADCMPTBRn slot. */
782 *tbl = ((uint32_t)low << (uint32_t)k_ra8_adcmptbr_bit_low) |
783 ((uint32_t)high << (uint32_t)k_ra8_adcmptbr_bit_high);
784
785 /* CMPMDn -> "inside-window" mode. ADCMPMDR0 carries CMPMD0..3,
786 * ADCMPMDR1 carries CMPMD4..7. Each field is 2 bits at byte stride. */
787 const uint8_t which = (uint8_t)((channel < k_ra8_adcmpmd_per_reg) ? 0U : 1U);
788 volatile uint32_t* mdr = ra8_adc_b_adcmpmdr(which);
789 if (mdr != nullptr) {
790 const uint8_t local_idx = (uint8_t)(channel % k_ra8_adcmpmd_per_reg);
791 const uint8_t shift = (uint8_t)(local_idx * k_ra8_adcmpmd_field_step);
792 const uint32_t mask = ((uint32_t)0x3UL) << shift;
793 const uint32_t code = ((uint32_t)k_ra8_adcmpmd_inside) << shift;
794 *mdr = (*mdr & ~mask) | code;
795 }
796
797 /* ADCMPENR.CMPENn is the main enable for table N. */
798 const uint32_t enable_bit = (uint32_t)(1UL << channel);
799 *ra8_adc_b_adcmpenr() |= enable_bit;
800
801 /* Per-channel CMPTBLEm in ADDOPCRB enables table N for this channel. */
802 volatile uint32_t* opcrb = ra8_adc_b_addopcrb(channel);
803 if (opcrb != nullptr) {
804 const uint32_t cmptble_bit =
805 ((uint32_t)(1UL << channel) << (uint32_t)k_ra8_addopcrb_bit_cmptblem) &
806 k_ra8_addopcrb_mask_cmptblem;
807 *opcrb =
808 (*opcrb & ~k_ra8_addopcrb_mask_cmptblem) | (cmptble_bit & k_ra8_addopcrb_mask_cmptblem);
809 }
810 return k_ra8_ok;
811}

813ra8_err_t ra8_adc_set_oversampling(uint8_t channel, ra8_adc_oversample_t mode)
814{
815 uint32_t avemd = 0U;
816 uint32_t adc = 0U;
817 const ra8_err_t enc_err = internal_oversample_encode(mode, &avemd, &adc);
818 RA8_RETURN_ON_ERROR(enc_err, s_tag, "set_oversampling: invalid mode"); /* GCOVR_EXCL_BR_LINE */
819
820 volatile uint32_t* opcrb = ra8_adc_b_addopcrb(channel);
821 if (opcrb == nullptr) {
822 return k_ra8_err_out_of_range;
823 }
824 /* HUM Ch 53 "16-bit A/D Converter (ADC16H)" p 3308 */
825 const uint32_t avemd_field =
826 (avemd << (uint32_t)k_ra8_addopcrb_bit_avemd) & k_ra8_addopcrb_mask_avemd;
827 const uint32_t adc_field = (adc << (uint32_t)k_ra8_addopcrb_bit_adc) & k_ra8_addopcrb_mask_adc;
828 const uint32_t cleared = *opcrb & ~(k_ra8_addopcrb_mask_avemd | k_ra8_addopcrb_mask_adc);
829 *opcrb = cleared | avemd_field | adc_field;
830 return k_ra8_ok;
