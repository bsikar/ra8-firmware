133[[nodiscard]] static bool kat_one(const uint8_t* msg, uint32_t len, const uint8_t* expect)
134{
135 uint8_t digest[k_ra8_rsip_sha256_digest_bytes] = {};
136 if (ra8_rsip_sha256(msg, len, digest) != k_ra8_ok) {
137 return false;
138 }
139 return memcmp(digest, expect, (size_t)k_ra8_rsip_sha256_digest_bytes) == 0;
140}

144int32_t main(void)
145{
146 kat_setup_or_halt();
147 ra8_isr_globals_enable();
148
149 while (1) {
150 bool ok = true;
151 if (!kat_one(k_msg_abc, (uint32_t)sizeof(k_msg_abc), k_exp_abc)) {
152 ok = false;
153 }
154 if (!kat_one(k_msg_two_block, (uint32_t)k_kat_msg2_len, k_exp_two_block)) {
155 ok = false;
156 }
157 if (!kat_one(k_msg_abc, 0U, k_exp_empty)) { /* zero-length: msg ignored */
158 ok = false;
159 }
160
161 if (ok) {
162 (void)ra8_board_uart_console_write(k_kat_msg_ok, (size_t)(sizeof(k_kat_msg_ok) - 1U));
163 (void)ra8_board_led_toggle(k_ra8_board_led1);
164 } else {
165 (void)ra8_board_uart_console_write(k_kat_msg_fail, (size_t)(sizeof(k_kat_msg_fail) - 1U));
166 (void)ra8_board_led_toggle(k_ra8_board_led2);
167 }
168 ra8_delay_ms(k_kat_period_ms);
169 }
170 kat_panic_halt();
171 return 0;
172}
