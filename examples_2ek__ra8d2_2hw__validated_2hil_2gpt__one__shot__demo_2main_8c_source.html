138static bool gpt_os_demo_wait_ovf(void)
139{
140 enum : uint32_t { k_poll_budget = 200000U };
141 for (uint32_t i = 0U; i < k_poll_budget; ++i) {
142 uint32_t status = 0U;
143 if (ra8_gpt_get_status((uint8_t)k_gpt_os_demo_channel, &status) != k_ra8_ok) {
144 return false;
145 }
146 if ((status & (uint32_t)k_ra8_gpt_status_overflow) != 0U) {
147 (void)ra8_gpt_clear_status((uint8_t)k_gpt_os_demo_channel,
148 (uint32_t)k_ra8_gpt_status_overflow);
149 return true;
150 }
151 }
152 return false;
153}

157int32_t main(void)
158{
159 gpt_os_demo_setup_or_halt();
160 ra8_isr_globals_enable();
161
162 if (gpt_os_demo_arm() != k_ra8_ok) {
163 gpt_os_demo_panic_halt();
164 }
165
166 while (1) {
167 if (ra8_gpt_start_free_run((uint8_t)k_gpt_os_demo_channel, (uint32_t)k_gpt_os_demo_period) !=
168 k_ra8_ok) {
169 g_gpt_one_shot_mismatch += 1U;
170 ra8_delay_ms((uint32_t)k_gpt_os_demo_rearm_delay_ms);
171 continue;
172 }
173 if (gpt_os_demo_wait_ovf()) {
174 g_gpt_one_shot_match += 1U;
175 (void)ra8_board_led_toggle(k_ra8_board_led1);
176 } else {
177 g_gpt_one_shot_mismatch += 1U;
178 }
179 ra8_delay_ms((uint32_t)k_gpt_os_demo_rearm_delay_ms);
180 }
181 gpt_os_demo_panic_halt();
182 return 0;
183}
