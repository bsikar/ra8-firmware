67typedef enum : uintptr_t {
68#ifdef RA8_PERIPH_NS_ALIAS
69 /* TrustZone Non-secure build (#96): reach MSTPCRA..E through the IDAU
70 * bit[28]=1 Non-secure alias (+0x1000_0000). Per-bit module-stop security
71 * follows the peripheral's PSAR attribution, so once the Secure side marks
72 * USBFS0 Non-secure (PSARB11) the NS image may clear MSTPCRB.MSTPB11 here;
73 * Secure-owned bits in the same word are write-ignored from NS. */
74 k_ra8_mstp_base_addr = 0x50203000UL,
75#else
76 k_ra8_mstp_base_addr = 0x40203000UL,
