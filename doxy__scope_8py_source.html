3"""Which files ``doxy_audit.py`` reads, in each of its two scopes.
4
5The auditor deliberately has two: the function gate covers ``libs/``, ``src/``
6and ``port/``, while the member gate is repo-wide. That difference is a
7recorded decision, not an oversight -- see the KNOWN SCOPE GAP note in
8``doxy_audit``'s docstring, which measures what closing it would cost -- so
9both lists live here, side by side, where the asymmetry is visible rather than
10buried 800 lines apart.
11
12Each scope also owns its own vacuity floor, and :func:`function_files` /
13:func:`member_files` are the materialising accessors every caller must use.
14A generator that yields nothing costs an auditor nothing to consume and makes
15it print ``gaps=0 (PASS)`` -- a perfectly documented tree and a completely
16broken walk are the same output. The floors live beside the scope lists they
17guard because that is the pair that has to stay consistent; both exit 2, so
18"the scan broke" stays distinguishable from "the tree is undocumented".
19"""

37def repo_root() -> Path:
38 """Return the root every path in this auditor is reported relative to."""

42@contextlib.contextmanager
43def override_repo_root(root: Path) -> Iterator[None]:
44 """Report paths relative to ``root`` for the duration of the block.
45
46 Used only by the selftest, which audits a synthetic tree in a temporary
47 directory. It used to rebind a module-global ``repo_root()`` instead, which
48 stopped working the moment the auditor was split across modules: the other
49 modules hold their own imported copy, so the rebinding would be invisible
50 and the suite would silently assert against the real checkout.
51 """
52 global _repo_root # noqa: PLW0603 # single-homed override; see docstring
53 previous = _repo_root
54 _repo_root = root
55 try:
56 yield
57 finally:
58 _repo_root = previous

61SCAN_DIRS = ["libs", "src", "port"]
62EXCLUDE_PARTS = {"third_party", "build", ".git"}
63
64# Top-level dirs scanned by the report-only member audit (--members). Wider
65# than SCAN_DIRS on purpose: the member/enum/macro documentation bar is
66# repo-wide (CLAUDE.md "these standards apply to EVERY first-party file"), so
67# the fallout report must cover examples/, tools/, and tests/ too. Vendored
68# SOUP (libs/third_party/) and generated tables (libs/ra8_fonts/) stay exempt, the
69# same as the function audit.
70MEMBER_SCAN_DIRS = ["libs", "src", "port", "examples", "tools", "tests"]
71MEMBER_EXCLUDE_PARTS = {"third_party", "ra8_fonts", "build", "build-cov", "_deps", ".git"}
72
73# A tree this size cannot legitimately collapse to a handful of files. If
74# either walk returns less than its floor, something broke (an unreachable
75# repo root, a renamed SCAN_DIRS entry) and reporting zero gaps would be a
76# lie -- an undocumented function cannot be found in a file nobody opened.
77# Measured 2026-07-28: 885 files in the function scope, 2116 in the member
78# scope. Same trip-wire as lint_targets.TRACKED_FLOOR.
79FUNCTION_FILE_FLOOR = 700
80MEMBER_FILE_FLOOR = 1700
81
82# Minimum path depth to form a two-segment module label (e.g. "libs/ra8_hal").
83MODULE_PATH_MIN_DEPTH = 2

86def iter_function_files() -> Iterator[Path]:
87 """Yield every first-party .c/.h the FUNCTION gate audits.
88
89 Both the strict gate and the report generator walked this identically
90 inline; sharing one iterator means the report can never describe a
91 different set of files than the gate enforces over.
92 """
93 for top in SCAN_DIRS:
94 root = repo_root() / top
95 if not root.is_dir():
96 continue
97 for dirpath, dirnames, filenames in os.walk(root):
98 dirnames[:] = [d for d in dirnames if d not in EXCLUDE_PARTS]
99 for fn in filenames:
100 if not fn.endswith((".c", ".h")):
101 continue
102 p = Path(dirpath) / fn
103 if any(part in EXCLUDE_PARTS for part in p.relative_to(repo_root()).parts):
104 continue
105 yield p

108def _iter_member_files(explicit: list[str]) -> Iterator[Path]:
109 """Yield first-party .c/.h paths for the member audit.
110
111 ``explicit`` is a list of user-supplied paths; if non-empty those exact
112 files are audited, otherwise MEMBER_SCAN_DIRS is walked.
113 """
114 if explicit:
115 for raw_path in explicit:
116 p = Path(raw_path)
117 if p.is_file(): # relative to CWD or absolute
118 yield p
119 elif (repo_root() / raw_path).is_file(): # relative to the repo root
120 yield repo_root() / raw_path
121 return
122 for top in MEMBER_SCAN_DIRS:
123 root = repo_root() / top
124 if not root.is_dir():
125 continue
126 for dirpath, dirnames, filenames in os.walk(root):
127 dirnames[:] = [d for d in dirnames if d not in MEMBER_EXCLUDE_PARTS]
128 for fn in filenames:
129 if not fn.endswith((".c", ".h")):
130 continue
131 p = Path(dirpath) / fn
132 if any(part in MEMBER_EXCLUDE_PARTS for part in p.relative_to(repo_root()).parts):
133 continue
134 yield p

137def _fatal_below_floor(kind: str, count: int, floor: int) -> None:
138 """Abort with exit 2 when a scope walk enumerated fewer files than its floor.
139
140 Args:
141 kind: Which scope collapsed, named as the caller's flag would spell it
142 (e.g. ``"function"`` or ``"member"``).
143 count: How many files the walk actually yielded.
144 floor: The measured floor that count must reach.
145
146 Raises:
147 SystemExit: Always, with code 2, when ``count`` is below ``floor``.
148 """
149 if count >= floor:
150 return
151 sys.stderr.write(
152 f"doxy_audit.py: FATAL -- only {count} file(s) in the {kind} scope, "
153 f"floor is {floor}.\n"
154 " A collapsed scope reports a documented tree because it read nothing.\n"
155 )
156 sys.exit(2)

159def function_files() -> list[Path]:
160 """Every file the FUNCTION scope audits, materialised and floor-checked.
161
162 The accessor exists so no caller can consume :func:`iter_function_files`
163 lazily and never notice that it yielded nothing: a generator that produces
164 no items looks exactly like a fully documented tree to the auditor
165 downstream.
166
167 Returns:
168 The function-scope paths, in walk order.
169
170 Raises:
171 SystemExit: With code 2 when fewer than FUNCTION_FILE_FLOOR files were
172 found -- a collapsed walk must fail, never read as clean.
173 """
174 files = list(iter_function_files())
175 _fatal_below_floor("function", len(files), FUNCTION_FILE_FLOOR)
176 return files

179def member_files(explicit: list[str]) -> list[Path]:
180 """Every file the MEMBER scope audits, materialised and floor-checked.
181
182 The floor applies to the repo-wide walk only. An ``explicit`` list is a
183 deliberately narrowed scope supplied on the command line -- it is allowed
184 to be one file, or to filter to none -- while a collapsed walk is a broken
185 enumeration reporting a documented tree.
186
187 Args:
188 explicit: User-supplied paths; empty means walk MEMBER_SCAN_DIRS.
189
190 Returns:
191 The member-scope paths, in walk order.
192
193 Raises:
194 SystemExit: With code 2 when the repo-wide walk found fewer than
195 MEMBER_FILE_FLOOR files.
196 """
197 files = list(_iter_member_files(explicit))
198 if not explicit:
199 _fatal_below_floor("member", len(files), MEMBER_FILE_FLOOR)
200 return files
