222RA8_EXPECTS_LOCK("ra8_fs_lock")
223static ra8_err_t
224priv_setattr_locked(ra8_fs_mount_t* handle, const char* path, uint8_t set_mask, uint8_t clear_mask)
225{
226 if (handle == nullptr || path == nullptr) {
227 return k_ra8_err_null_ptr;
228 }
229 if (handle->in_use == 0U) {
230 return k_ra8_err_invalid_state;
231 }
232 /* Confine the change to the four host-controlled bits, and reject a bit told
233 * to be both set and cleared. Any of the three catches makes the whole request
234 * invalid_arg, so the DIRECTORY / VOLUME_ID / long-name bits can never be
235 * flipped and a caller cannot give contradictory instructions. */
236 const uint8_t settable = (uint8_t)k_ra8_fs_attr_settable;
237 if (((set_mask & (uint8_t)~settable) != 0U) || ((clear_mask & (uint8_t)~settable) != 0U) ||
238 ((set_mask & clear_mask) != 0U)) {
239 return k_ra8_err_invalid_arg;
240 }
241 const char* p = path;
242 while (*p == '/') {
243 p++;
244 }
245 if (*p == '\0') {
246 return k_ra8_err_invalid_arg; /* the volume root has no entry to patch */
247 }
248 if (handle->type == k_ra8_fs_type_exfat) {
249 return priv_setattr_exfat(handle, path, set_mask, clear_mask);
250 }
251 return priv_setattr_fat(handle, path, set_mask, clear_mask);
252}
