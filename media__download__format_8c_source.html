20RA8_PRIV ra8_err_t priv_media_download_memory_write_at(media_download_memory_t* memory,
21 uint64_t offset,
22 const uint8_t* source,
23 uint32_t requested,
24 uint32_t* written)
25{
26 if (written == nullptr) {
27 return k_ra8_err_null_ptr;
28 }
29 *written = 0U;
30 if (memory == nullptr) {
31 return k_ra8_err_null_ptr;
32 }
33 if (source == nullptr) {
34 return k_ra8_err_null_ptr;
35 }
36 if (offset > memory->capacity) {
37 return k_ra8_err_invalid_size;
38 }
39 if (requested > ((uint64_t)memory->capacity - offset)) {
40 return k_ra8_err_invalid_size;
41 }
42 if (requested != 0U) {
43 (void)memcpy(&memory->bytes[(size_t)offset], source, requested);
44 }
45 const uint32_t end = (uint32_t)offset + requested;
46 if (end > memory->length) {
47 memory->length = end;
48 }
49 *written = requested;
50 return k_ra8_ok;
51}

70 if (offset > memory->length) {
71 return k_ra8_err_invalid_size;
72 }
73 if (requested > (memory->length - offset)) {
74 return k_ra8_err_invalid_size;

76 if (requested != 0U) {

78 }
79 *out_read = requested;

234RA8_PRIV ra8_err_t priv_media_download_format_rabook(const uint8_t* source,
235 size_t source_size,
236 const media_download_format_config_t* config,
237 media_download_format_workspace_t* workspace,
238 uint64_t* packed_size)
239{
240 ra8_err_t result =
241 internal_validate_format_request(source, source_size, config, workspace, packed_size);
242 if (result != k_ra8_ok) {
243 return result;
244 }
245 media_download_memory_t image = {.bytes = workspace->image, .capacity = workspace->image_cap};
246 media_download_memory_t flat = {.bytes = workspace->flat, .capacity = workspace->flat_cap};
247 media_download_memory_t packed = {.bytes = workspace->packed, .capacity = workspace->packed_cap};
248 ra8_rabook_comic_t comic = {};
249 result = ra8_rabook_comic_init(&comic,
250 &workspace->comic,
251 priv_media_download_memory_read,
252 internal_image_write,
253 &image,
254 config->max_image_edge,
255 config->pixel_format);
256 if (result == k_ra8_ok) {
257 result = ra8_rabook_comic_add_page(&comic, config->page_id, source, source_size);
258 }
259 uint32_t flat_size = 0U;
260 if (result == k_ra8_ok) {
261 result =
262 ra8_rabook_comic_finish(&comic, &config->metadata, internal_flat_append, &flat, &flat_size);
263 }
264 if (result == k_ra8_ok) {
265 if (flat_size != flat.length) {
266 result = k_ra8_err_validation_failed;
267 }
268 }
269 if (result == k_ra8_ok) {
270 result = ra8_rabook_container_write(priv_media_download_memory_read,
271 &flat,
272 flat.length,
273 config->chunk_bytes,
274 internal_packed_write,
275 &packed,
276 &workspace->container,
277 packed_size);
278 }
279 if (result == k_ra8_ok) {
280 if (*packed_size != packed.length) {
281 *packed_size = 0U;
282 result = k_ra8_err_validation_failed;
283 }
284 }
285 return result;
286}
