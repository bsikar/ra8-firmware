105static bool s_ends_with_ci(const char* name, uint16_t len, const char* ext)
106{
107 size_t elen = 0U;
108 for (uint8_t i = 0U; i < (uint8_t)k_comic_ext_max; ++i) { /* bound: fixed max ext */
109 if (ext[i] == '\0') {
110 break;
111 }
112 elen = (size_t)i + 1U;
113 }
114 if ((size_t)len < elen) {
115 return false;
116 }
117 const char* tail = &name[(size_t)len - elen];
118 for (size_t i = 0U; i < elen; ++i) { /* bound: elen <= k_comic_ext_max */
119 if (s_lower((uint8_t)tail[i]) != s_lower((uint8_t)ext[i])) {
120 return false;
121 }
122 }
123 return true;
124}

263RA8_INTERNAL
264static void s_sort(ra8_comic_t* c)
265{
266 for (uint32_t i = 1U; i < c->page_count; ++i) { /* bound: page_count <= page_cap */
267 const ra8_comic_page_t key = c->pages[i];
268 const char* const kn = &c->names[key.name_off];
269 uint32_t j = i;
270 while (j > 0U) { /* bound: j strictly decreases toward 0 */
271 const ra8_comic_page_t prev = c->pages[j - 1U];
272 if (s_name_cmp(&c->names[prev.name_off], prev.name_len, kn, key.name_len) <= 0) {
273 break;
274 }
275 c->pages[j] = prev;
276 j -= 1U;
277 }
278 c->pages[j] = key;
279 }
280}

364RA8_INTERNAL
365static ra8_err_t s_open_detect(ra8_comic_t* c)
366{
367 uint8_t sig[k_ra8_comic_magic_len] = {};
368 const size_t got = c->read(c->ctx, 0U, sig, sizeof(sig));
369 if (got < (size_t)k_comic_sig_min) {
370 return k_ra8_err_invalid_size;
371 }
372 if (s_is_zip(sig)) {
373 c->kind = k_ra8_comic_kind_cbz;
374 c->stream.read = c->read;
375 c->stream.ctx = c->ctx;
376 c->stream.size = c->size;
377 return ra8_comic_cbz_open(c);
378 }
379 const ra8_err_t rerr = ra8_rar_open(&c->rar, c->read, c->ctx, c->size);
380 if (rerr == k_ra8_ok) {
381 c->kind = k_ra8_comic_kind_cbr;
382 return ra8_comic_cbr_open(c);
383 }
384 /* tar has no leading magic bytes; the walker's open probes block 0
385 * (ustar magic + checksum), so a non-tar file is rejected cheaply. */
386 const ra8_err_t terr = ra8_unarch_tar_open(&c->tar, c->read, c->ctx, c->size, nullptr);
387 if (terr == k_ra8_ok) {
388 c->kind = k_ra8_comic_kind_cbt;
389 return ra8_comic_cbt_open(c);
390 }
391 return k_ra8_err_not_supported;
392}
