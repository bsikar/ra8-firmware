115{
116 if (!s_gfx_text_state.initialized) {
117 return k_ra8_err_not_initialized;
118 }
119 if (src == nullptr) {
120 return k_ra8_err_invalid_arg;
121 }
122 if (zoom <= 0) {
123 return k_ra8_err_invalid_arg;
124 }
125 if ((src_w <= 0) || (src_h <= 0)) {
126 return k_ra8_err_invalid_arg;
127 }
128 /* Clamp the sampled window to the source image bounds; an off-image edge draws
129 * only the in-image portion at its natural (dst + offset*zoom) position. A
130 * non-positive sw/sh collapses the range, so nothing is drawn. */
131 const int32_t x_lo = (sx > 0) ? sx : 0;
132 const int32_t y_lo = (sy > 0) ? sy : 0;
133 const int32_t x_hi = ((sx + sw) < src_w) ? (sx + sw) : src_w;
134 const int32_t y_hi = ((sy + sh) < src_h) ? (sy + sh) : src_h;
135 for (int32_t py = y_lo; py < y_hi; ++py) {
136 const int32_t by = dst_y + ((py - sy) * zoom);
137 for (int32_t px = x_lo; px < x_hi; ++px) {
138 internal_gray4_block(dst_x + ((px - sx) * zoom),
139 by,
140 zoom,
141 internal_gray4_color(src, src_w, px, py));
142 }
143 }
144 return k_ra8_ok;
145}
