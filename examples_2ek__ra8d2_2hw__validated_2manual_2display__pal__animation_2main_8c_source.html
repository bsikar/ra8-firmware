201static void app_bringup_clocks(void)
202{
203 uint32_t cpuclk0_hz = 0U;
204 if (ra8_cgc_init() != k_ra8_ok) {
205 app_panic_halt();
206 }
207 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
208 app_panic_halt();
209 }
210 if (ra8_mstp_init() != k_ra8_ok) {
211 app_panic_halt();
212 }
213 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
214 app_panic_halt();
215 }
216 if (ra8_board_led_init(k_ra8_board_led_blue) != k_ra8_ok) {
217 app_panic_halt();
218 }
219 if (ra8_board_led_init(k_ra8_board_led_red) != k_ra8_ok) {
220 app_panic_halt();
221 }
222 ra8_isr_globals_enable();
223}

273static void app_paint_bars(uint32_t scroll_offset)
274{
275 uint16_t* pixels = (uint16_t*)s_fb.pixels;
276 const uint16_t bar_height = (uint16_t)(s_fb.height_px / (uint16_t)k_app_bar_count);
277 for (uint16_t y = 0U; y < s_fb.height_px; ++y) {
278 const uint32_t row = ((uint32_t)y + scroll_offset) / (uint32_t)bar_height;
279 const uint16_t colour = s_palette[row % (uint32_t)k_app_bar_count];
280 const uint32_t row_off = (uint32_t)y * (uint32_t)s_fb.width_px;
281 for (uint16_t x = 0U; x < s_fb.width_px; ++x) {
282 pixels[row_off + (uint32_t)x] = colour;
283 }
284 }
285}

289int32_t main(void)
290{
291 app_bringup_clocks();
292 ra8_delay_ms(k_app_powerup_ms);
293 app_bringup_display();
294
295 /* Pick the refresh hint that matches the bound backend. Continuous-
296 * refresh panels (LCD) prefer "fast" because every flush is just a
297 * memory barrier; e-ink picks GC16-like quality only on full
298 * repaints to avoid ghosting. */
299 const display_refresh_hint_t hint =
300 s_caps.continuous_refresh ? k_display_refresh_fast : k_display_refresh_quality;
301
302 /* Clear once with black so the first scroll-in is clean. */
303 (void)display_clear(s_display, 0x0000U);
304 (void)display_flush(s_display, display_full_rect(s_display), k_display_refresh_init);
305
306 while (1) {
307 app_paint_bars(s_scroll_offset);
308 (void)display_flush(s_display, display_full_rect(s_display), hint);
309 (void)ra8_board_led_toggle(k_ra8_board_led_blue);
310 s_scroll_offset = (s_scroll_offset + 1U) % (uint32_t)s_fb.height_px;
311 ra8_delay_ms((uint32_t)k_app_frame_period_ms);
312 }
313 return 0;
314}
