114static void scb_demo_write_hex32(uint32_t val)
115{
116 static const uint8_t k_hex[] = "0123456789ABCDEF";
117 uint8_t buf[2U + k_scb_demo_hex_digits];
118 buf[0] = (uint8_t)'0';
119 buf[1] = (uint8_t)'x';
120 for (uint32_t i = 0U; i < (uint32_t)k_scb_demo_hex_digits; i++) {
121 const uint32_t shift =
122 ((uint32_t)k_scb_demo_hex_digits - 1U - i) * (uint32_t)k_scb_demo_nibble_bits;
123 const uint32_t nibble = (val >> shift) & (uint32_t)k_scb_demo_nibble_mask;
124 buf[2U + i] = k_hex[nibble];
125 }
126 scb_demo_write(buf, (uint32_t)sizeof(buf));
127}

146static bool scb_demo_probe_and_report(void)
147{

149
150 scb_demo_write(k_scb_demo_vtor_prefix, (uint32_t)(sizeof(k_scb_demo_vtor_prefix) - 1U));

162 scb_demo_write_hex32(fs.cfsr);

197static void scb_demo_setup_or_halt(void)
198{
199 uint32_t cpuclk0_hz = 0U;
200 if (ra8_cgc_init() != k_ra8_ok) {
201 scb_demo_panic_halt();
202 }
203 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
204 scb_demo_panic_halt();
205 }
206 if (ra8_mstp_init() != k_ra8_ok) {
207 scb_demo_panic_halt();
208 }
209 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
210 scb_demo_panic_halt();
211 }
212 if (ra8_board_uart_console_init((uint32_t)k_scb_demo_baud) != k_ra8_ok) {
213 scb_demo_panic_halt();
214 }
215 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
216 scb_demo_panic_halt();
217 }
218}

222int32_t main(void)
223{
224 scb_demo_setup_or_halt();
225 ra8_isr_globals_enable();
226
227 while (1) {
228 const bool clean = scb_demo_probe_and_report();
229 if (clean) {
230 scb_demo_write(k_scb_demo_verdict_pass, (uint32_t)(sizeof(k_scb_demo_verdict_pass) - 1U));
231 } else {
232 scb_demo_write(k_scb_demo_verdict_fail, (uint32_t)(sizeof(k_scb_demo_verdict_fail) - 1U));
233 }
234 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
235 break;
236 }
237 ra8_delay_ms((uint32_t)k_scb_demo_period_ms);
238 }
239 scb_demo_panic_halt();
240 return 0;
241}
