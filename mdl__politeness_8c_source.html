74RA8_INTERNAL static uint32_t draw_range(uint64_t* state, uint32_t min_ms, uint32_t max_ms)
75{
76 if (max_ms < min_ms) {
77 max_ms = min_ms;
78 }
79 /* 64-bit span keeps the full-range (min=0, max=UINT32_MAX) case from wrapping
80 * while preserving the exact modulo of the original jitter for smaller spans. */
81 const uint64_t span = (uint64_t)(max_ms - min_ms) + 1U;
82 return min_ms + (uint32_t)(next_rand(state) % span);
83}

159RA8_INTERNAL static bool parse_http_date(const char* value, int64_t now_wall_s, uint32_t* out_ms)
160{
161 static const char* const k_fmts[] = {
162 "%a, %d %b %Y %H:%M:%S GMT", /* IMF-fixdate: Sun, 06 Nov 1994 08:49:37 GMT */
163 "%A, %d-%b-%y %H:%M:%S GMT", /* RFC 850: Sunday, 06-Nov-94 08:49:37 GMT */
164 };
165 for (size_t i = 0U; i < (sizeof(k_fmts) / sizeof(k_fmts[0])); ++i) {
166 struct tm tmv = {};
167 if (strptime(value, k_fmts[i], &tmv) != nullptr) {
168 const time_t t = timegm(&tmv);
169 if (t == (time_t)-1) {
170 continue;
171 }
172 *out_ms = ms_from_secs((int64_t)t - now_wall_s);
173 return true;
174 }
175 }
176 return false;
177}

179bool mdl_retry_after_parse(const char* value, int64_t now_wall_s, uint32_t* out_ms)
180{
181 if ((value == nullptr) || (out_ms == nullptr)) {
182 return false;
183 }
184 while ((*value == ' ') || (*value == '\t')) {
185 ++value;
186 }
187 if (*value == '\0') {
188 return false;
189 }
190 if (all_digits(value)) {
191 *out_ms = ms_from_secs((int64_t)strtoull(value, nullptr, k_dec_base));
192 return true;
193 }
194 return parse_http_date(value, now_wall_s, out_ms);
195}

214RA8_INTERNAL static int64_t gov_now(const mdl_governor_t* g)
215{
216 if (g->now_fn != nullptr) {
217 return g->now_fn(g->now_ctx);
218 }
219 /* CLOCK_MONOTONIC, not CLOCK_REALTIME (#509). The governor only ever
220 * subtracts two readings -- token refill, request spacing, the backoff
221 * gate -- and CLOCK_REALTIME is steppable: an NTP correction forward makes
222 * the governor believe the spacing has elapsed and hammer the remote host,
223 * and one backward makes it wait out the step. A ~4 minute backward step
224 * was measured happening repeatedly on a fleet host, so this is an observed
225 * hazard rather than a theoretical one. */
226 struct timespec ts = {};
227 (void)clock_gettime(CLOCK_MONOTONIC, &ts);
228 return ((int64_t)ts.tv_sec * (int64_t)k_ms_per_s) + ((int64_t)ts.tv_nsec / (int64_t)k_ns_per_ms);
229}

294RA8_INTERNAL static int64_t gov_schedule(mdl_governor_t* g, mdl_host_rec_t* rec, int64_t now)
295{
296 const int64_t interval = gov_interval_ms(&g->cfg);
297 const int64_t cap = gov_cap_ms(&g->cfg);
298 const int64_t elapsed = max_i64(now - rec->last_ms, 0);
299 rec->credit_ms = min_i64(rec->credit_ms + elapsed, cap);
300 const int64_t rate_wait = (rec->credit_ms < interval) ? (interval - rec->credit_ms) : 0;
301 const int64_t target = max_i64(now + rate_wait, rec->earliest_next_ms);
302 const int64_t wait = target - now;
303 rec->credit_ms = max_i64(min_i64(rec->credit_ms + wait, cap) - interval, 0);
304 rec->last_ms = target;
305 return wait;
306}

352ra8_err_t mdl_governor_acquire(mdl_governor_t* g,
353 const char* host,
354 uint32_t jitter_min_ms,
355 uint32_t jitter_max_ms)
356{
357 if (g == nullptr) {
358 return k_ra8_ok; /* pacing disabled -- matches a NULL jitter source */
359 }
360 const int64_t now = gov_now(g);
361 mdl_host_rec_t* rec = gov_get(g, host, now);
362 if (rec == nullptr) {
363 /* NULL host or table full: still space requests with jitter, no tracking. */
364 gov_sleep(g, (int64_t)draw_range(&g->rng, jitter_min_ms, jitter_max_ms));
365 return k_ra8_ok;
366 }
367 if (rec->inflight >= g->cfg.max_inflight) {
368 return k_ra8_err_would_block;
369 }
370 const int64_t wait = gov_schedule(g, rec, now);
371 const int64_t jitter = (int64_t)draw_range(&g->rng, jitter_min_ms, jitter_max_ms);
372 gov_sleep(g, wait + jitter);
373 rec->inflight += 1U;
374 return k_ra8_ok;
375}

389RA8_INTERNAL static int64_t gov_backoff_window(const mdl_gov_cfg_t* cfg, uint16_t level)
390{
391 int64_t window = (int64_t)cfg->backoff_base_ms;
392 const uint16_t shifts = (level > 1U) ? (uint16_t)(level - 1U) : 0U;
393 for (uint16_t i = 0U; (i < shifts) && (i < (uint16_t)k_mdl_gov_level_max); ++i) {
394 window <<= 1;
395 }
396 if ((window < 0) || (window > (int64_t)cfg->backoff_max_ms)) {
397 window = (int64_t)cfg->backoff_max_ms;
398 }
399 return window;
400}

403RA8_INTERNAL static void
404gov_on_throttle(mdl_governor_t* g, mdl_host_rec_t* rec, int64_t now, uint32_t retry_ms)
405{
406 if (rec->backoff_level < (uint16_t)k_mdl_gov_level_max) {
407 rec->backoff_level += 1U;
408 }
409 rec->success_streak = 0U;
410 const int64_t window = gov_backoff_window(&g->cfg, rec->backoff_level);
411 const int64_t backoff = (int64_t)draw_range(&g->rng, 0U, (uint32_t)window);
412 const int64_t gate = now + max_i64(backoff, (int64_t)retry_ms);
413 rec->earliest_next_ms = max_i64(rec->earliest_next_ms, gate);
414}

435void mdl_governor_observe(mdl_governor_t* g, const char* host, long status, const char* retry_after)
436{
437 if (g == nullptr) {
438 return;
439 }
440 const int64_t now = gov_now(g);
441 mdl_host_rec_t* rec = gov_get(g, host, now);
442 if (rec == nullptr) {
443 return; /* NULL host or table full */
444 }
445 uint32_t retry_ms = 0U;
446 const bool has_retry = (retry_after != nullptr) &&
447 mdl_retry_after_parse(retry_after, now / (int64_t)k_ms_per_s, &retry_ms);
448 const bool throttled =
449 (status == (long)k_http_too_many_req) || (status == (long)k_http_unavailable);
450 if (throttled) {
451 gov_on_throttle(g, rec, now, retry_ms);
452 } else {
453 gov_on_success(g, rec, now, has_retry, retry_ms);
454 }
455}
