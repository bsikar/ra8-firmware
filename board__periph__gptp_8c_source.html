212static bool gptp_decode_timer(uint64_t off, uint32_t* out_timer, uint64_t* out_inoff)
213{
214 if (off < (uint64_t)k_ra8_gptp_off_timer0) {
215 return false;
216 }
217 const uint64_t rel = off - (uint64_t)k_ra8_gptp_off_timer0;
218 const uint32_t timer = (uint32_t)(rel / (uint64_t)k_ra8_gptp_timer_stride);
219 if (timer >= (uint32_t)k_ra8_gptp_timer_count) {
220 return false;
221 }
222 *out_timer = timer;
223 *out_inoff = rel % (uint64_t)k_ra8_gptp_timer_stride;
224 return true;
225}

248static void gptp_time_now(uint32_t t, uint64_t* out_sec, uint32_t* out_nsec)
249{
250 const gptp_timer_t* u = &s_gptp.timer[t];
251 if (!u->enabled) {
252 /* HUM Ch 35.3.2.1 "PTPTMEC : Timer Enable Configuration Register" p 1927 --
253 * a stopped unit's counters read 0. */
254 *out_sec = 0U;
255 *out_nsec = 0U;
256 return;
257 }
258 const uint32_t free_ns = (uint32_t)(u->acc_fixed >> (uint64_t)k_gptp_subns_shift);
259 uint64_t total_nsec = (uint64_t)u->off_nsec + (uint64_t)free_ns;
260 uint64_t carry_sec = 0U;
261 if (total_nsec >= (uint64_t)k_gptp_ns_per_sec) {
262 total_nsec -= (uint64_t)k_gptp_ns_per_sec;
263 carry_sec = 1U;
264 }
265 const uint64_t total_sec = (u->off_sec + u->acc_sec + carry_sec) & (uint64_t)k_gptp_mask_sec_48;
266 *out_sec = total_sec;
267 *out_nsec = (uint32_t)(total_nsec & (uint64_t)k_gptp_mask_nsec);
268}

286static void gptp_tick_timer(uint32_t t)
287{
288 gptp_timer_t* u = &s_gptp.timer[t];
289 if (!u->enabled) {
290 return;
291 }
292 const uint64_t tiv_off = (uint64_t)k_ra8_gptp_off_timer0 +
293 ((uint64_t)t * (uint64_t)k_ra8_gptp_timer_stride) +
294 (uint64_t)k_ra8_gptp_off_t_ptptivc;
295 /* HUM Ch 35.3.2.3 "PTPTIVCt : Timer t Increment Value Configuration Register
296 * (t = 0, 1)" p 1928 -- 5.27 fixed-point nanoseconds per clk. */
297 const uint64_t tiv = (uint64_t)gptp_shadow_u32(tiv_off);
298 if (tiv == 0U) {
299 return;
300 }
301 u->acc_fixed += tiv * (uint64_t)k_gptp_clk_per_tick;
302 const uint64_t one_sec_fixed = (uint64_t)k_gptp_ns_per_sec << (uint64_t)k_gptp_subns_shift;
303 const uint64_t whole = u->acc_fixed / one_sec_fixed;
304 u->acc_sec += whole;
305 u->acc_fixed -= whole * one_sec_fixed;
306}

313static void gptp_apply_tmec(uint32_t value)
314{
315 /* HUM Ch 35.3.2.1 "PTPTMEC : Timer Enable Configuration Register" p 1927 --
316 * writing 1 to bit q sets TEq; other bits are unaffected (write-1-to-set). */
317 for (uint32_t t = 0U; t < (uint32_t)k_ra8_gptp_timer_count; ++t) {
318 if ((value & (1UL << t)) != 0U) {
319 s_gptp.timer[t].enabled = true;
320 }
321 }
322}

325static void gptp_apply_tmdc(uint32_t value)
326{
327 /* HUM Ch 35.3.2.2 "PTPTMDC : Timer Disable Configuration Register" p 1928 --
328 * writing 1 to bit q clears TEq; a stopped unit's counters read 0, so a stop
329 * is also a clear of the free-running accumulator. */
330 for (uint32_t t = 0U; t < (uint32_t)k_ra8_gptp_timer_count; ++t) {
331 if ((value & (1UL << t)) != 0U) {
332 s_gptp.timer[t].enabled = false;
333 s_gptp.timer[t].acc_sec = 0U;
334 s_gptp.timer[t].acc_fixed = 0U;
335 }
336 }
337}

368static void gptp_commit_offset(uint32_t t)
369{
370 const uint64_t base =
371 (uint64_t)k_ra8_gptp_off_timer0 + ((uint64_t)t * (uint64_t)k_ra8_gptp_timer_stride);
372 /* HUM Ch 35.3.2.6 "PTPTOVCtU : Timer t Offset Value Configuration Register U"
373 * p 1930 -- seconds [47:32]. */
374 const uint64_t ovcu = (uint64_t)gptp_shadow_u32(base + (uint64_t)k_ra8_gptp_off_t_ptptovcu);
375 /* HUM Ch 35.3.2.5 "PTPTOVCMt : Timer t Offset Value Configuration Register M"
376 * p 1929 -- seconds [31:0]. */
377 const uint64_t ovcm = (uint64_t)gptp_shadow_u32(base + (uint64_t)k_ra8_gptp_off_t_ptptovcm);
378 /* HUM Ch 35.3.2.4 "PTPTOVCtL : Timer t Offset Value Configuration Register L"
379 * p 1929 -- nanoseconds [29:0]; this write commits the whole offset. */
380 const uint64_t ovcl = (uint64_t)gptp_shadow_u32(base + (uint64_t)k_ra8_gptp_off_t_ptptovcl);
381 s_gptp.timer[t].off_sec =
382 ((ovcu & (uint64_t)k_gptp_mask_sec_u) << (uint64_t)k_gptp_shift_sec_u) | ovcm;
383 s_gptp.timer[t].off_nsec = (uint32_t)(ovcl & (uint64_t)k_gptp_mask_nsec);
384}

409static bool gptp_read_monitor(uint32_t t, uint64_t inoff, uint32_t* out)
410{
411 gptp_timer_t* u = &s_gptp.timer[t];
412 if (inoff == (uint64_t)k_ra8_gptp_off_t_ptpgptptml) {
413 uint64_t sec = 0U;
414 uint32_t nsec = 0U;
415 gptp_time_now(t, &sec, &nsec);
416 u->latch_gptp_m = (uint32_t)sec;
417 u->latch_gptp_u =
418 (uint32_t)((sec >> (uint64_t)k_gptp_shift_sec_u) & (uint64_t)k_gptp_mask_sec_u);
419 *out = nsec;
420 return true;
421 }
422 if (inoff == (uint64_t)k_ra8_gptp_off_t_ptpgptptmm) {
423 *out = u->latch_gptp_m;
424 return true;
425 }
426 if (inoff == (uint64_t)k_ra8_gptp_off_t_ptpgptptmu) {
427 *out = u->latch_gptp_u;
428 return true;
429 }
430 if (inoff == (uint64_t)k_ra8_gptp_off_t_ptpavtptml) {
431 uint64_t sec = 0U;
432 uint32_t nsec = 0U;
433 gptp_time_now(t, &sec, &nsec);
434 const uint64_t avtp = (sec * (uint64_t)k_gptp_ns_per_sec) + (uint64_t)nsec;
435 u->latch_avtp_u = (uint32_t)(avtp >> (uint64_t)k_gptp_shift_sec_u);
436 *out = (uint32_t)avtp;
437 return true;
438 }
439 if (inoff == (uint64_t)k_ra8_gptp_off_t_ptpavtptmu) {
440 *out = u->latch_avtp_u;
441 return true;
442 }
443 return false;
444}

465static uint64_t gptp_read(uc_engine* uc, uint64_t addr, unsigned size)
466{
467 (void)uc;
468 const uint64_t off = addr - (uint64_t)k_gptp_win_base;
469 if (off == (uint64_t)k_ra8_gptp_off_ptpipv) {
470 /* HUM Ch 35.3.1.1 "PTPIPV : IP Version Register" p 1927 -- read-only, reset
471 * value nonzero; the apps use it as the block-presence probe. */
472 return (uint64_t)k_gptp_ipv_reset;
473 }
474 if (off == (uint64_t)k_ra8_gptp_off_ptptmec) {
475 return (uint64_t)gptp_read_tmec();
476 }
477 if (off == (uint64_t)k_ra8_gptp_off_ptptmdc) {
478 /* HUM Ch 35.3.2.2 "PTPTMDC : Timer Disable Configuration Register" p 1928 --
479 * write-only; reads back 0. */
480 return 0U;
481 }
482 uint32_t timer = 0U;
483 uint64_t inoff = 0U;
484 if (gptp_decode_timer(off, &timer, &inoff)) {
485 uint32_t val = 0U;
486 if (gptp_read_monitor(timer, inoff, &val)) {
487 return (uint64_t)val;
488 }
489 }
490 return gptp_shadow_read(off, size);
491}

508static void gptp_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
509{
510 (void)uc;
511 const uint64_t off = addr - (uint64_t)k_gptp_win_base;
512 if (off == (uint64_t)k_ra8_gptp_off_ptptmec) {
513 gptp_apply_tmec((uint32_t)value);
514 return;
515 }
516 if (off == (uint64_t)k_ra8_gptp_off_ptptmdc) {
517 gptp_apply_tmdc((uint32_t)value);
518 return;
519 }
520 if (off == (uint64_t)k_ra8_gptp_off_ptpipv) {
521 return; /* PTPIPV is read-only (HUM 35.3.1.1 p 1927) -- drop the write. */
522 }
523 gptp_shadow_write(off, size, value);
524 uint32_t timer = 0U;
525 uint64_t inoff = 0U;
526 if (gptp_decode_timer(off, &timer, &inoff) && (inoff == (uint64_t)k_ra8_gptp_off_t_ptptovcl)) {
527 gptp_commit_offset(timer);
528 }
529}

547static void gptp_report(void)
548{
549 for (uint32_t t = 0U; t < (uint32_t)k_ra8_gptp_timer_count; ++t) {
550 const gptp_timer_t* u = &s_gptp.timer[t];
551 if (!u->enabled && (u->acc_sec == 0U) && (u->acc_fixed == 0U)) {
552 continue; /* this unit never ran -- stay quiet. */
553 }
554 uint64_t sec = 0U;
555 uint32_t nsec = 0U;
556 gptp_time_now(t, &sec, &nsec);
557 (void)fprintf(stderr,
558 " GPTP timer%u : gPTP=%llu.%09u s (ESWCLK %llu MHz free-run)\n",
559 (unsigned)t,
560 (unsigned long long)sec,
561 (unsigned)nsec,
562 (unsigned long long)((uint64_t)k_gptp_eswclk_hz / (uint64_t)k_gptp_hz_per_mhz));
563 }
