313static void adc_convert_group(uint32_t group)
314{
315 uint32_t converted = 0U;
316 uint16_t last_code = 0U;
317 for (uint32_t slot = 0U; slot < (uint32_t)k_adc_max_channels; slot++) {
318 const uint32_t chcr = adc_chcr(slot);
319 if ((chcr & (uint32_t)k_adc_chcr_sgsel) != group) {
320 continue;
321 }
322 const uint32_t phys = (chcr & (uint32_t)k_adc_chcr_cnvcs_m) >> (uint32_t)k_adc_chcr_cnvcs_s;
323 if (phys >= (uint32_t)k_adc_ext_chan_base) {
324 /* An internal / extended-analog channel (self-diagnosis, temperature
325 * sensor, internal Vref) reports through ADEXDR[CNVCS - base], not ADDR[]. */
326 last_code = adc_ext_value(phys, group);
327 adc_set_ext_result(phys - (uint32_t)k_adc_ext_chan_base, last_code);
328 converted++;
329 } else if (slot < (uint32_t)k_adc_result_regs) {
330 /* Normal channel -> its virtual-channel result register ADDR[slot].
331 * Resolution is the slot's ADDOPCRCn.ADPRC data-format, so a mid-scale
332 * input converts to that format's half-full-scale code. */
333 last_code = adc_midscale_for_adprc(adc_adprc(slot));
334 adc_set_result(slot, last_code);
335 converted++;
336 }
337 /* else: a normal channel on a virtual slot with no ADDR result register
338 * (the diagnostic slot) produces no ordinary result. */
339 }
340 /* Console ADC tab: one line per scan-group conversion (coalesced -- a whole
341 * group is one push, never one per channel), skipped when the group is empty. */
342 if (converted != 0U) {
343 char ln[k_adc_console_line_cap];
344 (void)snprintf(ln,
345 sizeof(ln),
346 "ADC grp=%u %uch code=%u",
347 (unsigned)group,
348 (unsigned)converted,
349 (unsigned)last_code);
350 board_console_push(k_board_console_ch_adc, ln);
351 }
352}

355static uint64_t adc_read(uc_engine* uc, uint64_t addr, unsigned size)
356{
357 (void)uc;
358 (void)size;
359 const uint64_t off = addr - (uint64_t)k_adc_base;
360 if (off == (uint64_t)k_adc_off_adsr) {
361 /* Conversion is modelled as instantaneous: ADACT0 reads idle so the
362 * driver's busy-poll completes on its first read and fetches ADDR. */
363 return s_adc_reg[adc_word(off)] & ~(uint32_t)k_adc_adact0_mask;
364 }
365 if (off >= (uint64_t)k_adc_span) {
366 return 0U;
367 }
368 return s_adc_reg[adc_word(off)];
369}

372static void adc_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
373{
374 (void)size;
375 const uint64_t off = addr - (uint64_t)k_adc_base;
376 if (off >= (uint64_t)k_adc_span) {
377 return;
378 }
379 const uint64_t str_lo = (uint64_t)k_adc_off_adstr0;
380 const uint64_t str_hi = str_lo + ((uint64_t)k_adc_scan_groups * (uint64_t)k_adc_str_stride);
381 if ((off >= str_lo) && (off < str_hi) && ((value & (uint64_t)k_adc_adst_mask) != 0U)) {
382 const uint32_t group = (uint32_t)((off - str_lo) / (uint64_t)k_adc_str_stride);
383 s_adc_reg[adc_word(off)] = (uint32_t)value;
384 adc_convert_group(group);
385 s_adc_scans++;
386 board_periph_icu_raise_event(uc, (uint16_t)k_event_adc0_scan_end);
387 if (board_periph_trace()) {
388 (void)fprintf(stderr,
389 " [trace] ADC_B scan grp%u -> code=%u\n",
390 group,
391 (unsigned)s_adc_last_code);
392 }
393 return;
394 }
395 s_adc_reg[adc_word(off)] = (uint32_t)value;
396}
