64[[nodiscard]] ra8_err_t ra8_ulpt_init(void)
65{
66 for (uint8_t ch = 0U; ch < (uint8_t)k_ra8_ulpt_channel_count; ++ch) {
67 volatile r_ulpt_regs_t* reg = ra8_ulpt(ch);
68 if (reg == nullptr) {
69 return k_ra8_err_hw_init_failed;
70 }
71 /* HUM Ch 11.2.10 "MSTPCRE : Module Stop Control Register E" p 449 */
72 const ra8_err_t mst_err = ra8_mstp_enable(s_ulpt_mstp_table[ch]);
73 RA8_RETURN_ON_ERROR(mst_err, s_tag, "ulpt_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
74 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 */
75 reg->ULPTCR = 0U;
76 /* HUM Ch 25.2.2 "ULPTMR1 : ULPT Mode Register 1" p 1192 */
77 reg->ULPTMR1 = 0U;
78 /* HUM Ch 25.2.3 "ULPTMR2 : ULPT Mode Register 2" p 1192 */
79 reg->ULPTMR2 = 0U;
80 /* HUM Ch 25.2.4 "ULPTMR3 : ULPT Mode Register 3" p 1195 */
81 reg->ULPTMR3 = 0U;
82 /* HUM Ch 25.2.5 "ULPTIOC : ULPT I/O Control Register" p 1195 */
83 reg->ULPTIOC = 0U;
84 /* HUM Ch 25.2.6 "ULPT : ULPT Counter Register" p 1198 */
85 reg->ULPTCNT = 0U;
86 }
87 ra8_log_info(s_tag, "ulpt_init");
88 return k_ra8_ok;
89}

91[[nodiscard]] ra8_err_t ra8_ulpt_start(uint8_t channel, uint32_t period)
92{
93 if ((uint16_t)channel >= k_ra8_ulpt_channel_count) {
94 return k_ra8_err_invalid_arg;
95 }
96 volatile r_ulpt_regs_t* reg = ra8_ulpt(channel);
97 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
98
99 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 */
100 reg->ULPTCR = 0U;
101 /* HUM Ch 25.2.2 "ULPTMR1 : ULPT Mode Register 1" p 1192 -- timer mode,
102 * TCK1 = 0 selects ULPTLCLK (LOCO-derived 32.768 kHz, HUM Table 9.2
103 * p 320), which survives Software Standby. */
104 reg->ULPTMR1 = 0U;
105 /* HUM Ch 25.2.3 "ULPTMR2 : ULPT Mode Register 2" p 1192 */
106 reg->ULPTMR2 = 0U;
107 /* HUM Ch 25.2.4 "ULPTMR3 : ULPT Mode Register 3" p 1195 */
108 reg->ULPTMR3 = 0U;
109 /* HUM Ch 25.2.6 "ULPT : ULPT Counter Register" p 1198 */
110 reg->ULPTCNT = period;
111 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 */
112 /* TSTART = 1. */
113 reg->ULPTCR = k_ra8_ulpt_mask_tstart;
114
115 ra8_log_info_val(s_tag, "start channel", (uint32_t)channel);
116 return k_ra8_ok;
117}

119[[nodiscard]] ra8_err_t ra8_ulpt_stop(uint8_t channel)
120{
121 if ((uint16_t)channel >= k_ra8_ulpt_channel_count) {
122 return k_ra8_err_invalid_arg;
123 }
124 volatile r_ulpt_regs_t* reg = ra8_ulpt(channel);
125 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
126
127 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 */
128 /* TSTOP toggle. */
129 reg->ULPTCR = k_ra8_ulpt_mask_tstop;
130 reg->ULPTCR = 0U;
131
132 /* Confirm the down-counter actually halted before returning. ULPTCR.TCSTF
133 * (count-status flag) lags the TSTOP request by a few ULPTLCLK (32.768 kHz)
134 * cycles; a caller that re-arms for Software Standby while TCSTF is still
135 * set hangs on the second wake. Poll TCSTF low under a bounded budget
136 * (NASA Power of 10 Rule 2) -- k_ra8_ok when it clears, k_ra8_err_hw_timeout
137 * if the counter never stops. */
138 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 */
139 return ra8_hw_wait_flag_clear8(&reg->ULPTCR,
140 (uint8_t)k_ra8_ulpt_mask_tcstf,
141 (uint32_t)k_ra8_ulpt_stop_poll_max);
142}
