81/* Validate that a typed handle points into the static pool -- see implementation for details. */
82static bool internal_handle_valid(ra8_psa_key_t handle)
83{
84 if (handle == nullptr) {
85 return false;
86 }
87 const struct ra8_psa_key_handle* base = &s_key_pool[0];
88 const struct ra8_psa_key_handle* end = &s_key_pool[k_ra8_psa_max_keys];
89 if ((handle < base) || (handle >= end)) {
90 return false;
91 }
92 return handle->in_use;
93}

199/* Translate ``ra8_psa_key_usage_t`` bitmask to ``PSA_KEY_USAGE_*`` flags -- see implementation for details. */
200static psa_key_usage_t internal_map_usage(ra8_psa_key_usage_t usage)
201{
202 psa_key_usage_t out = 0;
203 if ((usage & k_ra8_psa_usage_sign) != 0U) {
204 out |= PSA_KEY_USAGE_SIGN_HASH;
205 }
206 if ((usage & k_ra8_psa_usage_verify) != 0U) {
207 out |= PSA_KEY_USAGE_VERIFY_HASH;
208 }
209 if ((usage & k_ra8_psa_usage_encrypt) != 0U) {
210 out |= PSA_KEY_USAGE_ENCRYPT;
211 }
212 if ((usage & k_ra8_psa_usage_decrypt) != 0U) {
213 out |= PSA_KEY_USAGE_DECRYPT;
214 }
215 if ((usage & k_ra8_psa_usage_derive) != 0U) {
216 out |= PSA_KEY_USAGE_DERIVE;
217 }
218 return out;
219}

277 slot->attr = *attr;

405ra8_err_t ra8_psa_verify_hash(ra8_psa_key_t handle,
406 ra8_psa_alg_t alg,
407 const uint8_t* hash,
408 size_t hash_len,
409 const uint8_t* sig,
410 size_t sig_len)
411{
412 if ((hash == nullptr) || (sig == nullptr)) {
413 return k_ra8_err_invalid_arg;
414 }
415 if (!s_initialized) {
416 return k_ra8_err_not_initialized;
417 }
418 if (!internal_handle_valid(handle)) {
419 return k_ra8_err_invalid_arg;
420 }
421 if (alg != k_ra8_psa_alg_ecdsa_sha_256) {
422 return k_ra8_err_not_supported;
423 }
424 if ((handle->attr.usage & k_ra8_psa_usage_verify) == 0U) {
425 return k_ra8_err_invalid_arg;
426 }
427 if (hash_len != (size_t)k_ra8_psa_sha256_len) {
428 return k_ra8_err_invalid_size;
429 }
430
431#ifdef RA8_OFF_TARGET
432 if (sig_len != (size_t)k_ra8_psa_sha256_len) {
433 return k_ra8_err_crc_mismatch;
434 }
435 uint8_t buf[k_ra8_psa_max_key_bytes + k_ra8_psa_sha256_len];
436 (void)memcpy(buf, handle->key, handle->key_len);
437 (void)memcpy(&buf[handle->key_len], hash, hash_len);
438 uint8_t digest[32];
439 ra8_psa_fake_sha256_oneshot(buf, handle->key_len + hash_len, digest);
440 /* Constant-time compare. */
441 uint8_t diff = 0U;
442 for (uint32_t i = 0U; i < (uint32_t)k_ra8_psa_sha256_len; ++i) {
443 diff |= (uint8_t)(digest[i] ^ sig[i]);
444 }
445 return (diff == 0U) ? k_ra8_ok : k_ra8_err_crc_mismatch;
446#else
447 const psa_status_t st =
448 psa_verify_hash(handle->psa_id, PSA_ALG_ECDSA(PSA_ALG_SHA_256), hash, hash_len, sig, sig_len);
449 if (st == PSA_SUCCESS) {
450 return k_ra8_ok;
451 }
452 if (st == PSA_ERROR_INVALID_SIGNATURE) {
453 return k_ra8_err_crc_mismatch;
454 }
455 return k_ra8_err_hw_error;
456#endif
457}

695ra8_err_t ra8_psa_crypto_random(uint8_t* out, size_t out_len)
696{
697 if (out == nullptr) {
698 return k_ra8_err_invalid_arg;
699 }
700 if (out_len == 0U) {
701 return k_ra8_err_invalid_size;
702 }
703 if (!s_initialized) {
704 return k_ra8_err_not_initialized;
705 }
706
707#ifdef RA8_OFF_TARGET
708 /* Deterministic xorshift32 -- reproducible host-test entropy.
709 * Constants from Marsaglia 2003 "Xorshift RNGs" J. Stat. Soft. 8(14). */
710 static uint32_t s_state = (uint32_t)k_xs32_seed;
711 for (size_t i = 0U; i < out_len; ++i) {
712 uint32_t x = s_state;
713 x ^= x << (uint32_t)k_xs32_shl_a;
714 x ^= x >> (uint32_t)k_xs32_shr_b;
715 x ^= x << (uint32_t)k_xs32_shl_c;
716 s_state = x;
717 out[i] = (uint8_t)(x & (uint32_t)k_xs32_byte_m);
718 }
719 return k_ra8_ok;
720#else
721 const psa_status_t st = psa_generate_random(out, out_len);
722 if (st != PSA_SUCCESS) {
723 ra8_log_error(k_ra8_psa_tag, "psa_generate_random failed");
724 return k_ra8_err_hw_error;
725 }
726 return k_ra8_ok;
727#endif
