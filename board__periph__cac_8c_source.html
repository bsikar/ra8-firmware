84static void cac_measure(void)
85{
86 /* Report the midpoint of the firmware's programmed window: in-band by
87 * construction, so FERRF stays clear and the measurement reads healthy. */
88 s_cac.cacntbr = (uint16_t)(((uint32_t)s_cac.callvr + (uint32_t)s_cac.caulvr) / 2U);
89 s_cac.castr = (uint8_t)k_cac_castr_mendf; /* MENDF set; FERRF / OVFF clear. */
90 s_cac.meas++;
91}

124static void cac_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
125{
126 (void)uc;
127 (void)size;
128 const uint64_t off = addr - (uint64_t)k_cac_base;
129 if (off == (uint64_t)k_cac_off_cacr0) {
130 s_cac.cacr0 = (uint8_t)value;
131 if (((uint8_t)value & (uint8_t)k_cac_cacr0_cfme) != 0U) {
132 cac_measure(); /* CFME=1 starts a measurement; it completes at once. */
133 }
134 } else if (off == (uint64_t)k_cac_off_caicr) {
135 /* CAICR clear bits are write-1-to-clear of the matching CASTR flags. */
136 if (((uint8_t)value & (uint8_t)k_cac_caicr_ferrfcl) != 0U) {
137 s_cac.castr &= (uint8_t)~(uint8_t)k_cac_castr_ferrf;
138 }
139 if (((uint8_t)value & (uint8_t)k_cac_caicr_mendfcl) != 0U) {
140 s_cac.castr &= (uint8_t)~(uint8_t)k_cac_castr_mendf;
141 }
142 if (((uint8_t)value & (uint8_t)k_cac_caicr_ovffcl) != 0U) {
143 s_cac.castr &= (uint8_t)~(uint8_t)k_cac_castr_ovff;
144 }
145 } else if (off == (uint64_t)k_cac_off_caulvr) {
146 s_cac.caulvr = (uint16_t)value;
147 } else if (off == (uint64_t)k_cac_off_callvr) {
148 s_cac.callvr = (uint16_t)value;
149 }
150}
