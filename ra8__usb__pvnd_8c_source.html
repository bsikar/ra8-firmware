169ra8_err_t ra8_usb_pvnd_init(ra8_usb_speed_t speed)
170{
171 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
172 return k_ra8_err_invalid_arg;
173 }
174 const ra8_err_t usb_err = ra8_usb_device_init(speed);
175 if (usb_err != k_ra8_ok) {
176 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
177 return k_ra8_err_hw_init_failed;
178 }
179 internal_reset_shadow(speed);
180
181 const ra8_err_t pipes_err = internal_configure_pipes(speed);
182 if (pipes_err != k_ra8_ok) {
183 /* GCOVR_EXCL_START: unreachable from the host. internal_configure_pipes
184 * only fails when ra8_usb_configure_endpoint rejects an argument, but the
185 * class layer feeds it compile-time-constant, always-valid pipe / endpoint
186 * / direction / type / max-packet values, so pipes_err is k_ra8_ok on every
187 * host run. No fake register seed can change these constants. */
188 (void)ra8_usb_device_deinit(speed);
189 return pipes_err;
190 /* GCOVR_EXCL_STOP */
191 }
192 s_state.initialized = true;
193 ra8_log_info_val(s_tag, "device-Vendor ready", (uint32_t)speed);
194 return k_ra8_ok;
195}

197ra8_err_t ra8_usb_pvnd_close(void)
198{
199 if (!s_state.initialized) {
200 return k_ra8_err_invalid_state;
201 }
202 (void)ra8_usb_device_attach(s_state.speed, false);
203 const ra8_err_t err = ra8_usb_device_deinit(s_state.speed);
204 s_state.initialized = false;
205 s_state.desc = nullptr;
206 s_state.setup_cb = nullptr;
207 s_state.setup_ctx = nullptr;
208 return err;
209}

235ra8_err_t ra8_usb_pvnd_send(const uint8_t* data, uint16_t len)
236{
237 if (!s_state.initialized) {
238 return k_ra8_err_invalid_state;
239 }
240 if ((data == nullptr) && (len != 0U)) {
241 return k_ra8_err_null_ptr;
242 }
243 if ((len == 0U) || (len > s_state.bulk_max_packet)) {
244 return k_ra8_err_invalid_arg;
245 }
246 return ra8_usb_queue_in(s_state.speed, k_ra8_pvnd_pipe_bulk_in, data, len);
247}

249ra8_err_t ra8_usb_pvnd_recv(uint8_t* buf, uint16_t max_len, uint16_t* got_len)
250{
251 RA8_CHECK_NULL_PTR(buf, s_tag, "recv: buf");
252 RA8_CHECK_NULL_PTR(got_len, s_tag, "recv: got_len");
253 if (!s_state.initialized) {
254 return k_ra8_err_invalid_state;
255 }
256 if (max_len == 0U) {
257 return k_ra8_err_invalid_arg;
258 }
259 uint16_t inout_len = max_len;
260 const ra8_err_t err =
261 ra8_usb_queue_out(s_state.speed, k_ra8_pvnd_pipe_bulk_out, buf, &inout_len, true);
262 if (err == k_ra8_ok) {
263 *got_len = inout_len;
264 } else {
265 *got_len = 0U;
266 }
267 return err;
268}

290ra8_err_t ra8_usb_pvnd_handle_setup(const ra8_usb_setup_t* setup)
291{
292 RA8_CHECK_NULL_PTR(setup, s_tag, "handle_setup: setup");
293 if (!s_state.initialized) {
294 return k_ra8_err_invalid_state;
295 }
296 if (!internal_is_vendor_envelope(setup->bm_request_type)) {
297 return k_ra8_err_not_supported;
298 }
299 if (s_state.setup_cb == nullptr) {
300 /* No application handler -> stall the unknown vendor request. */
301 return ra8_usb_control_response(s_state.speed, false);
302 }
303 const ra8_err_t cb_err = s_state.setup_cb(s_state.setup_ctx, setup);
304 if (cb_err != k_ra8_ok) {
305 return ra8_usb_control_response(s_state.speed, false);
306 }
307 return ra8_usb_control_response(s_state.speed, true);
308}
