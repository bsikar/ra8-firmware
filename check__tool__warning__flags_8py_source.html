4"""Assert the host tools compile first-party sources at the project warning bar.
5
6Why this exists
7---------------
8``tools/media_dl`` compiled ``libs/ra8_jof/*``, ``libs/ra8_hal/ra8_jpeg_sw*``,
9``libs/ra8_io/ra8_io_compress.c`` and three ``libs/ra8_core`` files under a
10blanket ``-w``, with a comment claiming "the repo lints these via its own
11build". It does not: ``-w`` turns off every diagnostic, so the one build in
12the tree that compiles those files for a 64-bit host could not report a
13host-only finding, and neither tool carried ``-Werror`` at all (#309).
14Removing the ``-w`` is a one-line edit that nothing stops a later change from
15re-adding, and it would come back silently -- the build would still pass.
16
17So the removal is checked, not just done. The check reads the compile
18database CMake emits, which records the exact argv per translation unit, and
19holds every first-party source in it to NASA Power of 10 Rule 10:
20
21 * blanket ``-Werror`` is present, and
22 * blanket ``-w`` is absent.
23
24Vendored SOUP (``libs/third_party/``) and generated data (``libs/ra8_fonts/``) are
25exempt by construction -- they are not hand-authored here and CLAUDE.md scopes
26the style bar to first-party code.
27
28Precision
29---------
30The traps this deliberately avoids, each of which a substring test gets wrong:
31
32 * ``-Werror=return-type`` promotes ONE diagnostic and is not blanket
33 ``-Werror``; ``"-Werror" in command`` would wrongly accept it.
34 * ``-Wno-error`` later on the line cancels an earlier ``-Werror``.
35 * ``-Wall`` / ``-Wwrite-strings`` are not ``-w``; a prefix test would
36 wrongly reject them.
37
38Matching is therefore on exact argv arguments, in order, never on substrings.
39
40Second compiler arm (#356)
41--------------------------
42gcc-14 catches warning families clang-18 misses -- its ``-Wformat-truncation``
43found a silent PATH_MAX path-join truncation in ``tools/media_dl`` that clang
44did not flag. The tools-build gate therefore compiles the host tools under BOTH
45pinned compilers and passes both sets of databases here. ``--require-compilers
46clang,gcc`` makes a silently-dropped arm a hard failure rather than a vacuous
47pass (the #348/#355 class): each named family must drive at least one database.
48
49Usage
50-----
51 check_tool_warning_flags.py COMPILE_COMMANDS_JSON [...]
52 check_tool_warning_flags.py --require-compilers clang,gcc COMPILE_COMMANDS_JSON [...]
53 check_tool_warning_flags.py --selftest
54"""

64# Path fragments whose sources are not hand-authored under this project's
65# style rules. Kept identical in spirit to check_no_silent_stubs.py's EXCLUDED.
66EXEMPT_FRAGMENTS = ("libs/third_party/", "libs/ra8_fonts/")

69def is_exempt(source: str) -> bool:
70 """True when `source` is vendored SOUP or generated data."""
71 normalised = source.replace("\\", "/")
72 return any(fragment in normalised for fragment in EXEMPT_FRAGMENTS)

75def tokens_of(entry: dict) -> list[str]:
76 """Return the argv of one compile-database entry.
77
78 CMake emits either `arguments` (a list) or `command` (a single string)
79 depending on generator and version; both spellings are accepted.
80 """
81 if isinstance(entry.get("arguments"), list):
82 return [str(arg) for arg in entry["arguments"]]
83 return shlex.split(str(entry.get("command", "")))

86def compiler_of(argv: list[str]) -> str:
87 """Classify the compiler family driving one compile-database entry.
88
89 Returns ``"clang"`` or ``"gcc"`` for the two pinned host-tool arms, or
90 ``"other"`` for anything else. The first recognised driver token wins, so a
91 launcher prefix (``ccache gcc-14 ...``) still classifies as gcc.
92 """
93 for token in argv:
94 base = token.replace("\\", "/").rsplit("/", 1)[-1]
95 if "clang" in base:
96 return "clang"
97 if base in ("gcc", "g++") or base.startswith(("gcc-", "g++-")):
98 return "gcc"
99 return "other"

102def missing_required_compilers(seen: set[str], required: list[str]) -> list[str]:
103 """Return the required compiler families that no database exercised.
104
105 This is the #356 second-arm guard: the tools-build gate compiles the host
106 tools under clang-18 AND gcc-14 and passes both sets of databases here. If
107 the gcc arm is ever silently dropped, its family never appears in `seen`,
108 and a silently-dropped arm otherwise reads as a pass -- the #348/#355
109 failure class. ``--selftest`` asserts this fires when an arm is absent.
110 """
111 return sorted(set(required) - set(seen))

114def flag_problem(argv: list[str]) -> str | None:
115 """Return a problem description for `argv`, or None when it is compliant.
116
117 Blanket `-Werror` must be in force at the end of the line and blanket `-w`
118 must never appear. Both are exact-token tests: `-Werror=xxx` is a targeted
119 promotion rather than the blanket flag, and `-Wno-error` cancels a blanket
120 `-Werror` that precedes it.
121 """
122 werror = False
123 for arg in argv:
124 if arg == "-Werror":
125 werror = True
126 elif arg in ("-Wno-error", "-Wno-error=all"):
127 werror = False
128 elif arg == "-w":
129 return "compiled with -w (all diagnostics disabled)"
130 if not werror:
131 return "compiled without blanket -Werror"
132 return None

135def check_database(path: Path) -> tuple[list[str], set[str]]:
136 """Return (violation lines, compiler families seen) for one database.
137
138 The compiler set is the #356 second-arm evidence: every translation unit in
139 one database is compiled by the same driver, so the union across the clang
140 and gcc databases the gate passes must contain both families.
141 """
142 try:
143 entries = json.loads(path.read_text())
144 except (OSError, json.JSONDecodeError) as exc:
145 sys.stderr.write(f"check_tool_warning_flags.py: FATAL -- cannot read {path}: {exc}\n")
146 raise SystemExit(2) from exc
147
148 if not entries:
149 sys.stderr.write(
150 f"check_tool_warning_flags.py: FATAL -- {path} lists no translation\n"
151 " units. An empty compile database would let this gate report a\n"
152 " pass for a build that never happened.\n"
153 )
154 raise SystemExit(2)
155
156 violations: list[str] = []
157 compilers: set[str] = set()
158 for entry in entries:
159 argv = tokens_of(entry)
160 compilers.add(compiler_of(argv))
161 source = str(entry.get("file", ""))
162 if not source or is_exempt(source):
163 continue
164 problem = flag_problem(argv)
165 if problem is not None:
166 violations.append(f"{source}: {problem}")
167 return violations, compilers

231# Compiler-classification cases: compiler_of must name the driver family from
232# an argv the way the real databases spell it (absolute path, versioned name,
233# a launcher prefix). Asserted so the #356 second-arm guard cannot be defeated
234# by a classifier that quietly folds gcc into clang or "other".
235COMPILER_SELFTEST_CASES: list[tuple[str, list[str], str]] = [
236 ("clang-18 absolute path", ["/usr/bin/clang-18", "-c", "f.c"], "clang"),
237 ("clang++-18 driver", ["clang++-18", "-c", "f.cc"], "clang"),
238 ("gcc-14 absolute path", ["/usr/local/bin/gcc-14", "-c", "f.c"], "gcc"),
239 ("g++-14 driver", ["g++-14", "-c", "f.cc"], "gcc"),
240 ("bare gcc", ["gcc", "-c", "f.c"], "gcc"),
241 ("ccache-wrapped gcc-14", ["ccache", "gcc-14", "-c", "f.c"], "gcc"),
242 ("unknown driver", ["tcc", "-c", "f.c"], "other"),
243]
244
245# Second-arm coverage cases: with both arms required, a missing family must be
246# reported (fires) and a complete set must report nothing (quiet). This is the
247# property the #356 acceptance names -- a silently-dropped gcc arm reads as a
248# pass unless its absence is turned into a hard finding here.
249COVERAGE_SELFTEST_CASES: list[tuple[str, set[str], list[str], list[str]]] = [
250 ("only clang seen -> gcc missing", {"clang"}, ["clang", "gcc"], ["gcc"]),
251 ("only gcc seen -> clang missing", {"gcc"}, ["clang", "gcc"], ["clang"]),
252 ("both seen -> nothing missing", {"clang", "gcc"}, ["clang", "gcc"], []),
253]
254
255
256def selftest() -> int:
257 """Prove the detector fires and stays quiet where it must."""
258 failures: list[str] = []
259 for name, source, argv, should_fire in SELFTEST_CASES:
260 fired = (not is_exempt(source)) and (flag_problem(argv) is not None)
261 if fired != should_fire:
262 failures.append(
263 f" {name}: expected {'a violation' if should_fire else 'no violation'}, "
264 f"got {'a violation' if fired else 'none'}"
265 )
266
267 for name, argv, want in COMPILER_SELFTEST_CASES:
268 got = compiler_of(argv)
269 if got != want:
270 failures.append(f" compiler_of {name}: expected {want!r}, got {got!r}")
271
272 for name, seen, required, want_missing in COVERAGE_SELFTEST_CASES:
273 got_missing = missing_required_compilers(seen, required)
274 if got_missing != want_missing:
275 failures.append(
276 f" coverage {name}: expected missing {want_missing}, got {got_missing}"
277 )
278
279 if failures:
280 sys.stderr.write("check_tool_warning_flags.py --selftest: FAILED\n")
281 sys.stderr.write("\n".join(failures) + "\n")
282 return 1
283
284 fires = sum(1 for case in SELFTEST_CASES if case[3])
285 quiet = len(SELFTEST_CASES) - fires
286 print(
287 f"check_tool_warning_flags.py --selftest: PASS "
288 f"({fires} must-fire, {quiet} must-stay-quiet flag cases; "
289 f"{len(COMPILER_SELFTEST_CASES)} classifier, "
290 f"{len(COVERAGE_SELFTEST_CASES)} second-arm coverage cases)"
291 )
292 return 0

295def _scan_databases(paths: list[str]) -> tuple[list[str], set[str]]:
296 """Scan every database path; return (violations, compiler families seen).
297
298 Exits (SystemExit 2) when a path is missing, matching check_database's
299 fatal handling of an empty or unreadable database -- a gate must never
300 report a pass for a database it could not inspect.
301 """
302 violations: list[str] = []
303 seen: set[str] = set()
304 for name in paths:
305 path = Path(name)
306 if not path.is_file():
307 sys.stderr.write(
308 f"check_tool_warning_flags.py: FATAL -- {path} does not exist.\n"
309 " Configure the tool with -DCMAKE_EXPORT_COMPILE_COMMANDS=ON first.\n"
310 )
311 raise SystemExit(2)
312 db_violations, db_compilers = check_database(path)
313 violations += db_violations
314 seen |= db_compilers
315 return violations, seen

318def _report_violations(violations: list[str]) -> None:
319 """Print every first-party translation unit found below the warning bar."""
320 sys.stderr.write(
321 f"check_tool_warning_flags.py: {len(violations)} first-party "
322 "translation unit(s) below the project warning bar:\n\n"
323 )
324 for line in sorted(violations):
325 sys.stderr.write(f" {line}\n")
326 sys.stderr.write(
327 "\nFirst-party sources are held to -Wall -Wextra -Werror everywhere\n"
328 "else in this tree (NASA Power of 10 Rule 10). Only libs/third_party/\n"
329 "SOUP may be compiled -w. Fix the warning; do not re-suppress it.\n"
330 )

333def _report_missing_arm(missing: list[str], seen: set[str]) -> None:
334 """Print the silently-dropped compiler-arm failure (#356)."""
335 sys.stderr.write(
336 "check_tool_warning_flags.py: FATAL -- required compiler arm(s) "
337 f"never exercised: {', '.join(missing)}\n"
338 f" databases were compiled by: {', '.join(sorted(seen)) or '(none)'}\n"
339 " The tools-build gate compiles the host tools under clang-18 AND\n"

345def main() -> int:
346 """Entry point."""
347 parser = argparse.ArgumentParser(description=__doc__)
348 parser.add_argument("databases", nargs="*", help="compile_commands.json paths")
349 parser.add_argument(
350 "--selftest",
351 action="store_true",
352 help="prove the detector fires and stays quiet, then exit",
353 )
354 parser.add_argument(
355 "--require-compilers",
356 default="",
357 metavar="FAMILY[,FAMILY...]",
358 help=(
359 "comma-separated compiler families (e.g. clang,gcc) that must each "
360 "drive at least one database; the #356 second-arm guard. A single "
361 "comma-joined value keeps it order-independent of the database list."
362 ),
363 )
364 args = parser.parse_args()
365
366 if args.selftest:
367 return selftest()
368 if not args.databases:
369 sys.stderr.write(
370 "check_tool_warning_flags.py: FATAL -- no compile database given.\n"
371 " This gate must never run with nothing to inspect: that reports a\n"
372 " pass for work never done.\n"
373 )
374 return 2
375
376 required = [fam for fam in args.require_compilers.split(",") if fam]
377 violations, seen = _scan_databases(args.databases)
378 if violations:
379 _report_violations(violations)
380 return 1
381 missing = missing_required_compilers(seen, required)
382 if missing:
383 _report_missing_arm(missing, seen)
384 return 1
385
386 arms = f", arms: {', '.join(sorted(seen))}" if required else ""
387 print(
388 f"check_tool_warning_flags.py: OK ({len(args.databases)} compile "
389 f"database(s), every first-party TU at -Wall -Wextra -Werror{arms})"
390 )
391 return 0
