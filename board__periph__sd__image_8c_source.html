54bool board_sd_attach(const char* path)
55{
56 if (path == nullptr) {
57 return false;
58 }
59 FILE* fp = fopen(path, "rb");
60 if (fp == nullptr) {
61 (void)fprintf(stderr, "ra8_emulator: --sd: cannot open '%s'\n", path);
62 return false;
63 }
64 (void)fseek(fp, 0L, SEEK_END);
65 const long size = ftell(fp);
66 (void)fseek(fp, 0L, SEEK_SET);
67 if (size <= 0L) {
68 (void)fclose(fp);
69 (void)fprintf(stderr, "ra8_emulator: --sd: empty image '%s'\n", path);
70 return false;
71 }
72 board_sd_release_image();
73 s_sd = (board_sd_state_t){};
74 s_sd.map_fd = -1;
75 s_sd.image = (uint8_t*)malloc((size_t)size);
76 if (s_sd.image == nullptr) {
77 (void)fclose(fp);
78 return false;
79 }
80 const size_t got = fread(s_sd.image, 1U, (size_t)size, fp);
81 (void)fclose(fp);
82 if (got != (size_t)size) {
83 board_sd_release_image();
84 return false;
85 }
86 s_sd.image_len = (uint64_t)size;
87 s_sd.attached = true;
88 (void)fprintf(stderr, "ra8_emulator: SD card attached (%ld bytes) from %s\n", size, path);
89 return true;
90}

121static uint8_t* board_sd_map_blank_image(uint64_t bytes, int* out_fd)
122{
123 char tmpl[] = "/tmp/ra8_emulator_sd.XXXXXX";
124 int fd = mkstemp(tmpl);
125 if (fd < 0) {
126 (void)fprintf(stderr, "ra8_emulator: --sd-new: mkstemp failed\n");
127 return nullptr;
128 }
129 (void)unlink(tmpl); /* anonymous: the storage lives until the fd is closed. */
130 if (ftruncate(fd, (off_t)bytes) != 0) {
131 (void)close(fd);
132 (void)fprintf(stderr,
133 "ra8_emulator: --sd-new: ftruncate to %llu bytes failed\n",
134 (unsigned long long)bytes);
135 return nullptr;
136 }
137 uint8_t* img = (uint8_t*)mmap(nullptr, (size_t)bytes, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
138 if (img == MAP_FAILED) {
139 (void)close(fd);
140 (void)fprintf(stderr,
141 "ra8_emulator: --sd-new: mmap of %llu bytes failed\n",
142 (unsigned long long)bytes);
143 return nullptr;
144 }
145 *out_fd = fd;
146 return img;
147}

185bool board_sd_attach_blank(uint32_t total_sectors, uint8_t fat_bits, const char* label)
186{
187 if (total_sectors < (uint32_t)k_sd_min_sectors) {
188 (void)fprintf(stderr, "ra8_emulator: --sd-new: size too small (need >= 32 KiB)\n");
189 return false;
190 }
191 const uint64_t bytes = (uint64_t)total_sectors * (uint64_t)k_fmt_sec_bytes;
192 int fd = -1;
193 uint8_t* img = board_sd_map_blank_image(bytes, &fd);
194 if (img == nullptr) {
195 return false;
196 }
197 const uint32_t spc = (fat_bits == (uint8_t)k_fat32_bits)
198 ? sd_format_fat32(img, total_sectors, label)
199 : sd_format_fat16(img, total_sectors, label);
200 board_sd_release_image();
201 s_sd = (board_sd_state_t){};
202 s_sd.image = img;
203 s_sd.image_len = bytes;
204 s_sd.mmapped = true;
205 s_sd.map_fd = fd;
206 s_sd.attached = true;
207 s_sd.fat_bits =
208 (fat_bits == (uint8_t)k_fat32_bits) ? (uint8_t)k_fat32_bits : (uint8_t)k_fat16_bits;
209 sd_label_field((uint8_t*)s_sd.label, label);
210 s_sd.label[k_fmt_label_len] = '\0';
211 board_sd_report_created(bytes, s_sd.fat_bits, spc);
212 return true;
213}

215bool board_sd_save(const char* path)
216{
217 if ((path == nullptr) || !s_sd.attached || (s_sd.image == nullptr)) {
218 return false;
219 }
220 /* A sparse multi-GB card would dump GBs of mostly-zeros; cap the dump so
221 * --save-sd stays sane. Inspect a large card by its live mount instead. */
222 if (s_sd.image_len > (uint64_t)k_sd_save_max_bytes) {
223 (void)fprintf(
224 stderr,
225 "ra8_emulator: --save-sd: card is %llu MiB (> %u MiB cap) -- skipped\n",
226 (unsigned long long)(s_sd.image_len / ((uint64_t)k_unit_kib * (uint64_t)k_unit_kib)),
227 (unsigned)((uint64_t)k_sd_save_max_bytes / ((uint64_t)k_unit_kib * (uint64_t)k_unit_kib)));
228 return false;
229 }
230 FILE* fp = fopen(path, "wb");
231 if (fp == nullptr) {
232 (void)fprintf(stderr, "ra8_emulator: --save-sd: cannot write '%s'\n", path);
233 return false;
234 }
235 const size_t put = fwrite(s_sd.image, 1U, (size_t)s_sd.image_len, fp);
236 (void)fclose(fp);
237 if (put != (size_t)s_sd.image_len) {
238 return false;
239 }
240 (void)fprintf(stderr,
241 "ra8_emulator: SD card image saved (%llu bytes) to %s\n",
242 (unsigned long long)s_sd.image_len,
243 path);
244 return true;
245}
