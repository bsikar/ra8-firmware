82static bool viewer_parse_args(int argc, char** argv, viewer_opts_t* opts)
83{
84 if (argc < k_viewer_min_args) {
85 return false;
86 }
87 memset(opts, 0, sizeof(*opts));
88 opts->path = argv[1];
89 opts->dump_tile = -1;
90 for (int i = 2; i < argc; ++i) {
91 if ((strcmp(argv[i], "--page") == 0) && ((i + 1) < argc)) {
92 opts->page = (uint32_t)strtoul(argv[++i], nullptr, (int)k_viewer_radix_dec);
93 } else if ((strcmp(argv[i], "--dump-ppm") == 0) && ((i + 1) < argc)) {
94 opts->dump_ppm = argv[++i];
95 } else if ((strcmp(argv[i], "--dump-tile") == 0) && ((i + 1) < argc)) {
96 opts->dump_tile = (int64_t)strtoll(argv[++i], nullptr, (int)k_viewer_radix_dec);
97 } else if (strcmp(argv[i], "--headless") == 0) {
98 opts->headless = true;
99 } else {
100 return false;
101 }
102 }
103 return true;
104}

153static int viewer_run_window(ra8_viewer_reader_t* reader)
154{
155 ra8_viewer_view_t* view = ra8_viewer_view_open(reader, "RA8 Viewer");
156 if (view == nullptr) {
157 (void)fprintf(stderr, "no window (headless host?); use --headless --dump-ppm PATH\n");
158 return 1;
159 }
160 const struct timespec frame = {.tv_sec = 0, .tv_nsec = k_viewer_frame_ns};
161 while (!ra8_viewer_view_pump(view)) {
162 (void)nanosleep(&frame, nullptr);
163 }
164 ra8_viewer_view_close(view);
165 return 0;
166}

193static bool viewer_dump_tile(ra8_viewer_reader_t* reader, uint32_t tile, const char* path)
194{
195 uint32_t w = 0U;
196 uint32_t h = 0U;
197 uint16_t* buf = nullptr;
198 const ra8_err_t rc = ra8_viewer_render_tile565(reader, tile, &w, &h, &buf);
199 if (rc != k_ra8_ok) {
200 (void)fprintf(stderr, "render tile %u failed: 0x%x\n", tile, (unsigned)rc);
201 return false;
202 }
203 const ra8_err_t wrc = ra8_viewer_write_ppm565(buf, w, h, path);
204 free(buf);
205 if (wrc != k_ra8_ok) {
206 (void)fprintf(stderr, "write tile ppm failed: 0x%x\n", (unsigned)wrc);
207 return false;
208 }
209 (void)fprintf(stderr, "wrote tile %u (%ux%u) -> %s\n", tile, w, h, path);
210 return true;
211}

223int main(int argc, char** argv)
224{
225 ra8_log_set_byte_sink(viewer_log_sink, nullptr);
226
227 viewer_opts_t opts = {};
228 if (!viewer_parse_args(argc, argv, &opts)) {
229 viewer_usage(argv[0]);
230 return 2;
231 }
232
233 ra8_viewer_reader_t* reader = nullptr;
234 const ra8_err_t rc = ra8_viewer_open(&reader, opts.path);
235 if (rc != k_ra8_ok) {
236 (void)fprintf(stderr, "open '%s' failed: 0x%x\n", opts.path, (unsigned)rc);
237 return 1;
238 }
239 const uint32_t count = ra8_viewer_page_count(reader);
240 const uint32_t page = viewer_clamp_page(opts.page, count);
241 (void)fprintf(stderr, "opened '%s': %u page(s)\n", opts.path, count);
242
243 /* Headless tile dump: verify the scroll render path independent of a window. */
244 if (opts.dump_tile >= 0) {
245 const char* out = (opts.dump_ppm != nullptr) ? opts.dump_ppm : "/tmp/ra8_tile.ppm";
246 const int status = viewer_dump_tile(reader, (uint32_t)opts.dump_tile, out) ? 0 : 1;
247 ra8_viewer_close(reader);
248 return status;
249 }
250
251 int status = 0;
252 if (opts.headless || (opts.dump_ppm != nullptr)) {
253 status = viewer_render_and_dump(reader, &opts, page) ? 0 : 1;
254 } else {
255 status = viewer_run_window(reader);
256 }
257 ra8_viewer_close(reader);
258 return status;
259}
