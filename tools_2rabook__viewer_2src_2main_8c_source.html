85RA8_INTERNAL static bool internal_parse_arguments(int argc, char** argv, viewer_options_t* options)
86{
87 if (argc < (int)k_viewer_min_args) {
88 return false;
89 }
90 *options = (viewer_options_t){.path = argv[1], .dump_tile = -1};
91 for (int index = 2; index < argc; ++index) {
92 if ((strcmp(argv[index], "--page") == 0) && ((index + 1) < argc)) {
93 options->page = (uint32_t)strtoul(argv[++index], nullptr, (int)k_viewer_radix_dec);
94 } else if ((strcmp(argv[index], "--dump-ppm") == 0) && ((index + 1) < argc)) {
95 options->dump_ppm = argv[++index];
96 } else if ((strcmp(argv[index], "--dump-tile") == 0) && ((index + 1) < argc)) {
97 options->dump_tile = strtoll(argv[++index], nullptr, (int)k_viewer_radix_dec);
98 } else if (strcmp(argv[index], "--headless") == 0) {
99 options->headless = true;
100 } else {
101 return false;
102 }
103 }
104 return true;
105}

146RA8_INTERNAL static bool internal_align_offset(size_t offset, size_t alignment, size_t* out)
147{
148 const size_t mask = alignment - 1U;
149 if ((alignment == 0U) || ((alignment & mask) != 0U) || (offset > (SIZE_MAX - mask))) {
150 return false;
151 }
152 *out = (offset + mask) & ~mask;
153 return true;
154}

288RA8_INTERNAL static int internal_run_window(ra8_viewer_reader_t* reader,
289 void* workspace,
290 size_t workspace_bytes,
291 ra8_io_stream_t* diagnostic)
292{
293 ra8_viewer_view_requirements_t requirements = {};
294 if (ra8_viewer_view_requirements(reader, &requirements) != k_ra8_ok) {
295 (void)priv_viewer_output_text(diagnostic,
296 "no window (headless host?); use --headless --dump-ppm PATH\n");
297 return 1;
298 }
299 ra8_viewer_view_t* view = nullptr;
300 ra8_viewer_workspace_report_t report = {};
301 const ra8_err_t error = ra8_viewer_view_open(&view,
302 reader,
303 "RA8 Viewer",
304 workspace,
305 workspace_bytes,
306 &requirements,
307 &report);
308 if (error != k_ra8_ok) {
309 if (error == k_ra8_err_invalid_size) {
310 internal_report_capacity(diagnostic, "view", &report);
311 }
312 (void)priv_viewer_output_text(diagnostic,
313 "no window (headless host?); use --headless --dump-ppm PATH\n");
314 return 1;
315 }
316 const struct timespec frame = {.tv_sec = 0, .tv_nsec = k_viewer_frame_ns};
317 while (!ra8_viewer_view_pump(view)) {
318 (void)nanosleep(&frame, nullptr);
319 }
320 ra8_viewer_view_close(view);
321 return 0;
322}

359 ra8_io_stream_posix_state_t* state)

361 struct sigaction action = {.sa_handler = SIG_IGN};

387RA8_INTERNAL static int internal_execute(ra8_viewer_reader_t* reader,
388 const ra8_viewer_reader_requirements_t* requirements,
389 const viewer_options_t* options,
390 uint32_t page,
391 ra8_io_stream_t* diagnostic)
392{
393 size_t scratch_offset = 0U;
394 if (!internal_align_offset(requirements->required_bytes, alignof(max_align_t), &scratch_offset) ||
395 (scratch_offset > sizeof(s_viewer_composition))) {
396 return 1;
397 }
398 void* scratch = &s_viewer_composition[scratch_offset];
399 const size_t scratch_bytes = sizeof(s_viewer_composition) - scratch_offset;
400 if (options->dump_tile >= 0) {
401 const char* output = (options->dump_ppm != nullptr) ? options->dump_ppm : "/tmp/ra8_tile.ppm";
402 return internal_dump_tile(reader,
403 (uint32_t)options->dump_tile,
404 output,
405 scratch,
406 scratch_bytes,
407 diagnostic)
408 ? 0
409 : 1;
410 }
411 if (options->headless || (options->dump_ppm != nullptr)) {
412 return internal_render_page(reader, options, page, diagnostic) ? 0 : 1;
413 }
414 return internal_run_window(reader, scratch, scratch_bytes, diagnostic);
415}

418int main(int argc, char** argv)
419{
420 ra8_io_stream_t diagnostic = {};
421 ra8_io_stream_posix_state_t diagnostic_state = {};
422 if (internal_bind_diagnostic(&diagnostic, &diagnostic_state) != k_ra8_ok) {
423 return 1;
424 }
425 ra8_log_set_byte_sink(internal_log_sink, &diagnostic);
426 viewer_options_t options = {};
427 if (!internal_parse_arguments(argc, argv, &options)) {
428 internal_usage(&diagnostic, argv[0]);
429 return 2;
430 }
431 ra8_viewer_reader_requirements_t requirements = {};
432 ra8_err_t error = ra8_viewer_reader_requirements(options.path, &requirements);
433 if (error != k_ra8_ok) {
434 (void)priv_viewer_output_open_error(&diagnostic, options.path, error);
435 return 1;
436 }
437 ra8_viewer_reader_t* reader = nullptr;
438 ra8_viewer_workspace_report_t report = {};
439 error = ra8_viewer_reader_bind(&reader,
440 s_viewer_composition,
441 sizeof(s_viewer_composition),
442 &requirements,
443 &report);
444 if (error != k_ra8_ok) {
445 internal_report_capacity(&diagnostic, "reader", &report);
446 (void)priv_viewer_output_open_error(&diagnostic, options.path, error);
447 return 1;
448 }
449 error = ra8_viewer_open(reader, options.path);
450 if (error != k_ra8_ok) {
451 (void)priv_viewer_output_open_error(&diagnostic, options.path, error);
452 return 1;
453 }
454 const uint32_t count = ra8_viewer_page_count(reader);
455 const uint32_t page = internal_clamp_page(options.page, count);
456 (void)priv_viewer_output_opened(&diagnostic, options.path, count);
457 const int status = internal_execute(reader, &requirements, &options, page, &diagnostic);
458 ra8_viewer_close(reader);
