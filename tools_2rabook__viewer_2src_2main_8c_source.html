101static bool viewer_parse_args(int argc, char** argv, viewer_opts_t* opts)
102{
103 if (argc < k_viewer_min_args) {
104 return false;
105 }
106 memset(opts, 0, sizeof(*opts));
107 opts->path = argv[1];
108 opts->dump_tile = -1;
109 for (int i = 2; i < argc; ++i) {
110 if ((strcmp(argv[i], "--page") == 0) && ((i + 1) < argc)) {
111 opts->page = (uint32_t)strtoul(argv[++i], nullptr, (int)k_viewer_radix_dec);
112 } else if ((strcmp(argv[i], "--dump-ppm") == 0) && ((i + 1) < argc)) {
113 opts->dump_ppm = argv[++i];
114 } else if ((strcmp(argv[i], "--dump-tile") == 0) && ((i + 1) < argc)) {
115 opts->dump_tile = (int64_t)strtoll(argv[++i], nullptr, (int)k_viewer_radix_dec);
116 } else if (strcmp(argv[i], "--headless") == 0) {
117 opts->headless = true;
118 } else {
119 return false;
120 }
121 }
122 return true;
123}

202static int viewer_run_window(ra8_viewer_reader_t* reader)
203{
204 ra8_viewer_view_t* view = ra8_viewer_view_open(reader, "RA8 Viewer");
205 if (view == nullptr) {
206 (void)fprintf(stderr, "no window (headless host?); use --headless --dump-ppm PATH\n");
207 return 1;
208 }
209 const struct timespec frame = {.tv_sec = 0, .tv_nsec = k_viewer_frame_ns};
210 while (!ra8_viewer_view_pump(view)) {
211 (void)nanosleep(&frame, nullptr);
212 }
213 ra8_viewer_view_close(view);
214 return 0;
215}

254static bool viewer_dump_tile(ra8_viewer_reader_t* reader, uint32_t tile, const char* path)
255{
256 uint32_t w = 0U;
257 uint32_t h = 0U;
258 uint16_t* buf = nullptr;
259 const ra8_err_t rc = ra8_viewer_render_tile565(reader, tile, &w, &h, &buf);
260 if (rc != k_ra8_ok) {
261 (void)fprintf(stderr, "render tile %u failed: 0x%x\n", tile, (unsigned)rc);
262 return false;
263 }
264 const ra8_err_t wrc = ra8_viewer_write_ppm565(buf, w, h, path);
265 free(buf);
266 if (wrc != k_ra8_ok) {
267 (void)fprintf(stderr, "write tile ppm failed: 0x%x\n", (unsigned)wrc);
268 return false;
269 }
270 (void)fprintf(stderr, "wrote tile %u (%ux%u) -> %s\n", tile, w, h, path);
271 return true;
272}

284int main(int argc, char** argv)
285{
286 ra8_log_set_byte_sink(viewer_log_sink, nullptr);
287
288 viewer_opts_t opts = {};
289 if (!viewer_parse_args(argc, argv, &opts)) {
290 viewer_usage(argv[0]);
291 return 2;
292 }
293
294 ra8_viewer_reader_t* reader = nullptr;
295 const ra8_err_t rc = ra8_viewer_open(&reader, opts.path);
296 if (rc != k_ra8_ok) {
297 (void)fprintf(stderr, "open '%s' failed: 0x%x\n", opts.path, (unsigned)rc);
298 return 1;
299 }
300 const uint32_t count = ra8_viewer_page_count(reader);
301 const uint32_t page = viewer_clamp_page(opts.page, count);
302 (void)fprintf(stderr, "opened '%s': %u page(s)\n", opts.path, count);
303
304 /* Headless tile dump: verify the scroll render path independent of a window. */
305 if (opts.dump_tile >= 0) {
306 const char* out = (opts.dump_ppm != nullptr) ? opts.dump_ppm : "/tmp/ra8_tile.ppm";
307 const int status = viewer_dump_tile(reader, (uint32_t)opts.dump_tile, out) ? 0 : 1;
308 ra8_viewer_close(reader);
309 return status;
310 }
311
312 int status = 0;
313 if (opts.headless || (opts.dump_ppm != nullptr)) {
314 status = viewer_render_and_dump(reader, &opts, page) ? 0 : 1;
315 } else {
316 status = viewer_run_window(reader);
317 }
318 ra8_viewer_close(reader);
319 return status;
320}
