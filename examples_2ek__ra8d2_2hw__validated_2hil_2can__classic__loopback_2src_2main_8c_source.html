213[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_can_demo_one_round_trip(uint8_t seq)
214{
215 ra8_canfd_frame_t tx = {
216 .id = (uint32_t)k_can_demo_id,
217 .dlc = (uint8_t)k_can_demo_dlc,
218 .is_extended = 0U,
219 .is_fd = 0U, /* classic */
220 .is_brs = 0U,
221 .data = {seq,
222 (uint8_t)k_can_demo_byte_marker_a,
223 (uint8_t)k_can_demo_byte_marker_b,
224 (uint8_t)k_can_demo_byte_marker_c,
225 (uint8_t)k_can_demo_byte_marker_d,
226 (uint8_t)k_can_demo_byte_marker_e,
227 (uint8_t)k_can_demo_byte_marker_f,
228 (uint8_t)k_can_demo_byte_marker_g},
229 };
230 if (ra8_canfd_transmit((uint8_t)k_can_demo_channel, &tx) != k_ra8_ok) {
231 return k_ra8_err_hw_error;
232 }
233 ra8_canfd_frame_t rx = {};
234 if (ra8_canfd_receive((uint8_t)k_can_demo_channel, &rx) != k_ra8_ok) {
235 return k_ra8_err_no_data;
236 }
237 return k_ra8_ok;
238}
