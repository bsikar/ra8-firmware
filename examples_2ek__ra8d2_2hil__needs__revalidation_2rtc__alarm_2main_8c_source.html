72static void rtc_demo_setup_or_halt(void)
73{
74 uint32_t cpuclk0_hz = 0U;
75 if (ra8_cgc_init() != k_ra8_ok) {
76 rtc_demo_panic_halt();
77 }
78 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
79 rtc_demo_panic_halt();
80 }
81 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
82 rtc_demo_panic_halt();
83 }
84 if (ra8_board_uart_console_init((uint32_t)k_rtc_demo_baud) != k_ra8_ok) {
85 rtc_demo_panic_halt();
86 }
87 /* Enable global interrupts before the RTC count-source bring-up: that
88 * step blocks on ra8_delay_ms for the sub-clock stabilization time, and
89 * with IRQs unmasked ra8_delay_ms advances off the SysTick tick counter
90 * (the demo registers no RTC NVIC handler -- it polls the status flag). */
91 ra8_isr_globals_enable();
92 /* Bring up and select the RTC count source BEFORE ra8_rtc_init(), or the
93 * counter never advances and the +5 s alarm never fires. Primary source is
94 * the EK-RA8D2 32.768 kHz sub-clock crystal (accurate). On a board without
95 * the crystal, swap the argument to k_ra8_rtc_clk_loco (internal LOCO,
96 * crystal-free) and re-flash. */
97 if (ra8_rtc_clock_init(k_ra8_rtc_clk_subclock) != k_ra8_ok) {
98 rtc_demo_panic_halt();
99 }
100 if (ra8_rtc_init() != k_ra8_ok) {
101 rtc_demo_panic_halt();
102 }
103 const ra8_rtc_datetime_t seed = {
104 .year = (uint16_t)(k_rtc_demo_year_base + k_rtc_demo_seed_year_lo),
105 .month = (uint8_t)k_rtc_demo_seed_month,
106 .day = (uint8_t)k_rtc_demo_seed_day,
107 .weekday = 0U,
108 .hour = 0U,
109 .minute = 0U,
110 .second = 0U,
111 };
112 if (ra8_rtc_set(&seed) != k_ra8_ok) {
113 rtc_demo_panic_halt();
114 }
115}

128[[nodiscard]] static ra8_err_t rtc_demo_arm_alarm(const ra8_rtc_datetime_t* now)
129{
130 ra8_rtc_datetime_t a = *now;
131 uint16_t s = (uint16_t)a.second + (uint16_t)k_rtc_demo_alarm_offset_s;
132 a.second = (uint8_t)(s % (uint16_t)k_rtc_demo_secs_per_min);
133 uint16_t add_min = (uint16_t)(s / (uint16_t)k_rtc_demo_secs_per_min);
134 uint16_t m = (uint16_t)a.minute + add_min;
135 a.minute = (uint8_t)(m % (uint16_t)k_rtc_demo_mins_per_hour);
136 uint16_t add_hr = (uint16_t)(m / (uint16_t)k_rtc_demo_mins_per_hour);
137 uint16_t h = (uint16_t)a.hour + add_hr;
138 a.hour = (uint8_t)(h % (uint16_t)k_rtc_demo_hours_per_day);
139 ra8_err_t err = ra8_rtc_set_alarm(&a);
140 if (err != k_ra8_ok) {
141 return err;
142 }
143 return ra8_rtc_set_irq_enable((uint8_t)k_ra8_rtc_irq_alarm);
144}

148int32_t main(void)
149{
150 rtc_demo_setup_or_halt();
151
152 /* Boot banner -- emit before the RTC poll loop so the HIL host can
153 * confirm the firmware booted even when the sub-clock crystal is not
154 * running and the alarm-fired path never reaches its own write.
155 * (Global IRQs were already enabled in rtc_demo_setup_or_halt.) */
156 (void)ra8_board_uart_console_write(k_rtc_demo_boot_msg,
157 (size_t)(sizeof(k_rtc_demo_boot_msg) - 1U));
158
159 while (1) {
160 ra8_rtc_datetime_t now = {};
161 if (ra8_rtc_get(&now) != k_ra8_ok) {
162 break;
163 }
164 if (rtc_demo_arm_alarm(&now) != k_ra8_ok) {
165 break;
166 }
167
168 /* Poll the alarm flag -- on real silicon RCR1.AIE drives an NVIC
169 * line, but for this demo we just spin against the status bit. */
170 uint8_t status = 0U;
171 do {
172 ra8_delay_ms(k_rtc_demo_poll_ms);
173 if (ra8_rtc_get_status(&status) != k_ra8_ok) {
174 rtc_demo_panic_halt();
175 }
176 } while ((status & (uint8_t)k_ra8_rtc_irq_alarm) == 0U);
177
178 if (ra8_board_uart_console_write(k_rtc_demo_log_msg,
179 (size_t)(sizeof(k_rtc_demo_log_msg) - 1U)) != k_ra8_ok) {
180 break;
181 }
182 if (ra8_rtc_clear_status((uint8_t)k_ra8_rtc_irq_alarm) != k_ra8_ok) {
183 break;
184 }
185 ra8_delay_ms((uint32_t)k_rtc_demo_advance_secs * (uint32_t)k_rtc_demo_ms_per_sec);
186 }
187 rtc_demo_panic_halt();
188 return 0;
189}
