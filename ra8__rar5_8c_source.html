112static uint32_t s_slot_to_length(ra8_rar5_state_t* st, uint32_t slot)
113{
114 uint32_t length = 2U;
115 if (slot < 8U) {
116 return length + slot;
117 }
118 const uint32_t lbits = (slot / 4U) - 1U;
119 length += (4U | (slot & 3U)) << lbits;
120 length += ra8_rar5_get(st, lbits);
121 return length;

140static uint64_t s_decode_distance(ra8_rar5_state_t* st)
141{
142 const uint32_t slot = ra8_rar5_decode_num(st, &st->dd);
143 if (slot < 4U) {
144 return (uint64_t)1U + slot;
145 }
146 const uint32_t dbits = (slot / 2U) - 1U;
147 uint64_t dist = (uint64_t)1U + ((uint64_t)(2U | (slot & 1U)) << dbits);
148 if (dbits < 4U) {
149 return dist + ra8_rar5_get(st, dbits);
150 }
151 if (dbits > 4U) {
152 dist += (uint64_t)ra8_rar5_get(st, dbits - 4U) << 4U;
153 }
154 dist += ra8_rar5_decode_num(st, &st->ldd);
155 return dist;
156}

213{
214 size_t pos = *out_pos;

216 return false;

270static ra8_err_t s_read_filter(ra8_rar5_state_t* st, size_t out_pos)
271{
272 if (st->filter_count >= (uint16_t)k_ra8_rar5_max_filters) {
273 return k_ra8_err_validation_failed;
274 }
275 const uint32_t rel = s_read_filter_data(st);
276 const uint32_t len = s_read_filter_data(st);
277 const uint32_t type = ra8_rar5_get(st, (uint32_t)k_r5_ftype_bits);
278 if (type > (uint32_t)k_ra8_rar5_filter_arm) {
279 return k_ra8_err_validation_failed;

284 }

286 f->start = (uint64_t)out_pos + (uint64_t)rel;
287 f->len = len;
288 f->type = (uint8_t)type;
289 f->channels = (uint8_t)channels;
290 st->filter_count += 1U;

295void ra8_rar5_filter_delta(ra8_rar5_state_t* st, uint8_t* d, uint32_t len, uint32_t channels)
296{
297 if ((len > (uint32_t)k_ra8_rar5_delta_scratch) || (channels == 0U)) {
298 return;

301 uint32_t dpos = 0U;

306 dpos += 1U;

353static void s_filter_x86(uint8_t* d, uint32_t len, uint64_t filepos, bool e9)
354{
355 if (len < (uint32_t)k_r5_x86_ilen) {
356 return;
357 }
358 const uint32_t limit = len - (uint32_t)k_r5_x86_ilen;
359 uint32_t i = 0U;
360 while (i <= limit) { /* bound: i <= limit < len */
361 if (s_x86_is_op(d[i], e9)) {
362 const uint32_t off = i + 1U;
363 const uint32_t pos = (uint32_t)(filepos + (uint64_t)off);
364 s_wr_le32(&d[off], s_rd_le32(&d[off]) - pos);
365 i += (uint32_t)k_r5_x86_ilen;
366 } else {
367 i += 1U;

387RA8_INTERNAL
388static void s_filter_arm(uint8_t* d, uint32_t len, uint64_t filepos)
389{
390 if (len < 4U) {
391 return;
392 }
393 const uint32_t limit = len - 4U;
394 uint32_t i = 0U;
395 while (i <= limit) { /* bound: i steps by 4 to limit */
396 if (d[i + 3U] == (uint8_t)k_r5_arm_bl) {
397 uint32_t v = (uint32_t)d[i] | ((uint32_t)d[i + 1U] << k_r5_byte_bits) |
398 ((uint32_t)d[i + 2U] << (k_r5_byte_bits * 2U));
399 const uint32_t p = (uint32_t)((filepos + (uint64_t)i) >> 2U);
400 v = (v - p) & (uint32_t)k_r5_arm_off_mask;
401 d[i] = (uint8_t)(v & (uint32_t)k_r5_byte_mask);
402 d[i + 1U] = (uint8_t)((v >> k_r5_byte_bits) & (uint32_t)k_r5_byte_mask);
403 d[i + 2U] = (uint8_t)((v >> (k_r5_byte_bits * 2U)) & (uint32_t)k_r5_byte_mask);
404 }
405 i += 4U;
406 }
407}

427static void
428s_apply_one_filter(ra8_rar5_state_t* st, uint8_t* out, size_t unp, const ra8_rar5_filter_t* f)
429{
430 if (f->start >= (uint64_t)unp) {
431 return;
432 }
433 const uint64_t avail = (uint64_t)unp - f->start;
434 const uint32_t len = ((uint64_t)f->len > avail) ? (uint32_t)avail : f->len;
435 if (len == 0U) {
436 return;
437 }
438 uint8_t* d = &out[(size_t)f->start];
439 if (f->type == (uint8_t)k_ra8_rar5_filter_delta) {
440 ra8_rar5_filter_delta(st, d, len, f->channels);
441 } else if (f->type == (uint8_t)k_ra8_rar5_filter_e8) {
442 s_filter_x86(d, len, f->start, false);
443 } else if (f->type == (uint8_t)k_ra8_rar5_filter_e8e9) {
444 s_filter_x86(d, len, f->start, true);
445 } else {
446 s_filter_arm(d, len, f->start);
447 }
448}

497s_do_match(ra8_rar5_state_t* st, uint32_t slot, uint8_t* out, size_t* out_pos, size_t unp)
498{
499 uint32_t length = s_slot_to_length(st, slot - (uint32_t)k_r5_sym_lenbase);
500 const uint64_t dist = s_decode_distance(st);
501 length = s_adjust_length(length, dist);

503 st->last_length = length;
504 if (!ra8_rar5_copy_match(out, out_pos, unp, length, dist)) {
505 return k_ra8_err_validation_failed;

597RA8_INTERNAL
598static ra8_err_t s_decode_token(ra8_rar5_state_t* st, uint8_t* out, size_t* out_pos, size_t unp)
599{
600 const uint32_t slot = ra8_rar5_decode_num(st, &st->ld);
601 if (slot < (uint32_t)k_r5_sym_filter) {
602 out[*out_pos] = (uint8_t)slot;
603 *out_pos += 1U;
604 return k_ra8_ok;
605 }
606 if (slot == (uint32_t)k_r5_sym_filter) {
607 return s_read_filter(st, *out_pos);
608 }
609 if (slot == (uint32_t)k_r5_sym_replast) {
610 return s_do_replast(st, out, out_pos, unp);
611 }
612 if (slot < (uint32_t)k_r5_sym_lenbase) {
613 return s_do_repdist(st, slot, out, out_pos, unp);
614 }
615 return s_do_match(st, slot, out, out_pos, unp);
616}

683RA8_INTERNAL
684static ra8_err_t s_decode_stream(ra8_rar5_state_t* st, uint8_t* out, size_t out_cap, size_t unp)
685{
686 (void)out_cap;
687 size_t out_pos = 0U;
688 bool last = false;
689 uint64_t block_end = 0U;
690 const uint64_t cap_bits = (st->packlen * (uint64_t)k_r5_byte_bits) + (uint64_t)k_r5_max_pad_bits;
691 ra8_err_t e = s_open_block(st, &block_end, &last);
692 if (e != k_ra8_ok) {
693 return e;
694 }
695 while ((out_pos < unp) && (st->consumed <= cap_bits)) { /* bound: consumed strictly rises */
696 if (st->consumed >= block_end) {
697 if (last) {
698 break;
699 }
700 e = s_open_block(st, &block_end, &last);
701 if (e != k_ra8_ok) {
702 return e;
703 }
704 continue;
705 }
706 e = s_decode_token(st, out, &out_pos, unp);
707 if (e != k_ra8_ok) {
708 return e;
709 }
710 }
711 if (out_pos != unp) {
712 return k_ra8_err_validation_failed;
713 }
714 return k_ra8_ok;
715}
