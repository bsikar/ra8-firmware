4"""Gate: ``tests/README.md`` documents exactly the subdirectories that exist.
5
6The README explains what each subdirectory of ``tests/`` is for. A plain prose
7README rots the moment someone adds ``tests/newthing/`` and forgets to describe
8it, or deletes ``tests/oldthing/`` and leaves a paragraph describing a directory
9that is gone -- and nothing notices. This gate makes that impossible in both
10directions:
11
121. **Every immediate subdirectory of ``tests/`` is described.** A new one that
13 the README does not name fails the gate, so it cannot be added silently.
14
152. **Every subdirectory the README names still exists.** A row describing a
16 directory that has been removed or renamed fails the gate, so a stale entry
17 cannot linger.
18
19The README is machine-read the same way it is human-read: only the FIRST cell
20of each Markdown table row counts, written as a code span with a trailing slash
21(```bench/```). Keying on the first column -- never the prose, never a
22description cell -- is what lets a description mention ``epub/real/`` without the
23checker mistaking ``epub`` for a top-level subdirectory of ``tests/``.
24
25Like every other detector in this tree it carries a **non-vacuity floor**: the
26real ``tests/`` has eight subdirectories, so a scan that finds almost none did
27not walk the tree it meant to, and reporting "no drift" against nothing would be
28the exact silent-pass failure this gate exists to prevent.
29
30``--selftest`` runs first in the gate. It builds throwaway ``tests/`` trees and
31asserts an undocumented subdirectory fires, a stale README entry fires, an
32in-sync tree stays quiet, and a collapsed scan is caught by the floor. Without
33it, "0 problems" is indistinguishable from "checked nothing".
34"""

49# A documented subdirectory is the first cell of a table row, written as a code
50# span ending in a slash: `name/`. Reading only the first column is deliberate
51# -- a name that appears in a row's description or in prose is never counted.
52ROW_NAME_RE = re.compile(r"^`([A-Za-z0-9._-]+)/`$")
53
54# The real tree carries eight subdirectories. A scan that finds far fewer walked
55# the wrong root or a tree that never checked out; refuse to certify it clean.
56MIN_SUBDIRS = 5
57
58EXIT_OK = 0
59EXIT_DRIFT = 1
60EXIT_VACUOUS = 2

63def immediate_subdirs(tests_dir: Path) -> set[str]:
64 """Return the names of the directories directly under `tests_dir`.
65
66 Args:
67 tests_dir: The ``tests/`` directory to scan.
68
69 Returns:
70 Every immediate child directory name, excluding dot-directories, which
71 are tooling rather than test content.
72 """
73 return {p.name for p in tests_dir.iterdir() if p.is_dir() and not p.name.startswith(".")}

76def documented_subdirs(readme_text: str) -> set[str]:
77 """Return the subdirectory names the README's table claims to describe.
78
79 Only the first cell of each table row is read, so a name appearing in a
80 row's description or in surrounding prose is never mistaken for a documented
81 top-level subdirectory.
82
83 Args:
84 readme_text: The full contents of ``tests/README.md``.
85
86 Returns:
87 Every ``name`` written as ```name/``` in a table's first column.
88 """
89 names: set[str] = set()
90 for line in readme_text.splitlines():
91 stripped = line.strip()
92 if not stripped.startswith("|"):
93 continue
94 first_cell = stripped.strip("|").split("|", 1)[0].strip()
95 match = ROW_NAME_RE.match(first_cell)
96 if match:
97 names.add(match.group(1))
98 return names

101def drift_problems(actual: set[str], documented: set[str]) -> list[str]:
102 """Return every way the tree and the README disagree.
103
104 Args:
105 actual: Subdirectory names that exist under ``tests/``.
106 documented: Subdirectory names the README describes.
107
108 Returns:
109 One message per undocumented subdirectory and per documented-but-absent
110 entry; empty when the two sets are equal.
111 """
112 problems = [
113 f"tests/{name}/ exists but is not documented in tests/README.md -- "
114 "add a table row whose first cell is `" + name + "/`"
115 for name in sorted(actual - documented)
116 ]
117 problems += [
118 f"tests/README.md documents tests/{name}/ but no such subdirectory "
119 "exists -- remove or rename that row"
120 for name in sorted(documented - actual)
121 ]
122 return problems

125def evaluate(
126 tests_dir: Path, readme: Path, min_subdirs: int = MIN_SUBDIRS
127) -> tuple[int, list[str]]:
128 """Compare a ``tests/`` tree against its README.
129
130 Args:
131 tests_dir: The ``tests/`` directory to scan.
132 readme: The README that must describe every subdirectory.
133 min_subdirs: Non-vacuity floor; a scan below it is treated as collapsed.
134
135 Returns:
136 An ``(exit_code, messages)`` pair: ``EXIT_VACUOUS`` when fewer than
137 ``min_subdirs`` directories were found, ``EXIT_DRIFT`` on any
138 disagreement, ``EXIT_OK`` when the README matches the tree.
139 """
140 actual = immediate_subdirs(tests_dir)
141 if len(actual) < min_subdirs:
142 return EXIT_VACUOUS, [
143 f"only {len(actual)} subdirectory(ies) found under {tests_dir} "
144 f"(floor is {min_subdirs}); the scan collapsed rather than the tree"
145 ]
146 readme_text = readme.read_text(encoding="utf-8") if readme.exists() else ""
147 problems = drift_problems(actual, documented_subdirs(readme_text))
148 return (EXIT_DRIFT if problems else EXIT_OK), problems

151@dataclass(frozen=True)
152class _Case:
153 """One selftest fixture and the verdict it must produce."""
154
155 name: str
156 subdirs: list[str]
157 documented: list[str]
158 floor: int
159 want_code: int
160 needle: str | None

163def _selftest_cases() -> list[_Case]:
164 """Return the fixtures the selftest asserts, one per direction.
165
166 Returns:
167 A case for the in-sync tree, the undocumented-subdir direction, the
168 stale-entry direction, and the non-vacuity floor. ``needle`` is a
169 substring one message must contain, or ``None`` when the case must
170 produce no messages.
171 """
172 trio = ["alpha", "beta", "gamma"]
173 return [
174 _Case("in-sync stays quiet", trio, trio, 3, EXIT_OK, None),
175 _Case("undocumented subdir fires", trio, ["alpha", "beta"], 3, EXIT_DRIFT, "tests/gamma/"),
176 _Case("stale doc entry fires", trio, [*trio, "ghost"], 3, EXIT_DRIFT, "ghost"),
177 _Case("collapsed scan is vacuous", ["alpha"], ["alpha"], 3, EXIT_VACUOUS, None),
178 ]

181def _write_fixture(root: Path, case: _Case) -> tuple[Path, Path]:
182 """Build a throwaway ``tests/`` tree and README for one selftest case.
183
184 Args:
185 root: Temporary directory to build inside.
186 case: The fixture to materialise.
187
188 Returns:
189 The ``(tests_dir, readme_path)`` pair to hand to :func:`evaluate`.
190 """
191 tests_dir = root / "tests"
192 tests_dir.mkdir()
193 for name in case.subdirs:
194 (tests_dir / name).mkdir()
195 rows = "".join(f"| `{name}/` | fixture description |\n" for name in case.documented)
196 readme = tests_dir / "README.md"
197 readme.write_text(
198 "# tests/\n\n| Subdirectory | What it holds |\n|---|---|\n" + rows,
199 encoding="utf-8",
200 )
201 return tests_dir, readme

204def _run_case(root: Path, case: _Case) -> list[str]:
205 """Run one selftest case and return the ways it misbehaved.
206
207 Args:
208 root: Fresh temporary directory for this case.
209 case: The fixture and its expected verdict.
210
211 Returns:
212 A message per assertion the case failed; empty when it behaved.
213 """
214 tests_dir, readme = _write_fixture(root, case)
215 code, messages = evaluate(tests_dir, readme, min_subdirs=case.floor)
216 failures = []
217 if code != case.want_code:
218 failures.append(f" {case.name}: exit {code}, expected {case.want_code}")
219 if case.needle is None:
220 if case.want_code == EXIT_OK and messages:
221 failures.append(f" {case.name}: expected no messages, got {messages}")
222 elif not any(case.needle in message for message in messages):
223 failures.append(f" {case.name}: no message mentioned '{case.needle}': {messages}")
224 return failures

226
227def _selftest() -> int:
228 """Prove both drift directions fire, a clean tree stays quiet, the floor holds.
229
230 Returns:
231 0 when every fixture behaves, 1 otherwise.
232 """
233 cases = _selftest_cases()
234 failures: list[str] = []
235 for case in cases:
236 with tempfile.TemporaryDirectory() as tmp:
237 failures += _run_case(Path(tmp), case)
238 if failures:
239 print("check_tests_readme selftest FAILED:", file=sys.stderr)
240 print("\n".join(failures), file=sys.stderr)
241 return 1
242 print(f"selftest OK: {len(cases)} cases (both drift directions + non-vacuity floor)")
243 return 0

246def main(argv: list[str] | None = None) -> int:
247 """Entry point.
248
249 Args:
250 argv: Command line, defaulting to ``sys.argv[1:]``.
251
252 Returns:
253 0 when the README matches the tree, 1 on drift, 2 on a collapsed scan.
254 """
255 parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
256 parser.add_argument("--selftest", action="store_true", help="prove both drift directions fire")
257 args = parser.parse_args(argv)
258 if args.selftest:
259 return _selftest()
260 code, messages = evaluate(TESTS_DIR, README)
261 if code == EXIT_OK:
262 count = len(immediate_subdirs(TESTS_DIR))
263 print(f"tests/README.md OK: {count} subdirectory(ies) documented, none stale")
264 return EXIT_OK
265 label = "collapsed scan" if code == EXIT_VACUOUS else "drift"
266 print(f"tests/README.md {label}: {len(messages)} problem(s):", file=sys.stderr)
267 for message in messages:
268 print(f" {message}", file=sys.stderr)
269 return code
