203ra8_err_t ra8_usb_pprn_init(ra8_usb_speed_t speed)
204{
205 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
206 return k_ra8_err_invalid_arg;
207 }
208 const ra8_err_t usb_err = ra8_usb_device_init(speed);
209 if (usb_err != k_ra8_ok) {
210 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
211 return k_ra8_err_hw_init_failed;
212 }
213 internal_reset_shadow(speed);
214
215 const ra8_err_t pipes_err = internal_configure_pipes(speed);
216 if (pipes_err != k_ra8_ok) {
217 /* GCOVR_EXCL_START -- host-unreachable defensive roll-back. internal_configure_pipes only
218 * returns non-ok when ra8_usb_configure_endpoint fails, which fails solely on argument
219 * validation (internal_check_ep_args) or a NULL controller block. This layer calls it with
220 * compile-time-constant valid arguments (PIPE3/PIPE4, EP1/EP2, valid dir/type, max-packet
221 * 64/512 <= 1024 ceiling) and speed is already validated to FS/HS above, so internal_pick
222 * never returns NULL. No fake register seed can force this leg. */
223 (void)ra8_usb_device_deinit(speed);
224 return pipes_err;
225 /* GCOVR_EXCL_STOP */
226 }
227 s_state.initialized = true;
228 ra8_log_info_val(s_tag, "device-Printer ready", (uint32_t)speed);
229 return k_ra8_ok;
230}

232ra8_err_t ra8_usb_pprn_close(void)
233{

235 return k_ra8_err_invalid_state;

239 s_state.initialized = false;
240 s_state.desc = nullptr;
241 s_state.device_id = nullptr;
242 s_state.setup_cb = nullptr;
243 s_state.setup_ctx = nullptr;
244 return err;

268 return k_ra8_err_invalid_arg;

270 s_state.desc = desc;
271 s_state.desc_len = desc_len;
272 s_state.device_id = device_id;
273 s_state.device_id_len = device_id_len;
274 return k_ra8_ok;

282ra8_err_t ra8_usb_pprn_recv(uint8_t* buf, uint16_t max_len, uint16_t* got_len)
283{
284 RA8_CHECK_NULL_PTR(buf, s_tag, "recv: buf");
285 RA8_CHECK_NULL_PTR(got_len, s_tag, "recv: got_len");
286 if (!s_state.initialized) {
287 return k_ra8_err_invalid_state;
288 }
289 if (max_len == 0U) {
290 return k_ra8_err_invalid_arg;
291 }
292 uint16_t inout_len = max_len;
293 const ra8_err_t err =
294 ra8_usb_queue_out(s_state.speed, k_ra8_pprn_pipe_bulk_out, buf, &inout_len, true);
295 if (err == k_ra8_ok) {
296 *got_len = inout_len;
297 } else {
298 *got_len = 0U;
299 }
300 return err;
301}

303ra8_err_t ra8_usb_pprn_send(const uint8_t* data, uint16_t len)
304{
305 if (!s_state.initialized) {
306 return k_ra8_err_invalid_state;

312 return k_ra8_err_invalid_arg;
313 }

356/* =============================================================================
357 * Class SETUP dispatch
358 * =============================================================================

360
361ra8_err_t ra8_usb_pprn_handle_setup(const ra8_usb_setup_t* setup)
362{
363 RA8_CHECK_NULL_PTR(setup, s_tag, "handle_setup: setup");
364 if (!s_state.initialized) {
365 return k_ra8_err_invalid_state;
366 }
367 if ((setup->bm_request_type != k_ra8_pprn_bm_class_iface_in) &&
368 (setup->bm_request_type != k_ra8_pprn_bm_class_iface_out)) {
369 return k_ra8_err_not_supported;
370 }
371 if (!internal_is_known_class_request(setup->b_request)) {
372 return k_ra8_err_not_supported;
373 }
374 /* SOFT_RESET (USB Printer 1.1 sec 4.2.3) flushes both bulk pipes;
375 * implemented here as a no-op shadow refresh so the application can
376 * re-arm pipes from its callback if it wants to. */
377 if (s_state.setup_cb != nullptr) {
378 const ra8_err_t cb_err = s_state.setup_cb(s_state.setup_ctx, setup);
379 if (cb_err != k_ra8_ok) {
380 return ra8_usb_control_response(s_state.speed, false);
381 }
382 }
383 return ra8_usb_control_response(s_state.speed, true);
