156RA8_INTERNAL static uint32_t internal_crc32(const uint8_t* data, uint32_t length)
157{
158 uint32_t crc = (uint32_t)k_ck_crc_seed;
159 for (uint32_t i = 0U; i < length; ++i) {
160 crc ^= data[i];
161 for (uint8_t bit = 0U; bit < (uint8_t)k_ck_bits_per_byte; ++bit) {
162 const uint32_t mask = (uint32_t)(0U - (crc & 1U));
163 crc = (crc >> 1U) ^ ((uint32_t)k_ck_crc_poly & mask);
164 }
165 }
166 return crc ^ (uint32_t)k_ck_crc_seed;
167}

187RA8_INTERNAL static bool internal_ranges_overlap(const void* first,
188 uint32_t first_bytes,
189 const void* second,
190 uint32_t second_bytes)
191{
192 const uintptr_t first_start = (uintptr_t)first;
193 const uintptr_t second_start = (uintptr_t)second;
194 if (first_bytes == 0U) {
195 return false;
196 }
197 if (second_bytes == 0U) {
198 return false;
199 }
200 if (first_start > UINTPTR_MAX - ((uintptr_t)first_bytes - 1U)) {
201 return true;
202 }
203 if (second_start > UINTPTR_MAX - ((uintptr_t)second_bytes - 1U)) {
204 return true;
205 }
206 const uintptr_t first_end = first_start + (uintptr_t)first_bytes - 1U;
207 const uintptr_t second_end = second_start + (uintptr_t)second_bytes - 1U;
208 if (first_start > second_end) {
209 return false;
210 }
211 return second_start <= first_end;
212}

260RA8_INTERNAL static ra8_err_t
261internal_size_values(uint32_t logical_blocks, uint32_t physical_blocks, uint32_t* out)
262{
263 if (logical_blocks == 0U) {
264 return k_ra8_err_invalid_size;
265 }
266 if (physical_blocks > (uint32_t)k_ra8_ftl_max_pblocks) {
267 return k_ra8_err_invalid_size;
268 }
269 if (physical_blocks <= logical_blocks) {
270 return k_ra8_err_invalid_size;
271 }
272 uint32_t total = (uint32_t)k_ck_fixed_bytes;
273 if (logical_blocks > (UINT32_MAX - total) / (uint32_t)k_ck_map_entry_bytes) {
274 return k_ra8_err_invalid_size;
275 }
276 total += logical_blocks * (uint32_t)k_ck_map_entry_bytes;
277 if (physical_blocks > (UINT32_MAX - total) / (uint32_t)k_ck_pblock_entry_bytes) {
278 return k_ra8_err_invalid_size;
279 }
280 *out = total + (physical_blocks * (uint32_t)k_ck_pblock_entry_bytes);
281 return k_ra8_ok;

426RA8_INTERNAL static ra8_err_t
427internal_native_window(const ra8_ftl_t* ftl, uint32_t base, uint32_t count)
428{
429 (void)memset(ftl->scratch, 0, (size_t)k_ck_scratch_bytes);
430 for (uint32_t lbn = 0U; lbn < ftl->logical_blocks; ++lbn) {
431 const uint16_t phys = ftl->map[lbn];
432 if (phys == (uint16_t)k_ra8_ftl_unmapped) {
433 continue;
434 }
435 const ra8_err_t marked = internal_window_mark(ftl, base, count, phys);
436 if (marked != k_ra8_ok) {
437 return marked;
438 }
439 }
440 for (uint32_t rel = 0U; rel < count; ++rel) {
441 const uint8_t state = ftl->pblocks[base + rel].state;
442 if (state > (uint8_t)k_ra8_ftl_pstate_stale) {
443 return k_ra8_err_invalid_state;
444 }
445 if (internal_bit_is_set(ftl->scratch, rel) != (state == (uint8_t)k_ra8_ftl_pstate_live)) {
446 return k_ra8_err_invalid_state;
447 }
448 }
449 return k_ra8_ok;

587RA8_INTERNAL static ra8_err_t
588internal_validate_header(const ra8_ftl_t* ftl, const uint8_t* buf, uint32_t buf_len, uint32_t need)
589{
590 if (buf_len < (uint32_t)k_ck_crc_bytes) {
591 return k_ra8_err_invalid_size;
592 }
593 const uint32_t magic = internal_get_le32(buf);
594 if (magic == (uint32_t)k_ck_legacy_magic_le) {
595 return k_ra8_err_not_supported;
596 }
597 if (magic == (uint32_t)k_ck_legacy_magic_swapped) {
598 return k_ra8_err_not_supported;
599 }
600 if (magic != (uint32_t)k_ck_magic) {
601 return k_ra8_err_invalid_state;
602 }
603 if (buf_len < (uint32_t)k_ck_fixed_bytes) {
604 return k_ra8_err_invalid_size;
605 }
606 if (internal_get_le16(&buf[k_ck_off_version]) != (uint16_t)k_ck_version) {
607 return k_ra8_err_not_supported;
608 }
609 if (internal_get_le16(&buf[k_ck_off_header_bytes]) != (uint16_t)k_ck_header_bytes) {
610 return k_ra8_err_invalid_size;
611 }
612 if (internal_get_le32(&buf[k_ck_off_total_bytes]) != buf_len) {
613 return k_ra8_err_invalid_size;
614 }
615 if (buf_len != need) {
616 return k_ra8_err_invalid_size;
617 }
618 if (internal_get_le32(&buf[k_ck_off_logical_blocks]) != ftl->logical_blocks) {
619 return k_ra8_err_invalid_arg;
620 }
621 if (internal_get_le32(&buf[k_ck_off_physical_blocks]) != ftl->physical_blocks) {
622 return k_ra8_err_invalid_arg;
623 }
624 const uint32_t crc_offset = buf_len - (uint32_t)k_ck_crc_bytes;
625 return (internal_crc32(buf, crc_offset) == internal_get_le32(&buf[crc_offset]))
626 ? k_ra8_ok
627 : k_ra8_err_crc_mismatch;
628}

644RA8_INTERNAL static void internal_encode(const ra8_ftl_t* ftl, uint8_t* buf, uint32_t need)
645{
646 internal_put_le32(buf, (uint32_t)k_ck_magic);
647 internal_put_le16(&buf[k_ck_off_version], (uint16_t)k_ck_version);
648 internal_put_le16(&buf[k_ck_off_header_bytes], (uint16_t)k_ck_header_bytes);
649 internal_put_le32(&buf[k_ck_off_total_bytes], need);
650 internal_put_le32(&buf[k_ck_off_logical_blocks], ftl->logical_blocks);
651 internal_put_le32(&buf[k_ck_off_physical_blocks], ftl->physical_blocks);
652 uint32_t offset = (uint32_t)k_ck_header_bytes;
653 for (uint32_t lbn = 0U; lbn < ftl->logical_blocks; ++lbn) {
654 internal_put_le16(&buf[offset], ftl->map[lbn]);
655 offset += (uint32_t)k_ck_map_entry_bytes;
656 }
657 for (uint32_t phys = 0U; phys < ftl->physical_blocks; ++phys) {
658 internal_put_le32(&buf[offset], ftl->pblocks[phys].erase_count);
659 buf[offset + (uint32_t)k_ck_pblock_state_offset] = ftl->pblocks[phys].state;
660 offset += (uint32_t)k_ck_pblock_entry_bytes;
661 }
662 internal_put_le32(&buf[offset], internal_crc32(buf, offset));
663}
