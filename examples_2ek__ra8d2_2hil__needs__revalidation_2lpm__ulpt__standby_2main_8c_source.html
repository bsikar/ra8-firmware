155[[nodiscard]] static ra8_err_t lus_wait_count_started(uint8_t channel)
156{
157 const uint8_t tcstf_mask = (uint8_t)(1U << (uint8_t)k_ra8_ulpt_bit_tcstf);
158 for (uint32_t i = 0U; i < (uint32_t)k_lus_tcstf_poll_limit; ++i) {
159 uint8_t status = 0U;
160 ra8_err_t err = ra8_ulpt_get_status(channel, &status);
161 if (err != k_ra8_ok) {
162 return err;
163 }
164 if ((status & tcstf_mask) != 0U) {
165 return k_ra8_ok;
166 }
167 }
168 return k_ra8_err_hw_timeout;
169}

249[[nodiscard]] static ra8_err_t lus_arm_wake(void)
250{
251 ra8_err_t err = ra8_isr_init();
252 if (err != k_ra8_ok) {
253 return err;
254 }
255 err = ra8_isr_register(k_ra8_elc_event_ulpt0_ulpti,
256 lus_ulpt_isr,
257 nullptr,
258 (uint8_t)k_ra8_isr_prio_default,
259 nullptr);
260 if (err != k_ra8_ok) {
261 return err;
262 }
263 return ra8_lpm_arm_wupen1_bits((uint32_t)k_ra8_lpm_wupen1_ulpt0u);
264}

268int32_t main(void)
269{
270 lus_setup_or_halt();
271
272 if (lus_arm_wake() != k_ra8_ok) {
273 lus_panic_halt();
274 }
275 ra8_isr_globals_enable();
276
277 (void)ra8_board_uart_console_write(k_lus_boot_msg, (size_t)(sizeof(k_lus_boot_msg) - 1U));
278
279 while (1) {
280 /* Re-arm the ULPT countdown, then drop into Software Standby until
281 * the underflow cancels it. ra8_ulpt_stop clears ULPTCR.TUNF so the
282 * next cycle starts clean. */
283 if (ra8_ulpt_start((uint8_t)k_lus_channel, (uint32_t)k_lus_period_ticks) != k_ra8_ok) {
284 break;
285 }
286 if (lus_wait_count_started((uint8_t)k_lus_channel) != k_ra8_ok) {
287 break;
288 }
289 if (ra8_lpm_enter_sleep(k_ra8_sleep_mode_software_std) != k_ra8_ok) {
290 break;
291 }
292 if (ra8_ulpt_stop((uint8_t)k_lus_channel) != k_ra8_ok) {
293 break;
294 }
295 if (ra8_board_uart_console_write(k_lus_wake_msg, (size_t)(sizeof(k_lus_wake_msg) - 1U)) !=
296 k_ra8_ok) {
297 break;
298 }
299 }
300 lus_panic_halt();
301 return 0;
302}
