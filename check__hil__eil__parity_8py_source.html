4"""Gate: EIL==HIL parity -- every HIL app is also exercised in the emulator.
5
6The owner invariant this gate enforces: **every hardware-in-the-loop (HIL) app
7MUST also be exercised in the emulator (EIL), automatically and dynamically**,
8so that *adding* a HIL app cannot silently escape EIL coverage.
9
10Two harnesses already exist:
11
12 * ``scripts/hil/all.sh`` flashes the physical EK-RA8D2 and scrapes its UART /
13 J-Link / RTT / wire for every app under
14 ``examples/ek_ra8d2/hw_validated/hil/``.
15 * ``scripts/emu/eil_all.sh`` boots the SAME ``.elf`` files in ``tools/ra8_emulator``
16 headless and checks the SAME per-app ``hil.conf`` expectations with NO board
17 attached. The ``eil-integration`` CI job runs it enforcing, 0 skips.
18
19Nothing, however, GUARANTEED that the two suites cover the same set of apps or
20that a newly-added HIL app is EIL-visible. A HIL app whose ``hil.conf`` is
21missing, or whose ``HIL_MODE`` ``ra8_emulator`` cannot check, would quietly fall
22out of EIL coverage -- exactly the drift this gate makes impossible.
23
24It is a CHEAP, hardware-free structural gate: it does not build or run either
25harness. It re-derives each harness's app-discovery *from the harness scripts
26themselves* (parsing ``HIL_DIR`` / ``EIL_RA8P1_DIR`` and the EIL-capable-mode
27``case`` list out of ``eil_all.sh``, ``HIL_DIR`` out of ``hil_all.sh``) and then
28globs the tree exactly as those scripts do. Because the roots and the mode list
29are parsed, not hardcoded, the gate tracks the harnesses as they evolve.
30
31It FAILS (non-zero) if any of the following hold, with a precise message per
32offender:
33
34 1. an app under ``examples/ek_ra8d2/hw_validated/hil/`` has no ``hil.conf``
35 (it is HIL-tiered but unspecified -- ``hil_all.sh`` fails loud on it, and
36 ``ra8_emulator`` never sees it);
37 2. the set of apps ``hil_all.sh`` would run is not covered by the set
38 ``eil_all.sh`` would run -- drift between the two harnesses. ``eil_all.sh``
39 legitimately runs a SUPERSET (it also emulates the RA8P1/NPU foundation
40 apps under ``examples/ra8p1_foundation/`` via ``--device ra8p1``, which the
41 RA8D2 bench cannot flash), so the relation checked is HIL subset-of EIL:
42 every HIL app must be in the EIL run set;
43 3. any ``hw_validated/hil`` app declares a ``HIL_MODE`` that ``eil_all.sh``
44 cannot check -- i.e. one that is not in its EIL-capable set
45 (``uart_scrape`` / ``alive`` / ``jlink_memprobe`` / ``rtt_scrape`` /
46 ``hil_eth_tcp``, parsed from ``eil_all.sh``). There must be NO EIL skips.
47
48Run::
49
50 check_hil_eil_parity.py # gate (exit 1 on any violation)
51 check_hil_eil_parity.py --list # enumerate the derived sets, no gate
52
53Exit 0 if the EIL==HIL invariant holds, exit 1 otherwise, exit 2 if the harness
54scripts could not be parsed (a structural change that must be reconciled -- the
55gate refuses to silently no-op).
56"""

58# ---------------------------------------------------------------------------
59# EIL==HIL is an ONGOING DISCIPLINE, not a one-time state.
60#
61# "EIL==HIL" means ra8_emulator (tools/ra8_emulator) must reproduce the ACTUAL
62# silicon behaviour of every path a HIL app exercises -- bugs INCLUDED, not an
63# idealised model. When a new HIL app drives a peripheral or code path that
64# ra8_emulator does not yet model, the correct response is NOT to skip it in EIL:
65# ra8_emulator MUST be extended so the EIL suite stays complete. This gate makes
66# skipping impossible (a missing hil.conf, an un-modelled HIL_MODE, or an app
67# invisible to eil_all.sh all fail the build), which is precisely what FORCES
68# the emulator to keep up with the hardware. Concretely, the upcoming
69# archive / compression work may need new ra8_emulator SD / decode models before
70# its HIL apps can pass EIL -- and that is by design: the emulator follows silicon.
71# ---------------------------------------------------------------------------

112def _parse_repo_root_dir(text: str, var: str) -> Path | None:
113 """Parse ``VAR="${REPO_ROOT}/<path>"`` and return REPO_ROOT / <path>.
114
115 This is how both harnesses anchor their discovery roots, so parsing it (per
116 harness) keeps the gate tracking a root the scripts may relocate.
117 """
118 pattern = re.compile(r"^\s*" + re.escape(var) + r'="\$\{REPO_ROOT\}/([^"]+)"', re.MULTILINE)
119 match = pattern.search(text)
120 if match is None:
121 return None
122 return REPO_ROOT / match.group(1)

125def _parse_eil_modes(text: str) -> list[str] | None:
126 """Parse the EIL-capable HIL_MODE set from eil_all.sh's run_one dispatch.
127
128 The authoritative list is the ``case`` label right after
129 ``case "${HIL_MODE:-}" in``::
130
131 uart_scrape | alive | jlink_memprobe | rtt_scrape | hil_eth_tcp) : ;;
132
133 Parsed (not hardcoded) so a mode added to the emulator is picked up here.
134 Returns None if the dispatch or its label line cannot be found/parsed.
135 """
136 lines = text.splitlines()
137 for idx, line in enumerate(lines):
138 if EIL_MODE_CASE_HEADER not in line:
139 continue
140 for follow in lines[idx + 1 :]:
141 stripped = follow.strip()
142 if not stripped or stripped.startswith("#"):
143 continue
144 match = re.match(r"^([a-z0-9_ |]+)\‍)\s*:\s*;;\s*$", stripped)
145 if match is None:
146 return None
147 return [tok.strip() for tok in match.group(1).split("|") if tok.strip()]
148 return None

151def _discover_apps(root: Path) -> list[str]:
152 """App names directly under a hil/ root -- mirrors hil_discover_apps().
153
154 An app is an immediate child DIRECTORY; the README.md file is skipped.
155 Returned sorted, matching the harness's ``| sort``.
156 """
157 if not root.is_dir():
158 return []
159 names = [c.name for c in root.iterdir() if c.is_dir() and c.name != DISCOVERY_SKIP]
160 return sorted(names)

163def _hil_mode_of(conf: Path) -> str | None:
164 """Parse ``HIL_MODE=<mode>`` from a hil.conf (quotes stripped)."""
165 if not conf.is_file():
166 return None
167 text = _read(conf)
168 if text is None:
169 return None
170 for raw in text.splitlines():
171 line = raw.strip()
172 if line.startswith("HIL_MODE="):
173 return line[len("HIL_MODE=") :].strip().strip('"').strip("'")
174 return None

190def build_model() -> tuple[Model | None, list[str]]:
191 """Derive the two harnesses' discovery, or a list of parse-error strings."""
192 errors: list[str] = []
193 hil_text = _read(HIL_ALL)
194 eil_text = _read(EIL_ALL)
195 if hil_text is None:
196 errors.append(f"cannot read {_rel(HIL_ALL)}")
197 if eil_text is None:
198 errors.append(f"cannot read {_rel(EIL_ALL)}")
199 if hil_text is None or eil_text is None:
200 return None, errors
201
202 hil_root_hilall = _parse_repo_root_dir(hil_text, "HIL_DIR")
203 hil_root_silall = _parse_repo_root_dir(eil_text, "HIL_DIR")
204 ra8p1_root = _parse_repo_root_dir(eil_text, "EIL_RA8P1_DIR")
205 eil_modes = _parse_eil_modes(eil_text)
206 if hil_root_hilall is None:
207 errors.append(f'{_rel(HIL_ALL)}: no HIL_DIR="${{REPO_ROOT}}/..." assignment')
208 if hil_root_silall is None:
209 errors.append(f'{_rel(EIL_ALL)}: no HIL_DIR="${{REPO_ROOT}}/..." assignment')
210 if ra8p1_root is None:
211 errors.append(f'{_rel(EIL_ALL)}: no EIL_RA8P1_DIR="${{REPO_ROOT}}/..." assignment')
212 if eil_modes is None:
213 errors.append(
214 f"{_rel(EIL_ALL)}: could not parse the EIL-capable mode set from the "
215 f"'{EIL_MODE_CASE_HEADER}' dispatch"
216 )
217 if None in (hil_root_hilall, hil_root_silall, ra8p1_root, eil_modes):
218 return None, errors
219
220 hil_apps = _discover_apps(hil_root_hilall)
221 eil = set(_discover_apps(hil_root_silall))
222 ra8p1_apps = [
223 name for name in _discover_apps(ra8p1_root) if (ra8p1_root / name / "hil.conf").is_file()
224 ]
225 eil.update(ra8p1_apps)
226 model = Model(
227 hil_root_hilall=hil_root_hilall,
228 hil_root_silall=hil_root_silall,
229 ra8p1_root=ra8p1_root,
230 eil_modes=eil_modes,
231 hil_apps=hil_apps,
232 ra8p1_apps=ra8p1_apps,
233 eil_apps=sorted(eil),
234 )
235 return model, []

275def check_missing_conf(model: Model) -> list[str]:
276 """Check 1: every hil/ app must declare a hil.conf."""
277 return [
278 _msg_no_conf(_rel(model.hil_root_hilall / app))
279 for app in model.hil_apps
280 if not (model.hil_root_hilall / app / "hil.conf").is_file()
281 ]

284def check_set_drift(model: Model) -> list[str]:
285 """Check 2: hil_all's run set must be covered by eil_all's run set."""
286 offenders: list[str] = []
287 if model.hil_root_hilall != model.hil_root_silall:
288 offenders.append(_msg_root_drift(model))
289 eil_set = set(model.eil_apps)
290 offenders.extend(_msg_escapes_eil(app) for app in model.hil_apps if app not in eil_set)
291 return offenders

293
294def check_unsupported_mode(model: Model) -> list[str]:
295 """Check 3: every hil/ app's HIL_MODE must be EIL-checkable (no skips)."""
296 capable = set(model.eil_modes)
297 offenders: list[str] = []
298 for app in model.hil_apps:
299 # A missing hil.conf / HIL_MODE is reported by check 1, not double-counted.
300 mode = _hil_mode_of(model.hil_root_hilall / app / "hil.conf")
301 if mode is not None and mode not in capable:
302 offenders.append(_msg_bad_mode(app, mode, capable))
303 return offenders

337def main(argv: list[str]) -> int:
338 """Fail when the HIL app set and the EIL app set have drifted apart.
339
340 A model-build failure short-circuits ahead of every parity check and is
341 reported on its own: if the two sets could not be derived, any verdict
342 about their agreement would be an artefact of the broken parse rather than
343 a fact about the tree.
344
345 ``--list`` prints the derived sets and modes and always exits 0. It is a
346 debugging aid, explicitly NOT a gate -- CI must invoke this bare, or the
347 step passes without ever comparing anything.
348
349 Returns 0 when parity holds or under ``--list``, 1 on drift (an app
350 missing a hil.conf, a set mismatch, or an unsupported mode) or on a
351 model-build error.
352 """
353 parser = argparse.ArgumentParser(
354 prog="check_hil_eil_parity.py",
355 description="Gate: every HIL app is also exercised in the emulator (EIL==HIL).",
356 )
357 parser.add_argument(
358 "--list",
359 action="store_true",
360 help="enumerate the derived hil/eil sets + modes, then exit 0 (no gating).",
361 )
362 args = parser.parse_args(argv[1:])
363
364 model, errors = build_model()
365 if model is None:
366 return _report_parse_errors(errors)
367
368 if args.list:
369 _print_list(model)
370 return 0
371
372 offenders = check_missing_conf(model) + check_set_drift(model) + check_unsupported_mode(model)
373 if not offenders:
374 print(
375 f"check_hil_eil_parity.py: EIL==HIL holds -- {len(model.hil_apps)} HIL "
376 "app(s), all with a hil.conf, all in an EIL-checkable mode "
377 f"({', '.join(model.eil_modes)}), all covered by eil_all.sh "
378 f"({len(model.eil_apps)} EIL app(s) incl. {len(model.ra8p1_apps)} RA8P1)."
379 )
380 return 0
381
382 print(
383 f"check_hil_eil_parity.py: {len(offenders)} EIL==HIL parity violation(s):\n",
384 file=sys.stderr,
385 )
386 for offender in offenders:
387 print(f" - {offender}", file=sys.stderr)
388 print(
389 "\nEIL==HIL is an ongoing discipline: ra8_emulator must reproduce the actual\n"
390 "silicon behaviour of every HIL path (bugs included). A HIL app may never\n"
391 "be skipped in EIL -- if ra8_emulator cannot yet model a path, extend\n"
392 "ra8_emulator + eil_all.sh so EIL stays complete. See the header comment.",
393 file=sys.stderr,
394 )
395 return 1
