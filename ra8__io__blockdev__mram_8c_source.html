129RA8_INTERNAL
130static ra8_err_t mram_window_ok(uintptr_t base, uint32_t block_count)
131{
132 const uintptr_t extra_start = (uintptr_t)k_ra8_flash_extra_start;
133 const uintptr_t extra_end = extra_start + (uintptr_t)k_ra8_flash_extra_size;
134 const uintptr_t span = (uintptr_t)block_count * (uintptr_t)k_ra8_io_block_size_bytes;
135 if ((base % (uintptr_t)k_ra8_io_mram_erase_block_bytes) != 0U) {
136 return k_ra8_err_invalid_arg;
137 }
138 if ((span % (uintptr_t)k_ra8_io_mram_erase_block_bytes) != 0U) {
139 return k_ra8_err_invalid_arg;
140 }
141 if (base < extra_start) {
142 return k_ra8_err_invalid_arg;
143 }
144 if (base >= extra_end) {
145 return k_ra8_err_invalid_arg;
146 }
147 if (span > extra_end - base) {
148 return k_ra8_err_invalid_arg;
149 }
150 return k_ra8_ok;
151}

226static ra8_err_t mram_write(void* ctx, uint32_t lba, uint32_t count, const uint8_t* buf)
227{
228 RA8_CHECK_NULL_PTR(ctx, s_tag, "ctx must not be nullptr");
229 RA8_CHECK_NULL_PTR(buf, s_tag, "buf must not be nullptr");
230 ra8_io_blockdev_mram_state_t* st = (ra8_io_blockdev_mram_state_t*)ctx;
231 if (st->read_only) {
232 return k_ra8_err_not_supported;
233 }
234 const ra8_err_t b = mram_bounds(st, lba, count);
235 if (b != k_ra8_ok) {
236 return b;
237 }
238 const uintptr_t off = (uintptr_t)lba * (uintptr_t)k_ra8_io_block_size_bytes;
239 const uintptr_t span = (uintptr_t)count * (uintptr_t)k_ra8_io_block_size_bytes;
240 const uint32_t step = (uint32_t)k_ra8_io_mram_program_bytes;
241 for (uintptr_t done = (uintptr_t)k_ra8_io_mram_zero_len; done < span; done += step) {
242 const ra8_err_t w =
243 ra8_flash_extra_mram_write((uint32_t)(st->base + off + done), &buf[done], step);
244 if (w != k_ra8_ok) {
245 return w;
246 }
247 }
248 return k_ra8_ok;

352ra8_err_t ra8_io_blockdev_mram_init(ra8_io_blockdev_t* bd,
353 ra8_io_blockdev_mram_state_t* state,
354 uintptr_t base_addr,
355 uint32_t block_count,
356 bool read_only)
357{
358 RA8_CHECK_NULL_PTR(bd, s_tag, "bd must not be nullptr");
359 RA8_CHECK_NULL_PTR(state, s_tag, "state must not be nullptr");
360 if (block_count < (uint32_t)k_ra8_io_mram_min_blocks) {
361 return k_ra8_err_invalid_arg;
362 }
363 const ra8_err_t w = mram_window_ok(base_addr, block_count);
364 if (w != k_ra8_ok) {
365 return w;
366 }
367 state->base = base_addr;
368 state->block_count = block_count;
369 state->read_only = read_only;
370 bd->iface = &k_mram_iface;
371 bd->ctx = state;
372 return k_ra8_ok;
