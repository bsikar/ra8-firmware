97RA8_INTERNAL static bool jof_is_webp(const uint8_t* data, size_t len)
98{
99 if ((data == nullptr) || (len < (size_t)k_jof_webp_head_len)) {
100 return false;
101 }
102 return (memcmp(&data[k_jof_webp_riff_ofs], s_jof_webp_riff, sizeof(s_jof_webp_riff)) == 0) &&
103 (memcmp(&data[k_jof_webp_fourcc_ofs], s_jof_webp_webp, sizeof(s_jof_webp_webp)) == 0);
104}

121RA8_INTERNAL static ra8_err_t jof_pull(void* ctx, uint8_t* buf, size_t cap, size_t* got)
122{
123 jof_pull_ctx_t* s = (jof_pull_ctx_t*)ctx;
124 size_t n = s->len - s->pos;
125 if (n > cap) {
126 n = cap;
127 }
128 memcpy(buf, s->data + s->pos, n);
129 s->pos += n;
130 *got = n;
131 return k_ra8_ok;

141RA8_INTERNAL static ra8_err_t slurp(const char* path, uint8_t** out_buf, size_t* out_len)
142{
143 FILE* f = fopen(path, "rb");
144 if (f == nullptr) {
145 return k_ra8_fail;
146 }
147 (void)fseek(f, 0, SEEK_END);
148 const long sz = ftell(f);
149 (void)fseek(f, 0, SEEK_SET);
150 if (sz <= 0) {
151 (void)fclose(f);
152 return k_ra8_fail;
153 }
154 uint8_t* buf = (uint8_t*)malloc((size_t)sz);
155 if ((buf == nullptr) || (fread(buf, 1U, (size_t)sz, f) != (size_t)sz)) {
156 free(buf);
157 (void)fclose(f);
158 return k_ra8_fail;
159 }
160 (void)fclose(f);
161 *out_buf = buf;
162 *out_len = (size_t)sz;
163 return k_ra8_ok;
164}

290RA8_INTERNAL static ra8_err_t jof_one(const char* in_path, const char* out_path)
291{
292 uint8_t* src = nullptr;
293 size_t slen = 0U;
294 ra8_err_t rc = slurp(in_path, &src, &slen);
295 if (rc != k_ra8_ok) {
296 return rc;
297 }
298 uint16_t w = 0U;
299 uint16_t h = 0U;
300 /* Probe through the producer's own dispatch, so every format the producer can
301 * decode (JPEG, PNG, WebP) is a format this exporter can size and write. */
302 if (ra8_jof_probe_dims(src, slen, &w, &h) != k_ra8_ok) {
303 free(src);
304 return k_ra8_err_not_supported;
305 }
306 const uint16_t tile_h = (h < (uint16_t)k_jof_band_h) ? h : (uint16_t)k_jof_band_h;
307 const uint32_t work_cap = ra8_jof_work_bytes(w, h, w, tile_h);
308 if (work_cap == 0U) {
309 free(src);
310 return k_ra8_err_invalid_size;
311 }
312 /* Produce into a sibling temp and rename on success: re-exporting a chapter
313 * must not destroy the `.jof` already sitting there if this page fails to
314 * decode (see mdl_atomic.h). */
315 char tmp[PATH_MAX];
316 if (!mdl_atomic_tmp_path(out_path, tmp, sizeof(tmp))) {
317 free(src);
318 return k_ra8_fail;
319 }
320 FILE* out = fopen(tmp, "wb");
321 if (out == nullptr) {
322 free(src);
323 return k_ra8_fail;
324 }
325 rc = jof_produce_page(src, slen, w, h, tile_h, work_cap, out);
326 if (fclose(out) != 0) {
327 rc = (rc == k_ra8_ok) ? k_ra8_fail : rc;
328 }
329 free(src);
330 if (rc != k_ra8_ok) {
331 mdl_atomic_abort(tmp);
332 return rc;
333 }
334 return mdl_atomic_commit(tmp, out_path) ? k_ra8_ok : k_ra8_fail;
335}

337RA8_PRIV ra8_err_t mdl_export_jof(const char* dir, const char names[][k_name_max], size_t count)
338{
339 ra8_log_set_byte_sink(jof_log_sink, nullptr);
340 ra8_err_t rc = k_ra8_ok;
341 for (size_t i = 0U; (rc == k_ra8_ok) && (i < count); ++i) {
342 char in_path[PATH_MAX];
343 char stem[k_name_max];
344 char out_path[PATH_MAX];
345 (void)snprintf(in_path, sizeof(in_path), "%s/%s", dir, names[i]);
346 (void)snprintf(stem, sizeof(stem), "%s", names[i]);
347 char* dot = strrchr(stem, '.');
348 if (dot != nullptr) {
349 *dot = '\0';
350 }
351 (void)snprintf(out_path, sizeof(out_path), "%s/%s.jof", dir, stem);
352 rc = jof_one(in_path, out_path);
353 }
354 return rc;
