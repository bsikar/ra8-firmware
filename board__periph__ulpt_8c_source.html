159static uint32_t ulpt_cell_value(const ulpt_channel_t* c, uint64_t off)
160{
161 /* HUM Ch 25.2.6 "ULPT : ULPT Counter Register" p 1198 */
162 if (off < (uint64_t)k_ulpt_off_cma) {
163 return c->counter;
164 }
165 /* HUM Ch 25 "Ultra-Low-Power Timer (ULPT)" p 1187 -- ULPTCMA */
166 if (off < (uint64_t)k_ulpt_off_cmb) {
167 return c->cmpa;
168 }
169 /* HUM Ch 25 "Ultra-Low-Power Timer (ULPT)" p 1187 -- ULPTCMB */
170 if (off < (uint64_t)k_ulpt_off_cr) {
171 return c->cmpb;
172 }
173 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 */
174 if (off == (uint64_t)k_ulpt_off_cr) {
175 return c->cr;
176 }
177 /* HUM Ch 25.2.2 "ULPTMR1 : ULPT Mode Register 1" p 1192 */
178 if (off == (uint64_t)k_ulpt_off_mr1) {
179 return c->mr1;
180 }
181 /* HUM Ch 25.2.3 "ULPTMR2 : ULPT Mode Register 2" p 1194 */
182 if (off == (uint64_t)k_ulpt_off_mr2) {
183 return c->mr2;
184 }
185 /* HUM Ch 25.2.4 "ULPTMR3 : ULPT Mode Register 3" p 1195 */
186 if (off == (uint64_t)k_ulpt_off_mr3) {
187 return c->mr3;
188 }
189 /* HUM Ch 25.2.5 "ULPTIOC : ULPT I/O Control Register" p 1196 */
190 return c->ioc;
191}

214static uint64_t ulpt_read(uc_engine* uc, uint64_t addr, unsigned size)
215{
216 (void)uc;
217 const uint64_t rel = addr - (uint64_t)k_ulpt_base;
218 const uint32_t ch = (uint32_t)(rel / (uint64_t)k_ulpt_stride);
219 if (ch >= (uint32_t)k_ulpt_count) {
220 return 0U;
221 }
222 const uint64_t off = rel % (uint64_t)k_ulpt_stride;
223 const uint32_t cell = ulpt_cell_value(&s_ulpt[ch], off);
224 const uint32_t lane = (uint32_t)(off - ulpt_cell_base(off));
225 uint64_t v = 0U;
226 for (unsigned i = 0U; i < size; i++) {
227 const uint32_t b = lane + (uint32_t)i;
228 if (b < (uint32_t)sizeof(uint32_t)) {
229 const uint32_t lane_byte =
230 (cell >> ((uint32_t)k_ulpt_byte_bits * b)) & (uint32_t)k_ulpt_byte_mask;
231 v |= (uint64_t)lane_byte << ((uint32_t)k_ulpt_byte_bits * (uint32_t)i);
232 }
233 }
234 return v;
235}

238static void ulpt_write_cr(ulpt_channel_t* c, uint8_t value)
239{
240 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 -- TUNDF is
241 * write-0-to-clear (ra8_ulpt_stop clears it by writing 0); TSTART is RW and
242 * TCSTF tracks TSTART. Mirrors the AGT AGTCR model. */
243 const uint8_t status_keep = (uint8_t)(c->cr & value & (uint8_t)k_ulpt_cr_tundf);
244 const uint8_t start = (uint8_t)(value & (uint8_t)k_ulpt_cr_tstart);
245 c->cr = (uint8_t)(start | (start != 0U ? (uint8_t)k_ulpt_cr_tcstf : 0U) | status_keep);
246}

249static void ulpt_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
250{
251 (void)uc;
252 const uint64_t rel = addr - (uint64_t)k_ulpt_base;
253 const uint32_t ch = (uint32_t)(rel / (uint64_t)k_ulpt_stride);
254 if (ch >= (uint32_t)k_ulpt_count) {
255 return;
256 }
257 ulpt_channel_t* c = &s_ulpt[ch];
258 const uint64_t off = rel % (uint64_t)k_ulpt_stride;
259 /* The ra8_ulpt driver only ever issues aligned, full-width stores: a 32-bit
260 * write to ULPTCNT/CMA/CMB and an 8-bit write to ULPTCR/MRn/IOC. */
261 if (off < (uint64_t)k_ulpt_off_cma) {
262 /* HUM Ch 25.2.6 "ULPT : ULPT Counter Register" p 1198 -- a write seeds both
263 * the live counter and the reload value (ra8_ulpt_start writes the period). */
264 c->counter = (uint32_t)value;
265 c->reload = (uint32_t)value;
266 } else if (off < (uint64_t)k_ulpt_off_cmb) {
267 /* HUM Ch 25 "Ultra-Low-Power Timer (ULPT)" p 1187 -- ULPTCMA */
268 c->cmpa = (uint32_t)value;
269 } else if (off < (uint64_t)k_ulpt_off_cr) {
270 /* HUM Ch 25 "Ultra-Low-Power Timer (ULPT)" p 1187 -- ULPTCMB */
271 c->cmpb = (uint32_t)value;
272 } else if (off == (uint64_t)k_ulpt_off_cr) {
273 ulpt_write_cr(c, (uint8_t)value);
274 } else if (off == (uint64_t)k_ulpt_off_mr1) {
275 /* HUM Ch 25.2.2 "ULPTMR1 : ULPT Mode Register 1" p 1192 */
276 c->mr1 = (uint8_t)value;
277 } else if (off == (uint64_t)k_ulpt_off_mr2) {
278 /* HUM Ch 25.2.3 "ULPTMR2 : ULPT Mode Register 2" p 1194 */
279 c->mr2 = (uint8_t)value;
280 } else if (off == (uint64_t)k_ulpt_off_mr3) {
281 /* HUM Ch 25.2.4 "ULPTMR3 : ULPT Mode Register 3" p 1195 */
282 c->mr3 = (uint8_t)value;
283 } else if (off == (uint64_t)k_ulpt_off_ioc) {
284 /* HUM Ch 25.2.5 "ULPTIOC : ULPT I/O Control Register" p 1196 */
285 c->ioc = (uint8_t)value;
286 }
287 (void)size;
288}

296static void ulpt_tick_channel(uc_engine* uc, uint32_t ch)
297{
298 ulpt_channel_t* c = &s_ulpt[ch];
299 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 -- TSTART gate */
300 if ((c->cr & (uint8_t)k_ulpt_cr_tstart) == 0U) {
301 return; /* stopped: counter holds */
302 }
303 const uint32_t step = (uint32_t)k_ulpt_step_per_chunk;
304 if (c->counter > step) {
305 c->counter -= step;
306 return;
307 }
308 /* Underflow: reload from the latched period and continue (periodic). */
309 const uint32_t span = c->reload + 1U;
310 const uint32_t deficit = step - c->counter;
311 c->counter = c->reload - ((deficit - 1U) % span);
312 /* Raise only on the rising edge of TUNDF -- one wake per armed period, until
313 * the firmware clears it by stopping/re-arming the channel. */
314 if ((c->cr & (uint8_t)k_ulpt_cr_tundf) == 0U) {
315 /* HUM Ch 25.2.1 "ULPTCR : ULPT Control Register" p 1190 -- TUNDF set */
316 c->cr |= (uint8_t)k_ulpt_cr_tundf;
317 c->underflows++;
318 if (ch == 0U) {
319 board_periph_icu_raise_event(uc, (uint16_t)k_ulpt_event_ch0_underflow);
320 }
321 if (board_periph_trace()) {
322 (void)fprintf(stderr, " ULPT%u : underflow (reload=0x%08X)\n", ch, c->reload);
323 }
324 }

349static void ulpt_report(void)
350{
351 for (uint32_t ch = 0U; ch < (uint32_t)k_ulpt_count; ch++) {
352 const ulpt_channel_t* c = &s_ulpt[ch];
353 if (c->underflows == 0U) {
354 continue;
355 }
356 (void)fprintf(stderr,
357 " ULPT%u : counter=0x%08X underflows=%u (running=%s)\n",
358 ch,
359 c->counter,
360 c->underflows,
361 (c->cr & (uint8_t)k_ulpt_cr_tstart) ? "yes" : "no");
362 }
