4"""Root-of-trust image signer + key ceremony for the RA8D2 secure-boot chain.
5
6The firmware verifier ``ra8_rot_verify_image`` (libs/ra8_dfu/src/ra8_rot.c) accepts a
7signed image of the form::
8
9 [ body (body_len bytes) ] [ ra8_rot_trailer_t ]
10
11and default-denies anything without a valid trailer. Nothing in-tree could
12*produce* that trailer, so ``RA8_ENABLE_ROOT_OF_TRUST`` could never be turned on.
13This tool closes that gap. It is a build/release-time host tool (never a CI
14gate); it shells out to ``openssl`` so it needs no third-party Python packages.
15
16The trailer is 116 bytes, little-endian, matching ra8_rot.h::ra8_rot_trailer_t::
17
18 uint32 magic = 0x524F5431 ("ROT1")
19 uint32 version = 1
20 uint32 img_version (monotonic anti-rollback counter)
21 uint32 body_len (bytes the digest covers)
22 uint32 sig_len = 64
23 uint8 digest[32] = SHA-256(body)
24 uint8 sig[64] = ECDSA-P256 raw r||s over the digest
25
26The public key is the 65-byte uncompressed P-256 point ``0x04 || X || Y`` that is
27embedded in ra8_rot.c::s_rot_root_pubkey.
28
29Commands::
30
31 rot_sign.py keygen --key priv.pem [--pubkey-c pubkey.h]
32 rot_sign.py sign --key priv.pem --image body.bin --out signed.bin
33 [--img-version N]
34 rot_sign.py selftest
35
36``selftest`` generates a throwaway key, signs a random body, then re-verifies the
37signature with openssl and checks the trailer layout -- proving the produced
38signature is valid ECDSA-P256 over SHA-256(body) with a 116-byte trailer.
39"""

51# Mirrors libs/ra8_dfu/inc/ra8_rot.h.
52ROT_MAGIC = 0x524F5431 # "ROT1"
53ROT_VERSION = 1
54DIGEST_BYTES = 32
55SIG_BYTES = 64
56PUBKEY_BYTES = 65
57BODY_MAX = 0x00100000 # 1 MiB signable-body cap (k_ra8_rot_body_max)
58TRAILER_BYTES = 5 * 4 + DIGEST_BYTES + SIG_BYTES # 116
59INT_BYTES = 32 # P-256 field element / half-signature width
60
61# ASN.1 DER / SEC1 tag bytes used when (de)serialising the openssl signature.
62DER_SEQUENCE_TAG = 0x30
63DER_INTEGER_TAG = 0x02
64DER_HIGH_BIT = 0x80
65EC_UNCOMPRESSED_TAG = 0x04

90def _der_int(buf: bytes, off: int) -> tuple[int, int]:
91 """Parse one DER INTEGER at ``buf[off]``; return (value_offset, value_len)."""
92 if buf[off] != DER_INTEGER_TAG:
93 _fail("malformed DER signature (expected INTEGER)")
94 length = buf[off + 1]
95 return off + 2, length

98def _der_sig_to_raw(der: bytes) -> bytes:
99 """Convert a DER ``SEQUENCE{INTEGER r, INTEGER s}`` to 64-byte r||s."""
100 if der[0] != DER_SEQUENCE_TAG:
101 _fail("malformed DER signature (expected SEQUENCE)")
102 # Skip the SEQUENCE tag + single-byte length (an ECDSA-P256 sig is < 128 B).
103 r_off, r_len = _der_int(der, 2)
104 s_off, s_len = _der_int(der, r_off + r_len)
105 r = int.from_bytes(der[r_off : r_off + r_len], "big")
106 s = int.from_bytes(der[s_off : s_off + s_len], "big")
107 return r.to_bytes(INT_BYTES, "big") + s.to_bytes(INT_BYTES, "big")
108

110def _raw_sig_to_der(r: int, s: int) -> bytes:
111 """Encode two integers as a DER ``SEQUENCE{INTEGER r, INTEGER s}``."""
112
113 def _int_enc(value: int) -> bytes:
114 data = value.to_bytes(INT_BYTES, "big").lstrip(b"\x00") or b"\x00"
115 if data[0] & DER_HIGH_BIT: # keep the ASN.1 integer positive
116 data = b"\x00" + data
117 return bytes([DER_INTEGER_TAG, len(data)]) + data
118
119 body = _int_enc(r) + _int_enc(s)
120 return bytes([DER_SEQUENCE_TAG, len(body)]) + body

123def keygen(key_path: Path) -> bytes:
124 """Generate a P-256 private key at ``key_path``; return the 65-byte pubkey."""
125 pem = _openssl(["ecparam", "-name", "prime256v1", "-genkey", "-noout"])
126 key_path.write_bytes(pem)
127 return public_key(key_path)

130def public_key(key_path: Path) -> bytes:
131 """Return the 65-byte uncompressed 0x04||X||Y public key for ``key_path``."""
132 der = _openssl(["ec", "-in", str(key_path), "-pubout", "-outform", "DER"])
133 # The SubjectPublicKeyInfo for P-256 ends in the 65-byte uncompressed point.
134 point = der[-PUBKEY_BYTES:]
135 if len(point) != PUBKEY_BYTES or point[0] != EC_UNCOMPRESSED_TAG:
136 _fail("could not extract uncompressed P-256 public key")
137 return point

140def sign_body(key_path: Path, body: bytes, img_version: int) -> bytes:
141 """Return ``body`` with an appended ra8_rot_trailer_t signed by ``key_path``."""
142 if len(body) == 0 or len(body) > BODY_MAX:
143 _fail(f"body length {len(body)} out of range (1..{BODY_MAX})")
144 digest = hashlib.sha256(body).digest()
145 # The signature authenticates SHA-256(img_version_le || body_digest), not the
146 # bare body digest, so a forged img_version cannot ride on a validly-signed
147 # body (T5-05). `openssl dgst -sha256 -sign` hashes its input, so signing the
148 # concatenation yields ECDSA(SHA-256(img_version_le || digest)) -- exactly
149 # what ra8_rot_verify_image recomputes via internal_bind_version. The trailer's
150 # own `digest` field still stores SHA-256(body) for the tamper pre-check.
151 to_sign = struct.pack("<I", img_version) + digest
152 with tempfile.NamedTemporaryFile() as sign_file:
153 sign_file.write(to_sign)
154 sign_file.flush()
155 der = _openssl(["dgst", "-sha256", "-sign", str(key_path), sign_file.name])
156 sig = _der_sig_to_raw(der)
157 header = struct.pack("<5I", ROT_MAGIC, ROT_VERSION, img_version, len(body), SIG_BYTES)
158 trailer = header + digest + sig
159 if len(trailer) != TRAILER_BYTES:
160 _fail(f"trailer size {len(trailer)} != {TRAILER_BYTES}")
161 return body + trailer

164def _pubkey_c_header(pubkey: bytes) -> str:
165 """Render the 65-byte pubkey as a C initialiser for s_rot_root_pubkey."""
166 rows = [
167 " " + ", ".join(f"0x{b:02X}U" for b in pubkey[i : i + 8]) + ","
168 for i in range(0, PUBKEY_BYTES, 8)
169 ]
170 body = "\n".join(rows)
171 return (
172 "/* Provisioned root public key -- paste into "
173 "libs/ra8_dfu/src/ra8_rot.c::s_rot_root_pubkey. */\n"
174 "static const uint8_t s_rot_root_pubkey[k_ra8_rot_pubkey_bytes] = {\n"
175 f"{body}\n}};\n"
176 )

179def cmd_keygen(args: argparse.Namespace) -> int:
180 """Run the `keygen` subcommand: mint a root keypair and emit its C header.
181
182 This is the key ceremony. The private key written to `--key` is the root of
183 the entire secure-boot chain: losing it means no future image can ever be
184 signed for a board already provisioned with the matching public key, and
185 leaking it means anyone can. It is written wherever the caller says, with no
186 passphrase and no permission hardening -- protecting it is the operator's
187 job, not this tool's.
188
189 The public half is emitted as a C array for
190 `libs/ra8_dfu/src/ra8_rot.c::s_rot_root_pubkey`. Without `--pubkey-c` it
191 goes to stdout, so the ceremony can be eyeballed before anything is pasted.
192
193 An existing `--key` path is overwritten without confirmation.
194
195 Args:
196 args: Parsed namespace with `key` and optional `pubkey_c`.
197
198 Returns:
199 0. Failures surface as exceptions from `keygen`, not a status code.
200 """
201 pubkey = keygen(Path(args.key))
202 sys.stdout.write(f"wrote private key: {args.key}\n")
203 if args.pubkey_c:
204 Path(args.pubkey_c).write_text(_pubkey_c_header(pubkey), encoding="ascii")
205 sys.stdout.write(f"wrote public-key C header: {args.pubkey_c}\n")
206 else:
207 sys.stdout.write(_pubkey_c_header(pubkey))
208 return 0

211def cmd_sign(args: argparse.Namespace) -> int:
212 """Run the `sign` subcommand: append a signed trailer to an image body.
213
214 Output is the input body followed by the 116-byte `ra8_rot_trailer_t`, so
215 `--out` is always exactly 116 bytes longer than `--image`. Signing is not
216 idempotent: feeding an already-signed image back in signs the body AND the
217 old trailer, producing a double-trailered image the verifier rejects. Always
218 sign the raw build artifact.
219
220 `--img-version` is the monotonic anti-rollback counter, and the device
221 refuses any image whose value is below what it has already accepted.
222 Signing with a number lower than one already deployed produces a
223 correctly-signed image that the target will still refuse -- and re-using a
224 number is what silently permits a rollback.
225
226 Args:
227 args: Parsed namespace with `key`, `image`, `out` and `img_version`.
228
229 Returns:
230 0. Failures surface as exceptions from `sign_body`.
231 """
232 body = Path(args.image).read_bytes()
233 signed = sign_body(Path(args.key), body, args.img_version)
234 Path(args.out).write_bytes(signed)
235 sys.stdout.write(f"signed {len(body)} body bytes -> {args.out} ({len(signed)} bytes)\n")
236 return 0

239def cmd_selftest(_args: argparse.Namespace) -> int:
240 """Run the `selftest` subcommand: a throwaway-key sign/verify round trip.
241
242 Proves this tool emits exactly the byte layout `ra8_rot_verify_image()`
243 parses -- magic, version, field widths and offsets, digest placement, and
244 that `img_version` survives the round trip. It checks the FORMAT contract,
245 not the crypto: openssl owns ECDSA correctness.
246
247 The keypair is generated into a temporary directory and destroyed with it,
248 so this touches no provisioned key and is safe to run anywhere. It needs
249 `openssl` on PATH and no hardware.
250
251 Failures exit non-zero from `_require` rather than returning, so the caller
252 never sees a false 0.
253
254 Args:
255 _args: Unused; the subcommand takes no options.
256
257 Returns:
258 0 when every check passes.
259 """
260 with tempfile.TemporaryDirectory() as tmp:
261 key = Path(tmp) / "k.pem"
262 pubkey = keygen(key)
263 _require(pubkey[0] == EC_UNCOMPRESSED_TAG, "pubkey point-format")
264 _require(len(pubkey) == PUBKEY_BYTES, "pubkey length")
265 body = hashlib.sha256(b"rot-selftest").digest() * 4 # 128 arbitrary bytes
266 signed = sign_body(key, body, img_version=7)
267 _require(len(signed) == len(body) + TRAILER_BYTES, "signed image length")
268
269 magic, ver, imgv, blen, slen = struct.unpack("<5I", signed[len(body) : len(body) + 20])
270 _require(magic == ROT_MAGIC, "trailer magic")
271 _require(ver == ROT_VERSION, "trailer version")
272 _require(imgv == 7, "img_version round-trip") # noqa: PLR2004
273 _require(blen == len(body), "body_len")
274 _require(slen == SIG_BYTES, "sig_len")
275 digest = signed[len(body) + 20 : len(body) + 20 + DIGEST_BYTES]
276 _require(digest == hashlib.sha256(body).digest(), "trailer digest == SHA-256(body)")
277
278 # Re-verify the raw r||s signature against the public key via openssl:
279 # rebuild the DER form and check ECDSA-P256 over the version-bound
280 # material SHA-256(img_version_le || digest) -- the same bytes
281 # ra8_rot_verify_image reconstructs. _openssl exits on a failed verify, so
282 # reaching the print means the signature is valid over that material.
283 signed_material_input = struct.pack("<I", imgv) + digest
284 raw = signed[len(body) + 20 + DIGEST_BYTES :]
285 der = _raw_sig_to_der(
286 int.from_bytes(raw[:INT_BYTES], "big"), int.from_bytes(raw[INT_BYTES:], "big")
287 )
288 with tempfile.TemporaryDirectory() as vtmp:
289 vdir = Path(vtmp)
290 (vdir / "pub.pem").write_bytes(_openssl(["ec", "-in", str(key), "-pubout"]))
291 (vdir / "body.bin").write_bytes(signed_material_input)
292 (vdir / "sig.der").write_bytes(der)
293 _openssl(
294 [
295 "dgst",
296 "-sha256",
297 "-verify",
298 str(vdir / "pub.pem"),
299 "-signature",
300 str(vdir / "sig.der"),
301 str(vdir / "body.bin"),
302 ]
303 )
304 sys.stdout.write("rot_sign.py selftest: PASS -- sign/verify round-trip + trailer layout OK.\n")
305 return 0

314def main() -> int:
315 """Parse the subcommand line and dispatch to the matching `cmd_*` handler.
316
317 Three subcommands -- `keygen`, `sign`, `selftest` -- and one is required, so
318 a bare invocation is an argparse error rather than a default action. That
319 is deliberate for a tool whose default could otherwise overwrite a root key.
320
321 Returns:
322 The handler's status, for `sys.exit`.
323 """
324 parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
325 sub = parser.add_subparsers(dest="cmd", required=True)
326
327 keygen_p = sub.add_parser("keygen", help="generate a P-256 root keypair")
328 keygen_p.add_argument("--key", required=True, help="output private-key PEM path")
329 keygen_p.add_argument("--pubkey-c", help="output C header with s_rot_root_pubkey")
330 keygen_p.set_defaults(func=cmd_keygen)
331
332 sign_p = sub.add_parser("sign", help="append a signed ra8_rot_trailer_t to an image")
333 sign_p.add_argument("--key", required=True, help="private-key PEM from keygen")
334 sign_p.add_argument("--image", required=True, help="raw image body .bin")
335 sign_p.add_argument("--out", required=True, help="output signed image path")
336 sign_p.add_argument("--img-version", type=int, default=1, help="anti-rollback version")
337 sign_p.set_defaults(func=cmd_sign)
338
339 selftest_p = sub.add_parser("selftest", help="sign+verify round-trip self-check")
340 selftest_p.set_defaults(func=cmd_selftest)
341
342 args = parser.parse_args()
343 return int(args.func(args))
