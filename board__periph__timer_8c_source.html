177static void agt_reg_write(uint32_t ch, uint64_t off, uint32_t value)
178{
179 agt_state_t* a = &s_agt[ch];
180 if (off == (uint64_t)k_agt_off_agt) {
181 a->counter = (uint16_t)value; /* AGT write both reloads and seeds count */
182 a->reload = (uint16_t)value;
183 } else if (off == (uint64_t)k_agt_off_cma) {
184 a->cmpa = (uint16_t)value;
185 } else if (off == (uint64_t)k_agt_off_cmb) {
186 a->cmpb = (uint16_t)value;
187 } else if (off == (uint64_t)k_agt_off_cr) {
188 /* TUNDF/TCMAF/TCMBF are write-0-to-clear (ra8_agt clears by writing 0);
189 * TSTART is RW. TCSTF tracks TSTART. */
190 const uint8_t status_keep =
191 (uint8_t)(a->cr & (uint8_t)value & (uint8_t)(k_agtcr_tundf | k_agtcr_tcmaf | k_agtcr_tcmbf));
192 const uint8_t start = (uint8_t)(value & (uint8_t)k_agtcr_tstart);
193 a->cr = (uint8_t)(start | (start != 0U ? (uint8_t)k_agtcr_tcstf : 0U) | status_keep);
194 } else if (off == (uint64_t)k_agt_off_mr1) {
195 a->mr1 = (uint8_t)value;
196 }
197}

200static void agt_tick_channel(uc_engine* uc, uint32_t ch)
201{
202 agt_state_t* a = &s_agt[ch];
203 if ((a->cr & (uint8_t)k_agtcr_tstart) == 0U) {
204 return; /* stopped: counter holds */
205 }
206 const uint32_t step = (uint32_t)k_agt_step_per_chunk;
207 if (a->counter > step) {
208 a->counter = (uint16_t)(a->counter - step);
209 return;
210 }
211 /* Underflow: reload and set TUNDF (and TCMAF if compare-match A is armed). */
212 const uint32_t span = (uint32_t)a->reload + 1U;
213 const uint32_t deficit = step - a->counter;
214 a->counter = (uint16_t)(a->reload - ((deficit - 1U) % span));
215 a->cr |= (uint8_t)k_agtcr_tundf;
216 a->underflows++;
217 if (ch == 0U) {
218 board_periph_icu_raise_event(uc, (uint16_t)k_event_agt0_int);
219 }
220}

265static void gpt_reg_write(uint32_t ch, uint64_t off, uint32_t value)
266{
267 gpt_state_t* g = &s_gpt[ch];
268 if (off == (uint64_t)k_gpt_off_gtcnt) {
269 g->cnt = value;
270 } else if (off == (uint64_t)k_gpt_off_gtpr) {
271 g->period = value;
272 } else if (off == (uint64_t)k_gpt_off_gtcr) {
273 g->cr = value; /* CST (bit0) starts / stops the count */
274 } else if (off == (uint64_t)k_gpt_off_gtstr) {
275 if ((value & 1U) != 0U) {
276 g->cr |= (uint32_t)k_gtcr_cst;
277 }
278 } else if (off == (uint64_t)k_gpt_off_gtstp) {
279 if ((value & 1U) != 0U) {
280 g->cr &= ~(uint32_t)k_gtcr_cst;
281 }
282 } else if (off == (uint64_t)k_gpt_off_gtclr) {
283 if ((value & 1U) != 0U) {
284 g->cnt = 0U;
285 }
286 } else if (off == (uint64_t)k_gpt_off_gtst) {
287 /* GTST bits are cleared by writing the value back with target bits zero
288 * (ra8_gpt_clear_status), so the model keeps only bits still set. */
289 g->st &= value;
290 }
291}

294static void gpt_tick_channel(uc_engine* uc, uint32_t ch)
295{
296 gpt_state_t* g = &s_gpt[ch];
297 if ((g->cr & (uint32_t)k_gtcr_cst) == 0U) {
298 return; /* stopped */
299 }
300 const uint32_t period = (g->period == 0U) ? (uint32_t)k_u16_max : g->period;
301 const uint32_t step = (uint32_t)k_gpt_step_per_chunk;
302 if ((g->cnt + step) <= period) {
303 g->cnt += step;
304 return;
305 }
306 /* Overflow past GTPR in saw mode: wrap and set TCFPO. */
307 g->cnt = (uint32_t)((g->cnt + step) - period - 1U);
308 g->st |= (uint32_t)k_gtst_tcfpo;
309 g->overflows++;
310 if (ch == 0U) {
311 board_periph_icu_raise_event(uc, (uint16_t)k_event_gpt0_ovf);
312 }
313}

361static void timer_report(void)
362{
363 for (uint32_t ch = 0U; ch < (uint32_t)k_agt_count; ch++) {
364 if (s_agt[ch].underflows > 0U) {
365 (void)fprintf(stderr,

371 }
372 }
373 for (uint32_t ch = 0U; ch < (uint32_t)k_gpt_count; ch++) {
374 if (s_gpt[ch].overflows > 0U) {
375 (void)fprintf(stderr,
376 " GPT%u : cnt=0x%08X period=0x%08X overflows=%u\n",
377 ch,
378 s_gpt[ch].cnt,
379 s_gpt[ch].period,
380 s_gpt[ch].overflows);
381 }
