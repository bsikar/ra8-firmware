117static uint32_t lcd_bringup_clocks(void)
118{
119 uint32_t cpuclk0_hz = 0U;
120 if (ra8_cgc_init() != k_ra8_ok) {
121 lcd_panic_halt();
122 }
123 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
124 lcd_panic_halt();
125 }
126 if (ra8_mstp_init() != k_ra8_ok) {
127 lcd_panic_halt();
128 }
129 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
130 lcd_panic_halt();
131 }
132 if (ra8_board_led_init(k_ra8_board_led_blue) != k_ra8_ok) {
133 lcd_panic_halt();
134 }
135 if (ra8_board_led_init(k_ra8_board_led_red) != k_ra8_ok) {
136 lcd_panic_halt();
137 }
138 ra8_isr_globals_enable();
139 return cpuclk0_hz;

160static void lcd_bringup_panel(void)
161{
162 /* Stabilization delays: PLLs, SDRAM, and the panel itself all need
163 * a few hundred ms after power-on to settle. Without these, the
164 * GLCDC sometimes starts before LCDCLK is stable and the panel
165 * comes up in its no-signal "white" state on cold boot. */
166 ra8_delay_ms(k_lcd_powerup_ms);
167
168 /* SDRAM is initialized so the framebuffer region at 0x68000000 is
169 * accessible for follow-on apps; this demo doesn't use it. */
170 if (ra8_sdramc_init() != k_ra8_ok) {
171 lcd_panic_halt();
172 }
173 ra8_delay_ms(k_lcd_sdram_settle_ms);
174
175 if (ra8_board_lcd_panel_power_on() != k_ra8_ok) {
176 lcd_panic_halt();
177 }
178 if (ra8_board_glcdc_init(k_ra8_board_glcdc_fmt_rgb888) != k_ra8_ok) {
179 lcd_panic_halt();
180 }
181 ra8_delay_ms(k_lcd_pin_settle_ms); /* let pins settle in output mode */
182
183 /* GLCDC: BG plane drives the panel on its own with both graphics
184 * layers held invisible by the driver, so the framebuffer pointer
185 * is never dereferenced -- leave it null. */
186 const ra8_glcdc_config_t cfg = {
187 .framebuffer_addr = 0UL,
188 .width_px = (uint16_t)k_lcd_panel_w,
189 .height_px = (uint16_t)k_lcd_panel_h,
190 .format = k_ra8_glcdc_fmt_rgb565,
191 .timing = k_ra8_panel_ek_ra8d2_timing,
192 };
193 if (ra8_glcdc_init(&cfg) != k_ra8_ok) {
194 lcd_panic_halt();
195 }
196 if (ra8_glcdc_set_background_color(k_bgc_red) != k_ra8_ok) {
197 lcd_panic_halt();
198 }
199 if (ra8_glcdc_start(true) != k_ra8_ok) {
200 lcd_panic_halt();
201 }
202}
