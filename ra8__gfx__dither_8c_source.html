100static uint8_t internal_quantise(uint8_t gray8, uint8_t thr)
101{
102 const uint8_t base = (uint8_t)(gray8 / (uint8_t)k_ra8_gfx_dither_step);
103 const uint8_t rem = (uint8_t)(gray8 - (uint8_t)(base * (uint8_t)k_ra8_gfx_dither_step));
104 uint8_t level = base;
105 if (((uint32_t)thr * (uint32_t)k_ra8_gfx_dither_step) <
106 ((uint32_t)rem * (uint32_t)k_ra8_gfx_dither_byte_levels)) {
107 level = (uint8_t)(level + 1U);
108 }
109 return level;
110}

166static void
167internal_pack_tile(const uint8_t* src, int32_t w, int32_t h, int32_t ox, int32_t oy, uint8_t* out)
168{
169 for (int32_t row = 0; row < h; ++row) {
170 for (int32_t col = 0; col < w; ++col) {
171 const uint32_t i = ((uint32_t)row * (uint32_t)w) + (uint32_t)col;
172 const uint8_t level =
173 internal_quantise(src[i], s_ra8_gfx_dither_mask[internal_mask_index(ox + col, oy + row)]);
174 const uint32_t byte_idx = i / (uint32_t)k_ra8_gfx_dither_ppb;
175 if ((i & 1U) == 0U) {
176 out[byte_idx] = (uint8_t)(level << (uint8_t)k_ra8_gfx_dither_nib_shift);
177 } else {
178 out[byte_idx] = (uint8_t)(out[byte_idx] | level);
179 }
180 }
181 }
182}

189ra8_err_t ra8_gfx_dither_gray8_to_gray4(const uint8_t* src,
190 int32_t w,
191 int32_t h,
192 int32_t origin_x,
193 int32_t origin_y,
194 uint8_t* out,
195 uint32_t out_cap,
196 uint32_t* out_size)
197{
198 static const char* const s_tag = "ra8_gfx_dither";
199 RA8_CHECK_NULL_PTR(src, s_tag, "src");
200 RA8_CHECK_NULL_PTR(out, s_tag, "out");
201 RA8_CHECK_NULL_PTR(out_size, s_tag, "out_size");
202
203 if ((w <= 0) || (h <= 0)) {
204 ra8_log_error(s_tag, "w or h is non-positive");
205 return k_ra8_err_invalid_arg;
206 }
207
208 const uint32_t n_pixels = (uint32_t)w * (uint32_t)h;
209 const uint32_t n_bytes = (n_pixels + 1U) / (uint32_t)k_ra8_gfx_dither_ppb;
210 if (out_cap < n_bytes) {
211 ra8_log_error(s_tag, "output buffer too small");
212 return k_ra8_err_no_mem;
213 }
214
215 internal_pack_tile(src, w, h, origin_x, origin_y, out);
216 *out_size = n_bytes;
217 return k_ra8_ok;

221ra8_gfx_blit_gray8_dither(const uint8_t* src, int32_t w, int32_t h, int32_t dst_x, int32_t dst_y)
222{
223 if (!s_gfx_text_state.initialized) {
224 return k_ra8_err_not_initialized;
225 }
226 if ((src == nullptr) || (w <= 0) || (h <= 0)) {
227 return k_ra8_err_invalid_arg;
228 }
229
230 for (int32_t row = 0; row < h; ++row) {
231 for (int32_t col = 0; col < w; ++col) {
232 const uint8_t level =
233 internal_quantise(src[((size_t)row * (size_t)w) + (size_t)col],
234 s_ra8_gfx_dither_mask[internal_mask_index(dst_x + col, dst_y + row)]);
235 s_gfx_text_plot(dst_x + col, dst_y + row, internal_level_to_color(level));
236 }
237 }
238 return k_ra8_ok;
