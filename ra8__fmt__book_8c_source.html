99static ra8_err_t priv_walk_table(const ra8_fmt_blob_t* src,
100 size_t table_off,
101 uint32_t count,
102 size_t payload,
103 const ra8_fmt_opts_t* opts)
104{
105 const size_t need = table_off + (((size_t)count + 1U) * sizeof(uint64_t));
106 if (need > src->len) {
107 (void)fprintf(opts->report, " ANOMALY offset table runs past the file end\n");
108 return k_ra8_err_validation_failed;
109 }
110 ra8_err_t verdict = k_ra8_ok;
111 uint64_t prev = 0U;
112 if (opts->verbose) {
113 (void)fprintf(opts->report, " entry start end length\n");
114 }
115 for (uint32_t i = 0U; i <= count; ++i) { /* bounded by count + 1 */
116 const uint64_t off = priv_rd_u64(&src->bytes[table_off + ((size_t)i * sizeof(uint64_t))]);
117 if ((i == 0U) && (off != 0U)) {
118 (void)fprintf(opts->report,
119 " ANOMALY offset[0] is %llu, must be 0\n",
120 (unsigned long long)off);
121 verdict = k_ra8_err_validation_failed;
122 }
123 /* RBKC specifies a STRICTLY increasing table, so an offset equal to its
124 * predecessor is as invalid as one that goes backwards: it declares a
125 * zero-length stream that no reader can inflate. Testing `<=` rather than
126 * `<` is what makes a table rewritten to a constant fail here. */
127 if ((i > 0U) && (off <= prev)) {
128 (void)fprintf(opts->report,
129 " ANOMALY offset[%u]=%llu does not exceed its predecessor %llu -- "
130 "entries %s\n",
131 (unsigned)i,
132 (unsigned long long)off,
133 (unsigned long long)prev,
134 (off < prev) ? "OVERLAP" : "are ZERO-LENGTH");
135 verdict = k_ra8_err_validation_failed;
136 }
137 if ((i > 0U) && opts->verbose && (i <= (uint32_t)k_ra8_fmt_max_dump_rows)) {
138 (void)fprintf(opts->report,
139 " %5u %10llu %10llu %10llu\n",
140 (unsigned)(i - 1U),
141 (unsigned long long)(payload + prev),
142 (unsigned long long)(payload + off),
143 (unsigned long long)(off - prev));
144 }
145 prev = off;
146 }
147 const uint64_t payload_len = (uint64_t)(src->len - payload);
148 if (prev != payload_len) {
149 (void)fprintf(opts->report,
150 " ANOMALY final offset %llu != payload length %llu (%s)\n",
151 (unsigned long long)prev,
152 (unsigned long long)payload_len,
153 (prev < payload_len) ? "trailing bytes" : "TRUNCATED");
154 verdict = k_ra8_err_validation_failed;
155 }
156 return verdict;
157}

250ra8_err_t ra8_fmt_rabook_inspect(const ra8_fmt_blob_t* src, const ra8_fmt_opts_t* opts)
251{
252 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
253 RA8_CHECK_NULL_PTR(opts, s_tag, "opts must not be nullptr");
254 if (src->len < (size_t)k_ra8_book_container_header_len) {
255 (void)fprintf(opts->report, "RBKC too short to hold a header\n");
256 return k_ra8_err_invalid_size;
257 }
258 const uint32_t chunk_bytes = ra8_fmt_rd_u32(&src->bytes[k_fmt_bk_off_chunk_bytes]);
259 const uint64_t inflated = priv_rd_u64(&src->bytes[k_fmt_bk_off_inflated]);
260 const uint32_t chunks = ra8_fmt_rd_u32(&src->bytes[k_fmt_bk_off_chunk_count]);
261 const uint32_t reserved = ra8_fmt_rd_u32(&src->bytes[k_fmt_bk_off_reserved]);
262 priv_print_rabook_hdr(opts->report, src->len, chunk_bytes, inflated, chunks, reserved);
263
264 ra8_err_t verdict = k_ra8_ok;
265 if (!priv_hdr_sanity(opts->report, reserved, chunk_bytes, "chunk_bytes", &verdict)) {
266 return verdict;
267 }
268 const uint64_t expect = (inflated + (uint64_t)chunk_bytes - 1U) / (uint64_t)chunk_bytes;
269 if ((uint64_t)chunks != expect) {
270 (void)fprintf(opts->report,
271 " ANOMALY chunk_count %u != ceil(inflated/chunk_bytes) = %llu\n",
272 (unsigned)chunks,
273 (unsigned long long)expect);
274 verdict = k_ra8_err_validation_failed;
275 }
276 const size_t table = (size_t)k_ra8_book_container_header_len;
277 const size_t payload = table + (((size_t)chunks + 1U) * sizeof(uint64_t));
278 const ra8_err_t trc = priv_walk_table(src, table, chunks, payload, opts);
279 if (trc != k_ra8_ok) {
280 verdict = trc;
281 }
282 (void)fprintf(opts->report,
283 "verdict: %s\n",
284 (verdict == k_ra8_ok) ? "VALID (chunk table monotonic and complete)"
285 : "INVALID (see anomalies above)");
286 return verdict;
287}
