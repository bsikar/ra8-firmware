131static bool internal_format_decode(ra8_ssie_format_t format,
132 uint8_t* out_omod,
133 uint8_t* out_frm,
134 uint8_t* out_pdta,
135 uint8_t* out_sdta)
136{
137 /* Defaults map to I2S (OMOD=00, FRM=00, PDTA=0, SDTA=0). */
138 *out_omod = k_ra8_ssie_omod_i2s;
139 *out_frm = k_ra8_ssie_frm_default;
140 *out_pdta = 0U;
141 *out_sdta = 0U;
142
143 switch (format) {
144 case k_ra8_ssie_format_i2s:
145 case k_ra8_ssie_format_left_just:
146 /* I2S and left-justified both map to OMOD=I2S with SDTA=0
147 * (already default). Shown in HUM Ch 46.3.1 "Stereo Format"
148 * Figures, p 3094-3096. */
149 return true;
150 case k_ra8_ssie_format_right_just:
151 /* Right-justified maps to OMOD=I2S with PDTA=1 (right-justify
152 * placement) per HUM Ch 46.2.1 "PDTA" p 3056. */
153 *out_pdta = 1U;
154 return true;
155 case k_ra8_ssie_format_monaural:
156 *out_omod = k_ra8_ssie_omod_monaural;
157 *out_frm = k_ra8_ssie_frm_default;
158 return true;
159 case k_ra8_ssie_format_tdm_4:
160 *out_omod = k_ra8_ssie_omod_tdm;
161 *out_frm = k_ra8_ssie_frm_alt1;
162 return true;
163 case k_ra8_ssie_format_tdm_6:
164 *out_omod = k_ra8_ssie_omod_tdm;
165 *out_frm = k_ra8_ssie_frm_alt2;
166 return true;
167 case k_ra8_ssie_format_tdm_8:
168 *out_omod = k_ra8_ssie_omod_tdm;
169 *out_frm = k_ra8_ssie_frm_alt3;
170 return true;
171 default:
172 return false;
173 }
174}

245{

247 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3057 */

249 ssicr |= k_ra8_ssie_mask_mst;
250 }
251 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3056 */
252 ssicr |= ((uint32_t)cfg->bclk_div << k_ra8_ssie_bit_ckdv0) & k_ra8_ssie_mask_ckdv;
253 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3057 */
254 ssicr |= ((uint32_t)cfg->system_word << k_ra8_ssie_bit_swl0) & k_ra8_ssie_mask_swl;
255 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3057 */
256 ssicr |= ((uint32_t)cfg->data_word << k_ra8_ssie_bit_dwl0) & k_ra8_ssie_mask_dwl;
257 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3057 */
258 ssicr |= ((uint32_t)frm << k_ra8_ssie_bit_frm0) & k_ra8_ssie_mask_frm;
259 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3056 */
260 if (cfg->long_frame) {
261 ssicr |= k_ra8_ssie_mask_del;
262 }
263 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3056 */
264 if (pdta != 0U) {
265 ssicr |= k_ra8_ssie_mask_pdta;
266 }
267 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3056 */
268 if (sdta != 0U) {
269 ssicr |= k_ra8_ssie_mask_sdta;
270 }
271 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3057 */
272 if (cfg->spdp_high) {
273 ssicr |= k_ra8_ssie_mask_spdp;

661ra8_err_t ra8_ssie_start(uint8_t channel, ra8_ssie_dir_t dir)
662{
663 if (dir != k_ra8_ssie_dir_rx && dir != k_ra8_ssie_dir_tx && dir != k_ra8_ssie_dir_tx_rx) {
664 return k_ra8_err_invalid_arg;
665 }
666 volatile r_ssie_regs_t* reg = ra8_ssie_internal_regs(channel);
667 if (reg == nullptr) {
668 return k_ra8_err_invalid_arg;
669 }
670
671 /* HUM Ch 46.6.2 "Transmission" p 3098 / 46.6.3 "Reception" p 3099:
672 * the SSIE must be idle before REN/TEN may be set. */
673 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3056 */
674 uint32_t ssicr = reg->SSICR;
675 const uint32_t current_ren_ten = ssicr & k_ra8_ssie_mask_ren_ten;
676 const uint32_t desired_ren_ten = (uint32_t)dir;
677 if (desired_ren_ten == (current_ren_ten & desired_ren_ten)) {
678 return k_ra8_ok; /* Already enabled, nothing to do. */
679 }
680 if (current_ren_ten != 0U) {
681 return k_ra8_err_busy;
682 }
683 /* HUM Ch 46.2.2 "SSISR : Status Register" p 3066 */
684 if ((reg->SSISR & k_ra8_ssie_mask_iirq) == 0U) {
685 return k_ra8_err_busy;
686 }
687
688 /* HUM Ch 46.2.3 "SSIFCR : FIFO Control Register" p 3077 */
689 uint32_t ssifcr = reg->SSIFCR;
690 const ra8_err_t rst_err = internal_pulse_fifo_reset(reg, ssifcr);
691 RA8_RETURN_ON_ERROR(rst_err, s_tag, "ssie_start: FIFO reset stuck"); /* GCOVR_EXCL_BR_LINE */
692
693 /* HUM Ch 46.2.1 "SSICR" p 3057 */ /* / HUM Ch 46.2.3 "SSIFCR" p 3077 */
694 internal_apply_dir_irq_bits(desired_ren_ten, &ssicr, &ssifcr);
695
696 reg->SSIFCR = ssifcr;
697 /* HUM Ch 46.2.1 "SSICR : Control Register" p 3056 */
698 reg->SSICR = ssicr | desired_ren_ten;
699 return k_ra8_ok;
700}

801 uint16_t samples,
802 uint16_t* out_written)
803{
804 RA8_CHECK_NULL_PTR(buffer, s_tag, "buffer must not be nullptr");
805 RA8_CHECK_NULL_PTR(out_written, s_tag, "out_written must not be nullptr");
806 volatile r_ssie_regs_t* reg = ra8_ssie_internal_regs(channel);
807 if (reg == nullptr) {

812 for (uint16_t i = 0U; i < samples; i++) {
813 /* HUM Ch 46.2.4 "SSIFSR : FIFO Status Register" p 3083 */
814 const uint8_t tdc = (uint8_t)((reg->SSIFSR & k_ra8_ssie_mask_tdc) >> k_ra8_ssie_shift_tdc);

816 break;
817 }
818 /* HUM Ch 46.2.5 "SSIFTDR : Transmit FIFO Data Register" p 3088 */

822 if (written > 0U) {
823 /* HUM Ch 46.2.4 "SSIFSR : FIFO Status Register" p 3083 */
