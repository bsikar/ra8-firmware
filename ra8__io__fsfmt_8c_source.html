139RA8_INTERNAL
140static bool fat_probe(const ra8_fs_backend_t* be)
141{
142 uint8_t blk[(size_t)k_ra8_io_block_size_bytes];
143 if (!internal_read_boot0(be, blk)) {
144 return false;
145 }
146 if (internal_is_exfat(blk)) {
147 return false;
148 }
149 if (blk[(uint32_t)k_off_boot_sig0] != (uint8_t)k_boot_sig0) {
150 return false;
151 }
152 if (blk[(uint32_t)k_off_boot_sig1] != (uint8_t)k_boot_sig1) {
153 return false;
154 }
155 return true;

224ra8_err_t ra8_io_fsfmt_register(const ra8_io_fsfmt_t* fmt)
225{
226 RA8_CHECK_NULL_PTR(fmt, s_tag, "fmt must not be nullptr");
227 RA8_CHECK_NULL_PTR(fmt->name, s_tag, "fmt->name must not be nullptr");
228 RA8_CHECK_NULL_PTR(fmt->probe, s_tag, "fmt->probe must not be nullptr");
229 if (s_count >= (uint32_t)k_ra8_io_fsfmt_max) {
230 return k_ra8_err_no_mem;
231 }
232 s_reg[s_count] = fmt;
233 s_count++;
234 return k_ra8_ok;
235}

237ra8_err_t ra8_io_fsfmt_probe(const ra8_fs_backend_t* backend, const ra8_io_fsfmt_t** out)
238{
239 RA8_CHECK_NULL_PTR(backend, s_tag, "backend must not be nullptr");
240 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
241 for (uint32_t i = 0; i < s_count; ++i) {
242 if (s_reg[i]->probe(backend)) {
243 *out = s_reg[i];
244 return k_ra8_ok;
245 }
246 }
247 return k_ra8_err_not_found;
