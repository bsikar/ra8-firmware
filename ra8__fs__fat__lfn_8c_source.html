22/* ===========================================================================
23 * VFAT long-filename (LFN) read support
24 *
25 * A long name is stored as a chain of attr-0x0F entries IMMEDIATELY before the
26 * 8.3 short entry, in reverse order: the entry tagged 0x40 (last logical group)
27 * is physically first, then group N-1 ... group 1, then the 8.3 entry. Each LFN
28 * entry carries 13 UTF-16LE chars (offsets 1/3/5/7/9, 14/16/18/20/22/24, 28/30)
29 * and a checksum of the 8.3 name (offset 13) that ties the chain to its entry.
30 * This is READ-only: 8.3 creation/writes are unchanged (see #101).
31 * ===========================================================================
32 */

104void priv_lfn_add(lfn_state_t* s, const uint8_t* ent)
105{
106 static const uint8_t k_off[k_lfn_chars_per_ent] = {(uint8_t)k_lfn_char_off_0,
107 (uint8_t)k_lfn_char_off_1,
108 (uint8_t)k_lfn_char_off_2,
109 (uint8_t)k_lfn_char_off_3,
110 (uint8_t)k_lfn_char_off_4,
111 (uint8_t)k_lfn_char_off_5,
112 (uint8_t)k_lfn_char_off_6,
113 (uint8_t)k_lfn_char_off_7,
114 (uint8_t)k_lfn_char_off_8,
115 (uint8_t)k_lfn_char_off_9,
116 (uint8_t)k_lfn_char_off_10,
117 (uint8_t)k_lfn_char_off_11,
118 (uint8_t)k_lfn_char_off_12};
119 const uint32_t order = (uint32_t)(ent[k_lfn_off_seq] & (uint8_t)k_lfn_seq_order_mask);
120 if ((order < 1U) || (order > (uint32_t)k_lfn_max_entries)) {
121 return; /* out-of-range sequence -> corrupt chain, ignore this entry */
122 }
123 s->checksum = ent[k_lfn_off_checksum];
124 s->have = 1U;
125 const uint32_t base = (order - 1U) * (uint32_t)k_lfn_chars_per_ent;
126 for (uint32_t i = 0U; i < (uint32_t)k_lfn_chars_per_ent; i++) {
127 const uint32_t off = (uint32_t)k_off[i];
128 const uint32_t val = (uint32_t)ent[off] | ((uint32_t)ent[off + 1U] << 8U);
129 const uint32_t pos = base + i;
130 if (pos >= ((uint32_t)k_lfn_name_cap - 1U)) {
131 /* Unreachable: max pos = (k_lfn_max_entries-1)*k_lfn_chars_per_ent +
132 * (k_lfn_chars_per_ent-1) = 18*13+12 = 246 < k_lfn_name_cap-1 = 255. */
133 break; /* GCOVR_EXCL_LINE */
134 }
135 if ((val == 0U) || (val == (uint32_t)k_lfn_unicode_pad)) {
136 s->name[pos] = '\0'; /* terminator / padding ends this group's name */
137 break;
138 }
139 s->name[pos] =
140 (char)((val <= (uint32_t)k_lfn_ascii_max) ? (unsigned char)val : (unsigned char)'?');
141 }
142}

238static ra8_fs_lfn_scan_t priv_dir_find_long_sector(const char* needle,
239 const uint8_t* buf,
240 uint32_t cur_lba,
241 lfn_state_t* lfn,
242 uint32_t* out_lba,
243 uint32_t* out_entry_off,
244 uint8_t out_entry[k_ra8_fs_dir_entry_bytes])
245{
246 for (uint32_t e = 0; e < k_dir_entries_per_sector; e++) {
247 const uint8_t* ent = &buf[(size_t)e * (size_t)k_ra8_fs_dir_entry_bytes];
248 if (ent[k_dir_off_name] == k_dir_marker_free_perm) {
249 return k_lfn_scan_eod;
250 }
251 if (ent[k_dir_off_name] == k_dir_marker_free_used) {
252 priv_lfn_reset(lfn); /* a deleted slot breaks the chain */
253 continue;
254 }
255 if (ent[k_dir_off_attr] == k_ra8_fs_attr_lfn) {
256 priv_lfn_add(lfn, ent);
257 continue;
258 }
259 const char* lname = priv_lfn_name_for(lfn, ent);
260 if ((lname != nullptr) && (priv_name_ieq(needle, lname) != 0U)) {
261 *out_lba = cur_lba;
262 *out_entry_off = e * (uint32_t)k_ra8_fs_dir_entry_bytes;
263 priv_byte_copy(out_entry, ent, k_ra8_fs_dir_entry_bytes);
264 return k_lfn_scan_found;
265 }
266 priv_lfn_reset(lfn); /* 8.3 entry consumed -> next chain starts fresh */
267 }
268 return k_lfn_scan_continue;
269}

272ra8_err_t priv_dir_find_long(const ra8_fs_mount_t* m,
273 const dir_loc_t* loc,

342/* `priv_free_chain()`: see header for the documented contract. */
343ra8_err_t priv_free_chain(const ra8_fs_mount_t* m, uint32_t start)
344{
345 uint32_t cur = start;
346 uint32_t guard = 0;
347 /* mcdc-deactivated: loop bound; `cur < k_cluster_first_data` only on a corrupt FAT chain (the first-cluster reservation 0/1 is enforced at allocation). */
348 while (cur >= k_cluster_first_data && (cur - k_cluster_first_data) < m->count_of_clusters) {
349 uint32_t next = 0;
350 ra8_err_t err = priv_fat_get(m, cur, &next);
351 if (err != k_ra8_ok) {
352 return err;
353 }
354 err = priv_fat_set(m, cur, k_cluster_free);
355 if (err != k_ra8_ok) {
356 return err;
357 }
358 if (priv_is_eoc(m, next) != 0U) {
359 break;
360 }
361 cur = next;
362 /* Bounded loop -- can't visit more clusters than exist. */
363 guard++;
364 if (guard > m->count_of_clusters) {
365 return k_ra8_err_protocol_error;
366 }
367 }
368 return k_ra8_ok;
