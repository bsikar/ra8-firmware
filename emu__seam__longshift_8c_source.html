58/* Armv8.1-M long-shift family (LSLL / LSRL / ASRL), both the immediate form
59 * (`lsll r4, r5, #7`) and the register form (`lsll r0, r1, ip`).
60 *
61 * Encoding (T1, two halfwords, little-endian), verified by assembling every
62 * member with `arm-none-eabi-as -march=armv8.1-m.main+mve` and reading back the
63 * bytes -- the family is dense and three neighbouring groups alias it:
64 *
65 * hw1 = 0xEA5n, where n[3:1] = RdaLo (always an EVEN register) and
66 * n[0] = 1 selects the SATURATING/ROUNDING subfamily
67 * (UQSHLL / URSHRL / SRSHRL / SQSHLL /
68 * UQRSHLL / SQRSHRL)
69 * hw2[11:8] = RdaHi, except 0b1111 which selects the 32-bit (non-long)
70 * UQRSHL / SQRSHR
71 * hw2[3:0] = 0b1111 -> immediate form, imm5 = hw2[14:12]:hw2[7:6]
72 * (imm5 == 0 encodes a shift of 32)
73 * = 0b1101 -> register form, Rm = hw2[15:12], hw2[7:6] == 0
74 * op = hw2[5:4] (00 = LSLL, 01 = LSRL, 10 = ASRL)
75 *
76 * The two exclusions matter and are not cosmetic. `uqshll r0, r1, #7` encodes
77 * as EA51 11CF: it shares the immediate form's 0b1111 tail and its op field, so
78 * a decoder that masks the low FOUR bits of hw1 as RdaLo (as this one once did)
79 * accepts it as `lsll r1, r1, #7` and computes an unsaturated shift into the
80 * wrong register -- silently wrong 64-bit arithmetic, which is the exact defect
81 * this seam exists to prevent. Likewise `uqrshl r0, r2` (EA50 2F0D) has the
82 * register form's 0b1101 tail and a clear hw1[0], and is excluded only by its
83 * RdaHi == 0b1111. Both subfamilies are therefore rejected here and reported at
84 * install time rather than emulated approximately. */

159static bool long_shift_decode(uint16_t hw1, uint16_t hw2, long_shift_insn_t* out)
160{
161 if ((hw1 & (uint16_t)k_lsh_hw1_mask) != (uint16_t)k_lsh_hw1_match) {
162 return false;
163 }
164 if ((hw1 & (uint16_t)k_lsh_hw1_sat_bit) != 0U) {
165 return false; /* UQSHLL / URSHRL / SRSHRL / SQSHLL / UQRSHLL / SQRSHRL. */
166 }
167 const uint32_t op = ((uint32_t)hw2 >> (uint32_t)k_lsh_op_shift) & (uint32_t)k_lsh_op_mask;
168 if (op > (uint32_t)k_lsh_op_asrl) {
169 return false; /* op == 0b11 is reserved -- not a long shift. */
170 }
171 const uint32_t rdahi = ((uint32_t)hw2 >> (uint32_t)k_lsh_rdahi_shift) & (uint32_t)k_lo4_mask;
172 if (rdahi == (uint32_t)k_lsh_rdahi_wide) {
173 return false; /* UQRSHL / SQRSHR -- the 32-bit, non-long forms. */
174 }
175 const uint32_t tail = (uint32_t)hw2 & (uint32_t)k_lsh_hw2_lo_mask;
176 *out = (long_shift_insn_t){
177 .rdalo = (uint32_t)hw1 & (uint32_t)k_lsh_rdalo_mask,
178 .rdahi = rdahi,
179 .op = op,
180 };
181 if (tail == (uint32_t)k_lsh_tail_imm) {
182 const uint32_t imm3 = ((uint32_t)hw2 >> (uint32_t)k_lsh_imm3_shift) & (uint32_t)k_lsh_imm3_mask;
183 const uint32_t imm2 = ((uint32_t)hw2 >> (uint32_t)k_lsh_imm2_shift) & (uint32_t)k_lsh_imm2_mask;
184 const uint32_t imm5 = (imm3 << (uint32_t)k_lsh_imm3_to_imm5) | imm2;
185 out->imm = (imm5 == 0U) ? (uint32_t)k_lsh_word_bits : imm5;
186 return true;
187 }
188 if (tail == (uint32_t)k_lsh_tail_reg) {
189 if (((uint32_t)hw2 & (uint32_t)k_lsh_reg_zero_msk) != 0U) {
190 return false; /* hw2[7:6] must be zero in the register form. */
191 }
192 if (out->op == (uint32_t)k_lsh_op_lsrl) {
193 return false; /* no LSRL register form exists in the ISA. */
194 }
195 out->by_reg = true;
196 out->rm = ((uint32_t)hw2 >> (uint32_t)k_lsh_rm_shift) & (uint32_t)k_lo4_mask;
197 return true;
198 }
199 return false;
200}

247static uint64_t long_shift_apply(uint64_t val, uint32_t op, int32_t shift)
248{
249 uint32_t left = (op == (uint32_t)k_lsh_op_lsll) ? 1U : 0U;
250 int32_t n = shift;
251 if (n < 0) {
252 n = -n;
253 left = (left != 0U) ? 0U : 1U;
254 }
255 const bool arith = (op == (uint32_t)k_lsh_op_asrl);
256 if (n >= (int32_t)k_lsh_pair_bits) {
257 if (left != 0U) {
258 return 0U;
259 }
260 return arith ? (uint64_t)((int64_t)val >> ((int32_t)k_lsh_pair_bits - 1)) : 0U;
261 }
262 if (left != 0U) {
263 return val << (uint32_t)n;
264 }
265 return arith ? (uint64_t)((int64_t)val >> (uint32_t)n) : (val >> (uint32_t)n);
266}

338{
339 for (uint32_t i = 0U; i < (uint32_t)k_lsh_pending_max; i++) {

345 return nullptr;

391static void long_shift_begin(uc_engine* uc, uint32_t address)
392{
393 uint8_t code[k_lsh_insn_len] = {};
394 if (uc_mem_read(uc, address, code, sizeof(code)) != UC_ERR_OK) {
395 return;
396 }
397 const uint16_t hw1 = (uint16_t)(code[0] | ((uint16_t)code[1] << (uint16_t)k_byte_bits));
398 const uint16_t hw2 = (uint16_t)(code[2] | ((uint16_t)code[3] << (uint16_t)k_byte_bits));
399 long_shift_insn_t insn = {};
400 if (!long_shift_decode(hw1, hw2, &insn)) {
401 return; /* scan false-positive at a never-executed address: leave it alone. */
402 }
403 uint32_t lo = 0U;
404 uint32_t hi = 0U;
405 (void)uc_reg_read(uc, k_arm_reg_id[insn.rdalo], &lo);
406 (void)uc_reg_read(uc, k_arm_reg_id[insn.rdahi], &hi);
407 const uint64_t val = ((uint64_t)hi << (uint64_t)k_lsh_word_bits) | (uint64_t)lo;
408 const uint64_t res = long_shift_apply(val, insn.op, long_shift_amount(uc, &insn));
409 uint32_t nzcv = 0U;
410 (void)uc_reg_read(uc, UC_ARM_REG_APSR_NZCV, &nzcv);
411 const uint32_t at = address + (uint32_t)k_lsh_insn_len;
412 long_shift_pending_t* slot = long_shift_slot(at, false);
413 if (slot == nullptr) {
414 return; /* table full: leave the core's (wrong) ORRS result rather than
415 * write back against the wrong instruction. Unreachable in practice
416 * -- it needs k_lsh_pending_max nested interrupts inside the
417 * one-instruction window. */
418 }
419 *slot = (long_shift_pending_t){
420 .active = true,
421 .at = at,
422 .rdalo = insn.rdalo,
423 .rdahi = insn.rdahi,
424 .lo = (uint32_t)res,
425 .hi = (uint32_t)(res >> (uint64_t)k_lsh_word_bits),
426 .nzcv = nzcv,
427 };
428}

488bool emulate_long_shift_reg(uc_engine* uc, uint32_t pc, const uint8_t code[4])
489{
490 const uint16_t hw1 = (uint16_t)(code[0] | ((uint16_t)code[1] << (uint16_t)k_byte_bits));
491 const uint16_t hw2 = (uint16_t)(code[2] | ((uint16_t)code[3] << (uint16_t)k_byte_bits));
492 long_shift_insn_t insn = {};
493 if (!long_shift_decode(hw1, hw2, &insn) || !insn.by_reg) {
494 return false; /* not ours: the immediate form never traps, it mis-executes. */
495 }
496 uint32_t lo = 0U;
497 uint32_t hi = 0U;
498 (void)uc_reg_read(uc, k_arm_reg_id[insn.rdalo], &lo);
499 (void)uc_reg_read(uc, k_arm_reg_id[insn.rdahi], &hi);
500 const uint64_t val = ((uint64_t)hi << (uint64_t)k_lsh_word_bits) | (uint64_t)lo;
501 const uint64_t res = long_shift_apply(val, insn.op, long_shift_amount(uc, &insn));
502 lo = (uint32_t)res;
503 hi = (uint32_t)(res >> (uint64_t)k_lsh_word_bits);
504 (void)uc_reg_write(uc, k_arm_reg_id[insn.rdalo], &lo);
505 (void)uc_reg_write(uc, k_arm_reg_id[insn.rdahi], &hi);
506 uint32_t next = pc + (uint32_t)k_lsh_insn_len;
507 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
508 return true;
509}

527static bool install_seg_hooks(uc_engine* uc,
528 const uint8_t* elf,
529 uint32_t p_offset,
530 uint32_t p_vaddr,
531 uint32_t p_filesz,
532 uint32_t* n_hooks)
533{
534 for (uint32_t off = 0U; (off + (uint32_t)k_lsh_insn_len) <= p_filesz; off += 2U) {
535 const uint8_t* p = elf + p_offset + off;
536 const uint16_t hw1 = (uint16_t)(p[0] | ((uint16_t)p[1] << (uint16_t)k_byte_bits));
537 const uint16_t hw2 = (uint16_t)(p[2] | ((uint16_t)p[3] << (uint16_t)k_byte_bits));
538 long_shift_insn_t insn = {};
539 if (!long_shift_decode(hw1, hw2, &insn)) {
540 continue;
541 }
542 if (insn.by_reg) {
543 /* The register form aliases to ORRS with Rm == SP, which the core refuses
544 * outright, so it arrives as a real undefined-instruction trap and is
545 * serviced by ::emulate_long_shift_reg off the seam dispatch chain. Only
546 * the immediate form -- which aliases to an ORRS the core happily
547 * MIS-EXECUTES -- has to be intercepted here. */
548 continue;
549 }
550 if (*n_hooks >= (uint32_t)k_lsh_sites_max) {
551 (void)fprintf(stderr, " long-shift seam: site cap %u reached\n", (unsigned)k_lsh_sites_max);
552 return false;
553 }
554 const uint64_t va = (uint64_t)p_vaddr + (uint64_t)off;
555 const uint64_t tail = va + (uint64_t)k_lsh_insn_len;
556 s_lsh_sites[s_lsh_site_count] = (uint32_t)va;
557 s_lsh_site_count++;
558 (void)uc_hook_add(uc,
559 &s_lsh_hooks[(size_t)(*n_hooks) * 2U],
560 UC_HOOK_CODE,
561 (void*)on_long_shift,
562 nullptr,
563 va,
564 va);
565 (void)uc_hook_add(uc,
566 &s_lsh_hooks[((size_t)(*n_hooks) * 2U) + 1U],
567 UC_HOOK_CODE,
568 (void*)on_long_shift,
569 nullptr,
570 tail,
571 tail);
572 (*n_hooks)++;
573 }
574 return true;
575}

601void long_shift_seam_install(uc_engine* uc, const uint8_t* elf, long len)
602{
603 if ((elf == nullptr) || (len < (long)k_elf_ehdr_size)) {
604 return;
605 }
606 uint32_t phoff = 0U;
607 (void)memcpy(&phoff, elf + (uint32_t)k_elf_e_phoff_off, 4);
608 const uint16_t phentsize = (uint16_t)(elf[42] | (elf[43] << 8));
609 const uint16_t phnum = (uint16_t)(elf[44] | (elf[45] << 8));
610 /* Bound the program-header table against the file before walking it: a
611 * malformed e_phoff/e_phnum would otherwise OOB-read past `elf`. load_elf has
612 * already validated the ELF/ARM magic, so this only guards a truncated image. */
613 if (((uint64_t)phoff + ((uint64_t)phnum * (uint64_t)phentsize)) > (uint64_t)len) {
614 return;
615 }
616 s_lsh_site_count = 0U;
617 (void)memset(s_lsh_pending, 0, sizeof(s_lsh_pending));
618 uint32_t n_hooks = 0U;
619 for (uint16_t i = 0U; i < phnum; i++) {
620 const uint8_t* ph = elf + phoff + ((size_t)(uint32_t)i * phentsize);
621 uint32_t p_type = 0U;
622 uint32_t p_offset = 0U;
623 uint32_t p_vaddr = 0U;
624 uint32_t p_filesz = 0U;
625 uint32_t p_flags = 0U;
626 (void)memcpy(&p_type, ph + 0, 4);
627 (void)memcpy(&p_offset, ph + (uint32_t)k_elf_ph_offset_off, 4);
628 (void)memcpy(&p_vaddr, ph + (uint32_t)k_elf_ph_vaddr_off, 4);
629 (void)memcpy(&p_filesz, ph + (uint32_t)k_elf_ph_filesz_off, 4);
630 (void)memcpy(&p_flags, ph + (uint32_t)k_elf_ph_flags_off, 4);
631 if ((p_type != (uint32_t)k_elf_pt_load) || (p_filesz < (uint32_t)k_lsh_insn_len) ||
632 ((p_flags & (uint32_t)k_elf_pf_x) == 0U)) {
633 continue;
634 }
635 if (((uint64_t)p_offset + (uint64_t)p_filesz) > (uint64_t)len) {
636 continue; /* truncated / out-of-file segment -- skip. */
637 }
638 if (!install_seg_hooks(uc, elf, p_offset, p_vaddr, p_filesz, &n_hooks)) {
639 return;
640 }
641 }
642 if (n_hooks > 0U) {
643 (void)fprintf(stderr,
644 " long-shift seam: emulating %u Armv8.1-M LSLL/LSRL/ASRL site(s)\n",
645 (unsigned)n_hooks);
646 }
