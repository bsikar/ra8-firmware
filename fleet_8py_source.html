4"""Drive the CI fleet from its declaration.
5
6``infra/fleet.yml`` says what machines exist and how much of each one CI may
7use. This is the tool that acts on it -- listing, validating, generating the
8inventory, converging a host, and changing a host's live capacity without
9killing the jobs it is running.
10
11Every command names a HOST from the declaration, never a class or a hostname,
12so adding a machine really is adding a block:
13
14 fleet.py list what is declared, and how it is sized
15 fleet.py show <host> one host in full, with its derived vars
16 fleet.py validate the fleet-declaration gate's check
17 fleet.py inventory write infra/ansible/inventory/hosts.ini
18 fleet.py ssh-config [--install] the fleet's host aliases, for ~/.ssh
19 fleet.py ssh-target <host> the ssh command that reaches one host
20 fleet.py check <host> [play] dry run -- report, change nothing
21 fleet.py apply <host> [play] converge, draining first where needed
22 fleet.py remove <host> tear down (classes that implement it)
23 fleet.py status [host] what each host is running, right now
24 fleet.py scale <host> <n> live capacity change; shrinking DRAINS
25
26``list``, ``show``, ``validate``, ``inventory``, ``ssh-config``,
27``ssh-target`` and ``status`` are read-only and safe at any time (bar
28``ssh-config --install``, which writes under ``~/.ssh``). ``apply``, ``remove``
29and ``scale`` are not.
30
31Reachability is DERIVED, not assumed. Every host declares a literal address, a
32login user and an optional jump through another fleet host, so every command
33here works from a machine whose ``~/.ssh/config`` is empty. ``ssh-config``
34generates the friendly aliases FROM that declaration, which is what makes any
35machine a control node in one command instead of by hand-copying somebody's
36private config (#526).
37
38Why capacity is a first-class verb here rather than only an Ansible run: a
39converge on a container host recreates containers, which cancels whatever they
40are running. ``apply`` therefore drains the host first through the same
41mechanism ``scale`` uses, so re-provisioning is not a way to lose a job.
42"""

95def _run(argv: list[str], stdin: str | bytes | None = None, cwd: Path | None = None) -> int:
96 """Run a command, streaming its output, and return its status.
97
98 Args:
99 argv: Fully resolved argument vector.
100 stdin: Text (or bytes, for a tar stream) to feed the command. The
101 ``bash -s`` transport carries whole scripts this way, which keeps
102 them clear of both the Windows shell's quoting and the remote
103 shell's.
104 cwd: Directory to run in. Ansible needs ``infra/ansible``: both the
105 playbook paths and ``ansible.cfg`` are resolved relative to it.
106
107 Returns:
108 The command's exit status.
109 """
110 proc = subprocess.run( # noqa: S603 -- argv is built from the declaration, never a shell string
111 argv,
112 input=stdin,
113 text=not isinstance(stdin, bytes),
114 cwd=cwd,
115 check=False,
116 )
117 return proc.returncode

126def _push_to_wsl(data: dict[str, Any], name: str) -> int:
127 """Copy the playbooks and the capacity script into a WSL host's distro.
128
129 WSL has no SSH daemon of its own, and giving a personal daily-driver
130 machine one -- plus the inbound firewall hole to reach it -- is not a
131 reasonable ask, so the play is shipped in and run with
132 ``--connection=local``. This is the codified form of what was previously a
133 hand-run pipeline in the playbook's header comment.
134
135 Args:
136 data: The parsed declaration.
137 name: The ``docker_wsl`` host's fleet name.
138
139 Returns:
140 0 on success, the first failing step's status otherwise.
141 """
142 host = data["hosts"][name]
143 ssh = fr.ssh_target(data, name)
144 shell = fm.remote_shell(host)
145 prepare = f"set -eu\nrm -rf {WSL_STAGE}\nmkdir -p {WSL_STAGE}\n"
146 rc = _run([*ssh, shell], stdin=prepare)
147 if rc:
148 return rc
149 tar_tool = shutil.which("tar") or "tar"
150 tar = subprocess.run( # noqa: S603 -- fixed argv, paths from the checkout
151 [
152 tar_tool,
153 # macOS bsdtar attaches com.apple.provenance to every member, and
154 # GNU tar in the distro then warns once per file -- 37 lines of
155 # noise around the one line that matters. The archive is identical
156 # without them.
157 "--no-xattrs",
158 "-czf",
159 "-",
160 "-C",
161 str(fm.REPO_ROOT),
162 "infra/ansible",
163 "scripts/ci/fleet_capacity.sh",
164 ],
165 capture_output=True,
166 check=False,
167 )
168 if tar.returncode:
169 sys.stderr.write(tar.stderr.decode("utf-8", "replace"))
170 return tar.returncode
171 distro = host["connect"]["distro"]
172 unpack = f"wsl -d {distro} -u root -e tar -xzf - -C {WSL_STAGE}"
173 return _run([*ssh, unpack], stdin=tar.stdout)

176def _capacity(data: dict[str, Any], name: str, args: list[str]) -> int:
177 """Run ``fleet_capacity.sh`` on a host, over that host's transport.
178
179 The script is piped from the checkout on every call rather than invoked
180 from a copy on the host, so an operator command always runs the version in
181 the tree. The copy the ``fleet_capacity`` role installs exists for the
182 unattended quiet-hours timer, which has no checkout to read from.
183
184 Args:
185 data: The parsed declaration.
186 name: Fleet host name.
187 args: Arguments after the fixed configuration flags.
188
189 Returns:
190 The script's exit status.
191 """
192 host = _host(data, name)
193 cls = fm.CLASSES[host["class"]]
194 if cls.capacity_kind == "none":
195 return _fail(f"{name} is a {host['class']} host and carries no runners to scale")
196 flags = ["--kind", cls.capacity_kind]
197 if cls.capacity_kind == "docker":
198 if fm.docker_command(host) != "docker":
199 flags.append("--sudo")
200 for container in fm.container_names(host):
201 flags += ["--container", container]
202 # An operator scale-down must reach the dev slice for the same reason
203 # the timer's does: `make infra-scale HOST=win-ci N=0` is the "I want
204 # to play a game for an hour" command, and it buys the owner nothing
205 # while a gate suite in the slice still has the machine.
206 if host.get("dev_slice"):
207 flags += ["--dev-slice", fm.DEV_SLICE_UNIT]
208 else:
209 flags += ["--scale-set", host["runners"]["labels"][0]]
210 # Never a quoted argument: for the WSL host this line is parsed by Windows'
211 # shell before `wsl -e` sees it, and quoting does not survive that. The
212 # capacity script's flags are shaped so none is ever needed.
213 remote = f"{fm.remote_shell(host)} -- {' '.join(flags)} {' '.join(args)}"
214 return _run(
215 [*fr.ssh_target(data, name), remote],
216 stdin=CAPACITY_SCRIPT.read_text(encoding="utf-8"),
217 )

335def cmd_ssh_config(data: dict[str, Any], args: argparse.Namespace) -> int:
336 """Print, or install, the fleet's generated SSH config fragment.
337
338 ``--install`` is the whole of making a machine a control node: it writes
339 ``~/.ssh/ra8-fleet.config`` and puts one ``Include`` line at the TOP of
340 ``~/.ssh/config``. Top, because ssh takes the FIRST value it obtains for
341 each keyword -- so the declaration wins over a stale hand-written alias of
342 the same name, while any keyword the fragment does not set (an
343 ``IdentityFile``, a ``Port``) still comes from the operator's own block
344 below it.
345
346 Args:
347 data: The parsed declaration.
348 args: Parsed command line; uses ``args.install``.
349
350 Returns:
351 0 on success, non-zero when the fragment could not be written.
352 """
353 body = fr.render_ssh_config(data)
354 if not args.install:
355 print(body, end="")
356 return 0
357 ssh_dir = Path.home() / ".ssh"
358 ssh_dir.mkdir(mode=0o700, parents=True, exist_ok=True)
359 fragment = ssh_dir / fr.SSH_FRAGMENT_NAME
360 fragment.write_text(body, encoding="utf-8")
361 fragment.chmod(0o600)
362 print(f"wrote {fragment} ({len(data['hosts'])} host(s))")
363 print(_ensure_include(ssh_dir / "config"))
364 return 0

368 """Put the fragment's ``Include`` line at the top of an ssh config.

370 Args:

389 return f"prepended the include line to {config}"

442def _converge_wsl(data: dict[str, Any], name: str, plays: list[str], extra: list[str]) -> int:
443 """Run a WSL host's plays inside its distro.
444
445 Args:
446 data: The parsed declaration.
447 name: Fleet host name.
448 plays: Plays to run, in order.
449 extra: Extra ansible flags (dry-run, teardown state).
450
451 Returns:
452 0 on success, the first failing play's status otherwise.
453 """
454 host = data["hosts"][name]
455 rc = _push_to_wsl(data, name)
456 if rc:
457 return rc
458 variables = shlex.quote(json.dumps(fm.role_vars(name, host)))
459 # The stage keeps the checkout's own layout, so the play's relative paths
460 # (roles/, and fleet_capacity_src reaching back into scripts/) resolve
461 # exactly as they do on the control node.
462 lines = [
463 "set -eu",
464 f"cd {WSL_STAGE}/infra/ansible",
465 *(
466 "ansible-playbook --connection=local -i localhost, "
467 f"playbooks/{fm.PLAYS[play].playbook} -e {variables} "
468 f"-e wsl_ci_host_id={shlex.quote(name)} "
469 f"-e fleet_capacity_src={WSL_STAGE}/scripts/ci/fleet_capacity.sh "
470 f"{' '.join(extra)}"
471 for play in plays
472 ),
473 ]
474 return _run([*fr.ssh_target(data, name), fm.remote_shell(host)], stdin="\n".join(lines) + "\n")

515def cmd_converge(data: dict[str, Any], args: argparse.Namespace) -> int:
516 """Run a host's plays, either as a dry run or for real.
517
518 On a container host a converge recreates containers, which cancels running
519 jobs, so a real apply drains the host to zero instances first and restores
520 the declared count afterwards. ``--no-drain`` skips that when the caller
521 knows the host is idle; a dry run never drains, because it changes nothing.
522
523 Args:
524 data: The parsed declaration.
525 args: Parsed command line.
526
527 Returns:
528 0 on success, non-zero from the first step that fails.
529 """
530 host = _host(data, args.host)
531 plays = _plays_for(host, args.play)
532 refusal = _converge_refusal(args, host, plays)
533 if refusal:
534 return _fail(refusal)
535 rc = cmd_inventory(data, argparse.Namespace(stdout=False))
536 if rc:
537 return rc
538 # Some tag sets cannot stop, start or recreate a container -- `capacity`
539 # refreshes the drain script and the quiet-hours timer, `dev-slice` tunes a
540 # cgroup beside them. Draining the host for either would cost it every
541 # running job's worth of runner time to protect against a change that
542 # cannot touch them. The whitelist lives in fleet_model.NO_DRAIN_TAGS so
543 # adding a tag is a deliberate act with the rule in front of you.
544 no_drain_tags = args.tags in fm.NO_DRAIN_TAGS
545 drain = (
546 args.mode == "apply"
547 and not args.no_drain
548 and not no_drain_tags
549 and fm.container_names(host)
550 )
551 if drain:
552 print(f"==> draining {args.host} before converging (a converge recreates containers)")
553 # drain-all, not `scale 0`: a converge that changes the instance count
554 # across the 1 <-> N boundary also renames the containers, so the drain
555 # has to walk what is really on the host rather than what the
556 # declaration predicts.
557 rc = _capacity(data, args.host, ["drain-all"])
558 if rc:
559 return _fail("could not drain the host; refusing to converge over running jobs")
560 extra = (["--check", "--diff"] if args.mode == "check" else []) + _remove_flags(host, args)
561 # SHORT-LIVED credentials only, and preferably by file reference.
562 #
563 # Anything given as KEY=VALUE lands in this process's argv and in
564 # ansible-playbook's, where `ps` on the control node can read it. That is
565 # tolerable for a registration or removal token -- one hour, one use,
566 # minted on demand -- and it is NOT tolerable for a PAT. Ansible reads
567 # `-e @file` as well, so a long-lived credential goes in a mode-0600 file
568 # outside the checkout and is passed as `-e @/path/to/secrets.yml`.
569 for pair in args.extra_var:
570 extra += ["-e", pair]
571 if args.tags:
572 extra += ["--tags", args.tags]
573 if fm.CLASSES[host["class"]].transport == "wsl":
574 rc = _converge_wsl(data, args.host, plays, extra)
575 else:
576 rc = _converge_ssh(data, args.host, plays, extra)
577 if rc or args.mode == "remove":
578 return rc
579 if drain:
580 return _capacity(data, args.host, ["scale", str(host["runners"]["instances"])])
581 return 0

637 information, not a failure of the question.
638 """
639 names = [args.host] if args.host else list(data["hosts"])
640 for name in names:
641 host = _host(data, name)
642 if fm.CLASSES[host["class"]].capacity_kind == "none":

644 print(f"{name} ({host['class']}, declared {host['runners']['instances']} instance(s)):")
645 # Flushed before handing the terminal to ssh, or Python's buffer holds

652def cmd_reach(data: dict[str, Any], _args: argparse.Namespace) -> int:
653 """Probe every declared machine over its own transport.
654
655 The transport is the point: ``win-ci`` is not an ssh alias but a jump
656 through the bench Pi into a Windows box and then into a WSL distro, and a
657 reachability check that did not know that would report the fleet broken.
658 Because the list AND every address come from the declaration, a machine
659 added there is probed from the next run with nothing else edited, on a
660 control node with no ``~/.ssh/config`` at all.
661
662 Args:
663 data: The parsed declaration.
664 _args: Unused; the command takes no arguments.
665
666 Returns:
667 0 when every machine answered, 1 otherwise.
668 """
669 rc = 0
670 for name, host in data["hosts"].items():
671 # Captured, not streamed: the probe's own "ok" belongs to this function,
672 # not to the operator's terminal, and a failing host's ssh chatter would
673 # otherwise bury the one line that says which host failed.
674 probe = subprocess.run( # noqa: S603 -- argv built from the declaration
675 [*fr.ssh_target(data, name), fm.remote_shell(host)],
676 input="echo ok\n",
677 text=True,
678 capture_output=True,
679 check=False,
680 ).returncode
681 where = fr.ssh_destination(host)
682 if probe:
683 print(f" MISS {name:<10} not reachable at {where}")
684 rc = 1
685 else:
686 print(f" ok {name:<10} reachable at {where}")
687 return rc
