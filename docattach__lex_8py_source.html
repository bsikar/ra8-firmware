3"""The checks that need only the source text, not a parse.
4
5Two of this gate's seven findings are answerable lexically, and answering them
6without libclang is deliberate: DOC004 (two doc blocks in a row with nothing
7between them) and DOC007 (banned definition-site boilerplate) are statements
8about the *comment stream*, which a parse discards.
9
10Everything here therefore works on text that has had comments and literals
11blanked out but line structure preserved, so a brace inside a string or a

34def _blank_comments_and_literals(text: str) -> str:
35 """Blank every comment and string/char literal, preserving line structure.
36
37 Length and newline positions are preserved so the result can be indexed by
38 the same line numbers as the original. Blanking (rather than deleting) is
39 what lets the DOC004 "is there code between these two blocks" test look at
40 code only -- an earlier version advanced past comments without clearing
41 them, so a member's own trailing ``/**< ... */`` counted as intervening
42 code and the check silently never fired on real member runs.
43 """
44 out = list(text)
45
46 def blank(lo: int, hi: int) -> None:
47 for k in range(lo, min(hi, len(text))):
48 if text[k] != "\n":
49 out[k] = " "
50
51 i, n = 0, len(text)
52 while i < n:
53 ch = text[i]
54 if ch in {'"', "'"}:
55 quote = ch
56 start = i
57 i += 1
58 while i < n:
59 if text[i] == "\\" and i + 1 < n:
60 i += 2
61 continue
62 if text[i] == quote:
63 i += 1
64 break
65 i += 1
66 blank(start, i)
67 continue
68 if ch == "/" and i + 1 < n and text[i + 1] == "/":
69 start = i
70 while i < n and text[i] != "\n":
71 i += 1
72 blank(start, i)
73 continue
74 if ch == "/" and i + 1 < n and text[i + 1] == "*":
75 start = i
76 i += 2
77 while i + 1 < n and not (text[i] == "*" and text[i + 1] == "/"):
78 i += 1
79 i += 2
80 blank(start, i)
81 continue
82 i += 1
83 return "".join(out)

86def _skip_literal(text: str, i: int, line: int) -> tuple[int, int]:
87 """Advance past the string/char literal starting at ``i``."""
88 quote = text[i]
89 n = len(text)
90 i += 1
91 while i < n:
92 if text[i] == "\\" and i + 1 < n:
93 i += 2
94 continue
95 if text[i] == quote:
96 i += 1
97 break
98 if text[i] == "\n":
99 line += 1
100 i += 1
101 return i, line

104def _scan_block_comment(text: str, i: int, line: int) -> tuple[int, int, bool]:
105 """Advance past the ``/* ... */`` at ``i``; report whether it is a doc block."""
106 n = len(text)
107 is_doc = i + 2 < n and text[i + 2] in {"*", "!"}
108 # `/**/` and `/***/` are terminators, not documentation.
109 if is_doc and i + 3 < n and text[i + 2] == "*" and text[i + 3] == "/":
110 is_doc = False
111 j = i + 2
112 while j + 1 < n and not (text[j] == "*" and text[j + 1] == "/"):
113 if text[j] == "\n":
114 line += 1
115 j += 1
116 return j + 2, line, is_doc

119def extract_doc_blocks(text: str) -> list[DocBlock]:
120 """Return every ``/**`` / ``/*!`` block in ``text``, in source order.
121
122 Blocks inside string literals are skipped. Single-line ``///`` and ``//!``
123 comments are captured too so the duplicate check sees them.
124 """
125 blocks: list[DocBlock] = []
126 i, n = 0, len(text)
127 line = 1
128 while i < n:
129 ch = text[i]
130 if ch == "\n":
131 line += 1
132 i += 1
133 continue
134 if ch in {'"', "'"}:
135 i, line = _skip_literal(text, i, line)
136 continue
137 if ch == "/" and i + 1 < n and text[i + 1] == "*":
138 start_line = line
139 end, line, is_doc = _scan_block_comment(text, i, line)
140 if is_doc:
141 blocks.append(DocBlock(start_line, line, text[i:end]))
142 i = end
143 continue
144 if ch == "/" and i + 1 < n and text[i + 1] == "/":
145 is_doc = i + 2 < n and text[i + 2] in {"/", "!"}
146 chunk_start = i
147 while i < n and text[i] != "\n":
148 i += 1
149 if is_doc:
150 blocks.append(DocBlock(line, line, text[chunk_start:i]))
151 continue
152 i += 1
153 for b in blocks:
154 b.standalone = bool(STANDALONE_TAG_RE.search(b.text) or GROUP_MARKER_RE.search(b.text))
155 b.trailing = bool(re.match(r"/(?:\*[*!]|//|/!)<", b.text))
156 return blocks

159def check_consecutive_blocks(path: str, text: str) -> list[Finding]:
160 """DOC004 -- two doc blocks in a row with nothing declared between them.
161
162 This is the owner's literal complaint and the shape that makes a presence
163 gate *reward* the defect. A ``@file`` / ``@defgroup`` / ``@{`` block
164 followed by a real block is normal and is exempted via ``standalone``.
165 """
166 findings: list[Finding] = []
167 blocks = extract_doc_blocks(text)
168 lines = _blank_comments_and_literals(text).splitlines()
169 raw_lines = text.splitlines()
170 for prev, cur in zip(blocks, blocks[1:]):
171 if prev.standalone or prev.trailing or cur.trailing:
172 continue
173 # A block carrying an explicit @def / @fn / @struct / ... tag is
174 # resolved by Doxygen *by name*, not by position, so it does not need a
175 # following declaration at all. Config headers rely on this to
176 # document deliberately-disabled options:
177 # /** \def MBEDTLS_AES_ROM_TABLES ... */
178 # //#define MBEDTLS_AES_ROM_TABLES
179 # The commented-out define is not code, so a purely positional test
180 # calls that a duplicate. Accept the block when the very next non-blank
181 # source line mentions the symbol it names -- which still leaves a block
182 # whose symbol is declared further down, past another symbol, reported.
183 refs = [ref for _, ref in EXPLICIT_REF_RE.findall(prev.text)]
184 if refs:
185 following = next(
186 (ln for ln in raw_lines[prev.end_line : cur.start_line - 1] if ln.strip()),
187 "",
188 ) or next((ln for ln in raw_lines[prev.end_line :] if ln.strip()), "")
189 if any(re.search(r"\b" + re.escape(r) + r"\b", following) for r in refs):
190 continue
191 # A *commented-out* preprocessor directive is the config-header idiom
192 # for "here is an option, documented and deliberately off":
193 # /** Enable the Everest ECDH backend ... */
194 # //#define MBEDTLS_ECDH_VARIANT_EVEREST_ENABLED
195 # The block documents that option, so it is attached -- but the
196 # directive is lexically a comment, so `_blank_comments_and_literals`
197 # erases it and a purely positional test sees an empty gap and calls
198 # the block a duplicate. Consult the raw text for this one shape.
199 # Deliberately narrow: only a commented-out #define / #undef counts,
200 # so two genuinely adjacent blocks are still reported.
201 raw_between = raw_lines[prev.end_line : cur.start_line - 1]
202 if any(COMMENTED_DEFINE_RE.match(seg) for seg in raw_between):
203 continue
204 # Anything but whitespace between the two blocks means the first one is
205 # attached to something -- not a duplicate.
206 between = lines[prev.end_line : cur.start_line - 1]
207 if any(seg.strip() for seg in between):
208 continue
209 findings.append(
210 Finding(
211 path,
212 prev.start_line,
213 "DOC004",
214 "(block)",
215 f"doc block is followed by another doc block at line {cur.start_line} "
216 "with no declaration between them; the first documents nothing",
217 )
218 )
219 return findings

222def check_banned_boilerplate(path: str, text: str) -> list[Finding]:
223 """DOC007 -- CLAUDE.md's banned pointer-only definition-site comment.
224
225 Only fires on blocks whose *entire* informational content is the pointer.
226 A block that says "see the header for the full contract" and then adds a
227 real implementation note is allowed by CLAUDE.md, so the presence of any
228 other Doxygen tag or a ``--`` note clause clears it.
229 """
230 findings: list[Finding] = []
231 for block in extract_doc_blocks(text):
232 if not BANNED_BOILERPLATE_RE.search(block.text):
233 continue
234 body = re.sub(r"^\s*/\*+<?|\*+/\s*$", " ", block.text)
235 body = re.sub(r"^\s*\*", " ", body, flags=re.MULTILINE)
236 tags = set(re.findall(_T + r"([a-z]+)", body))
237 # @brief alone is still boilerplate; any other tag means real content.
238 if tags - {"brief"}:
239 continue
240 # Strip everything the idiom is *allowed* to contain -- the @brief tag,
241 # the "Implementation of <symbol>()" lead-in (backticked or not), the
242 # pointer phrase itself, and punctuation. Whatever survives is the
243 # real implementation note CLAUDE.md requires. If nothing survives,
244 # the comment says only "see the header" and is banned.
245 residue = BANNED_BOILERPLATE_RE.sub(" ", body)
246 residue = re.sub(_T + r"brief", " ", residue)
247 residue = re.sub(
248 r"[Ii]mplementation of\s+`?[A-Za-z_][A-Za-z_0-9]*`?(?:\s*\‍(\‍))?`?", " ", residue
249 )
250 residue = re.sub(r"[`(){}.,;:*/\s-]", "", residue)
251 if residue:
252 continue
253 findings.append(
254 Finding(
255 path,
256 block.start_line,
257 "DOC007",
258 "(block)",
259 "pointer-only definition-site boilerplate (CLAUDE.md bans it): the comment "
260 "carries no information the signature does not. Delete it, or replace it "
261 "with the single-line form carrying a real implementation note",
262 )
263 )
264 return findings
