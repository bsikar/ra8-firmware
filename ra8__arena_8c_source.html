69ra8_err_t ra8_arena_carve(ra8_arena_t* arena, uint32_t bytes, uint32_t align, void** out_ptr)
70{
71 RA8_CHECK_NULL_PTR(arena, s_tag, "arena must not be nullptr");
72 RA8_CHECK_NULL_PTR(out_ptr, s_tag, "out_ptr must not be nullptr");
73 if (bytes == 0U) {
74 return k_ra8_err_invalid_size;
75 }
76 if (!priv_is_pow2(align)) {
77 return k_ra8_err_invalid_arg;
78 }
79 const uintptr_t cur = (uintptr_t)arena->base + (uintptr_t)arena->used;
80 const uintptr_t mask = (uintptr_t)align - 1U;
81 const uintptr_t aligned = (cur + mask) & ~mask;
82 const uintptr_t end = (uintptr_t)arena->base + (uintptr_t)arena->size;
83 if (aligned > end) {
84 return k_ra8_err_no_mem;
85 }
86 if ((uintptr_t)bytes > (end - aligned)) {
87 return k_ra8_err_no_mem;
88 }
89 *out_ptr = (void*)aligned;
90 arena->used = (uint32_t)((aligned + (uintptr_t)bytes) - (uintptr_t)arena->base);
91 return k_ra8_ok;
92}
