49def _plan(argv: list[str]) -> list[str]:
50 """Translate one explicit plan output mode."""
51 if len(argv) != PLAN_ARGC:
52 fail("plan requires notes, mode, and output slots")
53 notes, mode, output = argv
54 result = ["plan", notes]
55 modes = {"summary": "--summary", "commands": "--emit-commands"}
56 if mode in modes:
57 result.append(modes[mode])

67def _start(argv: list[str]) -> list[str]:

97def build_argv(argv: list[str]) -> list[str]:
98 """Translate one fixed Just recipe shape into the public CLI argv."""
99 if not argv:
100 fail("missing Just action")
101 board_tail = BOARD_NO_ARG_ACTIONS.get(argv[0])
102 if board_tail is not None:
103 if len(argv) != 1:
104 fail(f"{argv[0]} takes no Just arguments")
105 return ["board", *board_tail]
106 board_view = BOARD_SELECTOR_ACTIONS.get(argv[0])
107 if board_view is not None:
108 if len(argv) != BOARD_SELECTOR_ARGC:
109 fail(f"{argv[0]} requires one argument")
110 return ["board", board_view, argv[1]]
111 action, values = argv[0], argv[1:]
112 adapters: dict[str, Adapter] = {
113 "doctor": _doctor,
114 "plan": _plan,
115 "start": _start,
116 "status": _status,
117 "ready": lambda items: _phase("ready", items),
118 "landed": lambda items: _phase("landed", items),
119 }
120 adapter = adapters.get(action)
121 if adapter is None:
122 fail(f"invalid Just action: {action}")
123 return adapter(values)
