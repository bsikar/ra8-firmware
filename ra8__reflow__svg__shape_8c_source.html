239priv_parse_points(const uint8_t* v, size_t vlen, const svg_xform_t* t, int32_t* xs, int32_t* ys)
240{
241 size_t k = 0U;
242 int32_t n = 0;
243 /* Bounded: <= k_svg_poly_max pairs; k advances past each number. */
244 while (n < (int32_t)k_svg_poly_max) {
245 while ((k < vlen) && (ra8_svgp_ws((char)v[k]) || (v[k] == ','))) {
246 ++k;
247 }

254 ++n;
255 }

283static void priv_sort_i32(int32_t* a, int32_t m)
284{
285 /* Bounded: m <= k_svg_poly_max. */
286 for (int32_t i = 1; i < m; ++i) {
287 const int32_t key = a[i];
288 int32_t j = i - 1;
289 while ((j >= 0) && (a[j] > key)) {
290 a[j + 1] = a[j];
291 --j;
292 }
293 a[j + 1] = key;

328static int32_t
329priv_scanline_x(const int32_t* xs, const int32_t* ys, int32_t n, int32_t y, int32_t* xint)
330{
331 int32_t m = 0;
332 /* Bounded: one test per polygon edge (n edges). */
333 /* mcdc-deactivated: the polygon point count n is itself capped at k_svg_poly_max by the point parser, so the crossing count m (<= edges processed <= i < n <= k_svg_poly_max) can never reach k_svg_poly_max before i exhausts n; (m < k_svg_poly_max) is invariantly true and its false arm is unreachable. */
334 for (int32_t i = 0; (i < n) && (m < (int32_t)k_svg_poly_max); ++i) {
335 const int32_t j = (i + 1) % n;
336 const int32_t y0 = ys[i];
337 const int32_t y1 = ys[j];
338 /* mcdc-deactivated: the fourth condition (y < y0) is the exact boolean negation of the first (y0 <= y), so it cannot be flipped independently of C1; the C4 independence pair is structurally unreachable. */
339 if (((y0 <= y) && (y < y1)) || ((y1 <= y) && (y < y0))) {
340 const int32_t x0 = xs[i];
341 const int32_t x1 = xs[j];
342 xint[m] = x0 + (int32_t)(((int64_t)(y - y0) * (int64_t)(x1 - x0)) / (int64_t)(y1 - y0));
343 ++m;
344 }
345 }
346 return m;
347}

423static uint32_t priv_col_lerp(uint32_t a, uint32_t b, float f)
424{
425 const int32_t ra = (int32_t)((a >> (uint32_t)k_svg_sh_r) & (uint32_t)k_svg_chan_mask);
426 const int32_t ga = (int32_t)((a >> (uint32_t)k_svg_hex_chan) & (uint32_t)k_svg_chan_mask);
427 const int32_t ba = (int32_t)(a & (uint32_t)k_svg_chan_mask);
428 const int32_t rb = (int32_t)((b >> (uint32_t)k_svg_sh_r) & (uint32_t)k_svg_chan_mask);
429 const int32_t gb = (int32_t)((b >> (uint32_t)k_svg_hex_chan) & (uint32_t)k_svg_chan_mask);
430 const int32_t bb = (int32_t)(b & (uint32_t)k_svg_chan_mask);
431 const int32_t r = ra + (int32_t)((float)(rb - ra) * f);
432 const int32_t g = ga + (int32_t)((float)(gb - ga) * f);
433 const int32_t bl = ba + (int32_t)((float)(bb - ba) * f);
434 return ((uint32_t)r << (uint32_t)k_svg_sh_r) | ((uint32_t)g << (uint32_t)k_svg_hex_chan) |
435 (uint32_t)bl;
436}

470static uint32_t priv_grad_eval(const svg_grad_t* g, float px, float py)
471{
472 if (g->nstops == 0U) {
473 return (uint32_t)k_svg_def_fill;
474 }
475 float p = 0.0F;
476 if (g->kind == (uint8_t)k_svg_grad_radial) {
477 const float dx = px - g->x1;
478 const float dy = py - g->y1;
479 const float rr = (g->x2 > 0.0F) ? g->x2 : 1.0F;
480 p = sqrtf((dx * dx) + (dy * dy)) / rr;
481 } else {
482 const float vx = g->x2 - g->x1;
483 const float vy = g->y2 - g->y1;
484 const float len2 = (vx * vx) + (vy * vy);
485 p = (len2 > 0.0F) ? ((((px - g->x1) * vx) + ((py - g->y1) * vy)) / len2) : 0.0F;
486 }
487 if (p <= g->stops[0].off) {
488 return g->stops[0].col;
489 }
490 const uint8_t last = (uint8_t)(g->nstops - 1U);
491 if (p >= g->stops[last].off) {
492 return g->stops[last].col;
493 }
494 /* Bounded: at most nstops-1 stop brackets. */
495 for (uint8_t i = 0U; i < last; ++i) {
496 const float o0 = g->stops[i].off;
497 const float o1 = g->stops[i + 1U].off;
498 /* mcdc-deactivated: the loop is entered only when p > stops[0].off, and reaching bracket i without returning requires p to exceed every earlier stop's upper offset, so p >= o0 (== stops[i].off) invariantly holds regardless of stop ordering; (p >= o0) cannot be flipped false. */
499 if ((p >= o0) && (p <= o1)) {
500 const float f = (o1 > o0) ? ((p - o0) / (o1 - o0)) : 0.0F;
501 return priv_col_lerp(g->stops[i].col, g->stops[i + 1U].col, f);
502 }
503 }
504 return g->stops[last].col;
505}

732RA8_INTERNAL
733static void priv_arc_solve(svg_arc_t* a,
734 float x1p,
735 float y1p,
736 float rx,
737 float ry,
738 float mx,
739 float my,
740 bool large,
741 bool sweep)
742{
743 const float rx2 = rx * rx;
744 const float ry2 = ry * ry;
745 const float num = (rx2 * ry2) - (rx2 * y1p * y1p) - (ry2 * x1p * x1p);
746 const float den = (rx2 * y1p * y1p) + (ry2 * x1p * x1p);
747 float co = sqrtf((num > 0.0F) ? (num / den) : 0.0F);
748 if (large == sweep) {
749 co = -co;
750 }
751 const float cxp = co * ((rx * y1p) / ry);
752 const float cyp = -co * ((ry * x1p) / rx);
753 a->cx = (a->cos_phi * cxp) - (a->sin_phi * cyp) + mx;
754 a->cy = (a->sin_phi * cxp) + (a->cos_phi * cyp) + my;
755 a->t1 = atan2f((y1p - cyp) / ry, (x1p - cxp) / rx);
756 float dt = atan2f((-y1p - cyp) / ry, (-x1p - cxp) / rx) - a->t1;
757 if (!sweep && (dt > 0.0F)) {
758 dt -= s_svg_2pi;
759 } else if (sweep && (dt < 0.0F)) {
760 dt += s_svg_2pi;
761 }
762 a->dt = dt;
763 a->rx = rx;
764 a->ry = ry;
765 a->ok = true;
766}

802RA8_INTERNAL
803static svg_arc_t priv_arc_center(svg_pt_t p0,
804 svg_pt_t p_end,
805 int32_t rx_in,
806 int32_t ry_in,
807 int32_t rot_deg,
808 bool large,
809 bool sweep)
810{
811 svg_arc_t a = {};
812 float rx = priv_absf((float)rx_in);
813 float ry = priv_absf((float)ry_in);
814 if ((rx == 0.0F) || (ry == 0.0F) || ((p0.x == p_end.x) && (p0.y == p_end.y))) {
815 a.ok = false;
816 return a;
817 }
818 const float phi = (float)rot_deg * (s_svg_pi / s_svg_deg_half);
819 a.cos_phi = cosf(phi);
820 a.sin_phi = sinf(phi);
821 const float dx = (float)(p0.x - p_end.x) * s_svg_half;
822 const float dy = (float)(p0.y - p_end.y) * s_svg_half;
823 const float x1p = (a.cos_phi * dx) + (a.sin_phi * dy);
824 const float y1p = -(a.sin_phi * dx) + (a.cos_phi * dy);
825 const float lam = ((x1p * x1p) / (rx * rx)) + ((y1p * y1p) / (ry * ry));
826 if (lam > 1.0F) {
827 const float s = sqrtf(lam);
828 rx *= s;
829 ry *= s;
830 }
831 const float mx = (float)(p0.x + p_end.x) * s_svg_half;
832 const float my = (float)(p0.y + p_end.y) * s_svg_half;
833 priv_arc_solve(&a, x1p, y1p, rx, ry, mx, my, large, sweep);
834 return a;
835}
