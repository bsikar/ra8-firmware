68static ra8_err_t
69internal_validate_dma_cfg(const ra8_ssie_dma_cfg_t* dma, bool* want_tx, bool* want_rx)
70{
71 *want_tx = (dma->tx_dma_channel < k_ra8_ssie_dma_max_ch);
72 *want_rx = (dma->rx_dma_channel < k_ra8_ssie_dma_max_ch);
73 if (!*want_tx && !*want_rx) {
74 return k_ra8_err_invalid_arg;
75 }
76 if (*want_tx && (dma->tx_buffer == nullptr || dma->tx_samples == 0U)) {
77 return k_ra8_err_invalid_arg;
78 }
79 if (*want_rx && (dma->rx_buffer == nullptr || dma->rx_samples == 0U)) {
80 return k_ra8_err_invalid_arg;
81 }
82 return k_ra8_ok;
83}

222ra8_err_t ra8_ssie_attach_dma(uint8_t channel, const ra8_ssie_dma_cfg_t* dma)
223{
224 RA8_CHECK_NULL_PTR(dma, s_tag, "dma must not be nullptr");
225 volatile r_ssie_regs_t* reg = ra8_ssie_internal_regs(channel);
226 if (reg == nullptr) {
227 return k_ra8_err_invalid_arg;
228 }
229
230 bool want_tx = false;
231 bool want_rx = false;
232 const ra8_err_t v_err = internal_validate_dma_cfg(dma, &want_tx, &want_rx);
233 RA8_RETURN_ON_ERROR(v_err, s_tag, "ssie_attach_dma: bad cfg"); /* GCOVR_EXCL_BR_LINE */
234
235 const ra8_err_t d_err = internal_attach_dma_dirs(reg, channel, dma, want_tx, want_rx);
236 RA8_RETURN_ON_ERROR(d_err, s_tag, "ssie_attach_dma: dir start"); /* GCOVR_EXCL_BR_LINE */
237
238 s_ssie_runtime[channel].dma_attached = true;
239 return k_ra8_ok;

242ra8_err_t ra8_ssie_detach_dma(uint8_t channel)
243{
244 if ((uint16_t)channel >= k_ra8_ssie_channel_count) {
245 return k_ra8_err_invalid_arg;
246 }
247 ra8_ssie_runtime_t* rt = &s_ssie_runtime[channel];
248 if (rt->tx_dma_channel < k_ra8_ssie_dma_max_ch) {
249 (void)ra8_dmac_stop(rt->tx_dma_channel);
250 rt->tx_dma_channel = k_ra8_ssie_dma_ch_unused;
251 }
252 if (rt->rx_dma_channel < k_ra8_ssie_dma_max_ch) {
253 (void)ra8_dmac_stop(rt->rx_dma_channel);
254 rt->rx_dma_channel = k_ra8_ssie_dma_ch_unused;
255 }
256 rt->dma_attached = false;
257 return k_ra8_ok;

260ra8_err_t ra8_ssie_attach_dma_pair(uint8_t channel, uint8_t tx_dma_channel, uint8_t rx_dma_channel)
261{
262 if ((uint16_t)channel >= k_ra8_ssie_channel_count) {
263 return k_ra8_err_invalid_arg;
264 }
265 if (tx_dma_channel >= (uint8_t)k_ra8_ssie_dma_max_ch &&
266 rx_dma_channel >= (uint8_t)k_ra8_ssie_dma_max_ch) {
267 return k_ra8_err_invalid_arg;
268 }
269 ra8_ssie_runtime_t* rt = &s_ssie_runtime[channel];
270 /* HUM Ch 46.4.1 "Operation in DMAC Transfer" p 3104:
271 * the SSIE issues TX-empty / RX-full requests to the DMAC; the
272 * driver only records which DMAC channels to free at detach. */
273 rt->tx_dma_channel =
274 (tx_dma_channel < k_ra8_ssie_dma_max_ch) ? tx_dma_channel : (uint8_t)k_ra8_ssie_dma_ch_unused;
275 rt->rx_dma_channel =
276 (rx_dma_channel < k_ra8_ssie_dma_max_ch) ? rx_dma_channel : (uint8_t)k_ra8_ssie_dma_ch_unused;
277 rt->dma_attached = true;
278 return k_ra8_ok;
279}

281ra8_err_t ra8_ssie_send_iso(uint8_t channel, const uint32_t* buffer, uint16_t samples)
282{
283 RA8_CHECK_NULL_PTR(buffer, s_tag, "buffer must not be nullptr");
284 volatile r_ssie_regs_t* reg = ra8_ssie_internal_regs(channel);
285 if (reg == nullptr) {
286 return k_ra8_err_invalid_arg;
287 }
288 /* Bounded loop: at most samples * k_ra8_ssie_fifo_depth FIFO probes.
289 * HUM Ch 46.2.4 "SSIFSR : FIFO Status Register" p 3083 (TDC field). */
290 uint16_t sent = 0U;
291 while (sent < samples) {
292 /* HUM Ch 46.2.4 "SSIFSR" p 3083 */
293 const uint8_t tdc = (uint8_t)((reg->SSIFSR & k_ra8_ssie_mask_tdc) >> k_ra8_ssie_shift_tdc);
294 if (tdc < (uint8_t)k_ra8_ssie_fifo_depth) {
295 /* HUM Ch 46.2.5 "SSIFTDR" p 3088 */
296 reg->SSIFTDR = buffer[sent];
297 ++sent;
298 }
299 /* In fake/host builds the TDC count never advances because
300 * no clock is consuming TX samples; treat the slot as drained
301 * immediately on every loop pass to remain bounded. */
302 }
303 /* W1C of TDE keeps the RDF bit. */
304 reg->SSIFSR = k_ra8_ssie_mask_rdf_clear;
305 return k_ra8_ok;
306}

308ra8_err_t
309ra8_ssie_recv_iso(uint8_t channel, uint32_t* buffer, uint16_t max_samples, uint16_t* out_got)
310{
311 RA8_CHECK_NULL_PTR(buffer, s_tag, "buffer must not be nullptr");
312 RA8_CHECK_NULL_PTR(out_got, s_tag, "out_got must not be nullptr");
313 volatile r_ssie_regs_t* reg = ra8_ssie_internal_regs(channel);
314 if (reg == nullptr) {
315 return k_ra8_err_invalid_arg;
316 }
317
318 uint16_t got = 0U;
319 for (uint16_t i = 0U; i < max_samples; ++i) {
320 /* HUM Ch 46.2.4 "SSIFSR" p 3083 */ /* (RDC field) */
321 const uint8_t rdc = (uint8_t)((reg->SSIFSR & k_ra8_ssie_mask_rdc) >> k_ra8_ssie_shift_rdc);
322 if (rdc == 0U) {
323 break;
324 }
325 /* HUM Ch 46.2.6 "SSIFRDR" p 3089 */
326 buffer[got] = reg->SSIFRDR;
327 ++got;
328 }
329 if (got > 0U) {
330 /* W1C RDF, keep TDE. */
331 reg->SSIFSR = k_ra8_ssie_mask_tde_clear;
332 }
333 *out_got = got;
334 return k_ra8_ok;
