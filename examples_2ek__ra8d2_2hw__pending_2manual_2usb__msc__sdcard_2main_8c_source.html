225[[nodiscard]] static ra8_err_t sdmsc_card_up(uint32_t pclka_hz)
226{
227 const ra8_sdmmc_spi_sci_pins_t pins = {
228 .sck = (ra8_port_pin_t)k_ra8_board_pmod2_spi_sck,
229 .cipo = (ra8_port_pin_t)k_ra8_board_pmod2_spi_cipo,
230 .copi = (ra8_port_pin_t)k_ra8_board_pmod2_spi_copi,
231 .cs = (ra8_port_pin_t)k_ra8_board_pmod2_spi_cs,
232 };
233 ra8_sdmmc_spi_transport_t transport = {};
234
235 ra8_err_t err =
236 ra8_sdmmc_spi_transport_sci((uint8_t)k_sdmsc_sd_spi_channel, pclka_hz, &pins, &transport);
237 if (err != k_ra8_ok) {
238 s_dbg_sd_err = (uint32_t)err;
239 (void)sdmsc_print_fail("sd transport", err);
240 return err;
241 }
242 err = ra8_sdmmc_spi_init(&transport);
243 if (err != k_ra8_ok) {
244 s_dbg_sd_err = (uint32_t)err;
245 (void)sdmsc_print_fail("sd init (card seated?)", err);
246 return err;
247 }
248 uint32_t blocks = 0U;
249 err = ra8_sdmmc_spi_get_capacity(&blocks);
250 if (err != k_ra8_ok) {
251 s_dbg_sd_err = (uint32_t)err;
252 (void)sdmsc_print_fail("sd capacity", err);
253 return err;
254 }
255 s_usb_msc_sdcard_blocks = blocks;
256 s_dbg_sd_err = 0U;
257 return sdmsc_print_card_banner(blocks);
258}

308static void sdmsc_setup_or_halt(void)
309{
310 uint32_t cpuclk0_hz = 0U;
311 uint32_t pclka_hz = 0U;
312 if (ra8_cgc_init() != k_ra8_ok) {
313 sdmsc_panic_halt();
314 }
315 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
316 sdmsc_panic_halt();
317 }
318 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
319 sdmsc_panic_halt();
320 }
321 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_pclka, &pclka_hz) != k_ra8_ok) {
322 sdmsc_panic_halt();
323 }
324 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
325 sdmsc_panic_halt();
326 }
327 if (ra8_board_uart_console_init((uint32_t)k_sdmsc_baud) != k_ra8_ok) {
328 sdmsc_panic_halt();
329 }
330 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
331 sdmsc_panic_halt();
332 }
333 if (ra8_board_led_init(k_ra8_board_led2) != k_ra8_ok) {
334 sdmsc_panic_halt();
335 }
336 /* Enumerate the card and latch its capacity before the kernel starts.
337 * On failure s_dbg_sd_err records it, the FAIL banner is on the wire,
338 * and the worker refuses to attach -- the board does not hang. */
339 (void)sdmsc_card_up(pclka_hz);
340 sdmsc_route_usb_or_halt();
341}
