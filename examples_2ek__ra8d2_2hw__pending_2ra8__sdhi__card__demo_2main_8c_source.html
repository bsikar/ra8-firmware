169static void sdhi_card_setup_or_halt(void)
170{
171 uint32_t cpuclk0_hz = 0U;
172 if (ra8_cgc_init() != k_ra8_ok) {
173 sdhi_card_panic_halt();
174 }
175 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
176 sdhi_card_panic_halt();
177 }
178 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
179 sdhi_card_panic_halt();
180 }
181 if (ra8_board_uart_console_init((uint32_t)k_sdhi_card_uart_baud) != k_ra8_ok) {
182 sdhi_card_panic_halt();
183 }
184 if (ra8_board_sdhi_pins_init() != k_ra8_ok) {
185 sdhi_card_panic_halt();
186 }
187}

206static void sdhi_card_init_or_halt(void)
207{
208 /* Request the SD 4-bit wide bus (ACMD6). Best-effort inside
209 * ra8_sdcard_init: a card that declines is left at 1-bit and init
210 * still succeeds, so this never turns a working card into a failure. */
211 const ra8_sdcard_cfg_t cfg = {.instance = (uint8_t)k_sdhi_card_instance,
212 .bus_width = k_ra8_sdhi_bus_width_4bit};
213 if (ra8_sdcard_init(&cfg) != k_ra8_ok) {
214 SDHI_CARD_PUTS(k_msg_init_fail);
215 sdhi_card_panic_halt();
216 }
217 SDHI_CARD_PUTS(k_msg_card_ok);
218}

246static void sdhi_card_fill_payload(void)
247{
248 uint32_t state = (uint32_t)k_sdhi_card_prng_seed;
249 for (uint32_t i = 0U; i < (uint32_t)k_sdhi_card_block_bytes; i++) {
250 state = (state * (uint32_t)k_sdhi_card_prng_mul) + (uint32_t)k_sdhi_card_prng_add;
251 s_payload[i] = (uint8_t)((state >> k_sdhi_card_prng_byte_shift) & k_sdhi_card_byte_mask);
252 }
253}

275[[nodiscard]] static ra8_err_t sdhi_card_check_capacity(void)
276{
277 uint32_t blocks = 0U;
278 RA8_RETURN_ON_ERROR(ra8_sdcard_get_capacity(&blocks), s_tag, "capacity");
279 if (blocks <= (uint32_t)k_sdhi_card_test_lba) {

306[[nodiscard]] static ra8_err_t sdhi_card_block_roundtrip(void)
307{
308 RA8_RETURN_ON_ERROR(ra8_sdcard_write_blocks((uint32_t)k_sdhi_card_test_lba,
309 s_payload,
310 (uint32_t)k_sdhi_card_block_count),
311 s_tag,
312 "write");
313 memset(s_readback, 0, sizeof(s_readback));
314 RA8_RETURN_ON_ERROR(ra8_sdcard_read_blocks((uint32_t)k_sdhi_card_test_lba,
315 s_readback,
316 (uint32_t)k_sdhi_card_block_count),
317 s_tag,
318 "read");
319 if (memcmp(s_payload, s_readback, (size_t)k_sdhi_card_block_bytes) != 0) {
320 return k_ra8_err_checksum_mismatch;
321 }
322 return k_ra8_ok;
323}

343[[nodiscard]] static ra8_err_t sdhi_card_roundtrip(void)
344{
345 RA8_RETURN_ON_ERROR(sdhi_card_check_capacity(), s_tag, "capacity check");
346 RA8_RETURN_ON_ERROR(sdhi_card_block_roundtrip(), s_tag, "block roundtrip");
347 return k_ra8_ok;
348}
