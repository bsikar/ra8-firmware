81static ra8_err_t internal_reset_channel(uint8_t ch)
82{
83 volatile r_acmphs_regs_t* reg = ra8_acmphs(ch);
84 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE -- ch already bounded */
85 return k_ra8_err_hw_init_failed; /* GCOVR_EXCL_LINE */
86 }
87 if (ch < k_ra8_acmphs_mstp_id_count) {
88 /* HUM Ch 11.2.9 "MSTPCRD : Module Stop Control Register D", p 449 */
89 const ra8_err_t mst_err = ra8_mstp_enable(s_acmphs_mstp_table[ch]);
90 RA8_RETURN_ON_ERROR(mst_err, s_tag, "acmphs_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
91 }
92 reg->CMPCTL = 0U;
93 reg->CMPSEL0 = 0U;
94 reg->CMPSEL1 = 0U;
95 reg->CPIOC = 0U;
96 return k_ra8_ok;
97}

162ra8_err_t ra8_acmphs_channel_init(uint8_t channel, const ra8_acmphs_cfg_t* cfg)
163{
164 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
165 if ((uint16_t)channel >= k_ra8_acmphs_channel_count) {
166 return k_ra8_err_invalid_arg;
167 }
168 volatile r_acmphs_regs_t* reg = ra8_acmphs(channel);
169 RA8_CHECK_NULL_PTR(reg, s_tag, "channel mapping failed");
170
171 if (channel < k_ra8_acmphs_mstp_id_count) {
172 const ra8_err_t mst_err = ra8_mstp_enable(s_acmphs_mstp_table[channel]);
173 RA8_RETURN_ON_ERROR(mst_err, s_tag, "acmphs_init: mstp"); /* GCOVR_EXCL_BR_LINE */
174 }
175
176 reg->CMPSEL0 = cfg->ivpsel;
177 reg->CMPSEL1 = cfg->ivrefsel;
178
179 /* FSP CMPCTL packs the filter select into CDFS[1:0] @ [6:5]. There
180 * is no standalone CMPFIR register on RA8D2; the driver maps its
181 * boolean filter_en into CDFS = 0 (off) or 1 (base sampling). */
182 uint8_t ctl = k_ra8_acmphs_mask_hcen;
183 ctl |= (uint8_t)((uint8_t)cfg->edge << k_ra8_acmphs_ceg_shift);
184 if (cfg->invert_out) {
185 ctl |= k_ra8_acmphs_mask_cinv;
186 }
187 if (cfg->filter_en) {
188 ctl |= (uint8_t)(k_ra8_acmphs_cdfs_enabled_value << k_ra8_acmphs_cdfs_shift);
189 }
190 reg->CMPCTL = ctl;
191 return k_ra8_ok;
192}
