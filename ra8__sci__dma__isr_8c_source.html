42/* Build a DMA request descriptor for byte-stream to/from SCI TDR/RDR -- see surrounding code and HUM citations. */
43static ra8_dma_request_t internal_make_dma_request(uintptr_t src,
44 uintptr_t dst,
45 uint16_t len,
46 bool src_inc,
47 bool dst_inc,
48 ra8_dma_complete_fn_t on_complete,
49 void* ctx)
50{
51 ra8_dma_request_t req = {};
52 req.src_addr = src;
53 req.dst_addr = dst;
54 req.count = len;
55 req.width = k_ra8_dmac_width_byte;
56 req.src_inc = src_inc;
57 req.dst_inc = dst_inc;
58 /* HUM Ch 19 "Event Link Controller (ELC)" p 817 -- trigger routing is a
59 * task; until then use software-start and drive the first
60 * element from the polling path or rely on ra8_fake_dma for host tests. */
61 req.trigger = (ra8_elc_event_t)0;
62 req.on_complete = on_complete;
63 req.ctx = ctx;
64 return req;
65}

67ra8_err_t ra8_sci_write_dma(uint8_t channel,
68 const uint8_t* data,
69 uint16_t len,
70 ra8_dma_complete_fn_t on_complete,
71 void* ctx,
72 uint8_t* out_dma_channel)
73{
74 RA8_CHECK_NULL_PTR(data, s_tag, "write_dma: data");
75 RA8_CHECK_NULL_PTR(out_dma_channel, s_tag, "write_dma: out_dma_channel");
76 volatile r_sci_regs_t* reg = ra8_sci(channel);
77 if ((reg == nullptr) || (len == 0U)) {
78 return k_ra8_err_invalid_arg;
79 }
80 /* HUM Ch 38.2.3 "TDR : Transmit Data Register", p 2181 -- DMA writes
81 * land in TDR; the engine streams one byte per element while the
82 * source increments across data[]. */
83 const ra8_dma_request_t req = internal_make_dma_request((uintptr_t)data,
84 (uintptr_t)&reg->TDR,
85 len,
86 /*src_inc=*/true,
87 /*dst_inc=*/false,
88 on_complete,
89 ctx);
90 return ra8_dma_request(&req, out_dma_channel);
91}

125RA8_ISR_SAFE
126void ra8_sci_dispatch_txi(uint8_t channel)
127{
128 if (channel > k_ra8_sci_channel_max_index) {
129 return;
130 }
131 volatile r_sci_regs_t* reg = ra8_sci(channel);
132 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE -- bounds already validated */
133 return;
134 }
135 const ra8_sci_tx_fn_t cb = s_sci_state[channel].tx_fn;
136 void* const ctx = s_sci_state[channel].tx_ctx;
137 const uint32_t tie = (1U << k_ra8_sci_ccr0_bit_tie);
138
139 /* Async byte-stream path (ra8_sci_write). Mirrors FSP r_sci_b_uart
140 * `txi_isr` (r_sci_b_uart.c) which decrements `tx_src_bytes`
141 * each TXI and clears TIE when the count hits zero. */
142 if (s_sci_state[channel].tx_len > 0U) {
143 if (s_sci_state[channel].tx_idx < s_sci_state[channel].tx_len) {
144 const uint8_t byte = s_sci_state[channel].tx_buf[s_sci_state[channel].tx_idx];
145 s_sci_state[channel].tx_idx += 1U;
146 /* HUM Ch 38.2.3 "TDR : Transmit Data Register", p 2181 */
147 reg->TDR = (uint32_t)byte;
148 /* Fire user-attached TX visibility callback per byte; its
149 * boolean return is ignored here because the async path owns
150 * the byte-stream now. */
151 if (cb != nullptr) {
152 uint8_t echo = byte;
153 (void)cb(ctx, &echo);
154 }
155 }
156 if (s_sci_state[channel].tx_idx >= s_sci_state[channel].tx_len) {
157 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 */
158 reg->CCR0 = reg->CCR0 & ~tie;
159 s_sci_state[channel].tx_buf = nullptr;
160 s_sci_state[channel].tx_len = 0U;
161 s_sci_state[channel].tx_idx = 0U;
162 }
163 return;
164 }
165
166 /* Legacy callback-only path (ra8_sci_attach_tx_handler). */
167 if (cb == nullptr) {
168 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 */
169 reg->CCR0 = reg->CCR0 & ~tie;
170 return;
171 }
172 uint8_t byte = 0U;
173 if (cb(ctx, &byte)) {
174 /* HUM Ch 38.2.3 "TDR : Transmit Data Register", p 2181 */
175 reg->TDR = (uint32_t)byte;
176 } else {
177 reg->CCR0 = reg->CCR0 & ~tie;
178 }
179}

181RA8_ISR_SAFE
182void ra8_sci_dispatch_rxi(uint8_t channel)
183{
184 if (channel > k_ra8_sci_channel_max_index) {
185 return;
186 }
187 volatile r_sci_regs_t* reg = ra8_sci(channel);
188 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE -- bounds already validated */
189 return;
190 }
191 const ra8_sci_rx_fn_t cb = s_sci_state[channel].rx_fn;
192 void* const ctx = s_sci_state[channel].rx_ctx;
193 /* HUM Ch 38.2.2 "RDR : Receive Data Register", p 2180 */
194 const uint8_t b = (uint8_t)(reg->RDR & k_ra8_sci_rdr_mask_data8);
195
196 /* Async byte-stream path (ra8_sci_read). Mirrors FSP r_sci_b_uart
197 * `rxi_isr` which appends each byte into `p_rx_dest` until
198 * `rx_dest_bytes` is satisfied, then signals UART_EVENT_RX_COMPLETE. */
199 if (s_sci_state[channel].rx_len > 0U) {
200 if (s_sci_state[channel].rx_idx < s_sci_state[channel].rx_len) {
201 s_sci_state[channel].rx_buf[s_sci_state[channel].rx_idx] = b;
202 s_sci_state[channel].rx_idx += 1U;
203 }
204 if (s_sci_state[channel].rx_idx >= s_sci_state[channel].rx_len) {
205 const uint32_t rie = (1U << k_ra8_sci_ccr0_bit_rie);
206 /* HUM Ch 38.2.5 "CCR0 : Common Control Register 0", p 2182 */
207 reg->CCR0 = reg->CCR0 & ~rie;
208 s_sci_state[channel].rx_buf = nullptr;
209 s_sci_state[channel].rx_len = 0U;
210 s_sci_state[channel].rx_idx = 0U;
211 }
212 }
213
214 /* Legacy attach path: still fires per byte even when an async RX is
215 * in flight, so existing flow-control hooks keep working. */
216 if (cb != nullptr) {
217 cb(ctx, b);
218 }
219}
