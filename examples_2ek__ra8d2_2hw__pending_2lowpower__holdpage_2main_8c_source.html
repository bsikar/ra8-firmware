110[[noreturn]] static void park_low_power(volatile lowpower_mailbox_t* mb)
111{
112 (void)mb; /* Read only inside ra8_log_info_val(), which a non-Debug build elides. */
113 uint32_t wakes = 0U;
114 while (1) {
115 __asm volatile("wfi");
116 wakes++;
