286static void sbns_console_print_hex32(uint32_t value)
287{
288 if (!s_sbns_console_up) {
289 return;
290 }
291 uint8_t buf[k_sbns_hex_nibbles];
292 for (uint8_t i = 0U; i < (uint8_t)k_sbns_hex_nibbles; i++) {
293 const uint32_t shift =
294 (uint32_t)((uint8_t)k_sbns_hex_nibbles - 1U - i) * (uint32_t)k_sbns_hex_bits;
295 const uint8_t nib = (uint8_t)((value >> shift) & (uint32_t)k_sbns_hex_nibble_mask);
296 buf[i] = (nib <= (uint8_t)k_sbns_hex_dec_max)
297 ? (uint8_t)('0' + nib)
298 : (uint8_t)('a' + (nib - (uint8_t)k_sbns_hex_alpha_base));
299 }
300 uint8_t first = 0U;
301 while (first < (uint8_t)((uint8_t)k_sbns_hex_nibbles - 1U)) {
302 if (buf[first] != (uint8_t)'0') {
303 break;
304 }
305 first++;
306 }
307 (void)ra8_board_uart_console_write(&buf[first], (size_t)((uint8_t)k_sbns_hex_nibbles - first));
308}

334int32_t main(void)
335{
336 /* Bring up the crypto heap + PSA facade the root-of-trust verify needs. The
337 * SAU + NS-image copy already ran in ra8_trustzone_init (SystemInit). */
338 mbedtls_memory_buffer_alloc_init(s_sbns_heap, sizeof(s_sbns_heap));
339 const ra8_err_t psa_err = ra8_psa_crypto_init();
340 if ((psa_err != k_ra8_ok) && (psa_err != k_ra8_err_exists)) {
341 g_sbns_step = k_sbns_step_psa_fail;
342 sbns_park();
343 }
344 g_sbns_step = k_sbns_step_crypto;
345
346 /* Bring up the Secure UART console and announce the crypto milestone so the
347 * boot verdict is scrape-able on the bench. Best-effort: a dead console must
348 * never gate the security decision, so the prints no-op if it failed. */
349 sbns_console_bringup();
350 sbns_console_say(k_sbns_msg_crypto_ready, (uint32_t)(sizeof(k_sbns_msg_crypto_ready) - 1U));
351
352 /* Authenticate + jump. The RA8_ENABLE_ROOT_OF_TRUST gate inside jump_ns reads
353 * the NS RoT header for the body length, verifies SHA-256 + ECDSA-P256, and
354 * BLXNS-es ONLY on success. A genuine image never returns here. */
355 g_sbns_step = k_sbns_step_armed;
356 sbns_console_say(k_sbns_msg_verify_enter, (uint32_t)(sizeof(k_sbns_msg_verify_enter) - 1U));
357 const ra8_err_t jump_err =
358 ra8_tz_secure_boot_jump_ns((const uint32_t*)(uintptr_t)k_sbns_ns_run_base);
359
360 /* Reached only when the gate DENIED the NS image (tampered / unsigned): the
361 * default-deny path. Latch + print the outcome for the bench and halt -- the
362 * NS world never ran, so g_sbns_ns_alive stays 0. */
363 g_sbns_jump_err = (uint32_t)jump_err;
364 g_sbns_denied = 1U;
365 g_sbns_step = k_sbns_step_denied;
366 sbns_console_say_reject(jump_err);
367 sbns_park();
368 return 0;
369}
