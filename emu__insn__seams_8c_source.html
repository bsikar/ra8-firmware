33/* Armv8.1-M conditional-select family (CSEL/CSINC/CSINV/CSNEG). The RA8D2 is
34 * Cortex-M85 (Armv8.1-M) but Unicorn's nearest core is M33 (Armv8-M), which
35 * lacks these. GCC emits them for branchless index/modulo math -- notably in
36 * the firmware's event_post on the touch path -- so the invalid-instruction
37 * hook decodes and executes them and lets the real firmware path continue.
38 * Encoding (T1, two halfwords, little-endian): hw1 = 0xEA5n (n = Rn);
39 * hw2 bit15=1, bit14=0, op=bits[13:12], Rd=bits[11:8], cond=bits[7:4],
40 * Rm=bits[3:0]. Verified against the GNU assembler for armv8.1-m.main. */

44 k_cs_hw2_b15 = 0x8000U,
45 k_cs_hw2_b14 = 0x4000U,
46 k_cs_op_csel = 0U,
47 k_cs_op_csinc = 1U,
48 k_cs_op_csinv = 2U,
49 k_cs_op_csneg = 3U,
50 k_cs_insn_len = 4U,
51 k_cs_reg_sp = 13U,
52 k_cs_reg_pc = 15U,
53} cond_select_t;

92static bool cond_holds(uint32_t cond, uint32_t xpsr)
93{
94 const bool n = ((xpsr >> (uint32_t)k_apsr_n) & 1U) != 0U;
95 const bool z = ((xpsr >> (uint32_t)k_apsr_z) & 1U) != 0U;
96 const bool c = ((xpsr >> (uint32_t)k_apsr_c) & 1U) != 0U;
97 const bool v = ((xpsr >> (uint32_t)k_apsr_v) & 1U) != 0U;
98 switch (cond & (uint32_t)k_lo4_mask) {
99 case k_cond_eq:
100 return z;
101 case k_cond_ne:
102 return !z;
103 case k_cond_cs:
104 return c;
105 case k_cond_cc:
106 return !c;
107 case k_cond_mi:
108 return n;
109 case k_cond_pl:
110 return !n;
111 case k_cond_vs:
112 return v;
113 case k_cond_vc:
114 return !v;
115 case k_cond_hi:
116 return c && !z;
117 case k_cond_ls:
118 return !c || z;
119 case k_cond_ge:
120 return n == v;
121 case k_cond_lt:
122 return n != v;
123 case k_cond_gt:
124 return !z && (n == v);
125 case k_cond_le:
126 return z || (n != v);
127 default:
128 return true; /* AL (k_cond_al) / 0xF */
129 }
130}

168static bool emulate_cond_select(uc_engine* uc, uint32_t pc, const uint8_t* code)
169{
170 const uint16_t hw1 = (uint16_t)(code[0] | ((uint16_t)code[1] << 8));
171 const uint16_t hw2 = (uint16_t)(code[2] | ((uint16_t)code[3] << 8));
172 if (((hw1 & (uint16_t)k_cs_hw1_mask) != (uint16_t)k_cs_hw1_match) ||
173 ((hw2 & (uint16_t)k_cs_hw2_b15) == 0U) || ((hw2 & (uint16_t)k_cs_hw2_b14) != 0U)) {
174 return false;
175 }
176 const uint32_t rn = (uint32_t)(hw1 & (uint32_t)k_lo4_mask);
177 const uint32_t op = (uint32_t)((hw2 >> (uint32_t)k_cs_op_shift) & (uint32_t)k_cs_op_mask);
178 const uint32_t rd = (uint32_t)((hw2 >> 8) & (uint32_t)k_lo4_mask);
179 const uint32_t cond = (uint32_t)((hw2 >> 4) & (uint32_t)k_lo4_mask);
180 const uint32_t rm = (uint32_t)(hw2 & (uint32_t)k_lo4_mask);
181
182 /* The Armv8.1-M long shifts share this family's 0xEA5n prefix and put 0b1101
183 * (SP) or 0b1111 (PC) in what reads here as Rm, so a register-form long shift
184 * such as LSLL r2, r3, r8 (EA52 830D) satisfies every condition above -- it
185 * would execute here as CSEL r3, r2, sp, EQ and silently corrupt r3. Reserved
186 * operands mean this is not a conditional select; fall through and let the
187 * long-shift seam claim it. */
188 if (cs_reserved_reg(rn) || cs_reserved_reg(rd) || cs_reserved_reg(rm)) {
189 return false;
190 }
191
192 uint32_t xpsr = 0U;
193 uint32_t vn = 0U;
194 uint32_t vm = 0U;
195 (void)uc_reg_read(uc, UC_ARM_REG_XPSR, &xpsr);
196 (void)uc_reg_read(uc, k_arm_reg_id[rn], &vn);
197 (void)uc_reg_read(uc, k_arm_reg_id[rm], &vm);
198
199 uint32_t result;
200 if (cond_holds(cond, xpsr)) {
201 result = vn;
202 } else {
203 switch (op) {
204 case (uint32_t)k_cs_op_csinc:
205 result = vm + 1U;
206 break;
207 case (uint32_t)k_cs_op_csinv:
208 result = ~vm;
209 break;
210 case (uint32_t)k_cs_op_csneg:
211 result = (uint32_t)(-(int32_t)vm);
212 break;
213 case (uint32_t)k_cs_op_csel:
214 default:
215 result = vm;
216 break;
217 }
218 }
219 (void)uc_reg_write(uc, k_arm_reg_id[rd], &result);
220 uint32_t next = pc + (uint32_t)k_cs_insn_len;
221 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
222 return true;
223}

305/* ---------------------------------------------------------------------------
306 * Low-Overhead-Branch (LOB) emulation. The M85 has the hardware-loop extension
307 * (DLS/WLS/LE) that GCC's -Og uses for counted loops; the M33 does not, so they
308 * trap as invalid. Handle the two forms the toolchain emits:
309 * DLS lr, Rn -- start a loop: LR = Rn (the iteration count)
310 * LE lr, <label> -- loop end: LR -= 1; branch back to <label> while LR != 0
311 * (the toolchain's <label> offset is (hw2 & 0x7FF) 2-byte units backward from
312 * PC+4, verified against its disassembly). Both sit amid valid code, so the same
313 * stop-then-relaunch contract as the cond-select seam applies. WLS/LETP/DLSTP are
314 * not emitted here and fall through to the normal invalid-instruction report.
315 * ===========================================================================
316 */
317enum : uint32_t {
318 k_lob_dls_hw1 = 0xF040U,
319 k_lob_dls_h1m = 0xFFF0U,
320 k_lob_dls_hw2 = 0xE001U,
321 k_lob_le_hw1 = 0xF00FU,
322 k_lob_le_hw2 = 0xC000U,
323 k_lob_le_h2m = 0xF000U,
324 k_lob_le_imm10 = 0x03FFU,
325 k_lob_le_lsb = 11U,
326 k_lob_rn_mask = 0xFU,
327 k_lob_insn_len = 4U,
328};
330static uint64_t s_lob_emulated = 0U;

333static bool emulate_lob(uc_engine* uc, uint32_t pc, const uint8_t code[4])
334{
335 const uint16_t hw1 = (uint16_t)(code[0] | ((uint16_t)code[1] << (uint16_t)k_byte_bits));
336 const uint16_t hw2 = (uint16_t)(code[2] | ((uint16_t)code[3] << (uint16_t)k_byte_bits));
337 /* DLS lr, Rn -> LR = Rn (start the counted loop). */
338 if (((hw1 & (uint16_t)k_lob_dls_h1m) == (uint16_t)k_lob_dls_hw1) &&
339 (hw2 == (uint16_t)k_lob_dls_hw2)) {
340 /* Map Rn via k_arm_reg_id[] -- Unicorn's UC_ARM_REG_* enum is NOT contiguous
341 * (UC_ARM_REG_R0 + n != UC_ARM_REG_Rn), so the prior arithmetic read the
342 * wrong register for the loop count -> bogus iteration count -> over-run and
343 * a wild branch (#233). This is the same table the CSEL/long-shift seams use. */
344 const uint32_t rn = (uint32_t)hw1 & (uint32_t)k_lob_rn_mask;
345 uint32_t v = 0U;
346 (void)uc_reg_read(uc, k_arm_reg_id[rn], &v);
347 (void)uc_reg_write(uc, UC_ARM_REG_LR, &v);
348 const uint32_t next = pc + (uint32_t)k_lob_insn_len;
349 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
350 s_lob_emulated++;
351 return true;
352 }
353 /* LE lr, label -> LR -= 1; branch back while LR != 0, else fall through. */
354 if ((hw1 == (uint16_t)k_lob_le_hw1) &&
355 ((hw2 & (uint16_t)k_lob_le_h2m) == (uint16_t)k_lob_le_hw2)) {
356 uint32_t lr = 0U;
357 (void)uc_reg_read(uc, UC_ARM_REG_LR, &lr);
358 lr -= 1U;
359 (void)uc_reg_write(uc, UC_ARM_REG_LR, &lr);
360 uint32_t next = pc + (uint32_t)k_lob_insn_len;
361 if (lr != 0U) {
362 /* Backward branch to the loop top: the 11-bit immediate is scattered --
363 * hw2[10:1] are the high bits and hw2[11] is the offset LSB (2-byte
364 * granularity) -- so offset = (imm10 << 2) | (lsb << 1), taken from PC+4.
365 * Verified against objdump for all 95 LE sites in the -O1 reader image
366 * (#233). The previous (hw2 & 0x7FF) << 1 decode ignored the scatter and
367 * mis-branched at 55 of them, running loops the wrong number of times
368 * (memory corruption + a wild branch to zeros at -O1). */
369 const uint32_t imm10 = ((uint32_t)hw2 >> 1U) & (uint32_t)k_lob_le_imm10;
370 const uint32_t lsb = ((uint32_t)hw2 >> (uint32_t)k_lob_le_lsb) & 1U;
371 next -= ((imm10 << 2U) | (lsb << 1U)); /* backward to loop top. */
372 }
373 (void)uc_reg_write(uc, UC_ARM_REG_PC, &next);
374 s_lob_emulated++;
375 return true;
376 }
377 return false;
378}

405static bool dispatch_armv81_seam(uc_engine* uc, uint32_t pc, const uint8_t code[4])
406{
407 /* The RA8D2 firmware is built for Cortex-M85 (Armv8.1-M); the nearest core
408 * Unicorn offers is M33 (Armv8-M), which lacks the conditional-select family.
409 * GCC emits those for branchless index math on the touch path, so
410 * execute them here. emulate_cond_select writes Rd and advances PC past the
411 * 4-byte instruction; then uc_emu_stop so the chunked run loop relaunches from
412 * the new PC -- editing PC and continuing in-place corrupts Unicorn's block /
413 * Thumb state (it then misdecodes the next valid instruction), so the
414 * stop-then-relaunch contract the SysTick / touch stubs use is required here. */
415 if (emulate_cond_select(uc, pc, code)) {
416 return true; /* handled -- run loop resumes at the advanced PC */
417 }
418
419 /* Older Unicorn builds (the runner's 2.0.1) trap DSB/DMB/ISB as invalid; a
420 * barrier is a NOP in this emulator, so advance past it and relaunch. */
421 if (emulate_barrier(uc, pc, code)) {
422 return true; /* handled -- run loop resumes past the barrier */
423 }
424
425 /* MVE (Helium): the M85 has it, Unicorn's M33 does not, so the auto-vectoriser's
426 * Helium ops trap here. Emulate the handled subset and relaunch past it. */
427 if (emulate_mve(uc, pc, code)) {
428 return true; /* handled -- run loop resumes past the MVE instruction */
429 }
430
431 /* Armv8.1-M long shift, REGISTER form (LSLL/ASRL by Rm). Its encoding aliases
432 * to an ORR.W with Rm == SP, which the core refuses rather than
433 * mis-executing, so it lands here as a real trap. (The immediate form aliases
434 * to an ORRS the core happily executes with a wrong result, so that one is
435 * repaired by the code-hook seam in emu_seam_longshift.c instead.) */
436 if (emulate_long_shift_reg(uc, pc, code)) {
437 return true; /* handled -- run loop resumes past the long shift */
438 }
439
440 /* Low-Overhead-Branch (DLS/LE): the M85's hardware-loop ops, absent on the M33.
441 * Emulate the loop counter / branch and relaunch. */
442 if (emulate_lob(uc, pc, code)) {
443 return true; /* handled -- run loop resumes at the loop top or past the loop */
444 }
445 return false;
446}

449static bool dispatch_insn_seam(uc_engine* uc, uint32_t pc, const uint8_t code[4])
450{
451 /* Divide-by-zero trap: a UDIV/SDIV overwritten with UDF once the firmware set
452 * CCR.DIV_0_TRP. emulate_div0_patched either latches a UsageFault (zero divisor)
453 * or emulates the divide and advances PC; either way stop + relaunch. Checked
454 * first: it matches only the exact patched addresses, never a real UDF. */
455 if (emulate_div0_patched(uc, pc, code)) {
456 (void)uc_emu_stop(uc);
457 return true;
458 }
459
460 /* Armv8-M Security Extension register scrub (CLRM / VSCCLRM) on a cmse
461 * Non-Secure-Callable return: a NOP in ra8_emulator's single-domain model. */
462 if (emulate_sec_scrub(uc, pc, code)) {
463 (void)uc_emu_stop(uc);
464 return true;
465 }
466
467 if (dispatch_armv81_seam(uc, pc, code)) {
468 (void)uc_emu_stop(uc);
469 return true; /* handled -- run loop resumes past the emulated instruction */
470 }
471 return false;
472}

475static void report_unhandled_insn(uint32_t pc, const uint8_t code[4])
476{
477 (void)fprintf(stderr,
478 " INVALID INSN @ 0x%08X: bytes %02X %02X %02X %02X\n",
479 pc,
480 code[0],
481 code[1],
482 code[2],
483 code[3]);
484
485 csh cs;
486 if (cs_open(CS_ARCH_ARM, (cs_mode)(CS_MODE_THUMB | CS_MODE_MCLASS), &cs) == CS_ERR_OK) {
487 cs_insn* insn = nullptr;
488 /* `code` decays to a pointer here, so sizeof(code) would hand capstone the
489 * pointer width instead of the four instruction bytes that are valid. */
490 const size_t n = cs_disasm(cs, code, (size_t)k_cs_insn_len, pc, 1, &insn);
491 if (n > 0U) {
492 (void)fprintf(stderr, " disasm: %s %s\n", insn[0].mnemonic, insn[0].op_str);
493 cs_free(insn, n);
494 } else {
495 (void)fprintf(stderr, " disasm: capstone could not decode it either\n");
496 }
497 cs_close(&cs);
498 }

502bool on_invalid_insn(uc_engine* uc, void* user)
503{
504 (void)user;
505 uint32_t pc = 0U;
506 (void)uc_reg_read(uc, UC_ARM_REG_PC, &pc);
507 uint8_t code[4] = {};
508 (void)uc_mem_read(uc, pc, code, sizeof(code));
509
510 if (dispatch_insn_seam(uc, pc, code)) {
511 return true; /* handled -- run loop resumes at the advanced PC */
512 }
513
514 /* Unicorn re-reports the instruction FOLLOWING a serviced NoCP fault as
515 * invalid even when it is perfectly valid, because the seam wrote PC and
516 * stopped the engine from inside the interrupt hook. Absorb exactly that one
517 * report, at exactly that one address, and relaunch. */
518 if (emu_mve_nocp_spurious(pc)) {
519 (void)uc_emu_stop(uc);
520 return true;
521 }
522
523 report_unhandled_insn(pc, code);
524 return false; /* not handled -> stop emulation with UC_ERR_INSN_INVALID */
525}
