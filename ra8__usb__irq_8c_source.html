124RA8_ISR_SAFE
125void ra8_usb_dispatch(ra8_usb_speed_t speed)
126{
127 volatile r_usb_regs_t* reg = internal_pick(speed);
128 if (reg == nullptr) { /* GCOVR_EXCL_BR_LINE -- speeds always valid */
129 return; /* GCOVR_EXCL_LINE */
130 }
131 /* HUM Ch 36.2.14 "INTSTS0 : Interrupt Status Register 0", p 1986.
132 *
133 * INTSTS0 is W0C: writing 0 to a bit clears it, writing 1 has no
134 * effect. The previous mask-and-back pattern
135 * INTSTS0 = mask & ~ack_bits
136 * unconditionally writes 0 to every bit in ack_bits, even bits that
137 * were 0 in the snapshot. That means a CTRT (or DVST) edge that
138 * asserts in HW *between* the read of `mask` and the write of the
139 * ack pattern is silently cleared before the dispatcher can act on
140 * it. With a free-running polled worker (1.7M ticks/sec) and SETUP
141 * tokens arriving at ~1 ms cadence, almost every CTRT edge falls
142 * inside this race window, which is why s_ctrt_irq_count stayed at
143 * 0 even though DVST was caught (DVST edges hold longer because the
144 * device-state remains in default until the next reset).
145 *
146 * Correct W0C ack: write 0 ONLY to the bits we observed set, and 1
147 * (no-op) everywhere else. ``~(mask & ack_bits)`` does exactly that:
148 * for any bit not in ack_bits, we write 1 (preserve); for an
149 * ack_bits position that was 0 in the snapshot, we write 1 (no-op,
150 * does not clobber a freshly-set edge); for an ack_bits position
151 * that was 1 in the snapshot, we write 0 (clear).
152 *
153 * VALID is intentionally NOT in ack_bits -- the SETUP handlers
154 * (ra8_usb_read_setup_if_valid / _unconditional) drain
155 * USBREQ/USBVAL/USBINDX/USBLENG and clear VALID themselves.
156 * CTSQ/DVSQ are read-only fields. */
157 const uint16_t mask = reg->INTSTS0;
158
159 const uint16_t ack_bits = (uint16_t)((1U << k_ra8_int0_bit_ctrt) | (1U << k_ra8_int0_bit_dvst) |
160 (1U << k_ra8_int0_bit_bemp) | (1U << k_ra8_int0_bit_brdy) |
161 (1U << k_ra8_int0_bit_nrdy) | (1U << k_ra8_int0_bit_vbse) |
162 (1U << k_ra8_int0_bit_rsme) | (1U << k_ra8_int0_bit_sofr));
163 reg->INTSTS0 = (uint16_t)~(uint16_t)(mask & ack_bits);
164
165 const uint8_t slot = internal_cb_slot(speed);
166 const ra8_usb_event_fn_t fn = s_usb_fn[slot];
167 void* const ctx = s_usb_ctx[slot];
168 if (fn != nullptr) {
169 fn(ctx, speed, mask);
170 }
171}

278RA8_INTERNAL
279static uint16_t internal_host_syscfg_word(ra8_usb_speed_t speed)
280{
281 uint16_t syscfg = (uint16_t)(1U << k_ra8_syscfg_bit_scke);
282 syscfg = (uint16_t)(syscfg | (uint16_t)(1U << k_ra8_syscfg_bit_dcfm));
283 syscfg = (uint16_t)(syscfg | (uint16_t)(1U << k_ra8_syscfg_bit_drpd));
284 syscfg = (uint16_t)(syscfg | (uint16_t)(1U << k_ra8_syscfg_bit_usbe));
285 if (speed == k_ra8_usb_speed_hs) {
286 syscfg = (uint16_t)(syscfg | (uint16_t)(1U << k_ra8_syscfg_bit_hse));
287 }
288 return syscfg;
289}

389ra8_err_t ra8_usb_host_init(ra8_usb_speed_t speed)
390{
391 volatile r_usb_regs_t* reg = internal_pick(speed);
392 if (reg == nullptr) {
393 return k_ra8_err_invalid_arg;
394 }
395
396 /* HUM Ch 11.2.7 "MSTPCRB : Module Stop Control Register B", p 444 */
397 const ra8_err_t mst_err = ra8_mstp_enable(internal_mstp(speed));
398 RA8_RETURN_ON_ERROR(mst_err, s_tag, "host_init: mstp enable"); /* GCOVR_EXCL_BR_LINE */
399
400 if (speed == k_ra8_usb_speed_hs) {
401 const ra8_err_t hs_err = internal_host_hs_bringup(reg);
402 RA8_RETURN_ON_ERROR(hs_err, s_tag, "host_init: HS bring-up"); /* GCOVR_EXCL_BR_LINE */
403 } else {
404 /* HUM Ch 36.2.1 "SYSCFG : System Configuration Control Register", p 1967 */
405 reg->SYSCFG = internal_host_syscfg_word(speed);
406 }
407
408 /* HUM Ch 36.2.5 "DVSTCTR0 : Device State Control Register 0", p 1971 */
409 reg->DVSTCTR0 = 0U;
410 if (speed == k_ra8_usb_speed_hs) {
411 /* HUM Ch 37.2.5 DVSTCTR0.VBUSEN: the USBHS jack's external VBUS
412 * switch is driven by this bit (FSP hw_usb_hmodule_init); without
413 * it an attached device never powers and LNST stays SE0. */
414 internal_rmw16(&reg->DVSTCTR0, (uint16_t)(1U << k_ra8_dvstctr_bit_vbusen), 0U);
415 }
416
417 internal_host_init_defaults(reg);
418
419 ra8_log_info_val(s_tag, "usb host init speed", (uint32_t)speed);
420 return k_ra8_ok;
421}

467ra8_err_t ra8_usb_host_bus_reset(ra8_usb_speed_t speed, bool assert_reset)
468{
469 volatile r_usb_regs_t* reg = internal_pick(speed);
470 if (reg == nullptr) {
471 return k_ra8_err_invalid_arg;
472 }
473 /* HUM Ch 36.2.5 "DVSTCTR0 : Device State Control Register 0", p 1971 */
474 const uint16_t rst_bit = (uint16_t)(1U << k_ra8_dvstctr_bit_usbrst);
475 const uint16_t uact_bit = (uint16_t)(1U << k_ra8_dvstctr_bit_uact);
476 if (assert_reset) {
477 /* USBRST=1 forces UACT low; FSP atomically sets RST + clears UACT. */
478 internal_rmw16(&reg->DVSTCTR0, rst_bit, uact_bit);
479 } else {
480 internal_rmw16(&reg->DVSTCTR0, 0U, rst_bit);
481 }
482 return k_ra8_ok;
483}

529ra8_err_t ra8_usb_host_setup_request(ra8_usb_speed_t speed, const ra8_usb_setup_t* setup)
530{
531 RA8_CHECK_NULL_PTR(setup, s_tag, "host_setup_request: setup");
532 volatile r_usb_regs_t* reg = internal_pick(speed);
533 if (reg == nullptr) {
534 return k_ra8_err_invalid_arg;
535 }
536 /* HUM Ch 36.2.21 "DCPCTR : DCP Control Register", p 1999 -- guard
537 * against a still-pending request. */
538 const uint16_t sureq_bit = (uint16_t)(1U << k_ra8_dcpctr_bit_sureq);
539 if ((reg->DCPCTR & sureq_bit) != 0U) {
540 return k_ra8_err_busy;
541 }
542
543 /* HUM Ch 36.2.17 "USBREQ : USB Request Type Register", p 1995 */
544 const uint16_t req = (uint16_t)((uint16_t)setup->bm_request_type |
545 (uint16_t)((uint16_t)setup->b_request << k_ra8_usb_byte_bits));
546 reg->USBREQ = req;
547 reg->USBVAL = setup->w_value;
548 reg->USBINDX = setup->w_index;
549 reg->USBLENG = setup->w_length;
550
551 /* SUREQ tells the SIE to issue the SETUP token on the next frame. */
552 internal_rmw16(&reg->DCPCTR, sureq_bit, 0U);
553 return k_ra8_ok;
554}
