4"""mcdc_compound_ratchet.py -- MC/DC compound-decision ratchet (vs baseline).
5
6WHY THIS EXISTS
7---------------
8This project targets DO-178C Level B, where MC/DC coverage of every compound
9boolean decision is the core evidence. `check_new_compound_has_mcdc.py` is the
10detector for that rule and it works -- but it audited nothing in CI.
11
12`scripts/ci/gates/checks.sh` ran only its `--selftest`. The blocking `--range`
13invocation existed solely as a commented-out line (added commented, never
14enabled), while issue #426 carried a comment asserting it was blocking. So a
15newly-added uncovered compound decision passed CI, and the gate meant to stop
16exactly that was decorative -- this tree's signature defect.
17
18Turning the `--range` delta scan on was not the fix. A delta scan flags any
19decision whose normalized source line is new, so with a backlog this size it
20fails on a *reformat* of a pre-existing uncovered decision. That is a cliff,
21and a cliff gets bypassed. This ratchet is the mechanism the tree already uses
22for a measured debt it intends to burn down rather than grandfather -- it is
23the same shape as `tidy_ratchet.py` and `misra_ratchet.py`:
24
25* NEW findings (any per-file-per-function count above the baseline, or any
26 file/function bucket absent from the baseline) FAIL.
27* Shrinkage PASSES with a notice to re-baseline, which locks the progress in so
28 the debt can never quietly grow back.
29* Reformatting, renaming a variable inside, or moving an existing uncovered
30 decision changes no count, so it passes -- a ratchet, not a cliff.
31
32Closing the debt means the baseline reaching zero rows and being deleted -- not
33being regenerated larger. `--update` refuses to grow a bucket for exactly that
34reason; a genuine increase has to be justified by a human editing the file,
35which leaves a reviewable diff.
36
37BASELINE NORMALISATION -- per-file-per-FUNCTION counts, not raw finding lines.
38Raw findings carry line numbers, which churn on every unrelated edit above
39them. A `(file, function) -> count` map is invariant under that, still trips
40the moment a function gains another uncovered decision, and is keyed on exactly
41the granularity an MC/DC citation uses (`path@function`) -- so a baseline row
42names the function whose vectors are missing. It is strictly tighter than a
43per-file count: covering one decision while adding another elsewhere in the
44same file still fails.
45
46The measurement itself is `check_new_compound_has_mcdc.py::audit_tree`, the
47same detection primitives the delta modes use, imported rather than re-parsed
48from console output. There is therefore exactly one definition of "this
49decision lacks MC/DC vectors", and no text seam between detector and gate that
50could silently stop matching.
51
52USAGE
53 python3 scripts/checks/mcdc_compound_ratchet.py --selftest # assert it fires
54 python3 scripts/checks/mcdc_compound_ratchet.py --check # the CI gate
55 python3 scripts/checks/mcdc_compound_ratchet.py --update # re-baseline
56 python3 scripts/checks/mcdc_compound_ratchet.py --list # burn-down list
57
58Copyright (c) 2026 Brighton Sikarskie
59SPDX-License-Identifier: MIT

105def scan(root: Path) -> tuple[Counter, int, int]:
106 """Scan ``root`` and return ``(counts, production_file_count, citation_count)``.
107
108 ``counts`` maps ``(repo-relative file, enclosing function)`` to the number
109 of compound decisions in that function with no matching MC/DC vector set.
110 The two scalars accompany it so the caller can refuse a scan whose scope
111 never got established -- a count is only trustworthy once the thing that
112 produced it is known to have looked at the tree.
113 """
114 files, findings = audit_tree(root)
115 counts: Counter = Counter()
116 for path, function, _line, _snippet in findings:
117 counts[(path, function)] += 1
118 return counts, len(files), len(collect_tree_citations(root))

121def load_baseline() -> Counter:
122 """Read the committed baseline into a Counter. Missing file means empty."""
123 counts: Counter = Counter()
124 if not BASELINE_FILE.is_file():
125 return counts
126 for raw in BASELINE_FILE.read_text(encoding="ascii").splitlines():
127 line = raw.strip()
128 if not line or line.startswith("#"):
129 continue
130 parts = line.split("\t")
131 if len(parts) != BASELINE_COLUMNS:
132 continue
133 counts[(parts[0], parts[1])] = int(parts[2])
134 return counts

137def write_baseline(counts: Counter) -> None:
138 """Write `counts` out in the committed, sorted, diffable form."""
139 total = sum(counts.values())
140 lines = [
141 "# MC/DC compound-decision ratchet baseline -- per-file-per-function counts.",
142 "# Consumed by scripts/checks/mcdc_compound_ratchet.py --check",
143 "# (CI gate: pre-commit-checks).",
144 "#",
145 "# Each row is a function holding N compound boolean decisions (`&&` / `||`)",
146 "# with NO matching `@par MC/DC:` `path@function` citation in any",
147 "# tests/test_*.c. These predate enforcement (issue #426). The gate fails",
148 "# on any INCREASE, so the debt is frozen and can only be burned down; a",
149 "# newly-added uncovered decision raises a count and fails.",
150 "#",
151 f"# Total at this baseline: {total} uncovered compound decision(s)",
152 f"# across {len({p for p, _ in counts})} file(s).",
153 "#",
154 "# Burn one down: add a `test_mcdc_<decision>` function with N+1 vectors",
155 "# to the matching tests/test_<module>.c whose `@par MC/DC:` block cites",
156 "# `path@function`. See docs/MCDC.md for the worked example.",
157 "#",
158 "# Regenerate after burning findings down:",
159 "# python3 scripts/checks/mcdc_compound_ratchet.py --update",
160 "#",
161 "# RENAMING a function (or moving a file) retires its row and creates a",
162 "# new one, which reads as growth -- so the gate fails and --update",
163 "# refuses, by design. Rename the row(s) here BY HAND, keeping the counts",
164 "# identical. That is a one-line reviewable diff, and it is deliberately",
165 "# not automated: rename detection that guessed wrong would silently",
166 "# absorb a genuinely new uncovered decision.",
167 "#",
168 "# Closing this out means this file reaching zero rows and being",
169 "# DELETED -- never regenerated larger.",
170 "#",
171 "# file<TAB>function<TAB>count",
172 ]
173 for (path, function), n in sorted(counts.items()):
174 if n:
175 lines.append(f"{path}\t{function}\t{n}")
176 BASELINE_FILE.write_text("\n".join(lines) + "\n", encoding="ascii")

179def scope_reason(files: int, citations: int) -> str | None:
180 """Return a refusal reason when the scan's scope is not credible, else None.
181
182 Guards BOTH directions of the ratchet: a scan that examined a fragment of
183 the tree, or read a fragment of the citation index, produces a number that
184 must not be compared against the baseline and must certainly never be
185 written as one.
186 """
187 if files < MIN_PRODUCTION_FILES:
188 return (
189 f"only {files} production .c file(s) found under libs/ src/ port/, "
190 f"below the {MIN_PRODUCTION_FILES} floor.\n"
191 " The scan is not looking at this repository (wrong --root, or a\n"
192 " partial checkout). A partial scan reports FEWER uncovered\n"
193 " decisions, which reads as a burn-down; refusing it is the only\n"
194 " way that cannot be mistaken for progress."
195 )
196 if citations < MIN_CITATIONS:
197 return (
198 f"only {citations} MC/DC citation(s) found in tests/test_*.c, "
199 f"below the {MIN_CITATIONS} floor.\n"
200 " Every finding is 'this function is not cited', so a truncated\n"
201 " citation index makes the whole tree look uncovered. What changed\n"
202 " is the index, not the code, so the count is meaningless."
203 )
204 return None

207def report(current: Counter, baseline: Counter) -> int:
208 """Compare and print. Returns the process exit code."""
209 grown = []
210 for key, n in sorted(current.items()):
211 was = baseline.get(key, 0)
212 if n > was:
213 grown.append((key, was, n))
214
215 total_now = sum(current.values())
216 total_was = sum(baseline.values())
217
218 if grown:
219 print(file=sys.stderr)
220 print(
221 "MC/DC ratchet: NEW uncovered compound decisions above the baseline",
222 file=sys.stderr,
223 )
224 print(file=sys.stderr)
225 for (path, function), was, now in grown[:MAX_DETAIL_LINES]:
226 print(f" {path}\n {function}(): {was} -> {now}", file=sys.stderr)
227 if len(grown) > MAX_DETAIL_LINES:
228 print(f" ... and {len(grown) - MAX_DETAIL_LINES} more bucket(s)", file=sys.stderr)
229 print(file=sys.stderr)
230 print(
231 "Every compound boolean decision under libs/ src/ port/ needs MC/DC\n"
232 "vectors: add a `test_mcdc_<decision>` function with N+1 vectors to\n"
233 "the matching tests/test_<module>.c and cite the decision as\n"
234 "`path@function` in its `@par MC/DC:` block (see docs/MCDC.md).\n"
235 "\n"
236 "The baseline is a burn-down of debt that predates enforcement, not\n"
237 "a place to record new debt. Do not --update to make this pass.",
238 file=sys.stderr,
239 )
240 return 1
241
242 print(
243 f"MC/DC ratchet: {total_now} uncovered compound decision(s), "
244 f"baseline {total_was} -- no growth."
245 )
246 if total_now < total_was:
247 burned = total_was - total_now
248 print(f" {burned} decision(s) burned down. Re-baseline to lock it in:")
249 print(" python3 scripts/checks/mcdc_compound_ratchet.py --update")
250 return 0

257_ST_COVERED_C = """\
258ra8_err_t ra8_covered_fn(int c, int d)
259{
260 if (c && d) {
261 return k_ra8_ok;
262 }
263 return k_ra8_err;
264}
265"""
266
267_ST_UNCOVERED_C = """\
268ra8_err_t ra8_uncovered_fn(int a, int b)
269{
270 if (a || b) {
271 return k_ra8_ok;
272 }
273 return k_ra8_err;
274}
275"""
276
277_ST_SOUP_C = """\
278int soup_fn(int e, int f)
279{
280 if (e && f) {
281 return 1;
282 }
283 return 0;
284}
285"""
286
287_ST_TEST_C = """\
288/**
289 * @test ra8_covered_fn_mcdc
290 *
291 * @par MC/DC:
292 * Decision: `if (c && d)` cites libs/covered.c@ra8_covered_fn
293 * - Vector 1: c=1, d=1 -> true
294 * - Vector 2: c=0, d=1 -> false (varies c)
295 * - Vector 3: c=1, d=0 -> false (varies d)
296 */
297void test_mcdc_ra8_covered_fn(void) {}
298"""
299
300# A second uncovered decision appended to the ALREADY-uncovered function: the
301# "new decision lands in a function the baseline already knows about" case,
302# which a whole-file-count ratchet would catch but only a count-based one
303# catches at all -- a delta scan keyed on new source lines would also fire, and
304# both must.
305_ST_UNCOVERED_C_GROWN = """\
306ra8_err_t ra8_uncovered_fn(int a, int b)
307{
308 if (a || b) {
309 return k_ra8_ok;
310 }
311 if (a && !b) {
312 return k_ra8_err;
313 }
314 return k_ra8_err;
315}
316"""

320 """Write ``body`` to ``rel`` under ``root``, creating parent dirs."""
321 dst = root / rel
322 dst.parent.mkdir(parents=True, exist_ok=True)
323 dst.write_text(body, encoding="ascii")

326def _st_build_tree(root: Path) -> None:
327 """Lay down a fixture tree: one covered, one uncovered, one SOUP decision."""
328 _st_write(root, "libs/covered.c", _ST_COVERED_C)
329 _st_write(root, "libs/uncovered.c", _ST_UNCOVERED_C)

334def _selftest_scan(tmp: Path) -> list[str]:
335 """Assert the whole-tree measurement counts the right decisions and no others."""
336 failures: list[str] = []
337 root = tmp / "tree"
338 _st_build_tree(root)
339 counts, files, cites = scan(root)
340
341 if counts.get(("libs/uncovered.c", "ra8_uncovered_fn")) != 1:
342 failures.append("scan() did not count the uncovered decision in libs/uncovered.c")
343 if any(path == "libs/covered.c" for path, _fn in counts):
344 failures.append("scan() counted libs/covered.c, which carries MC/DC vectors")
345 if any("third_party" in path for path, _fn in counts):
346 failures.append("scan() counted a libs/third_party/ decision (SOUP is exempt)")
347 if sum(counts.values()) != 1:
348 failures.append(f"scan() found {sum(counts.values())} decision(s), expected exactly 1")
349
350 # The measurement must SEE the growth it is meant to gate on.
351 _st_write(root, "libs/uncovered.c", _ST_UNCOVERED_C_GROWN)
352 grown, _files, _cites = scan(root)
353 if grown.get(("libs/uncovered.c", "ra8_uncovered_fn")) != 2: # noqa: PLR2004 -- fixture count
354 failures.append("scan() did not see a second uncovered decision added to a known function")
355
356 # Scope scalars must be real, or the guards below have nothing to guard.
357 if files < 2 or cites < 1: # noqa: PLR2004 -- fixture plants 2 files, 1 citation
358 failures.append(
359 f"scan() reported an implausible scope: {files} file(s), {cites} citation(s)"
360 )
361 return failures

364def _selftest_ratchet() -> list[str]:
365 """Assertions about the growth verdict and baseline round-tripping."""
366 failures: list[str] = []
367
368 base = Counter({("libs/f.c", "fn_a"): 1})
369 # FAILS when a known bucket grows ...
370 if report(Counter({("libs/f.c", "fn_a"): 2}), base) == 0:
371 failures.append("report() passed a bucket that GREW above the baseline")
372 # ... when a new function in a known file appears ...
373 if report(Counter({("libs/f.c", "fn_a"): 1, ("libs/f.c", "fn_b"): 1}), base) == 0:
374 failures.append("report() passed a NEW function bucket in a baselined file")
375 # ... and when a file the baseline never saw appears.
376 if report(Counter({("libs/new.c", "fn_a"): 1}), base) == 0:
377 failures.append("report() passed a finding in a file absent from the baseline")
378 # PASSES when unchanged or shrinking -- the ratchet must not be a cliff.
379 if report(Counter({("libs/f.c", "fn_a"): 1}), base) != 0:
380 failures.append("report() failed an unchanged bucket")
381 if report(Counter(), base) != 0:
382 failures.append("report() failed a fully burned-down baseline")
383
384 # Round-trip: what is written must read back identically, or a re-baseline
385 # would silently reshape the debt it claims to be freezing.
386 original = BASELINE_FILE.read_text(encoding="ascii") if BASELINE_FILE.is_file() else None
387 try:
388 fixture = Counter({("libs/a.c", "fn_x"): 3, ("src/b.c", "fn_y"): 1})
389 write_baseline(fixture)
390 if load_baseline() != fixture:
391 failures.append("write_baseline()/load_baseline() did not round-trip")
392 finally:
393 if original is None:
394 BASELINE_FILE.unlink(missing_ok=True)
395 else:
396 BASELINE_FILE.write_text(original, encoding="ascii")
397
398 return failures
399

401def _selftest_scope_guard() -> list[str]:
402 """Assertions about the scope guards, in both directions."""
403 failures: list[str] = []
404 if scope_reason(MIN_PRODUCTION_FILES - 1, MIN_CITATIONS) is None:
405 failures.append("scope_reason() accepted a scan that saw too few production files")
406 if scope_reason(MIN_PRODUCTION_FILES, MIN_CITATIONS - 1) is None:
407 failures.append("scope_reason() accepted a scan with a truncated citation index")
408 if scope_reason(MIN_PRODUCTION_FILES, MIN_CITATIONS) is not None:
409 failures.append("scope_reason() rejected a scan that is exactly at both floors")
410 return failures

413def selftest() -> int:
414 """Assert the measurement and the ratchet fire, in BOTH directions.
415
416 Runs the REAL scan against a throwaway fixture tree, so a detector that
417 quietly stopped matching cannot pass as clean: it must COUNT the uncovered
418 decision, must NOT count the vector-covered or SOUP ones, and must see a
419 second uncovered decision appear in a function it already knew about.
420 """
421 with tempfile.TemporaryDirectory() as td:
422 failures = _selftest_scan(Path(td))
423 failures += _selftest_ratchet() + _selftest_scope_guard()
424
425 if failures:
426 print("SELFTEST FAILED:", file=sys.stderr)
427 for problem in failures:
428 print(f" - {problem}", file=sys.stderr)
429 return 1
430 print(
431 "selftest: MC/DC compound-decision ratchet OK "
432 "(counts an uncovered decision; ignores vector-covered and SOUP ones; "
433 "fails on growth, passes on shrinkage; refuses an implausible scope)."
434 )
435 return 0

452def main() -> int:
453 """Ratchet uncovered compound decisions against the committed baseline.
454
455 A ratchet, not a floor: ``--check`` fails when a count RISES, and
456 ``--update`` lowers the baseline once vectors are written. The baseline can
457 therefore only move downward, which is what stops a large legacy count from
458 being permanently accepted while still letting CI block a new one today.
459
460 Returns 0 when every count is at or below the baseline, 1 when one grew or
461 the scan's scope was not credible.
462 """
463 parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
464 parser.add_argument("--root", default=str(REPO_ROOT), help="tree to scan (default: this repo)")
465 parser.add_argument("--check", action="store_true", help="gate against the baseline")
466 parser.add_argument("--update", action="store_true", help="rewrite the baseline")
467 parser.add_argument("--list", action="store_true", help="print the whole backlog")
468 parser.add_argument("--selftest", action="store_true", help="assert this gate still fires")
469 args = parser.parse_args()
470
471 if args.selftest:
472 return selftest()
473 root = Path(args.root).resolve()
474 if args.list:
475 return _list_backlog(root)
476 if not (args.check or args.update):
477 parser.error("one of --check / --update / --list / --selftest is required")
478
479 current, files, citations = scan(root)
480
481 # Refuse to ratchet a scan whose scope never got established, in EITHER
482 # direction, before comparing or writing anything.
483 broken = scope_reason(files, citations)
484 if broken:
485 verb = "--update" if args.update else "--check"
486 print(f"refusing to {verb}: {broken}", file=sys.stderr)
487 return 1
488
489 if args.update:
490 baseline = load_baseline()
491 # Seeding the very first baseline necessarily "grows" every bucket from
492 # nothing, so the no-growth rule applies only once a baseline exists.
493 seeding = not BASELINE_FILE.is_file()
494 grew = [] if seeding else [k for k, n in current.items() if n > baseline.get(k, 0)]
495 if grew:
496 print(
497 f"refusing to --update: {len(grew)} bucket(s) would GROW. "
498 "The baseline is a burn-down; write the missing MC/DC vectors instead.",
499 file=sys.stderr,
500 )
501 for path, function in grew[:MAX_DETAIL_LINES]:
502 print(f" {path} {function}()", file=sys.stderr)
503 return 1
504 write_baseline(current)
505 print(
506 f"baseline updated: {sum(current.values())} uncovered compound decision(s) "
507 f"recorded across {len({p for p, _ in current})} file(s) "
508 f"({files} production file(s) scanned)."
509 )
510 return 0
511
512 print(
513 f"MC/DC ratchet: scanned {files} production file(s), "
514 f"{citations} citation(s) in tests/test_*.c."
515 )
516 return report(current, load_baseline())
