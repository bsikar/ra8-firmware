93static uint32_t es_framebuffer_hash(void)
94{
95 const uint8_t* p = (const uint8_t*)s_framebuffer;
96 const size_t n = sizeof(s_framebuffer);
97 uint32_t hsh = (uint32_t)k_es_fnv_offset;
98 for (size_t i = 0U; i < n; i++) {
99 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_es_fnv_prime;
100 }
101 return hsh;
102}

105static void es_print_hex(uint32_t value)
106{
107 uint8_t buf[k_es_hex_nibbles];
108 for (uint32_t i = 0U; i < (uint32_t)k_es_hex_nibbles; i++) {
109 const uint32_t shift = ((uint32_t)k_es_hex_nibbles - 1U - i) * (uint32_t)k_es_nibble_bits;
110 const uint32_t nib = (value >> shift) & (uint32_t)k_es_nibble_mask;
111 buf[i] = (uint8_t)((nib < (uint32_t)k_es_dec_ten) ? ('0' + nib) : ('A' + (nib - k_es_dec_ten)));
112 }
113 es_print(buf, (uint32_t)k_es_hex_nibbles);
114}

117static void es_print_uint(uint32_t value)
118{
119 uint8_t buf[k_es_dec_ten];
120 uint32_t n = 0U;
121 if (value == 0U) {
122 buf[n] = '0';
123 n++;
124 }
125 while ((value > 0U) && (n < (uint32_t)k_es_dec_ten)) {
126 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_es_dec_ten));
127 n++;
128 value /= (uint32_t)k_es_dec_ten;
129 }
130 for (uint32_t i = 0U; i < n; i++) {
131 es_print(&buf[n - 1U - i], 1U);
132 }
133}
