180RA8_INTERNAL
181static void priv_decode_one(const uint8_t* raw, ra8_touch_point_t* out)
182{
183 out->track_id = raw[k_ra8_touch_gt911_point_off_track];
184 out->x = (uint16_t)((uint32_t)raw[k_ra8_touch_gt911_point_off_x_lsb] |
185 ((uint32_t)raw[k_ra8_touch_gt911_point_off_x_msb] << k_ra8_touch_byte_shift));
186 out->y = (uint16_t)((uint32_t)raw[k_ra8_touch_gt911_point_off_y_lsb] |
187 ((uint32_t)raw[k_ra8_touch_gt911_point_off_y_msb] << k_ra8_touch_byte_shift));
188 /* Pressure is the low byte of the 16-bit "size" field; the high byte
189 * is rarely populated and is dropped here so the public type stays
190 * one byte. */
191 out->pressure = raw[k_ra8_touch_gt911_point_off_size_lsb];
192}

211RA8_INTERNAL
212static void priv_decode_block(const uint8_t* raw,
213 uint8_t n_points,
214 ra8_touch_point_t* out,
215 uint8_t max_count,
216 uint8_t* got_count)
217{
218 uint8_t emit = n_points;
219 if (emit > max_count) {
220 emit = max_count;
221 }
222 if (emit > k_ra8_touch_gt911_max_points) {
223 emit = k_ra8_touch_gt911_max_points;
224 }
225 for (uint8_t i = 0U; i < emit; i++) {
226 const size_t off = (size_t)i * (size_t)k_ra8_touch_gt911_point_bytes;
227 priv_decode_one(&raw[off], &out[i]);
228 }
229 *got_count = emit;
230}

509RA8_INTERNAL
510static ra8_err_t
511priv_read_inner(ra8_touch_point_t* out_points, uint8_t max_count, uint8_t* got_count)
512{
513 /* Step 1: read the status byte. */
514 uint8_t status = 0U;
515 const ra8_err_t status_err = priv_gt911_read(k_ra8_touch_gt911_reg_status, &status, 1U);
516 if (status_err != k_ra8_ok) {
517 *got_count = 0U;
518 return k_ra8_err_hw_error;
519 }
520
521 /* Step 2: bit7 must be set, otherwise no frame is ready. */
522 if ((status & k_ra8_touch_gt911_status_ready_mask) == 0U) {
523 *got_count = 0U;
524 return k_ra8_ok;
525 }
526
527 /* Step 3 + 4: clamp, pull the per-point block, decode. */
528 const uint8_t emit = priv_clamp_emit(status, max_count);
529 uint8_t raw[(size_t)k_ra8_touch_gt911_max_points * (size_t)k_ra8_touch_gt911_point_bytes] = {};
530 if (emit > 0U) {
531 const uint32_t bytes = (uint32_t)emit * (uint32_t)k_ra8_touch_gt911_point_bytes;
532 const ra8_err_t blk_err = priv_gt911_read(k_ra8_touch_gt911_reg_point0, raw, bytes);
533 if (blk_err != k_ra8_ok) {
534 *got_count = 0U;
535 priv_ack_frame();
536 return k_ra8_err_hw_error;
537 }
538 priv_decode_block(raw, emit, out_points, max_count, got_count);
539 } else {
540 /* Frame-ready with zero contacts: a full-release frame. Nothing to
541 * decode -- fall through to the ack so the IC latches the next one. */
542 *got_count = 0U;
543 }
544
545 /* Step 5: ack the frame so the IC latches the next one. */
546 priv_ack_frame();
547 return k_ra8_ok;
548}
