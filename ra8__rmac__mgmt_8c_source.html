118ra8_err_t ra8_rmac_clear_status(ra8_rmac_port_t port,
119 uint32_t err_mask,
120 uint32_t mon0_mask,
121 uint32_t mon1_mask,
122 uint32_t mon2_mask)
123{
124 if (!internal_port_ok(port)) {
125 ra8_log_error(s_tag, "rmac_clear_status: port out of range");
126 return k_ra8_err_invalid_arg;
127 }
128 volatile r_rmac_regs_t* reg = ra8_rmac(port);
129 /* The disable registers act as the clear-on-write counterpart of
130 * each status register; writing 1 to a bit clears the matching bit
131 * in MEIS / MMIS{0,1,2}. The driver also writes the explicit
132 * masked-out value so fake backings (which lack RW1C) end up
133 * in the same observable state as real hardware. */
134 /* HUM Ch 33.4 "MEID : MAC Error Interrupt Disable Register" p 1706 */
135 reg->MEID = err_mask;
136 /* HUM Ch 33.4 "MMID0 : MAC Monitoring Interrupt Disable Register 0" p 1706 */
137 reg->MMID0 = mon0_mask;
138 /* HUM Ch 33.4 "MMID1 : MAC Monitoring Interrupt Disable Register 1" p 1706 */
139 reg->MMID1 = mon1_mask;
140 /* HUM Ch 33.4 "MMID2 : MAC Monitoring Interrupt Disable Register 2" p 1706 */
141 reg->MMID2 = mon2_mask;
142 /* HUM Ch 33.4 "MEIS : MAC Error Interrupt Status Register" p 1706 */
143 reg->MEIS = reg->MEIS & ~err_mask;
144 /* HUM Ch 33.4 "MMIS0 : MAC Monitoring Interrupt Status Register 0" p 1706 */
145 reg->MMIS0 = reg->MMIS0 & ~mon0_mask;
146 /* HUM Ch 33.4 "MMIS1 : MAC Monitoring Interrupt Status Register 1" p 1706 */
147 reg->MMIS1 = reg->MMIS1 & ~mon1_mask;
148 /* HUM Ch 33.4 "MMIS2 : MAC Monitoring Interrupt Status Register 2" p 1706 */
149 reg->MMIS2 = reg->MMIS2 & ~mon2_mask;
150 return k_ra8_ok;
151}

332ra8_err_t ra8_rmac_phy_reset(ra8_rmac_port_t port, uint8_t phy_addr)
333{
334 if (!internal_phy_args_ok(port, phy_addr)) {
335 ra8_log_error(s_tag, "phy_reset: bad args");
336 return k_ra8_err_invalid_arg;
337 }
338 /* IEEE 802.3 Clause 22 sec 22.2.4.1.1 "BMCR.RESET" -- self-clearing.
339 * HUM Ch 33.4.1.1 "MPSM" p 1707 carries the MDIO write. */
340 const ra8_err_t w = ra8_rmac_mdio_c22_write(port,
341 phy_addr,
342 (uint8_t)k_ra8_rmac_phy_reg_bmcr,
343 (uint16_t)k_ra8_rmac_phy_bmcr_reset);
344 if (w != k_ra8_ok) {
345 /* Reached on host by arming the ra8_fake_mmio seam on MPSM so the
346 * MDIO write's drain or post-wait times out. */
347 ra8_log_error(s_tag, "phy_reset: bmcr write");
348 return w;
349 }
350 for (uint32_t i = 0U; i < (uint32_t)k_ra8_rmac_phy_reset_iter_cap; ++i) {
351 uint16_t bmcr = 0U;
352 const ra8_err_t r =
353 ra8_rmac_mdio_c22_read(port, phy_addr, (uint8_t)k_ra8_rmac_phy_reg_bmcr, &bmcr);
354 if (r != k_ra8_ok) {
355 /* Reached on host via ra8_fake_mmio_fail_nth_wait on MPSM: the
356 * write's two MPSM wait-loops succeed, the read's drain fails. */
357 ra8_log_error(s_tag, "phy_reset: bmcr read");
358 return r;
359 }
360 if ((bmcr & (uint16_t)k_ra8_rmac_phy_bmcr_reset) == 0U) {
361 return k_ra8_ok;
362 }
363 /* GCOVR_EXCL_START
364 * Off-target MDIO returns BMCR = 0 on every read (ra8_rmac.c
365 * internal_mpsm_issue writes PRD = 0 for reads and the pure-RAM
366 * MPSM readback returns that word), so BMCR.RESET always reads
367 * clear and the loop returns above on the first iteration. The
368 * natural loop exit and this timeout leg require a PHY that holds
369 * BMCR.RESET asserted -- read DATA the ra8_fake_mmio wait seam
370 * cannot synthesize. */
371 }
372 ra8_log_error(s_tag, "phy_reset: bmcr.reset never cleared");
373 return k_ra8_err_hw_timeout;
374 /* GCOVR_EXCL_STOP */
375}

403ra8_err_t ra8_rmac_phy_auto_neg_wait(ra8_rmac_port_t port,
404 uint8_t phy_addr,
405 uint32_t timeout_ms,
406 ra8_rmac_phy_link_t* out_link)
407{
408 RA8_CHECK_NULL_PTR(out_link, s_tag, "phy_auto_neg_wait: out_link null");
409 if (!internal_phy_args_ok(port, phy_addr)) {
410 ra8_log_error(s_tag, "phy_auto_neg_wait: bad args");
411 return k_ra8_err_invalid_arg;
412 }
413 out_link->up = false;
414 out_link->speed = k_ra8_rmac_phy_speed_unknown;
415
416 /* 100 iters per ms = 10us per spin (k_ra8_rmac_phy_anwait_us_per_iter). */
417 uint32_t cap = (timeout_ms == 0U) ? (uint32_t)k_ra8_rmac_phy_anwait_iter_cap
418 : (timeout_ms * (uint32_t)k_ra8_rmac_phy_anwait_iters_per_ms);
419 if (cap == 0U) {
420 cap = 1U;
421 }
422
423 for (uint32_t i = 0U; i < cap; ++i) {
424 uint16_t bmsr = 0U;
425 /* IEEE 802.3 Clause 22 sec 22.2.4.2 "BMSR" -- AN_COMPLETE bit 5,
426 * LINK_STATUS bit 2. Need both set for a "ready to forward" link. */
427 const ra8_err_t r =
428 ra8_rmac_mdio_c22_read(port, phy_addr, (uint8_t)k_ra8_rmac_phy_reg_bmsr, &bmsr);
429 if (r != k_ra8_ok) {
430 /* Reached on host by arming the ra8_fake_mmio seam on MPSM. */
431 return r;
432 }
433 const uint16_t need =
434 (uint16_t)((uint16_t)k_ra8_rmac_phy_bmsr_an_done | (uint16_t)k_ra8_rmac_phy_bmsr_link_up);
435 if ((bmsr & need) == need) {
436 /* GCOVR_EXCL_START
437 * Off-target MDIO delivers BMSR = 0 on every read (ra8_rmac.c
438 * internal_mpsm_issue writes PRD = 0 for reads -- read DATA the
439 * ra8_fake_mmio wait seam cannot synthesize), so (bmsr & need)
440 * is never equal to need and this link-up read-back body -- the
441 * ANLPAR fetch, its MDIO-error return, and the resolved-speed
442 * decode -- is unreachable from the host. */
443 uint16_t anlpar = 0U;
444 /* IEEE 802.3 Clause 28.2.4.4 "ANLPAR" -- resolved capability. */
445 const ra8_err_t lp =
446 ra8_rmac_mdio_c22_read(port, phy_addr, (uint8_t)k_ra8_rmac_phy_reg_anlpar, &anlpar);
447 if (lp != k_ra8_ok) {
448 return lp;
449 }
450 out_link->up = true;
451 out_link->speed = internal_decode_anlpar(anlpar);
452 return k_ra8_ok;
453 /* GCOVR_EXCL_STOP */
454 }
455 }
456 ra8_log_error(s_tag, "phy_auto_neg_wait: timeout");
457 return k_ra8_err_hw_timeout;
458}

461ra8_rmac_phy_link_status(ra8_rmac_port_t port, uint8_t phy_addr, ra8_rmac_phy_link_t* out_link)
462{
463 RA8_CHECK_NULL_PTR(out_link, s_tag, "phy_link_status: out_link null");
464 if (!internal_phy_args_ok(port, phy_addr)) {
465 ra8_log_error(s_tag, "phy_link_status: bad args");
466 return k_ra8_err_invalid_arg;
467 }
468 out_link->up = false;
469 out_link->speed = k_ra8_rmac_phy_speed_unknown;
470
471 uint16_t bmsr = 0U;
472 /* IEEE 802.3 Clause 22 sec 22.2.4.2 "BMSR.LINK_STATUS" (bit 2). */
473 const ra8_err_t r =
474 ra8_rmac_mdio_c22_read(port, phy_addr, (uint8_t)k_ra8_rmac_phy_reg_bmsr, &bmsr);
475 if (r != k_ra8_ok) {
476 /* Reached on host by arming the ra8_fake_mmio seam on MPSM. */
477 return r;
478 }
479 out_link->up = (bmsr & (uint16_t)k_ra8_rmac_phy_bmsr_link_up) != 0U;
480 // mcdc-deactivated: ra8_rmac_phy_auto_neg_start link-up + an-done gate; both bits come from the same BMSR read; PHY hardware sets BMSR.AN_DONE only after BMSR.LINK_STATUS asserts (IEEE 802.3 Clause 22 22.2.4.2 ordering) -- the second condition cannot be true while the first is false on any conformant PHY.
481 if (out_link->up && ((bmsr & (uint16_t)k_ra8_rmac_phy_bmsr_an_done) != 0U)) {
482 /* GCOVR_EXCL_START
483 * Same off-target MDIO limitation: BMSR reads 0, so out_link->up is
484 * always false and this AN-resolved-speed read-back body (ANLPAR
485 * fetch, MDIO-error return, and decode) is unreachable from the
486 * host. */
487 uint16_t anlpar = 0U;
488 const ra8_err_t lp =
489 ra8_rmac_mdio_c22_read(port, phy_addr, (uint8_t)k_ra8_rmac_phy_reg_anlpar, &anlpar);
490 if (lp != k_ra8_ok) {
491 return lp;
492 }
493 out_link->speed = internal_decode_anlpar(anlpar);
494 }
495 /* GCOVR_EXCL_STOP */
496 return k_ra8_ok;
497}
