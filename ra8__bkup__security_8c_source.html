120{
121 RA8_CHECK_NULL_PTR(cfg, s_bkup_tag, "security cfg must not be nullptr");
122 const ra8_err_t v_err = internal_validate_security_cfg(cfg);
123 RA8_RETURN_ON_ERROR(v_err, s_bkup_tag, "security_apply: cfg bad"); /* GCOVR_EXCL_BR_LINE */
124
125 /* BBFSAR / VBRSABAR / VBRPABARS / VBRPABARNS are security-attribution
126 * registers, so they sit behind PRC4 -- not the PRC1 that guards the rest
127 * of this block (HUM Ch 13.1 Table 13.1 p 521). */
128 RA8_PROTECTED_WRITE(k_ra8_prcr_unlock_sar)
129 {
130 /* HUM Ch 12.2.1 "BBFSAR : Battery Backup Function Security Attribute Register", p 500 */
131 *ra8_bkup_bbfsar() = (uint32_t)(cfg->bbfsar & k_ra8_bkup_bbfsar_mask_all);
132 /* HUM Ch 12.2.2 "VBRSABAR : VBATT Backup Register Security Attribute Boundary Address Register", p 502 */
133 *ra8_bkup_vbrsabar() = cfg->saba;
134 /* HUM Ch 12.2.3 "VBRPABARS : VBATT Backup Register Privilege Attribute Boundary Address Register for Secure Region", p 502 */
135 *ra8_bkup_vbrpabars() = cfg->pabas;
136 /* HUM Ch 12.2.4 "VBRPABARNS : VBATT Backup Register Privilege Attribute Boundary Address Register for Non-secure Region", p 503 */
137 *ra8_bkup_vbrpabarns() = cfg->pabans;
138 }
139 return k_ra8_ok;
140}
