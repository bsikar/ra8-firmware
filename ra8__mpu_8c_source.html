72/* Map a (priv, unpriv) permission pair onto the Armv8-M AP code -- see implementation for details. */
73static uint8_t ra8_mpu_encode_ap(ra8_mpu_perm_t priv, ra8_mpu_perm_t unpriv)
74{
75 /* Arm Cortex-M85 TRM "MPU_RBAR" -- AP[1:0] table. */
76 if (priv == k_ra8_mpu_perm_rw && unpriv == k_ra8_mpu_perm_none) {
77 return (uint8_t)k_ra8_mpu_ap_priv_rw_unpriv_none;
78 }
79 if (priv == k_ra8_mpu_perm_rw && unpriv == k_ra8_mpu_perm_rw) {
80 return (uint8_t)k_ra8_mpu_ap_priv_rw_unpriv_rw;
81 }
82 if (priv == k_ra8_mpu_perm_ro && unpriv == k_ra8_mpu_perm_none) {
83 return (uint8_t)k_ra8_mpu_ap_priv_ro_unpriv_none;
84 }
85 if (priv == k_ra8_mpu_perm_ro && unpriv == k_ra8_mpu_perm_ro) {
86 return (uint8_t)k_ra8_mpu_ap_priv_ro_unpriv_ro;
87 }
88 return (uint8_t)k_ra8_mpu_ap_invalid;
89}

100/* Validate a single region descriptor -- see implementation for details. */
101static ra8_err_t ra8_mpu_validate_region(const ra8_mpu_region_t* r)
102{
103 if (!ra8_mpu_is_pow2(r->size) || r->size < (uint32_t)k_ra8_mpu_min_region_size) {
104 return k_ra8_err_invalid_arg;
105 }
106 if ((r->base & (uintptr_t)(r->size - 1U)) != 0U) {
107 return k_ra8_err_invalid_arg;
108 }
109 if (ra8_mpu_encode_ap(r->priv, r->unpriv) == (uint8_t)k_ra8_mpu_ap_invalid) {
110 return k_ra8_err_invalid_arg;
111 }
112 return k_ra8_ok;
113}

115/* Build the MPU_RBAR value for a region descriptor -- see implementation for details. */
116static uint32_t ra8_mpu_build_rbar(const ra8_mpu_region_t* r)
117{
118 /* Arm Cortex-M85 TRM "MPU_RBAR": BASE[31:5] | SH[4:3] | AP[2:1] | XN[0]. */
119 const uint32_t base = (uint32_t)r->base & (uint32_t)k_ra8_mpu_rbar_base_mask;
120 const uint32_t ap =
121 ((uint32_t)ra8_mpu_encode_ap(r->priv, r->unpriv) << (uint32_t)k_ra8_mpu_rbar_ap_shift) &
122 (uint32_t)k_ra8_mpu_rbar_ap_mask;
123 const uint32_t sh = ((uint32_t)r->shareable << (uint32_t)k_ra8_mpu_rbar_sh_shift) &
124 (uint32_t)k_ra8_mpu_rbar_sh_mask;
125 const uint32_t xn = r->executable ? 0U : (uint32_t)k_ra8_mpu_rbar_xn_mask;
126 return base | sh | ap | xn;
127}

129/* Build the MPU_RLAR value for a region descriptor -- see implementation for details. */
130static uint32_t ra8_mpu_build_rlar(const ra8_mpu_region_t* r)
131{
132 /* Arm Cortex-M85 TRM "MPU_RLAR": LIMIT[31:5] | AttrIndx[3:1] | EN[0]. */
133 const uint32_t limit_inclusive = (uint32_t)(r->base + r->size - 1U);
134 const uint32_t limit = limit_inclusive & (uint32_t)k_ra8_mpu_rlar_limit_mask;
135 const uint32_t idx = ((uint32_t)r->attr_idx << (uint32_t)k_ra8_mpu_rlar_attridx_shift) &
136 (uint32_t)k_ra8_mpu_rlar_attridx_mask;
137 return limit | idx | (uint32_t)k_ra8_mpu_rlar_en_mask;
138}

186/* Validate the whole config block before any register write -- see implementation for details. */
187static ra8_err_t ra8_mpu_validate_cfg(const ra8_mpu_cfg_t* cfg)
188{
189 const uint8_t implemented = ra8_mpu_dregion_count();
190 if (cfg->region_count > implemented) {
191 return k_ra8_err_invalid_arg;
192 }
193 if (cfg->region_count > 0U && cfg->regions == nullptr) {
194 return k_ra8_err_null_ptr;
195 }
196 for (uint8_t i = 0U; i < cfg->region_count; ++i) {
197 const ra8_err_t err = ra8_mpu_validate_region(&cfg->regions[i]);
198 if (err != k_ra8_ok) {

205ra8_err_t ra8_mpu_configure(const ra8_mpu_cfg_t* cfg)
206{
207 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
208 const ra8_err_t verr = ra8_mpu_validate_cfg(cfg);
209 if (verr != k_ra8_ok) {
210 return verr;
211 }
212
213 /* Arm Cortex-M85 TRM "MPU_CTRL": disable before reprogramming. */
214 ra8_mpu_write_ctrl(0U);
215 ra8_mpu_write_mair(cfg->mair0, cfg->mair1);
216
217 for (uint8_t i = 0U; i < cfg->region_count; ++i) {
218 ra8_mpu_program_region(i, &cfg->regions[i]);
219 }
220 const uint8_t implemented = ra8_mpu_dregion_count();
221 for (uint8_t i = cfg->region_count; i < implemented; ++i) {
222 ra8_mpu_clear_region(i);
223 }
224 ra8_mpu_write_ctrl(ra8_mpu_build_ctrl(cfg));
225
226 /* Enable MemManage fault dispatch via SHCSR.MEMFAULTENA (bit 16).
227 * Without this bit, every MPU permission violation silently escalates
228 * to HardFault -- the strong MemManage_Handler the application
229 * installed never runs. Armv8-M ARM "B3.2.10 SHCSR, System Handler
230 * Control and State Register" + Cortex-M85 TRM SCB+0x24. The MPU is
231 * useless as a runtime safety mechanism without this enable; making
232 * it the default of ra8_mpu_configure means every consumer gets
233 * working MemFault delivery without having to know about the
234 * separate SHCSR concern. Callers that genuinely want
235 * MemFault->HardFault escalation can clear SHCSR.MEMFAULTENA
236 * themselves after the configure call. */
237 volatile uint32_t* shcsr = (volatile uint32_t*)k_ra8_mpu_shcsr_addr;
238 *shcsr |= (uint32_t)k_ra8_mpu_shcsr_memfaultena;
239 return k_ra8_ok;
240}

416ra8_err_t ra8_mpu_apply_boot_map(void)
417{
418 /* Precondition: the silicon must implement at least the map's region count.
419 On the RA8D2 M85 (16 regions) this always holds; the guard keeps the boot
420 path from installing a truncated, half-covered map on an unexpected part. */
421 const uint8_t implemented = ra8_mpu_dregion_count();
422 if (implemented < (uint8_t)k_ra8_mpu_boot_region_count) {
423 return k_ra8_err_invalid_arg;
424 }
425
426 /* Arm Cortex-M85 TRM "MPU_CTRL": disable before reprogramming. */
427 ra8_mpu_write_ctrl(0U);
428 ra8_mpu_write_mair((uint32_t)k_ra8_mpu_boot_mair0, (uint32_t)k_ra8_mpu_boot_mair1);
429
430 /* Clear any stale high-index regions FIRST, then install the map, so an
431 aborted reprogram never leaves a stale enabled region beside the new map,
432 and the last register write reflects the highest map region. */
433 for (uint8_t i = (uint8_t)k_ra8_mpu_boot_region_count; i < implemented; ++i) {
434 ra8_mpu_clear_region(i);
435 }
436 for (uint8_t i = 0U; i < (uint8_t)k_ra8_mpu_boot_region_count; ++i) {
437 ra8_mpu_program_region(i, &s_ra8_mpu_boot_regions[i]);
438 }
439
440 /* Retire the region writes, enable the MPU with PRIVDEFENA, then flush the
441 pipeline so the very next access sees the new attribute map. */
442 ra8_hw_dsb();
443 ra8_mpu_write_ctrl((uint32_t)k_ra8_mpu_ctrl_enable | (uint32_t)k_ra8_mpu_ctrl_privdefena);
444 ra8_hw_dsb();
445 ra8_hw_isb();
446 return k_ra8_ok;
447}
