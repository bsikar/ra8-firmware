66/* Map a (priv, unpriv) permission pair onto the Armv8-M AP code -- see implementation for details. */
67static uint8_t ra8_mpu_encode_ap(ra8_mpu_perm_t priv, ra8_mpu_perm_t unpriv)
68{
69 /* Arm Cortex-M85 TRM "MPU_RBAR" -- AP[1:0] table. */
70 if (priv == k_ra8_mpu_perm_rw && unpriv == k_ra8_mpu_perm_none) {
71 return (uint8_t)k_ra8_mpu_ap_priv_rw_unpriv_none;
72 }
73 if (priv == k_ra8_mpu_perm_rw && unpriv == k_ra8_mpu_perm_rw) {
74 return (uint8_t)k_ra8_mpu_ap_priv_rw_unpriv_rw;
75 }
76 if (priv == k_ra8_mpu_perm_ro && unpriv == k_ra8_mpu_perm_none) {
77 return (uint8_t)k_ra8_mpu_ap_priv_ro_unpriv_none;
78 }
79 if (priv == k_ra8_mpu_perm_ro && unpriv == k_ra8_mpu_perm_ro) {
80 return (uint8_t)k_ra8_mpu_ap_priv_ro_unpriv_ro;
81 }
82 return (uint8_t)k_ra8_mpu_ap_invalid;
83}

85/* Read the implemented region count from `MPU_TYPE -- see implementation for details. */

94/* Validate a single region descriptor -- see implementation for details. */
95static ra8_err_t ra8_mpu_validate_region(const ra8_mpu_region_t* r)
96{
97 if (!ra8_mpu_is_pow2(r->size) || r->size < (uint32_t)k_ra8_mpu_min_region_size) {
98 return k_ra8_err_invalid_arg;
99 }
100 if ((r->base & (uintptr_t)(r->size - 1U)) != 0U) {
101 return k_ra8_err_invalid_arg;
102 }
103 if (ra8_mpu_encode_ap(r->priv, r->unpriv) == (uint8_t)k_ra8_mpu_ap_invalid) {
104 return k_ra8_err_invalid_arg;
105 }
106 return k_ra8_ok;
107}

109/* Build the MPU_RBAR value for a region descriptor -- see implementation for details. */
110static uint32_t ra8_mpu_build_rbar(const ra8_mpu_region_t* r)
111{
112 /* Arm Cortex-M85 TRM "MPU_RBAR": BASE[31:5] | SH[4:3] | AP[2:1] | XN[0]. */
113 const uint32_t base = (uint32_t)r->base & (uint32_t)k_ra8_mpu_rbar_base_mask;
114 const uint32_t ap =
115 ((uint32_t)ra8_mpu_encode_ap(r->priv, r->unpriv) << (uint32_t)k_ra8_mpu_rbar_ap_shift) &
116 (uint32_t)k_ra8_mpu_rbar_ap_mask;
117 const uint32_t sh = ((uint32_t)r->shareable << (uint32_t)k_ra8_mpu_rbar_sh_shift) &
118 (uint32_t)k_ra8_mpu_rbar_sh_mask;
119 const uint32_t xn = r->executable ? 0U : (uint32_t)k_ra8_mpu_rbar_xn_mask;
120 return base | sh | ap | xn;
121}

123/* Build the MPU_RLAR value for a region descriptor -- see implementation for details. */
124static uint32_t ra8_mpu_build_rlar(const ra8_mpu_region_t* r)
125{
126 /* Arm Cortex-M85 TRM "MPU_RLAR": LIMIT[31:5] | AttrIndx[3:1] | EN[0]. */
127 const uint32_t limit_inclusive = (uint32_t)(r->base + r->size - 1U);
128 const uint32_t limit = limit_inclusive & (uint32_t)k_ra8_mpu_rlar_limit_mask;
129 const uint32_t idx = ((uint32_t)r->attr_idx << (uint32_t)k_ra8_mpu_rlar_attridx_shift) &
130 (uint32_t)k_ra8_mpu_rlar_attridx_mask;
131 return limit | idx | (uint32_t)k_ra8_mpu_rlar_en_mask;
132}

164/* Validate the whole config block before any register write -- see implementation for details. */
165static ra8_err_t ra8_mpu_validate_cfg(const ra8_mpu_cfg_t* cfg)
166{
167 const uint8_t implemented = ra8_mpu_dregion_count();
168 if (cfg->region_count > implemented) {
169 return k_ra8_err_invalid_arg;
170 }
171 if (cfg->region_count > 0U && cfg->regions == nullptr) {
172 return k_ra8_err_null_ptr;
173 }
174 for (uint8_t i = 0U; i < cfg->region_count; ++i) {
175 const ra8_err_t err = ra8_mpu_validate_region(&cfg->regions[i]);
176 if (err != k_ra8_ok) {
177 return err;
178 }
179 }
180 return k_ra8_ok;
181}

183ra8_err_t ra8_mpu_configure(const ra8_mpu_cfg_t* cfg)
184{
185 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
186 const ra8_err_t verr = ra8_mpu_validate_cfg(cfg);
187 if (verr != k_ra8_ok) {
188 return verr;
189 }
190
191 volatile r_mpu_regs_t* mpu = ra8_mpu_regs();
192 /* Arm Cortex-M85 TRM "MPU_CTRL": disable before reprogramming. */
193 mpu->CTRL = 0U;
194 mpu->MAIR0 = cfg->mair0;
195 mpu->MAIR1 = cfg->mair1;
196
197 for (uint8_t i = 0U; i < cfg->region_count; ++i) {
198 ra8_mpu_program_region(i, &cfg->regions[i]);
199 }
200 const uint8_t implemented = ra8_mpu_dregion_count();
201 for (uint8_t i = cfg->region_count; i < implemented; ++i) {
202 ra8_mpu_clear_region(i);
203 }
204 mpu->CTRL = ra8_mpu_build_ctrl(cfg);
205
206 /* Enable MemManage fault dispatch via SHCSR.MEMFAULTENA (bit 16).
207 * Without this bit, every MPU permission violation silently escalates
208 * to HardFault -- the strong MemManage_Handler the application
209 * installed never runs. Armv8-M ARM "B3.2.10 SHCSR, System Handler
210 * Control and State Register" + Cortex-M85 TRM SCB+0x24. The MPU is
211 * useless as a runtime safety mechanism without this enable; making
212 * it the default of ra8_mpu_configure means every consumer gets
213 * working MemFault delivery without having to know about the
214 * separate SHCSR concern. Callers that genuinely want
215 * MemFault->HardFault escalation can clear SHCSR.MEMFAULTENA
216 * themselves after the configure call. */
217 volatile uint32_t* shcsr = (volatile uint32_t*)k_ra8_mpu_shcsr_addr;
218 *shcsr |= (uint32_t)k_ra8_mpu_shcsr_memfaultena;
219 return k_ra8_ok;
220}
