249ra8_err_t ra8_usb_paud_init(ra8_usb_speed_t speed)
250{
251 if ((speed != k_ra8_usb_speed_fs) && (speed != k_ra8_usb_speed_hs)) {
252 return k_ra8_err_invalid_arg;
253 }
254 const ra8_err_t usb_err = ra8_usb_device_init(speed);
255 if (usb_err != k_ra8_ok) {
256 ra8_log_error_val(s_tag, "ra8_usb_device_init failed", (uint32_t)usb_err);
257 return k_ra8_err_hw_init_failed;
258 }
259 internal_reset_shadow(speed);
260
261 const ra8_err_t pipes_err = internal_configure_pipes(speed);
262 if (pipes_err != k_ra8_ok) {
263 (void)ra8_usb_device_deinit(speed);
264 return pipes_err;
265 }
266 s_state.initialized = true;
267 ra8_log_info_val(s_tag, "device-Audio ready", (uint32_t)speed);
268 return k_ra8_ok;
269}
270
271ra8_err_t ra8_usb_paud_close(void)
272{
273 if (!s_state.initialized) {
274 return k_ra8_err_invalid_state;
275 }
276 (void)ra8_usb_device_attach(s_state.speed, false);
277 const ra8_err_t err = ra8_usb_device_deinit(s_state.speed);
278 s_state.initialized = false;
279 s_state.desc = nullptr;
280 s_state.setup_cb = nullptr;
281 s_state.setup_ctx = nullptr;
282 return err;
283}

309ra8_err_t ra8_usb_paud_send_frame(const uint8_t* frame, uint16_t len)
310{
311 if (!s_state.initialized) {
312 return k_ra8_err_invalid_state;
313 }
314 if ((frame == nullptr) && (len != 0U)) {
315 return k_ra8_err_null_ptr;
316 }
317 if ((len == 0U) || (len > s_state.iso_max_packet)) {
318 return k_ra8_err_invalid_arg;
319 }
320 return ra8_usb_queue_in(s_state.speed, k_ra8_paud_pipe_iso_in, frame, len);
321}

323ra8_err_t ra8_usb_paud_recv_frame(uint8_t* buf, uint16_t max_len, uint16_t* got_len)
324{
325 RA8_CHECK_NULL_PTR(buf, s_tag, "recv_frame: buf");
326 RA8_CHECK_NULL_PTR(got_len, s_tag, "recv_frame: got_len");
327 if (!s_state.initialized) {
328 return k_ra8_err_invalid_state;
329 }
330 if (max_len == 0U) {
331 return k_ra8_err_invalid_arg;
332 }
333 uint16_t inout_len = max_len;
334 const ra8_err_t err =
335 ra8_usb_queue_out(s_state.speed, k_ra8_paud_pipe_iso_out, buf, &inout_len, true);
336 if (err == k_ra8_ok) {
337 *got_len = inout_len;
338 } else {
339 *got_len = 0U;
340 }
341 return err;
342}

417
418ra8_err_t ra8_usb_paud_handle_setup(const ra8_usb_setup_t* setup)
419{
420 RA8_CHECK_NULL_PTR(setup, s_tag, "handle_setup: setup");
421 if (!s_state.initialized) {
422 return k_ra8_err_invalid_state;
423 }
424 if (!internal_is_class_envelope(setup->bm_request_type)) {
425 return k_ra8_err_not_supported;
426 }
427 if (!internal_is_known_class_request(setup->b_request)) {
428 return k_ra8_err_not_supported;
429 }
430 if (s_state.setup_cb != nullptr) {
431 const ra8_err_t cb_err = s_state.setup_cb(s_state.setup_ctx, setup);
432 if (cb_err != k_ra8_ok) {
433 return ra8_usb_control_response(s_state.speed, false);
434 }
435 }
436 return ra8_usb_control_response(s_state.speed, true);
