81static void fake_record(fake_net_t* f, const char* url, const mdl_net_req_t* req)

534static bool atom_write(const char* path, const char* body)
535{
536 FILE* f = fopen(path, "wb");
537 if (f == nullptr) {
538 return false;
539 }
540 const size_t n = strlen(body);
541 const bool ok = (fwrite(body, 1U, n, f) == n);
542 return (fclose(f) == 0) && ok;
543}

546static long atom_read(const char* path, char* out, size_t cap)
547{
548 FILE* f = fopen(path, "rb");
549 if (f == nullptr) {
550 return -1;
551 }
552 const size_t n = fread(out, 1U, cap - 1U, f);
553 out[n] = '\0';
554 (void)fclose(f);
555 return (long)n;
556}

559static bool atom_has_debris(const char* dir)
560{
561 DIR* d = opendir(dir);
562 if (d == nullptr) {
563 return false;
564 }
565 bool found = false;
566 const struct dirent* e = readdir(d);
567 while (e != nullptr) {
568 if (strncmp(e->d_name, ".mdl-tmp-", strlen(".mdl-tmp-")) == 0) {
569 found = true;
570 break;
571 }
572 e = readdir(d);
573 }
574 (void)closedir(d);
575 return found;
576}
