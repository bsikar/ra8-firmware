193static uint32_t priv_fnv1a(const void* key, uint32_t key_bytes)
194{
195 const uint8_t* p = (const uint8_t*)key;
196 uint32_t h = (uint32_t)k_keycache_fnv_offset;
197 for (uint32_t i = 0U; i < key_bytes; ++i) {

367RA8_INTERNAL
368static void priv_hash_remove(ra8_keycache_t* kc, int32_t f)
369{
370 const uint32_t b = priv_hash(kc, priv_key_ptr(kc, (uint32_t)f));
371 int32_t cur = kc->cfg.buckets[b];
372 if (cur == f) {
373 kc->cfg.buckets[b] = kc->cfg.meta[f].hash_next;
374 } else {
375 for (uint32_t guard = 0U; guard < kc->cfg.cell_count; ++guard) {
376 if (cur == -1) {
377 break;
378 }
379 if (kc->cfg.meta[cur].hash_next == f) {
380 kc->cfg.meta[cur].hash_next = kc->cfg.meta[f].hash_next;
381 break;
382 }
383 cur = kc->cfg.meta[cur].hash_next;
384 }
385 }
386 kc->cfg.meta[f].hash_next = -1;
387}

520RA8_INTERNAL
521static void priv_slru_access(ra8_keycache_t* kc, int32_t f)
522{
523 if (kc->cfg.meta[f].seg == (uint8_t)k_keycache_seg_protected) {
524 priv_unlink(kc, f, &kc->pt_head, &kc->pt_tail);
525 priv_push_head(kc, f, &kc->pt_head, &kc->pt_tail);
526 return;
527 }
528 priv_unlink(kc, f, &kc->pb_head, &kc->pb_tail);
529 if (kc->protected_count >= kc->protected_cap) {
530 const int32_t d = kc->pt_tail;
531 if (d != -1) {
532 priv_unlink(kc, d, &kc->pt_head, &kc->pt_tail);
533 kc->cfg.meta[d].seg = (uint8_t)k_keycache_seg_probation;
534 priv_push_head(kc, d, &kc->pb_head, &kc->pb_tail);
535 kc->protected_count--;
536 }
537 }
538 kc->cfg.meta[f].seg = (uint8_t)k_keycache_seg_protected;
539 priv_push_head(kc, f, &kc->pt_head, &kc->pt_tail);
540 kc->protected_count++;
541}

823RA8_INTERNAL
824static ra8_err_t priv_miss(ra8_keycache_t* kc, const void* key, ra8_keycache_view_t* out_view)
825{
826 const int32_t v = priv_pick_victim(kc);
827 if (v < 0) {
828 return k_ra8_err_no_mem;
829 }
830 ra8_keycache_cell_t* m = &kc->cfg.meta[v];
831 if (m->valid != 0U) {
832 priv_hash_remove(kc, v);
833 kc->evictions++;
834 }
835 const uint8_t seg = m->seg;
836 priv_unlink(kc,
837 v,
838 (seg == (uint8_t)k_keycache_seg_protected) ? &kc->pt_head : &kc->pb_head,
839 (seg == (uint8_t)k_keycache_seg_protected) ? &kc->pt_tail : &kc->pb_tail);
840 if (seg == (uint8_t)k_keycache_seg_protected) {
841 kc->protected_count--;
842 }
843 uint8_t* cell = priv_cell_ptr(kc, (uint32_t)v);
844 void* user = priv_user_ptr(kc, (uint32_t)v);
845 const ra8_err_t rerr = kc->cfg.render(kc->cfg.render_ctx, key, cell, kc->cfg.cell_bytes, user);
846 if (rerr != k_ra8_ok) {
847 m->valid = 0U;
848 m->seg = (uint8_t)k_keycache_seg_probation;
849 priv_push_head(kc, v, &kc->pb_head, &kc->pb_tail);
850 return rerr;
851 }
852 (void)memcpy(priv_key_ptr(kc, (uint32_t)v), key, (size_t)kc->cfg.key_bytes);
853 m->valid = 1U;
854 m->pin_count = 1U;
855 m->seg = (uint8_t)k_keycache_seg_probation;
856 priv_hash_insert(kc, v);
857 priv_push_head(kc, v, &kc->pb_head, &kc->pb_tail);
858 *out_view = (ra8_keycache_view_t){.data = cell, .user = user};
859 return k_ra8_ok;
860}

862ra8_err_t ra8_keycache_get(ra8_keycache_t* kc, const void* key, ra8_keycache_view_t* out_view)
863{
864 RA8_CHECK_NULL_PTR(kc, s_tag, "kc must not be nullptr");
865 RA8_CHECK_NULL_PTR(key, s_tag, "key must not be nullptr");
866 RA8_CHECK_NULL_PTR(out_view, s_tag, "out_view must not be nullptr");
867 const int32_t f = priv_hash_lookup(kc, key);
868 if (f >= 0) {
869 kc->hits++;
870 kc->cfg.meta[f].pin_count++;
871 priv_access(kc, f);
872 *out_view = (ra8_keycache_view_t){.data = priv_cell_ptr(kc, (uint32_t)f),
873 .user = priv_user_ptr(kc, (uint32_t)f)};
874 return k_ra8_ok;
875 }
876 kc->misses++;
877 return priv_miss(kc, key, out_view);
878}

894ra8_err_t ra8_keycache_put(ra8_keycache_t* kc, const uint8_t* data)
895{
896 RA8_CHECK_NULL_PTR(kc, s_tag, "kc must not be nullptr");
897 RA8_CHECK_NULL_PTR(data, s_tag, "data must not be nullptr");
898 const uintptr_t base = (uintptr_t)kc->cfg.cell_mem;
899 const uintptr_t addr = (uintptr_t)data;
900 if (addr < base) {
901 return k_ra8_err_invalid_arg;
902 }
903 const uintptr_t off = addr - base;
904 const uintptr_t span = (uintptr_t)kc->cfg.cell_count * (uintptr_t)kc->cfg.cell_bytes;
905 if (off >= span) {
906 return k_ra8_err_invalid_arg;
907 }
908 if ((off % (uintptr_t)kc->cfg.cell_bytes) != 0U) {
909 return k_ra8_err_invalid_arg;
910 }
911 const uint32_t idx = (uint32_t)(off / (uintptr_t)kc->cfg.cell_bytes);
912 if (kc->cfg.meta[idx].pin_count == 0U) {
913 return k_ra8_err_invalid_arg;
914 }
915 kc->cfg.meta[idx].pin_count--;
916 return k_ra8_ok;
917}
