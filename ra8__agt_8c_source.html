98static ra8_err_t agt_mstp_acquire(uint8_t channel)
99{
100 if (channel >= k_ra8_agt_mstp_id_count) {
101 return k_ra8_ok;
102 }
103 if (s_agt_mstp_held[channel]) {
104 return k_ra8_ok;
105 }
106 /* HUM Ch 11.2.9 "MSTPCRD : Module Stop Control Register D" p 448 */
107 const ra8_err_t err = ra8_mstp_enable(s_agt_mstp_table[channel]);
108 if (err != k_ra8_ok) { /* GCOVR_EXCL_BR_LINE -- target-only read-back path */
109 return err; /* GCOVR_EXCL_LINE */
110 }
111 s_agt_mstp_held[channel] = true;
112 return k_ra8_ok;
113}

600static void agt_pulse_program_registers(volatile r_agt_regs_t* reg, const ra8_agt_pulse_cfg_t* cfg)
601{
602 /* Stop first. */
603 /* HUM Ch 24.2.4 "AGTCR : AGT Control Register" p 1167 */
604 reg->AGTCR = 0U;
605 /* TMOD=001b (pulse output mode), TCK = PCLKB. */
606 /* HUM Ch 24.2.5 "AGTMR1 : AGT Mode Register 1" p 1168 */
607 reg->AGTMR1 = (uint8_t)k_ra8_agt_agtmr1_tmod_pulse_output;
608 /* HUM Ch 24.2.6 "AGTMR2 : AGT Mode Register 2" p 1169 */
609 reg->AGTMR2 = 0U;
610 /* TOE = 1, polarity from the config. */
611 /* HUM Ch 24.2.7 "AGTIOC : AGT I/O Control Register" p 1170 */
612 reg->AGTIOC = agt_pulse_agtioc_value(cfg->polarity);
613 /* HUM Ch 24.2.9 "AGTCMSR : AGT Compare Match Function Select" p 1172 */
614 reg->AGTCMSR = agt_pulse_agtcmsr_value(cfg->compare);
615 agt_pulse_program_compare(reg, cfg->compare, cfg->duty);
616 /* HUM Ch 24.2.1 "AGT : AGT Counter" p 1165 */
617 reg->AGT = cfg->period;
618}

649{
650 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
651 volatile r_agt_regs_t* reg = ra8_agt(channel);
652 RA8_CHECK_NULL_PTR(reg, s_tag, "channel out of range");
653

655 RA8_RETURN_ON_ERROR(verr, s_tag, "agt_pulse: cfg validation");

671 (void)cfg->mode;

674 return k_ra8_ok;
