501static void dtc_coh_program_ti(void)
502{
503 const uint8_t mra = (uint8_t)(((uint8_t)k_dtc_coh_md_block << k_dtc_coh_mra_md_pos) |
504 ((uint8_t)k_dtc_coh_sz_word << k_dtc_coh_mra_sz_pos) |
505 ((uint8_t)k_dtc_coh_sm_inc << k_dtc_coh_mra_sm_pos));
506 const uint8_t mrb = (uint8_t)((uint8_t)k_dtc_coh_dm_inc << k_dtc_coh_mrb_dm_pos);
507 /* MR[31:24]=MRA, MR[23:16]=MRB, MR[15:8]=MRC(0); HUM Ch 18.2.2 p 786 /
508 * 18.2.3 p 787 / 18.2.4 p 789, Figure 18.4 p 799. SRAM (not MMIO). */
509 s_dtc_ti.MR =
510 ((uint32_t)mra << k_dtc_coh_mra_byte_pos) | ((uint32_t)mrb << k_dtc_coh_mrb_byte_pos);
511 s_dtc_ti.SAR = (uint32_t)(uintptr_t)s_src;
512 s_dtc_ti.DAR = (uint32_t)(uintptr_t)s_dst;
513 s_dtc_ti.CRB = (uint16_t)k_dtc_coh_crb_one_block;
514 s_dtc_ti.CRA = (uint16_t)k_dtc_coh_cra_block_256;
515 s_dtc_vt[s_dtc_slot] = (uint32_t)(uintptr_t)&s_dtc_ti;
516}

537[[nodiscard]] static ra8_err_t dtc_coh_arm_slot(void)
538{
539 volatile uint32_t* slot_reg = ra8_icu_ielsr(s_dtc_slot);
540 if (slot_reg == nullptr) {
541 return k_ra8_err_hw_error;
542 }
543 /* HUM Ch 14.2.10 "IELSRn : ICU Event Link Setting Register n" p 524:
544 * DTCE[24] enables DTC activation; the IELS event field is preserved. */
545 *slot_reg = *slot_reg | (uint32_t)k_ra8_ielsr_dtce_mask;
546 return k_ra8_ok;
547}

620[[nodiscard]] static ra8_err_t dtc_coh_run_once(uint8_t* out_ok)
621{
622 RA8_CHECK_NULL_PTR(out_ok, s_tag, "out_ok must not be nullptr");
623 *out_ok = 0U;
624
625 /* clean-before (app owns the TI + source): dirty lines -> SRAM so the DTC
626 * reads the real descriptor and the real pattern from memory. */
627 if (ra8_cache_dcache_clean_by_addr(s_src, (uint32_t)sizeof(s_src)) != k_ra8_ok) {
628 return k_ra8_err_hw_error;
629 }
630 if (ra8_cache_dcache_clean_by_addr(&s_dtc_ti, (uint32_t)sizeof(s_dtc_ti)) != k_ra8_ok) {
631 return k_ra8_err_hw_error;
632 }
633
634 /* The HAL cleans the caller-populated vector table here, then DTCST = 1. */
635 const ra8_err_t en = ra8_dtc_enable();
636 if (en != k_ra8_ok) {
637 return en;
638 }
639 const ra8_err_t arm = dtc_coh_arm_slot();
640 if (arm != k_ra8_ok) {
641 return arm;
642 }
643 const ra8_err_t trig = ra8_elc_software_trigger((uint8_t)k_dtc_coh_swevt_index);
644 if (trig != k_ra8_ok) {
645 return trig;
646 }
647
648 /* Completion + coherency gate: invalidate-after each poll so the verify reads
649 * what the DTC wrote to SRAM (not the dirty sentinel still in cache), and stop
650 * as soon as the full block has landed. Bounded so a stale-descriptor copy
651 * (the regression this app guards against) times out instead of hanging. */
652 bool done = false;
653 for (uint32_t i = 0U; i < (uint32_t)k_dtc_coh_poll_limit; ++i) {
654 if (ra8_cache_dcache_invalidate_by_addr(s_dst, (uint32_t)sizeof(s_dst)) != k_ra8_ok) {
655 return k_ra8_err_hw_error;
656 }
657 if (dtc_coh_verify() != 0U) {
658 done = true;
659 break;
660 }
661 }
662 if (!done) {
663 return k_ra8_err_hw_timeout;
664 }
665 *out_ok = 1U;
666 return k_ra8_ok;
667}

723{
