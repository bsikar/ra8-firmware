62static ra8_err_t internal_validate(const ra8_io_i2c_bus_t* bus)
63{
64 if (bus == nullptr) {
65 return k_ra8_err_null_ptr;

231ra8_err_t ra8_io_i2c_bus_as_ops(const ra8_io_i2c_bus_t* bus, ra8_i2c_bus_ops_t* out)
232{
233 const ra8_err_t v = internal_validate(bus);
234 if (v != k_ra8_ok) {
235 return v;
236 }
237 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
238 out->write = internal_ops_write;
239 out->read = internal_ops_read;
240 out->transfer = internal_ops_transfer;
241 out->ctx = (void*)(uintptr_t)bus;
242 return k_ra8_ok;
243}
