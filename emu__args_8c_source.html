145static bool args_try_mode(int argc, char** argv, int* i, emu_args_t* out)
146{
147 const int idx = *i;
148 if (strncmp(argv[idx], "--view", sizeof("--view")) == 0) {
149 out->want_view = true;
150 } else if (strncmp(argv[idx], "--usbhs-loop", sizeof("--usbhs-loop")) == 0) {
151 /* Model the on-chip USBHS host controller (board_periph_usbhs_host.c) and
152 * bridge it to the USBFS device, instead of the ra8_usb_host_* function
153 * seam: for apps whose HS host enumerates the SAME chip's FS device. */
154 out->usbhs_loop = true;
155 } else if (strncmp(argv[idx], "--trace", sizeof("--trace")) == 0) {
156 out->want_trace = true;
157 } else if (strncmp(argv[idx], "--low-power", sizeof("--low-power")) == 0) {
158 emu_set_low_power(true);
159 } else if (strncmp(argv[idx], "--fast-sd", sizeof("--fast-sd")) == 0) {
160 emu_fast_sd_enable();
161 } else if (strncmp(argv[idx], "--eink", sizeof("--eink")) == 0) {
162 (void)board_eink_attach(); /* answer the ra8_epaper SPI path (IT8951 model) */
163 } else if (strncmp(argv[idx], "--modem", sizeof("--modem")) == 0) {
164 (void)board_modem_attach(); /* answer the ra8_modem_at AT path (SCI7 modem model) */
165 } else if ((strncmp(argv[idx], "--primary-core", sizeof("--primary-core")) == 0) &&
166 ((idx + 1) < argc)) {
167 emu_set_primary_core((strncmp(argv[idx + 1], "m33", sizeof("m33")) == 0) ? k_core_m33
168 : k_core_m85);
169 (*i)++;
170 } else if ((strncmp(argv[idx], "--device", sizeof("--device")) == 0) && ((idx + 1) < argc)) {
171 /* Select the emulated part; anything but "ra8p1" (incl. "ra8d2") = RA8D2. */
172 out->emu_device = (strncmp(argv[idx + 1], "ra8p1", sizeof("ra8p1")) == 0)
173 ? k_board_device_ra8p1
174 : k_board_device_ra8d2;
175 (*i)++;
176 } else {
177 return false;
178 }
179 return true;
180}

312static void args_attach_blank_sd(const char* spec)
313{
314 char* endp = nullptr;
315 const long num = strtol(spec, &endp, (int)k_strtol_base10);
316 uint64_t mult =
317 (uint64_t)k_sectors_per_mib * (uint64_t)k_bytes_per_sector; /* default unit: MiB. */
318 /* Cast the whole conditional: it promotes both arms to int. */
319 const char unit = (char)(((endp != nullptr) && (*endp != '\0')) ? tolower((int)*endp) : 'm');
320 if (unit == 'k') {
321 mult = (uint64_t)k_size_kib;
322 } else if (unit == 'g') {
323 mult = (uint64_t)k_size_kib * (uint64_t)k_size_kib * (uint64_t)k_size_kib;
324 } else if (unit == 't') {
325 mult =
326 (uint64_t)k_size_kib * (uint64_t)k_size_kib * (uint64_t)k_size_kib * (uint64_t)k_size_kib;
327 }
328 const uint64_t bytes = (num > 0L) ? ((uint64_t)num * mult) : 0ULL;
329 const uint64_t sectors = bytes / (uint64_t)k_bytes_per_sector;
330 const char* colon = (endp != nullptr) ? strchr(endp, ':') : nullptr;
331 const uint64_t fat32_min_bytes =
332 (uint64_t)k_fat32_min_mib * (uint64_t)k_size_kib * (uint64_t)k_size_kib;
333 uint8_t fat = (bytes >= fat32_min_bytes) ? (uint8_t)32U : (uint8_t)16U;
334 if (colon != nullptr) {
335 fat = (strstr(colon, "32") != nullptr) ? (uint8_t)32U : (uint8_t)16U;
336 }
337 if ((sectors > 0ULL) && (sectors <= (uint64_t)k_sd_u32_max)) {
338 (void)board_sd_attach_blank((uint32_t)sectors, fat, "RA8EMU");
339 } else if (sectors > (uint64_t)k_sd_u32_max) {
340 (void)fprintf(stderr, "ra8_emulator: --sd-new: size exceeds the 2 TiB FAT limit\n");
341 }
342}

364static bool args_try_sd(int argc, char** argv, int* i, emu_args_t* out)
365{
366 const int idx = *i;
367 if ((strncmp(argv[idx], "--sd", sizeof("--sd")) == 0) && ((idx + 1) < argc)) {
368 (void)board_sd_attach(argv[idx + 1]); /* serve this FAT image to ra8_sdmmc_spi */
369 (*i)++;
370 } else if ((strncmp(argv[idx], "--sd-new", sizeof("--sd-new")) == 0) && ((idx + 1) < argc)) {
371 /* "<N>[k|m|g|t][:fat16|fat32]" -- create + attach a blank formatted card.
372 * A bare number is MiB; a k/m/g/t suffix sets the unit (so "30g" = 30 GiB).
373 * Format defaults by size (FAT32 >= 512 MiB) like a real SD card; FAT16
374 * cannot exceed its cluster ceiling, so multi-GB cards are FAT32. */
375 args_attach_blank_sd(argv[idx + 1]);
376 (*i)++;
377 } else if ((strncmp(argv[idx], "--save-sd", sizeof("--save-sd")) == 0) && ((idx + 1) < argc)) {
378 out->save_sd_path = argv[idx + 1];
379 (*i)++;
380 } else {
381 return false;
382 }
383 return true;
384}

407static bool args_try_sym(int argc, char** argv, int* i, emu_args_t* out)
408{
409 const int idx = *i;
410 if ((strncmp(argv[idx], "--dump-sym", sizeof("--dump-sym")) == 0) && ((idx + 1) < argc)) {
411 if (out->dump_sym_n < (uint32_t)k_dump_sym_max) {
412 out->dump_sym_names[out->dump_sym_n] = argv[idx + 1];
413 out->dump_sym_n++;
414 }
415 (*i)++;
416 } else if ((strncmp(argv[idx], "--stop-sym", sizeof("--stop-sym")) == 0) && ((idx + 2) < argc)) {
417 /* End the run the instant 32-bit global <name> reaches <threshold> -- the
418 * counter analog of RA8_EMU_STOP_ON, for the jlink_memprobe EIL mode: a
419 * free-running progress counter that hits its floor stops the run at once,
420 * so a passing probe finishes in a few chunks instead of the full budget. */
421 out->stop_sym_name = argv[idx + 1];
422 out->stop_sym_thresh = (uint32_t)strtoul(argv[idx + 2], nullptr, (int)k_strtol_base10);
423 *i += 2;
424 } else if ((strncmp(argv[idx], "--trace-sym", sizeof("--trace-sym")) == 0) &&
425 ((idx + 1) < argc)) {
426 if (out->trace_sym_n < (uint32_t)k_trace_sym_max) {
427 out->trace_sym_names[out->trace_sym_n] = argv[idx + 1];
428 out->trace_sym_n++;
429 }
430 (*i)++;
431 } else {
432 return false;
433 }
434 return true;

479bool emu_args_parse(int argc, char** argv, emu_args_t* out)
480{
481 if (argc < 2) {
482 args_print_usage();
483 return false;
484 }
485 args_defaults(argv[1], out);
486 for (int i = 2; i < argc; i++) {
487 (void)(args_try_mode(argc, argv, &i, out) || args_try_display(argc, argv, &i, out) ||
488 args_try_input(argc, argv, &i, out) || args_try_sd(argc, argv, &i, out) ||
489 args_try_sym(argc, argv, &i, out) || args_try_control(argc, argv, &i, out));
490 }
491 return true;
