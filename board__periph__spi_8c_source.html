134static void spi_spcr_write(spi_state_t* s, uint32_t value)
135{
136 s->enabled = ((value & (uint32_t)k_spi_spcr_spe) != 0U);
137 if (s->enabled) {
138 /* Running: the model drains TX instantly, so SPTEF is always ready for
139 * the first frame; SPRF stays clear until a word is written. */
140 s->spsr |= (uint32_t)k_spi_spsr_sptef;
141 s->spsr &= ~(uint32_t)k_spi_spsr_sprf;
142 } else {
143 s->spsr = 0U;
144 }
145}

148static void spi_spcr2_write(spi_state_t* s, uint32_t value)
149{
150 s->loopback = ((value & ((uint32_t)k_spi_spcr2_splp | (uint32_t)k_spi_spcr2_splp2)) != 0U);
151}

154static void spi_spdr_write(spi_state_t* s, uint32_t value)
155{
156 /* SPLP2 ties the outgoing line back to the receive shifter (rx = tx);
157 * SPLP inverts it (rx = ~tx). Without loopback, if a `--sd` card is
158 * attached it answers the exchange (the genuine ra8_sdmmc_spi path); an
159 * `--eink` IT8951 controller answers the ra8_epaper path; else nothing
160 * drives the line and the receive shifter clocks in an idle 0. */
161 if ((s->reg[spi_word((uint64_t)k_spi_off_spcr2)] & (uint32_t)k_spi_spcr2_splp) != 0U) {
162 s->rx = (~value) & (uint32_t)k_spi_byte_mask;
163 } else if (s->loopback) {
164 s->rx = value;
165 } else if (board_sd_attached()) {
166 s->rx = (uint32_t)board_sd_exchange((uint8_t)(value & (uint32_t)k_spi_byte_mask));
167 } else if (board_eink_attached()) {
168 s->rx = (uint32_t)board_eink_exchange((uint8_t)(value & (uint32_t)k_spi_byte_mask));
169 } else {
170 s->rx = 0U;
171 }
172 /* TX drains immediately (SPTEF stays ready) and the echoed word lands in
173 * the receive holding register, so SPRF asserts for the driver's RX poll. */
174 s->spsr |= ((uint32_t)k_spi_spsr_sptef | (uint32_t)k_spi_spsr_sprf | (uint32_t)k_spi_spsr_spdrf);
175 s->frames++;
176}

179static void spi_spsrc_write(spi_state_t* s, uint32_t value)
180{
181 /* Every SPSRC clear bit sits at the same position as its SPSR flag, so a
182 * written 1 clears that flag; SPTEF is re-asserted because the model's TX
183 * buffer is always empty once the channel runs. */
184 s->spsr &= ~value;
185 if (s->enabled) {
186 s->spsr |= (uint32_t)k_spi_spsr_sptef;
187 }
188}

191static uint64_t spi_reg_read(spi_state_t* s, uint64_t off)
192{
193 if (off == (uint64_t)k_spi_off_spsr) {
194 return s->spsr;
195 }
196 if (off == (uint64_t)k_spi_off_spdr) {
197 return s->rx; /* receive holding register: the echoed frame */
198 }
199 return s->reg[spi_word(off)]; /* reflect every other register */
200}

203static void spi_reg_write(spi_state_t* s, uint64_t off, uint32_t value)
204{
205 s->reg[spi_word(off)] = value; /* shadow keeps "configure then verify" working */
206 if (off == (uint64_t)k_spi_off_spcr) {
207 spi_spcr_write(s, value);
208 } else if (off == (uint64_t)k_spi_off_spcr2) {
209 spi_spcr2_write(s, value);
210 } else if (off == (uint64_t)k_spi_off_spdr) {
211 spi_spdr_write(s, value);
212 } else if (off == (uint64_t)k_spi_off_spsrc) {
213 spi_spsrc_write(s, value);
214 }
215}

258static void spi_report(void)
259{
260 for (uint32_t ch = 0U; ch < (uint32_t)k_spi_count; ch++) {
261 if (s_spi[ch].frames > 0U) {
262 (void)fprintf(stderr,
263 " SPI%u : %u frame(s) echoed (loopback=%s)\n",
264 ch,
265 s_spi[ch].frames,
266 s_spi[ch].loopback ? "yes" : "no");
267 char ln[k_spi_console_line_cap];
268 (void)snprintf(ln,
269 sizeof(ln),
270 "SPI%u: %u frames lb=%s",
271 ch,
272 s_spi[ch].frames,
273 s_spi[ch].loopback ? "on" : "off");
274 board_console_push(k_board_console_ch_spi, ln);
275 }
276 }
277 board_eink_report(); /* an attached IT8951 controller prints its own line */
278}
