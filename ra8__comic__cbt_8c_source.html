88ra8_err_t ra8_comic_cbt_open(ra8_comic_t* c)
89{
90 RA8_CHECK_NULL_PTR(c, s_tag_cbt, "cbt open: null c");
91 if (!c->tar.live) {
92 return k_ra8_err_invalid_state;
93 }
94 uint64_t off = 0U;
95 for (uint32_t n = 0U; n < (uint32_t)k_cbt_max_members; ++n) { /* bound: static member cap */
96 char nb[k_cbt_name_max] = {};
97 ra8_unarch_tar_entry_t ent = {};
98 const ra8_err_t nerr = ra8_unarch_tar_next(&c->tar, off, nb, (uint16_t)sizeof(nb), &ent);
99 if (nerr == k_ra8_err_not_found) {
100 return k_ra8_ok; /* clean end of archive */
101 }
102 if (nerr != k_ra8_ok) {
103 return nerr; /* malformed / lying / over-policy member: reject */
104 }
105 const ra8_err_t ae = s_add_member(c, &ent, nb, ent.name_len);
106 if (ae != k_ra8_ok) {
107 return ae; /* index / arena full -- a real capacity error, propagate */
108 }
109 off = ent.next_off;
110 }
111 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE -- the walker's policy
112 * entry cap (4096) fires inside tar_next
113 * far below the static member cap */
114}
