4"""Doxygen documentation gap auditor for ra8-firmware.
5
6Walks every .c/.h under libs/, src/, port/ (excluding libs/third_party/),
7locates every function definition/prototype, and checks the immediately
8preceding Doxygen block for the required tags listed in CLAUDE.md
9("Doxygen Documentation Requirements").
10
11KNOWN SCOPE GAP -- this function gate does NOT cover examples/, tools/ or
12tests/, which CLAUDE.md ("these standards apply to EVERY first-party file")
13says it should. ``--members`` already covers them (MEMBER_SCAN_DIRS); the
14function mode does not. Measured on the current tree, adding all three to
15SCAN_DIRS reports 9635 gaps -- tests/ 7052, examples/ 1500, tools/ 1083 --
16across roughly a thousand functions in tools/ alone. Only 52 of the tools/
17gaps are functions with no block at all; the rest carry a block that is
18missing required tags, so closing them means writing real ``@pre`` / ``@post``
19/ ``@retval`` / ``@note`` prose per function. That is a documentation
20campaign, tracked in its own issue, and it must be closed by writing real
21documentation, never by generating tag-shaped filler -- which is exactly what
22the required-tag list rewards if applied mechanically.
23
24Modes
25-----
26 (no args) Function audit report -> docs/DOXYGEN_GAPS.csv + .md
27 --check Strict function gate (exit 1 on any gap). Wired into CI
28 and the pre-commit hook.
29 --selftest Regression-test the auditor itself, in both directions,
30 for both enforcing modes. Runs before the real check in
31 the gate: a parser-driven gate that stops recognising a
32 construct reports nothing and looks like a documented
33 tree.
34 --members REPORT-ONLY member audit. Enumerates every undocumented
35 enum value, struct/union member, and macro across the
36 first-party tree and prints an offender count per top-level
37 dir. This mode never fails the build. Optionally takes
38 explicit file paths to audit just those, and --out=PATH to
39 dump the full offender list as CSV.
40 --members --check ENFORCING member gate (exit 1 on any undocumented enum
41 value, struct/union member, or macro). Wired into CI and
42 the pre-commit hook alongside the function gate (issue
43 #246). Optionally takes explicit file paths to gate just
44 those.
45 --style ENFORCING gate for the two docs/STYLE_GUIDE.md tag rules
46 that attach to no symbol, so neither the function nor the
47 member gate ever saw them (#532): the file-header block
48 (@file present and naming this file, @brief, @details) and
49 the @param direction bracket. Unlike --members there is no
50 report-only twin -- a mode that measures instead of failing
51 is how the member gate was once mis-wired.
52 --style --update-baseline
53 Shrink .github/doxy-details-baseline.txt to the files that
54 still lack @details. Refuses to GROW it.
55
56CLAUDE.md ("Doxygen Documentation Requirements") demands that *every* enum
57value, struct/union member, and macro carry documentation -- an inline
58``/**< ... */`` (or a preceding ``/** ... */`` block) on each aggregate member
59and a ``@brief``/``@def`` block on each macro. ``--members --check`` enforces
60exactly that (issue #246); plain ``--members`` is the sizing report.
61
62Output:
63 - docs/DOXYGEN_GAPS.csv full row-per-function table
64 - docs/DOXYGEN_GAPS.md human-readable summary
65 - (--members) stdout summary; optional --out=PATH member-gap CSV
66
67Audit-only: never edits source files.
68
69Exit 0 when clean or report-only, 1 when an enforcing mode found gaps, 2 when
70a scope walk collapsed below its floor (doxy_scope.FUNCTION_FILE_FLOOR /
71MEMBER_FILE_FLOOR).
72"""

94def run_check() -> int:
95 """Strict gate: exit 0 if zero gaps, else exit 1 with offender list.
96
97 Used by the pre-commit hook to keep documentation coverage at 100%.
98 Does not write CSV/MD outputs -- read-only audit.
99
100 Scope comes from :func:`doxy_scope.function_files`, which exits 2 rather
101 than let a collapsed walk report ``gaps=0 (PASS)``.
102 """
103 all_rows = []
104 for path in function_files():
105 all_rows.extend(audit_file(path))
106
107 gap_rows = [r for r in all_rows if r[3]]
108 if not gap_rows:
109 print("doxy_audit --check: gaps=0 (PASS)")
110 return 0
111
112 print(f"doxy_audit --check: gaps={len(gap_rows)} (FAIL)")
113 print("Offending functions (file:line function -- missing tags):")
114 # cap output to 50 lines so the hook stays readable
115 cap = OFFENDER_CAP
116 for src, line, name, missing, _sev in gap_rows[:cap]:
117 print(f" {src}:{line} {name} -- {';'.join(missing)}")
118 if len(gap_rows) > cap:
119 print(f" ... and {len(gap_rows) - cap} more")
120 print()
121 print("Refresh the audit report by running:")
122 print(" python3 scripts/checks/doxy_audit.py")
123 return 1

126def run_members_report(explicit: list[str], out_csv: str | None) -> int:
127 """REPORT-ONLY member/enum/macro audit. Always returns 0.
128
129 Enumerates every undocumented enum value, struct/union member, and macro
130 across the first-party tree and prints an offender count per top-level dir
131 so the owner can size the fallout wave (issue #246). This mode intentionally
132 never fails: promoting the member checks to a hard gate is a follow-up.
133
134 "Never fails" covers findings, not infrastructure:
135 :func:`doxy_scope.member_files` still exits 2 on a collapsed repo-wide
136 walk, since a report sized from nothing is worse than no report.
137 """
138 all_rows = []
139 for p in member_files(explicit):
140 all_rows.extend(audit_members_file(p))
141
142 by_dir_kind = Counter((_top_dir(r[0]), r[2]) for r in all_rows)
143 by_dir = Counter(_top_dir(r[0]) for r in all_rows)
144 by_kind = Counter(r[2] for r in all_rows)
145 by_file = Counter(r[0] for r in all_rows)
146
147 if out_csv is not None:
148 out_path = Path(out_csv)
149 if not out_path.is_absolute():
150 out_path = repo_root() / out_path
151 with out_path.open("w", encoding="ascii") as f:
152 f.write("source_file,line,element,name,reason\n")
153 for rel, line, kind, name, reason in sorted(all_rows, key=lambda r: (r[0], r[1])):
154 f.write(f"{rel},{line},{kind},{name},{reason}\n")
155
156 kinds = ["enum-value", "member", "macro"]
157 dirs = sorted(by_dir)
158
159 print("doxy_audit --members: REPORT-ONLY member/enum/macro documentation audit")
160 print("(this mode never fails the build; see issue #246)")
161 print()
162 print("Undocumented offenders per top-level dir:")
163 header = f" {'dir':<12}" + "".join(f"{k:>13}" for k in kinds) + f"{'total':>10}"
164 print(header)
165 print(" " + "-" * (len(header) - 2))
166 for d in dirs:
167 cells = "".join(f"{by_dir_kind[(d, k)]:>13}" for k in kinds)
168 print(f" {d:<12}{cells}{by_dir[d]:>10}")
169 totals = "".join(f"{by_kind[k]:>13}" for k in kinds)
170 print(" " + "-" * (len(header) - 2))
171 print(f" {'TOTAL':<12}{totals}{len(all_rows):>10}")
172 print()
173 print(f"Files scanned with at least one offender: {len(by_file)}")
174 print("Worst 15 files by offender count:")
175 for rel, cnt in by_file.most_common(15):
176 print(f" {cnt:>6} {rel}")
177 if out_csv is not None:
178 print()
179 print(f"Full offender list written to: {out_csv}")
180 return 0

183def run_members_check(explicit: list[str]) -> int:
184 """ENFORCING member gate: exit 0 if zero offenders, else exit 1.
185
186 Enforces the CLAUDE.md rule that every enum value, struct/union member, and
187 macro carries documentation (issue #246). Wired into the pre-commit hook and
188 CI alongside the function gate. Read-only -- writes no CSV/MD outputs.
189
190 Scope comes from :func:`doxy_scope.member_files`, which exits 2 rather than
191 let a collapsed repo-wide walk report ``offenders=0 (PASS)``.
192 """
193 all_rows = []
194 for p in member_files(explicit):
195 all_rows.extend(audit_members_file(p))
196 all_rows.sort(key=lambda r: (r[0], r[1]))
197
198 if not all_rows:
199 print("doxy_audit --members --check: offenders=0 (PASS)")
200 return 0
201
202 by_kind = Counter(r[2] for r in all_rows)
203 print(f"doxy_audit --members --check: offenders={len(all_rows)} (FAIL)")
204 print(
205 " by kind: "
206 + ", ".join(f"{k}={by_kind[k]}" for k in ("enum-value", "member", "macro") if by_kind[k])
207 )
208 print("Undocumented members (file:line kind name -- reason):")
209 cap = OFFENDER_CAP
210 for rel, line, kind, name, reason in all_rows[:cap]:
211 print(f" {rel}:{line} {kind} {name} -- {reason}")
212 if len(all_rows) > cap:
213 print(f" ... and {len(all_rows) - cap} more")
214 print()
215 print("Every enum value, struct/union member, and macro needs a doc comment:")
216 print(" - aggregate members: an inline /**< ... */ or a preceding /** ... */ block")
217 print(" - macros: a preceding /** @brief ... */ (or @def) block")
218 print("Size the full fallout with: python3 scripts/checks/doxy_audit.py --members")
219 return 1

222def _parse_members_args(argv: list[str]) -> tuple[list[str], str | None]:
223 """Split --members argv into (explicit_paths, out_csv)."""
224 explicit = []
225 out_csv = None
226 for a in argv:
227 if a == "--members":
228 continue
229 if a.startswith("--out="):
230 out_csv = a[len("--out=") :]
231 elif not a.startswith("--"):
232 explicit.append(a)
233 return explicit, out_csv

236def main() -> int:
237 """Dispatch to the function gate, the member gate, the style gate, or a report.
238
239 The two member modes are NOT interchangeable: ``--members`` alone is
240 report-only and always exits 0 (it exists to size the #246 fallout), while
241 ``--members --check`` is the enforcing gate. CI must pass both flags, or
242 the step measures the problem instead of failing on it. ``--style`` has no
243 such pair on purpose -- it is always enforcing, so there is no spelling of
244 it that measures instead of failing.
245
246 Returns 0 on a clean gate, a passing selftest, or any report-only run;
247 1 when an enforcing mode found offenders. Exits 2 when a mode could not
248 run at all -- either scope walk collapsing below its measured file floor
249 (:mod:`doxy_scope`), or ``--style``'s file / ``@param`` floors
250 (:mod:`doxy_style`). A scan that read nothing must never report a
251 documented tree.
252 """
253 args = sys.argv[1:]
254 if "--selftest" in args:
255 return run_selftest()
256 if "--style" in args:
257 return run_style_update() if "--update-baseline" in args else run_style_check()
258 if "--members" in args:
259 explicit, out_csv = _parse_members_args(args)
260 return (
261 run_members_check(explicit)
262 if "--check" in args
263 else run_members_report(explicit, out_csv)
264 )
265 if "--check" in args:
266 return run_check()
267 return run_report()
