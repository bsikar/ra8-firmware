282static void wd_print_hex(uint32_t value)
283{
284 uint8_t buf[k_wd_hex_nibbles];
285 for (uint32_t i = 0U; i < (uint32_t)k_wd_hex_nibbles; i++) {
286 const uint32_t shift = ((uint32_t)k_wd_hex_nibbles - 1U - i) * (uint32_t)k_wd_nibble_bits;
287 const uint32_t nib = (value >> shift) & (uint32_t)k_wd_nibble_mask;
288 buf[i] = (uint8_t)((nib < (uint32_t)k_wd_dec_ten) ? ('0' + nib) : ('A' + (nib - k_wd_dec_ten)));
289 }
290 wd_print(buf, (uint32_t)k_wd_hex_nibbles);
291}

294static void wd_print_uint(uint32_t value)
295{
296 uint8_t buf[k_wd_dec_ten];
297 uint32_t n = 0U;
298 if (value == 0U) {
299 buf[n] = '0';
300 n++;
301 }
302 while ((value > 0U) && (n < (uint32_t)k_wd_dec_ten)) {
303 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_wd_dec_ten));
304 n++;
305 value /= (uint32_t)k_wd_dec_ten;
306 }
307 for (uint32_t i = 0U; i < n; i++) {
308 wd_print(&buf[n - 1U - i], 1U);
309 }
310}

313static uint32_t wd_framebuffer_hash(void)
314{
315 const uint8_t* p = (const uint8_t*)s_framebuffer;
316 const size_t n = sizeof(s_framebuffer);
317 uint32_t hsh = (uint32_t)k_wd_fnv_offset;
318 for (size_t i = 0U; i < n; i++) {
319 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_wd_fnv_prime;
320 }
321 return hsh;
322}

416static bool wd_tabbar_on_input(ra8_widget_t* w, const ra8_widget_event_t* ev)
417{
418 if (ev->kind != k_ra8_widget_ev_touch) {
419 return false;
420 }
421 const wa_chrome_t* ch = (const wa_chrome_t*)w->ctx;
422 uint16_t n = 0U;
423 if (ra8_app_count(ch->reg, &n) != k_ra8_ok) {
424 return false;
425 }
426 if (n == 0U) {
427 return false;
428 }
429 const int32_t tabw = w->rect.w / (int32_t)n;
430 if (tabw <= 0) {
431 return false;
432 }
433 const int32_t col = (ev->x - w->rect.x) / tabw;
434 if (col < 0) {
435 return false;
436 }
437 if (col >= (int32_t)n) {
438 return false;
439 }
440 ra8_app_t* app = nullptr;
441 if (ra8_app_at(ch->reg, (uint16_t)col, &app) != k_ra8_ok) {
442 return false;
443 }
444 g_wd_request = (int16_t)app->id;
445 return true;
446}

465static int16_t wd_neighbor_id(int32_t dir)
466{
467 uint16_t n = 0U;
468 if (ra8_app_count(&s_reg, &n) != k_ra8_ok) {
469 return (int16_t)k_ra8_app_none;
470 }
471 if (n == 0U) {
472 return (int16_t)k_ra8_app_none;
473 }
474 int32_t cur = (int32_t)s_reg.active;
475 if (cur < 0) {
476 cur = 0;
477 }
478 const int32_t nx = (cur + dir + (int32_t)n) % (int32_t)n;
479 ra8_app_t* app = nullptr;
480 if (ra8_app_at(&s_reg, (uint16_t)nx, &app) != k_ra8_ok) {
481 return (int16_t)k_ra8_app_none;
482 }
483 return (int16_t)app->id;
484}

487static bool wd_app_on_input(ra8_app_t* a, const ra8_widget_event_t* ev)
488{
489 wa_app_state_t* st = (wa_app_state_t*)a->ctx;
490 if (ev->kind == k_ra8_widget_ev_button) {
491 int32_t dir = 0;
492 if (ev->button_id == (uint16_t)k_wd_btn_prev) {
493 dir = -1;
494 }
495 if (ev->button_id == (uint16_t)k_wd_btn_next) {
496 dir = 1;
497 }
498 if (dir == 0) {
499 return false;
500 }
501 g_wd_request = wd_neighbor_id(dir);
502 return true;
503 }
504 bool handled = false;
505 (void)ra8_widget_dispatch(st->widgets, (uint16_t)k_wd_w_count, ev, &handled);
506 return handled;
507}

693static void wd_service_request(void)
694{
695 const int16_t want = g_wd_request;
696 g_wd_request = (int16_t)k_ra8_app_none;
697 if (want == (int16_t)k_ra8_app_none) {
698 return;
699 }
700 ra8_app_t* act = nullptr;
701 (void)ra8_app_active(&s_reg, &act);
702 if (act != nullptr) {
703 if (act->id == (uint16_t)want) {
704 return; /* already focused -- no flicker. */
705 }
706 }
707 if (ra8_app_launch(&s_reg, (uint16_t)want) == k_ra8_ok) {
708 wd_render_active();
709 }
710}

812static bool wd_selfcheck(uint32_t* out_lib, uint32_t* out_rdr)
813{
814 if (ra8_app_launch(&s_reg, (uint16_t)k_wd_app_library) != k_ra8_ok) {
815 return false;
816 }
817 (void)ra8_app_render(&s_reg);
818 const uint32_t lib = wd_framebuffer_hash();
819 if (ra8_app_launch(&s_reg, (uint16_t)k_wd_app_reader) != k_ra8_ok) {
820 return false;
821 }
822 (void)ra8_app_render(&s_reg);
823 const uint32_t rdr = wd_framebuffer_hash();
824 if (lib == rdr) {
825 return false;
826 }
827 if (s_library.leaves != 1U) {
828 return false;
829 }
830 if (s_reader.enters != 1U) {
831 return false;
832 }
833 if (s_library.enters != 1U) {
834 return false;
835 }
836 if (!wd_check_touch_dispatch()) {
837 return false;
838 }
839 if (!wd_check_damage()) {
840 return false;
841 }
842 *out_lib = lib;
843 *out_rdr = rdr;
844 return true;
845}

932 wd_poll_buttons();
933 wd_service_request();

935 }
936}
