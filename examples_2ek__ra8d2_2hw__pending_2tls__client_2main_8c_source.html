416static int tls_bio_send(void* ctx, const uint8_t* buf, size_t len)
417{
418 (void)ctx;
419 if (buf == nullptr || len == 0U) {
420 return 0;
421 }
422 NX_PACKET* pkt = NX_NULL;
423 UINT s = nx_packet_allocate(&s_packet_pool, &pkt, NX_TCP_PACKET, NX_WAIT_FOREVER);
424 if (s != NX_SUCCESS || pkt == NX_NULL) {
425 return MBEDTLS_ERR_SSL_INTERNAL_ERROR;
426 }
427 s =
428 nx_packet_data_append(pkt, (VOID*)(uintptr_t)buf, (ULONG)len, &s_packet_pool, NX_WAIT_FOREVER);
429 if (s != NX_SUCCESS) {
430 (void)nx_packet_release(pkt);
431 return MBEDTLS_ERR_SSL_INTERNAL_ERROR;
432 }
433 s = nx_tcp_socket_send(&s_tls_socket, pkt, (ULONG)k_demo_recv_timeout);
434 if (s != NX_SUCCESS) {
435 (void)nx_packet_release(pkt);
436 if (s == NX_NO_PACKET || s == NX_WINDOW_OVERFLOW) {
437 return MBEDTLS_ERR_SSL_WANT_WRITE;
438 }
439 return MBEDTLS_ERR_NET_SEND_FAILED;
440 }
441 return (int)len;
442}

459static int tls_bio_recv(void* ctx, uint8_t* buf, size_t len)
460{
461 (void)ctx;
462 if (buf == nullptr || len == 0U) {
463 return 0;
464 }
465 NX_PACKET* pkt = NX_NULL;
466 UINT s = nx_tcp_socket_receive(&s_tls_socket, &pkt, (ULONG)k_demo_recv_timeout);
467 if (s == NX_NO_PACKET) {
468 return MBEDTLS_ERR_SSL_WANT_READ;
469 }
470 if (s != NX_SUCCESS || pkt == NX_NULL) {
471 return MBEDTLS_ERR_NET_RECV_FAILED;
472 }
473 ULONG copied = 0U;
474 (void)nx_packet_data_retrieve(pkt, (VOID*)buf, &copied);
475 (void)nx_packet_release(pkt);
476 if (copied > (ULONG)len) {
477 copied = (ULONG)len;
478 }
479 return (int)copied;
480}

521

523 if (s != NX_SUCCESS) {
524 return s;

530 (void)nx_icmp_enable(&s_ip);

532 (void)nx_ip_fragment_enable(&s_ip);

608static ra8_err_t demo_exchange(ra8_tls_session_t session, uint8_t* got, size_t got_cap)
609{
610 if (got == nullptr || got_cap == 0U) {
611 return k_ra8_err_invalid_arg;
612 }
613 size_t sent = 0U;
614 ra8_err_t err = k_ra8_err_would_block;
615 for (uint32_t i = 0U; (i < (uint32_t)k_demo_io_max) && (err == k_ra8_err_would_block); ++i) {
616 err = ra8_tls_send(session, k_demo_record, sizeof(k_demo_record) - 1U, &sent);
617 }
618 if (err != k_ra8_ok) {
619 return err;
620 }
621 size_t got_n = 0U;
622 err = k_ra8_err_would_block;
623 for (uint32_t i = 0U; (i < (uint32_t)k_demo_io_max) && (err == k_ra8_err_would_block); ++i) {
624 err = ra8_tls_recv(session, got, got_cap, &got_n);
625 }
626 return err;

670static ra8_err_t demo_run_tls(void)
671{
672 uint16_t mss = 0U;
673 if (ra8_tls_mss_clamp((uint16_t)k_demo_mtu, &mss) != k_ra8_ok) {
674 return k_ra8_err_invalid_arg;
675 }
676 {
677 char line[k_demo_line_buf];
678 int n = snprintf(line,
679 sizeof(line),
680 "[tls] MTU=%u -> MSS clamp=%u\r\n",
681 (unsigned)k_demo_mtu,
682 (unsigned)mss);
683 if (n > 0 && (size_t)n < sizeof(line)) {
684 demo_print(line);
685 }
686 }
687
688 ra8_err_t err = ra8_tls_global_init();
689 if (err != k_ra8_ok) {
690 return err;
691 }
692
693 ra8_tls_session_cfg_t cfg = {};
694 cfg.bio_send = tls_bio_send;
695 cfg.bio_recv = tls_bio_recv;
696 cfg.bio_ctx = &s_tls_socket;
697 cfg.server_name = k_demo_server_name;
698 /* The test endpoint typically presents a self-signed cert; report the
699 * verification result rather than aborting the exchange. */
700 cfg.verify_mode = k_ra8_tls_verify_optional;
701
702 ra8_tls_session_t session = nullptr;
703 err = ra8_tls_session_open(&session, &cfg);
704 if (err != k_ra8_ok) {
705 (void)ra8_tls_global_deinit();
706 return err;
707 }
708
709 err = demo_handshake(session);
710 if (err == k_ra8_ok) {
711 uint8_t got[k_demo_record_cap] = {};
712 err = demo_exchange(session, got, sizeof(got));
713 demo_report(session);
714 }
715
716 (void)ra8_tls_session_close(session);
717 (void)ra8_tls_global_deinit();
718 return err;
719}
