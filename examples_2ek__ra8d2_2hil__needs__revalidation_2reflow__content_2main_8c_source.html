104static void rc_print_hex(uint32_t value)
105{
106 uint8_t buf[k_rc_hex_nibbles];
107 for (uint32_t i = 0U; i < (uint32_t)k_rc_hex_nibbles; i++) {
108 const uint32_t shift = ((uint32_t)k_rc_hex_nibbles - 1U - i) * (uint32_t)k_rc_nibble_bits;
109 const uint32_t nib = (value >> shift) & (uint32_t)k_rc_nibble_mask;
110 buf[i] = (uint8_t)((nib < (uint32_t)k_rc_dec_ten) ? ('0' + nib) : ('A' + (nib - k_rc_dec_ten)));
111 }
112 rc_print(buf, (uint32_t)k_rc_hex_nibbles);
113}

116static void rc_print_uint(uint32_t value)
117{
118 uint8_t buf[k_rc_dec_ten];
119 uint32_t n = 0U;
120 if (value == 0U) {
121 buf[n] = '0';
122 n++;
123 }
124 while ((value > 0U) && (n < (uint32_t)k_rc_dec_ten)) {
125 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_rc_dec_ten));
126 n++;
127 value /= (uint32_t)k_rc_dec_ten;
128 }
129 for (uint32_t i = 0U; i < n; i++) {
130 rc_print(&buf[n - 1U - i], 1U);
131 }
132}

140static uint32_t rc_render_all(uint32_t* out_hash)
141{
142 uint32_t pages = 0U;
143 (void)ra8_reflow_get_page_count(&s_engine, &pages);
144 uint32_t hsh = (uint32_t)k_rc_fnv_offset;
145 const size_t nbytes = (size_t)k_rc_fb_w * (size_t)k_rc_fb_h * sizeof(uint16_t);
146 for (uint32_t p = 0U; p < pages; p++) {
147 (void)ra8_gfx_clear((uint32_t)k_rc_bg);
148 (void)ra8_reflow_render_page(&s_engine, p, s_framebuffer);
149 const uint8_t* fb = (const uint8_t*)s_framebuffer;
150 for (size_t i = 0U; i < nbytes; i++) {
151 hsh = (hsh ^ (uint32_t)fb[i]) * (uint32_t)k_rc_fnv_prime;
152 }
153 }
154 *out_hash = hsh;
155 return pages;
156}

159static void rc_setup_or_halt(void)
160{
161 uint32_t cpuclk0_hz = 0U;
162 if ((ra8_cgc_init() != k_ra8_ok) || (ra8_mstp_init() != k_ra8_ok)) {
163 rc_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);

167 }
168 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
169 rc_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
170 }
171 if (ra8_board_uart_console_init((uint32_t)k_rc_uart_baud) != k_ra8_ok) {
172 rc_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);

192 rc_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);

199 }
200 if (ra8_reflow_init((uint16_t)k_rc_fb_w,
201 (uint16_t)k_rc_fb_h,
202 k_ahem_ttf,

204 (uint16_t)k_rc_font_px,
205 (uint32_t)k_rc_ink,

207 &s_engine) != k_ra8_ok) {
208 rc_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
209 }
210 uint32_t pages = 0U;
211 if (ra8_reflow_layout_chapter(&s_engine,
212 (const uint8_t*)k_rc_chapter,
213 (uint32_t)(sizeof(k_rc_chapter) - 1U),
214 &pages) != k_ra8_ok) {
215 rc_panic_halt(k_msg_lerr, (uint32_t)sizeof(k_msg_lerr) - 1U);
216 }
217
218 uint32_t crc1 = 0U;
219 const uint32_t pages1 = rc_render_all(&crc1);
220
221 /* Re-flow at a larger size on the cached chapter; render again. */
222 if (ra8_reflow_set_font_size(&s_engine, (uint16_t)k_rc_reflow_px) != k_ra8_ok) {
223 rc_panic_halt(k_msg_lerr, (uint32_t)sizeof(k_msg_lerr) - 1U);
224 }
225 uint32_t crc2 = 0U;
226 const uint32_t pages2 = rc_render_all(&crc2);
227
228 rc_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
229 rc_print_uint(pages1);
230 rc_print(k_msg_crc, (uint32_t)sizeof(k_msg_crc) - 1U);
231 rc_print_hex(crc1);

233 rc_print_uint(pages2);
234 rc_print(k_msg_crc, (uint32_t)sizeof(k_msg_crc) - 1U);
235 rc_print_hex(crc2);
236 rc_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
237
238 while (1) {
239 __asm__ volatile("wfi");
240 }
