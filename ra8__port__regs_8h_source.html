156RA8_HW_REGISTER_ACCESS
157static inline volatile r_port_regs_t* ra8_port(ra8_port_t port)
158{
159 if ((uint8_t)port > k_ra8_port_max) {
160 return nullptr;
161 }
162 const uintptr_t addr = k_ra8_port0_base_addr + ((uintptr_t)port * k_ra8_port_stride);
163 return (volatile r_port_regs_t*)addr;
164}
