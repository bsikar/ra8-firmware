118void ra8_epub_internal_join_path(const char* dir, const char* name, char* dst, size_t cap)
119{
120 if (dst == nullptr || cap == 0U) {
121 return;
122 }
123 dst[0] = '\0';
124 size_t off = 0U;
125 if (dir != nullptr) {
126 while (off + 1U < cap && dir[off] != '\0') {
127 dst[off] = dir[off];
128 ++off;
129 }
130 }
131 if (name != nullptr) {
132 size_t i = 0U;
133 while (off + 1U < cap && name[i] != '\0') {
134 dst[off] = name[i];
135 ++off;
136 ++i;
137 }
138 }
139 dst[off] = '\0';
140}

165static ra8_err_t priv_locate_extract(mz_zip_archive* zip,
166 const char* prefixed_path,
167 const char* bare_path,
168 uint8_t* out_buf,
169 size_t max_len,
170 size_t* got_len)
171{
172 int32_t file_idx = mz_zip_reader_locate_file(zip, prefixed_path, nullptr, 0U);
173 if (file_idx < 0) {
174 file_idx = mz_zip_reader_locate_file(zip, bare_path, nullptr, 0U);
175 }
176 if (file_idx < 0) {
177 return k_ra8_err_not_found;
178 }
179 mz_zip_archive_file_stat st;
180 /* Defensive: stat can only fail here if the archive central directory is
181 * corrupt after a successful locate step (data race or memory error). */
182 if (mz_zip_reader_file_stat(zip, (mz_uint)file_idx, &st) == MZ_FALSE) {
183 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE */
184 }
185 const ra8_err_t gerr = ra8_epub_zip_guard_entry(&st);
186 if (gerr != k_ra8_ok) {
187 return gerr; /* lying header / declared bomb: reject before inflation */
188 }
189 if ((size_t)st.m_uncomp_size > max_len) {
190 return k_ra8_err_no_mem;
191 }
192 /* Defensive: extract can only fail here if the compressed stream is
193 * corrupt (bad CRC or LZ data), which requires a malformed archive. */
194 if (mz_zip_reader_extract_to_mem(zip, (mz_uint)file_idx, out_buf, max_len, 0U) == MZ_FALSE) {
195 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE */
196 }
197 *got_len = (size_t)st.m_uncomp_size;
198 return k_ra8_ok;
199}

219static ra8_err_t priv_font_init(const ra8_epub_book_t* book, stbtt_fontinfo* out_font)
220{
221 /* Guard against malformed font blobs: stbtt_GetFontOffsetForIndex
222 * returns -1 for non-TTF input, and stbtt_InitFont dereferences past
223 * the buffer when handed that offset. */
224 const int32_t offset = stbtt_GetFontOffsetForIndex(book->font_data, 0);
225 if (offset < 0) {
226 return k_ra8_err_validation_failed;
227 }
228 /* Bound-check the sfnt table directory before stbtt_InitFont walks it: the
229 * font bytes are attacker-controlled (an EPUB @font-face / ra8_epub_set_font
230 * blob) and stb_truetype reads the directory with no length check (#217). */
231 if (!ra8_stbtt_sfnt_dir_in_bounds(book->font_data, book->font_size, (uint32_t)offset)) {
232 return k_ra8_err_validation_failed;
233 }
234 if (stbtt_InitFont(out_font, book->font_data, (int)book->font_size, offset) == 0) {
235 return k_ra8_err_validation_failed;
236 }
237 return k_ra8_ok;
238}

264RA8_INTERNAL
265static ra8_err_t priv_render_into(const stbtt_fontinfo* font,
266 int32_t codepoint,
267 float font_size,
268 uint8_t* out_bitmap,
269 size_t max_pixels,
270 uint32_t* out_w,
271 uint32_t* out_h)
272{
273 /* No-allocation glyph rasterisation. stbtt_GetCodepointBitmap()
274 * calls STBTT_malloc internally; instead use the two-step "Box"
275 * + "Make" path that writes into the caller's buffer. NASA Rule 3
276 * compliance. */
277 const float scale = stbtt_ScaleForPixelHeight(font, font_size);
278 int x0 = 0;
279 int y0 = 0;
280 int x1 = 0;
281 int y1 = 0;
282 stbtt_GetCodepointBitmapBox(font, codepoint, scale, scale, &x0, &y0, &x1, &y1);
283 const int w = x1 - x0;
284 const int h = y1 - y0;
285 /* Defensive: stbtt_GetCodepointBitmapBox guarantees x1 >= x0 and y1 >= y0
286 * for any glyph in a well-formed font that passed stbtt_InitFont. */
287 if (ra8_epub_internal_glyph_dim_invalid(w, h)) {
288 return k_ra8_err_validation_failed; /* GCOVR_EXCL_LINE */
289 }
290 const size_t total = (size_t)w * (size_t)h;
291 if (total > max_pixels) {
292 return k_ra8_err_no_mem;
293 }
294 if (total > 0U) {
295 /* Stride = w (tightly packed alpha-8). MakeCodepointBitmap writes
296 * directly into out_bitmap with no malloc. */
297 stbtt_MakeCodepointBitmap(font, out_bitmap, w, h, w, scale, scale, codepoint);
298 }
299 *out_w = (uint32_t)w;
300 *out_h = (uint32_t)h;
301 return k_ra8_ok;
302}

392ra8_err_t ra8_epub_toc_entry_to_chapter(const ra8_epub_book_t* book,
393 uint16_t toc_idx,
394 uint16_t* out_chapter_idx)
395{
396 if (book == nullptr || out_chapter_idx == nullptr) {
397 return k_ra8_err_null_ptr;
398 }
399 if (book->in_use == 0U) {
400 return k_ra8_err_not_initialized;
401 }
402 if (toc_idx >= book->toc_count) {
403 return k_ra8_err_out_of_range;
404 }
405
406 /* Match the entry href up to (but excluding) any "#fragment": the spine
407 * stores whole-document paths, while a TOC href may target an anchor. */
408 const char* href = book->toc[toc_idx].href;
409 const char* hash = strchr(href, '#');
410 const size_t base_len = (hash != nullptr) ? (size_t)(hash - href) : strlen(href);
411
412 for (uint16_t i = 0U; i < book->chapter_count; i++) {
413 const char* path = book->chapter_paths[i];
414 if (strlen(path) != base_len) {
415 continue;
416 }
417 if (strncmp(path, href, base_len) != 0) {
418 continue;
419 }
420 *out_chapter_idx = i;
421 return k_ra8_ok;
422 }
423 return k_ra8_err_not_found;
424}
