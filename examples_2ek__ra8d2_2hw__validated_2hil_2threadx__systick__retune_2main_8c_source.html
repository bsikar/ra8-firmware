213/* NOLINTNEXTLINE(misc-use-internal-linkage) -- exported symbol expected by ThreadX. */
214void tx_application_define(void* first_unused_memory)
215{
216 (void)first_unused_memory;
217
218 /* Reprogram SysTick.LOAD from the live CPUCLK0 (issue #287). */
219 const ra8_err_t retune_err = ra8_threadx_systick_retune();
220
221 /* Independently recompute the expected reload from the live clock and
222 * compare against what retune actually left in SYST_RVR. Any deviation
223 * means the kernel tick would drift, so flag it for the probe. */
224 uint32_t cpuclk_hz = 0U;
225 uint32_t expected = 0U;
226 const ra8_err_t clk_err = ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk_hz);
227 const ra8_err_t rel_err =
228 ra8_threadx_systick_reload_for(cpuclk_hz, (uint32_t)k_ra8_threadx_tick_hz, &expected);
229 const uint32_t live_rvr = retune_read_syst_rvr();
230
231 g_threadx_retune_reload = expected;
232 if ((retune_err != k_ra8_ok) || (clk_err != k_ra8_ok) || (rel_err != k_ra8_ok) ||
233 (live_rvr != expected)) {
234 g_threadx_retune_bad = 1U;
235 }
236
237 const UINT err = tx_thread_create(&s_retune_thread,
238 "retune",
239 retune_thread_entry,
240 0U,
241 s_retune_stack,
242 (ULONG)k_retune_thread_stack_bytes,
243 (UINT)k_retune_thread_priority,
244 (UINT)k_retune_thread_priority,
245 TX_NO_TIME_SLICE,
246 TX_AUTO_START);
247 if (err != TX_SUCCESS) {
248 while (1) {
249 __asm__ volatile("wfi");
250 }
251 }
252}
253#endif /* !RA8_OFF_TARGET */
