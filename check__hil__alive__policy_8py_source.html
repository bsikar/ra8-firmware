4"""Enforce the HIL "honest contract" gate.
5
6Every hil.conf under examples/ek_ra8d2/hw_validated/hil/ must declare
7HIL_MODE to one of:
8 - uart_scrape (UART success-banner assertion)
9 - usb_cdc (Pi-side USB host probe)
10 - usb_hid (Pi-side USB HID host-binding probe)
11 - usb_msc (Pi-side USB Mass Storage SCSI-attach probe)
12 - jlink_memprobe (named-symbol counter advance via SWD)
13 - hil_eth_tcp (Pi-as-peer TCP/UDP/HTTP probe)
14 - rtt_scrape (SEGGER RTT banner assertion via JLink mem dump)
15 - alive ONLY if HIL_FAULT_EXPECTED=1 is also set
16 (legitimate fault-recovery test path -- the
17 handler is required to latch CFSR != 0 and the
18 app must emit a positive UART banner that is
19 not in hil_check_alive.sh's negative regex)
20
21Plain HIL_MODE=alive without HIL_FAULT_EXPECTED is the historical
22"PC happens to be in MRAM" loose check; it lets silently-broken apps
23pass CI and is forbidden under hw_validated/hil/. Apps that genuinely
24have no observable signal yet (panic-halt at init, no
25instrumentation) must live in hw_pending/ until they are fixed AND
26instrumented.
27
28Exit:
29 0 every hil.conf in hw_validated/hil/ satisfies the policy
30 1 one or more hil.confs violate
31 2 usage / unreachable repo root, or a collapsed scan (see HIL_CONF_FLOOR)
32"""
33
34from __future__ import annotations

36import pathlib
37import re
38import sys
39from collections.abc import Iterable

41# A hw_validated HIL suite this size cannot legitimately collapse to a handful
42# of apps. If the glob returns less than this, something broke (a bad repo
43# root, a renamed hil/ directory) and reporting "0 findings" would be a lie:
44# every app would be trivially compliant because none was read. Measured
45# 2026-07-28: 114 hil.conf files. Same trip-wire as check_ruff.py.
46HIL_CONF_FLOOR = 90
47
48ALLOWED_MODES: frozenset[str] = frozenset(
49 {
50 "uart_scrape",
51 "usb_cdc",
52 "usb_hid",
53 "usb_msc",
54 "jlink_memprobe",
55 "hil_eth_tcp",
56 "rtt_scrape",
57 }
58)

61def _iter_hil_confs(repo_root: pathlib.Path) -> Iterable[pathlib.Path]:
62 """Yield every hil.conf under examples/ek_ra8d2/hw_validated/hil/."""
63 hil_dir = repo_root / "examples" / "ek_ra8d2" / "hw_validated" / "hil"
64 if not hil_dir.is_dir():
65 return ()
66 return hil_dir.glob("*/hil.conf")

69def _parse_kv(conf: pathlib.Path) -> dict[str, str]:
70 """Extract simple KEY=VALUE lines (no shell expansion) from a hil.conf."""
71 out: dict[str, str] = {}
72 pat = re.compile(r"^\s*([A-Z_][A-Z_0-9]*)\s*=\s*(.*?)\s*$")
73 for raw in conf.read_text().splitlines():
74 line = raw.split("#", 1)[0].rstrip()
75 m = pat.match(line)
76 if not m:
77 continue
78 key, val = m.group(1), m.group(2)
79 if val.startswith(('"', "'")) and val.endswith(val[0]) and len(val) >= 2: # noqa: PLR2004 # min quoted-string length
80 val = val[1:-1]
81 out[key] = val
82 return out

85def main() -> int:
86 """Fail any hw_validated HIL app whose hil.conf does not assert a real outcome.
87
88 The rule being enforced is that ``HIL_MODE=alive`` proves only that the
89 board did not hang -- it cannot distinguish a working app from one that
90 booted and did nothing. It is therefore allowed under hw_validated/ only
91 when ``HIL_FAULT_EXPECTED=1``, i.e. when not faulting IS the assertion.
92 Any other app claiming validation must scrape a banner or probe a symbol,
93 or move to hw_pending/.
94
95 A missing HIL_MODE is treated as a violation rather than a default, since
96 a silently defaulted mode is how an unasserted app would slip in.
97
98 Enforces HIL_CONF_FLOOR before reading anything and exits 2 below it. An
99 empty glob would otherwise report "0 findings" -- indistinguishable from a
100 fully compliant suite, and produced by the scan having read nothing.
101
102 Returns 1 with one remediation-bearing message per offending hil.conf, 0
103 when every conf under hw_validated/hil/ declares an asserting mode, 2 when
104 the enumeration is too small to trust.
105 """
106 repo_root = pathlib.Path(__file__).resolve().parents[2]
107 violations: list[str] = []
108
109 confs = sorted(_iter_hil_confs(repo_root))
110 if len(confs) < HIL_CONF_FLOOR:
111 sys.stderr.write(
112 f"check_hil_alive_policy.py: FATAL -- only {len(confs)} hil.conf file(s) in "
113 f"scope, floor is {HIL_CONF_FLOOR}.\n"
114 " A collapsed scope reports a compliant suite because it checked nothing.\n"
115 )
116 return 2
117
118 for conf in confs:
119 kv = _parse_kv(conf)
120 mode = kv.get("HIL_MODE", "")
121 fault_expected = kv.get("HIL_FAULT_EXPECTED", "0")
122 app = conf.parent.name
123
124 if mode in ALLOWED_MODES:
125 continue
126 if mode == "alive" and fault_expected == "1":
127 continue
128
129 if mode == "alive":
130 violations.append(
131 f"{conf.relative_to(repo_root)}: HIL_MODE=alive without "
132 f"HIL_FAULT_EXPECTED=1 is forbidden under "
133 f"hw_validated/hil/. Either:\n"
134 f" - instrument the app (g_<x>_match symbol + "
135 f"HIL_MODE=jlink_memprobe), OR\n"
136 f" - add a UART success banner + HIL_MODE=uart_scrape, OR\n"
137 f" - move the app to examples/ek_ra8d2/hw_pending/{app}/."
138 )
139 elif mode == "":
140 violations.append(f"{conf.relative_to(repo_root)}: missing HIL_MODE")
141 else:
142 violations.append(
143 f"{conf.relative_to(repo_root)}: HIL_MODE={mode!r} is not "
144 f"in the allowed set "
145 f"({[*sorted(ALLOWED_MODES), 'alive (with HIL_FAULT_EXPECTED=1)']})"
146 )
147
148 if violations:
149 sys.stderr.write(
150 f"check_hil_alive_policy.py: {len(violations)} violation(s) in hw_validated/hil/:\n"
151 )
152 for v in violations:
153 sys.stderr.write(f" - {v}\n")
154 sys.stderr.write(
155 "\nThe HIL 'honest contract' rule: every app under "
156 "hw_validated/hil/ must prove its feature works end-to-end on "
157 "real hardware, not just that the chip booted.\n"
158 )
159 return 1
160
161 print(f"check_hil_alive_policy.py: 0 findings across {len(confs)} hil.conf file(s).")
162 return 0
163
164
165if __name__ == "__main__":
166 sys.exit(main())
