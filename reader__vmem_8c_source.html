113static uint64_t rv_rng(uint64_t* s)
114{
115 uint64_t x = *s;
116 x ^= x << (uint32_t)k_rv_xs_shift_a;
117 x ^= x >> (uint32_t)k_rv_xs_shift_b;
118 x ^= x << (uint32_t)k_rv_xs_shift_c;
119 *s = x;
120 return x;
121}

146static void rv_layout_book(rv_driver_t* d)
147{
148 uint32_t cur = (uint32_t)k_rv_header_frames;
149 for (uint32_t c = 0U; c < (uint32_t)k_rv_chapters; ++c) {
150 const uint32_t count =
151 (uint32_t)k_rv_chap_min_fr + rv_below(&d->rng, (uint32_t)k_rv_chap_span_fr);
152 d->chapters[c].first = cur;
153 d->chapters[c].count = count;
154 cur += count;
155 }
156 d->total_frames = cur;
157}

160static void rv_touch(rv_driver_t* d, uint32_t frame)
161{
162 const uint64_t offset = (uint64_t)frame * (uint64_t)k_rv_frame_bytes;
163 void* page = nullptr;
164 const ra8_err_t err = ra8_vmem_get(d->vm, d->object_id, offset, &page);
165 if (err == k_ra8_ok) {
166 (void)ra8_vmem_put(d->vm, page);
167 } else {
168 (void)fprintf(stderr, "vm_get(frame %u) failed: %d\n", frame, (int)err);
169 }
170 (void)fprintf(d->trace, "%u %u\n", (unsigned)d->object_id, (unsigned)frame);
171 d->accesses++;
172}

183static void rv_phase_linear(rv_driver_t* d)
184{
185 for (uint32_t c = 0U; c < (uint32_t)k_rv_chapters; ++c) {
186 const rv_chapter_t ch = d->chapters[c];
187 for (uint32_t i = 0U; i < ch.count; ++i) {
188 const uint32_t frame = ch.first + i;
189 rv_touch_header(d); /* the reader consults the header every page turn */
190 rv_touch(d, frame);
191 if ((rv_below(&d->rng, (uint32_t)k_rv_pct_base) < (uint32_t)k_rv_backglance_pct) &&
192 (i > 0U)) {
193 rv_touch(d, frame - 1U); /* glance back at the previous page */
194 }
195 }
196 }
197}

200static void rv_phase_toc_jumps(rv_driver_t* d)
201{
202 for (uint32_t r = 0U; r < (uint32_t)k_rv_toc_rounds; ++r) {
203 rv_touch_header(d); /* open the TOC */
204 const uint32_t c = rv_below(&d->rng, (uint32_t)k_rv_chapters);
205 const rv_chapter_t ch = d->chapters[c];
206 for (uint32_t i = 0U; (i < (uint32_t)k_rv_toc_read_fr) && (i < ch.count); ++i) {
207 rv_touch(d, ch.first + i);
208 }
209 }

213static void rv_phase_scan_resist(rv_driver_t* d)
214{
215 uint32_t scan_at = (uint32_t)k_rv_header_frames + (uint32_t)k_rv_sr_hot_fr;
216 for (uint32_t r = 0U; r < (uint32_t)k_rv_sr_rounds; ++r) {
217 for (uint32_t p = 0U; p < (uint32_t)k_rv_sr_hot_pass; ++p) {
218 for (uint32_t h = 0U; h < (uint32_t)k_rv_sr_hot_fr; ++h) {
219 rv_touch(d, (uint32_t)k_rv_header_frames + h); /* hot working set */
220 }
221 }
222 for (uint32_t s = 0U; s < (uint32_t)k_rv_sr_scan_fr; ++s) {
223 const uint32_t span = d->total_frames - (uint32_t)k_rv_header_frames;
224 const uint32_t frame = (uint32_t)k_rv_header_frames + ((scan_at + s) % span);
225 rv_touch(d, frame); /* one-shot linear flood that must not evict the hot set */
226 }
227 scan_at += (uint32_t)k_rv_sr_scan_fr;
228 }

249static bool rv_alloc_book(rv_driver_t* d)
250{
251 s_book_bytes = (uint64_t)d->total_frames * (uint64_t)k_rv_frame_bytes;
252 s_book_backing = (uint8_t*)malloc((size_t)s_book_bytes);
253 if (s_book_backing == nullptr) {
254 (void)fprintf(stderr,
255 "reader_vmem: OOM allocating %llu-byte book\n",
256 (unsigned long long)s_book_bytes);
257 return false;
258 }
259 for (uint64_t i = 0U; i < s_book_bytes; ++i) {
260 s_book_backing[i] = (uint8_t)(i * (uint64_t)k_rv_fill_mul >> (uint32_t)k_rv_fill_shift);
261 }
262 return true;
263}

266static bool rv_vmem_setup(ra8_vmem_t* vm, rv_res_t* res, uint32_t budget, ra8_vsource_t* vs)

289static void
290rv_run_and_report(rv_driver_t* d, ra8_vmem_t* vm, uint32_t budget, const char* trace_path)
291{
292 rv_phase_linear(d);
293 rv_phase_toc_jumps(d);
294 rv_phase_scan_resist(d);
295 (void)fclose(d->trace);
296
297 uint32_t hits = 0U;
298 uint32_t miss = 0U;
299 uint32_t ev = 0U;
300 (void)ra8_vmem_stats(vm, &hits, &miss, &ev);
301 const double hit_pct =
302 (d->accesses == 0U) ? 0.0 : ((double)k_rv_pct_base * (double)hits / (double)d->accesses);
303 (void)fprintf(stderr,
304 "reader_vmem: book=%u frames (%llu bytes), budget=%u frames\n"
305 " accesses=%llu trace=%s\n"
306 " ra8_vmem SLRU: hits=%u misses=%u evictions=%u hit_rate=%.2f%%\n",
307 d->total_frames,
308 (unsigned long long)s_book_bytes,
309 budget,
310 (unsigned long long)d->accesses,
311 trace_path,
312 hits,
313 miss,
314 ev,
315 hit_pct);
316}

318int main(int argc, char** argv)
319{
320 const char* trace_path = (argc > 1) ? argv[1] : "reader_vmem.trace";
321 const uint32_t budget = (argc > 2) ? (uint32_t)strtoul(argv[2], nullptr, (int)k_rv_decimal_base)
322 : (uint32_t)k_rv_def_budget;
323
324 rv_driver_t d = {};
325 d.rng = (uint64_t)k_rv_rng_seed;
326 rv_layout_book(&d);
327 if (!rv_alloc_book(&d)) {
328 return 1;
329 }
330
331 ra8_vsource_obj_t objs[k_rv_max_objs] = {};
332 ra8_vsource_t vs = {};
333 uint32_t id = 0U;
334 if ((ra8_vsource_init(&vs, objs, (uint32_t)k_rv_max_objs) != k_ra8_ok) ||
335 (ra8_vsource_add_paged(&vs, rv_read, nullptr, 0U, s_book_bytes, &id) != k_ra8_ok)) {
336 free(s_book_backing);
337 return 1;
338 }
339 d.object_id = id;
340
341 ra8_vmem_t vm = {};
342 rv_res_t res = {};
343 if (!rv_vmem_setup(&vm, &res, budget, &vs)) {
344 rv_res_free(&res);
345 free(s_book_backing);
346 return 1;
347 }
348 d.vm = &vm;
349 d.trace = fopen(trace_path, "w");
350 if (d.trace == nullptr) {
351 (void)fprintf(stderr, "reader_vmem: cannot open trace file %s\n", trace_path);
352 rv_res_free(&res);
353 free(s_book_backing);
354 return 1;
355 }
356
357 rv_run_and_report(&d, &vm, budget, trace_path);
358
359 rv_res_free(&res);
360 free(s_book_backing);
361 return 0;
362}
