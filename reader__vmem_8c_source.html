134static uint64_t rv_rng(uint64_t* s)
135{
136 uint64_t x = *s;
137 x ^= x << (uint32_t)k_rv_xs_shift_a;
138 x ^= x >> (uint32_t)k_rv_xs_shift_b;
139 x ^= x << (uint32_t)k_rv_xs_shift_c;
140 *s = x;
141 return x;
142}

233static void rv_layout_book(rv_driver_t* d)
234{
235 uint32_t cur = (uint32_t)k_rv_header_frames;
236 for (uint32_t c = 0U; c < (uint32_t)k_rv_chapters; ++c) {
237 const uint32_t count =
238 (uint32_t)k_rv_chap_min_fr + rv_below(&d->rng, (uint32_t)k_rv_chap_span_fr);
239 d->chapters[c].first = cur;
240 d->chapters[c].count = count;
241 cur += count;
242 }

266static void rv_touch(rv_driver_t* d, uint32_t frame)
267{
268 const uint64_t offset = (uint64_t)frame * (uint64_t)k_rv_frame_bytes;
269 void* page = nullptr;
270 const ra8_err_t err = ra8_vmem_get(d->vm, d->object_id, offset, &page);
271 if (err == k_ra8_ok) {
272 (void)ra8_vmem_put(d->vm, page);
273 } else {
274 (void)fprintf(stderr, "vm_get(frame %u) failed: %d\n", frame, (int)err);
275 }
276 (void)fprintf(d->trace, "%u %u\n", (unsigned)d->object_id, (unsigned)frame);
277 d->accesses++;

325static void rv_phase_linear(rv_driver_t* d)
326{
327 for (uint32_t c = 0U; c < (uint32_t)k_rv_chapters; ++c) {
328 const rv_chapter_t ch = d->chapters[c];
329 for (uint32_t i = 0U; i < ch.count; ++i) {
330 const uint32_t frame = ch.first + i;
331 rv_touch_header(d); /* the reader consults the header every page turn */
332 rv_touch(d, frame);
333 if ((rv_below(&d->rng, (uint32_t)k_rv_pct_base) < (uint32_t)k_rv_backglance_pct) &&
334 (i > 0U)) {
335 rv_touch(d, frame - 1U); /* glance back at the previous page */
336 }
337 }
338 }

361static void rv_phase_toc_jumps(rv_driver_t* d)
362{
363 for (uint32_t r = 0U; r < (uint32_t)k_rv_toc_rounds; ++r) {
364 rv_touch_header(d); /* open the TOC */
365 const uint32_t c = rv_below(&d->rng, (uint32_t)k_rv_chapters);
366 const rv_chapter_t ch = d->chapters[c];
367 for (uint32_t i = 0U; (i < (uint32_t)k_rv_toc_read_fr) && (i < ch.count); ++i) {
368 rv_touch(d, ch.first + i);
369 }
370 }

392static void rv_phase_scan_resist(rv_driver_t* d)
393{
394 uint32_t scan_at = (uint32_t)k_rv_header_frames + (uint32_t)k_rv_sr_hot_fr;
395 for (uint32_t r = 0U; r < (uint32_t)k_rv_sr_rounds; ++r) {
396 for (uint32_t p = 0U; p < (uint32_t)k_rv_sr_hot_pass; ++p) {
397 for (uint32_t h = 0U; h < (uint32_t)k_rv_sr_hot_fr; ++h) {
398 rv_touch(d, (uint32_t)k_rv_header_frames + h); /* hot working set */
399 }
400 }
401 for (uint32_t s = 0U; s < (uint32_t)k_rv_sr_scan_fr; ++s) {
402 const uint32_t span = d->total_frames - (uint32_t)k_rv_header_frames;
403 const uint32_t frame = (uint32_t)k_rv_header_frames + ((scan_at + s) % span);
404 rv_touch(d, frame); /* one-shot linear flood that must not evict the hot set */
405 }
406 scan_at += (uint32_t)k_rv_sr_scan_fr;
407 }

468static bool rv_alloc_book(rv_driver_t* d)
469{
470 s_book_bytes = (uint64_t)d->total_frames * (uint64_t)k_rv_frame_bytes;
471 s_book_backing = (uint8_t*)malloc((size_t)s_book_bytes);
472 if (s_book_backing == nullptr) {
473 (void)fprintf(stderr,
474 "reader_vmem: OOM allocating %llu-byte book\n",
475 (unsigned long long)s_book_bytes);
476 return false;
477 }
478 for (uint64_t i = 0U; i < s_book_bytes; ++i) {
479 s_book_backing[i] = (uint8_t)(i * (uint64_t)k_rv_fill_mul >> (uint32_t)k_rv_fill_shift);
480 }
481 return true;
482}

585int main(int argc, char** argv)
586{
587 const char* trace_path = (argc > 1) ? argv[1] : "reader_vmem.trace";
588 const uint32_t budget = (argc > 2) ? (uint32_t)strtoul(argv[2], nullptr, (int)k_rv_decimal_base)
589 : (uint32_t)k_rv_def_budget;
590
591 rv_driver_t d = {};
592 d.rng = (uint64_t)k_rv_rng_seed;
593 rv_layout_book(&d);
594 if (!rv_alloc_book(&d)) {
595 return 1;
596 }
597
598 ra8_vsource_obj_t objs[k_rv_max_objs] = {};
599 ra8_vsource_t vs = {};
600 uint32_t id = 0U;
601 if ((ra8_vsource_init(&vs, objs, (uint32_t)k_rv_max_objs) != k_ra8_ok) ||
602 (ra8_vsource_add_paged(&vs, rv_read, nullptr, 0U, s_book_bytes, &id) != k_ra8_ok)) {
603 free(s_book_backing);
604 return 1;
605 }
606 d.object_id = id;
607
608 ra8_vmem_t vm = {};
609 rv_res_t res = {};
610 if (!rv_vmem_setup(&vm, &res, budget, &vs)) {
611 rv_res_free(&res);
612 free(s_book_backing);
613 return 1;
614 }
615 d.vm = &vm;
616 d.trace = fopen(trace_path, "w");
617 if (d.trace == nullptr) {
618 (void)fprintf(stderr, "reader_vmem: cannot open trace file %s\n", trace_path);
619 rv_res_free(&res);
620 free(s_book_backing);
621 return 1;
622 }
623
624 rv_run_and_report(&d, &vm, budget, trace_path);
625
626 rv_res_free(&res);
627 free(s_book_backing);
628 return 0;
629}
