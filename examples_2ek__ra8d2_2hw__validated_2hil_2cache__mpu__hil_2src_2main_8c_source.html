267static bool cache_mpu_test_cacheable_rw(void)
268{
269 for (uint32_t i = 0U; i < (uint32_t)k_cache_mpu_buf_bytes; i++) {
270 s_cache_mpu_buf[i] =
271 (uint8_t)((i * (uint32_t)k_cache_mpu_pat_mul) + (uint32_t)k_cache_mpu_pat_add);
272 }
273
274 /* Force the freshly written (dirty) lines back to SRAM and drop them, so the
275 * verify pass below must refill from memory -- the full write-back + refill
276 * round-trip with the D-cache on. */
277 if (ra8_cache_dcache_clean_invalidate_by_addr(s_cache_mpu_buf,
278 (uint32_t)sizeof(s_cache_mpu_buf)) != k_ra8_ok) {
279 return false;
280 }
281
282 for (uint32_t i = 0U; i < (uint32_t)k_cache_mpu_buf_bytes; i++) {
283 const uint8_t expected =
284 (uint8_t)((i * (uint32_t)k_cache_mpu_pat_mul) + (uint32_t)k_cache_mpu_pat_add);
285 if (s_cache_mpu_buf[i] != expected) {
286 return false;
287 }
288 }
289 return true;
290}

475void main(void)
476{
477 ra8_log_init();
478 ra8_log_info("CACHE_MPU", "==== cache_mpu_hil: M85 L1 cache + MPU self-test ====");
479
480 if (!cache_mpu_setup()) {
481 ra8_log_info("CACHE_MPU", "setup FAILED (CGC / console) -- halting");
482 cache_mpu_emit(k_cache_mpu_fail_setup, sizeof(k_cache_mpu_fail_setup) - 1U);
483 cache_mpu_wfi_forever();
484 }
485
486 const cache_mpu_result_t result = cache_mpu_run_selftest();
487
488 switch (result) {
489 case k_cache_mpu_result_pass:
490 ra8_log_info("CACHE_MPU", "cache+mpu PASS (cacheable RW + MRAM const + Device MMIO)");
491 cache_mpu_emit(k_cache_mpu_pass_banner, sizeof(k_cache_mpu_pass_banner) - 1U);
492 break;
493 case k_cache_mpu_result_fail_sram:
494 ra8_log_info("CACHE_MPU", "cacheable-SRAM step FAILED");
495 cache_mpu_emit(k_cache_mpu_fail_sram, sizeof(k_cache_mpu_fail_sram) - 1U);
496 break;
497 case k_cache_mpu_result_fail_mram:
498 ra8_log_info("CACHE_MPU", "MRAM-const step FAILED");
499 cache_mpu_emit(k_cache_mpu_fail_mram, sizeof(k_cache_mpu_fail_mram) - 1U);
500 break;
501 case k_cache_mpu_result_fail_mmio:
502 default:
503 ra8_log_info("CACHE_MPU", "device-MMIO step FAILED");
504 cache_mpu_emit(k_cache_mpu_fail_mmio, sizeof(k_cache_mpu_fail_mmio) - 1U);
505 break;
506 }
507
508 cache_mpu_wfi_forever();
509}
