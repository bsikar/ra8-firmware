104static void pdctr_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
105{
106 (void)uc;
107 (void)addr;
108 (void)size;
109 /* HUM Ch 13.1 Table 13.1 p 521 lists PDCTRGD under PRC1: a write with that
110 * group locked is discarded by the hardware with no fault and no flag. */
111 if (!board_prcr_group_unlocked((uint16_t)k_board_prcr_grp1_lpm)) {
112 s_pdctr.blocked++;
113 return;
114 }
115 const bool power_off = ((uint8_t)value & (uint8_t)k_pdctr_pdde_mask) != 0U;
116 if (power_off) {
117 s_pdctr.pdctrgd = (uint8_t)(k_pdctr_pdde_mask | k_pdctr_pdpgsf_mask);
118 return;
119 }
120 if (!board_pdctr_graphics_powered()) {
121 s_pdctr.power_on++;
122 }
123 s_pdctr.pdctrgd = 0U; /* PDDE=0, PDCSF=0 (settled), PDPGSF=0 (powered). */
124}

127static void pdctr_report(void)
128{
129 if (s_pdctr.blocked != 0U) {
130 /* Loud: PDCTRGD writes need PRCR.PRC1 unlocked or the domain stays dark. */
131 (void)fprintf(stderr,
132 " PWR-GRAPHICS : PDCTRGD writes DROPPED=%u (PRCR.PRC1 locked: unlock 0xA502)\n",
133 s_pdctr.blocked);
134 return;
135 }
136 if (s_pdctr.power_on == 0U) {
137 return; /* Untouched: stay quiet (domain still gated, as at reset). */
138 }
139 (void)fprintf(stderr, " PWR-GRAPHICS : graphics power domain on (DRW/GLCDC/MIPI/VIN)\n");
140}
