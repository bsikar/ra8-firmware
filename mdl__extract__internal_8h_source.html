81{
82 const size_t nlen = strlen(needle);
83 if ((nlen == 0U) || (hay_len < nlen)) {
84 return nullptr;
85 }
86 for (size_t i = 0U; i + nlen <= hay_len; ++i) {
87 size_t j = 0U;
88 while ((j < nlen) && (internal_lc(hay[i + j]) == internal_lc(needle[j]))) {
89 ++j;
90 }
91 if (j == nlen) {
92 return hay + i;

94 }

143internal_read_eq_value(const char* tag, size_t tag_len, size_t k, char* out, size_t out_cap)
144{
145 k = internal_skip_ws(tag, tag_len, k);
146 if (!((k < tag_len) && (tag[k] == '='))) {
147 return false;
148 }
149 k = internal_skip_ws(tag, tag_len, k + 1U);
150 if (!((k < tag_len) && ((tag[k] == '"') || (tag[k] == '\'')))) {
151 return false;
152 }
153 const char quote = tag[k];
154 ++k;
155 size_t n = 0U;
156 while ((k < tag_len) && (tag[k] != quote) && (n + 1U < out_cap)) {
157 out[n] = tag[k];
158 ++n;
159 ++k;
160 }
161 if ((k < tag_len) && (tag[k] == quote)) {
162 out[n] = '\0';
163 return true;
164 }
165 return false;

188RA8_INTERNAL static inline bool internal_find_attr_value(const char* tag,
189 size_t tag_len,
190 const char* attr,
191 char* out,
192 size_t out_cap)
193{
194 const size_t alen = strlen(attr);
195 size_t pos = 0U;
196 while (pos + alen <= tag_len) {
197 const char* hit = internal_find_ci(tag + pos, tag_len - pos, attr);
198 if (hit == nullptr) {
199 return false;
200 }
201 const size_t idx = (size_t)(hit - tag);
202 const bool left_ok = (idx == 0U) || internal_is_attr_boundary(tag[idx - 1U]);
203 if (left_ok && internal_read_eq_value(tag, tag_len, idx + alen, out, out_cap)) {
204 return true;
205 }
206 pos = idx + alen;
207 }
208 return false;
209}
