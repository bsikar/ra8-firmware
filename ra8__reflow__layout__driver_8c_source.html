54static ra8_err_t priv_layout_tokens(ra8_reflow_t* engine, const stbtt_fontinfo* font)
55{
56 priv_cursor_t cur = {
57 .x = (int32_t)k_ra8_reflow_margin_px,
58 .y = (int32_t)k_ra8_reflow_margin_px,
59 .line_top = (int32_t)k_ra8_reflow_margin_px,
60 .line_height_px = ra8_reflow_layout_line_height(engine->font_px),
61 .active_font_px = engine->font_px,
62 .active_style = k_ra8_reflow_style_normal,
63 .indent_px = 0U,
64 .page_first_glyph = 0U,
65 .page_first_image = 0U,
66 .line_first_glyph = 0U,
67 .line_has_content = 0U,
68 .align = (uint8_t)k_ra8_reflow_align_left,
69 .reserved8 = {0U, 0U},
70 };
71
72 uint32_t i = 0U;
73 while (i < engine->token_count) {
74 const ra8_reflow_token_t* tok = &engine->tokens[i];
75 if ((tok->kind == (uint8_t)k_ra8_reflow_tok_block_start) &&
76 (tok->tag == (uint8_t)k_ra8_reflow_tag_table)) {
77 ra8_err_t err = ra8_reflow_layout_table(engine, &cur, font, i, &i);
78 if (err != k_ra8_ok) {
79 return err;
80 }
81 } else {
82 ra8_err_t err = ra8_reflow_layout_apply_token(engine, &cur, font, tok);
83 if (err != k_ra8_ok) {
84 return err;
85 }
86 i++;
87 }
88 }
89
90 /* Flush the final page if it has glyph or image content. */
91 if ((engine->glyph_count > cur.page_first_glyph) ||
92 (engine->image_box_count > cur.page_first_image)) {
93 if (!ra8_reflow_layout_finish_page(engine, &cur)) {
94 return k_ra8_err_no_mem;
95 }
96 }
97 return k_ra8_ok;
98}

105ra8_err_t ra8_reflow_run_layout(ra8_reflow_t* engine)
106{
107 if (engine == nullptr) {
108 return k_ra8_err_null_ptr;
109 }
110 if (engine->in_use == 0U) {
111 return k_ra8_err_not_initialized;
112 }
113
114 /* Wipe glyph, page, image, link and anchor state but keep the token stream. */
115 engine->glyph_count = 0U;
116 engine->page_count = 0U;
117 engine->image_box_count = 0U;
118 engine->link_rect_count = 0U;
119 engine->anchor_count = 0U;
120
121 stbtt_fontinfo font;
122 ra8_err_t err = ra8_reflow_layout_init_font(engine, &font);
123 if (err != k_ra8_ok) {
124 return err;
125 }
126 err = priv_layout_tokens(engine, &font);
127 if (err != k_ra8_ok) {
128 return err;
129 }
130 /* Guarantee at least one page on non-empty input. */
131 if (ra8_reflow_internal_final_page_needed(engine->page_count, engine->token_count)) {
132 if (engine->page_count >= k_ra8_reflow_max_pages) {
133 return k_ra8_err_no_mem;
134 }
135 engine->pages[0].glyph_first = 0U;
136 engine->pages[0].glyph_count = engine->glyph_count;
137 engine->page_count = (uint32_t)k_priv_min_chapter_pages;
138 }
139 /* Build tappable link rectangles from the link-tagged glyphs (#110). */
140 ra8_reflow_layout_build_link_rects(engine, &font);
141 return k_ra8_ok;
142}

302ra8_err_t ra8_reflow_bind_font(ra8_reflow_t* engine, const uint8_t* font_data, size_t font_len)
303{
304 if (engine == nullptr || font_data == nullptr) {
305 return k_ra8_err_null_ptr;
306 }
307 if (engine->in_use == 0U) {
308 return k_ra8_err_not_initialized;
309 }
310 if (font_len < (size_t)k_ra8_reflow_min_font_bytes) {
311 return k_ra8_err_invalid_size;
312 }
313
314 /* Validate the blob BEFORE swapping; on failure the current face is kept
315 * (graceful degradation). The probe fontinfo is a stack temporary -- the
316 * layout / render paths re-init their own per-call fontinfo from font_data. */
317 const int32_t offset = stbtt_GetFontOffsetForIndex(font_data, 0);
318 /* Bound-check the sfnt table directory before stbtt_InitFont walks it: the
319 * @font-face bytes are attacker-controlled and stb_truetype reads the
320 * directory with no length check (#217). A non-sfnt blob (offset < 0) is
321 * left to the existing decision below. */
322 if (offset >= 0) {
323 if (!ra8_stbtt_sfnt_dir_in_bounds(font_data, font_len, (uint32_t)offset)) {
324 return k_ra8_err_not_supported;
325 }
326 }
327 stbtt_fontinfo probe;
328 if (offset < 0 || stbtt_InitFont(&probe, font_data, (int)font_len, offset) == 0) {
329 return k_ra8_err_not_supported;
330 }
331
332 engine->font_data = font_data;
333 engine->font_len = font_len;
334
335 /* If a chapter is already laid out, re-flow it against the new face so the
336 * change is immediate (mirrors ra8_reflow_set_font_size); otherwise the next
337 * layout picks it up. */
338 if (!ra8_reflow_internal_xhtml_invalid(engine->xhtml_buf, engine->xhtml_len)) {
339 uint32_t pages = 0U;
340 return ra8_reflow_layout_chapter(engine, engine->xhtml_buf, engine->xhtml_len, &pages);
341 }
342 return k_ra8_ok;
343}

346ra8_err_t ra8_reflow_register_face(ra8_reflow_t* engine,
347 uint8_t css_face_idx,
348 const uint8_t* blob,
349 size_t len)
350{
351 if (engine == nullptr || blob == nullptr) {
352 return k_ra8_err_null_ptr;
353 }
354 if (engine->in_use == 0U) {
355 return k_ra8_err_not_initialized;
356 }
357 if (len < (size_t)k_ra8_reflow_min_font_bytes) {
358 return k_ra8_err_invalid_size;
359 }
360 if (engine->face_count >= (uint8_t)k_ra8_reflow_max_faces) {
361 return k_ra8_err_no_mem;
362 }
363 /* Validate the blob before recording it; a bad face is rejected so layout falls
364 * back to the default face (graceful degradation). The probe is a stack
365 * temporary -- the render pass re-inits its own per-face fontinfo. */
366 const int32_t offset = stbtt_GetFontOffsetForIndex(blob, 0);
367 /* Bound-check the sfnt table directory before stbtt_InitFont walks it: the
368 * @font-face bytes are attacker-controlled and stb_truetype reads the
369 * directory with no length check (#217). A non-sfnt blob (offset < 0) is
370 * left to the existing decision below. */
371 if (offset >= 0) {
372 if (!ra8_stbtt_sfnt_dir_in_bounds(blob, len, (uint32_t)offset)) {
373 return k_ra8_err_not_supported;
374 }
375 }
376 stbtt_fontinfo probe;
377 if (offset < 0 || stbtt_InitFont(&probe, blob, (int)len, offset) == 0) {
378 return k_ra8_err_not_supported;
379 }
380 ra8_reflow_face_t* face = &engine->faces[engine->face_count];
381 face->blob = blob;
382 face->len = len;
383 face->css_face_idx = css_face_idx;
384 engine->face_count++;
385 return k_ra8_ok;
