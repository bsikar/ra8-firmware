77[[noreturn]] static void cpu1_main(void)
78{
79 volatile dualcore_bg_t* bg = dualcore_bg();
80
81 /* Announce we are alive. The M85 polls this field before waiting on done
82 * so that a boot failure is distinguishable from a counting stall. */
83 bg->m33_sig = (uint32_t)k_bg_m33_signature;
84 __asm volatile("dsb" ::: "memory");
85
86 /* Increment the shared counter k_bg_target_count times. Each store is
87 * individually visible to the M85 once the write buffer drains, but we
88 * only signal completion after all increments so the M85 reads a stable
89 * final value. */
90 RA8_LOOP_BOUND(k_bg_target_count);
91 for (uint32_t i = 0U; i < (uint32_t)k_bg_target_count; i++) {
92 bg->counter = bg->counter + 1U;
93 }
94
95 /* Drain the write buffer so the M85 sees the final `counter` value before
96 * it sees the `done` flag flip. */
97 __asm volatile("dsb" ::: "memory");
98 bg->done = 1U;
99
100 while (1) {
101 __asm volatile("nop");
102 }
103}

123[[noreturn]] void cpu1_reset_handler(void)
124{
125 uint32_t* dst = &g_ra8_ls_cpu1_data_start;
126 uint32_t* src = &g_ra8_ls_cpu1_data_load;
127 RA8_LOOP_BOUND_RUNTIME(g_ra8_ls_cpu1_data_end);
128 while (dst < &g_ra8_ls_cpu1_data_end) {
129 *dst = *src;
130 dst++;
131 src++;
132 }
133
134 uint32_t* bss = &g_ra8_ls_cpu1_bss_start;
135 RA8_LOOP_BOUND_RUNTIME(g_ra8_ls_cpu1_bss_end);
136 while (bss < &g_ra8_ls_cpu1_bss_end) {
137 *bss = 0U;
138 bss++;
139 }
140
141 cpu1_main();
142}
