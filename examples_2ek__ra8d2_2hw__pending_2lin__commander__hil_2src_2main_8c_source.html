216static ra8_err_t lin_hil_send_frame(uint8_t id)
217{
218 const ra8_err_t herr = ra8_sci_lin_send_header((uint8_t)k_lin_hil_channel, id);

239void main(void)
240{
241 lin_hil_setup_or_halt();
242 ra8_isr_globals_enable();
243 (void)ra8_board_uart_console_write(k_lin_hil_banner, (size_t)(sizeof(k_lin_hil_banner) - 1U));
244
245 uint8_t idx = 0U;
246 while (1) {
247 const uint8_t id = k_lin_hil_ids[idx];
248 const ra8_err_t err = lin_hil_send_frame(id);
249 if (err == k_ra8_ok) {
250 (void)ra8_board_uart_console_write(k_lin_hil_frame_msg,
251 (size_t)(sizeof(k_lin_hil_frame_msg) - 1U));
252 } else {
253 (void)ra8_board_uart_console_write(k_lin_hil_fault_msg,
254 (size_t)(sizeof(k_lin_hil_fault_msg) - 1U));
255 }
256
257 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
258 break;
259 }
260 idx = (uint8_t)((idx + 1U) % (uint8_t)k_lin_hil_id_count);
261 ra8_delay_ms((uint32_t)k_lin_hil_period_ms);
262 }
263
264 lin_hil_panic_halt();
265}
