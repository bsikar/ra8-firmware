103static void lk_print_hex(uint32_t value)
104{
105 uint8_t buf[k_lk_hex_nibbles];
106 for (uint32_t i = 0U; i < (uint32_t)k_lk_hex_nibbles; i++) {
107 const uint32_t shift = ((uint32_t)k_lk_hex_nibbles - 1U - i) * (uint32_t)k_lk_nibble_bits;
108 const uint32_t nib = (value >> shift) & (uint32_t)k_lk_nibble_mask;
109 buf[i] = (uint8_t)((nib < (uint32_t)k_lk_dec_ten) ? ('0' + nib) : ('A' + (nib - k_lk_dec_ten)));
110 }
111 lk_print(buf, (uint32_t)k_lk_hex_nibbles);
112}

115static void lk_print_uint(uint32_t value)
116{
117 uint8_t buf[k_lk_dec_ten];
118 uint32_t n = 0U;
119 if (value == 0U) {
120 buf[n] = '0';
121 n++;
122 }
123 while ((value > 0U) && (n < (uint32_t)k_lk_dec_ten)) {
124 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_lk_dec_ten));
125 n++;
126 value /= (uint32_t)k_lk_dec_ten;
127 }
128 for (uint32_t i = 0U; i < n; i++) {
129 lk_print(&buf[n - 1U - i], 1U);
130 }
131}

144static uint32_t lk_geom_hash(void)
145{
146 const uint8_t* p = (const uint8_t*)s_engine.link_rects;
147 const size_t n = (size_t)s_engine.link_rect_count * sizeof(s_engine.link_rects[0]);
148 uint32_t hsh = (uint32_t)k_lk_fnv_offset;
149 for (size_t i = 0U; i < n; i++) {
150 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_lk_fnv_prime;
151 }
152 return hsh;
153}

163static void lk_probe_rect(uint32_t idx, bool* out_cross, bool* out_frag)
164{
165 const ra8_reflow_link_rect_t* rect = &s_engine.link_rects[idx];
166 const int32_t cx = rect->x + (rect->w / 2);
167 const int32_t cy = rect->y + (rect->h / 2);
168 uint32_t off = 0U;
169 uint32_t len = 0U;
170 if (ra8_reflow_hit_test_link(&s_engine, rect->page_index, cx, cy, &off, &len) != k_ra8_ok) {
171 return;
172 }
173 ra8_reflow_href_kind_t kind = k_ra8_reflow_href_empty;
174 uint32_t pl = 0U;
175 uint32_t fo = 0U;
176 uint32_t fl = 0U;
177 if (ra8_reflow_href_split((const char*)&s_engine.text_pool[off], len, &kind, &pl, &fo, &fl) !=
178 k_ra8_ok) {
179 return;
180 }
181 if ((kind == k_ra8_reflow_href_chapter) || (kind == k_ra8_reflow_href_chapter_fragment)) {
182 *out_cross = true;
183 }
184 if (kind == k_ra8_reflow_href_fragment) {
185 *out_frag = true;
186 }
187}

190static void lk_setup_or_halt(void)
191{
192 uint32_t cpuclk0_hz = 0U;
193 if ((ra8_cgc_init() != k_ra8_ok) || (ra8_mstp_init() != k_ra8_ok)) {
194 lk_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
195 }
196 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
197 lk_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
198 }
199 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
200 lk_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
201 }
202 if (ra8_board_uart_console_init((uint32_t)k_lk_uart_baud) != k_ra8_ok) {
203 lk_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
204 }

219int32_t main(void)
220{
221 lk_setup_or_halt();
222 ra8_isr_globals_enable();
223 lk_print(k_msg_boot, (uint32_t)sizeof(k_msg_boot) - 1U);
224
225 if (ra8_reflow_init((uint16_t)k_lk_view_w,
226 (uint16_t)k_lk_view_h,
227 k_ahem_ttf,
228 (size_t)k_ahem_ttf_len,
229 (uint16_t)k_lk_font_px,
230 (uint32_t)k_lk_ink,
231 (uint32_t)k_lk_link_col,
232 &s_engine) != k_ra8_ok) {
233 lk_panic_halt(k_msg_fail, (uint32_t)sizeof(k_msg_fail) - 1U);
234 }
235 uint32_t pages = 0U;
236 if (ra8_reflow_layout_chapter(&s_engine,
237 (const uint8_t*)k_lk_chapter,
238 (uint32_t)(sizeof(k_lk_chapter) - 1U),
239 &pages) != k_ra8_ok) {
240 lk_panic_halt(k_msg_lerr, (uint32_t)sizeof(k_msg_lerr) - 1U);
241 }
242
243 /* Synthesise a tap at the centre of every link rect; classify each. */
244 bool cross = false;
245 bool frag = false;
246 for (uint32_t i = 0U; i < s_engine.link_rect_count; ++i) {
247 lk_probe_rect(i, &cross, &frag);
248 }
249
250 /* Resolve the same-chapter "#foot" fragment to its page. */
251 uint32_t apage = 0U;
252 if (ra8_reflow_find_anchor(&s_engine, "foot", (uint32_t)(sizeof("foot") - 1U), &apage) !=
253 k_ra8_ok) {
254 apage = (uint32_t)k_lk_view_h; /* sentinel: not found */
255 }
256
257 lk_print(k_msg_pre, (uint32_t)sizeof(k_msg_pre) - 1U);
258 lk_print_uint(s_engine.link_target_count);
259 lk_print(k_msg_cross, (uint32_t)sizeof(k_msg_cross) - 1U);
260 lk_print_bool(cross);
261 lk_print(k_msg_frag, (uint32_t)sizeof(k_msg_frag) - 1U);
262 lk_print_bool(frag);
263 lk_print(k_msg_apage, (uint32_t)sizeof(k_msg_apage) - 1U);
264 lk_print_uint(apage);
265 lk_print(k_msg_geom, (uint32_t)sizeof(k_msg_geom) - 1U);
266 lk_print_hex(lk_geom_hash());
267 lk_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);
268
269 while (1) {
270 __asm__ volatile("wfi");
271 }
272}
