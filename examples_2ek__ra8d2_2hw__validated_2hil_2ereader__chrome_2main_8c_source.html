126static void ch_print_hex(uint32_t value)
127{
128 uint8_t buf[k_ch_hex_nibbles];
129 for (uint32_t i = 0U; i < (uint32_t)k_ch_hex_nibbles; i++) {
130 const uint32_t shift = ((uint32_t)k_ch_hex_nibbles - 1U - i) * (uint32_t)k_ch_nibble_bits;
131 const uint32_t nib = (value >> shift) & (uint32_t)k_ch_nibble_mask;
132 buf[i] = (uint8_t)((nib < (uint32_t)k_ch_dec_ten) ? ('0' + nib) : ('A' + (nib - k_ch_dec_ten)));
133 }
134 ch_print(buf, (uint32_t)k_ch_hex_nibbles);
135}

138static void ch_print_uint(uint32_t value)
139{
140 uint8_t buf[k_ch_dec_ten];
141 uint32_t n = 0U;
142 if (value == 0U) {
143 buf[n] = '0';
144 n++;
145 }
146 while ((value > 0U) && (n < (uint32_t)k_ch_dec_ten)) {
147 buf[n] = (uint8_t)('0' + (value % (uint32_t)k_ch_dec_ten));
148 n++;
149 value /= (uint32_t)k_ch_dec_ten;
150 }
151 for (uint32_t i = 0U; i < n; i++) {
152 ch_print(&buf[n - 1U - i], 1U);
153 }
154}

162static int16_t ch_build_chrome(ra8_box_tree_t* tree)
163{
164 for (uint32_t i = 0U; i < (uint32_t)k_ch_max_nodes; i++) {
165 s_label[i] = nullptr;
166 }
167 (void)ra8_box_tree_init(tree, s_box_nodes, (uint16_t)k_ch_max_nodes);
168 const ra8_box_t root_n = {.kind = (uint8_t)k_ra8_box_stack_v,
169 .pad = (int16_t)k_ch_root_pad,
170 .gap = (int16_t)k_ch_root_pad,
171 .flex = 1U,
172 .fill = (uint32_t)k_ch_col_page};
173 const int16_t root = ra8_box_add(tree, (int16_t)k_ra8_box_none, &root_n);
174
175 const ra8_box_t bar_n = {.kind = (uint8_t)k_ra8_box_leaf,
176 .fixed = (int16_t)k_ch_status_h,
177 .fill = (uint32_t)k_ch_col_bar};
178 const int16_t bar = ra8_box_add(tree, root, &bar_n);
179 s_label[bar] = "Library";
180
181 const ra8_box_t grid_n = {.kind = (uint8_t)k_ra8_box_grid,
182 .grid_cols = 2U,
183 .flex = 1U,
184 .pad = (int16_t)k_ch_grid_pad,
185 .gap = (int16_t)k_ch_grid_gap};
186 const int16_t grid = ra8_box_add(tree, root, &grid_n);
187
188 for (uint32_t i = 0U; i < (uint32_t)k_ch_cell_count; i++) {
189 const ra8_box_t cell_n = {.kind = (uint8_t)k_ra8_box_leaf,
190 .flex = 1U,
191 .pad = (int16_t)k_ch_cell_pad,
192 .fill = (uint32_t)k_ch_col_cell,
193 .border = (uint32_t)k_ch_col_edge,
194 .border_w = (int16_t)k_ch_border_w,
195 .tag = (int16_t)i};
196 const int16_t cell = ra8_box_add(tree, grid, &cell_n);
197 s_label[cell] = k_ch_titles[i];
198 }
199 return root;
200}

222static void ch_render(const ra8_box_tree_t* tree)
223{
224 (void)ra8_gfx_clear((uint32_t)k_ch_col_page);
225 for (uint16_t i = 0U; i < tree->count; i++) {
226 const ra8_box_t* b = &tree->nodes[i];
227 if (b->fill != (uint32_t)k_ra8_box_no_colour) {
228 ch_fill(&b->rect, b->fill);
229 }
230 if ((b->border != (uint32_t)k_ra8_box_no_colour) && (b->border_w > 0)) {
231 ch_border(&b->rect, b->border);
232 }
233 if (s_label[i] != nullptr) {
234 const uint32_t fg =
235 (b->fill == (uint32_t)k_ch_col_bar) ? (uint32_t)k_ch_col_bar_tx : (uint32_t)k_ch_col_ink;
236 const uint32_t bg =
237 (b->fill != (uint32_t)k_ra8_box_no_colour) ? b->fill : (uint32_t)k_ch_col_page;
238 (void)ra8_gfx_text_out(b->rect.x + (int16_t)k_ch_cell_pad,
239 b->rect.y + (int16_t)k_ch_cell_pad,
240 s_label[i],
241 &ra8_gfx_font_8x16,
242 fg,
243 bg);
244 }
245 }

249static uint32_t ch_framebuffer_hash(void)
250{
251 const uint8_t* p = (const uint8_t*)s_framebuffer;
252 const size_t n = sizeof(s_framebuffer);
253 uint32_t hsh = (uint32_t)k_ch_fnv_offset;
254 for (size_t i = 0U; i < n; i++) {
255 hsh = (hsh ^ (uint32_t)p[i]) * (uint32_t)k_ch_fnv_prime;
256 }
257 return hsh;
258}

266static void ch_setup_or_halt(void)
267{
268 uint32_t cpuclk0_hz = 0U;
269 if ((ra8_cgc_init() != k_ra8_ok) || (ra8_mstp_init() != k_ra8_ok)) {
270 ch_panic_halt();
271 }
272 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
273 ch_panic_halt();
274 }
275 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
276 ch_panic_halt();
277 }
278 if (ra8_board_uart_console_init((uint32_t)k_ch_uart_baud) != k_ra8_ok) {
279 ch_panic_halt();
280 }
281}

310 (void)ra8_box_layout(&tree, root, &frame);

317 ch_print(k_msg_eol, (uint32_t)sizeof(k_msg_eol) - 1U);

320 __asm__ volatile("wfi");
321 }
