174static void sdhi_make_csd(void)
175{
176 uint64_t bytes = 0U;
177 board_sd_info(nullptr, &bytes, nullptr, nullptr);
178 uint32_t cap_blocks = (uint32_t)(bytes / (uint64_t)k_sdhi_block_bytes);
179 if (cap_blocks < (uint32_t)k_sdhi_csize_unit) {
180 cap_blocks = (uint32_t)k_sdhi_csize_unit; /* floor: keep c_size non-negative. */
181 }
182 const uint32_t c_size = (cap_blocks / (uint32_t)k_sdhi_csize_unit) - 1U;
183 s_sdhi.rsp[3] = (uint32_t)k_sdhi_csd_v2_struct;
184 s_sdhi.rsp[2] = (c_size >> (uint32_t)k_sdhi_csize_shift) & (uint32_t)k_sdhi_csize_lo_mask;
185 s_sdhi.rsp[1] = (c_size & (uint32_t)k_sdhi_csize_mid_mask) << (uint32_t)k_sdhi_csize_shift;
186 s_sdhi.rsp[0] = 0U;
187}

239static void sdhi_exec_ident(uint32_t idx, uint32_t arg)
240{
241 (void)arg;
242 if ((idx == (uint32_t)k_sdhi_acmd41_op_cond) && (s_sdhi.app_cmd != 0U)) {
243 s_sdhi.rsp[0] = (uint32_t)k_sdhi_ocr_ready;
244 return;
245 }
246 switch (idx) {
247 case (uint32_t)k_sdhi_cmd8_if_cond:
248 s_sdhi.rsp[0] = (uint32_t)k_sdhi_r7_if_cond;
249 break;
250 case (uint32_t)k_sdhi_cmd55_app_cmd:
251 s_sdhi.app_cmd = 1U;
252 /* Echo APP_CMD so the ACMD6 SET_BUS_WIDTH that may follow is
253 * honoured by the driver's 4-bit negotiation check. */
254 s_sdhi.rsp[0] = (uint32_t)k_sdhi_r1_ready | (uint32_t)k_sdhi_r1_app_cmd;
255 break;
256 case (uint32_t)k_sdhi_cmd2_send_cid:
257 s_sdhi.rsp[0] = (uint32_t)k_sdhi_cid_word;
258 s_sdhi.rsp[1] = (uint32_t)k_sdhi_cid_word;
259 s_sdhi.rsp[2] = (uint32_t)k_sdhi_cid_word;
260 s_sdhi.rsp[3] = (uint32_t)k_sdhi_cid_word;
261 break;
262 case (uint32_t)k_sdhi_cmd3_send_rca:
263 s_sdhi.rsp[0] = (uint32_t)k_sdhi_rca_value;
264 break;
265 case (uint32_t)k_sdhi_cmd9_send_csd:
266 sdhi_make_csd();
267 break;
268 default:
269 s_sdhi.rsp[0] = (uint32_t)k_sdhi_r1_ready;
270 break;
271 }
272}

275static void sdhi_exec_command(uint32_t cmd)
276{
277 const uint32_t idx = cmd & (uint32_t)k_sdhi_cmd_index_mask;
278 const uint32_t arg = s_sdhi.regs[sdhi_word(k_sdhi_off_sd_arg)];
279
280 s_sdhi.rsp[0] = (uint32_t)k_sdhi_r1_ready;
281 s_sdhi.rsp[1] = 0U;
282 s_sdhi.rsp[2] = 0U;
283 s_sdhi.rsp[3] = 0U;
284
285 switch (idx) {
286 case (uint32_t)k_sdhi_cmd17_read1:
287 s_sdhi.lba = arg;
288 sdhi_begin_read(false);
289 break;
290 case (uint32_t)k_sdhi_cmd18_readm:
291 s_sdhi.lba = arg;
292 sdhi_begin_read(true);
293 break;
294 case (uint32_t)k_sdhi_cmd24_write1:
295 s_sdhi.lba = arg;
296 sdhi_begin_write(false);
297 break;
298 case (uint32_t)k_sdhi_cmd25_writem:
299 s_sdhi.lba = arg;
300 sdhi_begin_write(true);
301 break;
302 case (uint32_t)k_sdhi_cmd12_stop:
303 s_sdhi.xfer = (uint8_t)k_sdhi_xfer_none;
304 break;
305 default:
306 sdhi_exec_ident(idx, arg);
307 break;
308 }
309 /* CMD55 sets app_cmd; every other command clears the one-shot prefix. */
310 if (idx != (uint32_t)k_sdhi_cmd55_app_cmd) {
311 s_sdhi.app_cmd = 0U;
312 }
313 sdhi_publish_response();
314 s_sdhi.regs[sdhi_word(k_sdhi_off_sd_info1)] |= (uint32_t)k_sdhi_info1_rspend;
315}

318static uint32_t sdhi_buf_read(void)
319{
320 if (s_sdhi.xfer != (uint8_t)k_sdhi_xfer_read) {
321 return 0U;
322 }
323 const uint32_t b = s_sdhi.word_idx * 4U;
324 const uint32_t word = (uint32_t)s_sdhi.stage[b] |
325 ((uint32_t)s_sdhi.stage[b + 1U] << (uint32_t)k_sdhi_byte_bits) |
326 ((uint32_t)s_sdhi.stage[b + 2U] << (2U * (uint32_t)k_sdhi_byte_bits)) |
327 ((uint32_t)s_sdhi.stage[b + 3U] << (3U * (uint32_t)k_sdhi_byte_bits));
328 s_sdhi.word_idx++;
329 if (s_sdhi.word_idx < (uint32_t)k_sdhi_words_per_blk) {
330 return word;
331 }
332 /* Block drained: advance a multi-block read, else end the phase. */
333 s_sdhi.reads++;
334 if (s_sdhi.blocks_left > 0U) {
335 s_sdhi.blocks_left--;
336 }
337 if (s_sdhi.blocks_left > 0U) {
338 s_sdhi.lba++;
339 sdhi_load_block();
340 } else {
341 s_sdhi.xfer = (uint8_t)k_sdhi_xfer_none;
342 s_sdhi.regs[sdhi_word(k_sdhi_off_sd_info2)] &= ~(uint32_t)k_sdhi_info2_bre;
343 }
344 return word;
345}

348static void sdhi_buf_write(uint32_t word)
349{
350 if (s_sdhi.xfer != (uint8_t)k_sdhi_xfer_write) {
351 return;
352 }
353 const uint32_t b = s_sdhi.word_idx * 4U;
354 s_sdhi.stage[b] = (uint8_t)(word & (uint32_t)k_sdhi_byte_mask);
355 s_sdhi.stage[b + 1U] =
356 (uint8_t)((word >> (uint32_t)k_sdhi_byte_bits) & (uint32_t)k_sdhi_byte_mask);
357 s_sdhi.stage[b + 2U] =
358 (uint8_t)((word >> (2U * (uint32_t)k_sdhi_byte_bits)) & (uint32_t)k_sdhi_byte_mask);
359 s_sdhi.stage[b + 3U] =
360 (uint8_t)((word >> (3U * (uint32_t)k_sdhi_byte_bits)) & (uint32_t)k_sdhi_byte_mask);
361 s_sdhi.word_idx++;
362 if (s_sdhi.word_idx < (uint32_t)k_sdhi_words_per_blk) {
363 return;
364 }
365 /* Block filled: commit it, advance a multi-block write, else end the phase. */
366 (void)board_sd_write_block(s_sdhi.lba, s_sdhi.stage);
367 s_sdhi.writes++;
368 s_sdhi.word_idx = 0U;
369 if (s_sdhi.blocks_left > 0U) {
370 s_sdhi.blocks_left--;
371 }
372 if (s_sdhi.blocks_left > 0U) {
373 s_sdhi.lba++;
374 } else {
375 s_sdhi.xfer = (uint8_t)k_sdhi_xfer_none;
376 s_sdhi.regs[sdhi_word(k_sdhi_off_sd_info2)] &= ~(uint32_t)k_sdhi_info2_bwe;
377 }
378}

381static uint64_t sdhi_read(uc_engine* uc, uint64_t addr, unsigned size)
382{
383 (void)uc;
384 (void)size;
385 const uint64_t off = addr - (uint64_t)k_sdhi_base;
386 if (off == (uint64_t)k_sdhi_off_sd_buf0) {
387 return (uint64_t)sdhi_buf_read();
388 }
389 if (sdhi_word(off) >= (uint32_t)k_sdhi_words) {
390 return 0U;
391 }
392 return (uint64_t)s_sdhi.regs[sdhi_word(off)];

396static void sdhi_write(uc_engine* uc, uint64_t addr, unsigned size, uint64_t value)
397{
398 (void)uc;
399 (void)size;
400 const uint64_t off = addr - (uint64_t)k_sdhi_base;
401 if (off == (uint64_t)k_sdhi_off_sd_buf0) {
402 sdhi_buf_write((uint32_t)value);
403 return;
404 }
405 if (off == (uint64_t)k_sdhi_off_soft_rst) {
406 /* The driver writes 0 then 1; clear the FIFO/command engine either way. */
407 s_sdhi.xfer = (uint8_t)k_sdhi_xfer_none;
408 s_sdhi.word_idx = 0U;
409 s_sdhi.blocks_left = 0U;
410 s_sdhi.app_cmd = 0U;
411 s_sdhi.regs[sdhi_word(off)] = (uint32_t)value;
412 s_sdhi.regs[sdhi_word(k_sdhi_off_sd_info1)] = 0U;
413 s_sdhi.regs[sdhi_word(k_sdhi_off_sd_info2)] = 0U;
414 return;
415 }
416 if (sdhi_word(off) < (uint32_t)k_sdhi_words) {
417 s_sdhi.regs[sdhi_word(off)] = (uint32_t)value;
418 }
419 if (off == (uint64_t)k_sdhi_off_sd_cmd) {
420 sdhi_exec_command((uint32_t)value);
421 }
422}
