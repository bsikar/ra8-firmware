63/*
64 * The RSIP-E50D asymmetric signature / key-agreement family (ECDSA sign /
65 * verify, ECDH, Ed25519 PureEdDSA) is NOT backed by a documented register
66 * interface on this silicon. HUM Ch 52 "Renesas Secure IP (RSIP-E50D)" is a
67 * six-page feature overview (p 3302-3307) with no command-register map; the
68 * vendor engine is driven through an encrypted firmware mailbox, not the MMIO
69 * opcodes modelled below. The command-path bodies here only round-trip the
70 * host register fake; they do NOT compute a real ECDSA / ECDH / RFC 8032
71 * result. They compile only under the insecure-stub / off-target guard so a
72 * production image gets the fail-closed #else and can never mistake these
73 * bytes for a valid signature or shared secret. The real ECDSA-P256 / ECDH /
74 * Ed25519 backend is tf-psa-crypto on the M85, silicon-proven in
75 * psa_crypto_hil (issues #214 + #181). The register pokes below therefore
76 * carry NO HUM citation: there is no real register map to cite.
77 */

118{

127 case k_ra8_rsip_curve_secp256k1:

141ra8_err_t ra8_rsip_ecdsa_sign(const ra8_rsip_key_handle_t* key,
142 ra8_rsip_curve_t curve,
143 const uint8_t* digest,
144 uint32_t digest_len,
145 uint8_t* signature)
146{
147 RA8_CHECK_NULL_PTR(key, s_tag, "key must not be nullptr");
148 RA8_CHECK_NULL_PTR(digest, s_tag, "digest must not be nullptr");
149 RA8_CHECK_NULL_PTR(signature, s_tag, "signature must not be nullptr");
150 if (curve == k_ra8_rsip_curve_ed25519) {
151 /* Ed25519 is PureEdDSA (RFC 8032), not ECDSA -- routing it through
152 * the ECDSA opcode would not produce a valid signature. Callers
153 * must use ra8_rsip_eddsa_sign(). */
154 return k_ra8_err_invalid_arg;
155 }
156 const uint32_t curve_bytes = internal_curve_bytes(curve);
157 if (curve_bytes == 0U) {
158 return k_ra8_err_invalid_arg;
159 }
160 internal_load_handle(key);
161 *ra8_rsip_reg32(k_ra8_rsip_off_asym_curve) = (uint32_t)curve;
162 internal_asym_push(k_ra8_rsip_off_asym_msg_in, digest, digest_len);
163 *ra8_rsip_reg32(k_ra8_rsip_off_asym_ctrl) = k_ra8_rsip_asym_op_ecdsa_sign;
164 *ra8_rsip_reg32(k_ra8_rsip_off_mbox_op) = k_ra8_rsip_asym_op_ecdsa_sign;
165
166 const ra8_err_t err = internal_complete(k_ra8_rsip_mask_isr_asym_done);
167 if (err != k_ra8_ok) {
168 return err;
169 }
170 /* (r || s) */
171 internal_asym_pull(k_ra8_rsip_off_asym_sig_out, signature, curve_bytes * 2U);
172 return k_ra8_ok;
173}

213 const uint8_t* msg,
214 uint32_t msg_len,
215 uint8_t* signature)
216{
217 RA8_CHECK_NULL_PTR(key, s_tag, "key must not be nullptr");
218 RA8_CHECK_NULL_PTR(signature, s_tag, "signature must not be nullptr");
219 if ((msg == nullptr) && (msg_len != 0U)) {
220 return k_ra8_err_null_ptr;
221 }
222 if (key->alg != (uint32_t)k_ra8_rsip_oem_cmd_ecc_ed25519_priv) {
223 return k_ra8_err_invalid_arg;
224 }
225 internal_load_handle(key);
226 *ra8_rsip_reg32(k_ra8_rsip_off_asym_curve) = (uint32_t)k_ra8_rsip_curve_ed25519;
227 /* PureEdDSA signs the message itself, not a pre-computed digest (RFC 8032). */
228 if (msg_len > 0U) {
229 internal_asym_push(k_ra8_rsip_off_asym_msg_in, msg, msg_len);
230 }
231 *ra8_rsip_reg32(k_ra8_rsip_off_asym_ctrl) = k_ra8_rsip_asym_op_eddsa_sign;
232 *ra8_rsip_reg32(k_ra8_rsip_off_mbox_op) = k_ra8_rsip_asym_op_eddsa_sign;
233
234 const ra8_err_t err = internal_complete(k_ra8_rsip_mask_isr_asym_done);
235 if (err != k_ra8_ok) {
236 return err;
237 }
238 /* (R || S) */
239 internal_asym_pull(k_ra8_rsip_off_asym_sig_out,
240 signature,
241 (uint32_t)k_ra8_rsip_ed25519_sig_bytes);
242 return k_ra8_ok;

291static void internal_ecdh_pull_shared(ra8_rsip_key_handle_t* out)
292{
293 /* The wrapped shared secret is delivered as an HMAC-SHA-256 handle. */
294 out->alg = k_ra8_rsip_oem_cmd_hmac_sha256;
295 out->body_words = (uint32_t)k_ra8_rsip_handle_words_hmac_sha256;
296 for (uint32_t w = 0U; w < out->body_words; ++w) {
297 out->body[w] = *ra8_rsip_reg32(k_ra8_rsip_off_asym_shared);
298 }
299 internal_zero_handle_tail(out, out->body_words);
300}
