203RA8_EXPECTS_LOCK("ra8_fs_lock")
204static ra8_err_t priv_stat_fat(const ra8_fs_mount_t* handle, const char* path, ra8_fs_stat_t* out)
205{
206 dir_loc_t parent = {};
207 const char* leaf = nullptr;
208 const ra8_err_t rerr = priv_resolve_parent(handle, path, &parent, &leaf);
209 if (rerr != k_ra8_ok) {
210 return rerr;
211 }
212 uint8_t name83[k_max_8_3_name] = {};
213 const uint8_t have83 = priv_path_to_83(leaf, name83);
214 uint64_t lba = 0U;
215 uint32_t off = 0U;
216 uint8_t entry[k_ra8_fs_dir_entry_bytes] = {};
217 ra8_err_t err = k_ra8_err_not_found;
218 if (have83 != 0U) {
219 err = priv_dir_find(handle, &parent, name83, &lba, &off, entry);
220 }
221 if (err == k_ra8_err_not_found) {
222 /* Same long-name fallback as priv_open_locked: a leaf that is not
223 * 8.3-representable still resolves through its VFAT chain. */
224 err = priv_dir_find_long(handle, &parent, leaf, &lba, &off, entry);
225 }
226 if (err != k_ra8_ok) {
227 return err;
228 }
229 priv_entry_to_stat(entry, out);
230 return k_ra8_ok;
231}

262RA8_INTERNAL
263RA8_EXPECTS_LOCK("ra8_fs_lock")
264static ra8_err_t priv_stat_locked(ra8_fs_mount_t* handle, const char* path, ra8_fs_stat_t* out)
265{
266 if (handle == nullptr) {
267 return k_ra8_err_null_ptr;
268 }
269 if (path == nullptr) {
270 return k_ra8_err_null_ptr;
271 }
272 if (out == nullptr) {
273 return k_ra8_err_null_ptr;
274 }
275 if (handle->in_use == 0U) {
276 return k_ra8_err_invalid_state;
277 }
278 *out = (ra8_fs_stat_t){};
279 if (priv_path_is_root(path)) {
280 priv_stat_root(handle, out);
281 return k_ra8_ok;
282 }
283 if (handle->type == k_ra8_fs_type_exfat) {
284 return priv_stat_exfat(handle, path, out);
285 }
286 return priv_stat_fat(handle, path, out);
287}
