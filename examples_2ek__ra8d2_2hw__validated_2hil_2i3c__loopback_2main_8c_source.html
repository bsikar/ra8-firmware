75/* Controller bring-up banners. Reaching this loop at all proves
76 * ra8_i3c_init configured + powered the I3C controller on real silicon
77 * (init failure panic-halts before the loop), so every banner leads with
78 * "i3c: ctrl init ok". The scan outcome is reported but not gated: on a
79 * bare EVM the ch0/J27 bus floats (no device, no pull-ups) so the
80 * transaction cannot complete -- attach a device to J27 to see ack=1. */

210int32_t main(void)
211{
212 i3c_demo_setup_or_halt();
213 ra8_isr_globals_enable();
214
215 while (1) {
216 bool acked = false;
217 const ra8_err_t err =
218 ra8_i3c_scan((uint8_t)k_i3c_demo_iic_channel, (uint8_t)k_i3c_demo_probe_addr, &acked);
219 const uint8_t* msg = k_i3c_demo_msg_err;
220 uint32_t msg_len = (uint32_t)(sizeof(k_i3c_demo_msg_err) - 1U);
221 if (err == k_ra8_ok) {
222 if (acked) {
223 msg = k_i3c_demo_msg_ack;
224 msg_len = (uint32_t)(sizeof(k_i3c_demo_msg_ack) - 1U);
225 } else {
226 msg = k_i3c_demo_msg_nack;
227 msg_len = (uint32_t)(sizeof(k_i3c_demo_msg_nack) - 1U);
228 }
229 }
230 if (ra8_board_uart_console_write(msg, (size_t)msg_len) != k_ra8_ok) {
231 break;
232 }
233 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
234 break;
235 }
236 ra8_delay_ms(k_i3c_demo_period_ms);
237 }
238
239 i3c_demo_panic_halt();
240 return 0;
241}
