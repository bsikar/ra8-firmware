4"""Gate: no first-party shell function is called with its errexit masked.
5
6The defect class
7----------------
8Under ``set -e``, calling a function on the left of ``||`` puts it into bash's
9inherited "ignoring errors" state::
10
11 run_suite "$fast" || rc=$?
12
13Two things then go wrong. A failure part-way through the function body no
14longer aborts it, so every remaining statement still runs and the caller sees
15only the *last* command's status. Worse, that state propagates into nested
16subshells where a plain ``set -e`` cannot clear it -- ``$-`` reports ``e`` set
17while a failing command still does not abort. This tree shipped exactly that:
18the gate suite silently degraded to "did each gate's LAST command succeed", so
19a gate failing part-way -- including ``require_cmd`` reporting an absent tool
20-- reported PASS.
21
22The remedy, already documented at the ``run_suite`` call site in
23``scripts/ci.sh``, is to disable errexit around the CALL only::
24
25 set +e
26 run_suite "$fast"
27 rc=$?
28 set -e
29
30The callee then runs in a normal errexit context, so its own ``( set -e; ... )``
31subshells re-arm and a mid-body failure is not swallowed.
32
33Why this check and not ShellCheck's SC2310
34------------------------------------------
35``check-set-e-suppressed`` covers this defect but is unsatisfiable here: it
36fires on any function in a condition, including a bare one-line predicate and
37the rewrite its own help text recommends. Measured on this tree it produces 90
38findings, and the only two source forms it accepts are worse than what it
39rejects -- ``set +e; fn; rc=$?; set -e`` (which it passes) and a bare subshell
40(which aborts the parent). Adopting it would mean ~90 inline disables rather
41than ~90 fixes. See #363 for the full form-by-form evidence.
42
43This check keeps the signal and drops the noise: it fires only where a
44first-party function whose body has more than one command is invoked with its
45status masked. A one-command predicate is exempt by construction -- there is
46no statement after the failure for errexit to have protected.
47
48Run::
49
50 check_errexit_masking.py # gate (fail on any finding)
51 check_errexit_masking.py --selftest # prove it fires AND stays quiet
52
53Exit 0 if clean, exit 1 on findings, exit 2 on an internal error.
54"""

65# A function body with a single command cannot hide a mid-body failure: the
66# status the caller captures IS that command's status. Two or more is where
67# the defect becomes possible, so that is where the check starts firing.
68MIN_BODY_COMMANDS = 2

70# The bad fixture masks exactly three calls to the multi-command `work`.
71# Pinned as a constant so the selftest asserts a number rather than a
72# literal that could be edited to match whatever the checker happens to do.
73SELFTEST_EXPECTED_HITS = 3

75_DEF_RE = re.compile(r"^\s*(?:function\s+)?([A-Za-z_][A-Za-z0-9_]*)\s*\‍(\‍)\s*[({]")
76
77# `fn ... || true`, `|| :`, `|| rc=$?` -- the forms that swallow the status.
78# The callee must sit at the start of a command position: line start, or after
79# a separator, a `case` pattern's `)`, or a `then`/`do`/`else` keyword.
80_MASK_RE = re.compile(
81 r"(?:^|[;&|()]|\bthen\b|\bdo\b|\belse\b)\s*"
82 r"([A-Za-z_][A-Za-z0-9_]*)"
83 r"(?:\s+[^|;&]*?)?"
84 r"\|\|\s*(?:true\b|:\s|:$|[A-Za-z_][A-Za-z0-9_]*=\$\?)"
85)

88def _strip(line: str) -> str:
89 """Drop comments and neutralise quoted spans, preserving column count."""
90 out: list[str] = []
91 quote: str | None = None
92 for index, char in enumerate(line):
93 if quote is not None:
94 out.append(char)
95 if char == quote:
96 quote = None
97 elif char in "\"'":
98 quote = char
99 out.append(char)
100 elif char == "#" and (index == 0 or line[index - 1].isspace()):
101 break
102 else:
103 out.append(char)
104 return "".join(out)
105
106
107# Control keywords. A compound block counts as its WORST branch, never the sum,
108# and a condition counts zero: a non-zero status in a condition is control flow,
109# not a failure errexit was ever going to catch.
110_BLOCK_OPEN = frozenset({"if", "while", "until", "for", "case"})
111_BODY_START = frozenset({"then", "do", "in"})
112_BLOCK_CLOSE = frozenset({"fi", "done", "esac"})
113_KEYWORD_RE = re.compile(r"^(if|elif|while|until|for|case|then|do|else|fi|done|esac|\{|\})\b\s*")

116def _tokens(body: str) -> list[str]:
117 r"""Split a function body into statements at separator positions.
118
119 Newlines and `;` only separate at paren depth zero: a multi-line array
120 literal (`local -a args=(\n --hex ...\n)`) is ONE assignment, not one
121 command per element.
122 """
123 joined = re.sub(r"\\\n", " ", body)
124 depth = 0
125 flat: list[str] = []
126 for char in joined:
127 if char == "(":
128 depth += 1
129 elif char == ")":
130 depth = max(0, depth - 1)
131 if char in "\n;" and depth > 0:
132 flat.append(" ")
133 else:
134 flat.append(char)
135 out: list[str] = []
136 for raw in re.split(r"[\n;]", "".join(flat)):
137 piece = raw.strip()
138 while piece:
139 match = _KEYWORD_RE.match(piece)
140 if match is None:
141 break
142 out.append(match.group(1))
143 piece = piece[match.end() :].strip()
144 if piece:
145 out.append(piece)
146 return out

149# An assignment with no command substitution in it cannot fail, so it is not a
150# step errexit was ever protecting. Counting `local app="$1"` as a command is
151# what made an early revision of this check fire on functions whose only
152# failable statement was the last one -- the same false-positive problem that
153# makes SC2310 unusable.
154_ASSIGN_RE = re.compile(
155 r"^(?:(?:local|declare|readonly|export|typeset)\s+(?:-\w+\s+)*)?"
156 r"[A-Za-z_][A-Za-z0-9_]*(?:\‍[[^\‍]]*\‍])?\+?=|"
157 r"^(?:local|declare|readonly|export|typeset)\s+-?\w*\s*[A-Za-z_]"
158)
159_SUBST_RE = re.compile(r"\$\‍(|`")
160

162def _is_failable(token: str) -> bool:
163 """False for assignments that cannot fail -- no substitution, no exit status."""
164 if _SUBST_RE.search(token):
165 return True
166 return _ASSIGN_RE.match(token) is None

169def _max_sequential(body: str) -> int:
170 """Max commands executed sequentially on any ONE path through `body`.
171
172 This is the number that decides whether errexit had anything to protect:
173 with two or more commands in a row, a failure in the first still lets the
174 rest run and the caller sees only the last one's status.
175 """
176 stack: list[list[int]] = []
177 run = 0
178 skipping = False
179 for word in _tokens(body):
180 if word in _BLOCK_OPEN:
181 stack.append([0, run])
182 run = 0
183 skipping = True
184 elif word in _BODY_START:
185 skipping = False
186 elif word in ("elif", "else"):
187 if stack:
188 stack[-1][0] = max(stack[-1][0], run)
189 run = 0
190 skipping = word == "elif"
191 elif word in _BLOCK_CLOSE:
192 if stack:
193 best, saved = stack.pop()
194 run = saved + max(best, run)
195 skipping = False
196 elif word not in ("{", "}") and not skipping and _is_failable(word):
197 run += 1
198 while stack:
199 best, saved = stack.pop()
200 run = saved + max(best, run)
201 return run

204def _function_sizes(lines: list[str]) -> dict[str, int]:
205 """Map every function defined in `lines` to its max sequential command count."""
206 sizes: dict[str, int] = {}
207 index = 0
208 while index < len(lines):
209 match = _DEF_RE.match(lines[index])
210 if match is None:
211 index += 1
212 continue
213 end, body = _body_span(lines, index)
214 sizes[match.group(1)] = _max_sequential(body)
215 index = end + 1
216 return sizes

219def _body_span(lines: list[str], start: int) -> tuple[int, str]:
220 """Return (last line index, body text) for the function opening at `start`."""
221 depth = 0
222 opened = False
223 collected: list[str] = []
224 index = start
225 while index < len(lines):
226 stripped = _strip(lines[index])
227 for char in stripped:
228 if char in "{(":
229 depth += 1
230 opened = True
231 elif char in "})":
232 depth -= 1
233 collected.append(stripped.split("{", 1)[-1] if index == start else stripped)
234 if opened and depth <= 0:
235 return index, "\n".join(collected)
236 index += 1
237 return index - 1, "\n".join(collected)

240def _scan_file(rel: str) -> list[str]:
241 """Return one message per errexit-masked first-party call in `rel`."""
242 text = (REPO_ROOT / rel).read_text(encoding="utf-8", errors="replace")
243 lines = text.splitlines()
244 sizes = _function_sizes(lines)
245 findings: list[str] = []
246 for number, line in enumerate(lines, 1):
247 for match in _MASK_RE.finditer(_strip(line)):
248 name = match.group(1)
249 size = sizes.get(name)
250 if size is None or size < MIN_BODY_COMMANDS:
251 continue
252 findings.append(
253 f"{rel}:{number}: `{name}` (~{size} commands) is invoked with its "
254 f"exit status masked; a failure part-way through its body is "
255 f"silently swallowed.\n"
256 f" {line.strip()}\n"
257 f" Use: set +e; {name} ...; rc=$?; set -e"
258 )
259 return findings

308def _selftest() -> int:
309 """Assert the check fires on the defect AND stays silent on the remedy."""
310 import tempfile # noqa: PLC0415 -- selftest-only dependency
311
312 failures: list[str] = []
313 with tempfile.TemporaryDirectory(dir=REPO_ROOT) as tmp:
314 holder = Path(tmp)
315 bad = holder / "bad.sh"
316 good = holder / "good.sh"
317 bad.write_text(_SELFTEST_BAD, encoding="utf-8")
318 good.write_text(_SELFTEST_GOOD, encoding="utf-8")
319 bad_hits = _scan_file(str(bad.relative_to(REPO_ROOT)))
320 good_hits = _scan_file(str(good.relative_to(REPO_ROOT)))
321
322 # MUST-FIRE: the three masked calls on the multi-command `work`.
323 if len(bad_hits) != SELFTEST_EXPECTED_HITS:
324 failures.append(
325 f"must-fire: expected {SELFTEST_EXPECTED_HITS} findings on the bad "
326 f"fixture, got {len(bad_hits)}"
327 )
328 if any("predicate" in hit for hit in bad_hits):
329 failures.append("must-fire: flagged the one-command predicate, which is exempt")
330 if any("grep" in hit for hit in bad_hits):
331 failures.append("must-fire: flagged an external command, which is out of scope")
332
333 # MUST-BE-SILENT: the documented `set +e` remedy and the exempt forms.
334 if good_hits:
335 failures.append(f"must-be-silent: {len(good_hits)} finding(s) on the good fixture")
336
337 for line in failures:
338 sys.stderr.write(f"check_errexit_masking.py: SELFTEST FAIL -- {line}\n")
339 if failures:
340 return 1
341 sys.stdout.write(
342 f"check_errexit_masking.py: selftest OK "
343 f"(fires on {len(bad_hits)} masked calls, silent on the remedy)\n"
344 )
345 return 0
