1#!/usr/bin/env python3
2"""check_inclusive_terminology.py -- inclusive-terminology gate for ra8-firmware.
3
4Bans the legacy master/slave/MOSI/MISO/SS vocabulary from FIRST-PARTY source
5under libs/, src/, examples/, tests/, port/, scripts/, docs/, and the top-
6level CMake / Makefile / workflow files. CLAUDE.md "Terminology Standard"
7mandates Controller/Peripheral, COPI/CIPO, CS/Chip Select, Primary/Main.
8
9Per-line opt-out: append a `LEGACY-OK: <reason>` annotation on the offending
10line. Reserved for unavoidable upstream-symbol citations (e.g. the literal
11spelling of a Renesas HUM register-bit name where the symbol must appear
12verbatim in the source comment).
13
14Exit code:
15 0 -- no violations (gate clean), or warn-only mode is on
16 1 -- violations exist (only when WARN_ONLY_MODE is False)
17
18The script is intentionally fast (pure-Python regex scan, no libclang) so
19the pre-commit hook stays interactive.
20

40# verbatim (a Renesas HUM section title, a literal FSP / USBX API symbol),

101# Prose patterns: word-boundary matches that catch the legacy vocabulary
102# wherever it appears as a standalone word -- prose comments ("master
103# enable"), HUM section titles, "Slave Select", the bare "SS" pin name.
104PATTERNS: tuple[tuple[str, re.Pattern[str]], ...] = (
105 ("master", re.compile(r"\bmaster(s|ed|ing|ship)?\b", re.IGNORECASE)),
106 ("slave", re.compile(r"\bslave(s|d)?\b", re.IGNORECASE)),
107 ("mosi", re.compile(r"\bMOSI\b")),
108 ("miso", re.compile(r"\bMISO\b")),
109 ("slave_select", re.compile(r"\bSlave[ _-]Select\b", re.IGNORECASE)),
110 ("ss_pin", re.compile(r"\bSS\b")), # SPI Slave Select pin abbreviation -- use CS
111)
112
113# Identifier-embedded detection.
114#
115# Python's `\b` does not fire between an underscore and a letter (`_` is a
116# word character), so the prose patterns above miss the legacy words when
117# they are welded into a snake_case / SCREAMING_CASE symbol
118# (`internal_spcr_master`, `make_master_i2s_cfg`,
119# ...). Instead of enumerating our own prefixes, flag ANY identifier that
120# carries `master`/`slave` as a leading component and is NOT part of a
121# vendored upstream namespace -- those APIs are referenced verbatim in our
122# glue code and cannot be renamed.
123IDENT_RE: re.Pattern[str] = re.compile(r"[A-Za-z_][A-Za-z0-9_]*")
124# `master`/`slave` as a component: at the identifier start or just after an
125# underscore (so `MASTEREN`, `ra8_spi_master_init`, `_slave` all match, while
126# camelCase `offsetFromMaster` and substrings like `enslave` do not).
127IDENT_TERM_RE: re.Pattern[str] = re.compile(r"(?:^|_)(?:master|slave)", re.IGNORECASE)
128# Upstream namespaces whose symbols legitimately appear in first-party glue
129# and are cited verbatim: Express Logic USBX / ThreadX / NetX / FileX /
130# LevelX, Renesas FSP (r_iic_* / r_sce_*), Mbed TLS, NimBLE / Mynewt, and
131# Espressif esp-hosted. Matched at the identifier head.
132VENDOR_IDENT_RE: re.Pattern[str] = re.compile(
133 r"^(?:_?ux_|r_iic|r_sce|_?nx_|ble_|mynewt|mbedtls|tls_|pre_?master"
134 r"|premaster|resumption_master|_?lx_|_?tx_|_?gx_|_?fx_|netx|threadx"
135 r"|usbx|filex|levelx|esp_hosted|dcd_sim_slave|hcd_sim_host)",
136 re.IGNORECASE,
137)
138# Hardware register-bit names that spell a legacy token verbatim (Renesas
139# MIPI D-PHY DPHYMDC.MASTEREN); these are silicon names, not our symbols.
140HW_TOKENS: frozenset[str] = frozenset({"MASTEREN"})
141
142# Per-line opt-out marker.
143LEGACY_OK_RE: re.Pattern[str] = re.compile(r"LEGACY-OK\s*:")

146def identifier_violation(line: str) -> str | None:
147 """The offending identifier when a line names a symbol with legacy terminology.
148
149 Checks IDENTIFIERS, not prose: a comment discussing the legacy term --
150 often required when mapping a vendor document onto our names -- is fine,
151 while a symbol carrying it is not, because the symbol propagates.
152
153 Returns None when the line is clean.
154
155 Vendor-namespace identifiers and hardware register-bit names are
156 skipped: they are upstream contracts spelled verbatim, not symbols
157 this project is free to rename.
158 """
159 for m in IDENT_RE.finditer(line):
160 tok = m.group(0)
161 if not IDENT_TERM_RE.search(tok):
162 continue
163 if tok in HW_TOKENS:
164 continue
165 if VENDOR_IDENT_RE.match(tok):
166 continue
167 return tok
168 return None

201def should_scan(path: Path) -> bool:
202 """Whether a file is in scope, by directory, basename and suffix."""
203 parts = set(path.parts)
204 if parts & SKIP_DIR_NAMES:
205 return False
206 if path.name in SCAN_BASENAMES:
207 return True
208 return path.suffix in SCAN_EXTS

211def _is_skip_dir(name: str) -> bool:
212 """Whether a directory is build output and therefore skipped.
213
214 Matches the ``build-*`` family by prefix so CMake variant directories are
215 excluded without being enumerated.
216 """
217 if name in SKIP_DIR_NAMES:
218 return True
219 # Glob-style: any CMake/build artefact dir like build-fuzz, build-bench.
220 return bool(name.startswith("build-") or name == "build")

223def iter_source_files(root: Path) -> list[Path]:
224 """Every in-scope file beneath the configured scan roots."""
225 out: list[Path] = []
226 for scan_root in SCAN_ROOTS:
227 base = root / scan_root
228 if not base.exists():
229 continue
230 for dirpath, dirnames, filenames in os.walk(base):
231 dirnames[:] = [d for d in dirnames if not _is_skip_dir(d)]
232 for fn in filenames:
233 p = Path(dirpath) / fn
234 if should_scan(p):
235 out.append(p)
236 for top in ("Makefile", "CMakeLists.txt", "README.md"):
237 p = root / top
238 if p.exists():
239 out.append(p)
240 return out

243def scan_file(path: Path, root: Path) -> list[tuple[Path, int, str, str]]:
244 """Report every legacy-terminology identifier in one file.
245
246 Self-exempt files -- this checker, the terminology policy -- are skipped
247 whole, since they must name the banned terms to define them.
248 """
249 rel = path.relative_to(root)
250 rel_str = str(rel)
251 if rel_str in SELF_EXEMPT_FILES:
252 return []
253 try:
254 text = path.read_text(encoding="utf-8")
255 except (OSError, UnicodeDecodeError):
256 return []
257 out: list[tuple[Path, int, str, str]] = []
258 for lineno, line in enumerate(text.splitlines(), start=1):
259 if LEGACY_OK_RE.search(line):
260 continue
261 matched = False
262 for term, regex in PATTERNS:
263 if not regex.search(line):
264 continue
265 out.append((rel, lineno, term, line.rstrip()))
266 matched = True
267 break
268 if matched:
269 continue
270 ident = identifier_violation(line)
271 if ident is not None:
272 out.append((rel, lineno, f"symbol:{ident}", line.rstrip()))
273 return out

276def main() -> int:
277 """Enforce OSHWA-inclusive terminology on first-party identifiers.
278
279 Scoped to identifiers rather than all text on purpose: vendor manuals and
280 external APIs still use the legacy terms, and comments mapping our names
281 onto theirs are required elsewhere in this tree. The rule governs what
282 this codebase NAMES, not what it may mention.
283
284 Returns 1 listing each offending symbol, 0 when the tree is clean.
285 """
286 root = Path(__file__).resolve().parents[2]
287 files = iter_source_files(root)
288 findings: list[tuple[Path, int, str, str]] = []
289 for f in files:
290 findings.extend(scan_file(f, root))
291
292 if not findings:
293 print("inclusive-terminology: 0 violations -- gate clean.")
294 return 0
295
296 print(f"inclusive-terminology: {len(findings)} violations found.")
297 for rel, lineno, term, line in findings[:MAX_FINDINGS_SHOWN]:
298 snippet = line if len(line) <= MAX_SNIPPET_LEN else line[:SNIPPET_TRUNCATE_LEN] + "..."
299 print(f" {rel}:{lineno} [{term}] {snippet}")
300 if len(findings) > MAX_FINDINGS_SHOWN:
301 print(f" ... {len(findings) - MAX_FINDINGS_SHOWN} more (truncated)")
302 print()
303 print("Per-line opt-out: append `LEGACY-OK: <reason>` on the offending line.")
304 print("See CLAUDE.md 'Terminology Standard' for the policy.")
305
306 if WARN_ONLY_MODE:
307 print("WARN_ONLY_MODE=True -- not failing the gate.")
308 return 0
309 return 1
