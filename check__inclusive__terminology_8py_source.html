4"""check_inclusive_terminology.py -- inclusive-terminology gate for ra8-firmware.
5
6Bans the legacy master/slave/MOSI/MISO/SS vocabulary from FIRST-PARTY source
7under libs/, src/, examples/, tests/, port/, scripts/, docs/, and the top-
8level CMake / Makefile / workflow files. CLAUDE.md "Terminology Standard"
9mandates Controller/Peripheral, COPI/CIPO, CS/Chip Select, Primary/Main.
10
11Per-line opt-out: append a `LEGACY-OK: <reason>` annotation on the offending
12line. Reserved for unavoidable upstream-symbol citations (e.g. the literal
13spelling of a Renesas HUM register-bit name where the symbol must appear
14verbatim in the source comment).
15
16Exit code:
17 0 -- no violations (gate clean), or warn-only mode is on
18 1 -- violations exist (only when WARN_ONLY_MODE is False)
19
20The script is intentionally fast (pure-Python regex scan, no libclang) so
21the pre-commit hook stays interactive.
22
23@copyright Copyright (c) 2026 Brighton Sikarskie
24SPDX-License-Identifier: MIT

42# + sweep brought first-party source to 0 violations and the gate is
43# now strict (False). If a future agent reintroduces a violation the gate
44# fails the commit; either rewrite the wording (preferred for our own
45# prose and symbols) or, only where an upstream string must appear
46# verbatim (a Renesas HUM section title, a literal FSP / USBX API symbol),
47# annotate that single line with `LEGACY-OK: <reason>`. There is no

49WARN_ONLY_MODE: bool = False
50
51# Directories whose CONTENT is vendored or generated and not ours to police:
52# docs/reference/ holds committed vendor PDFs and register maps (HUM register
53# names literally spell MASTEREN and the like), and docs/**/doxygen,
54# docs/**/html are generated Doxygen output. first_party_paths already drops
55# third_party/ and build output; these three are the docs-side equivalents the
56# old SKIP_DIR_NAMES carried. Matched by path component, as the walk did.
57DOCS_VENDOR_DIRS: frozenset[str] = frozenset({"reference", "doxygen", "html"})
58
59# File extensions we scan. Anything else is binary / not-our-source.
60SCAN_EXTS: frozenset[str] = frozenset(
61 {
62 ".c",
63 ".h",
64 ".cpp",
65 ".hpp",
66 ".cc",
67 ".cmake",
68 ".mk",
69 ".md",
70 ".yml",
71 ".yaml",
72 ".sh",
73 ".py",
74 ".txt",
75 ".csv", # docs/MCDC_GAPS.csv and other generated/authored tables

79# Filenames (no extension) we still want to scan.
80SCAN_BASENAMES: frozenset[str] = frozenset(
81 {"Makefile", "GNUmakefile", "Dockerfile", "CMakeLists.txt"}
82)

84# A tree this size cannot legitimately collapse to a handful of files. A scan
85# that enumerates almost nothing reports a clean tree because it read almost
86# nothing -- the exact failure the gate-honesty epic (#190) exists to prevent.
87# Measured 2026-08-02: 3415 first-party files in the derived scope. Same
88# trip-wire as check_ruff.py.
89FILE_FLOOR = 2500
90
91# Prose patterns: word-boundary matches that catch the legacy vocabulary
92# wherever it appears as a standalone word -- prose comments ("master
93# enable"), HUM section titles, "Slave Select", the bare "SS" pin name.
94PATTERNS: tuple[tuple[str, re.Pattern[str]], ...] = (
95 ("master", re.compile(r"\bmaster(s|ed|ing|ship)?\b", re.IGNORECASE)),
96 ("slave", re.compile(r"\bslave(s|d)?\b", re.IGNORECASE)),
97 ("mosi", re.compile(r"\bMOSI\b")),
98 ("miso", re.compile(r"\bMISO\b")),
99 ("slave_select", re.compile(r"\bSlave[ _-]Select\b", re.IGNORECASE)),
100 ("ss_pin", re.compile(r"\bSS\b")), # SPI Slave Select pin abbreviation -- use CS
101)
102
103# Identifier-embedded detection.
104#
105# Python's `\b` does not fire between an underscore and a letter (`_` is a
106# word character), so the prose patterns above miss the legacy words when
107# they are welded into a snake_case / SCREAMING_CASE symbol
108# (`internal_spcr_master`, `make_master_i2s_cfg`,
109# ...). Instead of enumerating our own prefixes, flag ANY identifier that
110# carries `master`/`slave` as a leading component and is NOT part of a
111# vendored upstream namespace -- those APIs are referenced verbatim in our
112# glue code and cannot be renamed.
113IDENT_RE: re.Pattern[str] = re.compile(r"[A-Za-z_][A-Za-z0-9_]*")
114# `master`/`slave` as a component: at the identifier start or just after an
115# underscore (so `MASTEREN`, `ra8_spi_master_init`, `_slave` all match, while
116# camelCase `offsetFromMaster` and substrings like `enslave` do not).
117IDENT_TERM_RE: re.Pattern[str] = re.compile(r"(?:^|_)(?:master|slave)", re.IGNORECASE)
118# Upstream namespaces whose symbols legitimately appear in first-party glue
119# and are cited verbatim: Express Logic USBX / ThreadX / NetX / FileX /
120# LevelX, Renesas FSP (r_iic_* / r_sce_*), Mbed TLS, NimBLE / Mynewt, and
121# Espressif esp-hosted. Matched at the identifier head.
122VENDOR_IDENT_RE: re.Pattern[str] = re.compile(
123 r"^(?:_?ux_|r_iic|r_sce|_?nx_|ble_|mynewt|mbedtls|tls_|pre_?master"
124 r"|premaster|resumption_master|_?lx_|_?tx_|_?gx_|_?fx_|netx|threadx"
125 r"|usbx|levelx|esp_hosted|dcd_sim_slave|hcd_sim_host)",
126 re.IGNORECASE,
127)
128# Hardware register-bit names that spell a legacy token verbatim (Renesas
129# MIPI D-PHY DPHYMDC.MASTEREN); these are silicon names, not our symbols.
130HW_TOKENS: frozenset[str] = frozenset({"MASTEREN"})
131
132# Per-line opt-out marker.
133LEGACY_OK_RE: re.Pattern[str] = re.compile(r"LEGACY-OK\s*:")

136def identifier_violation(line: str) -> str | None:
137 """The offending identifier when a line names a symbol with legacy terminology.
138
139 Checks IDENTIFIERS, not prose: a comment discussing the legacy term --
140 often required when mapping a vendor document onto our names -- is fine,
141 while a symbol carrying it is not, because the symbol propagates.
142
143 Returns None when the line is clean.
144
145 Vendor-namespace identifiers and hardware register-bit names are
146 skipped: they are upstream contracts spelled verbatim, not symbols
147 this project is free to rename.
148 """
149 for m in IDENT_RE.finditer(line):
150 tok = m.group(0)
151 if not IDENT_TERM_RE.search(tok):
152 continue
153 if tok in HW_TOKENS:
154 continue
155 if VENDOR_IDENT_RE.match(tok):
156 continue
157 return tok
158 return None

194def iter_source_files(root: Path) -> list[Path]:
195 """Every in-scope first-party file, derived from git rather than a root list.
196
197 Enumeration goes through ``lint_targets.first_party_paths`` -- the shared
198 derived-scope primitive -- so ``infra/`` and ``mk/`` (the roots a hardcoded
199 list silently dropped, #549) are covered, and any future top-level
200 directory is in scope the day it lands. The only subtractions on top of what
201 that primitive already exempts (third_party/, generated fonts, build output)
202 are the docs-side vendored/generated directories in ``DOCS_VENDOR_DIRS``.
203 """
204 rels = set(first_party_paths(tuple(SCAN_EXTS)))
205 for name in SCAN_BASENAMES:
206 rels |= {rel for rel in first_party_paths((name,)) if Path(rel).name == name}
207 out: list[Path] = []
208 for rel in sorted(rels):
209 if set(Path(rel).parts) & DOCS_VENDOR_DIRS:
210 continue
211 out.append(root / rel)
212 return out

215def scan_file(path: Path, root: Path) -> list[tuple[Path, int, str, str]]:
216 """Report every legacy-terminology identifier in one file.
217
218 Self-exempt files -- this checker, the terminology policy -- are skipped
219 whole, since they must name the banned terms to define them.
220 """
221 rel = path.relative_to(root)
222 rel_str = str(rel)
223 if rel_str in SELF_EXEMPT_FILES:
224 return []
225 try:
226 text = path.read_text(encoding="utf-8")
227 except (OSError, UnicodeDecodeError):
228 return []
229 out: list[tuple[Path, int, str, str]] = []
230 for lineno, line in enumerate(text.splitlines(), start=1):
231 if LEGACY_OK_RE.search(line):
232 continue
233 matched = False
234 for term, regex in PATTERNS:
235 if not regex.search(line):
236 continue
237 out.append((rel, lineno, term, line.rstrip()))
238 matched = True
239 break
240 if matched:
241 continue
242 ident = identifier_violation(line)
243 if ident is not None:
244 out.append((rel, lineno, f"symbol:{ident}", line.rstrip()))
245 return out

248def selftest() -> int:
249 """Prove the detector fires on legacy terms, spares the legitimate ones, and scans real files.
250
251 Both directions plus a scope probe: a legacy identifier and the bare prose
252 tokens must FIRE, while an inclusive rewrite, a vendored-namespace symbol
253 and a hardware register-bit name must stay QUIET; the derived scope must
254 clear ``FILE_FLOOR`` and reach the roots a hardcoded list had dropped
255 (``infra/``, ``mk/``). A clean run over a scope that never sees those roots,
256 or one whose detector had stopped matching, proves nothing.
257
258 Returns:
259 0 when every assertion held in both directions, 1 otherwise.
260 """
261 failures: list[str] = []
262 # Prose / identifier detection, driven through scan_line-equivalent logic.
263 fire_lines = (
264 ("a legacy identifier", "ra8_err_t ra8_spi_master_init(void);"),
265 ("the bare MOSI token", "// route MOSI to the header"),
266 ('the "Slave Select" phrase', "assert the Slave Select line"),
267 )
268 for label, line in fire_lines:
269 prose = any(regex.search(line) for _, regex in PATTERNS)
270 ident = identifier_violation(line) is not None
271 expect(prose or ident, f"MUST FIRE: {label}", failures)
272
273 quiet_lines = (
274 ("an inclusive rewrite", "ra8_err_t ra8_spi_controller_init(void);"),
275 ("a vendored ux_ symbol", "ux_device_class_storage_master_read();"),
276 ("the MASTEREN register bit", "DPHYMDC.MASTEREN = 1U; // silicon name"),
277 )
278 for label, line in quiet_lines:
279 prose = any(regex.search(line) for _, regex in PATTERNS)
280 ident = identifier_violation(line) is not None
281 expect(not (prose or ident), f"MUST NOT FIRE: {label}", failures)
282
283 root = Path(__file__).resolve().parents[2]
284 files = iter_source_files(root)
285 rels = {str(p.relative_to(root)) for p in files if p.is_relative_to(root)}
286 expect(
287 len(files) >= FILE_FLOOR,
288 f"derived scope sees {len(files)} file(s) (floor {FILE_FLOOR})",
289 failures,
290 )
291 for root_name in ("infra", "mk"):
292 expect(
293 any(rel.startswith(root_name + "/") for rel in rels),
294 f"the derived scope reaches {root_name}/ (previously omitted)",
295 failures,
296 )
297 return report(failures)

300def main() -> int:
301 """Enforce OSHWA-inclusive terminology on first-party identifiers.
302
303 Scoped to identifiers rather than all text on purpose: vendor manuals and
304 external APIs still use the legacy terms, and comments mapping our names
305 onto theirs are required elsewhere in this tree. The rule governs what
306 this codebase NAMES, not what it may mention.
307
308 Returns 1 listing each offending symbol, 0 when the tree is clean, 2 when
309 the derived scope collapsed below FILE_FLOOR.
310 """
311 if "--selftest" in sys.argv[1:]:
312 return selftest()
313 root = Path(__file__).resolve().parents[2]
314 files = iter_source_files(root)
315 if len(files) < FILE_FLOOR:
316 sys.stderr.write(
317 f"inclusive-terminology: FATAL -- only {len(files)} file(s) in scope, floor "
318 f"is {FILE_FLOOR}. A collapsed scope reports a clean tree because it scanned "
319 "nothing.\n"
320 )
321 return 2
322 findings: list[tuple[Path, int, str, str]] = []
323 for f in files:
324 findings.extend(scan_file(f, root))
325
326 if not findings:
327 print("inclusive-terminology: 0 violations -- gate clean.")
328 return 0
329
330 print(f"inclusive-terminology: {len(findings)} violations found.")
331 for rel, lineno, term, line in findings[:MAX_FINDINGS_SHOWN]:
332 snippet = line if len(line) <= MAX_SNIPPET_LEN else line[:SNIPPET_TRUNCATE_LEN] + "..."
333 print(f" {rel}:{lineno} [{term}] {snippet}")
334 if len(findings) > MAX_FINDINGS_SHOWN:
335 print(f" ... {len(findings) - MAX_FINDINGS_SHOWN} more (truncated)")
336 print()
337 print("Per-line opt-out: append `LEGACY-OK: <reason>` on the offending line.")
338 print("See CLAUDE.md 'Terminology Standard' for the policy.")
339
340 if WARN_ONLY_MODE:
341 print("WARN_ONLY_MODE=True -- not failing the gate.")
342 return 0
343 return 1
