179static ra8_err_t priv_tile_content(const ra8_fmt_blob_t* atlas,
180 const ra8_jof_info_t* info,
181 uint32_t idx,
182 const fmt_decode_buf_t* buf,
183 fmt_tile_rec_t* rec)
184{
185 ra8_jof_memstore_t store = {.buf = atlas->bytes, .cap = atlas->len, .len = atlas->len};
186 uint16_t tw = 0U;
187 uint16_t th = 0U;
188 const ra8_err_t rc = ra8_jof_read_tile(ra8_jof_memstore_pread,
189 &store,
190 info,
191 (uint16_t)(idx % (uint32_t)info->tile_cols),
192 (uint16_t)(idx / (uint32_t)info->tile_cols),
193 buf->scratch,
194 buf->scratch_cap,
195 buf->cell,
196 buf->cell_cap,
197 &tw,
198 &th);
199 if (rc != k_ra8_ok) {
200 return rc;
201 }
202 rec->hash = priv_fnv1a(buf->cell, (size_t)rec->payload);
203 bool same = true;
204 const uint8_t first = buf->cell[0];
205 for (uint32_t i = 1U; (i < rec->payload) && same; ++i) { /* bounded by payload */
206 same = (buf->cell[i] == first);
207 }
208 rec->uniform = same;
209 return k_ra8_ok;

228RA8_INTERNAL
229static ra8_err_t priv_tile_rec(const ra8_fmt_blob_t* atlas,
230 const ra8_jof_info_t* info,
231 uint32_t idx,
232 fmt_tile_rec_t* rec)
233{
234 const size_t ent = (size_t)info->index_off + ((size_t)idx * (size_t)k_ra8_jof_index_entry);
235 if ((ent + (size_t)k_ra8_jof_index_entry) > atlas->len) {
236 return k_ra8_err_validation_failed;
237 }
238 rec->offset = ra8_fmt_rd_u32(&atlas->bytes[ent + (size_t)k_ra8_jof_idx_ofs_offset]);
239 rec->length = ra8_fmt_rd_u32(&atlas->bytes[ent + (size_t)k_ra8_jof_idx_ofs_length]);
240 const uint16_t tx = (uint16_t)(idx % (uint32_t)info->tile_cols);
241 const uint16_t ty = (uint16_t)(idx / (uint32_t)info->tile_cols);
242 const ra8_err_t drc = ra8_jof_tile_dims(info, tx, ty, &rec->tw, &rec->th);
243 if (drc != k_ra8_ok) {
244 return drc;
245 }
246 rec->payload = (uint32_t)rec->tw * (uint32_t)rec->th * (uint32_t)info->bpp;
247 if (((size_t)rec->offset + (size_t)rec->length) > (size_t)info->index_off) {
248 return k_ra8_err_validation_failed;
249 }
250 return k_ra8_ok;
251}

273RA8_INTERNAL
274static ra8_err_t
275priv_check_table(const fmt_tile_rec_t* recs, uint32_t count, uint32_t first, FILE* out)
276{
277 ra8_err_t verdict = k_ra8_ok;
278 uint32_t cursor = first;
279 for (uint32_t i = 0U; i < count; ++i) { /* bounded by tile_count */
280 if (recs[i].offset != cursor) {
281 (void)fprintf(out,
282 " ANOMALY tile %u: offset %u but previous tile ends at %u (%s)\n",
283 (unsigned)i,
284 (unsigned)recs[i].offset,
285 (unsigned)cursor,
286 (recs[i].offset < cursor) ? "OVERLAP" : "GAP");
287 verdict = k_ra8_err_validation_failed;
288 }
289 cursor = recs[i].offset + recs[i].length;
290 }
291 /* Identical CONTENT is only suspicious when the band actually carries detail.
292 * A solid-colour band -- the gutter between panels -- is uniform, and a real
293 * long-strip page repeats those many times over. Reporting them would cry
294 * wolf on most genuine pages, which is worse for debugging than not checking:
295 * the reader would learn to ignore the verdict. */
296 uint32_t uniform_pairs = 0U;
297 for (uint32_t i = 1U; i < count; ++i) { /* bounded by tile_count */
298 for (uint32_t j = 0U; j < i; ++j) { /* bounded by i */
299 if ((recs[i].hash != recs[j].hash) || (recs[i].payload != recs[j].payload)) {
300 continue;
301 }
302 if (recs[i].uniform && recs[j].uniform) {
303 uniform_pairs++;
304 continue;
305 }
306 (void)fprintf(out,
307 " ANOMALY tiles %u and %u decode to IDENTICAL non-uniform "
308 "content (%u bytes, hash=%08X) -- duplication is in the FILE\n",
309 (unsigned)j,
310 (unsigned)i,
311 (unsigned)recs[i].payload,
312 (unsigned)recs[i].hash);
313 verdict = k_ra8_err_validation_failed;
314 }
315 }
316 if (uniform_pairs > 0U) {
317 (void)fprintf(out,
318 " note: %u tile pair(s) share solid-colour content "
319 "(gutter bands -- legitimate, not duplication)\n",
320 (unsigned)uniform_pairs);
321 }
322 return verdict;
323}

382static ra8_err_t priv_collect_tiles(const ra8_fmt_blob_t* atlas,
383 const ra8_jof_info_t* info,
384 fmt_tile_rec_t* recs,
385 const ra8_fmt_opts_t* opts)
386{

388 fmt_decode_buf_t buf = {.cell = (uint8_t*)malloc((size_t)tmax),
389 .cell_cap = tmax,
390 .scratch = nullptr,

392 buf.scratch = (uint8_t*)malloc((size_t)buf.scratch_cap);
393 if ((buf.cell == nullptr) || (buf.scratch == nullptr)) {
394 free(buf.cell);
395 free(buf.scratch);
396 return k_ra8_err_no_mem;
397 }
398 ra8_err_t rc = k_ra8_ok;
399 for (uint32_t i = 0U; (rc == k_ra8_ok) && (i < info->tile_count); ++i) { /* bounded */
400 rc = priv_tile_rec(atlas, info, i, &recs[i]);
401 if (rc == k_ra8_ok) {
402 rc = priv_tile_content(atlas, info, i, &buf, &recs[i]);
403 }
404 if (rc != k_ra8_ok) {
405 (void)fprintf(opts->report, " tile %u could not be read or decoded\n", (unsigned)i);
406 }
407 }
408 free(buf.cell);
409 free(buf.scratch);
410 return rc;

413ra8_err_t ra8_fmt_jof_inspect(const ra8_fmt_blob_t* src, const ra8_fmt_opts_t* opts)
414{
415 RA8_CHECK_NULL_PTR(src, s_tag, "src must not be nullptr");
416 RA8_CHECK_NULL_PTR(opts, s_tag, "opts must not be nullptr");
417 ra8_jof_memstore_t store = {.buf = src->bytes, .cap = src->len, .len = src->len};
418 ra8_jof_info_t info = {};
419 ra8_err_t rc = ra8_jof_parse(ra8_jof_memstore_pread, &store, (uint64_t)src->len, &info);
420 if (rc != k_ra8_ok) {
421 (void)fprintf(opts->report, "JOF parse FAILED (rc=%d) -- container is invalid\n", (int)rc);
422 ra8_fmt_hex_dump(opts->report,
423 "header",
424 src->bytes,
425 (src->len < (size_t)k_fmt_hdr_dump_bytes) ? src->len
426 : (size_t)k_fmt_hdr_dump_bytes,
427 0U);
428 return rc;
429 }
430 priv_print_geom(opts->report, &info, src->len);
431 fmt_tile_rec_t* recs = (fmt_tile_rec_t*)calloc((size_t)info.tile_count, sizeof(*recs));
432 if (recs == nullptr) {
433 return k_ra8_err_no_mem;
434 }
435 rc = priv_collect_tiles(src, &info, recs, opts);
436 if (rc != k_ra8_ok) {
437 free(recs);
438 return rc;
439 }
440 if (opts->verbose) {
441 ra8_fmt_hex_dump(opts->report, "header", src->bytes, (size_t)k_fmt_hdr_dump_bytes, 0U);
442 ra8_fmt_hex_dump(opts->report,
443 "footer",
444 &src->bytes[src->len - (size_t)k_fmt_ftr_dump_bytes],
445 (size_t)k_fmt_ftr_dump_bytes,
446 (uint64_t)(src->len - (size_t)k_fmt_ftr_dump_bytes));
447 priv_print_table(opts->report, recs, info.tile_count);
448 }
449 const ra8_err_t verdict =
450 priv_check_table(recs, info.tile_count, (uint32_t)k_ra8_jof_hdr_bytes, opts->report);
451 free(recs);
452 (void)fprintf(opts->report,
453 "verdict: %s\n",
454 (verdict == k_ra8_ok) ? "VALID (coverage exact, no duplicate tiles)"
455 : "INVALID (see anomalies above)");
456 return verdict;
457}
