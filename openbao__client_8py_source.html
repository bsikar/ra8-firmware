3"""Shared HTTP client for the EXISTING OpenBao server.
4
5This does NOT run, embed, or spin up a vault. It is a thin urllib client
6(no `bao` CLI needed) that talks to the OpenBao server you already run --
7the k3s pod reached at BAO_ADDR (e.g. http://100.64.0.1:32200). It performs
8AppRole login plus KV v2 read / write / metadata, so the same operator
9identity backs both the HIL Tapo secrets (hil_secrets.py) and the
10root-of-trust key store (rot_keystore.py). No server = fall back to the
11local path (see rot_keystore.py's `local` backend / the .env pattern).
12
13Consumer config comes from an operator-controlled 0600 creds file (default
14~/.config/hil/openbao.env, override with HIL_OPENBAO_ENV) or the matching
15environment variables (e.g. CI injecting GitHub Actions secrets):
16
17 BAO_ADDR base URL, e.g. http://100.64.0.1:32200
18 ROLE_ID AppRole role id
19 SECRET_ID AppRole secret id
20 BAO_APPROLE_PATH auth mount (default "approle")
21 BAO_KV_MOUNT KV v2 mount (default "secret")
22
23The file never holds the secrets themselves -- only how to reach the vault
24and the AppRole identity.
25"""

43def creds_path() -> Path:
44 """Return the consumer creds file path, honouring HIL_OPENBAO_ENV."""
45 override = os.environ.get("HIL_OPENBAO_ENV", "").strip()

49def _parse_env_file(path: Path) -> dict[str, str]:
50 out: dict[str, str] = {}
51 # External, non-repo file: decode permissively rather than ASCII-strict.
52 for raw in path.read_text(encoding="utf-8").splitlines():
53 line = raw.strip()
54 if not line or line.startswith("#") or "=" not in line:
55 continue
56 key, val = line.split("=", 1)
57 out[key.strip()] = val.strip()
58 return out

61def load_config() -> dict[str, str]:
62 """Load consumer config from the 0600 creds file, else the environment."""
63 path = creds_path()
64 if path.exists():
65 return _parse_env_file(path)
66 keys = ("BAO_ADDR", "ROLE_ID", "SECRET_ID", "BAO_APPROLE_PATH", "BAO_KV_MOUNT")
67 return {k: os.environ[k] for k in keys if k in os.environ}

70def _http_json(
71 url: str,
72 headers: dict[str, str],
73 payload: dict | None = None,
74 method: str | None = None,
75) -> dict:
76 if not url.startswith(("http://", "https://")):
77 msg = "refusing non-http(s) OpenBao URL"
78 raise OpenBaoError(msg)
79 body = json.dumps(payload).encode("ascii") if payload is not None else None
80 verb = method or ("POST" if payload is not None else "GET")
81 hdrs = {"Content-Type": "application/json"} if body is not None else {}
82 hdrs.update(headers)
83 # Scheme guarded above; BAO_ADDR comes from a local 0600 operator-controlled
84 # creds file, never untrusted input.
85 req = urllib.request.Request(url, data=body, headers=hdrs, method=verb) # noqa: S310
86 try:
87 with urllib.request.urlopen(req, timeout=_HTTP_TIMEOUT_S) as resp: # noqa: S310
88 raw = resp.read().decode("utf-8")
89 except urllib.error.HTTPError as exc:
90 msg = f"OpenBao HTTP {exc.code} for {verb} {url}"
91 raise OpenBaoError(msg) from exc
92 except (urllib.error.URLError, TimeoutError, OSError) as exc:
93 msg = f"OpenBao unreachable: {exc}"
94 raise OpenBaoError(msg) from exc
95 return json.loads(raw) if raw else {}

98class OpenBaoClient:
99 """AppRole-authenticated KV v2 client for a single OpenBao server."""
100
101 def __init__(self, cfg: dict[str, str] | None = None) -> None:
102 """Read connection settings from ``cfg``, or from the environment when None.
103
104 Performs NO I/O and never raises on a missing setting -- every field
105 defaults to empty. Reachability is a separate question answered by
106 ``configured`` and ``login``, which is what lets a caller construct a
107 client and then decide to fall back to a local store instead.
108 """
109 cfg = cfg if cfg is not None else load_config()
110 self.addr = cfg.get("BAO_ADDR", "").rstrip("/")
111 self.role_id = cfg.get("ROLE_ID", "")
112 self.secret_id = cfg.get("SECRET_ID", "")
113 self.approle = cfg.get("BAO_APPROLE_PATH", "approle")
114 self.mount = cfg.get("BAO_KV_MOUNT", "secret")
115 self._token: str | None = None
116
117 @property
118 def configured(self) -> bool:
119 """True when address + AppRole identity are all present."""
120 return bool(self.addr and self.role_id and self.secret_id)
121
122 def login(self) -> str:
123 """AppRole login; caches and returns the client token."""
124 if self._token:
125 return self._token
126 if not self.configured:
127 msg = "OpenBao not configured (missing BAO_ADDR / ROLE_ID / SECRET_ID)"
128 raise OpenBaoError(msg)
129 resp = _http_json(
130 f"{self.addr}/v1/auth/{self.approle}/login",
131 {},
132 {"role_id": self.role_id, "secret_id": self.secret_id},
133 )
134 token = resp.get("auth", {}).get("client_token")
135 if not token:
136 msg = "OpenBao login returned no client_token"
137 raise OpenBaoError(msg)
138 self._token = token
139 return token
140
141 def _mount(self, mount: str | None) -> str:
142 return mount or self.mount
143
144 def kv_get(self, path: str, version: int | None = None, mount: str | None = None) -> dict:
145 """Return one KV v2 version's secret data (latest unless version given)."""
146 url = f"{self.addr}/v1/{self._mount(mount)}/data/{path}"
147 if version is not None:
148 url = f"{url}?version={int(version)}"
149 resp = _http_json(url, {"X-Vault-Token": self.login()})
150 return resp.get("data", {}).get("data", {})
151
152 def kv_put(self, path: str, data: dict, mount: str | None = None) -> int:
153 """Write a new KV v2 version; return the created version number."""
154 resp = _http_json(
155 f"{self.addr}/v1/{self._mount(mount)}/data/{path}",
156 {"X-Vault-Token": self.login()},
157 {"data": data},
158 )
159 return int(resp.get("data", {}).get("version", 0))
160
161 def kv_metadata(self, path: str, mount: str | None = None) -> dict:
162 """Return KV v2 metadata (per-version created_time, custom_metadata)."""
163 url = f"{self.addr}/v1/{self._mount(mount)}/metadata/{path}"
164 try:
165 resp = _http_json(url, {"X-Vault-Token": self.login()})
166 except OpenBaoError as exc:
167 if "HTTP 404" in str(exc):
168 return {}
169 raise
170 return resp.get("data", {})
171

173 self,

178 """Set path-level custom_metadata (tags) on a KV v2 secret."""
179 _http_json(

181 {"X-Vault-Token": self.login()},

183 )
