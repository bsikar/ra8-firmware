75RA8_INTERNAL static mdl_robots_fetch_result_t
76session_fetch(void* ctx, const char* robots_url, char* buf, size_t cap, size_t* out_len)
77{
78 mdl_session_t* s = (mdl_session_t*)ctx;
79 const mdl_net_req_t req = {.user_agent = s->user_agent,
80 .referer = nullptr,
81 .timeout_ms = k_robots_timeout_ms};
82 mdl_net_resp_t resp = {};
83 if (mdl_net_get_buf(s->net, robots_url, &req, buf, cap, out_len, &resp) == k_ra8_ok) {
84 return k_mdl_robots_fetch_ok;
85 }
86 if ((resp.status >= (long)k_http_server_err_min) &&
87 (resp.status <= (long)k_http_server_err_max)) {
88 return k_mdl_robots_fetch_denied;
89 }
90 return k_mdl_robots_fetch_absent;
91}

93bool mdl_session_url_allowed(mdl_session_t* session, const char* url, uint32_t* crawl_delay_ms)
94{
95 if (crawl_delay_ms != nullptr) {
96 *crawl_delay_ms = 0U;
97 }
98 if ((session == nullptr) || (url == nullptr) || !session->honor_robots) {
99 return true;
100 }
101 char host[k_mdl_robots_host_max];
102 if (!mdl_url_host(url, host, sizeof(host))) {
103 return true; /* cannot identify the host -> cannot apply its rules */
104 }
105 char path[k_mdl_robots_path_max];
106 if (!mdl_url_path(url, path, sizeof(path))) {
107 (void)snprintf(path, sizeof(path), "/");
108 }
109 const mdl_robots_t* rules = mdl_robots_cache_consult(&session->cache,
110 url_scheme(url),
111 host,
112 s_ua_token,
113 session_fetch,
114 session,
115 session->scratch,
116 sizeof(session->scratch));
117 if (rules == nullptr) {
118 (void)
119 fprintf(stderr, "media_dl: robots.txt (5xx) disallows all of %s; skipping %s\n", host, path);
120 return false;
121 }
122 if ((crawl_delay_ms != nullptr) && rules->have_crawl_delay) {
123 *crawl_delay_ms = rules->crawl_delay_ms;
124 }
125 const char* reason = mdl_robots_disallow_reason(rules, path);
126 if (reason != nullptr) {
127 (void)fprintf(stderr,
128 "media_dl: robots.txt disallows %s on %s (rule \"%s\"); skipping\n",
129 path,
130 host,
131 reason);
132 return false;
133 }
134 return true;
135}
