4"""check_mcdc_floor.py -- per-file MC/DC FLOOR gate (no allowlist).
5
6Per CLAUDE.md "IEC 61508 SIL 3 / DO-178C Level B" every compound boolean
7decision in first-party code must be covered to full Modified
8Condition/Decision Coverage. The historical MC/DC gate only looked at the
9project-wide TOTAL row (`mcdc_report.sh` + `.github/mcdc-baseline.txt`),
10which a well-covered majority can hold above a baseline while an individual
11file rots: an aggregate is an average, and averages hide per-file holes.
12This gate closes that hole the same way `check_coverage_floor.py` closes it
13for line coverage -- it fails CI if ANY first-party file drops below the
14floor, and it deliberately has NO allowlist / no per-file exemption table.
15
16Unit of measure: the llvm-cov "MC/DC Decision Region" (one compound `&&` /
17`||` decision), the same unit `regen_mcdc_gaps.py` classifies. A file's
18score is its *reachable* MC/DC rate:
19
20 reachable_total = total_decisions - deactivated_decisions
21 reachable_covered = decisions at 100% MC/DC
22 reachable_pct = 100 * reachable_covered / reachable_total
23
24Deactivated decisions (DO-178C 6.4.4.3 -- documented as unreachable on any
25public-API path, catalogued in docs/MCDC_DEACTIVATIONS.md and driven by an
26explicit `// mcdc-deactivated:` annotation or a conservative structural
27classifier) are dropped from BOTH numerator and denominator, exactly as the
28line-coverage floor drops `gcovr/excluded` lines. This is the deactivation
29mechanism, not an allowlist: a decision cannot escape the floor without a
30catalogued rationale that the whole tree can audit. A file whose only gaps
31are deactivated therefore stays at 100% reachable and passes.
32
33Input: `build/mcdc-report/mcdc_per_file.json`, written by
34`regen_mcdc_gaps.py` from the live `make mcdc` report
35(`build/mcdc-report/mcdc.txt`). Run the MC/DC build first (which runs the
36regenerator), then this. Scope mirrors the line-coverage floor: files under
37`libs/` or `src/`, excluding vendored SOUP (`libs/third_party/`) and
38generated font tables (`libs/ra8_fonts/`).
39
40Exit 0 if every in-scope file with at least one reachable decision is
41>= FLOOR_PCT, else exit 1 with the offenders.
42
43Copyright (c) 2026 Brighton Sikarskie
44SPDX-License-Identifier: MIT
45"""

47from __future__ import annotations

49import json

51from pathlib import Path

53FLOOR_PCT = 100.0
54"""Per-file reachable-MC/DC floor, in percent. DO-178C Level B mandates full
55MC/DC of every reachable compound decision; a reachable gap in any single
56file is a hard CI failure. Deactivated (documented-unreachable) decisions are
57excluded per DO-178C 6.4.4.3, so 100% is the honest bar the tree meets today.
58No allowlist: a file below this is fixed at the root (a real MC/DC vector, or
59a catalogued `// mcdc-deactivated:` rationale), never grandfathered."""
60
61REPO_ROOT = Path(__file__).resolve().parents[2]
62MCDC_JSON = REPO_ROOT / "build" / "mcdc-report" / "mcdc_per_file.json"
63
64IN_SCOPE_PREFIXES = ("libs/", "src/")
65"""First-party source roots held to the floor (mirrors the line-coverage floor)."""
66
67OUT_OF_SCOPE_PREFIXES = ("libs/third_party/", "libs/ra8_fonts/")
68"""Vendored SOUP and generated font tables -- exempt from first-party rules
69per the CLAUDE.md coding-standards scope, so exempt from the floor too."""

72def normalize(path: str) -> str:
73 """Normalise a coverage-JSON ``file`` field to a repo-relative POSIX path.
74
75 The field may arrive absolute or already relative, so both are handled.
76
77 The absolute-path split marker is derived from the checkout directory
78 basename (`REPO_ROOT.name`, itself resolved from this file's location)
79 rather than a hardcoded project name, so the gate strips the prefix
80 correctly from any clone regardless of what the repo directory is named.
81 """
82 p = path.replace("\\", "/")
83 marker = "/" + REPO_ROOT.name + "/"
84 if marker in p:
85 p = p.split(marker, 1)[1]
86 return p.lstrip("./")
87
88
89def in_scope(rel: str) -> bool:
90 """True if `rel` is a first-party file subject to the MC/DC floor."""
91 if not rel.startswith(IN_SCOPE_PREFIXES):
92 return False
93 return not rel.startswith(OUT_OF_SCOPE_PREFIXES)
94

96def file_reachable(entry: dict) -> tuple[int, int]:
97 """Return (reachable_covered, reachable_total) decisions for one file.
98
99 Deactivated decisions (DO-178C 6.4.4.3) are removed from the denominator;
100 covered decisions are never deactivated, so the numerator is just the
101 count of decisions at 100% MC/DC.
102 """
103 total = int(entry.get("total_decisions", 0))
104 covered = int(entry.get("covered_decisions", 0))
105 deactivated = int(entry.get("deactivated_decisions", 0))
106 reachable_total = total - deactivated
107 return covered, reachable_total
108
109
110def _collect_offenders(
111 files: list[dict],
112) -> tuple[list[tuple[float, str, int, int]], int]:
113 """Return ``(offenders, checked)`` for every in-scope file with a decision.
114
115 A file with no reachable decision is skipped rather than counted as a pass:
116 it has nothing to measure, and scoring it 100% would dilute the floor.
117 """
118 offenders: list[tuple[float, str, int, int]] = []
119 checked = 0
120 for entry in files:
121 rel = normalize(entry.get("file", ""))
122 if not in_scope(rel):
123 continue
124 covered, reachable_total = file_reachable(entry)
125 if reachable_total <= 0:
126 continue
127 checked += 1
128 pct = 100.0 * covered / reachable_total
129 if pct < FLOOR_PCT:
130 offenders.append((pct, rel, covered, reachable_total))
131 return offenders, checked

134def _report_offenders(offenders: list[tuple[float, str, int, int]]) -> None:
135 """Print the below-floor table, worst first, with the remedy."""
136 offenders.sort()
137 print(
138 f"check_mcdc_floor.py: {len(offenders)} first-party file(s) below "
139 f"the {FLOOR_PCT:.0f}% reachable-MC/DC floor (NO allowlist):"
140 )
141 print(" mc/dc covered/reachable file")
142 for pct, rel, covered, reachable_total in offenders:
143 print(f" {pct:5.1f}% {covered:5d}/{reachable_total:<5d} {rel}")
144 print(
145 "Fix each at the root -- add the missing MC/DC vector (N+1 vectors "
146 "for N conditions; see docs/MCDC.md), or, if the gap is genuinely "
147 "unreachable on any public-API path, catalogue it with a "
148 "`// mcdc-deactivated:` rationale per DO-178C 6.4.4.3. Do NOT add "
149 "an allowlist."
150 )

153def main() -> int:
154 """Fail when any in-scope file sits below the reachable MC/DC floor.
155
156 The floor is REACHABLE coverage, not absolute: decisions classified as
157 deactivated are excluded, so the number this gates is what tests could
158 actually cover rather than a figure no test can ever reach.
159
160 A missing JSON report exits 1 rather than passing vacuously.
161 """
162 if not MCDC_JSON.is_file():
163 print(
164 f"check_mcdc_floor.py: ERROR -- {MCDC_JSON} not found; "
165 f"run `bash scripts/report/mcdc_report.sh` first."
166 )
167 return 1
168
169 try:
170 data = json.loads(MCDC_JSON.read_text(encoding="utf-8"))
171 except (OSError, json.JSONDecodeError) as exc:
172 print(f"check_mcdc_floor.py: ERROR -- cannot read MC/DC JSON: {exc}")
173 return 1
174
175 files = data.get("files", [])
176 if not files:
177 print("check_mcdc_floor.py: ERROR -- MC/DC JSON has no files.")
178 return 1
179
180 offenders, checked = _collect_offenders(files)
181
182 if checked == 0:
183 print(
184 "check_mcdc_floor.py: ERROR -- no in-scope files matched; check the JSON path / scope."
185 )
186 return 1
187
188 if offenders:
189 _report_offenders(offenders)
190 return 1
191
192 print(
193 f"check_mcdc_floor.py: PASS -- all {checked} first-party file(s) "
194 f"with a reachable decision are >= {FLOOR_PCT:.0f}% MC/DC."
195 )
196 return 0

200 sys.exit(main())
