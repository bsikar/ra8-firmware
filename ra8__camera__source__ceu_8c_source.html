113RA8_INTERNAL static ra8_err_t internal_ceu_wait_for_frame(ra8_camera_source_ceu_state_t* state,
114 uint32_t* out_bytes)
115{
116 for (uint32_t attempt = 0U; attempt < state->poll_attempts; attempt += 1U) {
117 ra8_ceu_status_t status = {};
118 ra8_err_t err = ra8_ceu_status_snapshot(&status);
119 if (err != k_ra8_ok) {
120 return err;
121 }
122 state->last_events |= status.events;
123 if ((status.events & (uint32_t)k_ceu_fatal_events) != 0U) {
124 (void)ra8_ceu_clear_status(status.events);
125 const ra8_err_t reset_err = ra8_ceu_reset();
126 (void)ra8_ceu_clear_status((uint32_t)k_ra8_ceu_evt_mask_all);
127 if (reset_err != k_ra8_ok) {
128 return reset_err;
129 }
130 return k_ra8_err_hw_error;
131 }
132 if ((status.events & (uint32_t)k_ra8_ceu_evt_cpe) != 0U) {
133 internal_ceu_frame_bytes(state, &status, out_bytes);
134 (void)ra8_ceu_clear_status(state->last_events);
135 return k_ra8_ok;
136 }
137 ra8_delay_ms(state->poll_interval_ms);
138 }
139 const ra8_err_t reset_err = ra8_ceu_reset();
140 (void)ra8_ceu_clear_status((uint32_t)k_ra8_ceu_evt_mask_all);
141 if (reset_err != k_ra8_ok) {
142 return reset_err;
143 }
144 return k_ra8_err_hw_timeout;
145}

166RA8_INTERNAL static ra8_err_t
167internal_ceu_capture(void* ctx, const ra8_camera_buffer_t* buffer, ra8_camera_frame_t* out_frame)
168{
169 ra8_camera_source_ceu_state_t* state = (ra8_camera_source_ceu_state_t*)ctx;
170 if (state == nullptr) {
171 return k_ra8_err_not_initialized;
172 }
173 if (!state->initialized) {
174 return k_ra8_err_not_initialized;
175 }
176 if (buffer->capacity < state->info.frame_bytes_max) {
177 return k_ra8_err_invalid_size;
178 }
179 if (((uintptr_t)buffer->data & (uintptr_t)k_ceu_buffer_alignment_mask) != 0U) {
180 return k_ra8_err_invalid_arg;
181 }
182 state->last_events = 0U;
183 ra8_err_t err =
184 ra8_cache_dcache_clean_invalidate_by_addr(buffer->data, state->info.frame_bytes_max);
185 if (err != k_ra8_ok) {
186 return err;
187 }
188 const ra8_ceu_buffers_t buffers = {
189 .y_top = buffer->data,
190 };
191 err = ra8_ceu_capture_start_ex(&buffers);
192 if (err != k_ra8_ok) {
193 return err;
194 }
195 uint32_t captured_bytes = 0U;
196 err = internal_ceu_wait_for_frame(state, &captured_bytes);
197 if (err != k_ra8_ok) {
198 return err;
199 }
200 if (captured_bytes == 0U) {
201 return k_ra8_err_invalid_size;
202 }
203 if (captured_bytes > buffer->capacity) {
204 return k_ra8_err_invalid_size;
205 }
206 err = ra8_cache_dcache_invalidate_by_addr(buffer->data, captured_bytes);
207 if (err != k_ra8_ok) {
208 return err;
209 }
210 *out_frame = (ra8_camera_frame_t){
211 .data = buffer->data,
212 .bytes = captured_bytes,
213 .stride_bytes = state->info.stride_bytes,
214 .width = state->info.width,
215 .height = state->info.height,
216 .format = state->info.format,
217 };
218 return k_ra8_ok;
219}

261RA8_INTERNAL static ra8_err_t internal_ceu_cfg_valid(const ra8_camera_source_ceu_cfg_t* cfg)
262{
263 if (cfg->output.frame_bytes_max == 0U) {
264 return k_ra8_err_invalid_arg;
265 }
266 if (cfg->output.width == 0U) {
267 return k_ra8_err_invalid_arg;
268 }
269 if (cfg->output.height == 0U) {
270 return k_ra8_err_invalid_arg;
271 }
272 if (cfg->poll_interval_ms == 0U) {
273 return k_ra8_err_invalid_arg;
274 }
275 if (cfg->poll_attempts == 0U) {
276 return k_ra8_err_invalid_arg;
277 }
278 if ((cfg->ceu.capture_format == k_ra8_ceu_fmt_data_enable) !=
279 (cfg->output.format == k_ra8_camera_format_jpeg)) {
280 return k_ra8_err_invalid_arg;
281 }
282 if (cfg->output.format == k_ra8_camera_format_jpeg) {
283 if (cfg->output.stride_bytes != 0U) {
284 return k_ra8_err_invalid_arg;
285 }
286 if (cfg->ceu.image_area_size != cfg->output.frame_bytes_max) {
287 return k_ra8_err_invalid_arg;
288 }
289 }
290 return k_ra8_ok;
291}
