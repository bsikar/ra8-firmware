570static ra8_err_t app_write_bootsel(uint8_t bank)
571{
572 uint8_t rec[k_app_mram_block_bytes] = {};
573 for (uint8_t i = 0U; i < (uint8_t)k_app_rec_off_bank; ++i) {
574 rec[(uint8_t)k_app_rec_off_magic + i] =
575 (uint8_t)((uint32_t)k_app_bootsel_magic >> ((uint32_t)i * (uint32_t)k_app_octet_bits));
576 }
577 rec[k_app_rec_off_bank] = bank;
578 ++s_bootsel_seq;
579 for (uint8_t i = 0U; i < (uint8_t)k_app_rec_off_bank; ++i) {
580 rec[(uint8_t)k_app_rec_off_seq + i] =
581 (uint8_t)(s_bootsel_seq >> ((uint32_t)i * (uint32_t)k_app_octet_bits));
582 }
583 const ra8_err_t e = ra8_flash_extra_mram_erase((uint32_t)k_app_bootsel_addr);
584 if (e != k_ra8_ok) {
585 return e;
586 }
587 return ra8_flash_extra_mram_write((uint32_t)k_app_bootsel_addr, rec, (uint32_t)sizeof rec);
588}

631static ra8_err_t app_read_bootsel(uint8_t* out_bank, bool* out_valid)
632{
633 RA8_CHECK_NULL_PTR(out_bank, s_tag, "bootsel out_bank");
634 RA8_CHECK_NULL_PTR(out_valid, s_tag, "bootsel out_valid");
635 const uint8_t* rec = (const uint8_t*)(uintptr_t)k_app_bootsel_addr;
636 uint32_t magic = 0U;
637 for (uint8_t i = 0U; i < (uint8_t)k_app_rec_off_bank; ++i) {
638 magic |= ((uint32_t)rec[(uint8_t)k_app_rec_off_magic + i])
639 << ((uint32_t)i * (uint32_t)k_app_octet_bits);
640 }
641 *out_valid = (magic == (uint32_t)k_app_bootsel_magic);
642 *out_bank = rec[k_app_rec_off_bank];
643 return k_ra8_ok;
644}

794static app_outcome_t
795app_run_attempt(const uint8_t* source, const ra8_ota_manifest_t* m, uint8_t* out_bank)
796{
797 *out_bank = (uint8_t)k_app_bank_a;
798 if (app_write_bootsel((uint8_t)k_app_bank_a) != k_ra8_ok) {
799 return k_app_outcome_indeterminate;
800 }
801 s_net_source.data = source;
802 s_net_source.len = (uint32_t)k_app_image_bytes;
803 s_net_source.pos = 0U;
804
805 ra8_ota_cfg_t cfg;
806 app_make_cfg(&cfg);
807 (void)ra8_ota_deinit();
808 if (ra8_ota_init(&cfg) != k_ra8_ok) {
809 return k_app_outcome_indeterminate;
810 }
811
812 ra8_err_t result = ra8_ota_download_to_inactive_bank(m);
813 if (result == k_ra8_ok) {
814 result = ra8_ota_verify_signature(m);
815 }
816 if (result == k_ra8_ok) {
817 result = ra8_ota_commit_and_reboot();
818 }
819 const ra8_ota_state_t state = ra8_ota_get_state();
820
821 bool valid = false;
822 (void)app_read_bootsel(out_bank, &valid);
823 (void)ra8_ota_deinit();
824 return app_ab_classify(result, state);
825}

844static ra8_err_t app_stage_images(void)
845{
846 for (uint32_t i = 0U; i < (uint32_t)k_app_image_bytes; ++i) {
847 s_image_good[i] = (uint8_t)(i * (uint32_t)k_app_img_seed_mul + (uint32_t)k_app_img_seed_add);
848 s_image_bad[i] = s_image_good[i];
849 }
850 /* Corrupt the rollback image so its re-hash cannot match the manifest digest
851 * -- the "bad download" that must NOT be committed. */
852 s_image_bad[0] = (uint8_t)(s_image_good[0] ^ (uint8_t)k_app_byte_all_ones);
853 return ra8_rsip_sha256(s_image_good, (uint32_t)k_app_image_bytes, s_image_good_digest);
854}

927int32_t main(void)
928{
929 if (app_setup() != k_ra8_ok) {
930 /* NASA Rule 2 exemption: terminal panic spin on a fatal bring-up error. */
931 while (true) {
932 }
933 }
934
935 const bool staged = (app_stage_images() == k_ra8_ok);
936
937 bool committed = false;
938 bool rolled_back = false;
939 if (staged) {
940 ra8_ota_manifest_t m;
941 app_make_manifest(&m);
942
943 uint8_t bank_after = (uint8_t)k_app_bank_a;
944 const app_outcome_t good = app_run_attempt(s_image_good, &m, &bank_after);
945 committed = (good == k_app_outcome_committed) && (bank_after == (uint8_t)k_app_bank_b);
946
947 const app_outcome_t bad = app_run_attempt(s_image_bad, &m, &bank_after);
948 rolled_back = (bad == k_app_outcome_rolled_back) && (bank_after == (uint8_t)k_app_bank_a);
949 }
950
951 (void)app_print("\r\nra8d2 ota_ab_orchestration (A/B stage/verify/commit/rollback)\r\n");
952
953 /* NASA Rule 2 exemption: deliberate report loop; the verdict is latched, the
954 * banner just repeats so a UART scrape reliably captures it. */
955 while (true) {
956 app_print_banner(staged, committed, rolled_back);
957 for (volatile uint32_t d = 0U; d < (uint32_t)k_app_report_delay; ++d) {
958 /* coarse inter-report busy delay (bounded). */
959 }
960 }
961}
