125{

209static ra8_vreg_ocp_t internal_ocp_from_dcdcctl(uint8_t dcdcctl)
210{
211 if ((dcdcctl & k_ra8_vreg_mask_ocpen) == 0U) {

239{
240 uint8_t v = 0U;

279static void internal_dcdc_enable_sequence(bool fast)
280{
281 uint8_t cur = *ra8_vreg_dcdcctl();
282
283 /* Step 1: enable IO buffer (STOPZA = 1).
284 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
285 cur |= k_ra8_vreg_mask_stopza;
286 *ra8_vreg_dcdcctl() = cur;
287
288 /* Step 2: turn on DCDC Vref (PD = 0).
289 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
290 cur &= (uint8_t)~k_ra8_vreg_mask_pd;
291 *ra8_vreg_dcdcctl() = cur;
292
293 if (fast) {
294 /* Fast path: jump straight to the final value with FST asserted.
295 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
296 *ra8_vreg_dcdcctl() = k_ra8_vreg_dcdc_step_fast_on;
297 } else {
298 /* Step 3: switch Vref to low-power.
299 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
300 *ra8_vreg_dcdcctl() = k_ra8_vreg_dcdc_step_lp_vref;
301 /* Step 4: turn off LDO and turn on DCDC.
302 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
303 *ra8_vreg_dcdcctl() = k_ra8_vreg_dcdc_step_dcdc_on;
304 /* Step 5: enable DCDC over-current protection.
305 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
306 *ra8_vreg_dcdcctl() = k_ra8_vreg_dcdc_step_with_ocp;
307 }
308
309 if (fast) {
310 s_state.dcdcctl = k_ra8_vreg_dcdc_step_fast_on;
311 } else {
312 s_state.dcdcctl = k_ra8_vreg_dcdc_step_with_ocp;
313 }
314}

330static void internal_dcdc_disable_sequence(bool keep_lcboost)
331{
332 /* The LDO selection is "everything off" -- optionally with LCBOOST
333 * still asserted so the LDO can run at subosc-speed VDD.
334 * HUM Ch 68.2.2 "External VDD Mode" p 4033
335 *
336 * NB: do NOT touch s_state.dcdcctl here. enter_standby uses this
337 * helper transiently and relies on the cached value to restore on
338 * exit; set_mode(LDO) updates the cache itself for the permanent
339 * mode-switch path. */
340 uint8_t v = 0U;
341 if (keep_lcboost) {
342 v = k_ra8_vreg_mask_lcboost;
343 }
344 *ra8_vreg_dcdcctl() = v;
345}

352[[nodiscard]] ra8_err_t ra8_vreg_init(const ra8_vreg_cfg_t* cfg)
353{
354 RA8_CHECK_NULL_PTR(cfg, s_tag, "cfg must not be nullptr");
355 const ra8_err_t verr = internal_validate_cfg(cfg);
356 if (verr != k_ra8_ok) {
357 return verr;
358 }
359
360 /* Clear LVOCR first so a stale low-voltage profile cannot affect
361 * the DCDC handshake (LV profiles widen the timing budget).
362 * HUM Ch 68.3 "Usage Notes" p 4034 */
363 *ra8_vreg_lvocr() = 0U;
364
365 /* VCCSEL biases the DCDC for the actual VCC range; program before
366 * turning DCDC on.
367 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
368 *ra8_vreg_vccsel() = (uint8_t)((uint8_t)cfg->vccsel & k_ra8_vreg_vccsel_mask);
369
370 if (cfg->mode == k_ra8_vreg_mode_dcdc) {
371 /* Use the staged handshake so the chip ends in the documented
372 * final state (STOPZA + DCDCON + OCPEN, optionally + FST). */
373 internal_dcdc_enable_sequence(cfg->fast_startup);
374 if (!cfg->ldo_boost) {
375 /* LCBOOST has no effect in DCDC mode; nothing to apply. */
376 }
377 } else {
378 /* LDO mode: write the boost / OCP / FST bits directly without
379 * staging. OCPEN has no DCDC to protect when DCDCON==0 but the
380 * caller may have asked for it so the value is restored across
381 * a future LDO->DCDC transition.
382 * HUM Ch 68.2.2 "External VDD Mode" p 4033 */
383 const uint8_t ldo = internal_pack_ldo_dcdcctl(cfg);
384 *ra8_vreg_dcdcctl() = ldo;
385 s_state.dcdcctl = ldo;
386 }
387
388 /* Apply LV profile last so the chip is in the right power mode
389 * before relaxing the timing.
390 * HUM Ch 68.3 "Usage Notes" p 4034 */
391 const uint8_t lvocr = internal_lvocr_from_profile(cfg->lv_profile);
392 *ra8_vreg_lvocr() = lvocr;
393
394 s_state.vccsel = cfg->vccsel;
395 s_state.lvocr = lvocr;
396 s_state.ocp = cfg->ocp;
397 s_state.lv_profile = cfg->lv_profile;
398 s_state.fast_startup = cfg->fast_startup;
399 s_state.ldo_boost = cfg->ldo_boost;
400 s_state.last_standby = k_ra8_vreg_standby_software;
401 s_state.initialized = true;
402
403 ra8_log_info(s_tag, "vreg_init");
404 return k_ra8_ok;
405}

407[[nodiscard]] ra8_err_t ra8_vreg_deinit(void)
408{
409 /* DCDCCTL = 0 returns to LDO with all features (OCP, fast-start,
410 * low-power Vref) disabled.
411 * HUM Ch 68.2.2 "External VDD Mode" p 4033 */
412 *ra8_vreg_dcdcctl() = 0U;
413 /* VCCSEL irrelevant in LDO; clear it.
414 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
415 *ra8_vreg_vccsel() = 0U;
416 /* LVOCR cleared so we do not silently leave a low-voltage profile
417 * set across re-init.
418 * HUM Ch 68.3 "Usage Notes" p 4034 */
419 *ra8_vreg_lvocr() = 0U;
420
421 s_state.dcdcctl = 0U;
422 s_state.vccsel = k_ra8_vreg_vccsel_2v4_to_2v7;
423 s_state.lvocr = 0U;
424 s_state.ocp = k_ra8_vreg_ocp_off;
425 s_state.lv_profile = k_ra8_vreg_lv_off;
426 s_state.fast_startup = false;
427 s_state.ldo_boost = false;
428 s_state.initialized = false;
429 return k_ra8_ok;
430}

437[[nodiscard]] ra8_err_t ra8_vreg_set_mode(ra8_vreg_mode_t mode)
438{
439 if ((mode != k_ra8_vreg_mode_ldo) && (mode != k_ra8_vreg_mode_dcdc)) {
440 return k_ra8_err_invalid_arg;
441 }
442
443 if (mode == k_ra8_vreg_mode_dcdc) {
444 internal_dcdc_enable_sequence(s_state.fast_startup);
445 } else {
446 internal_dcdc_disable_sequence(s_state.ldo_boost);
447 /* Permanent mode switch: cache the LDO selection so a subsequent
448 * standby/restore cycle keeps the regulator in LDO. */
449 uint8_t cached = 0U;
450 if (s_state.ldo_boost) {
451 cached = k_ra8_vreg_mask_lcboost;
452 }
453 s_state.dcdcctl = cached;
454 }
455 return k_ra8_ok;
456}

543[[nodiscard]] ra8_err_t ra8_vreg_get_status(ra8_vreg_status_t* out)
544{
545 RA8_CHECK_NULL_PTR(out, s_tag, "out must not be nullptr");
546 /* DCDCCTL is the main status word.
547 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
548 const uint8_t dcdcctl = *ra8_vreg_dcdcctl();
549 /* VCCSEL is read-back-able.
550 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
551 const uint8_t vccsel = *ra8_vreg_vccsel();
552 /* LVOCR holds the low-voltage profile bit.
553 * HUM Ch 68.3 "Usage Notes" p 4034 */
554 const uint8_t lvocr = *ra8_vreg_lvocr();
555
556 out->dcdcctl = dcdcctl;
557 out->vccsel = vccsel;
558 out->lvocr = lvocr;
559 out->mode =
560 ((dcdcctl & k_ra8_vreg_mask_dcdcon) != 0U) ? k_ra8_vreg_mode_dcdc : k_ra8_vreg_mode_ldo;
561 out->vccsel_dec = (ra8_vreg_vccsel_t)(vccsel & k_ra8_vreg_vccsel_mask);
562 out->lv_profile = internal_profile_from_lvocr(lvocr);
563 out->ocp = internal_ocp_from_dcdcctl(dcdcctl);
564 const bool dcdcon_set = (dcdcctl & k_ra8_vreg_mask_dcdcon) != 0U;
565 const bool pd_clear = (dcdcctl & k_ra8_vreg_mask_pd) == 0U;
566 out->dcdc_ready = false;
567 if (dcdcon_set && pd_clear) {
568 out->dcdc_ready = true;
569 }
570 out->fast_startup = ((dcdcctl & k_ra8_vreg_mask_fst) != 0U);
571 out->ldo_boost = ((dcdcctl & k_ra8_vreg_mask_lcboost) != 0U);
572 out->io_buf_on = ((dcdcctl & k_ra8_vreg_mask_stopza) != 0U);
573 return k_ra8_ok;
574}

576[[nodiscard]] ra8_err_t ra8_vreg_clear_status(uint8_t mask)
577{
578 /* Reject any bit outside the documented writable mask -- writing a
579 * reserved bit is a programmer error.
580 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
581 if ((mask & (uint8_t)~k_ra8_vreg_mask_dcdcctl_all) != 0U) {
582 return k_ra8_err_invalid_arg;
583 }
584
585 /* Read-modify-write DCDCCTL to clear the requested control bits
586 * without touching the rest.
587 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
588 const uint8_t cur = *ra8_vreg_dcdcctl();
589 const uint8_t next = (uint8_t)(cur & (uint8_t)~mask);
590 *ra8_vreg_dcdcctl() = next;
591 s_state.dcdcctl = next;
592 return k_ra8_ok;
593}

609[[nodiscard]] ra8_err_t ra8_vreg_enter_standby(ra8_vreg_standby_t variant)
610{
611 if ((uint8_t)variant > k_ra8_vreg_standby_battery_backup) {
612 return k_ra8_err_invalid_arg;
613 }
614 s_state.last_standby = variant;
615
616 /* Software Standby (and every Deep variant + Battery Backup) is
617 * not supported in DCDC mode, so park the regulator in LDO before
618 * standby. The cached state lets us restore DCDC on exit.
619 * HUM Ch 68.2.2 "External VDD Mode" p 4033 */
620 internal_dcdc_disable_sequence(false);
621 return k_ra8_ok;
622}

632 return k_ra8_ok;

634 /* Restore VCCSEL before re-enabling DCDC.
635 * HUM Ch 68.2.1 "DCDC Mode" p 4032 */
636 *ra8_vreg_vccsel() = (uint8_t)((uint8_t)s_state.vccsel & k_ra8_vreg_vccsel_mask);
