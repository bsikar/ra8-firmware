71/* Strong SysTick override: route the tick into BOTH the ra8_time millisecond
72 * counter (for ra8_delay_ms) AND ThreadX's timer (for tx_thread_sleep and
73 * semaphore timeouts). The default weak ra8_time SysTick handler only advances
74 * the ms counter; without _tx_timer_interrupt ThreadX time never advances and
75 * tx_thread_sleep / USBX class-thread scheduling stall. The project's
76 * tx_initialize_low_level.S configures SysTick but relies on the application
77 * to publish the handler. */
78
79extern void _tx_timer_interrupt(void);
80void SysTick_Handler(void);
81void SysTick_Handler(void)
82{
83 ra8_time_on_tick();
84 _tx_timer_interrupt();
85 /* Re-enable the USB IRQ at the NVIC level: the bridge's storm guard
86 * masks it to break the USBFS event-less interrupt storm, and this
87 * 1 ms pulse is its recovery clock -- a masked line is re-enabled
88 * within one period so real USB events are never lost. */
89 ux_dcd_ra8_usb_irq_reenable();
90}
91#endif

346static UINT demo_msc_read(VOID* storage,
347 ULONG lun,
348 UCHAR* data_pointer,
349 ULONG number_blocks,
350 ULONG lba,
351 ULONG* media_status)
352{
353 (void)storage;
354 (void)lun;
355 s_dbg_read_calls++;
356 s_dbg_read_last_lba = (uint32_t)lba;
357 s_dbg_read_last_n = (uint32_t)number_blocks;
358 if ((lba + number_blocks) > (ULONG)k_fat_total_sectors) {
359 *media_status = UX_DEVICE_CLASS_STORAGE_SENSE_STATUS(k_scsi_sense_illegal_request,
360 k_scsi_asc_lba_out_of_range,
361 k_scsi_ascq_none);
362 return UX_ERROR;
363 }
364 for (ULONG i = 0UL; i < number_blocks; i++) {
365 demo_fat_fill_sector((uint32_t)(lba + i), &data_pointer[i * (ULONG)k_demo_block_size]);
366 }
367 *media_status = 0UL;
368 (void)ra8_board_led_toggle(k_ra8_board_led1);
369 return UX_SUCCESS;
370}

522static VOID demo_worker(ULONG arg)
523{
524 (void)arg;
525
526 if (demo_usbx_stack_up() != UX_SUCCESS) {
527 return;
528 }
529 if (demo_msc_class_register() != UX_SUCCESS) {
530 return;
531 }

538
539 /* Idle. USBX runs the SCSI/BBB state machine on its own threads. */

643int32_t main(void)
644{
645 uint32_t cpuclk0_hz = 0U;
646
647 if (ra8_cgc_init() != k_ra8_ok) {
648 demo_panic_halt();
649 }
650 /* Bring up PLL2 -> USBCKCR / USBCKDIVCR so USBFS sees a spec-compliant
651 * 48 MHz reference (PLL2P 240 MHz / 5). Must run BEFORE any caller
652 * releases MSTPB11 (USBFS) -- the SREQ -> SRDY handshake silently
653 * hangs otherwise (HUM Ch 9 "Clock selection switching procedure"
654 * step 1). Without this the SIE never sees a 48 MHz clock and the
655 * host never enumerates the device. */
656 if (ra8_cgc_usbfs_clock_enable() != k_ra8_ok) {
657 demo_panic_halt();
658 }
659 if (ra8_cgc_get_clock_hz(k_ra8_clock_id_cpuclk0, &cpuclk0_hz) != k_ra8_ok) {
660 demo_panic_halt();
661 }
662 if (ra8_time_init(cpuclk0_hz) != k_ra8_ok) {
663 demo_panic_halt();
664 }
665 if (ra8_board_led_init(k_ra8_board_led1) != k_ra8_ok) {
666 demo_panic_halt();
667 }
668 if (demo_pins_init() != k_ra8_ok) {
669 demo_panic_halt();
670 }
671
672 ra8_isr_globals_enable();
673
674#ifndef RA8_OFF_TARGET
675 /* tx_kernel_enter is __noreturn -- it never comes back. */
676 tx_kernel_enter();
677#endif
678
679 demo_panic_halt();
680 return 0;
681}
