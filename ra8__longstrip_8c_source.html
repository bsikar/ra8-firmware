156ra8_err_t ra8_longstrip_open(ra8_longstrip_t* wt, const ra8_longstrip_cfg_t* cfg)
157{
158 const ra8_err_t ptr_err = priv_longstrip_check_ptrs(wt, cfg);
159 if (ptr_err != k_ra8_ok) {
160 return ptr_err;
161 }
162 /* Decision: reject a zero-area viewport (2 conditions -- MC/DC in tests). */
163 if ((cfg->viewport_w == 0U) || (cfg->viewport_h == 0U)) {
164 return k_ra8_err_invalid_arg;
165 }
166 ra8_jof_info_t info = {};
167 const ra8_err_t err = ra8_jof_parse(cfg->pread, cfg->pread_ctx, cfg->atlas_size, &info);
168 if (err != k_ra8_ok) {
169 return err;
170 }
171 /* A longstrip strip is one full-width band column; anything else is not a
172 * scrollable strip. `ra8_jof_parse` derives tile_cols as
173 * ceil(width / tile_w), so `tile_w == width` already implies tile_cols == 1
174 * -- the full-width test alone is necessary and sufficient. */
175 if (info.tile_w != info.width) {
176 return k_ra8_err_not_supported;
177 }
178 priv_longstrip_bind(wt, cfg, &info);
179 return k_ra8_ok;
180}

385bool ra8_longstrip_tick(ra8_longstrip_t* wt)
386{
387 if (wt == nullptr) {
388 return false;
389 }
390 if (wt->velocity == 0) {
391 return false;
392 }
393 wt->scroll_y = ra8_longstrip_clamp_scroll(wt, wt_sat_add(wt->scroll_y, wt->velocity));
394 wt->velocity = wt_apply_friction(wt->velocity);
395 /* Decision: come to rest when the speed hits zero or a boundary is reached. */
396 if ((wt->velocity == 0) || wt_fling_should_stop(wt)) {
397 wt->velocity = 0;
398 return false;
399 }
400 return true;
401}

432ra8_err_t ra8_longstrip_prefetch(ra8_longstrip_t* wt, uint16_t depth)
433{
434 RA8_CHECK_NULL_PTR(wt, s_tag, "wt must not be nullptr");
435 uint16_t first = 0;
436 uint16_t last = 0;
437 const ra8_err_t err = ra8_longstrip_visible_bands(wt, wt->scroll_y, &first, &last);
438 if (err != k_ra8_ok) {
439 return err;
440 }
441 if (depth > (uint16_t)k_ra8_longstrip_max_prefetch) {
442 depth = (uint16_t)k_ra8_longstrip_max_prefetch;
443 }
444 const uint16_t visible_span = (uint16_t)((last - first) + 1U);
445 /* Decision: nothing to prefetch with zero depth or a strip that fits (2). */
446 if ((depth == 0U) || (wt->band_count <= visible_span)) {
447 return k_ra8_ok;
448 }
449 const bool downward = (wt->velocity >= 0);
450 for (uint16_t i = 1U; i <= depth; i++) { /* bounded by k_ra8_longstrip_max_prefetch */
451 uint16_t target = 0;
452 if (downward) {
453 const uint32_t t = (uint32_t)last + (uint32_t)i;
454 if (t >= (uint32_t)wt->band_count) {
455 break;
456 }
457 target = (uint16_t)t;
458 } else {
459 if (first < i) {
460 break;
461 }
462 target = (uint16_t)(first - i);
463 }
464 wt_warm_band(wt, target);
465 }
466 return k_ra8_ok;
467}

525RA8_INTERNAL
526static ra8_err_t
527wt_draw_band(ra8_longstrip_t* wt, uint16_t band, int32_t dst_x, ra8_longstrip_render_stats_t* stats)
528{
529 ra8_tile_key_t key = {};
530 key.image_id = wt->image_id;
531 key.tile_x = (uint16_t)k_wt_tile_x;
532 key.tile_y = band;
533 key.zoom = (uint16_t)k_wt_zoom;
534 ra8_tile_t tile = {};
535 if (ra8_tile_cache_get(wt->cache, &key, &tile) != k_ra8_ok) {
536 stats->skipped++; /* record a gap; the frame still completes */
537 return k_ra8_ok;
538 }
539 const int32_t dst_y = (int32_t)((uint32_t)band * (uint32_t)wt->band_h) - wt->scroll_y;
540 const ra8_err_t berr =
541 wt->blit(wt->blit_ctx, tile.pixels, tile.width, tile.height, wt->info.bpp, dst_x, dst_y);
542 if (berr == k_ra8_ok) {
543 stats->bands_drawn++;
544 wt_accumulate_coverage(wt, dst_y, tile.height, stats);
545 } else {
546 stats->skipped++;
547 }
548 const ra8_err_t perr = ra8_tile_cache_put(wt->cache, tile.pixels);
549 return (berr != k_ra8_ok) ? berr : perr;
550}

552ra8_err_t ra8_longstrip_render(ra8_longstrip_t* wt, ra8_longstrip_render_stats_t* stats)
553{
554 RA8_CHECK_NULL_PTR(wt, s_tag, "wt must not be nullptr");
555 ra8_longstrip_render_stats_t local = {};
556 wt->scroll_y = ra8_longstrip_clamp_scroll(wt, wt->scroll_y);
557 uint16_t first = 0;
558 uint16_t last = 0;
559 ra8_err_t err = ra8_longstrip_visible_bands(wt, wt->scroll_y, &first, &last);
560 if (err != k_ra8_ok) {
561 return err;
562 }
563 const int32_t dst_x = ((int32_t)wt->viewport_w - (int32_t)wt->canvas_w) / 2;
564 for (uint32_t b = (uint32_t)first; b <= (uint32_t)last; b++) { /* bounded by band_count */
565 err = wt_draw_band(wt, (uint16_t)b, dst_x, &local);
566 if (err != k_ra8_ok) {
567 break;
568 }
569 }
570 if (stats != nullptr) {
571 *stats = local;
572 }
573 return err;
