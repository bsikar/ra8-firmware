130static uint64_t gb_rng(uint64_t* s)
131{
132 uint64_t x = *s;

134 x ^= x >> (uint32_t)k_gb_xs_b;

190static uint32_t gb_pick_letter(uint64_t* s)
191{
192 uint32_t r = gb_below(s, (uint32_t)k_gb_freq_total);
193 for (uint32_t i = 0U; i < (uint32_t)k_gb_ascii_az_z; ++i) {
194 if (r < (uint32_t)k_gb_letter_freq[i]) {
195 return (uint32_t)k_gb_ascii_az_a + i;
196 }
197 r -= (uint32_t)k_gb_letter_freq[i];
198 }
199 return (uint32_t)k_gb_ascii_az_a; /* fallthrough: 'a' */
200}

229static uint32_t gb_pick_codepoint(uint64_t* s)
230{
231 const uint32_t roll = gb_below(s, (uint32_t)k_gb_pct_base);
232 if (roll < (uint32_t)k_gb_punct_pct) {
233 const uint32_t n = (uint32_t)(sizeof(k_gb_punct) / sizeof(k_gb_punct[0]));
234 return k_gb_punct[gb_below(s, n)];
235 }
236 if (roll < (uint32_t)k_gb_punct_pct + (uint32_t)k_gb_digit_pct) {
237 return (uint32_t)k_gb_ascii_0 + gb_below(s, (uint32_t)k_gb_digits);
238 }
239 uint32_t cp = gb_pick_letter(s);
240 if (gb_below(s, (uint32_t)k_gb_pct_base) < (uint32_t)k_gb_cap_pct) {
241 cp -= (uint32_t)k_gb_case_delta; /* uppercase variant */
242 }
243 return cp;
244}

296static gb_access_t* gb_build_stream(uint64_t* rng, uint64_t* out_n)
297{
298 const uint64_t total =
299 (uint64_t)k_gb_pages * (uint64_t)k_gb_reread_pages * (uint64_t)k_gb_page_glyphs;
300 gb_access_t* seq = (gb_access_t*)malloc((size_t)total * sizeof(gb_access_t));
301 if (seq == nullptr) {
302 *out_n = 0U;
303 return nullptr;
304 }
305 uint64_t w = 0U;
306 for (uint32_t p = 0U; p < (uint32_t)k_gb_pages; ++p) {
307 /* Generate the page's glyph stream once, then replay it on each re-render
308 * so a page-turn re-references exactly the same working set. */
309 const uint64_t page_start = w;
310 for (uint32_t g = 0U; g < (uint32_t)k_gb_page_glyphs; ++g) {
311 const bool heading = gb_below(rng, (uint32_t)k_gb_pct_base) < (uint32_t)k_gb_heading_pct;
312 const uint16_t size_px = heading ? (uint16_t)k_gb_heading_px : (uint16_t)k_gb_body_px;
313 seq[w].cp = gb_pick_codepoint(rng);
314 seq[w].size_px = size_px;
315 ++w;
316 }
317 for (uint32_t r = 1U; r < (uint32_t)k_gb_reread_pages; ++r) {
318 (void)memcpy(&seq[w], &seq[page_start], (size_t)k_gb_page_glyphs * sizeof(gb_access_t));
319 w += (uint64_t)k_gb_page_glyphs;
320 }
321 }
322 *out_n = w;
323 return seq;
324}

354static double gb_run(const gb_access_t* seq, uint64_t n, uint32_t budget, uint32_t* out_rasters)
355{
356 uint8_t* cell_mem = (uint8_t*)malloc((size_t)budget * (size_t)k_gb_cell_bytes);
357 ra8_glyph_key_t* keys = (ra8_glyph_key_t*)calloc((size_t)budget, sizeof(ra8_glyph_key_t));
358 ra8_glyph_dims_t* dims = (ra8_glyph_dims_t*)calloc((size_t)budget, sizeof(ra8_glyph_dims_t));
359 ra8_keycache_cell_t* meta =
360 (ra8_keycache_cell_t*)calloc((size_t)budget, sizeof(ra8_keycache_cell_t));
361 int32_t* buckets = (int32_t*)malloc((size_t)k_gb_buckets * sizeof(int32_t));
362 if ((cell_mem == nullptr) || (keys == nullptr) || (dims == nullptr) || (meta == nullptr) ||
363 (buckets == nullptr)) {
364 free(cell_mem);
365 free(keys);
366 free(dims);
367 free(meta);
368 free(buckets);
369 *out_rasters = 0U;
370 return 0.0;
371 }
372 ra8_glyph_atlas_cfg_t cfg = {.cell_mem = cell_mem,
373 .cell_bytes = (uint32_t)k_gb_cell_bytes,
374 .cell_count = budget,
375 .meta = meta,
376 .keys = keys,
377 .dims = dims,
378 .buckets = buckets,
379 .bucket_count = (uint32_t)k_gb_buckets,
380 .render = gb_render,
381 .render_ctx = nullptr};
382 ra8_glyph_atlas_t atlas = {};
383 (void)ra8_glyph_atlas_init(&atlas, &cfg);
384 for (uint64_t i = 0U; i < n; ++i) {
385 ra8_glyph_key_t k = {};
386 k.glyph_id = seq[i].cp;
387 k.size_px = seq[i].size_px;
388 ra8_glyph_t g = {};
389 if (ra8_glyph_atlas_get(&atlas, &k, &g) == k_ra8_ok) {
390 (void)ra8_glyph_atlas_put(&atlas, g.bitmap);
391 }
392 }
393 uint32_t hits = 0U;
394 uint32_t miss = 0U;
395 (void)ra8_glyph_atlas_stats(&atlas, &hits, &miss, nullptr);
396 *out_rasters = miss;
397 const double hit_pct = (n == 0U) ? 0.0 : (100.0 * (double)hits / (double)n);
398 free(cell_mem);
399 free(keys);
400 free(dims);
401 free(meta);
402 free(buckets);
403 return hit_pct;
404}
