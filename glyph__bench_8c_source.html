108static uint64_t gb_rng(uint64_t* s)
109{
110 uint64_t x = *s;
111 x ^= x << (uint32_t)k_gb_xs_a;
112 x ^= x >> (uint32_t)k_gb_xs_b;
113 x ^= x << (uint32_t)k_gb_xs_c;
114 *s = x;
115 return x;

125static uint32_t gb_pick_letter(uint64_t* s)
126{
127 uint32_t r = gb_below(s, (uint32_t)k_gb_freq_total);
128 for (uint32_t i = 0U; i < (uint32_t)k_gb_ascii_az_z; ++i) {
129 if (r < (uint32_t)k_gb_letter_freq[i]) {
130 return (uint32_t)k_gb_ascii_az_a + i;
131 }
132 r -= (uint32_t)k_gb_letter_freq[i];
133 }
134 return (uint32_t)k_gb_ascii_az_a; /* fallthrough: 'a' */
135}

141static uint32_t gb_pick_codepoint(uint64_t* s)
142{
143 const uint32_t roll = gb_below(s, (uint32_t)k_gb_pct_base);
144 if (roll < (uint32_t)k_gb_punct_pct) {
145 const uint32_t n = (uint32_t)(sizeof(k_gb_punct) / sizeof(k_gb_punct[0]));
146 return k_gb_punct[gb_below(s, n)];
147 }
148 if (roll < (uint32_t)k_gb_punct_pct + (uint32_t)k_gb_digit_pct) {
149 return (uint32_t)k_gb_ascii_0 + gb_below(s, (uint32_t)k_gb_digits);
150 }
151 uint32_t cp = gb_pick_letter(s);
152 if (gb_below(s, (uint32_t)k_gb_pct_base) < (uint32_t)k_gb_cap_pct) {
153 cp -= (uint32_t)k_gb_case_delta; /* uppercase variant */
154 }
155 return cp;
156}

181static gb_access_t* gb_build_stream(uint64_t* rng, uint64_t* out_n)
182{
183 const uint64_t total =
184 (uint64_t)k_gb_pages * (uint64_t)k_gb_reread_pages * (uint64_t)k_gb_page_glyphs;
185 gb_access_t* seq = (gb_access_t*)malloc((size_t)total * sizeof(gb_access_t));
186 if (seq == nullptr) {
187 *out_n = 0U;
188 return nullptr;
189 }
190 uint64_t w = 0U;
191 for (uint32_t p = 0U; p < (uint32_t)k_gb_pages; ++p) {
192 /* Generate the page's glyph stream once, then replay it on each re-render
193 * so a page-turn re-references exactly the same working set. */
194 const uint64_t page_start = w;
195 for (uint32_t g = 0U; g < (uint32_t)k_gb_page_glyphs; ++g) {
196 const bool heading = gb_below(rng, (uint32_t)k_gb_pct_base) < (uint32_t)k_gb_heading_pct;
197 const uint16_t size_px = heading ? (uint16_t)k_gb_heading_px : (uint16_t)k_gb_body_px;
198 seq[w].cp = gb_pick_codepoint(rng);
199 seq[w].size_px = size_px;
200 ++w;
201 }
202 for (uint32_t r = 1U; r < (uint32_t)k_gb_reread_pages; ++r) {
203 (void)memcpy(&seq[w], &seq[page_start], (size_t)k_gb_page_glyphs * sizeof(gb_access_t));
204 w += (uint64_t)k_gb_page_glyphs;
205 }
206 }
207 *out_n = w;
208 return seq;
209}

212static double gb_run(const gb_access_t* seq, uint64_t n, uint32_t budget, uint32_t* out_rasters)
213{
214 uint8_t* cell_mem = (uint8_t*)malloc((size_t)budget * (size_t)k_gb_cell_bytes);
215 ra8_glyph_key_t* keys = (ra8_glyph_key_t*)calloc((size_t)budget, sizeof(ra8_glyph_key_t));
216 ra8_glyph_dims_t* dims = (ra8_glyph_dims_t*)calloc((size_t)budget, sizeof(ra8_glyph_dims_t));
217 ra8_keycache_cell_t* meta =
218 (ra8_keycache_cell_t*)calloc((size_t)budget, sizeof(ra8_keycache_cell_t));
219 int32_t* buckets = (int32_t*)malloc((size_t)k_gb_buckets * sizeof(int32_t));
220 if ((cell_mem == nullptr) || (keys == nullptr) || (dims == nullptr) || (meta == nullptr) ||
221 (buckets == nullptr)) {
222 free(cell_mem);
223 free(keys);
224 free(dims);
225 free(meta);
226 free(buckets);
227 *out_rasters = 0U;
228 return 0.0;
229 }
230 ra8_glyph_atlas_cfg_t cfg = {.cell_mem = cell_mem,
231 .cell_bytes = (uint32_t)k_gb_cell_bytes,
232 .cell_count = budget,
233 .meta = meta,
234 .keys = keys,
235 .dims = dims,
236 .buckets = buckets,
237 .bucket_count = (uint32_t)k_gb_buckets,
238 .render = gb_render,
239 .render_ctx = nullptr};
240 ra8_glyph_atlas_t atlas = {};
241 (void)ra8_glyph_atlas_init(&atlas, &cfg);
242 for (uint64_t i = 0U; i < n; ++i) {
243 ra8_glyph_key_t k = {};
244 k.glyph_id = seq[i].cp;
245 k.size_px = seq[i].size_px;
246 ra8_glyph_t g = {};
247 if (ra8_glyph_atlas_get(&atlas, &k, &g) == k_ra8_ok) {
248 (void)ra8_glyph_atlas_put(&atlas, g.bitmap);
249 }
250 }
251 uint32_t hits = 0U;
252 uint32_t miss = 0U;
253 (void)ra8_glyph_atlas_stats(&atlas, &hits, &miss, nullptr);
254 *out_rasters = miss;
255 const double hit_pct = (n == 0U) ? 0.0 : (100.0 * (double)hits / (double)n);
256 free(cell_mem);
257 free(keys);
258 free(dims);
259 free(meta);
260 free(buckets);
261 return hit_pct;
262}
