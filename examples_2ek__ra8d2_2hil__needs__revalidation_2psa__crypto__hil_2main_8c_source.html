72static uint8_t hex_nibble(char c)
73{
74 uint8_t v = 0U;
75 if (c >= 'a') {
76 v = (uint8_t)((c - 'a') + (int)k_hex_ten);
77 } else if (c >= 'A') {
78 v = (uint8_t)((c - 'A') + (int)k_hex_ten);
79 } else {
80 v = (uint8_t)(c - '0');
81 }
82 return v;
83}

86static void unhex(const char* hex, uint8_t* out, size_t n)
87{
88 for (size_t i = 0U; i < n; ++i) {
89 uint8_t hi = hex_nibble(hex[2U * i]);
90 uint8_t lo = hex_nibble(hex[(2U * i) + 1U]);
91 out[i] = (uint8_t)((int)hi << (int)k_hex_shift) | lo;
92 }
93}

164{

173 k_gcm_pt);

178 k_gcm_ct);
179
180 psa_key_attributes_t attr = PSA_KEY_ATTRIBUTES_INIT;
181 psa_set_key_usage_flags(&attr, PSA_KEY_USAGE_ENCRYPT | PSA_KEY_USAGE_DECRYPT);
182 psa_set_key_algorithm(&attr, PSA_ALG_GCM);
183 psa_set_key_type(&attr, PSA_KEY_TYPE_AES);
184 psa_key_id_t kid = 0;
185 if (psa_import_key(&attr, key, sizeof(key), &kid) != PSA_SUCCESS) {
186 return false;

191 ok = false;
192 }
193 (void)psa_destroy_key(kid);

198static bool kat_ecdsa_p256(void)
199{
200 uint8_t pub[k_ec_point];
201 pub[0] = 0x04U;
202 unhex("60FED4BA255A9D31C961EB74C6356D68C049B8923B61FA6CE669622E60F29FB6",

207 (size_t)k_ec_coord);

209 unhex("EFD48B2AACB6A8FD1140DD9CD45E81D69D2C877B56AAF991C34D0EA84EAF3716",

212 unhex("F7CB1C942D657C41D436C7A1B6E29F65F3E900DBB9AFF4064DC4AB2F843ACDA8",
213 &sig[(size_t)k_ec_coord],
214 (size_t)k_ec_coord);
215
216 uint8_t hash[k_sha_len];
217 size_t hl = 0U;
218 (void)psa_hash_compute(PSA_ALG_SHA_256,
219 (const uint8_t*)"sample",
220 (size_t)k_smpl_len,
221 hash,
222 sizeof(hash),
223 &hl);
224
225 psa_key_attributes_t attr = PSA_KEY_ATTRIBUTES_INIT;
226 psa_set_key_usage_flags(&attr, PSA_KEY_USAGE_VERIFY_HASH);
227 psa_set_key_algorithm(&attr, PSA_ALG_ECDSA(PSA_ALG_SHA_256));
228 psa_set_key_type(&attr, PSA_KEY_TYPE_ECC_PUBLIC_KEY(PSA_ECC_FAMILY_SECP_R1));
229 psa_key_id_t kid = 0;
230 if (psa_import_key(&attr, pub, sizeof(pub), &kid) != PSA_SUCCESS) {
231 return false;
232 }
233
234 bool ok = true;
235 if (psa_verify_hash(kid, PSA_ALG_ECDSA(PSA_ALG_SHA_256), hash, sizeof(hash), sig, sizeof(sig)) !=
236 PSA_SUCCESS) {
237 ok = false;
238 }
239 sig[0] ^= 0x01U; /* flip one bit -> must be rejected */
240 if (psa_verify_hash(kid, PSA_ALG_ECDSA(PSA_ALG_SHA_256), hash, sizeof(hash), sig, sizeof(sig)) !=
241 PSA_ERROR_INVALID_SIGNATURE) {
242 ok = false;
243 }
244 (void)psa_destroy_key(kid);
245 return ok;
