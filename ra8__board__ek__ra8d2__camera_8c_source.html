59ra8_err_t ra8_board_camera_xclk_start(uint32_t frequency_hz)
60{
61 if (frequency_hz == 0U) {
62 return k_ra8_err_invalid_arg;
63 }
64 uint32_t pclkd_hz = 0U;
65 ra8_err_t err = ra8_cgc_get_clock_hz(k_ra8_clock_id_pclkd, &pclkd_hz);
66 if (err != k_ra8_ok) {
67 return err;
68 }
69 const uint32_t period = pclkd_hz / frequency_hz;
70 if ((period < 2U) || (period > (uint32_t)k_camera_gpt_period_max)) {
71 return k_ra8_err_invalid_arg;
72 }
73 const ra8_gpt_cfg_t timer_cfg = {
74 .mode = k_ra8_gpt_mode_saw_pwm,
75 .prescaler = k_ra8_gpt_ps_div_1,
76 .period = period - 1U,
77 .duty_a = period / 2U,
78 .duty_b = 0U,
79 .auto_start = true,
80 };
81 err = ra8_gpt_init((uint8_t)k_camera_xclk_gpt_channel, &timer_cfg);
82 if (err != k_ra8_ok) {
83 return err;
84 }
85 const ra8_gpt_pwm_pin_cfg_t pin_cfg = {
86 .output_enable = true,
87 .polarity = k_ra8_gpt_pol_active_high,
88 .stop_level = k_ra8_gpt_stop_low,
89 .disable_on_fault = k_ra8_gpt_disable_none,
90 };
91 err = ra8_gpt_pwm_pin_configure((uint8_t)k_camera_xclk_gpt_channel, k_ra8_gpt_pin_a, &pin_cfg);
92 if (err != k_ra8_ok) {
93 return err;
94 }
95 err = ra8_pfs_route_peripheral((ra8_port_pin_t)k_ra8_board_cam_xclk,
96 k_ra8_psel_gpt0,
97 "board.camera.xclk");
98 if (err != k_ra8_ok) {
99 return err;
100 }
101 return ra8_pfs_set_drive_strength((ra8_port_pin_t)k_ra8_board_cam_xclk,
102 k_ra8_pfs_dscr_high_speed_high);
103}

129{
130 const ra8_port_pin_t reset_pin = (ra8_port_pin_t)k_ra8_board_cam_rst;
131 ra8_err_t err = ra8_gpio_output_init(reset_pin, k_ra8_level_low);

137 if (err != k_ra8_ok) {

140 ra8_delay_ms((uint32_t)k_camera_reset_high_ms);
141 return k_ra8_ok;
