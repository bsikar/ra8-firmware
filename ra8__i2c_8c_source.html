268static void internal_i2c_restart(volatile r_i2c_regs_t* reg)
269{
270 /* HUM Ch 39.2.2 "ICCR2 : I2C Bus Control Register 2" p 2371 */
271 reg->ICCR2 = (uint8_t)(reg->ICCR2 | (uint8_t)k_ra8_i2c_msk_iccr2_rs);
272 /* RS auto-clears once the restart condition is issued. The peripheral
273 * address must be written to ICDRT only after RS reads 0 -- a write
274 * while RS = 1 is silently dropped (HUM Ch 39.11 "Issuing a Restart
275 * Condition" Note, p 2434). After a send_stop=false write TDRE is
276 * already 1, so without this wait send_address would write the address
277 * before the restart completed and the transmit would never happen. */
278 for (uint32_t i = 0U; i < (uint32_t)k_ra8_i2c_poll_limit; i++) { /* GCOVR_EXCL_BR_LINE */
279 if ((reg->ICCR2 & (uint8_t)k_ra8_i2c_msk_iccr2_rs) == 0U) { /* GCOVR_EXCL_BR_LINE */
280 break;
281 }
282 }

624static ra8_err_t internal_i2c_drain_rx(volatile r_i2c_regs_t* reg, uint8_t* out, uint32_t len)
625{
626 /* First RDRF marks the address-phase completion. Arm the short-read
627 * end-of-frame controls before the dummy read kicks off the data clock.
628 * HUM Ch 39.3.4 "Controller Receive Operation" p 2400 */
629 ra8_err_t err = internal_i2c_wait_icsr2(reg, (uint8_t)k_ra8_i2c_msk_icsr2_rdrf);
630 if (err != k_ra8_ok) {
631 return err;
632 }
633 if (len <= (uint32_t)k_ra8_i2c_rx_short_len) {
634 /* HUM Ch 39.2.5 "ICMR3 : I2C Bus Mode Register 3 -- WAIT" p 2376 */
635 reg->ICMR3 = (uint8_t)(reg->ICMR3 | (uint8_t)k_ra8_i2c_msk_icmr3_wait);
636 }
637 if (len == (uint32_t)k_ra8_i2c_rx_single_len) {
638 internal_i2c_set_nack(reg);
639 }
640 /* Dummy read starts the data clock.
641 * HUM Ch 39.2.18 "ICDRR : I2C Bus Receive Data Register" p 2393 */
642 (void)reg->ICDRR;
643
644 for (uint32_t loaded = 0U; loaded < len; loaded++) {
645 err = internal_i2c_wait_icsr2(reg, (uint8_t)k_ra8_i2c_msk_icsr2_rdrf);
646 if (err != k_ra8_ok) {
647 break;
648 }
649 const uint32_t remain = len - loaded;
650 if (remain == (uint32_t)k_ra8_i2c_rx_remain_wait) {
651 /* HUM Ch 39.2.5 "ICMR3 : I2C Bus Mode Register 3 -- WAIT" p 2376 */
652 reg->ICMR3 = (uint8_t)(reg->ICMR3 | (uint8_t)k_ra8_i2c_msk_icmr3_wait);
653 } else if (remain == (uint32_t)k_ra8_i2c_rx_remain_nack) {
654 internal_i2c_set_nack(reg);
655 } else if (remain == (uint32_t)k_ra8_i2c_rx_remain_stop) {
656 internal_i2c_stop_request(reg);
657 } else {
658 /* Mid-stream byte: no end-of-frame control to arm. */
659 }
660 /* HUM Ch 39.2.18 "ICDRR : I2C Bus Receive Data Register" p 2393 */
661 out[loaded] = reg->ICDRR;
662 }
663 /* Clear WAIT (lets the requested STOP fire) and ACKBT for the next
664 * transaction, then wait for the bus to be released.
665 * HUM Ch 39.2.5 "ICMR3 : I2C Bus Mode Register 3" p 2376 */
666 reg->ICMR3 = (uint8_t)(reg->ICMR3 & (uint8_t)~(uint8_t)((uint8_t)k_ra8_i2c_msk_icmr3_wait |
667 (uint8_t)k_ra8_i2c_msk_icmr3_ackbt));
668 internal_i2c_wait_bus_free(reg);
669 return err;
670}

673{
674 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
675 RA8_CHECK_NULL_PTR(reg, s_i2c_tag, "i2c_read: channel");
676 RA8_CHECK_NULL_PTR(data, s_i2c_tag, "i2c_read: data");
677 if (len == 0U) {
678 return k_ra8_err_invalid_arg;
679 }
680
681 const bool bus_held = s_i2c_state[channel].bus_held;
682 const ra8_err_t busy_gate = internal_i2c_busy_gate(reg, bus_held);
683 if (busy_gate != k_ra8_ok) {
684 return busy_gate;
685 }
686
687 internal_i2c_clear_status(reg);
688 internal_i2c_open_phase(reg, bus_held);
689
690 const uint8_t address_byte =
691 (uint8_t)(((uint32_t)peripheral_7b << (uint32_t)k_ra8_i2c_addr_shift) |
692 (uint32_t)k_ra8_i2c_addr_rw_read);
693 ra8_err_t err = internal_i2c_send_address(reg, address_byte);
694 if (err != k_ra8_ok) {
695 internal_i2c_stop(reg);
696 s_i2c_state[channel].bus_held = false;
697 return err;
698 }
699
700 err = internal_i2c_drain_rx(reg, data, len);
701 const ra8_err_t status_err = internal_i2c_status_from_icsr2(reg->ICSR2);
702 if (status_err != k_ra8_ok) {
703 err = status_err;
704 }
705 internal_i2c_clear_status(reg);
706 s_i2c_state[channel].bus_held = false;
707 return err;
708}

710ra8_err_t ra8_i2c_transfer(uint8_t channel,
711 uint8_t peripheral_7b,
712 const uint8_t* wr,
713 uint32_t wr_len,
714 uint8_t* rd,
715 uint32_t rd_len)
716{
717 if (ra8_i2c_regs(channel) == nullptr) {
718 return k_ra8_err_null_ptr;
719 }
720 if ((wr_len == 0U) && (rd_len == 0U)) {
721 return k_ra8_err_invalid_arg;
722 }
723 if ((wr_len != 0U) && (wr == nullptr)) {
724 return k_ra8_err_null_ptr;
725 }
726 if ((rd_len != 0U) && (rd == nullptr)) {
727 return k_ra8_err_null_ptr;
728 }
729
730 /* Phase 1: write, holding the bus when a read phase follows. */
731 if (wr_len != 0U) {
732 const ra8_err_t w_err =
733 ra8_i2c_write(channel, peripheral_7b, wr, wr_len, /*send_stop=*/(rd_len == 0U));
734 if (w_err != k_ra8_ok) {
735 return w_err;
736 }
737 }
738
739 /* Phase 2: read (issues STOP and releases the bus). */
740 if (rd_len != 0U) {
741 return ra8_i2c_read(channel, peripheral_7b, rd, rd_len);
742 }
743 return k_ra8_ok;
744}

746ra8_err_t ra8_i2c_scan(uint8_t channel, uint8_t peripheral_7b, bool* out_acked)
747{
748 volatile r_i2c_regs_t* reg = ra8_i2c_regs(channel);
749 RA8_CHECK_NULL_PTR(reg, s_i2c_tag, "i2c_scan: channel");
750 RA8_CHECK_NULL_PTR(out_acked, s_i2c_tag, "i2c_scan: out_acked");
751
752 *out_acked = false;
753 const ra8_err_t busy_gate = internal_i2c_busy_gate(reg, s_i2c_state[channel].bus_held);
754 if (busy_gate != k_ra8_ok) {
755 return busy_gate;
756 }
757
758 internal_i2c_clear_status(reg);
759 internal_i2c_start(reg);
760
761 const uint8_t address_byte =
762 (uint8_t)(((uint32_t)peripheral_7b << (uint32_t)k_ra8_i2c_addr_shift) |
763 (uint32_t)k_ra8_i2c_addr_rw_write);
764 ra8_err_t err = internal_i2c_send_address(reg, address_byte);
765 /* Address NACK is a valid probe outcome, not a hard failure. */
766 if ((err != k_ra8_ok) && (err != k_ra8_err_nack)) {
767 internal_i2c_stop(reg);
768 s_i2c_state[channel].bus_held = false;
769 return err;
770 }
771
772 /* Wait for TEND (peripheral ACKed) or NACKF (no peripheral answered).
773 * HUM Ch 39.2.10 "ICSR2 : I2C Bus Status Register 2" p 2384 */
774 err = internal_i2c_wait_icsr2(
775 reg,
776 (uint8_t)((uint8_t)k_ra8_i2c_msk_icsr2_tend | (uint8_t)k_ra8_i2c_msk_icsr2_nackf));
777 if (err == k_ra8_ok) {
778 *out_acked = (reg->ICSR2 & (uint8_t)k_ra8_i2c_msk_icsr2_nackf) == 0U;
779 }
780
781 internal_i2c_stop(reg);
782 internal_i2c_clear_status(reg);
783 s_i2c_state[channel].bus_held = false;
784 return err;
