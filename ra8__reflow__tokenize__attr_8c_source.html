66static bool priv_attr_name_at(const uint8_t* tag, size_t i, const char* name, size_t name_len)
67{
68 for (size_t k = 0U; k < name_len; ++k) {
69 if ((tag[i + k] | 0x20U) != (uint8_t)name[k]) {
70 return false;
71 }
72 }
73 const char prev = (char)((i == 0U) ? '<' : tag[i - 1U]);
74 return !(((prev >= 'a') && (prev <= 'z')) || ((prev >= 'A') && (prev <= 'Z')));
75}

99static bool
100priv_attr_quoted_value(const uint8_t* tag, size_t tag_len, size_t pos, size_t* vstart, size_t* vlen)
101{
102 size_t j = pos;
103 while ((j < tag_len) && ra8_reflow_tok_is_xml_whitespace((char)tag[j])) {
104 ++j;
105 }
106 if ((j >= tag_len) || (tag[j] != '=')) {
107 return false;
108 }
109 ++j;
110 while ((j < tag_len) && ra8_reflow_tok_is_xml_whitespace((char)tag[j])) {
111 ++j;
112 }
113 if ((j >= tag_len) || ((tag[j] != '"') && (tag[j] != '\''))) {
114 return false;
115 }
116 const uint8_t quote = tag[j];
117 ++j;
118 *vstart = j;
119 while ((j < tag_len) && (tag[j] != quote)) {
120 ++j;
121 }
122 *vlen = j - *vstart;
123 return true;
124}

150 size_t tag_len,

152 size_t name_len,

192void ra8_reflow_tok_capture_attr(ra8_reflow_t* engine,
193 const uint8_t* tag,
194 size_t tag_len,
195 const char* name,
196 size_t name_len,
197 uint32_t* out_off,
198 uint32_t* out_len)
199{
200 *out_off = 0U;
201 *out_len = 0U;
202 size_t vstart = 0U;
203 size_t vlen = 0U;
204 if (!ra8_reflow_tok_find_attr(tag, tag_len, name, name_len, &vstart, &vlen)) {
205 return;
206 }
207 if ((vlen == 0U) ||
208 ((size_t)engine->text_pool_used + vlen > (size_t)k_ra8_reflow_text_pool_bytes)) {
209 return;
210 }
211 const uint32_t off = engine->text_pool_used;
212 memcpy(&engine->text_pool[off], &tag[vstart], vlen);
213 engine->text_pool_used += (uint32_t)vlen;
214 *out_off = off;
215 *out_len = (uint32_t)vlen;
216}

378uint8_t ra8_reflow_tok_resolve_face_slot(const ra8_reflow_t* engine, const ra8_css_style_t* comp)
379{
380 if ((engine->face_count == 0U) || ((comp->set & (uint8_t)k_ra8_css_set_family) == 0U) ||
381 (comp->family_len == 0U)) {
382 return 0U;
383 }
384 const char* family = (const char*)&engine->css.names[comp->family_off];
385 const bool bold = (comp->style & (uint8_t)k_ra8_reflow_style_bold) != 0U;

387 const int16_t ci = ra8_css_match_face(&engine->css, family, comp->family_len, bold, italic);
388 if (ci < 0) {
389 return 0U;
390 }
391 for (uint8_t k = 0U; k < engine->face_count; ++k) {
392 if (engine->faces[k].css_face_idx == (uint8_t)ci) {
393 return (uint8_t)(k + 1U);
394 }
395 }
