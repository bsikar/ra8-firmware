177RA8_INTERNAL static void internal_format_tick(uint32_t tick, uint8_t* digits)
178{
179 for (uint32_t i = 0U; i < (uint32_t)k_agt_pulse_hex_digits; ++i) {
180 const uint32_t shift =
181 ((uint32_t)k_agt_pulse_hex_digits - 1U - i) * (uint32_t)k_agt_pulse_hex_shift;
182 const uint32_t nib = (tick >> shift) & (uint32_t)k_agt_pulse_hex_mask;
183 digits[i] = (uint8_t)s_agt_pulse_hex_chars[nib];
184 }
185}

207[[nodiscard]] RA8_INTERNAL static ra8_err_t internal_emit_tick(uint32_t tick)
208{
209 uint8_t digits[k_agt_pulse_hex_digits] = {};
210 internal_format_tick(tick, digits);
211 ra8_err_t err =
212 ra8_board_uart_console_write(s_agt_pulse_msg_head, (size_t)(sizeof(s_agt_pulse_msg_head) - 1U));
213 if (err != k_ra8_ok) {
214 return err;
215 }
216 err = ra8_board_uart_console_write(digits, (size_t)sizeof(digits));
217 if (err != k_ra8_ok) {
218 return err;
219 }
220 return ra8_board_uart_console_write(s_agt_pulse_msg_tail,
221 (size_t)(sizeof(s_agt_pulse_msg_tail) - 1U));
222}

224void main(void)
225{
226 internal_setup_or_halt();
227 ra8_isr_globals_enable();
228
229 if (internal_arm() != k_ra8_ok) {
230 internal_panic_halt();
231 }
232
233 uint32_t tick = 0U;
234 while (1) {
235 uint8_t status = 0U;
236 if (ra8_agt_get_status((uint8_t)k_agt_pulse_channel, &status) != k_ra8_ok) {
237 break;
238 }
239 if ((status & (uint8_t)k_ra8_agt_agtcr_tundf_msk) != 0U) {
240 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
241 break;
242 }
243 ++tick;
244 if (internal_emit_tick(tick) != k_ra8_ok) {
245 break;
246 }
247 /* Stop + re-arm to clear AGTCR.TUNDF on real silicon. */
248 if (ra8_agt_stop((uint8_t)k_agt_pulse_channel) != k_ra8_ok) {
249 break;
250 }
251 if (internal_arm() != k_ra8_ok) {
252 break;
253 }
254 }
255 ra8_delay_ms((uint32_t)k_agt_pulse_poll_ms);
256 }
257 internal_panic_halt();
258}
