254[[nodiscard]] static ra8_err_t selftest_verify_ospi_raw(void)
255{
256 static uint8_t s_burst[k_selftest_burst_bytes] = {};
257 static uint8_t s_expect[k_selftest_block_size] = {};
258
259 s_dbg_verified_bytes = 0U;
260 s_dbg_mismatch_off = (uint32_t)k_selftest_no_mismatch;
261 const uint32_t t0 = ra8_time_ms();
262 for (uint32_t blk = 0U; blk < (uint32_t)k_fat_data_clusters;
263 blk += (uint32_t)k_selftest_burst_blocks) {
264 const uint32_t lba = (uint32_t)k_fat_data_lba + blk;
265 ra8_err_t err = ra8_usb_hmsc_read10((uint8_t)k_selftest_target_lun,
266 lba,
267 (uint16_t)k_selftest_burst_blocks,
268 s_burst);
269 if (err != k_ra8_ok) {
270 (void)selftest_print_fail("READ(10) burst", err);
271 return err;
272 }
273 for (uint32_t s = 0U; s < (uint32_t)k_selftest_burst_blocks; s++) {
274 const uint32_t win_sector = blk + s;
275 selftest_pattern_fill(win_sector, s_expect);
276 const uint32_t boff = s * (uint32_t)k_selftest_block_size;
277 if (memcmp(&s_burst[boff], s_expect, (size_t)k_selftest_block_size) != 0) {
278 s_dbg_mismatch_off = win_sector * (uint32_t)k_selftest_block_size;
279 (void)selftest_print_fail("OSPI pattern mismatch at 0x", (ra8_err_t)s_dbg_mismatch_off);
280 return k_ra8_err_invalid_state;
281 }
282 }
283 s_dbg_verified_bytes =
284 (blk + (uint32_t)k_selftest_burst_blocks) * (uint32_t)k_selftest_block_size;
285 }
286 s_dbg_verify_ms = ra8_time_ms() - t0;
287 return k_ra8_ok;
288}

456[[nodiscard]] ra8_err_t selftest_host_pass(void)
457{
458 ra8_usb_hmsc_device_t device = {};
459 ra8_err_t err = selftest_host_enumerate(&device);
460 if (err != k_ra8_ok) {
461 return err;
462 }
463
464 /* Mount proves the host can PARSE the FAT16 volume over the loop;
465 * then unmount and do the heavy integrity check with raw multi-block
466 * READ(10) (ra8_fs's per-cluster metadata re-reads are far too slow
467 * for a 1 MiB sweep). */
468 s_dbg_phase = (uint32_t)k_selftest_phase_mount;
469 ra8_fs_mount_t* mount = nullptr;
470 err = selftest_mount_volume(&mount);
471 if (err != k_ra8_ok) {
472 (void)ra8_usb_hmsc_close();
473 return err;
474 }
475 (void)ra8_fs_unmount(mount);
476
477 s_dbg_phase = (uint32_t)k_selftest_phase_verify;
478 err = selftest_verify_ospi_raw();
479 if (err == k_ra8_ok) {
480 err = selftest_print_verify_verdict();
481 }
482 if (err == k_ra8_ok) {
483 s_dbg_phase = (uint32_t)k_selftest_phase_wp;
484 err = selftest_write_protect_probe();
485 }
486 if (err != k_ra8_ok) {
487 (void)ra8_usb_hmsc_close();
488 return err;
489 }
490
491 s_dbg_phase = (uint32_t)k_selftest_phase_pass;
492 s_dbg_pass_count++;
493 err = selftest_print("ra8d2 selftest: USB SELFTEST OSPI PASS\r\n");
494 if (err != k_ra8_ok) {
495 return err;
496 }
497 (void)ra8_board_led_on(k_ra8_board_led2);
498 return k_ra8_ok;
499}
