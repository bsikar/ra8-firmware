45ra8_err_t ra8_icu_init(void)
46{
47 ra8_log_info(s_tag, "ra8_icu_init");
48
49 /* HUM Ch 14.2.12 "IRQCRi : IRQ Control Register", p 535 */
50 for (uint8_t i = 0U; i < (uint8_t)k_ra8_icu_num_irqs; ++i) {
51 volatile uint8_t* irqcr = ra8_icu_irqcr(i);
52 if (irqcr != nullptr) { /* GCOVR_EXCL_BR_LINE -- bounds validated */
53 *irqcr = 0U;
54 }
55 }
56
57 /* HUM Ch 14.2.14 "NMIER : NMI Enable Register", p 542 */
58 *ra8_icu_nmier() = 0UL;
59
60 /* HUM Ch 14.2.15 "NMICLR : NMI Status Clear Register", p 544 -- write
61 * all-ones to clear every latched NMI status bit. */
62 *ra8_icu_nmiclr() = k_ra8_icu_nmiclr_all;
63
64 /* HUM Ch 14 "WUPEN0/WUPEN1 : Wake Up Interrupt Enable Register",
65 * p 524..582 -- clear so no source can wake the core unexpectedly. */
66 *ra8_icu_wupen0() = k_ra8_icu_wupen_clear;
67 *ra8_icu_wupen1() = k_ra8_icu_wupen_clear;
68 return k_ra8_ok;
69}

76ra8_err_t ra8_icu_configure_irq_pin(uint8_t irq_num, const ra8_icu_irq_cfg_t* cfg)
77{
78 RA8_CHECK_NULL_PTR(cfg, s_tag, "irq cfg");
79 if (irq_num >= (uint8_t)k_ra8_icu_num_irqs) {
80 return k_ra8_err_invalid_arg;
81 }
82 volatile uint8_t* irqcr = ra8_icu_irqcr(irq_num);
83 if (irqcr == nullptr) { /* GCOVR_EXCL_BR_LINE -- bounds validated */
84 return k_ra8_err_invalid_arg;
85 }
86
87 const uint8_t irqmd = ((uint8_t)cfg->sense & k_ra8_icu_irqcr_mask_irqmd);
88 const uint8_t fclk = (uint8_t)(((uint8_t)cfg->filter_div << k_ra8_icu_irqcr_bit_fclksel) &
89 k_ra8_icu_irqcr_mask_fclksel);
90 uint8_t val = (uint8_t)(irqmd | fclk);
91 if (cfg->filter_en) {
92 val |= k_ra8_icu_irqcr_mask_flten;
93 }
94 /* HUM Ch 14.2.12 "IRQCRi : IRQ Control Register", p 535 */
95 *irqcr = val;
96 ra8_log_info_val(s_tag, "irqcr pin", (uint32_t)irq_num);
97 return k_ra8_ok;
98}
