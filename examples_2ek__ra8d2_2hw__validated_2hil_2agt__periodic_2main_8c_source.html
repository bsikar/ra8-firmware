118 break;
119 }
120 if ((status & (uint8_t)k_agt_periodic_undf_bit) != 0U) {
121 if (ra8_board_led_toggle(k_ra8_board_led1) != k_ra8_ok) {
122 break;
123 }
124 if (ra8_board_uart_console_write(k_agt_periodic_log_msg,
125 (size_t)(sizeof(k_agt_periodic_log_msg) - 1U)) != k_ra8_ok) {
126 break;
127 }
128 /* Re-arm: stop + start clears AGTCR.TUNDF on real silicon. */
129 if (ra8_agt_stop((uint8_t)k_agt_periodic_channel) != k_ra8_ok) {

133 break;
134 }
135 }
136 ra8_delay_ms((uint32_t)k_agt_periodic_poll_ms);
137 }
138 agt_periodic_panic_halt();
