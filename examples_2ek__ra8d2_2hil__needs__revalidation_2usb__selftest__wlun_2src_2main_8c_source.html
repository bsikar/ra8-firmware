404static UINT wlun_msc_read(VOID* storage,
405 ULONG lun,

407 ULONG number_blocks,

423 *media_status = 0UL;
424 (void)ra8_board_led_toggle(k_ra8_board_led1);
425 return UX_SUCCESS;
426}

644static VOID wlun_device_worker(ULONG arg)
645{
646 (void)arg;
647
648 UINT ux = wlun_usbx_stack_up();
649 if (ux != UX_SUCCESS) {
650 s_dbg_dev_err = (uint32_t)ux;
651 return;
652 }
653 s_dbg_dev_step = (uint32_t)k_wlun_dev_step_stack;
654 ux = wlun_class_register();
655 if (ux != UX_SUCCESS) {
656 s_dbg_dev_err = (uint32_t)ux;
657 return;
658 }
659 s_dbg_dev_step = (uint32_t)k_wlun_dev_step_class;
660 ra8_err_t e = ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_fs);
661 if (e != k_ra8_ok) {
662 s_dbg_dev_err = (uint32_t)e;
663 return;
664 }
665 s_dbg_dev_step = (uint32_t)k_wlun_dev_step_dcd;
666 e = ra8_usb_device_attach(k_ra8_usb_speed_fs, true);
667 if (e != k_ra8_ok) {
668 s_dbg_dev_err = (uint32_t)e;
669 return;
670 }
671 s_dbg_dev_step = (uint32_t)k_wlun_dev_step_attach;
672
673 while (1) {
674 s_dbg_dev_step = (uint32_t)k_wlun_dev_step_parked;
675 tx_thread_sleep(k_wlun_idle_ticks);
676 }
677}
