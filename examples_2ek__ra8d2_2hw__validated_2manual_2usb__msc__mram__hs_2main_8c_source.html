72/* Strong SysTick override: route the tick into BOTH the ra8_time millisecond
73 * counter (for ra8_delay_ms) AND ThreadX's timer (for tx_thread_sleep and
74 * semaphore timeouts). The default weak ra8_time SysTick handler only advances
75 * the ms counter; without _tx_timer_interrupt ThreadX time never advances and
76 * tx_thread_sleep / USBX class-thread scheduling stall. The project's
77 * tx_initialize_low_level.S configures SysTick but relies on the application
78 * to publish the handler. */
79
80extern void _tx_timer_interrupt(void);
81void SysTick_Handler(void);
82void SysTick_Handler(void)
83{
84 ra8_time_on_tick();
85 _tx_timer_interrupt();
86 /* Re-enable the USB IRQ at the NVIC level: the bridge's storm guard
87 * masks it to break the USBFS event-less interrupt storm, and this
88 * 1 ms pulse is its recovery clock -- a masked line is re-enabled
89 * within one period so real USB events are never lost. */
90 ux_dcd_ra8_usb_irq_reenable();
91}
92#endif

679static VOID demo_worker(ULONG arg)
680{
681 (void)arg;
682
683 if (demo_usbx_stack_up() != UX_SUCCESS) {
684 return;
685 }
686 if (demo_msc_class_register() != UX_SUCCESS) {
687 return;
688 }
689 if (ux_dcd_ra8_usb_initialize(k_ra8_usb_speed_hs) != k_ra8_ok) {
690 return;
691 }
692 if (ra8_usb_device_attach(k_ra8_usb_speed_hs, true) != k_ra8_ok) {
693 return;
694 }
695
696 /* Idle. USBX runs the SCSI/BBB state machine on its own threads. */
697 while (1) {
698 s_dbg_dev_state = (uint32_t)_ux_system_slave->ux_system_slave_device.ux_slave_device_state;
699 s_dbg_ux_speed = (uint32_t)_ux_system_slave->ux_system_slave_speed;
700 s_dbg_class_inst =
701 (uint32_t)(uintptr_t)_ux_system_slave->ux_system_slave_class_array[0].ux_slave_class_instance;
702 s_dbg_framework = (uint32_t)(uintptr_t)_ux_system_slave->ux_system_slave_device_framework;
703 s_dbg_fw_len = (uint32_t)_ux_system_slave->ux_system_slave_device_framework_length;
704 s_dbg_thr_state = (uint32_t)_ux_system_slave->ux_system_slave_class_array[0]
705 .ux_slave_class_thread.tx_thread_state;
706 s_dbg_thr_runs = (uint32_t)_ux_system_slave->ux_system_slave_class_array[0]
707 .ux_slave_class_thread.tx_thread_run_count;
708 /* Diagnostic kick: if the class thread is parked SUSPENDED while the
709 * device is CONFIGURED, the activate-time resume (ISR context) did
710 * not take; re-resume from thread context and count it. */
711 if (s_dbg_dev_state == (uint32_t)UX_DEVICE_CONFIGURED) {
712 s_dbg_state3_seen++;
713 if (s_dbg_thr_state == (uint32_t)TX_SUSPENDED) {
714 (void)tx_thread_resume(
715 &_ux_system_slave->ux_system_slave_class_array[0].ux_slave_class_thread);
716 s_dbg_kicks++;
717 }
718 }
719 tx_thread_sleep(1U);
720 }
721}
