79static void run_print_sd_summary(void)
80{
81 bool sd_att = false;

83 uint8_t sd_f = 0U;
84 const char* sd_l = nullptr;
85 board_sd_info(&sd_att, &sd_b, &sd_f, &sd_l);
86 const uint64_t gib_bytes = (uint64_t)k_size_kib * (uint64_t)k_size_kib * (uint64_t)k_size_kib;

89 const unsigned long sd_sz =
90 sd_gb ? (unsigned long)(sd_b / gib_bytes) : (unsigned long)(sd_b / mib_bytes);
91 const char* sd_u = sd_gb ? "GB" : "MB";
92 if (sd_att && (sd_f != 0U)) {
93 (void)fprintf(stderr,
94 " SD card : %lu %s FAT%u '%s' (created by --sd-new)\n",
95 sd_sz,
96 sd_u,
97 (unsigned)sd_f,
98 (sd_l != nullptr) ? sd_l : "");
99 } else if (sd_att) {
100 (void)fprintf(stderr, " SD card : %lu %s image attached\n", sd_sz, sd_u);
101 }

122static void run_print_stop_summary(uc_engine* uc, const run_stop_t* st)
123{
124 (void)fprintf(stderr,
125 "\nra8_emulator: stopped -- %s%s%s%s%s%s\n",
126 uc_strerror(st->err),
127 st->timed_out ? " (wall-clock budget reached)" : "",
128 st->idle_stopped ? " (idle steady-state)" : "",
129 st->usb_stopped ? " (USB enumerated)" : "",
130 st->stop_sym_hit ? " (--stop-sym threshold reached)" : "",
131 st->prof_stopped ? " (profile: boot complete)" : "");
132 (void)fprintf(stderr, " final PC : 0x%08X\n", st->run_pc);
133 if (emu_exc_bkpt_hit()) {
134 (void)fprintf(stderr,
135 " => firmware executed a BKPT @ 0x%08X (deliberate trap: "
136 "Default_Handler / failed assert / fault give-up)\n",
137 emu_exc_bkpt_pc());
138 }
139 (void)fprintf(stderr,
140 " chunks run : %u SysTick ticks: %u\n",
141 st->chunks,
142 emu_exc_systick_fires());
143 if (emu_mve_emulated_count() > 0U) {
144 (void)fprintf(stderr,
145 " MVE (Helium) : %llu instruction(s) emulated (M85 vector ops the M33 "
146 "core lacks)\n",
147 (unsigned long long)emu_mve_emulated_count());
148 }
149 if (emu_lob_emulated_count() > 0U) {
150 (void)fprintf(stderr,
151 " LOB (loop) : %llu DLS/LE instruction(s) emulated (M85 hardware-loop "
152 "ops)\n",
153 (unsigned long long)emu_lob_emulated_count());
154 }
155 prof_report();
156 (void)fprintf(stderr,
157 " exceptions : %u PendSV %u SVCall (real Cortex-M entry/return)\n",
158 emu_exc_pendsv_takes(),
159 emu_exc_svc_takes());
160 (void)fprintf(stderr,
161 " touch clicks : %u drained via ra8_touch -> I3C -> GT911\n",
162 board_periph_touch_reported());
163 /* Emit any console bytes still buffered without a trailing newline. */
164 console_flush_line(board_periph_sci_console_channel());
165 /* Peripheral-model observability: LED transitions, timer totals, IRQ counts,
166 * SCI byte totals. */
167 board_periph_report(uc);
168 board_net_report();
169 emu_mmio_print_counts();
170 run_print_sd_summary();
171 emu_mmio_print_bgc_and_table();

189static void run_print_verdict(const run_stop_t* st)
190{
191 if (st->timed_out && !st->idle_stopped && !st->usb_stopped && !st->stop_sym_hit &&
192 !st->prof_stopped && !emu_exc_bkpt_hit()) {
193 /* The wall-clock guard fired (clock() is CPU-time, so a heavily-loaded host
194 * burns the budget faster). This is a TRUNCATED run, NOT a completed one --
195 * say so plainly, and do NOT print the "EXECUTED to the run budget" line a
196 * full-budget run prints. Conflating the two let a load-correlated truncation
197 * masquerade as success, dropping a deterministic banner without failing the
198 * gate (#168). A caller that wants the run bounded by a deterministic event
199 * (not by CPU-time) should pass RA8_EMU_STOP_ON / RA8_EMU_MAX_CHUNKS. */
200 (void)fprintf(stderr,
201 " => ra8_emulator TRUNCATED by the wall-clock guard at chunk %u of %u "
202 "(%.0fs CPU-time elapsed); this is NOT a full-budget run (host "
203 "overloaded -- see #168).\n",
204 st->chunks,
205 st->max_chunks,
206 st->wall_s);
207 } else if (((st->err == UC_ERR_OK) || st->idle_stopped || st->usb_stopped || st->stop_sym_hit ||
208 st->prof_stopped) &&
209 !emu_exc_bkpt_hit()) {
210 if (st->idle_stopped) {
211 (void)fprintf(stderr,
212 " => firmware EXECUTED to the run budget (idle steady-state: no "
213 "observable change for %u chunks, stopped at chunk %u).\n",
214 st->idle_stop_chunks,
215 st->chunks);
216 } else if (st->usb_stopped) {
217 (void)fprintf(stderr,
218 " => firmware EXECUTED to the run budget (USB enumerated: device "
219 "CONFIGURED, stopped at chunk %u).\n",
220 st->chunks);
221 } else {
222 (void)fprintf(stderr,
223 " => firmware EXECUTED to the run budget (no invalid opcode / fault).\n");
224 }
225 }

244static void run_print_dump_syms(uc_engine* uc, const emu_run_cfg_t* cfg)
245{
246 /* --dump-sym: read each resolved global from Unicorn memory and print its
247 * 32-bit value (and the address), so a test can probe firmware state (e.g. an
248 * init-step or mismatch counter) after the run without a debugger. */
249 for (uint32_t d = 0U; d < cfg->dump_sym_n; d++) {
250 if (cfg->dump_sym_addrs[d] == 0U) {
251 (void)fprintf(stderr, " dump-sym : %s = <unresolved>\n", cfg->dump_sym_names[d]);
252 continue;
253 }
254 uint32_t v = 0U;
255 if (uc_mem_read(uc, (uint64_t)cfg->dump_sym_addrs[d], &v, sizeof(v)) == UC_ERR_OK) {
256 (void)fprintf(stderr,
257 " dump-sym : %s @0x%08X = %u (0x%08X)\n",
258 cfg->dump_sym_names[d],
259 cfg->dump_sym_addrs[d],
260 v,
261 v);
262 } else {
263 (void)fprintf(stderr,
264 " dump-sym : %s @0x%08X = <unreadable>\n",
265 cfg->dump_sym_names[d],
266 cfg->dump_sym_addrs[d]);
267 }
268 }

288 run_print_verdict(&stop);

328{

353int run_cleanup(const run_loop_t* st)
354{
355 const emu_run_cfg_t* cfg = st->cfg;
356 if (cfg->save_sd_path != nullptr) {
357 (void)board_sd_save(cfg->save_sd_path);
358 }
359 free(st->panel_fb);
360 free(st->rot_fb);
361 free(st->composite);
362 free(cfg->elf); /* kept alive for the whole run so a warm reboot can re-load it */
363 (void)uc_close(cfg->uc);
364
365 emu_exit_t exit_code = k_emu_exit_ok;
366 if (emu_exc_bkpt_hit()) {
367 exit_code = k_emu_exit_bkpt;
368 } else if (st->err != UC_ERR_OK) {
369 exit_code = k_emu_exit_fault;
370 } else if (st->timed_out) {
371 exit_code = k_emu_exit_timeout;
372 }
373 return (int)exit_code;
