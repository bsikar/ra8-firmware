361static VOID dfu_device_worker(ULONG arg)
362{
363 (void)arg;
364
365 /* Config A device half: the FS controller runs the DFU device class wired
366 * to real MRAM (libs/ra8_dfu). DFU_DNLOAD programs the inactive Slot B. */
367 ra8_dfu_device_set_target(k_ra8_dfu_slot_a);
368 const ra8_err_t e = ra8_dfu_device_start(k_ra8_usb_speed_fs,
369 s_usbx_pool,
370 (uint32_t)sizeof(s_usbx_pool),
371 s_device_framework,
372 (uint32_t)sizeof(s_device_framework),
373 s_string_framework,
374 (uint32_t)sizeof(s_string_framework),
375 s_language_id_framework,
376 (uint32_t)sizeof(s_language_id_framework));
377 if (e != k_ra8_ok) {
378 s_dbg_host_err = (uint32_t)e;
379 return;
380 }
381 s_dbg_dev_step = 4U;
382 /* Each DFU_DNLOAD block is programmed synchronously inside the DFU write
383 * callback; this worker only finalizes the image header once the host
384 * signals end-of-download (the self-test uses DFU_ABORT, so that path is
385 * the bootloader's, not this twin's). */
386 while (1) {
387 (void)ra8_dfu_device_worker_step();
388 tx_thread_sleep(k_dfu_idle_ticks);
389 }
390}

488[[nodiscard]] static ra8_err_t dfu_print_hex(uint32_t value, uint8_t digits)
489{
490 uint8_t out[k_dfu_hex_chars_u32] = {};
491 uint8_t width = digits;
492 if (width > (uint8_t)k_dfu_hex_chars_u32) {
493 width = (uint8_t)k_dfu_hex_chars_u32;
494 }
495 for (uint8_t i = 0U; i < width; i++) {
496 const uint8_t shift = (uint8_t)((width - 1U - i) * k_dfu_nibble_bits);
497 out[i] = dfu_nibble_to_hex((value >> shift) & k_dfu_nibble_mask);
498 }
499 return dfu_sci_write(out, (uint32_t)width);
500}

594[[nodiscard]] static ra8_err_t dfu_host_pass(void)
595{
596 s_dbg_phase = (uint32_t)k_dfu_phase_init;
597 ra8_err_t err =
598 dfu_print("ra8d2 dfu-boot: host up, DFU-flashing a bootable Slot-A payload...\r\n");
599 if (err != k_ra8_ok) {
600 return err;
601 }
602
603 /* Real DFU flash flow: DFU_DNLOAD the bootable Slot-A image, then a
604 * zero-length DFU_DNLOAD (manifest) so the DEVICE commits the slot header --
605 * exactly what dfu-util drives, but over the self-loop so no external host is
606 * needed. The device half runs on this same chip (Slot A target). */
607 s_dbg_phase = (uint32_t)k_dfu_phase_download;
608 ra8_dfu_host_result_t res = {};
609 err = ra8_dfu_host_program(k_ra8_usb_speed_hs,
610 s_boot_payload,
611 (uint32_t)sizeof(s_boot_payload),
612 &res);
613 s_dbg_pid = res.pid;
614 if (err != k_ra8_ok) {
615 (void)dfu_print_fail("dfu program", err);
616 return err;
617 }
618
619 /* Wait for the device worker to commit the header, then confirm Slot A is
620 * bootable (CRC over the just-programmed body matches) from this side. */
621 bool committed = false;
622 for (uint32_t i = 0U; i < (uint32_t)k_dfu_commit_tries; i++) {
623 if (ra8_dfu_device_committed()) {
624 committed = true;
625 break;
626 }
627 tx_thread_sleep(k_dfu_idle_ticks);
628 }
629 if (!committed) {
630 (void)dfu_print_fail("commit timeout", k_ra8_err_hw_timeout);
631 return k_ra8_err_hw_timeout;
632 }
633 const ra8_err_t cerr = ra8_dfu_device_last_error();
634 if (cerr != k_ra8_ok) {
635 (void)dfu_print_fail("device commit", cerr);
636 return cerr;
637 }
638 if (!ra8_dfu_slot_valid(k_ra8_dfu_slot_a)) {
639 (void)dfu_print_fail("Slot A not bootable after commit", k_ra8_err_crc_mismatch);
640 return k_ra8_err_crc_mismatch;
641 }
642
643 s_dbg_phase = (uint32_t)k_dfu_phase_pass;
644 s_dbg_pass_count++;
645 err = dfu_print(
646 "ra8d2 dfu-boot: Slot A DFU-committed + bootable -- USB SELFTEST DFU-BOOT COMMIT PASS\r\n");
647 if (err != k_ra8_ok) {
648 return err;
649 }
650 (void)ra8_board_led_on(k_ra8_board_led2);
651 return k_ra8_ok;
652}
