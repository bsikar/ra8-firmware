326static void priv_coalesce(void)
327{
328 priv_blk_t* b = priv_cell_at(0U);
329 for (size_t guard = 0U; guard < (size_t)k_priv_walk_max; guard++) {
330 if ((uint8_t*)b >= priv_end()) {
331 break;
332 }
333 if (b->is_free == (size_t)k_priv_blk_free) {
334 priv_blk_t* n = priv_next(b);
335 /* Decision (MC/DC): swallow the successor while it exists and is free. */
336 while (((uint8_t*)n < priv_end()) && (n->is_free == (size_t)k_priv_blk_free)) {
337 b->size += (size_t)k_priv_hdr_bytes + n->size;
338 n = priv_next(b);
339 }
340 }
341 b = priv_next(b);
342 }

380static void priv_split(priv_blk_t* b, size_t need)
381{
382 /* Only split when the remainder can hold a header plus a minimum payload. */
383 if (b->size >= (need + (size_t)k_priv_hdr_bytes + (size_t)k_priv_align)) {
384 const uint8_t* const rem_ptr = priv_payload(b) + need;
385 const uint8_t* const pool = priv_base();
386 const size_t rem_off = (size_t)((uintptr_t)rem_ptr - (uintptr_t)pool);
387 priv_blk_t* rem = priv_cell_at(rem_off);
388 rem->size = b->size - need - (size_t)k_priv_hdr_bytes;
389 rem->is_free = (size_t)k_priv_blk_free;
390 b->size = need;
391 }

394void* ra8_epub_miniz_alloc(void* opaque, size_t items, size_t size)
395{
396 (void)opaque;
397 /* Overflow guard (MC/DC): a non-zero element size whose count would wrap. */
398 if ((size != 0U) && (items > (SIZE_MAX / size))) {
399 return nullptr;
400 }
401 /* Oversize guard (MC/DC): a request larger than the whole pool can never be
402 * satisfied, and rounding it up in priv_align_up would overflow
403 * (bytes + align - 1) into a tiny under-allocation -- reject it first. */
404 if ((items * size) > (size_t)k_ra8_epub_miniz_pool_bytes) {
405 return nullptr;
406 }
407 priv_init();
408 size_t need = priv_align_up(items * size);
409 if (need == 0U) {
410 need = (size_t)k_priv_align; /* never hand back a zero-size block */
411 }
412
413 priv_blk_t* b = priv_cell_at(0U);
414 for (size_t guard = 0U; guard < (size_t)k_priv_walk_max; guard++) {
415 if ((uint8_t*)b >= priv_end()) {
416 break;
417 }
418 /* First-fit decision (MC/DC): a free block big enough for the request. */
419 if ((b->is_free == (size_t)k_priv_blk_free) && (b->size >= need)) {
420 priv_split(b, need);
421 b->is_free = (size_t)k_priv_blk_used;
422 return priv_payload(b);
423 }
424 b = priv_next(b);
425 }
426 return nullptr;
427}

440void* ra8_epub_miniz_realloc(void* opaque, void* address, size_t items, size_t size)
441{
442 if (address == nullptr) {
443 return ra8_epub_miniz_alloc(opaque, items, size);
444 }
445 if ((size != 0U) && (items > (SIZE_MAX / size))) {
446 return nullptr;
447 }
448 /* Oversize guard (MC/DC): see ra8_epub_miniz_alloc -- reject a larger-than-pool
449 * request before priv_align_up can overflow it; the old block stays valid. */
450 if ((items * size) > (size_t)k_ra8_epub_miniz_pool_bytes) {
451 return nullptr;
452 }
453 const size_t need = priv_align_up(items * size);
454 if (need == 0U) {
455 ra8_epub_miniz_free(opaque, address);
456 return nullptr;
457 }
458 priv_blk_t* b = priv_header(address);
459 if (b->size >= need) {
460 return address; /* already big enough -- keep in place */
461 }
462 void* moved = ra8_epub_miniz_alloc(opaque, items, size);
463 if (moved == nullptr) {
464 return nullptr; /* old block still valid */
465 }
466 (void)memcpy(moved, address, b->size);
467 ra8_epub_miniz_free(opaque, address);
468 return moved;
