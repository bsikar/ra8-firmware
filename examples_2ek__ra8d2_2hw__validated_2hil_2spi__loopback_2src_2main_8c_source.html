177RA8_INTERNAL static bool internal_round_trip_ok(void)
178{
179 for (uint8_t i = 0U; i < (uint8_t)k_spi_demo_pattern_len; i++) {
180 const uint8_t tx = (uint8_t)((uint8_t)k_spi_demo_pattern_base + i);
181 uint8_t rx = 0U;
182 if (ra8_spi_xfer8((uint8_t)k_spi_demo_spi_channel, tx, &rx) != k_ra8_ok) {
183 return false;
184 }
185 if (rx != tx) {
186 return false;
187 }
188 }
189 return true;
190}

224 break;
225 }
226 ra8_delay_ms(k_spi_demo_period_ms);
227 }
228
229 internal_panic_halt();
